\begin{proof}[Proof of \cref{obj:lem:weighted-capacity-mass-homogeneity}]
Fix a covariate cell \(x\).  By \cref{obj:def:observable-capacities}, the capacity totals formed from the weighted array \(c^w\) are
\[
q_0^w(x)=\sum_{i\in\mathcal Y}\ell_i^w(x),
\qquad
q_1^w(x)=\sum_{j\in\mathcal Y}h_j^w(x),
\qquad
m^w(x)=\min\{q_0^w(x),q_1^w(x)\}.
\]

\begin{enumerate}
\item Substituting the definition of \(c^w\) into the first weighted total gives
\[
q_0^w(x)
=\sum_{i\in\mathcal Y} w_x\ell_i(x)
=w_x\sum_{i\in\mathcal Y}\ell_i(x)
=w_x q_0(x),
\]
where the middle equality is finite-sum linearity. % lean: Finset.mul_sum

\item The same computation for the upper-arm total gives
\[
q_1^w(x)
=\sum_{j\in\mathcal Y} w_xh_j(x)
=w_x\sum_{j\in\mathcal Y}h_j(x)
=w_x q_1(x).
\]
This finite-sum identity also covers the degenerate case in which the outcome index set is empty, since both sums are then zero. % lean: Finset.mul_sum

\item Since the weight satisfies \(w_x\ge 0\), multiplication by \(w_x\) commutes with the binary minimum:
\[
\min\{w_xq_0(x),w_xq_1(x)\}
=
w_x\min\{q_0(x),q_1(x)\}.
\]
Combining this identity with the two displayed total identities yields
\[
m^w(x)
=\min\{q_0^w(x),q_1^w(x)\}
=\min\{w_xq_0(x),w_xq_1(x)\}
=w_xm(x),
\]
again using the definition of \(m(x)\) in \cref{obj:def:observable-capacities}. % lean: mul_min_of_nonneg
\end{enumerate}
\end{proof}
