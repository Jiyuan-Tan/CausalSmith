\begin{proof}[Proof of \cref{obj:prop:identification-and-extension}]
Write \(\mathcal J=\{0,1\}\times\{0,1\}\).  For a four-vector \(u\), set
\[
s_a(u)=u_{a0}+u_{a1},
\qquad
s(u)=s_0(u)+s_1(u),
\qquad
g_{a,\epsilon}(u)=
\begin{cases}
0, & s(u)=0,\\[2mm]
\dfrac{s(u)u_{a1}}{\max\{s_a(u),\epsilon s(u)\}}, & s(u)>0,
\end{cases}
\]
so that \(\bar f_\epsilon(u)=\max\{g_{0,\epsilon}(u),g_{1,\epsilon}(u)\}\) on
\(\mathbb R_+^4\), as in \cref{obj:def:global-extension}.

\begin{enumerate}
\item Fix \(x\in[d]\).  The coordinates of \(q_x\) are observed atom masses, hence
\(q_{\jmath,x}\ge 0\) for every \(\jmath\in\mathcal J\).  Moreover
\[
s(q_x)=p_x,
\qquad
s_1(q_x)=q_{10,x}+q_{11,x}=p_x\pi_x .
\]
If \(p_x>0\), \cref{obj:def:observed-model-class} gives
\[
\epsilon\le \frac{s_1(q_x)}{s(q_x)}\le 1-\epsilon,
\]
and therefore
\[
\epsilon s(q_x)\le s_1(q_x)\le (1-\epsilon)s(q_x).
\]
If \(p_x=0\), nonnegativity and \(s(q_x)=0\) force \(q_x=0\), so the same two inequalities hold.  Thus \(q_x\in\mathcal C_\epsilon\) by \cref{obj:def:overlap-cone}. % lean: cellVector_mem_overlapCone_optimal

\item For each arm \(a\in\{0,1\}\), define the totalized cell mean
\[
\mu^{\mathrm{tot}}_{ax}
=
\begin{cases}
q_{a1,x}/s_a(q_x), & s_a(q_x)>0,\\
0, & s_a(q_x)=0.
\end{cases}
\]
On an occupied cell, the cone inequalities from the previous step imply
\[
s_1(q_x)\ge \epsilon s(q_x)>0,
\qquad
s_0(q_x)=s(q_x)-s_1(q_x)\ge \epsilon s(q_x)>0,
\]
and \(\mu^{\mathrm{tot}}_{ax}\) agrees with the outcome mean \(\mu_{ax}\) in
\cref{obj:def:observed-model-class}.  Hence, for both arms,
\[
\max\{s_a(q_x),\epsilon s(q_x)\}=s_a(q_x),
\qquad
q_{a1,x}=s_a(q_x)\mu^{\mathrm{tot}}_{ax},
\]
and therefore
\[
g_{a,\epsilon}(q_x)
=\frac{s(q_x)q_{a1,x}}{s_a(q_x)}
=p_x\mu^{\mathrm{tot}}_{ax}.
\]
On a null cell all observed atom masses vanish, and both sides of this last identity are zero.  Taking the maximum over \(a\) gives
\[
\bar f_\epsilon(q_x)=p_x\max\{\mu_{0x},\mu_{1x}\},
\]
with the displayed value unchanged by the permitted null-cell choice of \(\mu_{ax}\).  Summing over \(x\in[d]\) and using \cref{obj:def:observed-optimal-value} yields
\[
\Psi(\mathbb P)=\sum_{x=1}^d \bar f_\epsilon(q_x).
\] % lean: globalCellValue_cellVector

\item Let \(u\in\mathbb R_+^4\).  For every arm \(a\),
\(0\le u_{a1}\le s_a(u)\le \max\{s_a(u),\epsilon s(u)\}\).  If \(s(u)=0\), then
\(g_{a,\epsilon}(u)=0\).  If \(s(u)>0\), the denominator is positive and
\[
0\le g_{a,\epsilon}(u)
=\frac{s(u)u_{a1}}{\max\{s_a(u),\epsilon s(u)\}}
\le s(u).
\]
Taking the maximum over the two arms gives
\[
0\le \bar f_\epsilon(u)\le s(u).
\] % lean: globalCellValue_bounds

\item It remains to prove the Lipschitz estimate for the global extension.  For nonnegative
\(\phi,\zeta,\rho,\upsilon\), put
\[
\operatorname{scal}_{\epsilon}(\phi,\zeta,\rho,\upsilon)=
\begin{cases}
0, & \phi+\zeta+\rho+\upsilon=0,\\[2mm]
\dfrac{(\phi+\zeta+\rho+\upsilon)\zeta}
{\max\{\phi+\zeta,\epsilon(\phi+\zeta+\rho+\upsilon)\}},
& \phi+\zeta+\rho+\upsilon>0 .
\end{cases}
\]
For two quadruples, write
\[
\operatorname{dist}_{\mathrm{cell}}
=
|\phi-\phi'|+|\zeta-\zeta'|+|\rho-\rho'|+|\upsilon-\upsilon'|,
\qquad
\operatorname{tot}=\phi+\zeta+\rho+\upsilon,
\quad
\operatorname{tot}'=\phi'+\zeta'+\rho'+\upsilon'.
\]
When both totals are positive, split according to whether
\(\phi+\zeta\le \epsilon\operatorname{tot}\) and
\(\phi'+\zeta'\le \epsilon\operatorname{tot}'\).  In the low--low case,
\[
|\operatorname{scal}_{\epsilon}-\operatorname{scal}_{\epsilon}'|
=\left|\frac{\zeta-\zeta'}{\epsilon}\right|
\le \epsilon^{-1}\operatorname{dist}_{\mathrm{cell}}.
\]
In the low--high case,
\[
\frac{\zeta}{\epsilon}
-\operatorname{tot}'\frac{\zeta'}{\phi'+\zeta'}
=
\frac{\zeta-\zeta'}{\epsilon}
+\frac{\zeta'}{\phi'+\zeta'}
  \frac{(\phi'+\zeta')-\epsilon\operatorname{tot}'}{\epsilon}.
\]
The second summand is nonnegative, and \(0\le \zeta'/(\phi'+\zeta')\le 1\).  Using
\(\phi+\zeta\le \epsilon\operatorname{tot}\) gives
\[
\frac{\zeta-\zeta'}{\epsilon}
+\frac{(\phi'+\zeta')-\epsilon\operatorname{tot}'}{\epsilon}
\le
\epsilon^{-1}|\phi-\phi'|+|\operatorname{tot}-\operatorname{tot}'|
\le (1+\epsilon^{-1})\operatorname{dist}_{\mathrm{cell}},
\]
and the lower bound follows from
\[
\frac{\zeta-\zeta'}{\epsilon}\ge
-\epsilon^{-1}\operatorname{dist}_{\mathrm{cell}}.
\]
The high--low case is the same calculation with the two quadruples interchanged.  In the high--high
case,
\[
\operatorname{tot}\frac{\zeta}{\phi+\zeta}
-\operatorname{tot}'\frac{\zeta'}{\phi'+\zeta'}
=
(\operatorname{tot}-\operatorname{tot}')\frac{\zeta'}{\phi'+\zeta'}
+\operatorname{tot}\left(
\frac{\zeta}{\phi+\zeta}-\frac{\zeta'}{\phi'+\zeta'}
\right),
\]
and
\[
\left|\frac{\zeta}{\phi+\zeta}-\frac{\zeta'}{\phi'+\zeta'}\right|
\le
\frac{|\zeta-\zeta'|+|\phi-\phi'|}{\phi+\zeta},
\]
because
\[
\frac{\zeta}{\phi+\zeta}-\frac{\zeta'}{\phi'+\zeta'}
=
\frac{\phi'(\zeta-\zeta')+\zeta'(\phi'-\phi)}
{(\phi+\zeta)(\phi'+\zeta')}.
\]
Together with \(0\le\zeta'/(\phi'+\zeta')\le1\) and
\(\operatorname{tot}/(\phi+\zeta)\le\epsilon^{-1}\), this gives
\[
|\operatorname{scal}_{\epsilon}-\operatorname{scal}_{\epsilon}'|
\le (1+\epsilon^{-1})\operatorname{dist}_{\mathrm{cell}}.
\]
If one total is zero, nonnegativity forces that quadruple to vanish, and the bound follows from
\(0\le \operatorname{scal}_{\epsilon}\le\operatorname{tot}\) and
\(1\le1+\epsilon^{-1}\).  Applying this estimate to each arm, after merely permuting the four
coordinates, yields
\[
|g_{a,\epsilon}(u)-g_{a,\epsilon}(v)|
\le
(1+\epsilon^{-1})\sum_{\jmath\in\mathcal J}|u_{\jmath}-v_{\jmath}|.
\]
Finally,
\[
|\bar f_\epsilon(u)-\bar f_\epsilon(v)|
\le
\max_{a\in\{0,1\}}|g_{a,\epsilon}(u)-g_{a,\epsilon}(v)|,
\]
which proves the asserted global Lipschitz property. % lean: globalCellValue_lipschitz

\item Define
\[
\operatorname{bm}(\eta,\iota)=
\begin{cases}
\eta, & \iota=1,\\
1-\eta, & \iota=0,
\end{cases}
\qquad \eta\in[0,1],\ \iota\in\{0,1\}.
\]
Construct a full-data mass function by
\[
P^{\mathrm B}(x,a,y,y_0,y_1)
=
\mathbf 1\{y=y_a\}\,
q_{ay,x}\,
\operatorname{bm}(\mu_{1-a,x},y_{1-a}),
\]
where \(y_a=y_1\) for \(a=1\) and \(y_a=y_0\) for \(a=0\).  Since
\(\mu_{0x},\mu_{1x}\in[0,1]\), every displayed mass is nonnegative.  Also
\[
\sum_{x,a,y,y_0,y_1} P^{\mathrm B}(x,a,y,y_0,y_1)
=
\sum_{x,a,y}q_{ay,x}\sum_{y_{1-a}\in\{0,1\}}\operatorname{bm}(\mu_{1-a,x},y_{1-a})
=
\sum_{x,a,y}q_{ay,x}=1.
\]
Thus these masses define a full-data law. % lean: completionFullMass_nonneg_sum

The construction gives consistency because atoms with \(y\ne y_a\) have mass zero.  Its observed
margin is \(\mathbb P\), since
\[
\sum_{y_0,y_1\in\{0,1\}}P^{\mathrm B}(x,a,y,y_0,y_1)
=
q_{ay,x}\sum_{y_{1-a}\in\{0,1\}}\operatorname{bm}(\mu_{1-a,x},y_{1-a})
=q_{ay,x}.
\]
Using the observed-law factorization supplied by \cref{obj:def:observed-model-class},
\[
q_{ay,x}
=
p_x\,\operatorname{bm}(\pi_x,a)\,\operatorname{bm}(\mu_{ax},y),
\]
the potential-outcome atom obtained after summing over the observed outcome factors as
\[
\sum_{y\in\{0,1\}}P^{\mathrm B}(x,a,y,y_0,y_1)
=
p_x\,\operatorname{bm}(\pi_x,a)\,\operatorname{bm}(\mu_{0x},y_0)\,\operatorname{bm}(\mu_{1x},y_1).
\]
Consequently, conditionally on \(X=x\), treatment is Bernoulli with parameter \(\pi_x\), independent
of the joint potential-outcome vector, and \(Y(0),Y(1)\) are independent Bernoulli variables with
parameters \(\mu_{0x}\) and \(\mu_{1x}\).  The observed margin is \(\mathbb P\), so its overlap
properties are exactly those of the given observed law; together with consistency and conditional
exchangeability, \cref{obj:def:model-class} places \(P^{\mathrm B}\) in \(\mathcal P_{d,\epsilon}\). % lean: bernoulliCompletion; bernoulliCompletion_construction; bernoulliCompletion_mem_class

\item Conversely, if \(P\in\mathcal P_{d,\epsilon}\), then \cref{obj:def:model-class} includes the
assertion that its observed margin satisfies the observed fixed-overlap model restrictions.  Hence
\[
\operatorname{Obs}(P)\in\mathcal D_{d,\epsilon}^{\mathrm{obs}}.
\]
The preceding conclusions use the nonnegative-domain extension \(u\mapsto\bar f_\epsilon(u)\) from
\cref{obj:def:global-extension}, so they are exactly the cell-overlap, value, global-bound,
Lipschitz, completion, and observed-margin assertions stated above. % lean: identification_and_extension
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:causal-optimal-value-corollary}]
Let
\[
\mathbb P:=\operatorname{Obs}(P).
\]
For \(x\in[d]\) and \(a\in\{0,1\}\), write
\[
p_x:=\mathbb P(X=x),
\qquad
\omega_{a,x}:=\mathbb P(X=x,A=a),
\]
and define the totalized observed arm regression by
\[
\xi_{a,x}:=
\begin{cases}
\displaystyle
\frac{\mathbb P(X=x,A=a,Y=1)}{\omega_{a,x}},
& \omega_{a,x}>0,\\[1.2ex]
0,& \omega_{a,x}=0.
\end{cases}
\]
Also write
\[
\rho_x:=P(X=x).
\]
For \(\iota\in\{0,1\}\) and \(\tau,\upsilon\in\{0,1\}\), put
\[
\chi_{\iota,x}(\tau,\upsilon)
:=
P\{X=x,A=\tau,Y(\iota)=\upsilon\},
\]
and define the totalized potential-outcome regression by
\[
\eta_{\iota,x}:=
\begin{cases}
\displaystyle
\frac{\sum_{\tau\in\{0,1\}}\chi_{\iota,x}(\tau,1)}{\rho_x},
& \rho_x>0,\\[1.2ex]
0,& \rho_x=0.
\end{cases}
\]

\begin{enumerate}
\item By \cref{obj:def:observed-margin}, for every \(x,a,y\),
\[
\mathbb P(X=x,A=a,Y=y)
=
P\{X=x,A=a,Y=y\}.
\]
The support component of \cref{obj:def:model-class} makes \(Y(0)\) and \(Y(1)\) binary, so the event \(\{X=x,A=a,Y=y\}\) is the disjoint union, over \(y_0,y_1\in\{0,1\}\), of the events
\[
\{X=x,A=a,Y=y,Y(0)=y_0,Y(1)=y_1\}.
\]
Finite additivity therefore gives
\[
\mathbb P(X=x,A=a,Y=y)
=
\sum_{y_0\in\{0,1\}}\sum_{y_1\in\{0,1\}}
P\{X=x,A=a,Y=y,Y(0)=y_0,Y(1)=y_1\}.
\]
Summing over \(a\) and \(y\) gives
\[
p_x=\rho_x .
\]
% lean: observedMarginal_jointMass; observedMarginal_cellMass

\item Fix \(x\in[d]\) and \(\iota\in\{0,1\}\), and suppose
\[
p_x>0,
\qquad
\omega_{\iota,x}>0.
\]
The consistency component of \cref{obj:def:model-class} gives, for each \(\upsilon\in\{0,1\}\),
\[
\mathbb P(X=x,A=\iota,Y=\upsilon)
=
\chi_{\iota,x}(\iota,\upsilon),
\]
and hence
\[
\omega_{\iota,x}
=
\sum_{\upsilon\in\{0,1\}}\chi_{\iota,x}(\iota,\upsilon).
\]
The exchangeability component of \cref{obj:def:model-class}, applied to \(Y(\iota)\), \(A=\iota\), and outcome value \(1\), gives
\[
\chi_{\iota,x}(\iota,1)
\sum_{\tau\in\{0,1\}}\sum_{\upsilon\in\{0,1\}}
\chi_{\iota,x}(\tau,\upsilon)
=
\left(\sum_{\tau\in\{0,1\}}\chi_{\iota,x}(\tau,1)\right)
\left(\sum_{\upsilon\in\{0,1\}}\chi_{\iota,x}(\iota,\upsilon)\right).
\]
The double sum is \(\rho_x=p_x\).  Dividing by the positive product
\(p_x\omega_{\iota,x}\) yields
\[
\eta_{\iota,x}
=
\frac{\sum_{\tau\in\{0,1\}}\chi_{\iota,x}(\tau,1)}{\rho_x}
=
\frac{\chi_{\iota,x}(\iota,1)}{\omega_{\iota,x}}
=
\xi_{\iota,x}.
\]
% lean: poArmAtom_total_eq_cellMass; consistent_observed_selected_atom; consistent_observed_armMass; poRegression_eq_poArmAtom_ratio; poRegression_eq_outcomeMean_of_pos

\item The support component of \cref{obj:def:model-class} gives the finite partition \(\{X=x\}_{x\in[d]}\). Expanding the outer expectation in \cref{obj:def:oracle-value} over this partition gives
\[
V^\star(P)
=
\sum_{x:\rho_x>0}\rho_x
\max\left\{
\frac{\sum_{\tau\in\{0,1\}}\chi_{0,x}(\tau,1)}{\rho_x},
\frac{\sum_{\tau\in\{0,1\}}\chi_{1,x}(\tau,1)}{\rho_x}
\right\}.
\]
By the defining convention for \(\eta_{\iota,x}\) on null cells, this is equivalently
\[
V^\star(P)
=
\sum_{x\in[d]}\rho_x\max\{\eta_{0,x},\eta_{1,x}\}.
\]
Since \(P\in\mathcal P_{d,\epsilon}\), \cref{obj:def:model-class} places \(\mathbb P\) in the observed-law class of \cref{obj:def:observed-model-class}. Choose the representing parameters \(p_x^{\mathrm{rep}}\), \(\pi_x\), and \(\mu_{ax}\) from that representation. Summing its displayed atom factorization over \(a\) and \(y\) gives
\[
p_x^{\mathrm{rep}}=\mathbb P(X=x)=p_x,
\]
and summing over \(y\) at fixed \(a\) gives
\[
\omega_{a,x}
=
\mathbb P(X=x,A=a)
=
p_x\pi_x^a(1-\pi_x)^{1-a}.
\]
If \(p_x>0\), fixed overlap in \cref{obj:def:observed-model-class} gives \(\omega_{a,x}>0\), and hence
\[
\xi_{a,x}
=
\frac{\mathbb P(X=x,A=a,Y=1)}{\omega_{a,x}}
=
\frac{p_x\pi_x^a(1-\pi_x)^{1-a}\mu_{ax}}
{p_x\pi_x^a(1-\pi_x)^{1-a}}
=
\mu_{ax}.
\]
If \(p_x=0\), the cell contribution to the value is zero for any permitted choice of \(\mu_{ax}\). Therefore \cref{obj:def:observed-optimal-value} gives
\[
\Psi(\mathbb P)
=
\sum_{x\in[d]}p_x\max\{\xi_{0,x},\xi_{1,x}\}.
\]

Using \(p_x=\rho_x\), fix \(x\).  If \(p_x=0\), then
\[
\rho_x\max\{\eta_{0,x},\eta_{1,x}\}
=
p_x\max\{\xi_{0,x},\xi_{1,x}\}
=
0.
\]
If \(p_x>0\), the overlap component of \cref{obj:def:model-class} gives
\[
\epsilon\le \frac{\omega_{1,x}}{p_x}\le 1-\epsilon .
\]
Since \(\epsilon>0\),
\[
0<\epsilon p_x\le \omega_{1,x}.
\]
Also
\[
p_x=\omega_{0,x}+\omega_{1,x},
\qquad
\omega_{1,x}\le (1-\epsilon)p_x,
\]
so
\[
\omega_{0,x}=p_x-\omega_{1,x}\ge \epsilon p_x>0.
\]
The identity from the previous step applies for \(\iota=0\) and \(\iota=1\), giving
\[
\eta_{0,x}=\xi_{0,x},
\qquad
\eta_{1,x}=\xi_{1,x}.
\]
Thus
\[
\rho_x\max\{\eta_{0,x},\eta_{1,x}\}
=
p_x\max\{\xi_{0,x},\xi_{1,x}\}.
\]
Summing over \(x\in[d]\) proves
\[
V^\star(P)=\Psi\{\operatorname{Obs}(P)\}.
\]
% lean: causal_optimal_value_corollary
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:jackson-factorial-upper}]
\begin{enumerate}
\item Choose the universal tuning supplied by
\cref{obj:lem:centered-factorial-pilot-control}.  Thus \(H_0\) is the
universal self-normalized Poisson pilot-radius constant,
\[
\kappa=\frac{1}{100000},
\qquad
D_0=3,
\]
so \(H_0>0\), \(\kappa\in(0,1)\), and \(D_0>2\).  Fix
\(0<\epsilon<1/2\), let \(C_{\epsilon}^{\mathrm{cell}}>0\) be the cell
constant supplied by \cref{obj:lem:centered-factorial-pilot-control}, and put
\[
\Xi_\epsilon:=1+\epsilon^{-1}+\epsilon^{-2}.
\]
Define
\[
C_\epsilon
:=
20000000\{1+C_{\epsilon}^{\mathrm{cell}}+(C_{\epsilon}^{\mathrm{cell}})^2\}
+
2000\Xi_\epsilon+10 .
\]
Then \(C_\epsilon>0\).  For \(d\ge2\), write
\[
L_d:=\log(ed).
\]
Since \(L_d=1+\log d\) and \(\log d\le d-1\),
\[
0<L_d\le d,
\qquad
1\le L_2\le2 .
\]
Fix \(n\ge1\), \(d\ge2\), and
\(\mathbb P\in\mathcal D_{d,\epsilon}^{\mathrm{obs}}\).  Let
\(q_x=(q_{00,x},q_{01,x},q_{10,x},q_{11,x})\) and let \(p_x\) be the
covariate-cell mass.  By \cref{obj:prop:identification-and-extension},
\[
\Psi(\mathbb P)=\sum_{x=1}^d \bar f_\epsilon(q_x),
\qquad
0\le \bar f_\epsilon(q_x)\le s(q_x).
\]
The four-cell vectors partition the one-observation law, so
\[
\sum_{x=1}^d s(q_x)=1 .
\]
Consequently
\[
0\le \Psi(\mathbb P)\le1 .
\]
% lean: jackson_factorial_upper

\item Consider the bounded-alphabet branch \(d<D_0\).  Since \(D_0=3\) and
\(d\ge2\), this branch has \(d=2\).  By
\cref{obj:def:jackson-factorial-estimator},
\[
\widehat V_{n,2}^{\mathrm{JF}}
=
\widehat V_{n,2}^{\mathrm{ER}} .
\]
Assume first that \(n\ge4\).  For \(x\in[2]\) and \(a\in\{0,1\}\), define
\[
\zeta_{a,x}
:=
\frac{\overline N_x}{n}\,
\frac{\overline S_{ax}}{\overline N_{ax}},
\qquad
\zeta^0_{a,x}:=p_x\mu_{ax},
\]
with the \(0/0=0\) convention from
\cref{obj:def:empirical-ratio-estimator}, and put
\[
\delta_{a,x}:=\zeta_{a,x}-\zeta^0_{a,x}.
\]
If \(p_x=0\), then \(\zeta^0_{a,x}=0\) and \(\zeta_{a,x}=0\) almost surely.
If \(p_x>0\), the fixed-overlap representation in
\cref{obj:def:observed-model-class} gives
\[
\eta_{a,x}:=\mathbb P(A=a\mid X=x)\ge\epsilon ,
\]
and the conditional success mean in that arm and cell is \(\mu_{ax}\).  Define
\[
\delta^{\mathrm{res}}_{a,x}
:=
\frac{\overline N_x}{n}\,
\mathbf 1\{\overline N_{ax}>0\}
\frac{1}{\overline N_{ax}}
\sum_{i=1}^n
\mathbf 1\{X_i=x,A_i=a\}(Y_i-\mu_{ax}),
\]
\[
\delta^{\mathrm{cell}}_{a,x}
:=
\left(\frac{\overline N_x}{n}-p_x\right)\mu_{ax},
\qquad
\delta^{\mathrm{zero}}_{a,x}
:=
\mu_{ax}\frac{\overline N_x}{n}\mathbf 1\{\overline N_{ax}=0\}.
\]
The totalized-ratio identity is
\[
\delta_{a,x}
=
\delta^{\mathrm{res}}_{a,x}
+
\delta^{\mathrm{cell}}_{a,x}
-
\delta^{\mathrm{zero}}_{a,x}.
\]
Conditional on \(\overline N_x=k\), the arm count is binomial with success
probability \(\eta_{a,x}\).  For
\(\mathcal B_{k,\eta}\sim\operatorname{Bin}(k,\eta)\) with
\(\eta\ge\epsilon\),
\[
E\!\left[\frac{\mathbf 1\{\mathcal B_{k,\eta}>0\}}{\mathcal B_{k,\eta}}\right]
\le
2E\!\left[\frac{1}{\mathcal B_{k,\eta}+1}\right]
=
\frac{2\{1-(1-\eta)^{k+1}\}}{(k+1)\eta}
\le
\frac{2}{k\epsilon}
\]
for \(k>0\), while the left-hand side is \(0\) for \(k=0\).  Since
\(Y_i\in\{0,1\}\) and \(\mu_{ax}\in[0,1]\), the centered residual has
conditional second moment at most one; hence
\[
E_{\mathbb P}\{(\delta^{\mathrm{res}}_{a,x})^2\}
\le
E_{\mathbb P}\!\left[
\frac{\overline N_x^2}{n^2}
\frac{\mathbf 1\{\overline N_{ax}>0\}}{\overline N_{ax}}
\right]
\le
\frac{2p_x}{n\epsilon}.
\]
Also
\[
E_{\mathbb P}\{(\delta^{\mathrm{cell}}_{a,x})^2\}
\le
E_{\mathbb P}\left(\frac{\overline N_x}{n}-p_x\right)^2
\le
\frac1n .
\]
Let
\[
\Lambda_{n,x}:=
p_x\exp\!\left[-\frac{n-2}{2}\epsilon p_x\right].
\]
Expanding
\(\overline N_x^2\mathbf 1\{\overline N_{ax}=0\}\) into diagonal and
ordered-pair contributions gives
\[
E_{\mathbb P}\{(\delta^{\mathrm{zero}}_{a,x})^2\}
\le
\frac1n+\Lambda_{n,x}^2 .
\]
The elementary inequality \(t\exp(-ut)\le u^{-1}\), applied with
\(u=(n-2)\epsilon/2\), and the condition \(n\ge4\) give
\[
\Lambda_{n,x}\le \frac{4}{n\epsilon}.
\]
Using \(p_x\le1\) and \(n^{-2}\le n^{-1}\), we obtain
\[
E_{\mathbb P}\delta_{a,x}^2
\le
4E_{\mathbb P}\{(\delta^{\mathrm{res}}_{a,x})^2\}
+
2E_{\mathbb P}\{(\delta^{\mathrm{cell}}_{a,x})^2\}
+
4E_{\mathbb P}\{(\delta^{\mathrm{zero}}_{a,x})^2\}
\le
\frac{100\Xi_\epsilon}{n}.
\]
Moreover,
\[
\widehat V_{n,2}^{\mathrm{ER}}
=
\sum_{x=1}^2 \max\{\zeta_{0,x},\zeta_{1,x}\},
\qquad
\Psi(\mathbb P)
=
\sum_{x=1}^2 \max\{\zeta^0_{0,x},\zeta^0_{1,x}\}.
\]
The maximum map is one-Lipschitz in the sup norm, and Cauchy--Schwarz over the
four pairs \((a,x)\) yields
\[
\left(
\widehat V_{n,2}^{\mathrm{ER}}-\Psi(\mathbb P)
\right)^2
\le
4\sum_{x=1}^2\sum_{a\in\{0,1\}}\delta_{a,x}^2 .
\]
Therefore
\[
\operatorname{Risk}_{\mathbb P}
(\widehat V_{n,2}^{\mathrm{ER}},\Psi)
\le
\frac{1600\Xi_\epsilon}{n}.
\]
Since \(n\ge4\), \(L_2\ge1\), and \(L_2\le2\),
\[
\min\left\{1,\frac{2}{nL_2}\right\}
=
\frac{2}{nL_2},
\qquad

so

\sum_{x=1}^d

\right\}^2
\right].

finite-Poisson marked law restricted to prefixes of length at most \(n\), so
the nonoverflow contribution is bounded by
\[

\right].

\(\Psi(\mathbb P)\in[0,1]\), so the overflow contribution is at most
\[
\Pr\{\operatorname{Pois}(n/4)>n\}.
\]

\[
\operatorname{Risk}_{\mathbb P}
(\widehat V_{n,d}^{\mathrm{JF}},\Psi)
\le

\left\{\widetilde V^{\mathrm{unc}}-\Psi(\mathbb P)\right\}^2
\right]
+

\]
To bound the Poisson overflow term, let
\[
\mathcal N_{\mathrm{ov}}\sim\operatorname{Pois}(n/4),
\qquad
\lambda_{\mathrm{ov}}:=\frac n4,
\qquad
\gamma_{\mathrm{ov}}:=\frac n8 .
\]
Since \(n\ge1\), both \(\lambda_{\mathrm{ov}}\) and \(\gamma_{\mathrm{ov}}\) are
positive.  For every nonnegative Chernoff tilt \(\varpi_{\mathrm{ov}}\), the
centered Poisson moment-generating function gives
\[
E\exp\{\varpi_{\mathrm{ov}}(\mathcal N_{\mathrm{ov}}-\lambda_{\mathrm{ov}})\}
=
\exp\{\lambda_{\mathrm{ov}}(e^{\varpi_{\mathrm{ov}}}-1-\varpi_{\mathrm{ov}})\}.
\]
Consequently, for every \(\Delta_{\mathrm{ov}}>0\), Markov's inequality yields
\[
\Pr\{\mathcal N_{\mathrm{ov}}-\lambda_{\mathrm{ov}}\ge \Delta_{\mathrm{ov}}\}
\le
\inf_{\varpi_{\mathrm{ov}}\ge0}
\exp\{-\varpi_{\mathrm{ov}}\Delta_{\mathrm{ov}}
+\lambda_{\mathrm{ov}}(e^{\varpi_{\mathrm{ov}}}-1-\varpi_{\mathrm{ov}})\}.
\]
Taking
\(
\varpi_{\mathrm{ov}}
=\log(1+\Delta_{\mathrm{ov}}/\lambda_{\mathrm{ov}})
\)
gives the Bennett form
\[
\Pr\{\mathcal N_{\mathrm{ov}}-\lambda_{\mathrm{ov}}\ge \Delta_{\mathrm{ov}}\}
\le
\exp\left[-\left\{(\lambda_{\mathrm{ov}}+\Delta_{\mathrm{ov}})
\log\left(1+\frac{\Delta_{\mathrm{ov}}}{\lambda_{\mathrm{ov}}}\right)
-\Delta_{\mathrm{ov}}\right\}\right].
\]
The elementary inequality
\[
\log(1+\rho_{\mathrm{ov}})\ge \frac{2\rho_{\mathrm{ov}}}{\rho_{\mathrm{ov}}+2},
\qquad \rho_{\mathrm{ov}}\ge0,
\]
implies
\[
(\lambda_{\mathrm{ov}}+\Delta_{\mathrm{ov}})
\log\left(1+\frac{\Delta_{\mathrm{ov}}}{\lambda_{\mathrm{ov}}}\right)
-\Delta_{\mathrm{ov}}
\ge
\frac{\Delta_{\mathrm{ov}}^2}{2\lambda_{\mathrm{ov}}+\Delta_{\mathrm{ov}}}.
\]
Now choose
\[
\Delta_{\mathrm{ov}}
:=
\sqrt{2\lambda_{\mathrm{ov}}\gamma_{\mathrm{ov}}}+2\gamma_{\mathrm{ov}}.
\]
Then
\[
\Delta_{\mathrm{ov}}^2
-\gamma_{\mathrm{ov}}(2\lambda_{\mathrm{ov}}+\Delta_{\mathrm{ov}})
=
3\gamma_{\mathrm{ov}}\sqrt{2\lambda_{\mathrm{ov}}\gamma_{\mathrm{ov}}}
+2\gamma_{\mathrm{ov}}^2
\ge0,
\]
so
\[
\Pr\{\mathcal N_{\mathrm{ov}}-\lambda_{\mathrm{ov}}>\Delta_{\mathrm{ov}}\}
\le e^{-\gamma_{\mathrm{ov}}}.
\]
For the present values,
\[
\Delta_{\mathrm{ov}}
=\sqrt{2(n/4)(n/8)}+2(n/8)
=\frac n2,
\qquad
n-\lambda_{\mathrm{ov}}=\frac{3n}{4},
\]
and therefore
\[
\{\mathcal N_{\mathrm{ov}}>n\}
\subseteq
\{\mathcal N_{\mathrm{ov}}-\lambda_{\mathrm{ov}}>\Delta_{\mathrm{ov}}\}.
\]
Thus
\[
\Pr\{\operatorname{Pois}(n/4)>n\}
\le e^{-n/8}.
\]
Finally, \(\gamma_{\mathrm{ov}}\le e^{\gamma_{\mathrm{ov}}}\) follows from
\(1+\gamma_{\mathrm{ov}}\le e^{\gamma_{\mathrm{ov}}}\), and hence
\[
\Pr\{\operatorname{Pois}(n/4)>n\}
\le e^{-n/8}
\le \frac{1}{n/8}
=
\frac8n .
\]
Using \(L_d\le d\),
\[
\frac8n\le 8\frac{d}{nL_d}=8\chi_{n,d}.
\]
Combining this overflow bound with the uncapped bound from the previous step,
\[
\operatorname{Risk}_{\mathbb P}
(\widehat V_{n,d}^{\mathrm{JF}},\Psi)
\le
\left[
10000000
\{1+C_{\epsilon}^{\mathrm{cell}}+(C_{\epsilon}^{\mathrm{cell}})^2\}
+8
\right]\chi_{n,d}
\le
C_\epsilon\chi_{n,d}.
\]
In the present branch \(\chi_{n,d}\le1\), hence
\[
C_\epsilon\chi_{n,d}
=
C_\epsilon
\min\left\{1,\frac{d}{n\log(ed)}\right\}.
\]
The three branches cover every \(n\ge1\), \(d\ge2\), and
\(\mathbb P\in\mathcal D_{d,\epsilon}^{\mathrm{obs}}\), giving the stated
uniform bound.

\end{enumerate}

\begin{proof}[Proof of \cref{obj:prop:equal-propensity-l1-reduction}]

\[
\mathbb P=\operatorname{Obs}\{\mathrm{L1Emb}(\mathbf P,\mathbf Q)\}.
\]
The sampling condition in \cref{obj:ass:iid-sampling} supplies the product-sampling convention used in \cref{obj:def:minimax-risk}.

\begin{enumerate}
\item For \(x\in[d]\), define
\[
q_{ay,x}=\mathbb P\{X=x,A=a,Y=y\},
\qquad a,y\in\{0,1\}.
\]
By \cref{obj:synth_6,obj:def:observed-margin},
\begin{equation}
\label{eq:eqprop:atoms}
q_{11,x}=\frac{P_x}{4},
\qquad
q_{00,x}=\frac{P_x}{4},
\qquad
q_{10,x}=\frac{Q_x}{4},
\qquad
q_{01,x}=\frac{Q_x}{4}.
\end{equation}
Hence
\[
p_x=\sum_{a,y\in\{0,1\}}q_{ay,x}
=\frac{P_x+Q_x}{2}.
\]
The vector \(p=(p_x)_{x\in[d]}\) lies in \(\Delta_d\), since \(p_x\ge0\) and
\[
\sum_{x\in[d]}p_x
=
\frac12\sum_{x\in[d]}P_x+\frac12\sum_{x\in[d]}Q_x
=1.
\]
Set \(\pi_x=1/2\) for every cell, and set
\[
\mu_{1x}=
\begin{cases}
P_x/(P_x+Q_x),& P_x+Q_x>0,\\
0,& P_x+Q_x=0,
\end{cases}
\qquad
\mu_{0x}=
\begin{cases}
Q_x/(P_x+Q_x),& P_x+Q_x>0,\\
0,& P_x+Q_x=0.
\end{cases}
\]
On cells with \(P_x+Q_x>0\), these quantities lie in \([0,1]\), and the four identities in
\cref{eq:eqprop:atoms} are exactly the factorization required in \cref{obj:def:observed-model-class}.
On cells with \(P_x+Q_x=0\), nonnegativity gives \(P_x=Q_x=0\), so all four atom masses vanish and
the same factorization holds. Moreover,
\[
\sum_{y\in\{0,1\}}q_{1y,x}
=
\sum_{y\in\{0,1\}}q_{0y,x}
=
\frac{P_x+Q_x}{4},
\]
so whenever \(p_x>0\),
\[
\pi_x
=
\frac{\sum_y q_{1y,x}}{p_x}
=
\frac{(P_x+Q_x)/4}{(P_x+Q_x)/2}
=
\frac12.
\]
Because \(0<\epsilon<1/2\), this propensity lies in \([\epsilon,1-\epsilon]\). Thus
\(\mathbb P\in\mathcal D_{d,\epsilon}^{\mathrm{obs}}\). % lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.l1Embedding_observed_reduction

\item Write
\[
\|\mathbf P-\mathbf Q\|_1:=\sum_{x\in[d]}|P_x-Q_x|.
\]
By \cref{obj:def:observed-optimal-value},
\[
\Psi(\mathbb P)
=
\sum_{x\in[d]}p_x\max\{\mu_{0x},\mu_{1x}\}.
\]
A null cell contributes zero. On a cell with \(P_x+Q_x>0\),
\[
p_x\max\{\mu_{0x},\mu_{1x}\}
=
\frac{P_x+Q_x}{2}
\max\left\{\frac{Q_x}{P_x+Q_x},\frac{P_x}{P_x+Q_x}\right\}
=
\frac{\max\{P_x,Q_x\}}2.
\]
For each \(x\),
\[
\frac{\max\{P_x,Q_x\}}2
=
\frac{P_x+Q_x}{4}+\frac{|P_x-Q_x|}{4}.
\]
Summing over \(x\) and using \(\sum_xP_x=\sum_xQ_x=1\) gives
\begin{equation}
\label{eq:eqprop:value}
\Psi(\mathbb P)
=
\frac12+\frac14\|\mathbf P-\mathbf Q\|_1
=
\frac12+\frac14\sum_{x\in[d]}|P_x-Q_x|.
\end{equation} % lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.l1Embedding_observed_reduction

\item Define the single-observation kernel \(K_1\) from a paired draw
\(\eta=(\eta_{\mathrm P},\eta_{\mathrm Q})\in[d]\times[d]\) by
\[
K_1\bigl(\eta,\{(x,a,y)\}\bigr)
=
\begin{cases}
1/4,& (x,a,y)\in
\{(\eta_{\mathrm P},0,0),(\eta_{\mathrm P},1,1),
  (\eta_{\mathrm Q},0,1),(\eta_{\mathrm Q},1,0)\},\\
0,& \text{otherwise}.
\end{cases}
\]
Under one paired draw from \(\mathbf P\otimes\mathbf Q\),
\[
\Pr\{\eta=(x_{\mathrm P},x_{\mathrm Q})\}=P_{x_{\mathrm P}}Q_{x_{\mathrm Q}}.
\]
Therefore
\[
\{(\mathbf P\otimes\mathbf Q)K_1\}\{(x,1,1)\}
=
\sum_{x_{\mathrm Q}\in[d]}\frac14P_xQ_{x_{\mathrm Q}}
=
\frac{P_x}{4},
\]
and similarly
\[
\{(\mathbf P\otimes\mathbf Q)K_1\}\{(x,0,0)\}=\frac{P_x}{4},
\qquad
\{(\mathbf P\otimes\mathbf Q)K_1\}\{(x,1,0)\}=\frac{Q_x}{4},
\qquad
\{(\mathbf P\otimes\mathbf Q)K_1\}\{(x,0,1)\}=\frac{Q_x}{4}.
\]
These are the atom masses in \cref{eq:eqprop:atoms}, so the one-observation kernelized paired
draw has law \(\mathbb P\). % lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.l1SingleKernel_comp_pair

\item Define the fixed-\(n\) product kernel \(K_n\) on \(([d]\times[d])^n\) by
\[
K_n\bigl(\xi,B_1\times\cdots\times B_n\bigr)
=
\prod_{i=1}^n K_1(\xi_i,B_i),
\qquad
\xi=(\xi_1,\ldots,\xi_n).
\]
For \(n=0\), this is the empty product kernel; for \(n>0\), it applies \(K_1\) independently in
each coordinate. Since the fixed paired-sample law is \((\mathbf P\otimes\mathbf Q)^{\otimes n}\)
and the one-coordinate kernelized law is \(\mathbb P\), finite-product independence gives
\begin{equation}
\label{eq:eqprop:product-kernel}
K_n\{(\mathbf P\otimes\mathbf Q)^{\otimes n}\}
=
\mathbb P^{\otimes n}.
\end{equation}
This is the explicit fixed-\(n\) product Markov kernel asserted in the proposition. % lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.l1Embedding_observed_reduction

\item Let \(\widehat V\) be any measurable observed-sample estimator. Since the observed sample
space is finite, \(\widehat V\) is bounded. Define the paired-sample pullback
\[
\widehat L_{\widehat V}(\xi)
=
4\left\{\int \widehat V(o)\,K_n(\xi,do)-\frac12\right\},
\qquad
\xi\in([d]\times[d])^n .
\]
For each fixed paired sample \(\xi\), \cref{eq:eqprop:value} and Jensen's inequality for
\(t\mapsto t^2\) give
\[
\frac1{16}
\left\{
\widehat L_{\widehat V}(\xi)-\|\mathbf P-\mathbf Q\|_1
\right\}^2
=
\left[
\int\{\widehat V(o)-\Psi(\mathbb P)\}\,K_n(\xi,do)
\right]^2
\le
\int\{\widehat V(o)-\Psi(\mathbb P)\}^2\,K_n(\xi,do).
\]
Integrating under \((\mathbf P\otimes\mathbf Q)^{\otimes n}\) and using
\cref{eq:eqprop:product-kernel} yields
\[
\frac1{16}
E_{(\mathbf P\otimes\mathbf Q)^{\otimes n}}
\left[
\left\{
\widehat L_{\widehat V}-\|\mathbf P-\mathbf Q\|_1
\right\}^2
\right]
\le
E_{\mathbb P^{\otimes n}}
\left[
\{\widehat V-\Psi(\mathbb P)\}^2
\right].
\]
Taking the supremum over \((\mathbf P,\mathbf Q)\in\Delta_d^2\), and using the admissibility
proved above, gives
\[
\sup_{(\mathbf P,\mathbf Q)\in\Delta_d^2}
E_{(\mathbf P\otimes\mathbf Q)^{\otimes n}}
\left[
\left\{
\widehat L_{\widehat V}-\|\mathbf P-\mathbf Q\|_1
\right\}^2
\right]
\le
16
\sup_{\widetilde{\mathbb P}\in\mathcal D_{d,\epsilon}^{\mathrm{obs}}}
E_{\widetilde{\mathbb P}^{\otimes n}}
\left[
\{\widehat V-\Psi(\widetilde{\mathbb P})\}^2
\right].
\]
The estimator \(\widehat L_{\widehat V}\) is admissible for the paired problem in
\cref{obj:synth_7}; the zero estimator and the fixed pair chosen at the start give the required
nonempty estimator and parameter classes. Hence
\[
\mathfrak R^{L_1}_{n,d}
\le
16
\sup_{\widetilde{\mathbb P}\in\mathcal D_{d,\epsilon}^{\mathrm{obs}}}
E_{\widetilde{\mathbb P}^{\otimes n}}
\left[
\{\widehat V-\Psi(\widetilde{\mathbb P})\}^2
\right].
\]
Taking the infimum over all observed-sample estimators \(\widehat V\), as in
\cref{obj:def:minimax-risk}, gives
\[
\frac1{16}\mathfrak R^{L_1}_{n,d}
\le
\mathfrak R_{n,d,\epsilon}.
\] % lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.l1_minimax_transfer

\item Combining the observed-model membership, the displayed formulas for \(p_x\), \(\pi_x\), and
\(\Psi(\mathbb P)\), the product-kernel identity, and the minimax transfer proves every asserted
conclusion for the fixed pair \((\mathbf P,\mathbf Q)\). Since the pair was arbitrary, the proposition
follows. % lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.equal_propensity_l1_reduction
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:all-estimator-lower}]
Fix \(0<\epsilon<1/2\).  For an alphabet size \(r\ge2\), set
\[
L_r=\log(er),
\qquad
L_n^\circ=\log(en),
\]
where \(L_n^\circ\) is used only for sample-size logarithms.  Let
\(\mathfrak R^{L_1}_{n,r}\) be the fixed-sample two-sample \(L_1\) risk of
\cref{obj:synth_7}.  Also define the Poissonized two-sample risk
\[
\mathfrak R^{L_1,\mathrm{Pois}}_{n,r}
:=
\inf_{\widehat D}
\sup_{(\mathbf P,\mathbf Q)\in\Delta_r^2}
E_{\mathcal H_{n,\mathbf P,\mathbf Q}}
\left[
\left\{\widehat D(N^{\mathbf P},N^{\mathbf Q})
-\|\mathbf P-\mathbf Q\|_1\right\}^2
\right],
\]
where, under \(\mathcal H_{n,\mathbf P,\mathbf Q}\), the two histograms have
independent coordinates
\[
N^{\mathbf P}_x\sim\operatorname{Pois}(2nP_x),
\qquad
N^{\mathbf Q}_x\sim\operatorname{Pois}(2nQ_x),
\qquad x\in[r],
\]
and the infimum ranges over measurable histogram statistics whose squared loss
is integrable for every \((\mathbf P,\mathbf Q)\in\Delta_r^2\) and whose
displayed worst-case risk is finite.

\begin{enumerate}
\item
\Cref{obj:lem:dense-moment-matching-lower}, using the moment-matching priors of
\citep[Lemma 1 and Section 3.1]{CaiLow2011L1}, supplies an integer
\(d_{\mathrm{dense}}\ge2\) and a constant
\(\gamma_{\mathrm{dense},\epsilon}>0\) such that
\[
r\ge d_{\mathrm{dense}},\qquad r^2<n
\quad\Longrightarrow\quad
\gamma_{\mathrm{dense},\epsilon}\frac{r}{nL_r}
\le
\mathfrak R_{n,r,\epsilon}.
\]
The Poissonized lower bound of
\citep[Theorem 3, equation (24), and Section 3.1]{JiaoHanWeissman2018L1},
with constants \(1/4\) and \(2\), gives a universal
\(\gamma_{\mathrm{JHW}}>0\) such that
\[
r\ge2,\qquad
\frac{r}{4L_r}\le n,\qquad
L_n^\circ\le2L_r
\quad\Longrightarrow\quad
\gamma_{\mathrm{JHW}}
\min\left\{1,\frac{r}{nL_n^\circ}\right\}
\le
\mathfrak R^{L_1,\mathrm{Pois}}_{n,r}.
\]
For fixed-sample estimators this yields
\[
\gamma_{\mathrm{JHW}}
\min\left\{1,\frac{r}{nL_n^\circ}\right\}
-
8\exp\{-n(1-\log2)\}
\le
\mathfrak R^{L_1}_{n,r}.
\]
Indeed, from any fixed-sample estimator \(\widehat D\), construct a histogram
statistic as follows.  Let
\[
\mathcal E_n^{\mathrm{pois}}
=
\left\{
\sum_{x=1}^r N^{\mathbf P}_x\ge n,\ 
\sum_{x=1}^r N^{\mathbf Q}_x\ge n
\right\}.
\]
On \(\mathcal E_n^{\mathrm{pois}}\), take the conditional expectation of
\(\widehat D\) applied to ordered \(n\)-samples drawn from the two histograms;
on \((\mathcal E_n^{\mathrm{pois}})^c\), output \(0\).  On
\(\mathcal E_n^{\mathrm{pois}}\), the conditional reconstruction has the
fixed \(n\)-sample law, so Jensen's inequality gives the fixed-sample risk.
Since \(\|\mathbf P-\mathbf Q\|_1\in[0,2]\), the squared loss on
\((\mathcal E_n^{\mathrm{pois}})^c\) is at most \(4\).  The Chernoff bound for
\(\operatorname{Pois}(2n)\) gives
\[
\Pr\left\{\sum_x N^{\mathbf P}_x<n\right\}
=
\Pr\left\{\sum_x N^{\mathbf Q}_x<n\right\}
\le
\exp\{-n(1-\log2)\},
\]
and hence the displayed fixed-sample lower bound follows after taking the
infimum over \(\widehat D\).

Choose \(d_{\log}\) so that
\[
r\ge d_{\log}\quad\Longrightarrow\quad L_r\le\frac r2,
\]
and choose \(n_{\mathrm{tail}}\ge2\) so that, for every
\(n\ge n_{\mathrm{tail}}\),
\[
8\exp\{-n(1-\log2)\}\le\frac{\gamma_{\mathrm{JHW}}}{2n},
\qquad
8\exp\{-n(1-\log2)\}\le\frac{\gamma_{\mathrm{JHW}}}{2}.
\]
Set
\[
d_\star=\max\{d_{\mathrm{dense}},d_{\log}\},
\qquad
\Xi_\star=\max\{d_\star,n_{\mathrm{tail}}\},
\]
and
\[
c_\epsilon=
\min\left\{
\gamma_{\mathrm{dense},\epsilon},
\frac{\gamma_{\mathrm{JHW}}}{64},
\frac{1}{100\Xi_\star}
\right\}>0.
\]
% lean: all_estimator_lower

\item
For every \(n\ge1\) and \(d\ge2\),
\[
\frac{1}{100n}\le \mathfrak R_{n,d,\epsilon}.
\]
Put
\[
\delta_n=\frac{2}{5\sqrt n}.
\]
In every cell \(x\in[d]\), take \(p_x=1/d\), \(\pi_x=1/2\), and
\(\mu_{0x}=1/2\).  Under \(\mathbb P^{\mathrm{reg}}_{0,n,d}\), set
\(\mu_{1x}=1/2\); under \(\mathbb P^{\mathrm{reg}}_{1,n,d}\), set
\(\mu_{1x}=1/2+\delta_n\).  Since \(d\ge2\), the vector with coordinates \(p_x=1/d\) lies in
\(\Delta_d\).  The fixed range \(0<\epsilon<1/2\) gives
\(1/2\in[\epsilon,1-\epsilon]\), so the displayed propensities satisfy
\cref{obj:ass:fixed-overlap}.  Finally \(0\le\delta_n\le2/5\) places all
arm-specific means in \([0,1]\).  Thus both laws belong to
\(\mathcal D_{d,\epsilon}^{\mathrm{obs}}\), and
\[
\Psi(\mathbb P^{\mathrm{reg}}_{0,n,d})=\frac12,
\qquad
\Psi(\mathbb P^{\mathrm{reg}}_{1,n,d})=\frac12+\delta_n .
\]
The one-observation chi-square divergence is
\[
\chi^2(\mathbb P^{\mathrm{reg}}_{1,n,d},\mathbb P^{\mathrm{reg}}_{0,n,d})
=
\frac{(1/2)\delta_n^2}{(1/2)(1-1/2)}
=
2\delta_n^2,
\]
so tensorization and \(1+t\le e^t\) give
\[
1+
\chi^2\!\left(
(\mathbb P^{\mathrm{reg}}_{1,n,d})^{\otimes n},
(\mathbb P^{\mathrm{reg}}_{0,n,d})^{\otimes n}
\right)
=
(1+2\delta_n^2)^n
\le
\exp(2n\delta_n^2)
\le
2,
\]
because \(2n\delta_n^2=8/25\le\log2\).  Therefore
\[
\operatorname{TV}\!\left(
(\mathbb P^{\mathrm{reg}}_{0,n,d})^{\otimes n},
(\mathbb P^{\mathrm{reg}}_{1,n,d})^{\otimes n}
\right)
\le
\frac12
\sqrt{
\chi^2\!\left(
(\mathbb P^{\mathrm{reg}}_{1,n,d})^{\otimes n},
(\mathbb P^{\mathrm{reg}}_{0,n,d})^{\otimes n}
\right)}
\le
\frac12 .
\]
For two model points with target separation \(2s\) and product-law total
variation at most \(1/2\), thresholding any estimator at the target midpoint
and using
\[
E_j\{(\widehat V-\Psi_j)^2\}
\ge
s^2\,\Pr_j\{|\widehat V-\Psi_j|\ge s\},
\qquad j\in\{0,1\},
\]
gives a worst-case risk at least \(s^2/4\).  With \(s=\delta_n/2\),
\[
\mathfrak R_{n,d,\epsilon}
\ge
\frac{(\delta_n/2)^2}{4}
=
\frac{1}{100n}.
\]
% lean: regular_parametric_lower

\item
If \(n<n_{\mathrm{tail}}\), then \(n\le\Xi_\star\), and therefore
\[
c_\epsilon\min\left\{1,\frac{d}{nL_d}\right\}
\le c_\epsilon
\le\frac{1}{100\Xi_\star}
\le\frac{1}{100n}
\le\mathfrak R_{n,d,\epsilon}.
\]
% lean: all_estimator_lower

\item
Assume \(n\ge n_{\mathrm{tail}}\).  If \(d<d_\star\), then
\(d\le\Xi_\star\).  Since \(L_d\ge1\),
\[
c_\epsilon\min\left\{1,\frac{d}{nL_d}\right\}
\le
c_\epsilon\frac{d}{nL_d}
\le
\frac{d}{100\Xi_\star nL_d}
\le
\frac{1}{100n}
\le
\mathfrak R_{n,d,\epsilon}.
\]
% lean: all_estimator_lower

\item
Assume \(n\ge n_{\mathrm{tail}}\), \(d\ge d_\star\), and \(d^2<n\).  The
dense lower bound from \cref{obj:lem:dense-moment-matching-lower} gives
\[
\gamma_{\mathrm{dense},\epsilon}\frac{d}{nL_d}
\le
\mathfrak R_{n,d,\epsilon}.
\]
Since \(c_\epsilon\le\gamma_{\mathrm{dense},\epsilon}\) and
\(\min\{1,d/(nL_d)\}\le d/(nL_d)\),
\[
c_\epsilon\min\left\{1,\frac{d}{nL_d}\right\}
\le
\mathfrak R_{n,d,\epsilon}.
\]
% lean: dense_moment_matching_lower

\item
It remains to treat \(n\ge n_{\mathrm{tail}}\), \(d\ge d_\star\), and
\(n\le d^2\).  Then
\[
L_n^\circ
\le
\log(ed^2)
=
1+2\log d
\le
2(1+\log d)
=
2L_d.
\]
Because \(d\ge d_{\log}\), also \(L_d\le d/2\), and hence
\[
L_n^\circ\le2L_d\le d.
\]
% lean: all_estimator_lower

\item
Suppose first that
\[
n<\frac{d}{4L_d}.
\]
Define
\[
s_n=\left\lceil nL_n^\circ\right\rceil .
\]
The preceding inequality implies \(nL_d<d/4\).  Since \(L_d\ge1\), this gives
\(n<d\), hence \(L_n^\circ\le L_d\), and therefore
\[
nL_n^\circ\le nL_d<\frac d4<d,
\qquad
2\le s_n\le d.
\]
The ceiling definition also yields the lower bound
\[
nL_n^\circ\le s_n.
\]
Because \(L_n^\circ\ge1\) and \(n\ge n_{\mathrm{tail}}\ge2\), this implies
\(n\le s_n\), and monotonicity of the logarithm gives
\[
L_n^\circ=\log(en)\le \log(es_n).
\]
The upper ceiling estimate is just as important: since \(nL_n^\circ\ge2\),
\[
s_n=\lceil nL_n^\circ\rceil
< nL_n^\circ+1
\le 2nL_n^\circ .
\]
The denominator below is positive because \(s_n\ge2\).  Consequently,
\[
\frac{s_n}{4\log(es_n)}
\le
\frac{2nL_n^\circ}{4\log(es_n)}
\le
\frac n2
\le n,
\]
and the two remaining gate estimates are
\[
L_n^\circ\le2\log(es_n),
\qquad
1\le\frac{s_n}{nL_n^\circ},
\]
the second following by dividing \(nL_n^\circ\le s_n\) by the positive product
\(nL_n^\circ\).
The fixed-sample \(L_1\) bound at alphabet \(s_n\) yields
\[
\gamma_{\mathrm{JHW}}
-
8\exp\{-n(1-\log2)\}
\le
\mathfrak R^{L_1}_{n,s_n},
\]
and the choice of \(n_{\mathrm{tail}}\) gives
\[
\frac{\gamma_{\mathrm{JHW}}}{2}
\le
\mathfrak R^{L_1}_{n,s_n}.
\]
Embedding \(\Delta_{s_n}\) into \(\Delta_d\) by padding the remaining
coordinates with zero preserves the \(L_1\) distance, so
\[
\mathfrak R^{L_1}_{n,s_n}\le \mathfrak R^{L_1}_{n,d}.
\]
By \cref{obj:prop:equal-propensity-l1-reduction},
\[
\frac{1}{16}\mathfrak R^{L_1}_{n,d}
\le
\mathfrak R_{n,d,\epsilon},
\]
and consequently
\[
\frac{\gamma_{\mathrm{JHW}}}{32}
\le
\mathfrak R_{n,d,\epsilon}.
\]
The saturated inequality gives \(d/(nL_d)>4\), hence
\[
\min\left\{1,\frac{d}{nL_d}\right\}=1.
\]
Since \(c_\epsilon\le\gamma_{\mathrm{JHW}}/64\),
\[
c_\epsilon\min\left\{1,\frac{d}{nL_d}\right\}
\le
\mathfrak R_{n,d,\epsilon}.
\]
% lean: saturatedAlphabet_properties, jhw_fixedL1_lower_sub_expTail, fixedL1MinimaxRisk_mono_alphabet, l1_minimax_transfer

\item
Finally suppose that
\[
\frac{d}{4L_d}\le n.
\]
The fixed-sample \(L_1\) bound at alphabet \(d\) gives
\[
\gamma_{\mathrm{JHW}}
\min\left\{1,\frac{d}{nL_n^\circ}\right\}
-
8\exp\{-n(1-\log2)\}
\le
\mathfrak R^{L_1}_{n,d}.
\]
Since \(L_n^\circ\le d\),
\[
\frac1n
\le
\min\left\{1,\frac{d}{nL_n^\circ}\right\}.
\]
The choice of \(n_{\mathrm{tail}}\) therefore yields
\[
\frac{\gamma_{\mathrm{JHW}}}{2}
\min\left\{1,\frac{d}{nL_n^\circ}\right\}
\le
\mathfrak R^{L_1}_{n,d}.
\]
Also \(L_n^\circ\le2L_d\), so
\[
\frac{d}{nL_d}
\le
2\frac{d}{nL_n^\circ},
\qquad
\min\left\{1,\frac{d}{nL_d}\right\}
\le
2\min\left\{1,\frac{d}{nL_n^\circ}\right\}.
\]
Using \(c_\epsilon\le\gamma_{\mathrm{JHW}}/64\) and
\cref{obj:prop:equal-propensity-l1-reduction},
\[
c_\epsilon\min\left\{1,\frac{d}{nL_d}\right\}
\le
\frac{\gamma_{\mathrm{JHW}}}{32}
\min\left\{1,\frac{d}{nL_n^\circ}\right\}
\le
\frac{\mathfrak R^{L_1}_{n,d}}{16}
\le
\mathfrak R_{n,d,\epsilon}.
\]
% lean: all_estimator_lower, jhw_fixedL1_lower_sub_expTail, l1_minimax_transfer
\end{enumerate}

The displayed alternatives cover every \(n\ge1\) and \(d\ge2\).  Thus the
constant \(c_\epsilon>0\) satisfies
\[
c_\epsilon\min\left\{1,\frac{d}{n\log(ed)}\right\}
\le
\mathfrak R_{n,d,\epsilon}
\]
for all \(n\ge1\) and \(d\ge2\).
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:matched-minimax-frontier}]
\begin{enumerate}
\item \Cref{obj:ass:iid-sampling} identifies the sampling law with the canonical product law.  Apply \cref{obj:thm:jackson-factorial-upper} with this sampling law and choose its universal Jackson tuning constants \(H_0>0\), \(\kappa\in(0,1)\), and cutoff \(D_0>2\).  Fix \(\epsilon\in(0,1/2)\).  The same result gives a constant \(C_{\epsilon}^{\mathrm{JF}}>0\) such that, for all \(n\ge1\), \(d\ge2\), and \(\mathbb P\in\mathcal D_{d,\epsilon}^{\mathrm{obs}}\),
\[
E_{\mathbb P^{\otimes n}}
\left[
  \left(\widehat V_{n,d}^{\mathrm{JF}}-\Psi(\mathbb P)\right)^2
\right]
\le
C_{\epsilon}^{\mathrm{JF}}
\min\left\{1,\frac{d}{n\log(ed)}\right\}.
\]
The lower-bound input is \cref{obj:thm:all-estimator-lower}, using the moment-matching priors of \citep[Lemma 1 and Section 3.1]{CaiLow2011L1} and the Poissonized \(L_1\) minimax lower bound of \citep[Theorem 3 and equation (24)]{JiaoHanWeissman2018L1}.  It gives a constant \(c_{\epsilon}^{\mathrm{low}}>0\) such that, for all \(n\ge1\) and \(d\ge2\),
\[
c_{\epsilon}^{\mathrm{low}}
\min\left\{1,\frac{d}{n\log(ed)}\right\}
\le
\mathfrak R_{n,d,\epsilon}.
\]
Set
\[
c_\epsilon:=c_{\epsilon}^{\mathrm{low}},
\qquad
C_\epsilon:=C_{\epsilon}^{\mathrm{JF}}+c_{\epsilon}^{\mathrm{low}} .
\]
Then \(0<c_\epsilon\le C_\epsilon\).  For later comparison, when \(n\ge1\) and \(d\ge2\), \(L_d=\log(ed)>0\), and hence
\[
\rho_{n,d}:=
\min\left\{1,\frac{d}{nL_d}\right\}\ge0 .
\]
% lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.matched_minimax_frontier

\item The preceding lower bound gives
\[
c_\epsilon\rho_{n,d}\le \mathfrak R_{n,d,\epsilon}.
\]
For the middle inequality, define for each measurable observed-data estimator \(\widehat V\)
\[
\mathcal E_{n,d,\epsilon}(\widehat V)
:=
\sup_{\mathbb P\in\mathcal D_{d,\epsilon}^{\mathrm{obs}}}
E_{\mathbb P^{\otimes n}}
\left[
  \left(\widehat V-\Psi(\mathbb P)\right)^2
\right].
\]
By \cref{obj:def:minimax-risk},
\[
\mathfrak R_{n,d,\epsilon}
=
\inf_{\widehat V}\mathcal E_{n,d,\epsilon}(\widehat V),
\]
where the infimum is over the same measurable sample maps.  The estimator \(\widehat V_{n,d}^{\mathrm{JF}}\) in \cref{obj:def:jackson-factorial-estimator} is one admissible map, and squared-error risks are nonnegative:
\[
E_{\mathbb P^{\otimes n}}
\left[
  \left(\widehat V_{n,d}^{\mathrm{JF}}-\Psi(\mathbb P)\right)^2
\right]\ge0 .
\]
Therefore evaluating the infimum at this estimator gives
\[
\mathfrak R_{n,d,\epsilon}
\le
\mathcal E_{n,d,\epsilon}(\widehat V_{n,d}^{\mathrm{JF}}).
\]
% lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.matched_minimax_frontier

\item It remains to bound this particular worst-case risk.  If the set over which the displayed supremum is taken is empty, the real-supremum convention gives
\[
\mathcal E_{n,d,\epsilon}(\widehat V_{n,d}^{\mathrm{JF}})=0,
\]
and \(0\le C_\epsilon\rho_{n,d}\).  Otherwise, for every \(\mathbb P\in\mathcal D_{d,\epsilon}^{\mathrm{obs}}\), the upper bound from \cref{obj:thm:jackson-factorial-upper} gives
\[
E_{\mathbb P^{\otimes n}}
\left[
  \left(\widehat V_{n,d}^{\mathrm{JF}}-\Psi(\mathbb P)\right)^2
\right]
\le
C_{\epsilon}^{\mathrm{JF}}\rho_{n,d}
\le
(C_{\epsilon}^{\mathrm{JF}}+c_{\epsilon}^{\mathrm{low}})\rho_{n,d}
=
C_\epsilon\rho_{n,d},
\]
where the second inequality uses \(\rho_{n,d}\ge0\).  Taking the supremum over \(\mathbb P\) yields
\[
\mathcal E_{n,d,\epsilon}(\widehat V_{n,d}^{\mathrm{JF}})
\le
C_\epsilon
\min\left\{1,\frac{d}{nL_d}\right\}.
\]
% lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.matched_minimax_frontier

\item Finally fix a measurable observed-data estimator \(\widehat V\) and define its causal worst-case risk by
\[
\mathcal E^{\mathrm{causal}}_{n,d,\epsilon}(\widehat V)
:=
\sup_{P\in\mathcal P_{d,\epsilon}}
E_{\operatorname{Obs}(P)^{\otimes n}}
\left[
  \left(\widehat V-V^\star(P)\right)^2
\right].
\]
The two attained-risk sets are equal.  Indeed, if \(P\in\mathcal P_{d,\epsilon}\), then \cref{obj:prop:identification-and-extension} gives
\[
\operatorname{Obs}(P)\in\mathcal D_{d,\epsilon}^{\mathrm{obs}},
\]
and \cref{obj:prop:causal-optimal-value-corollary} gives
\[
V^\star(P)=\Psi\{\operatorname{Obs}(P)\}.
\]
Thus each causal risk value is an observed-law risk value:
\[
E_{\operatorname{Obs}(P)^{\otimes n}}
\left[
  \left(\widehat V-V^\star(P)\right)^2
\right]
=
E_{\operatorname{Obs}(P)^{\otimes n}}
\left[
  \left(\widehat V-\Psi\{\operatorname{Obs}(P)\}\right)^2
\right].
\]
Conversely, if \(\mathbb P\in\mathcal D_{d,\epsilon}^{\mathrm{obs}}\), then \cref{obj:prop:identification-and-extension} supplies \(P\in\mathcal P_{d,\epsilon}\) with
\[
\operatorname{Obs}(P)=\mathbb P,
\]
and \cref{obj:prop:causal-optimal-value-corollary} again gives
\[
V^\star(P)=\Psi(\mathbb P).
\]
Hence the corresponding observed-law risk is attained in the causal class:
\[
E_{\mathbb P^{\otimes n}}
\left[
  \left(\widehat V-\Psi(\mathbb P)\right)^2
\right]
=
E_{\operatorname{Obs}(P)^{\otimes n}}
\left[
  \left(\widehat V-V^\star(P)\right)^2
\right].
\]
Therefore
\[
\mathcal E^{\mathrm{causal}}_{n,d,\epsilon}(\widehat V)
=
\mathcal E_{n,d,\epsilon}(\widehat V)
\]
for every measurable observed-data estimator \(\widehat V\).  Taking the infimum over the common estimator class in \cref{obj:def:minimax-risk,obj:synth_4} gives
\[
\mathfrak R^{\mathrm{causal}}_{n,d,\epsilon}
=
\mathfrak R_{n,d,\epsilon}.
\]
Together with the three inequalities above, this proves the result.
% lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.matched_minimax_frontier
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:consistency-and-parametric-boundaries}]
Fix \(\epsilon\in(0,1/2)\).  The published moment-matching conclusion of \citep[Lemma 1 and Section 3.1]{CaiLow2011L1} and the published Poissonized \(L_1\) lower bound of \citep[Theorem 3, equation (24)]{JiaoHanWeissman2018L1} enter through \cref{obj:thm:matched-minimax-frontier,obj:prop:parent-reduction}.  By \cref{obj:thm:matched-minimax-frontier}, there are constants \(c_\epsilon,C_\epsilon\) with \(0<c_\epsilon\le C_\epsilon\) such that, for every \(n\ge1\) and \(d\ge2\),
\[
c_\epsilon \Phi_{n,d}
\le
\mathfrak R_{n,d,\epsilon}
\le
C_\epsilon \Phi_{n,d},
\qquad
\mathfrak R^{\mathrm{causal}}_{n,d,\epsilon}
=
\mathfrak R_{n,d,\epsilon},
\]
where
\[
\Phi_{n,d}
:=
\min\left\{1,\frac{d}{n\log(ed)}\right\}.
\]
The upper comparison uses the Jackson-factorial worst-case risk term displayed in \cref{obj:thm:matched-minimax-frontier}. % lean: consistency_and_parametric_boundaries

Let \(d_\bullet=(d_n)_{n\ge1}\) satisfy \(d_n\ge2\) for every \(n\).  Define
\[
\Phi_n:=\Phi_{n,d_n},
\qquad
\rho_n:=\frac{d_n}{n\log(ed_n)},
\qquad
\chi_n:=\frac{d_n}{n\log(en)}.
\]
All displayed denominators are positive.  We prove
\[
\Phi_n\to0
\quad\Longleftrightarrow\quad
\chi_n\to0,
\]
which is exactly \(d_n=o\{n\log(en)\}\).  Suppose first that \(\Phi_n\to0\).  Then eventually \(\Phi_n<1/2\), hence \(\Phi_n=\rho_n\), so \(\rho_n\to0\).  For all sufficiently large \(n\),
\[
0\le \chi_n\le \frac1{\log(en)}+2\rho_n .
\]
Indeed, if \(d_n<n\), then
\[
\chi_n=\frac{d_n/n}{\log(en)}
\le \frac1{\log(en)}.
\]
If \(n\le d_n\), set
\[
\tau_n:=\frac{d_n}{n}\ge1.
\]
Then
\[
\log(ed_n)=\log(en)+\log\tau_n,
\qquad
\rho_n=\frac{\tau_n}{\log(en)+\log\tau_n}.
\]
Since eventually \(\rho_n\le1/2\) and \(\log\tau_n\le\tau_n\),
\[
2\tau_n\le \log(en)+\log\tau_n\le \log(en)+\tau_n,
\]
so \(\tau_n\le\log(en)\).  Consequently,
\[
\log(ed_n)\le 2\log(en),
\qquad
\chi_n=\frac{\tau_n}{\log(en)}
\le
2\frac{\tau_n}{\log(ed_n)}
=
2\rho_n .
\]
Because \(1/\log(en)\to0\), this gives \(\chi_n\to0\).

Conversely, suppose \(\chi_n\to0\).  For all \(n\ge1\),
\[
0\le \rho_n\le n^{-1/2}+2\chi_n .
\]
If \(d_n\le\sqrt n\), then \(\log(ed_n)\ge1\), and therefore
\[
\rho_n\le\frac{d_n}{n}\le n^{-1/2}.
\]
If \(\sqrt n<d_n\), then \(n<d_n^2\), whence
\[
en\le (ed_n)^2,
\qquad
\log(en)\le2\log(ed_n),
\]
and therefore
\[
\rho_n\le2\chi_n .
\]
Since \(n^{-1/2}\to0\), we get \(\rho_n\to0\), and then
\[
\Phi_n=\min\{1,\rho_n\}\to0 .
\] % lean: phase_scale_tendsto_iff_littleO

The two-sided comparison with \(\Phi_n\) gives
\[
\mathfrak R_{n,d_n,\epsilon}\to0
\quad\Longleftrightarrow\quad
\Phi_n\to0 .
\]
For the forward implication, the lower comparison gives
\[
0\le\Phi_n\le c_\epsilon^{-1}\mathfrak R_{n,d_n,\epsilon}
\]
for all \(n\ge1\).  For the reverse implication, the upper comparison gives
\[
0\le\mathfrak R_{n,d_n,\epsilon}\le C_\epsilon\Phi_n
\]
for all \(n\ge1\).  Combining this equivalence with the preceding display proves observed consistency. % lean: risk_zero_iff_phase_scale

We next prove the observed parametric boundary.  Suppose
\[
\frac{\zeta_-}{n}
\le
\mathfrak R_{n,d_n,\epsilon}
\le
\frac{\zeta_+}{n}
\]
eventually for constants \(0<\zeta_-\le\zeta_+\).  The upper parametric bound and the lower frontier bound imply
\[
n\Phi_n\le \frac{\zeta_+}{c_\epsilon}
\]
eventually.  For all sufficiently large \(n>\zeta_+/c_\epsilon\), the alternative \(\rho_n>1\) would give \(\Phi_n=1\) and hence \(n\Phi_n=n>\zeta_+/c_\epsilon\), a contradiction.  Thus eventually \(\rho_n\le1\), and so
\[
\frac{d_n}{\log(ed_n)}
=
n\rho_n
=
n\Phi_n
\le
\frac{\zeta_+}{c_\epsilon}.
\]
For \(d_n\ge2\),
\[
\log(ed_n)
=
1+\log d_n
=
1+2\log\sqrt{d_n}
\le
1+2(\sqrt{d_n}-1)
\le
2\sqrt{d_n}.
\]
Therefore
\[
\sqrt{d_n}
\le
2\frac{d_n}{\log(ed_n)}
\le
2\frac{\zeta_+}{c_\epsilon},
\]
and hence \(d_n\) is eventually bounded.  This is \(d_n=O(1)\). % lean: parametric_implies_bounded_alphabet

Conversely, assume \(d_n=O(1)\).  Then there are a real number \(K_{\mathrm{bd}}\) and an index \(N_{\mathrm{bd}}\) such that \(d_n\le K_{\mathrm{bd}}\) for all \(n\ge N_{\mathrm{bd}}\).  Enlarging over the finite set \(1\le n<N_{\mathrm{bd}}\), choose an integer \(d_{\mathrm{bd}}\) such that
\[
d_n\le d_{\mathrm{bd}}
\qquad\text{for every } n\ge1 .
\]
This finite enlargement is the passage from an eventual \(O(1)\) bound to a single global alphabet bound. % lean: bounded_alphabet_sequence_global_bound

The bounded-alphabet sequence clause of \cref{obj:prop:parent-reduction} applied with this \(d_{\mathrm{bd}}\) supplies constants \(\eta_-,\eta_+\) with \(0<\eta_-\le\eta_+\) such that, for all sufficiently large \(n\),
\[
\frac{\eta_-}{n}
\le
\mathfrak R_{n,d_n,\epsilon}
\le
\frac{\eta_+}{n}.
\]
Together with the preceding implication, this proves the observed parametric equivalence. % lean: consistency_and_parametric_boundaries

It remains to transfer the two equivalences to the causal risk.  The equality supplied by \cref{obj:thm:matched-minimax-frontier} gives, for every \(n\ge1\),
\[
\mathfrak R^{\mathrm{causal}}_{n,d_n,\epsilon}
=
\mathfrak R_{n,d_n,\epsilon}.
\]
Hence
\[
\mathfrak R^{\mathrm{causal}}_{n,d_n,\epsilon}\to0
\quad\Longleftrightarrow\quad
\mathfrak R_{n,d_n,\epsilon}\to0,
\]
and observed consistency gives
\[
\mathfrak R^{\mathrm{causal}}_{n,d_n,\epsilon}\to0
\quad\Longleftrightarrow\quad
d_n=o\{n\log(en)\}.
\] % lean: consistency_and_parametric_boundaries

The same equality transports the parametric-rate property itself:
\[
\mathfrak R^{\mathrm{causal}}_{n,d_n,\epsilon}
\text{ has the parametric }n^{-1}\text{ rate}
\quad\Longleftrightarrow\quad
\mathfrak R_{n,d_n,\epsilon}
\text{ has the parametric }n^{-1}\text{ rate}.
\]
Indeed, the same eventual constants bound both sequences after replacing one risk by the other in the displayed equality.  Composing this equivalence with the observed parametric equivalence proves the causal parametric boundary. % lean: consistency_and_parametric_boundaries
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:parent-reduction}]
The published moment-matching construction and absolute-value approximation rate of
\citep[Lemma 1 and Section 3.1]{CaiLow2011L1}, together with the published Poissonized two-sample
\(L_1\) lower-bound argument of \citep[Theorem 3 and equation (24)]{JiaoHanWeissman2018L1},
are the literature inputs used by \cref{obj:thm:matched-minimax-frontier}. Apply
\cref{obj:thm:matched-minimax-frontier} and choose the tuning constants it supplies. Write them as
\(H_0\), \(\kappa\), and \(D_0\). They satisfy
\[
H_0>0,\qquad 0<\kappa<1,\qquad D_0>2 .
\]
For each fixed \(\epsilon\in(0,1/2)\), \cref{obj:thm:matched-minimax-frontier} gives constants
\(c_\epsilon^{\mathrm{mf}}\) and \(C_\epsilon^{\mathrm{mf}}\), with
\[
0<c_\epsilon^{\mathrm{mf}}\le C_\epsilon^{\mathrm{mf}},
\]
such that, for every \(n\ge1\) and \(d\ge2\), with
\[
L_d=\log(ed),
\]
one has
\[
c_\epsilon^{\mathrm{mf}}
\min\left\{1,\frac{d}{nL_d}\right\}
\le
\mathfrak R_{n,d,\epsilon}
\le
C_\epsilon^{\mathrm{mf}}
\min\left\{1,\frac{d}{nL_d}\right\}.
\]

1. Fix \(n\ge1\) and \(d\ge2\). The logarithms \(L_d\) and \(\log(en)\) are positive. Since
\[
L_d
=
1+\log d
=
1+2\log\sqrt d
\le
1+2(\sqrt d-1)
\le
2\sqrt d,
\]
we have
\[
\frac{\sqrt d}{n}
\le
2\,\frac{d}{nL_d}.
\]
Define
\[
\tau_{n,d}=\frac{d}{nL_d}.
\]
We prove
\[
\min\left\{1,\frac{\sqrt d}{n}+\frac{d}{n\log(en)}\right\}
\le
4\min\left\{1,\tau_{n,d}\right\}.
\]
If \(\tau_{n,d}\ge1\), the right-hand side is \(4\), while the left-hand side is at most \(1\).
Suppose \(\tau_{n,d}<1\). If \(d\le n\), then \(ed\le en\), so
\[
L_d\le \log(en)\le 2\log(en).
\]
If \(d>n\), put
\[
\eta_{n,d}=\frac dn .
\]
Then \(\eta_{n,d}>1\),
\[
L_d=\log(en)+\log\eta_{n,d},
\]
and \(\tau_{n,d}<1\) gives \(\eta_{n,d}<L_d\). For every \(\eta>1\),
\[
\log\eta
=
2\log\sqrt\eta
\le
2(\sqrt\eta-1)
\le
\frac{\eta}{2},
\]
where the last inequality is equivalent to \((\sqrt\eta-2)^2\ge0\). Hence
\[
\eta_{n,d}
<
\log(en)+\log\eta_{n,d}
\le
\log(en)+\frac{\eta_{n,d}}2,
\]
so \(\eta_{n,d}\le2\log(en)\), and therefore
\[
L_d
=
\log(en)+\log\eta_{n,d}
\le
\log(en)+\frac{\eta_{n,d}}2
\le
2\log(en).
\]
Thus, throughout the case \(\tau_{n,d}<1\),
\[
\frac{d}{n\log(en)}
\le
2\,\frac{d}{nL_d}.
\]
Combining this with the square-root bound gives
\[
\frac{\sqrt d}{n}+\frac{d}{n\log(en)}
\le
4\,\frac{d}{nL_d}
=
4\min\left\{1,\tau_{n,d}\right\},
\]
and hence the displayed comparison follows. Consequently,
\[
\frac{c_\epsilon^{\mathrm{mf}}}{4}
\min\left\{1,\frac{\sqrt d}{n}+\frac{d}{n\log(en)}\right\}
\le
c_\epsilon^{\mathrm{mf}}
\min\left\{1,\frac{d}{nL_d}\right\}
\le
\mathfrak R_{n,d,\epsilon}.
\]
For the upper bound, \(d\ge2\) gives \(L_d\ge1\), and therefore
\[
\mathfrak R_{n,d,\epsilon}
\le
C_\epsilon^{\mathrm{mf}}
\min\left\{1,\frac{d}{nL_d}\right\}
\le
C_\epsilon^{\mathrm{mf}}\frac dn .
\]
Taking \(c=c_\epsilon^{\mathrm{mf}}/4\) and \(C=C_\epsilon^{\mathrm{mf}}\) proves the matched-frontier assertion in the proposition. % lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.parent_reduction

2. Fix \(n\), \(d\), \(\epsilon\), and an observed sample with \(2\le d<D_0\). \Cref{obj:def:jackson-factorial-estimator} selects its empirical-ratio branch whenever \(d<D_0\). Hence, for that sample,
\[
\widehat V_{n,d}^{\mathrm{JF}}
=
\widehat V_{n,d}^{\mathrm{ER}} .
\]
% lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.parent_reduction

3. Fix \(\epsilon\in(0,1/2)\) and a sequence \((d_n)\) such that \(d_n\ge2\) for every \(n\) and
\(d_n\le d_{\max}\) for some finite \(d_{\max}\). Then \(d_{\max}\ge2\). Define
\[
c_{\epsilon,d_{\max}}
=
c_\epsilon^{\mathrm{mf}}\,\frac{2}{\log(ed_{\max})},
\qquad
C_{\epsilon,d_{\max}}
=
C_\epsilon^{\mathrm{mf}}\,d_{\max}.
\]
Since \(\log(ed_{\max})\ge1\) and \(d_{\max}\ge2\),
\[
\frac{2}{\log(ed_{\max})}\le d_{\max},
\]
so
\[
0<c_{\epsilon,d_{\max}}\le C_{\epsilon,d_{\max}}.
\]
For every \(n\ge d_{\max}\), one has \(d_n\le n\). With
\[
L_{d_n}=\log(ed_n),
\]
the bound \(d_n\ge2\) gives \(L_{d_n}\ge1\), and therefore
\[
\frac{d_n}{nL_{d_n}}\le1.
\]
Thus the minimum in the matched frontier equals \(d_n/(nL_{d_n})\). Monotonicity of the logarithm gives
\[
L_{d_n}\le \log(ed_{\max}),
\]
and, since \(d_n\ge2\),
\[
\frac{2}{\log(ed_{\max})}
\le
\frac{d_n}{L_{d_n}}.
\]
Therefore, for every \(n\ge d_{\max}\),
\[
\frac{c_{\epsilon,d_{\max}}}{n}
\le
c_\epsilon^{\mathrm{mf}}\frac{d_n}{nL_{d_n}}
\le
\mathfrak R_{n,d_n,\epsilon}.
\]
Similarly,
\[
\mathfrak R_{n,d_n,\epsilon}
\le
C_\epsilon^{\mathrm{mf}}\frac{d_n}{nL_{d_n}}
\le
C_\epsilon^{\mathrm{mf}}\frac{d_n}{n}
\le
\frac{C_{\epsilon,d_{\max}}}{n}.
\]
These inequalities hold for every \(n\ge d_{\max}\), hence for all sufficiently large \(n\). % lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.parent_reduction
\end{proof}
