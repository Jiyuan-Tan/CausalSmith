\begin{proof}[Proof of \cref{obj:thm:jackson-factorial-upper}]
\begin{enumerate}
\item Choose the universal tuning supplied by
\cref{obj:lem:centered-factorial-pilot-control}.  Thus \(H_0\) is the
universal self-normalized Poisson pilot-radius constant,
\[
\kappa=\frac{1}{100000},
\qquad
D_0=3,
\]
so \(H_0>0\), \(\kappa\in(0,1)\), and \(D_0>2\).  Fix
\(0<\epsilon<1/2\), let \(C_{\epsilon}^{\mathrm{cell}}>0\) be the cell
constant supplied by \cref{obj:lem:centered-factorial-pilot-control}, and put
\[
\Xi_\epsilon:=1+\epsilon^{-1}+\epsilon^{-2}.
\]
Define
\[
C_\epsilon
:=
20000000\{1+C_{\epsilon}^{\mathrm{cell}}+(C_{\epsilon}^{\mathrm{cell}})^2\}
+
2000\Xi_\epsilon+10 .
\]
Then \(C_\epsilon>0\).  For \(d\ge2\), write
\[
L_d:=\log(ed).
\]
Since \(L_d=1+\log d\) and \(\log d\le d-1\),
\[
0<L_d\le d,
\qquad
1\le L_2\le2 .
\]
Fix \(n\ge1\), \(d\ge2\), and
\(\mathbb P\in\mathcal D_{d,\epsilon}^{\mathrm{obs}}\).  Let
\(q_x=(q_{00,x},q_{01,x},q_{10,x},q_{11,x})\) and let \(p_x\) be the
covariate-cell mass.  By \cref{obj:prop:identification-and-extension},
\[
\Psi(\mathbb P)=\sum_{x=1}^d \bar f_\epsilon(q_x),
\qquad
0\le \bar f_\epsilon(q_x)\le s(q_x).
\]
The four-cell vectors partition the one-observation law, so
\[
\sum_{x=1}^d s(q_x)=1 .
\]
Consequently
\[
0\le \Psi(\mathbb P)\le1 .
\]
% lean: jackson_factorial_upper

\item Consider the bounded-alphabet branch \(d<D_0\).  Since \(D_0=3\) and
\(d\ge2\), this branch has \(d=2\).  By
\cref{obj:def:jackson-factorial-estimator},
\[
\widehat V_{n,2}^{\mathrm{JF}}
=
\widehat V_{n,2}^{\mathrm{ER}} .
\]
Assume first that \(n\ge4\).  For \(x\in[2]\) and \(a\in\{0,1\}\), define
\[
\zeta_{a,x}
:=
\frac{\overline N_x}{n}\,
\frac{\overline S_{ax}}{\overline N_{ax}},
\qquad
\zeta^0_{a,x}:=p_x\mu_{ax},
\]
with the \(0/0=0\) convention from
\cref{obj:def:empirical-ratio-estimator}, and put
\[
\delta_{a,x}:=\zeta_{a,x}-\zeta^0_{a,x}.
\]
If \(p_x=0\), then \(\zeta^0_{a,x}=0\) and \(\zeta_{a,x}=0\) almost surely.
If \(p_x>0\), the fixed-overlap representation in
\cref{obj:def:observed-model-class} gives
\[
\eta_{a,x}:=\mathbb P(A=a\mid X=x)\ge\epsilon ,
\]
and the conditional success mean in that arm and cell is \(\mu_{ax}\).  Define
\[
\delta^{\mathrm{res}}_{a,x}
:=
\frac{\overline N_x}{n}\,
\mathbf 1\{\overline N_{ax}>0\}
\frac{1}{\overline N_{ax}}
\sum_{i=1}^n
\mathbf 1\{X_i=x,A_i=a\}(Y_i-\mu_{ax}),
\]
\[
\delta^{\mathrm{cell}}_{a,x}
:=
\left(\frac{\overline N_x}{n}-p_x\right)\mu_{ax},
\qquad
\delta^{\mathrm{zero}}_{a,x}
:=
\mu_{ax}\frac{\overline N_x}{n}\mathbf 1\{\overline N_{ax}=0\}.
\]
The totalized-ratio identity is
\[
\delta_{a,x}
=
\delta^{\mathrm{res}}_{a,x}
+
\delta^{\mathrm{cell}}_{a,x}
-
\delta^{\mathrm{zero}}_{a,x}.
\]
Conditional on \(\overline N_x=k\), the arm count is binomial with success
probability \(\eta_{a,x}\).  For
\(\mathcal B_{k,\eta}\sim\operatorname{Bin}(k,\eta)\) with
\(\eta\ge\epsilon\),
\[
E\!\left[\frac{\mathbf 1\{\mathcal B_{k,\eta}>0\}}{\mathcal B_{k,\eta}}\right]
\le
2E\!\left[\frac{1}{\mathcal B_{k,\eta}+1}\right]
=
\frac{2\{1-(1-\eta)^{k+1}\}}{(k+1)\eta}
\le
\frac{2}{k\epsilon}
\]
for \(k>0\), while the left-hand side is \(0\) for \(k=0\).  Since
\(Y_i\in\{0,1\}\) and \(\mu_{ax}\in[0,1]\), the centered residual has
conditional second moment at most one; hence
\[
E_{\mathbb P}\{(\delta^{\mathrm{res}}_{a,x})^2\}
\le
E_{\mathbb P}\!\left[
\frac{\overline N_x^2}{n^2}
\frac{\mathbf 1\{\overline N_{ax}>0\}}{\overline N_{ax}}
\right]
\le
\frac{2p_x}{n\epsilon}.
\]
Also
\[
E_{\mathbb P}\{(\delta^{\mathrm{cell}}_{a,x})^2\}
\le
E_{\mathbb P}\left(\frac{\overline N_x}{n}-p_x\right)^2
\le
\frac1n .
\]
Let
\[
\Lambda_{n,x}:=
p_x\exp\!\left[-\frac{n-2}{2}\epsilon p_x\right].
\]
Expanding
\(\overline N_x^2\mathbf 1\{\overline N_{ax}=0\}\) into diagonal and
ordered-pair contributions gives
\[
E_{\mathbb P}\{(\delta^{\mathrm{zero}}_{a,x})^2\}
\le
\frac1n+\Lambda_{n,x}^2 .
\]
The elementary inequality \(t\exp(-ut)\le u^{-1}\), applied with
\(u=(n-2)\epsilon/2\), and the condition \(n\ge4\) give
\[
\Lambda_{n,x}\le \frac{4}{n\epsilon}.
\]
Using \(p_x\le1\) and \(n^{-2}\le n^{-1}\), we obtain
\[
E_{\mathbb P}\delta_{a,x}^2
\le
4E_{\mathbb P}\{(\delta^{\mathrm{res}}_{a,x})^2\}
+
2E_{\mathbb P}\{(\delta^{\mathrm{cell}}_{a,x})^2\}
+
4E_{\mathbb P}\{(\delta^{\mathrm{zero}}_{a,x})^2\}
\le
\frac{100\Xi_\epsilon}{n}.
\]
Moreover,
\[
\widehat V_{n,2}^{\mathrm{ER}}
=
\sum_{x=1}^2 \max\{\zeta_{0,x},\zeta_{1,x}\},
\qquad
\Psi(\mathbb P)
=
\sum_{x=1}^2 \max\{\zeta^0_{0,x},\zeta^0_{1,x}\}.
\]
The maximum map is one-Lipschitz in the sup norm, and Cauchy--Schwarz over the
four pairs \((a,x)\) yields
\[
\left(
\widehat V_{n,2}^{\mathrm{ER}}-\Psi(\mathbb P)
\right)^2
\le
4\sum_{x=1}^2\sum_{a\in\{0,1\}}\delta_{a,x}^2 .
\]
Therefore
\[
\operatorname{Risk}_{\mathbb P}
(\widehat V_{n,2}^{\mathrm{ER}},\Psi)
\le
\frac{1600\Xi_\epsilon}{n}.
\]
Since \(n\ge4\), \(L_2\ge1\), and \(L_2\le2\),
\[
\min\left\{1,\frac{2}{nL_2}\right\}
=
\frac{2}{nL_2},
\qquad
\frac{1600\Xi_\epsilon}{n}
\le
C_\epsilon\frac{2}{nL_2}.
\]

If \(n\le3\), the empirical ratios lie in \([0,1]\), their weights sum to
one, and therefore
\[
0\le \widehat V_{n,2}^{\mathrm{ER}}\le1.
\]
Together with \(0\le\Psi(\mathbb P)\le1\), this gives
\[
\operatorname{Risk}_{\mathbb P}
(\widehat V_{n,2}^{\mathrm{ER}},\Psi)
\le1 .
\]
Also
\[
\frac13\le \frac{2}{nL_2},
\qquad
\frac13\le1,
\]
so
\[
\frac13\le
\min\left\{1,\frac{2}{nL_2}\right\}.
\]
Since \(C_\epsilon\ge3\), the desired bound follows throughout the
bounded-alphabet branch.
% lean: fixedStratumArmScore_singleton_mse_le
% lean: empiricalRatio_sqRisk_two_le

\item Now suppose that \(d\ge D_0\) and
\[
n<\frac{d}{L_d}.
\]
In this branch, \cref{obj:def:jackson-factorial-estimator} outputs the constant
estimator \(1/2\).  Since \(\Psi(\mathbb P)\in[0,1]\),
\[
\operatorname{Risk}_{\mathbb P}\left(\frac12,\Psi\right)\le1 .
\]
The scale condition gives
\[
1<\frac{d}{nL_d},
\]
and hence
\[
\min\left\{1,\frac{d}{nL_d}\right\}=1 .
\]
Since \(C_\epsilon\ge1\), the asserted inequality holds in this branch.
% lean: jackson_factorial_upper

\item It remains to treat the Jackson branch
\[
d\ge D_0,
\qquad
\frac{d}{L_d}\le n .
\]
Set
\[
m:=\frac n8,
\qquad
\chi_{n,d}:=\frac{d}{nL_d}.
\]
Then \(0\le\chi_{n,d}\le1\).  In the uncapped finite-Poisson marked
experiment, draw
\[
M^{\mathrm{unc}}\sim\operatorname{Pois}(n/4),
\]
then draw \(M^{\mathrm{unc}}\) i.i.d. observations from \(\mathbb P\), each
with an independent fair pilot/evaluation mark.  Let
\(T_x^{\mathrm{JF},\mathrm{unc}}\) be the clipped cell statistic computed from
the resulting cell-\(x\) pilot and evaluation counts, and define
\[
\widetilde V^{\mathrm{unc}}
:=
\Pi_{[0,1]}
\left(
\sum_{x=1}^d T_x^{\mathrm{JF},\mathrm{unc}}
\right).
\]
By \cref{obj:lem:centered-factorial-pilot-control}, the cell blocks are
independent and, for every \(x\),
\[
\left|
E T_x^{\mathrm{JF},\mathrm{unc}}-\bar f_\epsilon(q_x)
\right|
\le
C_{\epsilon}^{\mathrm{cell}}
\left\{
\sqrt{\frac{p_x}{mL_d}}+\frac{1}{mL_d}
\right\},
\]
\[
\operatorname{Var}(T_x^{\mathrm{JF},\mathrm{unc}})
\le
C_{\epsilon}^{\mathrm{cell}}d^{1/16}
\left\{
\frac{p_xL_d}{m}+\frac{L_d^2}{m^2}
\right\}.
\]
Projection onto \([0,1]\) is nonexpansive for squared distance from
\(\Psi(\mathbb P)\in[0,1]\), so
\[
E\left[
\left\{\widetilde V^{\mathrm{unc}}-\Psi(\mathbb P)\right\}^2
\right]
\le
E\left[
\left\{
\sum_{x=1}^dT_x^{\mathrm{JF},\mathrm{unc}}
-\sum_{x=1}^d\bar f_\epsilon(q_x)
\right\}^2
\right].
\]
Using independence,
\[
E\left[
\left\{
\sum_{x=1}^dT_x^{\mathrm{JF},\mathrm{unc}}
-\sum_{x=1}^d\bar f_\epsilon(q_x)
\right\}^2
\right]
\le
2\sum_{x=1}^d\operatorname{Var}(T_x^{\mathrm{JF},\mathrm{unc}})
+
2\left(
\sum_{x=1}^dET_x^{\mathrm{JF},\mathrm{unc}}
-\sum_{x=1}^d\bar f_\epsilon(q_x)
\right)^2 .
\]
Since \(\sum_x p_x=1\), Cauchy--Schwarz gives
\[
\left|
\sum_{x=1}^dET_x^{\mathrm{JF},\mathrm{unc}}
-\sum_{x=1}^d\bar f_\epsilon(q_x)
\right|^2
\le
2(C_{\epsilon}^{\mathrm{cell}})^2
\left\{
\frac{d}{mL_d}+\frac{d^2}{m^2L_d^2}
\right\}.
\]
Summing the variance bounds gives
\[
\sum_{x=1}^d\operatorname{Var}(T_x^{\mathrm{JF},\mathrm{unc}})
\le
C_{\epsilon}^{\mathrm{cell}}d^{1/16}
\left\{
\frac{L_d}{m}+\frac{dL_d^2}{m^2}
\right\}.
\]
The elementary logarithmic absorption bounds
\[
L_d^2d^{-15/16}\le225,
\qquad
L_d^4d^{-15/16}\le50625
\]
hold for every integer \(d\ge2\): the one-variable bound
\((1+\log t)^2t^{-3/16}\le225\) for \(t\ge1\) gives the first inequality after
multiplication by \(d^{-12/16}\le1\), and the second after squaring and
multiplication by \(d^{-9/16}\le1\).  With \(m=n/8\) and
\(\chi_{n,d}\le1\), these imply
\[
\frac{d}{mL_d}+\frac{d^2}{m^2L_d^2}
\le
72\chi_{n,d},
\]
and the scale inequality \(d/L_d\le n\) gives
\[
d^{1/16}
\left\{
\frac{L_d}{m}+\frac{dL_d^2}{m^2}
\right\}
\le
3300000\chi_{n,d}.
\]
Consequently,
\[
E\left[
\left\{\widetilde V^{\mathrm{unc}}-\Psi(\mathbb P)\right\}^2
\right]
\le
10000000
\{1+C_{\epsilon}^{\mathrm{cell}}+(C_{\epsilon}^{\mathrm{cell}})^2\}
\chi_{n,d}.
\]
% lean: sqRisk_jacksonUncapped_le_rate

\item We pass from the uncapped comparison experiment to the fixed-sample
estimator in \cref{obj:def:jackson-factorial-estimator}.  Let
\(\mathbb B_n\) be the product law of \(n\) independent fair marks.  For
\[
o_{1:n}\in([d]\times\{0,1\}^2)^n,
\qquad
\beta_{1:n}\in\{0,1\}^n,
\qquad
M\in\mathbb N,
\]
define
\[
\Upsilon_{n,d}^{\mathrm{cap}}(o_{1:n},\beta_{1:n},M)
:=
\begin{cases}
\Pi_{[0,1]}\!\left\{
\sum_{x=1}^d
T_x^{\mathrm{JF}}(o_{1:M},\beta_{1:M})
\right\},
& M\le n,\\[0.4em]
0, & M>n .
\end{cases}
\]
Define the capped Poisson-prefix average
\[
\overline\Upsilon_{n,d}^{\mathrm{cap}}(o_{1:n},\beta_{1:n})
:=
\int_{\mathbb N}
\Upsilon_{n,d}^{\mathrm{cap}}(o_{1:n},\beta_{1:n},M)\,
d\operatorname{Pois}_{n/4}(M).
\]
In the present Jackson branch, the all-data estimator is the fair-mark average
\[
\widehat V_{n,d}^{\mathrm{JF}}(o_{1:n})
=
\int
\overline\Upsilon_{n,d}^{\mathrm{cap}}(o_{1:n},\beta_{1:n})\,
d\mathbb B_n(\beta_{1:n}).
\]
Jensen's inequality for the mark average gives
\[
\operatorname{Risk}_{\mathbb P}
(\widehat V_{n,d}^{\mathrm{JF}},\Psi)
\le
E_{\mathbb P^{\otimes n}\otimes\mathbb B_n}
\left[
\left\{
\overline\Upsilon_{n,d}^{\mathrm{cap}}(O_{1:n},B_{1:n})
-\Psi(\mathbb P)
\right\}^2
\right].
\]
A second application of Jensen, now to the Poisson-prefix average, gives
\[
E_{\mathbb P^{\otimes n}\otimes\mathbb B_n}
\left[
\left\{
\overline\Upsilon_{n,d}^{\mathrm{cap}}(O_{1:n},B_{1:n})
-\Psi(\mathbb P)
\right\}^2
\right]
\le
E_{\mathbb P^{\otimes n}\otimes\mathbb B_n\otimes\operatorname{Pois}_{n/4}}
\left[
\left\{
\Upsilon_{n,d}^{\mathrm{cap}}(O_{1:n},B_{1:n},M)-\Psi(\mathbb P)
\right\}^2
\right].
\]
On \(\{M\le n\}\), the prefix
\((O_1,B_1),\ldots,(O_M,B_M)\) has the corresponding uncapped
finite-Poisson marked law restricted to prefixes of length at most \(n\), so
the nonoverflow contribution is bounded by
\[
E\left[
\left\{\widetilde V^{\mathrm{unc}}-\Psi(\mathbb P)\right\}^2
\right].
\]
On \(\{M>n\}\), the capped statistic equals \(0\), and
\(\Psi(\mathbb P)\in[0,1]\), so the overflow contribution is at most
\[
\Pr\{\operatorname{Pois}(n/4)>n\}.
\]
Thus
\[
\operatorname{Risk}_{\mathbb P}
(\widehat V_{n,d}^{\mathrm{JF}},\Psi)
\le
E\left[
\left\{\widetilde V^{\mathrm{unc}}-\Psi(\mathbb P)\right\}^2
\right]
+
\Pr\{\operatorname{Pois}(n/4)>n\}.
\]
To bound the Poisson overflow term, let
\[
\mathcal N_{\mathrm{ov}}\sim\operatorname{Pois}(n/4),
\qquad
\lambda_{\mathrm{ov}}:=\frac n4,
\qquad
\gamma_{\mathrm{ov}}:=\frac n8 .
\]
Since \(n\ge1\), both \(\lambda_{\mathrm{ov}}\) and \(\gamma_{\mathrm{ov}}\) are
positive.  For every nonnegative Chernoff tilt \(\varpi_{\mathrm{ov}}\), the
centered Poisson moment-generating function gives
\[
E\exp\{\varpi_{\mathrm{ov}}(\mathcal N_{\mathrm{ov}}-\lambda_{\mathrm{ov}})\}
=
\exp\{\lambda_{\mathrm{ov}}(e^{\varpi_{\mathrm{ov}}}-1-\varpi_{\mathrm{ov}})\}.
\]
Consequently, for every \(\Delta_{\mathrm{ov}}>0\), Markov's inequality yields
\[
\Pr\{\mathcal N_{\mathrm{ov}}-\lambda_{\mathrm{ov}}\ge \Delta_{\mathrm{ov}}\}
\le
\inf_{\varpi_{\mathrm{ov}}\ge0}
\exp\{-\varpi_{\mathrm{ov}}\Delta_{\mathrm{ov}}
+\lambda_{\mathrm{ov}}(e^{\varpi_{\mathrm{ov}}}-1-\varpi_{\mathrm{ov}})\}.
\]
Taking
\(
\varpi_{\mathrm{ov}}
=\log(1+\Delta_{\mathrm{ov}}/\lambda_{\mathrm{ov}})
\)
gives the Bennett form
\[
\Pr\{\mathcal N_{\mathrm{ov}}-\lambda_{\mathrm{ov}}\ge \Delta_{\mathrm{ov}}\}
\le
\exp\left[-\left\{(\lambda_{\mathrm{ov}}+\Delta_{\mathrm{ov}})
\log\left(1+\frac{\Delta_{\mathrm{ov}}}{\lambda_{\mathrm{ov}}}\right)
-\Delta_{\mathrm{ov}}\right\}\right].
\]
The elementary inequality
\[
\log(1+\rho_{\mathrm{ov}})\ge \frac{2\rho_{\mathrm{ov}}}{\rho_{\mathrm{ov}}+2},
\qquad \rho_{\mathrm{ov}}\ge0,
\]
implies
\[
(\lambda_{\mathrm{ov}}+\Delta_{\mathrm{ov}})
\log\left(1+\frac{\Delta_{\mathrm{ov}}}{\lambda_{\mathrm{ov}}}\right)
-\Delta_{\mathrm{ov}}
\ge
\frac{\Delta_{\mathrm{ov}}^2}{2\lambda_{\mathrm{ov}}+\Delta_{\mathrm{ov}}}.
\]
Now choose
\[
\Delta_{\mathrm{ov}}
:=
\sqrt{2\lambda_{\mathrm{ov}}\gamma_{\mathrm{ov}}}+2\gamma_{\mathrm{ov}}.
\]
Then
\[
\Delta_{\mathrm{ov}}^2
-\gamma_{\mathrm{ov}}(2\lambda_{\mathrm{ov}}+\Delta_{\mathrm{ov}})
=
3\gamma_{\mathrm{ov}}\sqrt{2\lambda_{\mathrm{ov}}\gamma_{\mathrm{ov}}}
+2\gamma_{\mathrm{ov}}^2
\ge0,
\]
so
\[
\Pr\{\mathcal N_{\mathrm{ov}}-\lambda_{\mathrm{ov}}>\Delta_{\mathrm{ov}}\}
\le e^{-\gamma_{\mathrm{ov}}}.
\]
For the present values,
\[
\Delta_{\mathrm{ov}}
=\sqrt{2(n/4)(n/8)}+2(n/8)
=\frac n2,
\qquad
n-\lambda_{\mathrm{ov}}=\frac{3n}{4},
\]
and therefore
\[
\{\mathcal N_{\mathrm{ov}}>n\}
\subseteq
\{\mathcal N_{\mathrm{ov}}-\lambda_{\mathrm{ov}}>\Delta_{\mathrm{ov}}\}.
\]
Thus
\[
\Pr\{\operatorname{Pois}(n/4)>n\}
\le e^{-n/8}.
\]
Finally, \(\gamma_{\mathrm{ov}}\le e^{\gamma_{\mathrm{ov}}}\) follows from
\(1+\gamma_{\mathrm{ov}}\le e^{\gamma_{\mathrm{ov}}}\), and hence
\[
\Pr\{\operatorname{Pois}(n/4)>n\}
\le e^{-n/8}
\le \frac{1}{n/8}
=
\frac8n .
\]
Using \(L_d\le d\),
\[
\frac8n\le 8\frac{d}{nL_d}=8\chi_{n,d}.
\]
Combining this overflow bound with the uncapped bound from the previous step,
\[
\operatorname{Risk}_{\mathbb P}
(\widehat V_{n,d}^{\mathrm{JF}},\Psi)
\le
\left[
10000000
\{1+C_{\epsilon}^{\mathrm{cell}}+(C_{\epsilon}^{\mathrm{cell}})^2\}
+8
\right]\chi_{n,d}
\le
C_\epsilon\chi_{n,d}.
\]
In the present branch \(\chi_{n,d}\le1\), hence
\[
C_\epsilon\chi_{n,d}
=
C_\epsilon
\min\left\{1,\frac{d}{n\log(ed)}\right\}.
\]
The three branches cover every \(n\ge1\), \(d\ge2\), and
\(\mathbb P\in\mathcal D_{d,\epsilon}^{\mathrm{obs}}\), giving the stated
uniform bound.

% lean: poisson_quarter_overflow_le_inv
\end{enumerate}
\end{proof}
