\begin{proof}[Proof of \cref{obj:lem:simultaneous-jackson-certificate}]
Take
\[
A=2^{60}.
\]
Fix \(\epsilon\in(0,1/2)\), and set
\[
C_\epsilon=32(1+\epsilon^{-1}).
\]
Let \(K\ge2\) and let \(Q_x\) satisfy the displayed hypotheses. Applying
\cref{obj:lem:jackson-tensor-extraction} to this rectangle gives the tensor Jackson polynomial
\(P_{x,K}\) and a normalized centered polynomial \(\mathcal R_{x,K}\). The same result gives, first,
\[
\deg_{\jmath}(P_{x,K})\le 2(K-1)
\qquad(\jmath\in\mathcal J).
\]
% lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.jacksonTensorPolynomialData_exists

Fix \(v\in Q_x\). Choose a bijection \(\eta:\mathcal J\to\{1,2,3,4\}\), and write
\[
c_\omega=b_{\eta^{-1}(\omega),x},
\qquad
\rho_\omega=r_{\eta^{-1}(\omega),x},
\qquad
y_\omega=v_{\eta^{-1}(\omega)}
\qquad(\omega=1,\ldots,4).
\]
Define the normalized point and angle vector by
\[
z_\omega=\frac{y_\omega-c_\omega}{\rho_\omega},
\qquad
\varphi_\omega=\arccos z_\omega .
\]
Since \(v\in Q_x\) and every radius is positive, \(z\in[-1,1]^4\) and
\[
v_{\jmath}=b_{\jmath,x}+r_{\jmath,x}\cos\varphi_{\eta(\jmath)}
\qquad(\jmath\in\mathcal J).
\]
On \([-1,1]^4\), set
\[
F_x(w)
=
\bar f_\epsilon\!\left(
\bigl\{b_{\jmath,x}+r_{\jmath,x}w_{\eta(\jmath)}\bigr\}_{\jmath\in\mathcal J}
\right).
\]
The lower-endpoint hypothesis gives \(b_{\jmath,x}-r_{\jmath,x}=\ell_{\jmath,x}\ge0\), so the affine image of
\([-1,1]^4\) lies in \(\mathbb R_+^4\). The global Lipschitz property in
\cref{obj:prop:identification-and-extension} therefore implies that \(F_x\) is continuous on
\([-1,1]^4\) and that, for all \(w,w'\in[-1,1]^4\),
\[
|F_x(w)-F_x(w')|
\le
(1+\epsilon^{-1})
\sum_{\omega=1}^4
\rho_\omega |w_\omega-w'_\omega|.
\]
For every shift \(\tau\in\mathbb R^4\), the elementary inequality
\[
|\cos(\varphi_\omega-\tau_\omega)-\cos\varphi_\omega|
\le
|\sin\varphi_\omega|\,|\tau_\omega|+\frac{\tau_\omega^2}{2}
\]
therefore gives
\[
\left|
F_x\{\cos(\varphi-\tau)\}-F_x(\cos\varphi)
\right|
\le
(1+\epsilon^{-1})
\sum_{\omega=1}^4
\left(
\rho_\omega|\sin\varphi_\omega|\,|\tau_\omega|
+\frac{\rho_\omega}{2}\tau_\omega^2
\right).
\]
Applying \cref{obj:lem:jackson-convolution-modulus} with
\[
L=1+\epsilon^{-1},
\qquad
a_\omega=\rho_\omega|\sin\varphi_\omega|,
\qquad
b_\omega=\frac{\rho_\omega}{2}
\]
yields
\[
|P_{x,K}(v)-\bar f_\epsilon(v)|
\le
(1+\epsilon^{-1})
\sum_{\omega=1}^4
\left(
\frac{32\rho_\omega|\sin\varphi_\omega|}{K}
+
\frac{32\rho_\omega}{K^2}
\right),
\]
where the evaluation identity for \(P_{x,K}\) is the Jackson-convolution identity supplied by
\cref{obj:lem:jackson-tensor-extraction}.
Finally, for each \(\jmath\in\mathcal J\),
\[
r_{\jmath,x}|\sin\varphi_{\eta(\jmath)}|
=
\sqrt{(v_{\jmath}-\ell_{\jmath,x})(u_{\jmath,x}-v_{\jmath})},
\]
because
\[
(v_{\jmath}-\ell_{\jmath,x})(u_{\jmath,x}-v_{\jmath})
=
r_{\jmath,x}^2
\left\{
1-\left(\frac{v_{\jmath}-b_{\jmath,x}}{r_{\jmath,x}}\right)^2
\right\}.
\]
Reindexing the preceding sum over \(\mathcal J\) and using
\(C_\epsilon=32(1+\epsilon^{-1})\) gives the asserted pointwise bound.
% lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.tensorConvolution_approx_weighted

It remains to record the centered expansion and its coefficient envelope. By
\cref{obj:lem:jackson-tensor-extraction}, for every four-coordinate vector \(v\),
\[
P_{x,K}(v)-\bar f_\epsilon(b_x)
=
\mathcal R_{x,K}\left(
\left\{\frac{v_{\jmath}-b_{\jmath,x}}{r_{\jmath,x}}\right\}_{\jmath\in\mathcal J}
\right).
\]
Let \(\mathcal A_x\) be the image of \(\operatorname{supp}(\mathcal R_{x,K})\) under the coordinate reindexing
from \(\{1,2,3,4\}\) to \(\mathcal J\), and let \(b_{x,\alpha}\) be the corresponding reindexed coefficient.
Then the preceding display becomes
\[
P_{x,K}(v)-\bar f_\epsilon(b_x)
=
\sum_{\alpha\in\mathcal A_x}
b_{x,\alpha}
\prod_{\jmath\in\mathcal J}
\left(\frac{v_{\jmath}-b_{\jmath,x}}{r_{\jmath,x}}\right)^{\alpha_{\jmath}}.
\]
The normalized coordinate-degree conclusion of \cref{obj:lem:jackson-tensor-extraction} gives
\[
\alpha_{\jmath}\le 2(K-1)
\qquad
(\alpha\in\mathcal A_x,\ \jmath\in\mathcal J),
\]
and reindexing preserves the coefficient one-norm, so its coefficient envelope gives
\[
\sum_{\alpha\in\mathcal A_x}|b_{x,\alpha}|
\le
2^{40K+20}(1+\epsilon^{-1})
\sum_{\jmath\in\mathcal J}r_{\jmath,x}.
\]
Since \(K\ge2\),
\[
40K+20\le60K,
\qquad
2^{40K+20}\le (2^{60})^K=A^K.
\]
Combining the last two displays proves the stated coefficient bound.
% lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.simultaneous_jackson_certificate
\end{proof}
