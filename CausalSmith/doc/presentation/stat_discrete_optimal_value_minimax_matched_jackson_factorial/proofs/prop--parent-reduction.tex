\begin{proof}[Proof of \cref{obj:prop:parent-reduction}]
The published moment-matching construction and absolute-value approximation rate of
\citep[Lemma 1 and Section 3.1]{CaiLow2011L1}, together with the published Poissonized two-sample
\(L_1\) lower-bound argument of \citep[Theorem 3 and equation (24)]{JiaoHanWeissman2018L1},
are the literature inputs used by \cref{obj:thm:matched-minimax-frontier}. Apply
\cref{obj:thm:matched-minimax-frontier} and choose the tuning constants it supplies. Write them as
\(H_0\), \(\kappa\), and \(D_0\). They satisfy
\[
H_0>0,\qquad 0<\kappa<1,\qquad D_0>2 .
\]
For each fixed \(\epsilon\in(0,1/2)\), \cref{obj:thm:matched-minimax-frontier} gives constants
\(c_\epsilon^{\mathrm{mf}}\) and \(C_\epsilon^{\mathrm{mf}}\), with
\[
0<c_\epsilon^{\mathrm{mf}}\le C_\epsilon^{\mathrm{mf}},
\]
such that, for every \(n\ge1\) and \(d\ge2\), with
\[
L_d=\log(ed),
\]
one has
\[
c_\epsilon^{\mathrm{mf}}
\min\left\{1,\frac{d}{nL_d}\right\}
\le
\mathfrak R_{n,d,\epsilon}
\le
C_\epsilon^{\mathrm{mf}}
\min\left\{1,\frac{d}{nL_d}\right\}.
\]

1. Fix \(n\ge1\) and \(d\ge2\). The logarithms \(L_d\) and \(\log(en)\) are positive. Since
\[
L_d
=
1+\log d
=
1+2\log\sqrt d
\le
1+2(\sqrt d-1)
\le
2\sqrt d,
\]
we have
\[
\frac{\sqrt d}{n}
\le
2\,\frac{d}{nL_d}.
\]
Define
\[
\tau_{n,d}=\frac{d}{nL_d}.
\]
We prove
\[
\min\left\{1,\frac{\sqrt d}{n}+\frac{d}{n\log(en)}\right\}
\le
4\min\left\{1,\tau_{n,d}\right\}.
\]
If \(\tau_{n,d}\ge1\), the right-hand side is \(4\), while the left-hand side is at most \(1\).
Suppose \(\tau_{n,d}<1\). If \(d\le n\), then \(ed\le en\), so
\[
L_d\le \log(en)\le 2\log(en).
\]
If \(d>n\), put
\[
\eta_{n,d}=\frac dn .
\]
Then \(\eta_{n,d}>1\),
\[
L_d=\log(en)+\log\eta_{n,d},
\]
and \(\tau_{n,d}<1\) gives \(\eta_{n,d}<L_d\). For every \(\eta>1\),
\[
\log\eta
=
2\log\sqrt\eta
\le
2(\sqrt\eta-1)
\le
\frac{\eta}{2},
\]
where the last inequality is equivalent to \((\sqrt\eta-2)^2\ge0\). Hence
\[
\eta_{n,d}
<
\log(en)+\log\eta_{n,d}
\le
\log(en)+\frac{\eta_{n,d}}2,
\]
so \(\eta_{n,d}\le2\log(en)\), and therefore
\[
L_d
=
\log(en)+\log\eta_{n,d}
\le
\log(en)+\frac{\eta_{n,d}}2
\le
2\log(en).
\]
Thus, throughout the case \(\tau_{n,d}<1\),
\[
\frac{d}{n\log(en)}
\le
2\,\frac{d}{nL_d}.
\]
Combining this with the square-root bound gives
\[
\frac{\sqrt d}{n}+\frac{d}{n\log(en)}
\le
4\,\frac{d}{nL_d}
=
4\min\left\{1,\tau_{n,d}\right\},
\]
and hence the displayed comparison follows. Consequently,
\[
\frac{c_\epsilon^{\mathrm{mf}}}{4}
\min\left\{1,\frac{\sqrt d}{n}+\frac{d}{n\log(en)}\right\}
\le
c_\epsilon^{\mathrm{mf}}
\min\left\{1,\frac{d}{nL_d}\right\}
\le
\mathfrak R_{n,d,\epsilon}.
\]
For the upper bound, \(d\ge2\) gives \(L_d\ge1\), and therefore
\[
\mathfrak R_{n,d,\epsilon}
\le
C_\epsilon^{\mathrm{mf}}
\min\left\{1,\frac{d}{nL_d}\right\}
\le
C_\epsilon^{\mathrm{mf}}\frac dn .
\]
Taking \(c=c_\epsilon^{\mathrm{mf}}/4\) and \(C=C_\epsilon^{\mathrm{mf}}\) proves the matched-frontier assertion in the proposition. % lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.parent_reduction

2. Fix \(n\), \(d\), \(\epsilon\), and an observed sample with \(2\le d<D_0\). \Cref{obj:def:jackson-factorial-estimator} selects its empirical-ratio branch whenever \(d<D_0\). Hence, for that sample,
\[
\widehat V_{n,d}^{\mathrm{JF}}
=
\widehat V_{n,d}^{\mathrm{ER}} .
\]
% lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.parent_reduction

3. Fix \(\epsilon\in(0,1/2)\) and a sequence \((d_n)\) such that \(d_n\ge2\) for every \(n\) and
\(d_n\le d_{\max}\) for some finite \(d_{\max}\). Then \(d_{\max}\ge2\). Define
\[
c_{\epsilon,d_{\max}}
=
c_\epsilon^{\mathrm{mf}}\,\frac{2}{\log(ed_{\max})},
\qquad
C_{\epsilon,d_{\max}}
=
C_\epsilon^{\mathrm{mf}}\,d_{\max}.
\]
Since \(\log(ed_{\max})\ge1\) and \(d_{\max}\ge2\),
\[
\frac{2}{\log(ed_{\max})}\le d_{\max},
\]
so
\[
0<c_{\epsilon,d_{\max}}\le C_{\epsilon,d_{\max}}.
\]
For every \(n\ge d_{\max}\), one has \(d_n\le n\). With
\[
L_{d_n}=\log(ed_n),
\]
the bound \(d_n\ge2\) gives \(L_{d_n}\ge1\), and therefore
\[
\frac{d_n}{nL_{d_n}}\le1.
\]
Thus the minimum in the matched frontier equals \(d_n/(nL_{d_n})\). Monotonicity of the logarithm gives
\[
L_{d_n}\le \log(ed_{\max}),
\]
and, since \(d_n\ge2\),
\[
\frac{2}{\log(ed_{\max})}
\le
\frac{d_n}{L_{d_n}}.
\]
Therefore, for every \(n\ge d_{\max}\),
\[
\frac{c_{\epsilon,d_{\max}}}{n}
\le
c_\epsilon^{\mathrm{mf}}\frac{d_n}{nL_{d_n}}
\le
\mathfrak R_{n,d_n,\epsilon}.
\]
Similarly,
\[
\mathfrak R_{n,d_n,\epsilon}
\le
C_\epsilon^{\mathrm{mf}}\frac{d_n}{nL_{d_n}}
\le
C_\epsilon^{\mathrm{mf}}\frac{d_n}{n}
\le
\frac{C_{\epsilon,d_{\max}}}{n}.
\]
These inequalities hold for every \(n\ge d_{\max}\), hence for all sufficiently large \(n\). % lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.parent_reduction
\end{proof}
