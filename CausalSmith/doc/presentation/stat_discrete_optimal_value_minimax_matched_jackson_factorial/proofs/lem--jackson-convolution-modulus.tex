\begin{proof}[Proof of \cref{obj:lem:jackson-convolution-modulus}]
\begin{enumerate}
\item Let
\[
\mathfrak B_D=[-\pi,\pi]^D
\qquad\text{and}\qquad
\mathfrak k_K^{(D)}(\tau)=\prod_{i=1}^D J_K(\tau_i),
\quad \tau\in\mathfrak B_D .
\]
For the positive integer \(K\) in the statement, \(J_K\) is the normalized order-four Jackson factor on \([-\pi,\pi]\), with removable values filled continuously. The raw fourth-power factor is nonnegative and has positive integral, so its normalized version satisfies \(J_K(\rho)\ge0\) for every \(\rho\in[-\pi,\pi]\). The tensor product kernel is normalized to have unit mass:
\[
\mathfrak k_K^{(D)}(\tau)\ge0,
\qquad
\int_{\mathfrak B_D}\mathfrak k_K^{(D)}(\tau)\,d\tau
=
\prod_{i=1}^D\int_{-\pi}^{\pi}J_K(\rho)\,d\rho
=1.
\]
The convolution is written in the shifted-variable form
\[
(\mathcal J_K f)(\cos\varphi)
=
\int_{\mathfrak B_D}
f\{\cos(\varphi-\tau)\}\mathfrak k_K^{(D)}(\tau)\,d\tau .
\]
When \(D=0\), the product is the empty product and the sums below are empty sums, so these identities keep their displayed values. Continuity of \(f\) on the normalized cube, coordinatewise continuity of cosine, and compactness of \(\mathfrak B_D\) give the integrability used below. % lean: Causalean.Mathlib.Analysis.JacksonApproximation.tensorJackson_integral_eq_one

\item The preceding unit-mass identity gives
\[
\begin{aligned}
(\mathcal J_K f)(\cos\varphi)-f(\cos\varphi)
&=
\int_{\mathfrak B_D}
f\{\cos(\varphi-\tau)\}\mathfrak k_K^{(D)}(\tau)\,d\tau
-
f(\cos\varphi)
\int_{\mathfrak B_D}\mathfrak k_K^{(D)}(\tau)\,d\tau  \\
&=
\int_{\mathfrak B_D}
\bigl[
f\{\cos(\varphi-\tau)\}-f(\cos\varphi)
\bigr]\mathfrak k_K^{(D)}(\tau)\,d\tau .
\end{aligned}
\]
The subtraction is justified by the compact-domain integrability noted above. % lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.tensorConvolution_approx_weighted

\item Taking absolute values, using nonnegativity of \(\mathfrak k_K^{(D)}\), and applying the assumed weighted increment modulus pointwise yields
\[
\begin{aligned}
\left|(\mathcal J_K f)(\cos\varphi)-f(\cos\varphi)\right|
&\le
\int_{\mathfrak B_D}
\left|f\{\cos(\varphi-\tau)\}-f(\cos\varphi)\right|
\mathfrak k_K^{(D)}(\tau)\,d\tau  \\
&\le
\int_{\mathfrak B_D}
L\sum_{i=1}^D\left(a_i|\tau_i|+b_i\tau_i^2\right)
\mathfrak k_K^{(D)}(\tau)\,d\tau .
\end{aligned}
\] % lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.tensorConvolution_approx_weighted

\item Distributing the finite sum through the integral gives
\[
\begin{aligned}
&\int_{\mathfrak B_D}
L\sum_{i=1}^D\left(a_i|\tau_i|+b_i\tau_i^2\right)
\mathfrak k_K^{(D)}(\tau)\,d\tau  \\
&\qquad =
L\sum_{i=1}^D
\left[
a_i\int_{\mathfrak B_D}|\tau_i|\mathfrak k_K^{(D)}(\tau)\,d\tau
+
b_i\int_{\mathfrak B_D}\tau_i^2\mathfrak k_K^{(D)}(\tau)\,d\tau
\right].
\end{aligned}
\]
For each selected coordinate \(i\), product integration and the one-dimensional unit mass give
\[
\int_{\mathfrak B_D}|\tau_i|\mathfrak k_K^{(D)}(\tau)\,d\tau
=
\int_{-\pi}^{\pi}|\rho|J_K(\rho)\,d\rho
\]
and
\[
\int_{\mathfrak B_D}\tau_i^2\mathfrak k_K^{(D)}(\tau)\,d\tau
=
\int_{-\pi}^{\pi}\rho^2J_K(\rho)\,d\rho .
\] % lean: Causalean.Mathlib.Analysis.JacksonApproximation.tensorJackson_first_moment_eq; Causalean.Mathlib.Analysis.JacksonApproximation.tensorJackson_second_moment_eq

\item Write the raw kernel and its mass as
\[
\widetilde J_K(\rho)=
\begin{cases}
K^4, & \sin(\rho/2)=0,\\[2mm]
\left\{\dfrac{\sin(K\rho/2)}{\sin(\rho/2)}\right\}^4,
& \sin(\rho/2)\ne0,
\end{cases}
\qquad
\mathfrak m_K=\int_{-\pi}^{\pi}\widetilde J_K(\rho)\,d\rho,
\]
so that \(J_K(\rho)=\widetilde J_K(\rho)/\mathfrak m_K\). Let \(U_m\) denote the Chebyshev polynomial of the second kind. The removable-value convention and the identity
\[
U_{K-1}\{\cos(\rho/2)\}
=
\frac{\sin(K\rho/2)}{\sin(\rho/2)}
\]
away from the removable points give
\[
\widetilde J_K(\rho)
=
U_{K-1}\{\cos(\rho/2)\}^4 .
\]
The square expansion
\[
U_{K-1}\{\cos(\rho/2)\}^2
=
K+2\sum_{q\in\{0,\ldots,K-2\}}(K-1-q)\cos\{(q+1)\rho\}
\]
follows from the Chebyshev recurrence, with the displayed sum interpreted as an empty sum when \(K=1\). Using
\[
\int_{-\pi}^{\pi}\cos(r\rho)\,d\rho=0,
\qquad
\int_{-\pi}^{\pi}\cos(r\rho)\cos(s\rho)\,d\rho
=
\begin{cases}
\pi,& r=s,\\
0,& r\ne s,
\end{cases}
\]
for positive integers \(r,s\), one obtains
\[
\begin{aligned}
\mathfrak m_K
&=
\int_{-\pi}^{\pi}
\left[
K+2\sum_{q\in\{0,\ldots,K-2\}}(K-1-q)\cos\{(q+1)\rho\}
\right]^2\,d\rho  \\
&=
2\pi K^2+4\pi\sum_{q\in\{0,\ldots,K-2\}}(K-1-q)^2  \\
&=
2\pi K^2+\frac{2\pi}{3}(K-1)K(2K-1)
=
\frac{2\pi}{3}K(2K^2+1).
\end{aligned}
\]
In particular,
\[
\mathfrak m_K\ge \frac{4\pi}{3}K^3>0 .
\] % lean: Causalean.Mathlib.Analysis.JacksonApproximation.jrawMass_eq

\item The same Chebyshev-square expansion and the vanishing of all positive-frequency cosine integrals give
\[
\int_{-\pi}^{\pi}
U_{K-1}\{\cos(\rho/2)\}^2\,d\rho
=
\int_{-\pi}^{\pi}
\left[
K+2\sum_{q\in\{0,\ldots,K-2\}}(K-1-q)\cos\{(q+1)\rho\}
\right]d\rho
=
2\pi K .
\]
For \(\rho\in[-\pi,\pi]\),
\[
\rho^2\le \pi^2\sin^2(\rho/2).
\]
If \(\sin(\rho/2)=0\), then \(\rho=0\) on this interval and
\[
\rho^2\widetilde J_K(\rho)
\le
\pi^2U_{K-1}\{\cos(\rho/2)\}^2 .
\]
If \(\sin(\rho/2)\ne0\), put
\[
U_{K-1}\{\cos(\rho/2)\}=\frac{s_K(\rho)}{d(\rho)},
\qquad
s_K(\rho)=\sin(K\rho/2),
\qquad
d(\rho)=\sin(\rho/2).
\]
Since \(s_K(\rho)^2\le1\),
\[
\begin{aligned}
\rho^2\widetilde J_K(\rho)
&=
\rho^2\left\{\frac{s_K(\rho)}{d(\rho)}\right\}^4  \\
&=
\rho^2\left\{\frac{s_K(\rho)}{d(\rho)}\right\}^2
\left\{\frac{s_K(\rho)}{d(\rho)}\right\}^2  \\
&\le
\pi^2s_K(\rho)^2
\left\{\frac{s_K(\rho)}{d(\rho)}\right\}^2
\le
\pi^2
\left\{\frac{s_K(\rho)}{d(\rho)}\right\}^2
=
\pi^2U_{K-1}\{\cos(\rho/2)\}^2 .
\end{aligned}
\]
After integration and normalization,
\[
\begin{aligned}
\int_{-\pi}^{\pi}\rho^2J_K(\rho)\,d\rho
&=
\frac{1}{\mathfrak m_K}
\int_{-\pi}^{\pi}\rho^2\widetilde J_K(\rho)\,d\rho  \\
&\le
\frac{\pi^2}{\mathfrak m_K}
\int_{-\pi}^{\pi}U_{K-1}\{\cos(\rho/2)\}^2\,d\rho  \\
&\le
\frac{2\pi^3K}{(4\pi/3)K^3}
=
\frac{3\pi^2}{2K^2}
\le
\frac{24}{K^2}
\le
\frac{64}{K^2}.
\end{aligned}
\] % lean: Causalean.Mathlib.Analysis.JacksonApproximation.jackson_second_moment

\item The elementary inequality
\[
0\le (K|\rho|-2)^2
\]
is equivalent, since \(K>0\), to
\[
|\rho|\le \frac K4\rho^2+\frac1K .
\]
Multiplying by \(J_K(\rho)\ge0\), integrating, and using the preceding second-moment estimate together with \(\int_{-\pi}^{\pi}J_K(\rho)\,d\rho=1\), gives
\[
\begin{aligned}
\int_{-\pi}^{\pi}|\rho|J_K(\rho)\,d\rho
&\le
\frac K4\int_{-\pi}^{\pi}\rho^2J_K(\rho)\,d\rho
+
\frac1K\int_{-\pi}^{\pi}J_K(\rho)\,d\rho  \\
&\le
\frac K4\frac{24}{K^2}+\frac1K
=
\frac7K
\le
\frac{32}{K}.
\end{aligned}
\] % lean: Causalean.Mathlib.Analysis.JacksonApproximation.jackson_first_moment

\item Combining the integral bound from the previous steps with the tensor moment reductions gives
\[
\begin{aligned}
\left|(\mathcal J_K f)(\cos\varphi)-f(\cos\varphi)\right|
&\le
L\sum_{i=1}^D
\left[
a_i\frac{32}{K}
+
b_i\frac{64}{K^2}
\right]  \\
&=
L\sum_{i=1}^D
\left(\frac{32a_i}{K}+\frac{64b_i}{K^2}\right).
\end{aligned}
\]
Here \(L\ge0\), \(a_i\ge0\), and \(b_i\ge0\) justify multiplying the one-dimensional moment inequalities and summing them coordinatewise. This is the asserted bound. % lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.tensorConvolution_approx_weighted
\end{enumerate}
\end{proof}
