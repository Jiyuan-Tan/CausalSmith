\begin{proof}[Proof of \cref{obj:lem:jackson-normalized-coefficient-envelope}]
\begin{enumerate}
\item The ordered endpoints make \(Q_x\) a valid rectangle, and the remaining hypotheses give the lower-endpoint, positive-overlap, degree, and positive-radius conditions needed for \cref{obj:lem:jackson-tensor-extraction}. Apply that result to \(Q_x\). Under the same positive-overlap and positive-radius conditions, the construction in \cref{obj:def:jackson-kernel} identifies the extracted physical polynomial with \(P_{x,K}\).

Fix a bijection
\[
\iota_{\mathcal J}:\mathcal J\longrightarrow \{1,2,3,4\}.
\]
Write the normalized polynomial supplied by \cref{obj:lem:jackson-tensor-extraction} as
\[
\mathcal Q_{x,K}(\zeta)
=
\sum_{\beta\in\mathcal B_x}
\eta_\beta
\prod_{\varpi=1}^4 \zeta_\varpi^{\beta_\varpi},
\qquad
\mathcal B_x=\operatorname{supp}(\mathcal Q_{x,K})
\subset \mathbb N^{\{1,2,3,4\}},
\qquad
\eta_\beta=(\mathcal Q_{x,K})_\beta
\quad(\beta\in\mathcal B_x).
\]
Define the coordinate transport
\[
\Phi:\mathbb N^{\{1,2,3,4\}}\longrightarrow \mathbb N^{\mathcal J},
\qquad
(\Phi\beta)_\jmath=\beta_{\iota_{\mathcal J}(\jmath)},
\]
and set
\[
\mathcal S_x=\Phi(\mathcal B_x),
\qquad
\xi_{x,\alpha}=\eta_{\Phi^{-1}(\alpha)}
\quad\text{for }\alpha\in\mathcal S_x.
\]
Then, for every \(v\in\mathbb R^{\mathcal J}\),
\[
P_{x,K}(v)-\bar f_\epsilon(b_x)
=
\sum_{\alpha\in\mathcal S_x}
\xi_{x,\alpha}
\prod_{\jmath\in\mathcal J}
\left(\frac{v_\jmath-b_{\jmath,x}}{r_{\jmath,x}}\right)^{\alpha_\jmath}.
\]
The coefficient sum is unchanged by this relabeling:
\[
\sum_{\alpha\in\mathcal S_x}|\xi_{x,\alpha}|
=
\sum_{\beta\in\mathcal B_x}|\eta_\beta|.
\]
% lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.jacksonTensorPolynomialData_exists

\item\label{proof:jackson-normalized-coefficient-envelope:normalized-expansion} Substitute \(v_\jmath=b_{\jmath,x}+r_{\jmath,x}z_\jmath\) in the preceding display. Since every \(r_{\jmath,x}>0\),
\[
\frac{b_{\jmath,x}+r_{\jmath,x}z_\jmath-b_{\jmath,x}}{r_{\jmath,x}}
=
z_\jmath .
\]
Thus, as a polynomial in \(z\in\mathbb R^{\mathcal J}\),
\[
\mathcal R_{x,K}(z)
=
P_{x,K}(b_x+r_x\odot z)-\bar f_\epsilon(b_x)
=
\sum_{\alpha\in\mathcal S_x}
\xi_{x,\alpha}
\prod_{\jmath\in\mathcal J}z_\jmath^{\alpha_\jmath}.
\]
% lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.centeredNormalizedPolynomial_eq_of_expansion

\item The last identity implies that every coefficient of \(\mathcal R_{x,K}\) outside \(\mathcal S_x\) is zero, and that for \(\alpha\in\mathcal A_x=\operatorname{supp}(\mathcal R_{x,K})\) the coefficient \(b_{x,\alpha}\) equals \(\xi_{x,\alpha}\). Hence
\[
\sum_{\alpha\in\mathcal A_x}|b_{x,\alpha}|
\le
\sum_{\alpha\in\mathcal S_x}|\xi_{x,\alpha}|.
\]
% lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.centeredNormalizedPolynomial_coeffL1_le_of_expansion

\item By the coefficient-envelope part of \cref{obj:lem:jackson-tensor-extraction},
\[
\sum_{\beta\in\mathcal B_x}|\eta_\beta|
\le
2^{40K+20}(1+\epsilon^{-1})
\sum_{\jmath\in\mathcal J}r_{\jmath,x}.
\]
Since \(K\ge2\),
\[
40K+20\le 60K,
\qquad
2^{40K+20}\le 2^{60K}=(2^{60})^K.
\]
Also \((1+\epsilon^{-1})\sum_{\jmath\in\mathcal J}r_{\jmath,x}\ge0\), because \(\epsilon>0\) and every radius is positive. Combining these inequalities with the relabeling identity gives
\[
\sum_{\alpha\in\mathcal A_x}|b_{x,\alpha}|
\le
(2^{60})^{K}(1+\epsilon^{-1})
\sum_{\jmath\in\mathcal J}r_{\jmath,x}.
\]
% lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.jacksonCenteredNormalizedPolynomial_coeffL1_le

\item It remains to record the coordinate-degree bound. Let \(\alpha\in\mathcal A_x\). If \(\alpha\notin\mathcal S_x\), the polynomial identity in \cref{proof:jackson-normalized-coefficient-envelope:normalized-expansion} gives \(b_{x,\alpha}=0\), contradicting \(\alpha\in\operatorname{supp}(\mathcal R_{x,K})\). Hence \(\alpha=\Phi\beta\) for some \(\beta\in\mathcal B_x\). The normalized-degree part of \cref{obj:lem:jackson-tensor-extraction} gives
\[
\beta_\varpi\le 2(K-1)
\qquad\text{for every }\varpi\in\{1,2,3,4\}.
\]
Therefore, for each \(\jmath\in\mathcal J\),
\[
\alpha_\jmath
=
(\Phi\beta)_\jmath
=
\beta_{\iota_{\mathcal J}(\jmath)}
\le
2(K-1).
\]
% lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.jacksonCenteredNormalizedPolynomial_coeffL1_le
\end{enumerate}
\end{proof}
