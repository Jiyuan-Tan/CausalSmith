\begin{proof}[Proof of \cref{obj:lem:good-pilot-radius-sum}]
Fix \(x\) and set
\[
\zeta=\frac{L_d}{m}.
\]
Since \(d\ge 1\), \(ed\ge e>1\), so \(L_d=\log(ed)>0\). Hence \(\zeta>0\). For each \(\jmath\in\mathcal J\), write
\[
\sigma_{\jmath,x}=\sqrt{c_{\jmath,x}\zeta}.
\]
For the canonical tuning used in this lemma, the pilot half-width is calibrated with \(H_0=1024\). Hence
\[
h_{\jmath,x}=1024(\sigma_{\jmath,x}+\zeta).
\]
Also \(\sigma_{\jmath,x}\ge0\), and
\[
\sigma_{\jmath,x}\le \frac{c_{\jmath,x}}{512}+128\zeta
\]
because both sides are nonnegative and
\[
\left(\frac{c_{\jmath,x}}{512}+128\zeta\right)^2-c_{\jmath,x}\zeta
=
\left(\frac{c_{\jmath,x}}{512}-128\zeta\right)^2\ge0 .
\]
The good-pilot event gives
\[
c_{\jmath,x}\le q_\jmath+\frac{h_{\jmath,x}}4 .
\]
Substituting the displayed formula for \(h_{\jmath,x}\) and then using the preceding bound on
\(\sigma_{\jmath,x}\),
\[
c_{\jmath,x}
\le q_\jmath+256\sigma_{\jmath,x}+256\zeta
\le q_\jmath+\frac{c_{\jmath,x}}2+33024\zeta .
\]
Thus
\begin{equation}\label{eq:good-pilot-radius-center-bound}
c_{\jmath,x}\le 2q_\jmath+66048\zeta .
\end{equation}
% lean: canonicalPilotGoodRadiusSum_le

Define
\[
\rho_{\jmath,x}=\sqrt{q_\jmath\zeta}.
\]
Since \(q_\jmath\ge0\) and \(\zeta>0\), we have \(\rho_{\jmath,x}^2=q_\jmath\zeta\). From \cref{eq:good-pilot-radius-center-bound},
\[
\sigma_{\jmath,x}^2
=
c_{\jmath,x}\zeta
\le
(2q_\jmath+66048\zeta)\zeta .
\]
Moreover,
\[
(2\rho_{\jmath,x}+300\zeta)^2-(2q_\jmath+66048\zeta)\zeta
=
2q_\jmath\zeta+1200\rho_{\jmath,x}\zeta+23952\zeta^2
\ge0 .
\]
Taking square roots gives
\begin{equation}\label{eq:good-pilot-radius-sigma-bound}
\sigma_{\jmath,x}\le 2\sqrt{q_\jmath\zeta}+300\zeta .
\end{equation}
% lean: canonicalPilotGoodRadiusSum_le

For the zero-truncated pilot interval,
\[
r_{\jmath,x}
=
\frac{c_{\jmath,x}+h_{\jmath,x}-\max\{0,c_{\jmath,x}-h_{\jmath,x}\}}2 .
\]
If \(c_{\jmath,x}\le h_{\jmath,x}\), then \(r_{\jmath,x}=(c_{\jmath,x}+h_{\jmath,x})/2\le h_{\jmath,x}\). If
\(c_{\jmath,x}>h_{\jmath,x}\), then \(r_{\jmath,x}=h_{\jmath,x}\). Therefore
\[
r_{\jmath,x}\le h_{\jmath,x}.
\]
Combining this with the formula for \(h_{\jmath,x}\) and \cref{eq:good-pilot-radius-sigma-bound},
\begin{equation}\label{eq:good-pilot-radius-coordinate-bound}
r_{\jmath,x}
\le
1024(\sigma_{\jmath,x}+\zeta)
\le
2048\sqrt{q_\jmath\zeta}+308224\zeta .
\end{equation}
% lean: pilotRectangle_radius_bounds

Summing \cref{eq:good-pilot-radius-coordinate-bound} over the four coordinates,
\[
\sum_{\jmath\in\mathcal J}r_{\jmath,x}
\le
2048\sum_{\jmath\in\mathcal J}\sqrt{q_\jmath\zeta}
+
4\cdot 308224\,\zeta .
\]
By Cauchy's inequality on the four-point set \(\mathcal J\),
\[
\left(\sum_{\jmath\in\mathcal J}\sqrt{q_\jmath\zeta}\right)^2
\le
4\sum_{\jmath\in\mathcal J}\bigl(\sqrt{q_\jmath\zeta}\bigr)^2
=
4\left(\sum_{\jmath\in\mathcal J}q_\jmath\right)\zeta .
\]
All terms are nonnegative, so
\[
\sum_{\jmath\in\mathcal J}\sqrt{q_\jmath\zeta}
\le
2\sqrt{\left(\sum_{\jmath\in\mathcal J}q_\jmath\right)\zeta}.
\]
Consequently,
\[
\sum_{\jmath\in\mathcal J}r_{\jmath,x}
\le
4096\sqrt{\left(\sum_{\jmath\in\mathcal J}q_\jmath\right)\zeta}
+
1232896\,\zeta
\le
2\cdot 10^6
\left\{
\sqrt{\left(\sum_{\jmath\in\mathcal J}q_\jmath\right)\zeta}
+\zeta
\right\}.
\]
Since \(p_x=\sum_{\jmath\in\mathcal J}q_\jmath\) and \(\zeta=L_d/m\), the desired bound follows:
\[
\sum_{\jmath\in\mathcal J} r_{\jmath,x}
\le
2\cdot 10^6
\left\{
\sqrt{\frac{p_xL_d}{m}}+\frac{L_d}{m}
\right\}.
\]
% lean: canonicalPilotGoodRadiusSum_le
\end{proof}
