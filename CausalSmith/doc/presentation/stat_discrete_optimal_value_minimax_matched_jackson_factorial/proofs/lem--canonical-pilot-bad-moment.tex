\begin{proof}[Proof of \cref{obj:lem:canonical-pilot-bad-moment}]
1.  For each \(\jmath\in\mathcal J\), define
\[
\eta_\jmath=mq_\jmath,\qquad
\delta_\jmath=\left|N'_\jmath-\eta_\jmath\right|,
\qquad
\rho_\jmath=\sqrt{N'_\jmath L}+L,
\qquad
\sigma_\jmath=\delta_\jmath+1024\rho_\jmath,
\]
and put
\[
\Sigma=\sum_{\jmath\in\mathcal J}\sigma_\jmath .
\]
Since \(m>0\),
\[
\left|{N'_\jmath\over m}-q_\jmath\right|
={\delta_\jmath\over m},
\qquad
\sqrt{{N'_\jmath\over m}{L\over m}}+{L\over m}
={\rho_\jmath\over m}.
\]
Thus
\[
\Gamma(N')={\Sigma\over m},
\qquad
\mathsf G^c
=
\bigcup_{\jmath\in\mathcal J}
\{\delta_\jmath>256\rho_\jmath\},
\]
and hence
\[
E\!\left[\Gamma(N')^t\mathbf 1_{\mathsf G^c}\right]
=
m^{-t}
E\!\left[
\Sigma^t
\mathbf 1_{\cup_{\jmath\in\mathcal J}\{\delta_\jmath>256\rho_\jmath\}}
\right].
\]
% lean: Causalean.Stat.Concentration.PoissonSelfNormalized.independent_poisson_normalized_badAny_moment

2.  We first record the scalar estimate used in the product step.  Let
\(\mathsf W\sim\operatorname{Pois}(\Lambda)\), where \(\Lambda\ge0\), and define
\[
\delta_{\Lambda}(\mathsf W)=|\mathsf W-\Lambda|,
\qquad
\rho_L(\mathsf W)=\sqrt{\mathsf W L}+L,
\qquad
\sigma_{\Lambda,L}(\mathsf W)
=
\delta_{\Lambda}(\mathsf W)+1024\rho_L(\mathsf W),
\]
\[
\upsilon_{\Lambda,L}=\sqrt{\Lambda L}+L,
\qquad
\mathcal B_{\Lambda,L}
=
\{\delta_{\Lambda}(\mathsf W)>256\rho_L(\mathsf W)\}.
\]
For every real \(\xi\),
\[
E\exp\{\xi(\mathsf W-\Lambda)\}
=
\exp\{\Lambda(e^\xi-1-\xi)\}.
\]
The Chernoff--Bennett bounds obtained from this identity give, for every
\(\zeta_{\mathrm{tail}}\ge0\),
\[
\Pr\{\mathsf W-\Lambda>\sqrt{2\Lambda\zeta_{\mathrm{tail}}}
       +2\zeta_{\mathrm{tail}}\}\le e^{-\zeta_{\mathrm{tail}}},
\qquad
\Pr\{\Lambda-\mathsf W>\sqrt{2\Lambda\zeta_{\mathrm{tail}}}\}
\le e^{-\zeta_{\mathrm{tail}}}.
\]
With \(\zeta_{\mathrm{tail}}=80L\),
\[
\mathcal B_{\Lambda,L}
\subseteq
\{\mathsf W-\Lambda>\sqrt{160\Lambda L}+160L\}
\cup
\{\Lambda-\mathsf W>\sqrt{160\Lambda L}\}.
\]
Indeed, on \(\{\Lambda\le\mathsf W\}\),
\[
\sqrt{160\Lambda L}\le16\sqrt{\mathsf W L},
\]
so the bad-event inequality implies the displayed upper-tail event.  On
\(\{\mathsf W<\Lambda\}\), the inequality
\[
\Lambda-\mathsf W>256(\sqrt{\mathsf W L}+L)
\]
implies
\[
(\Lambda-\mathsf W)^2-160L(\Lambda-\mathsf W)-160\mathsf W L>0,
\]
and therefore \((\Lambda-\mathsf W)^2>160\Lambda L\), which is the displayed
lower-tail event.  Consequently
\[
\Pr(\mathcal B_{\Lambda,L})\le 2e^{-80L}.
\]

Next,
\[
\sqrt{\mathsf W L}
\le
\sqrt{\Lambda L}+\sqrt{\delta_{\Lambda}(\mathsf W)L}
\le
\sqrt{\Lambda L}+{\delta_{\Lambda}(\mathsf W)+L\over2},
\]
and hence
\[
\sigma_{\Lambda,L}(\mathsf W)
\le
1536\{\upsilon_{\Lambda,L}+\delta_{\Lambda}(\mathsf W)\}.
\]
Since \(L\ge1\), \(\upsilon_{\Lambda,L}\ge1\).  Set
\[
\xi_{\Lambda,L}={1\over 2\upsilon_{\Lambda,L}}.
\]
Then \(|\xi_{\Lambda,L}|\le1\) and
\(\Lambda\xi_{\Lambda,L}^2\le1/4\).  The preceding deterministic bound gives
\[
\exp\!\left\{
{\sigma_{\Lambda,L}(\mathsf W)\over4096\upsilon_{\Lambda,L}}
\right\}
\le
e^{1/2}
\left[
e^{\xi_{\Lambda,L}(\mathsf W-\Lambda)}
+
e^{-\xi_{\Lambda,L}(\mathsf W-\Lambda)}
\right].
\]
Using the moment-generating identity and
\(|e^r-1-r|\le r^2\) for \(|r|\le1\),
\[
E\exp\!\left\{
{\sigma_{\Lambda,L}(\mathsf W)\over4096\upsilon_{\Lambda,L}}
\right\}
\le
e^{1/2}(e^{1/4}+e^{1/4})
\le 8.
\]
The elementary exponential domination of powers then yields, for every integer
\(s\) with \(0\le s\le8\),
\[
E\sigma_{\Lambda,L}(\mathsf W)^s
\le
10^{4s+4}\upsilon_{\Lambda,L}^{\,s}.
\]
For \(t\in\{1,2,4\}\), define
\[
\zeta_{\Lambda,L,t}
=
e^{-40L}\upsilon_{\Lambda,L}^{-t}.
\]
With
\[
X_{\Lambda,L,t}(\mathsf W)=\sigma_{\Lambda,L}(\mathsf W)^t,
\]
the pointwise Young inequality gives
\[
\mathbf 1_{\mathcal B_{\Lambda,L}}X_{\Lambda,L,t}
\le
{ \zeta_{\Lambda,L,t}X_{\Lambda,L,t}^2
+\zeta_{\Lambda,L,t}^{-1}\mathbf 1_{\mathcal B_{\Lambda,L}}
\over2}.
\]
Combining the moment bound with \(s=2t\) and the bad-event probability bound,
\[
E\!\left[
\sigma_{\Lambda,L}(\mathsf W)^t\mathbf 1_{\mathcal B_{\Lambda,L}}
\right]
\le
{10^{8t+4}+2\over2}\,
e^{-40L}\upsilon_{\Lambda,L}^{\,t}
\le
10^{8t+8}e^{-40L}\upsilon_{\Lambda,L}^{\,t}.
\]
% lean: Causalean.Stat.Concentration.PoissonSelfNormalized.poisson_weighted_bad_moment

3.  Now let \(\mathcal I\) be a finite coordinate set with
\(|\mathcal I|\le4\), and let
\((\mathsf W_\iota)_{\iota\in\mathcal I}\) be independent with
\(\mathsf W_\iota\sim\operatorname{Pois}(\Lambda_\iota)\).  Define
\[
\delta_\iota=|\mathsf W_\iota-\Lambda_\iota|,
\qquad
\rho_\iota=\sqrt{\mathsf W_\iota L}+L,
\qquad
\sigma_\iota=\delta_\iota+1024\rho_\iota,
\]
\[
\mathcal B_\iota=\{\delta_\iota>256\rho_\iota\},
\qquad
\Sigma_{\mathcal I}=\sum_{\iota\in\mathcal I}\sigma_\iota,
\qquad
\Xi_{\mathcal I,L}
=
\sqrt{\left(\sum_{\iota\in\mathcal I}\Lambda_\iota\right)L}+L .
\]
If \(\mathcal I=\varnothing\), the corresponding integral is zero.  Otherwise,
since each \(\sigma_\iota\ge0\),
\[
\Sigma_{\mathcal I}^t
\le
|\mathcal I|^{t-1}
\sum_{\iota\in\mathcal I}\sigma_\iota^t ,
\]
for \(t\in\{1,2,4\}\), and therefore
\[
E\!\left[
\Sigma_{\mathcal I}^t
\mathbf 1_{\cup_{\iota'\in\mathcal I}\mathcal B_{\iota'}}
\right]
\le
|\mathcal I|^{t-1}
\sum_{\iota',\iota\in\mathcal I}
E\!\left[
\sigma_\iota^t\mathbf 1_{\mathcal B_{\iota'}}
\right].
\]
When \(\iota=\iota'\), the scalar bound from step 2 gives
\[
E\!\left[\sigma_\iota^t\mathbf 1_{\mathcal B_\iota}\right]
\le
10^{8t+8}e^{-40L}
\left(\sqrt{\Lambda_\iota L}+L\right)^t.
\]
When \(\iota\ne\iota'\), independence factors the expectation and the scalar
moment and probability bounds give
\[
E\!\left[
\sigma_\iota^t\mathbf 1_{\mathcal B_{\iota'}}
\right]
=
E\sigma_\iota^t\,\Pr(\mathcal B_{\iota'})
\le
2\cdot10^{4t+4}e^{-40L}
\left(\sqrt{\Lambda_\iota L}+L\right)^t.
\]
Thus, with
\[
\mathfrak c_t=10^{8t+8}+2\cdot10^{4t+4},
\]
\[
E\!\left[
\Sigma_{\mathcal I}^t
\mathbf 1_{\cup_{\iota'\in\mathcal I}\mathcal B_{\iota'}}
\right]
\le
|\mathcal I|^t\mathfrak c_t e^{-40L}
\sum_{\iota\in\mathcal I}
\left(\sqrt{\Lambda_\iota L}+L\right)^t .
\]
Cauchy--Schwarz gives
\[
\sum_{\iota\in\mathcal I}\sqrt{\Lambda_\iota L}
\le
|\mathcal I|
\sqrt{\left(\sum_{\iota\in\mathcal I}\Lambda_\iota\right)L},
\]
and hence
\[
\sum_{\iota\in\mathcal I}
\left(\sqrt{\Lambda_\iota L}+L\right)^t
\le
|\mathcal I|^{t+1}\Xi_{\mathcal I,L}^{\,t}.
\]
Since \(|\mathcal I|\le4\),
\[
|\mathcal I|^{2t+1}\mathfrak c_t
\le
10^{16(4+1)(t+1)}
=
10^{80(t+1)}.
\]
Also \(e^{-40L}\le e^{-20L}\).  Therefore
\[
E\!\left[
\Sigma_{\mathcal I}^t
\mathbf 1_{\cup_{\iota'\in\mathcal I}\mathcal B_{\iota'}}
\right]
\le
10^{80(t+1)}e^{-20L}
\left\{
\sqrt{\left(\sum_{\iota\in\mathcal I}\Lambda_\iota\right)L}+L
\right\}^{t}.
\]
% lean: Causalean.Stat.Concentration.PoissonSelfNormalized.independent_poisson_badAny_moment

4.  Apply the preceding product bound with
\[
\mathcal I=\mathcal J,
\qquad
\Lambda_\jmath=mq_\jmath .
\]
The product-Poisson pilot law supplies the displayed coordinate laws and
independence, and \(|\mathcal J|=4\).  Hence
\[
E\!\left[
\Sigma^t
\mathbf 1_{\cup_{\jmath\in\mathcal J}\{\delta_\jmath>256\rho_\jmath\}}
\right]
\le
10^{80(t+1)}e^{-20L}
\left\{
\sqrt{\left(\sum_{\jmath\in\mathcal J}mq_\jmath\right)L}+L
\right\}^{t}.
\]
Combining this estimate with step 1 and using \(m>0\),
\[
m^{-1}
\left\{
\sqrt{m\left(\sum_{\jmath\in\mathcal J}q_\jmath\right)L}+L
\right\}
=
\sqrt{{\left(\sum_{\jmath\in\mathcal J}q_\jmath\right)L\over m}}
+
{L\over m}.
\]
Therefore
\[
E\!\left[\Gamma(N')^t\mathbf 1_{\mathsf G^c}\right]
\le
10^{80(t+1)}e^{-20L}
\left\{
\sqrt{{\left(\sum_{\jmath\in\mathcal J}q_\jmath\right)L\over m}}
+
{L\over m}
\right\}^{t}.
\]
Since \(C_t=10^{80(t+1)}\), this is the asserted bound.
% lean: CausalSmith.Stat.DiscreteOptimalValueMinimaxMatched.canonicalPilotBadMoment
\end{proof}
