\item \emph{Tuning and constants.}
Use the canonical Jackson tuning specified in the statement, whose numerical components are
\[
H_0=1024,\qquad \kappa=\frac1{100000},\qquad D_0=3 .
\]
Fix \(\epsilon\in(0,1/2)\) and set
\[
F=1+\epsilon^{-1}.
\]
Let \(C_J>0\) be the constant supplied by \cref{obj:lem:simultaneous-jackson-certificate}, and let
\(C_1,C_2>0\) be the \(t=1\) and \(t=2\) constants in
\cref{obj:lem:canonical-pilot-bad-moment}.  Define
\[
\begin{aligned}
C_\epsilon
={}&1+\{5\cdot10^{16}C_J+450000000\,e^{1000}F\}+450FC_1\\
&\quad+8e^{1000}F^2(2\cdot10^6)^2+8F^2C_2 .
\end{aligned}
\]
All summands after the leading \(1\) are nonnegative, so \(C_\epsilon>0\).

\item \emph{Cell notation.}
Fix \(n,d,\mathbb P\), assume \(n>0\) and
\(\mathbb P\in\mathcal D_{d,\epsilon}^{\mathrm{obs}}\) as in
\cref{obj:def:observed-model-class}, and put
\[
m=\frac n8,\qquad L=L_d=\log(ed),\qquad
\mathcal J=\{0,1\}\times\{0,1\}.
\]
Then \(m>0\).  If \([d]\) is empty, the cellwise assertions are vacuous.  Otherwise fix
\(x\in[d]\).  Write
\[
q_\jmath=q_{\jmath,x},\qquad
q_x=(q_\jmath)_{\jmath\in\mathcal J},\qquad
p_x=\sum_{\jmath\in\mathcal J}q_\jmath .
\]
The coordinates satisfy \(q_\jmath\ge0\).  Since \(x\in[d]\), \(d\ge1\), and hence
\[
L=\log(ed)=1+\log d\ge1,\qquad L>0 .
\]

\item \emph{Poisson splitting.}
The uncapped experiment draws \(M\sim\operatorname{Pois}(n/4)\), giving the asserted marginal law of \(M\).  Conditional on inclusion, an observation in atom \((x',\jmath)\in[d]\times\mathcal J\) is independently marked pilot or evaluation with probability \(1/2\).  Since \(n/4=2m\), Poisson thinning gives pilot and evaluation intensities
\[
2m\,q_{\jmath,x'}\cdot\frac12=mq_{\jmath,x'}.
\]
Thus
\[
\operatorname{Law}\{(N'_{\jmath,x'},N_{\jmath,x'})_{x',\jmath}\}
=
\bigotimes_{x'\in[d]}\bigotimes_{\jmath\in\mathcal J}
\operatorname{Pois}(m q_{\jmath,x'})
\otimes
\bigotimes_{x'\in[d]}\bigotimes_{\jmath\in\mathcal J}
\operatorname{Pois}(m q_{\jmath,x'}) .
\]
Projecting this product law to the fixed cell \(x\) gives
\[
\bigotimes_{\jmath\in\mathcal J}\operatorname{Pois}(m q_{\jmath})
\otimes
\bigotimes_{\jmath\in\mathcal J}\operatorname{Pois}(m q_{\jmath}) .
\]

\item \label{item:pf-centered-factorial-moments}\emph{Centered factorial moments.}
By \cref{obj:def:jackson-factorial-estimator},
\[
U_h(N;z)=\sum_{a=0}^{h}\binom ha(-z)^{h-a}\frac{(N)_a}{m^a}.
\]
Equivalently,
\[
\sum_{h=0}^{\infty}U_h(N;z)\frac{w^h}{h!}
=
e^{-zw}\left(1+\frac wm\right)^N .
\]
For \(N\sim\operatorname{Pois}(mq)\),
\[
E\left[e^{-zw}\left(1+\frac wm\right)^N\right]
=
e^{(q-z)w},
\]
so coefficient comparison gives \(E\,U_h(N;z)=(q-z)^h\).  With two formal variables,
\[
E\left[
e^{-z(w_1+w_2)}
\left(1+\frac {w_1}m\right)^N
\left(1+\frac {w_2}m\right)^N
\right]
=
\exp\left\{(q-z)(w_1+w_2)+\frac qm w_1w_2\right\}.
\]
The coefficient of \(w_1^h w_2^t/(h!t!)\) is therefore
\[
\sum_{\ell=0}^{\min\{h,t\}}
\binom h\ell\binom t\ell\ell!
\left(\frac qm\right)^\ell
(q-z)^{h+t-2\ell}.
\]
The evaluation counts are independent of the pilot counts, so with
\(q=q_{\jmath,x}\) both identities hold conditionally on every fixed pilot table.

\item \label{item:pf-pilot-geometry-failure}\emph{Pilot geometry and failure terms.}
For a fixed pilot table, define
\[
c_{\jmath,x}=\frac{N'_{\jmath,x}}m,\qquad
h_{\jmath,x}=1024\left\{\sqrt{\frac{c_{\jmath,x}L}{m}}+\frac Lm\right\},
\]
\[
\ell_{\jmath,x}=\max\{0,c_{\jmath,x}-h_{\jmath,x}\},\qquad
u_{\jmath,x}=c_{\jmath,x}+h_{\jmath,x},
\]
and
\[
b_{\jmath,x}=\frac{\ell_{\jmath,x}+u_{\jmath,x}}2,\qquad
r_{\jmath,x}=\frac{u_{\jmath,x}-\ell_{\jmath,x}}2 .
\]
Let
\[
Q_x=\prod_{\jmath\in\mathcal J}[\ell_{\jmath,x},u_{\jmath,x}],
\qquad
S_x=F\sum_{\jmath\in\mathcal J}r_{\jmath,x},
\qquad
\varrho_x=d^{1/4}S_x,
\]
and define the good-pilot event
\[
\mathsf G_x
=
\bigcap_{\jmath\in\mathcal J}
\left\{
|c_{\jmath,x}-q_\jmath|\le \frac14h_{\jmath,x}
\right\}.
\]
On \(\mathsf G_x\), \(q_x\in Q_x\): if \(c_{\jmath,x}-h_{\jmath,x}\le0\), then
\(\ell_{\jmath,x}=0\le q_\jmath\) and
\(q_\jmath\le c_{\jmath,x}+h_{\jmath,x}/4\le u_{\jmath,x}\); if
\(c_{\jmath,x}-h_{\jmath,x}>0\), then
\(c_{\jmath,x}-h_{\jmath,x}\le q_\jmath\le c_{\jmath,x}+h_{\jmath,x}\).
Also \(r_{\jmath,x}>0\) and
\[
\frac12h_{\jmath,x}\le r_{\jmath,x}\le h_{\jmath,x}.
\]
Define
\[
\Gamma_x(N')
=
\sum_{\jmath\in\mathcal J}
\left[
|c_{\jmath,x}-q_\jmath|
+
1024\left\{\sqrt{\frac{c_{\jmath,x}L}{m}}+\frac Lm\right\}
\right].
\]
The interval definitions give
\[
\|b_x-q_x\|_1+\sum_{\jmath\in\mathcal J}r_{\jmath,x}
\le 2\Gamma_x(N').
\]
Using the notation of \cref{obj:def:overlap-cone,obj:def:global-extension}, for any two nonnegative four-vectors \(u,v\), the global cell extension satisfies the four-coordinate Lipschitz bound
\[
|\bar f_\epsilon(u)-\bar f_\epsilon(v)|
\le
F\sum_{\jmath\in\mathcal J}|u_\jmath-v_\jmath|.
\]
For a fixed arm \(a\), write
\[
\rho_a(w)=\max\{s_a(w),\epsilon s(w)\},
\]
so \(g_{a,\epsilon}(w)=0\) when \(s(w)=0\), and
\(g_{a,\epsilon}(w)=s(w)w_{a1}/\rho_a(w)\) otherwise.  If one of
\(s(u),s(v)\) is zero, all coordinates of that vector vanish and
\(0\le g_{a,\epsilon}(w)\le s(w)\) gives the bound.  On the positive-mass part, split according to whether
\(s_a(w)\le\epsilon s(w)\) or \(s_a(w)\ge\epsilon s(w)\) for \(w=u,v\).  In the threshold-threshold case the arm value is
\(w_{a1}/\epsilon\).  In the arm-active case use
\[
\left|\frac{u_{a1}}{s_a(u)}-\frac{v_{a1}}{s_a(v)}\right|
\le
\frac{|u_{a1}-v_{a1}|+|u_{a0}-v_{a0}|}{s_a(u)}
\]
and \(s(u)/s_a(u)\le\epsilon^{-1}\).  In the two mixed cases, the nonnegative gap between \(s_a(w)\) and \(\epsilon s(w)\) is bounded by the corresponding changes in \(s_a\) and \(s\), hence by
\(\sum_{\jmath}|u_\jmath-v_\jmath|\); the preceding estimates give the same \(F\)-Lipschitz bound for each \(g_{a,\epsilon}\).  Taking the maximum over the two arms preserves this bound.  The clipping interval in
\cref{obj:def:jackson-factorial-estimator} therefore gives
\[
|T_x^{\mathrm{JF}}-\bar f_\epsilon(q_x)|
\le
F\|b_x-q_x\|_1+d^{1/4}F\sum_{\jmath\in\mathcal J}r_{\jmath,x}
\le
2d^{1/4}F\,\Gamma_x(N').
\]
With \(\mu_x^{\mathrm{pil}}=\bigotimes_{\jmath}\operatorname{Pois}(m q_\jmath)\) and
\(\mu_x^{\mathrm{eval}}=\bigotimes_{\jmath}\operatorname{Pois}(m q_\jmath)\), set
\[
\mathrm{PF}_{1,x}
=
\int_{\mathsf G_x^c}
|T_x^{\mathrm{JF}}-\bar f_\epsilon(q_x)|
\,d(\mu_x^{\mathrm{pil}}\otimes\mu_x^{\mathrm{eval}})
\]
and
\[
\mathrm{PF}_{2,x}
=
\int_{\mathsf G_x^c}
(T_x^{\mathrm{JF}}-\bar f_\epsilon(q_x))^2
\,d(\mu_x^{\mathrm{pil}}\otimes\mu_x^{\mathrm{eval}}).
\]

v_x=L\sqrt{\frac{p_x}{mL}}+\frac Lm\le L^2\eta_x.

\left(\frac L{K_d}\right)^2\le150000^2 .

\(L>3\cdot10^5\), then

K_d\ge \frac L{100000}-1\ge \frac L{150000},

and the same bound follows.

The logarithmic calibrations used below are
\[
L d^{-3/16}\le7,
\qquad
L d^{1/4}e^{-20L}\le7,
\qquad
d^{1/2}e^{-20L}\le d^{1/16},

L^2d^{-3/16}\le225,

L^2d^{1/4}e^{-20L}\le225 .

\[
L=1+\log d\le d^{3/16}+\frac{16}{3}d^{3/16}\le7d^{3/16},

which proves \(Ld^{-3/16}\le7\).  Since \(L\ge1\),

Consequently

Ld^{1/4}e^{-20L}\le Ld^{-3/4}\le Ld^{-3/16}\le7,

and

d^{1/2}e^{-20L}\le d^{-1/2}\le d^{1/16}.

With \(a=3/32\), the same calculus inequality gives

so

L^2\le225d^{3/16}.

Multiplying by \(d^{-3/16}\) proves \(L^2d^{-3/16}\le225\), and using
\(e^{-20L}\le1/d\) once more gives

L^2d^{1/4}e^{-20L}\le L^2d^{-3/4}
\le L^2d^{-3/16}\le225 .
\]

\item \label{item:pf-good-pilot-factorial-ltwo}\emph{Good-pilot factorial \(L_2\) bound.}
Fix a pilot table in \(\mathsf G_x\), and write
\[
\chi_{\jmath,x}=\sqrt{\frac{c_{\jmath,x}L}{m}},
\qquad
h_{\jmath,x}=1024\left(\chi_{\jmath,x}+\frac Lm\right).
\]
First,
\[
\frac{q_\jmath}{m r_{\jmath,x}^2}\le\frac1L .
\]
Indeed, the good-pilot inequality gives
\[
q_\jmath\le c_{\jmath,x}+\frac14h_{\jmath,x}
\le 384\left(c_{\jmath,x}+\frac Lm\right),
\]
where the last inequality uses
\(\chi_{\jmath,x}\le (c_{\jmath,x}+L/m)/2\).  Since
\(r_{\jmath,x}\ge h_{\jmath,x}/2=512(\chi_{\jmath,x}+L/m)\),
\[
m r_{\jmath,x}^2
\ge
m\,512^2\left(\chi_{\jmath,x}+\frac Lm\right)^2
\ge
384L\left(c_{\jmath,x}+\frac Lm\right),
\]
because \(\chi_{\jmath,x}^2=c_{\jmath,x}L/m\).  Combining the last two displays proves the ratio bound.

Also \(q_x\in Q_x\) gives
\[
|q_\jmath-b_{\jmath,x}|\le r_{\jmath,x}.
\]
For any nonnegative integer \(h\), the centered-factorial second-moment identity from \cref{item:pf-centered-factorial-moments} yields
\[
E\left[
\left\{
\frac{U_h(N_{\jmath,x};b_{\jmath,x})}{r_{\jmath,x}^h}
\right\}^2
\middle|N'
\right]
\le
\sum_{\ell=0}^{h}\binom h\ell^2\ell!L^{-\ell}
\le
\exp(h^2/L).
\]
Let \(P_{x,K_d}\) be the Jackson tensor polynomial on \(Q_x\), and write the centered normalized expansion from
\cref{obj:lem:jackson-normalized-coefficient-envelope} as
\[
P_{x,K_d}(v)-\bar f_\epsilon(b_x)
=
\sum_{\alpha\in\mathcal A_x}b_{x,\alpha}
\prod_{\jmath\in\mathcal J}
\left(\frac{v_\jmath-b_{\jmath,x}}{r_{\jmath,x}}\right)^{\alpha_\jmath},
\]
with
\[
\sum_{\alpha\in\mathcal A_x}|b_{x,\alpha}|
\le (2^{60})^{K_d}F\sum_{\jmath\in\mathcal J}r_{\jmath,x},
\qquad
\alpha_\jmath\le2(K_d-1).
\]
Define the factorial lift
\[
\mathcal L_x(N)
=
\sum_{\alpha\in\mathcal A_x}b_{x,\alpha}
\prod_{\jmath\in\mathcal J}
\frac{U_{\alpha_\jmath}(N_{\jmath,x};b_{\jmath,x})}
{r_{\jmath,x}^{\alpha_\jmath}} .
\]
Independence of the evaluation coordinates and Minkowski's inequality give
\[
E\{\mathcal L_x(N)^2\mid N'\}
\le
(2^{60})^{2K_d}F^2
\left(\sum_{\jmath\in\mathcal J}r_{\jmath,x}\right)^2
\exp(16K_d^2/L).
\]
The calibrated coefficient growth bound is
\[
(2^{60})^{2K_d}\exp(16K_d^2/L)
\le e^{1000}d^{1/16}.
\]
Indeed, \(K_d=\max\{2,\lfloor L/100000\rfloor\}\) and the floor bound give
\[
K_d\le 2+\frac{L}{100000}.
\]
Since \(L\ge1\), squaring the last display gives
\[
K_d^2\le L\left(5+\frac{L}{10^{10}}\right),
\qquad
\frac{K_d^2}{L}\le 5+\frac{L}{10^{10}}.
\]
Also \(L=1+\log d\), and hence
\[
120K_d+16\frac{K_d^2}{L}
\le
120\left(2+\frac{L}{100000}\right)
+16\left(5+\frac{L}{10^{10}}\right)
\le
1000+\frac{\log d}{16}.
\]
Because \(2<e\),
\[
(2^{60})^{2K_d}=2^{120K_d}\le e^{120K_d}.
\]
Multiplying by the remaining exponential factor and using the preceding exponent comparison yields
\[
(2^{60})^{2K_d}\exp(16K_d^2/L)
\le
\exp\left(120K_d+16\frac{K_d^2}{L}\right)
\le
\exp\left(1000+\frac{\log d}{16}\right)
=e^{1000}d^{1/16}.
\]
Therefore
\[
E\{\mathcal L_x(N)^2\mid N'\}
\le
e^{1000}d^{1/16}S_x^2 .
\]

\item \label{item:pf-good-pilot-bias}\emph{Good-pilot bias.}
For the fixed good pilot, the factorial identities give
\[
E\{\mathcal L_x(N)\mid N'\}
=
P_{x,K_d}(q_x)-\bar f_\epsilon(b_x).
\]
Let
\[
Z_x=\bar f_\epsilon(b_x)+\mathcal L_x(N),
\qquad
T_x^{\mathrm{JF}}
=
\Pi_{[\bar f_\epsilon(b_x)-\varrho_x,\,
       \bar f_\epsilon(b_x)+\varrho_x]}(Z_x).
\]
Hence
\[
E(T_x^{\mathrm{JF}}\mid N')-\bar f_\epsilon(q_x)
=
\{P_{x,K_d}(q_x)-\bar f_\epsilon(q_x)\}
+
E(T_x^{\mathrm{JF}}-Z_x\mid N').
\]
For each coordinate,
\[
\sqrt{(q_\jmath-\ell_{\jmath,x})(u_{\jmath,x}-q_\jmath)}
\le
10^7\sqrt{\frac{q_\jmath L}{m}} .
\]
To see this, if \(c_{\jmath,x}-h_{\jmath,x}\le0\), then
\(\ell_{\jmath,x}=0\), and the quadratic inequality
\(c_{\jmath,x}\le h_{\jmath,x}\) gives
\(c_{\jmath,x}\le1.1\cdot10^6L/m\) and
\(h_{\jmath,x}\le1.2\cdot10^9L/m\), so
\[
(q_\jmath-\ell_{\jmath,x})(u_{\jmath,x}-q_\jmath)
\le q_\jmath(c_{\jmath,x}+h_{\jmath,x})
\le 10^{14}\frac{q_\jmath L}{m}.
\]
If \(c_{\jmath,x}-h_{\jmath,x}>0\), then the good-pilot inequality gives
\(q_\jmath\ge3c_{\jmath,x}/4\) and \(L/m\le c_{\jmath,x}\), whence
\[
h_{\jmath,x}\le4096\sqrt{\frac{q_\jmath L}{m}},
\qquad
\sqrt{(q_\jmath-\ell_{\jmath,x})(u_{\jmath,x}-q_\jmath)}
\le r_{\jmath,x}=h_{\jmath,x}.
\]
Applying \cref{obj:lem:simultaneous-jackson-certificate} at \(q_x\in Q_x\) gives
\[
|P_{x,K_d}(q_x)-\bar f_\epsilon(q_x)|
\le
C_J
\sum_{\jmath\in\mathcal J}
\left\{
\frac{10^7\sqrt{q_\jmath L/m}}{K_d}
+
\frac{r_{\jmath,x}}{K_d^2}
\right\}.
\]
Since \(\sum_{\jmath}\sqrt{q_\jmath L/m}\le2\sqrt{p_xL/m}\) and
\(\sum_{\jmath}r_{\jmath,x}\le2\cdot10^6v_x\),
\[
\sum_{\jmath\in\mathcal J}
\left\{
\frac{10^7\sqrt{q_\jmath L/m}}{K_d}
+
\frac{r_{\jmath,x}}{K_d^2}
\right\}
\le
22\cdot10^6\frac{\sqrt{p_xL/m}}{K_d}
+
2\cdot10^6\frac{L/m}{K_d^2}
\le
5\cdot10^{16}\eta_x .
\]
From the interval projection definition in \cref{obj:synth_13}, a direct case split gives the following elementary consequence: for every real midpoint \(\operatorname{mid}\), every real input \(\operatorname{raw}\), and every radius \(\operatorname{rad}>0\),
\[
\left|
\Pi_{[\operatorname{mid}-\operatorname{rad},\operatorname{mid}+\operatorname{rad}]}
(\operatorname{raw})-\operatorname{raw}
\right|
\le
\frac{(\operatorname{raw}-\operatorname{mid})^2}{\operatorname{rad}}.
\]
If \(\operatorname{raw}\) lies inside the interval, the left side is zero.  If
\(\operatorname{raw}<\operatorname{mid}-\operatorname{rad}\), then the inequality follows after multiplying by \(\operatorname{rad}\) from
\[
(\operatorname{raw}-\operatorname{mid})^2
+\operatorname{rad}(\operatorname{raw}-\operatorname{mid})
+\operatorname{rad}^2\ge0,
\]
and the case \(\operatorname{raw}>\operatorname{mid}+\operatorname{rad}\) is identical with the middle sign reversed.  Applying this with midpoint \(\bar f_\epsilon(b_x)\), radius \(\varrho_x\), and input \(Z_x\), and using
\(Z_x-\bar f_\epsilon(b_x)=\mathcal L_x(N)\), gives
\[
|T_x^{\mathrm{JF}}-Z_x|
\le
\frac{\mathcal L_x(N)^2}{\varrho_x}.
\]
Using the factorial \(L_2\) bound from \cref{item:pf-good-pilot-factorial-ltwo},
\[
|E(T_x^{\mathrm{JF}}-Z_x\mid N')|
\le
\frac{e^{1000}d^{1/16}S_x^2}{d^{1/4}S_x}
=
e^{1000}F d^{-3/16}\sum_{\jmath\in\mathcal J}r_{\jmath,x}.
\]
Together with the good-pilot radius bound and \(L^2d^{-3/16}\le225\), this implies
\[
\left|
E(T_x^{\mathrm{JF}}\mid N')-\bar f_\epsilon(q_x)
\right|
\le
\{5\cdot10^{16}C_J+450000000\,e^{1000}F\}\eta_x
\qquad\text{on }\mathsf G_x .
\]

2\left\{\frac{p_xL}{m}+\frac{L^2}{m^2}\right\},
\]
and \(d^{1/2}e^{-20L}\le d^{1/16}\), so
\[
\mathrm{PF}_{2,x}
\le
8F^2C_2d^{1/16}
\left\{\frac{p_xL}{m}+\frac{L^2}{m^2}\right\}.
\]
The displayed estimates are exactly the asserted pilot-failure bounds after enlarging by
\(C_\epsilon\).
% lean: canonicalPilotFailureBounds

\item \emph{Bias.}
Disintegrating over the pilot table, the good-pilot conditional bound from \cref{item:pf-good-pilot-bias} and the bad-pilot contribution give
\[
\left|
E T_x^{\mathrm{JF}}-\bar f_\epsilon(q_x)
\right|
\le
\{5\cdot10^{16}C_J+450000000\,e^{1000}F\}\eta_x+\mathrm{PF}_{1,x}.
\]
Using the estimate for \(\mathrm{PF}_{1,x}\) and the definition of \(C_\epsilon\),
\[
\left|
E T_x^{\mathrm{JF}}-\bar f_\epsilon(q_x)
\right|
\le
C_\epsilon
\left\{
\sqrt{\frac{p_x}{mL_d}}+\frac1{mL_d}
\right\}.
\]
% lean: centered_factorial_pilot_control

\item \emph{Variance.}
Combining the good-pilot second-moment estimate with \(\mathrm{PF}_{2,x}\) yields
\[
E\{T_x^{\mathrm{JF}}-\bar f_\epsilon(q_x)\}^2
\le
4e^{1000}d^{1/16}(2\cdot10^6Fv_x)^2+\mathrm{PF}_{2,x}.
\]
Using the bound on \(v_x^2\), the estimate for \(\mathrm{PF}_{2,x}\), and the definition of
\(C_\epsilon\),
\[
E\{T_x^{\mathrm{JF}}-\bar f_\epsilon(q_x)\}^2
\le
C_\epsilon d^{1/16}
\left\{
\frac{p_xL_d}{m}+\frac{L_d^2}{m^2}
\right\}.
\]
Variance is bounded by the mean squared deviation about any fixed center, so
\[
\operatorname{Var}(T_x^{\mathrm{JF}})
\le
E\{T_x^{\mathrm{JF}}-\bar f_\epsilon(q_x)\}^2.
\]
Thus the stated bias, variance, and pilot-failure estimates hold uniformly in the cell \(x\), in
\(n,d\), and in \(\mathbb P\in\mathcal D_{d,\epsilon}^{\mathrm{obs}}\).
% lean: cellStatisticVariance
% lean: centered_factorial_pilot_control
\end{enumerate}
\end{proof}
