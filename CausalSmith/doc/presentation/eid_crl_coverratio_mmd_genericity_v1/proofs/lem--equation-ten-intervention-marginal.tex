\begin{proof}[Proof of \cref{obj:lem:equation-ten-intervention-marginal}]
Put \(t_i=\pi(i)\in[n]\) and \(\mathcal A_i=A_i^\tau\subseteq[n]\).

\begin{enumerate}
\item First, \(\mathcal A_i\) is parent-closed:
\[
\lambda\in\mathcal A_i,\qquad \chi\in\operatorname{pa}_G(\lambda)
\quad\Longrightarrow\quad
\chi\in\mathcal A_i .
\]
Indeed, if \(\lambda=t_i=\pi(i)\), then \(\pi^{-1}(\chi)\to i\) in the transported environment-label graph, so the transported-DAG order gives
\[
\tau(\pi^{-1}(\chi))<\tau(i).
\]
Thus \(\pi^{-1}(\chi)\in B_i^\tau\), hence \(\chi\in\pi(B_i^\tau)\subseteq\mathcal A_i\). If instead \(\lambda=\pi(\xi)\) for some \(\xi\in B_i^\tau\), then \(\pi^{-1}(\chi)\to \xi\) in the transported graph, and the same order gives
\[
\tau(\pi^{-1}(\chi))<\tau(\xi)<\tau(i).
\]
Thus again \(\pi^{-1}(\chi)\in B_i^\tau\), hence \(\chi\in\mathcal A_i\). % lean: decoderRetainedLatentSet_parentClosed

\item Define, for \(\omega\in[0,1]^n\) and \(\lambda\in[n]\),
\[
\widetilde p_\lambda(\omega)
=
\begin{cases}
q_{t_i}(\omega_{t_i}), & \lambda=t_i,\\
p_\lambda\!\left(\omega_\lambda\mid
\omega_{\operatorname{pa}_G(\lambda)}\right), & \lambda\ne t_i .
\end{cases}
\]
The intervention density at target \(t_i\) is therefore
\[
\prod_{\lambda\in[n]}\widetilde p_\lambda(\omega)
=
q_{t_i}(\omega_{t_i})
\prod_{\lambda\ne t_i}
p_\lambda\!\left(\omega_\lambda\mid
\omega_{\operatorname{pa}_G(\lambda)}\right).
\]
For \(v\in[0,1]^n\) and \(r\in[0,1]^{[n]\setminus\mathcal A_i}\), write \(u^{v,\tau}(r)\) for the point satisfying
\[
u^{v,\tau}(r)_a=
\begin{cases}
v_a, & a\in\mathcal A_i,\\
r_a, & a\notin\mathcal A_i .
\end{cases}
\]
Since \(t_i\in\mathcal A_i\), the preceding display is the ordinary product density after replacing the \(t_i\)-factor by \(q_{t_i}\).

For a parent-closed subset \(\mathcal A\subseteq[n]\), an index \(\jmath\in\mathcal A\), and a normalized replacement density \(q_{\jmath}\), define
\[
\widehat p_\lambda(\omega)=
\begin{cases}
q_{\jmath}(\omega_{\jmath}), & \lambda=\jmath,\\
p_\lambda\!\left(\omega_\lambda\mid
\omega_{\operatorname{pa}_G(\lambda)}\right), & \lambda\ne\jmath,
\end{cases}
\]
and let
\[
\widehat p^{\mathrm{obs}}(\omega)
=
\prod_{\lambda\in[n]}
p_\lambda\!\left(\omega_\lambda\mid
\omega_{\operatorname{pa}_G(\lambda)}\right),
\qquad
\widehat p^{\mathrm{int}}(\omega)
=
\prod_{\lambda\in[n]}\widehat p_\lambda(\omega).
\]
The replacement construction gives
\[
\widehat p^{\mathrm{int}}(\omega)
=
q_{\jmath}(\omega_{\jmath})
\prod_{\lambda\ne\jmath}
p_\lambda\!\left(\omega_\lambda\mid
\omega_{\operatorname{pa}_G(\lambda)}\right),
\]
that is, the observational product factorization with the single \(\jmath\)-factor replaced. For this replaced factorization, parent-closed marginalization gives
\[
\int_{[0,1]^{[n]\setminus\mathcal A}}
\widehat p^{\mathrm{int}}(u)\,du_{[n]\setminus\mathcal A}
=
\prod_{\lambda\in\mathcal A}\widehat p_\lambda(\omega),
\]
where the integrated point \(u\) agrees with \(\omega\) on \(\mathcal A\). This is the usual finite product-density marginal identity for a parent-closed coordinate set: after the single-factor replacement, the factors again form a normalized directed factorization, and parent-closedness ensures that retaining \(\mathcal A\) leaves exactly the product of the retained factors. The sigma-finiteness and finite-coordinate integration bookkeeping are supplied by the unit-interval reference measure, while \cref{obj:ass:positive-normalized-smooth-mechanisms} supplies the normalized measurable factors and the normalized replacement density.

Apply this identity with \(\mathcal A=\mathcal A_i\) and \(\jmath=t_i\). Unfolding the retained product gives
\[
\prod_{\lambda\in\mathcal A_i}\widetilde p_\lambda(v)
=
q_{t_i}(v_{t_i})
\prod_{\lambda\in\mathcal A_i\setminus\{t_i\}}
p_\lambda\!\left(v_\lambda\mid v_{\operatorname{pa}_G(\lambda)}\right),
\]
because the erased product contains no \(t_i\)-factor. Therefore
\[
\int_{[0,1]^{[n]\setminus\mathcal A_i}}
\prod_{\lambda\in[n]}\widetilde p_\lambda(u^{v,\tau}(r))\,dr
=
q_{t_i}(v_{t_i})
\prod_{\lambda\in\mathcal A_i\setminus\{t_i\}}
p_\lambda\!\left(v_\lambda\mid v_{\operatorname{pa}_G(\lambda)}\right).
\] % lean: interventionMarginalDensity_eq_replacement_mul_partialDensity

\item Substituting \(t_i=\pi(i)\) and \(\mathcal A_i=A_i^\tau\) in the preceding identity gives, for every \(v\in[0,1]^n\),
\[
\int_{[0,1]^{[n]\setminus A_i^\tau}}
q_{\pi(i)}\!\left(u^{v,\tau}(r)_{\pi(i)}\right)
\prod_{\ell\ne \pi(i)}
p_\ell\!\left(u^{v,\tau}(r)_\ell\mid
u^{v,\tau}(r)_{\operatorname{pa}_G(\ell)}\right)\,dr
=
q_{\pi(i)}(v_{\pi(i)})
\prod_{\ell\in A_i^\tau\setminus\{\pi(i)\}}
p_\ell\!\left(v_\ell\mid v_{\operatorname{pa}_G(\ell)}\right),
\]
which is the asserted identity. % lean: equationTen_interventionMarginalDensity
\end{enumerate}
\end{proof}
