\begin{proof}[Proof of \cref{obj:lem:ratio-nonancestor-zero}]
\begin{enumerate}
\item Let \(\mu^0\) be the latent observational law with density \(p^0(v)\) on \([0,1]^n\), and let \(\mu^{\pi(j)}\) be the latent law for the single intervention at \(\pi(j)\), with density \(p^{\pi(j)}(v)\) on \([0,1]^n\).  On the cube define
\[
\rho_i(v)
=
\frac{q_{\pi(i)}(v_{\pi(i)})}
{p_{\pi(i)}(v_{\pi(i)}\mid v_{\operatorname{pa}_G(\pi(i))})},
\qquad v\in[0,1]^n .
\]
The denominator is positive on \([0,1]^n\) by \cref{obj:ass:positive-normalized-smooth-mechanisms}.  By \cref{obj:ass:one-perfect-intervention-per-node}, the observed laws are the \(f\)-pushforwards of the latent laws, and the likelihood-ratio formula stated there identifies the two ratio laws as
\[
(R_i)_\#P^0=(\rho_i)_\#\mu^0,
\qquad
(R_i)_\#P^j=(\rho_i)_\#\mu^{\pi(j)} .
\]
% lean: finiteDensityObservedWorldBridge_of_assumptions

\item Since \(j\ne i\) and \(\pi\) is injective, \(\pi(j)\ne\pi(i)\).  Together with the hypothesis that \(\pi(j)\) is not an ancestor of \(\pi(i)\), this gives
\[
\pi(j)\notin \mathcal A_i,
\qquad
\mathcal A_i
=
\{\pi(i)\}\cup
\{\mathfrak u\in[n]:\mathfrak u\text{ is an ancestor of }\pi(i)\text{ in }G\} .
\]
The set \(\mathcal A_i\) is parent-closed: if \(\mathfrak u\in\mathcal A_i\) and \(\mathfrak v\in\operatorname{pa}_G(\mathfrak u)\), then \(\mathfrak v\in\mathcal A_i\), because a parent of \(\pi(i)\) is an ancestor of \(\pi(i)\), and a parent of an ancestor of \(\pi(i)\) is again an ancestor of \(\pi(i)\).  Let \(\operatorname{pr}_{\mathcal A_i}:[0,1]^n\to[0,1]^{\mathcal A_i}\) be the coordinate projection.  For \(\mathfrak a\in[0,1]^{\mathcal A_i}\) and \(\mathfrak b\in[0,1]^{[n]\setminus\mathcal A_i}\), write \(\mathfrak a\oplus\mathfrak b\) for the point of \([0,1]^n\) whose \(\mathcal A_i\)-coordinates are \(\mathfrak a\) and whose complementary coordinates are \(\mathfrak b\).  For such \((\mathfrak a,\mathfrak b)\), set
\[
\Gamma_i^0(\mathfrak a,\mathfrak b)
=
\prod_{\ell\in[n]\setminus\mathcal A_i}
 p_\ell\bigl((\mathfrak a\oplus\mathfrak b)_\ell\mid
 (\mathfrak a\oplus\mathfrak b)_{\operatorname{pa}_G(\ell)}\bigr)
\]
and
\[
\Gamma_i^j(\mathfrak a,\mathfrak b)
=
q_{\pi(j)}(\mathfrak b_{\pi(j)})
\prod_{\substack{\ell\in[n]\setminus\mathcal A_i\\ \ell\ne\pi(j)}}
 p_\ell\bigl((\mathfrak a\oplus\mathfrak b)_\ell\mid
 (\mathfrak a\oplus\mathfrak b)_{\operatorname{pa}_G(\ell)}\bigr),
\]
where the coordinate \(\mathfrak b_{\pi(j)}\) is defined because \(\pi(j)\notin\mathcal A_i\).  Parent-closedness implies that, for every \(\ell\in\mathcal A_i\), the factor \(p_\ell\) evaluated at \(\mathfrak a\oplus\mathfrak b\) depends only on \(\mathfrak a\).  Hence
\[
p^0(\mathfrak a\oplus\mathfrak b)
=
\left\{
\prod_{\ell\in\mathcal A_i}
 p_\ell\bigl(\mathfrak a_\ell\mid
 \mathfrak a_{\operatorname{pa}_G(\ell)}\bigr)
\right\}
\Gamma_i^0(\mathfrak a,\mathfrak b)
\]
and
\[
p^{\pi(j)}(\mathfrak a\oplus\mathfrak b)
=
\left\{
\prod_{\ell\in\mathcal A_i}
 p_\ell\bigl(\mathfrak a_\ell\mid
 \mathfrak a_{\operatorname{pa}_G(\ell)}\bigr)
\right\}
\Gamma_i^j(\mathfrak a,\mathfrak b).
\]
The complement factors have unit iterated integral.  Indeed, if a nonempty collection of complementary vertices remains to be integrated, choose one that is maximal in a topological ordering of the induced remaining set.  Parent-closedness prevents this vertex from being a parent of any retained factor, and maximality prevents it from being a parent of any other remaining factor.  Its coordinate therefore appears in the current product only through its own factor; integrating that factor gives one by the normalization of \(p_\ell\), and, when the vertex is \(\pi(j)\), by the normalization of \(q_{\pi(j)}\).  Iterating this deletion, with the empty remaining collection as the base case, gives
\[
\int_{[0,1]^{[n]\setminus\mathcal A_i}}
\Gamma_i^0(\mathfrak a,\mathfrak b)\,d\mathfrak b=1,
\qquad
\int_{[0,1]^{[n]\setminus\mathcal A_i}}
\Gamma_i^j(\mathfrak a,\mathfrak b)\,d\mathfrak b=1 .
\]
Consequently the two complement-marginal densities on \([0,1]^{\mathcal A_i}\) are the same:
\[
\int_{[0,1]^{[n]\setminus\mathcal A_i}}
p^0(\mathfrak a\oplus\mathfrak b)\,d\mathfrak b
=
\prod_{\ell\in\mathcal A_i}
 p_\ell\bigl(\mathfrak a_\ell\mid
 \mathfrak a_{\operatorname{pa}_G(\ell)}\bigr)
=
\int_{[0,1]^{[n]\setminus\mathcal A_i}}
p^{\pi(j)}(\mathfrak a\oplus\mathfrak b)\,d\mathfrak b .
\]
Thus the projected latent laws agree:
\[
(\operatorname{pr}_{\mathcal A_i})_\#\mu^0
=
(\operatorname{pr}_{\mathcal A_i})_\#\mu^{\pi(j)} .
\]
Since \(\{\pi(i)\}\cup\operatorname{pa}_G(\pi(i))\subseteq\mathcal A_i\), the ratio map \(\rho_i\) factors through this projection.  Explicitly, the map \(\widetilde\rho_i:[0,1]^{\mathcal A_i}\to\mathbb R\) defined by
\[
\widetilde\rho_i(\mathfrak a)
=
\frac{q_{\pi(i)}(\mathfrak a_{\pi(i)})}
{p_{\pi(i)}(\mathfrak a_{\pi(i)}\mid
\mathfrak a_{\operatorname{pa}_G(\pi(i))})}
\]
satisfies \(\rho_i=\widetilde\rho_i\circ\operatorname{pr}_{\mathcal A_i}\).  Pushing the projected-law identity forward by \(\widetilde\rho_i\) gives
\[
(\rho_i)_\#\mu^0=(\rho_i)_\#\mu^{\pi(j)},
\]
and hence, by the ratio-law identifications in the preceding step,
\[
(R_i)_\#P^0=(R_i)_\#P^j .
\]
% lean: FiniteDensityObservedWorldBridge.ratioLaw_eq_of_nonancestor

\item The population discrepancy is the norm of the difference of the two Gaussian-kernel mean embeddings of these ratio laws.  Using the equality just proved,
\[
\begin{aligned}
D_{ji}
&=
\left\|
\int \phi(r)\,d\bigl((R_i)_\#P^0\bigr)(r)
-
\int \phi(r)\,d\bigl((R_i)_\#P^j\bigr)(r)
\right\|_{\mathcal H_k}  \\
&=
\left\|
\int \phi(r)\,d\bigl((R_i)_\#P^0\bigr)(r)
-
\int \phi(r)\,d\bigl((R_i)_\#P^0\bigr)(r)
\right\|_{\mathcal H_k}
=0 .
\end{aligned}
\]
% lean: populationDiscrepancy_eq_zero_of_ratioLaw_eq
\end{enumerate}
\end{proof}
