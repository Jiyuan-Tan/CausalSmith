\begin{proof}[Proof of \cref{obj:lem:exact-ratio-decoder-continuous-rank-clauses}]
Fix an injective topological ordering \(\tau\) of \(H_D\), and fix \(i\in[n]\).  Write
\[
B_i^\tau=\{j\in[n]:\tau(j)<\tau(i)\},
\qquad
L_{B_i^\tau}(x)=(L_j(x))_{j\in B_i^\tau}.
\]

\begin{enumerate}
\item First record the order-recovery consequence that is passed to the ordered decoder argument.  If \(G^\pi\) contains \(j\to k\), then the one-edge path \(j\to k\) is an element of \(\operatorname{TC}(G^\pi)\).  The assumed equality
\[
\operatorname{TC}(H_D)=\operatorname{TC}(G^\pi)
\]
therefore gives a directed \(H_D\)-path from \(j\) to \(k\).  Along each edge \(a\to b\) of this path, topologicality of \(\tau\) for \(H_D\) gives \(\tau(a)<\tau(b)\); induction over the path and transitivity of \(<\) yield
\[
G^\pi:j\to k
\quad\Longrightarrow\quad
\tau(j)<\tau(k).
\]
Thus \(\tau\) is also topological for the transported graph \(G^\pi\), and in particular every \(G^\pi\)-parent of \(k\) lies in \(B_k^\tau\). % lean: permutedGraphOrdered_of_transitiveClosure

\item The supplied observed laws are probability laws.  The observational law is the pushforward by \(f\) of the normalized observational latent density, and each target-intervention law is the pushforward by the same \(f\) of the corresponding normalized intervention density.  Hence, for every \(e=0,\ldots,n\),
\[
P^e(\mathbb R^n)=1.
\]
Consequently the decoder is computed from these laws themselves: writing \(\mathsf P\) for the law family it uses,
\[
\mathsf P_e=P^e,\qquad e=0,\ldots,n.
\]
In each subsequent conditional-CDF statement, the selected family is rewritten by this identity to the stated observed laws. % lean: observedWorld_laws_isProbabilityMeasure

\item Let \(\kappa_i^\tau(\ell,\cdot)\) be the probability kernel from \cref{obj:lem:equation-eleven-kernel-factorization}, extended to all \(\ell\in\mathbb R^{B_i^\tau}\) by first retracting \(\ell\) onto the realized predecessor-score range.  Thus, for every \(\ell\in\mathbb R^{B_i^\tau}\), \(\kappa_i^\tau(\ell,\cdot)\) is the law of
\[
\log\frac{q_{\pi(i)}(W_i)}
{p_{\pi(i)}\!\left(\widetilde v_i^\tau(z_i^\tau(\ell),W_i)\right)},
\qquad
W_i\sim q_{\pi(i)}(w)\,dw\ \text{on }[0,1],
\]
where the retraction in \(z_i^\tau\) supplies the off-range values.  Define the total candidate
\[
C_i^\star(t\mid \ell)
=
\kappa_i^\tau\bigl(\ell,(-\infty,t]\bigr),
\qquad
(t,\ell)\in\mathbb R\times\mathbb R^{B_i^\tau}.
\]
The same formula applies when \(B_i^\tau=\varnothing\), with \(\mathbb R^{B_i^\tau}\) the one-point product space.  Put
\[
\mu_i^\tau=(L_{B_i^\tau})_{\#}P^i,
\qquad
\zeta_i^\tau=(L_{B_i^\tau},L_i)_{\#}P^i .
\]
By \cref{obj:lem:equation-eleven-kernel-factorization},
\[
\zeta_i^\tau
=
\mu_i^\tau\otimes\kappa_i^\tau .
\]
The candidate \(C_i^\star\) has the four properties required of a continuous conditional-CDF version.  First, \((t,\ell)\mapsto C_i^\star(t\mid\ell)\) is Borel measurable because it is the lower-half-line evaluation of a measurable probability kernel.  Second, uniqueness of regular conditional distributions in the displayed product disintegration gives, for every fixed \(t\in\mathbb R\),
\[
C_i^\star(t\mid \ell)
=
P^i(L_i\le t\mid L_{B_i^\tau}=\ell)
\quad
\text{for }\mu_i^\tau\text{-a.e. }\ell .
\]
The same disintegration, pulled back along the argument map \(x\mapsto(L_i(x),L_{B_i^\tau}(x))\), gives the joint almost-everywhere version
\[
C_i^\star(r\mid \ell)
=
P^i(L_i\le r\mid L_{B_i^\tau}=\ell)
\quad
\text{for }(L_i,L_{B_i^\tau})_{\#}P^i\text{-a.e. }(r,\ell).
\]
For continuity, put
\[
\mathcal R_i^\tau
=
\{L_{B_i^\tau}(f(u)):u\in[0,1]^n\}.
\]
On \(\mathbb R\times\mathcal R_i^\tau\), the retraction in the definition of \(\kappa_i^\tau\) is the identity, and the predecessor-score homeomorphism rewrites \(C_i^\star\) as the lower-level integral of the one-dimensional score
\[
w\longmapsto
\log\frac{q_{\pi(i)}(w)}{p_{\pi(i)}(\widetilde v_i^\tau(z,w))},
\qquad z=(\Lambda_i^\tau)^{-1}(\ell).
\]
The positivity and smoothness in \cref{obj:ass:positive-normalized-smooth-mechanisms}, together with the fixed own-derivative sign in \cref{obj:ass:fixed-own-derivative-sign}, make this score continuous in \((z,w)\) and strictly monotone in \(w\) on \([0,1]\); hence its lower-level integral varies continuously in \((t,\ell)\) on \(\mathbb R\times\mathcal R_i^\tau\).  The observed conditional-ratio support satisfies
\[
\mathcal S_i^\tau
\subseteq
\mathbb R\times\mathcal R_i^\tau,
\]
because under \(P^i\) the predecessor projection is supported on realized predecessor scores.  Thus \(C_i^\star\) is continuous on \(\mathcal S_i^\tau\), and so it is a continuous conditional-CDF version of \(L_i\) given \(L_{B_i^\tau}\). % lean: equationElevenAmbientCDF_isContinuousVersion

\item Since the class of continuous conditional-CDF versions is nonempty, the law-selected conditional CDF \(C_i\) used by \cref{obj:def:population-decoder} is itself one of these versions.  This proves the first asserted clause:
\[
C_i
\quad\text{is a continuous conditional-CDF version of }L_i\text{ given }L_{B_i^\tau}.
\]
Taking \(C=C_i^\star\) proves the asserted existence of a continuous version once its pointwise latent formula is verified in the next step. % lean: observedConditionalRatioCDF_isContinuousVersion

\item Fix \(v\in[0,1]^n\), and set
\[
\ell_v=L_{B_i^\tau}(f(v)).
\]
The realized score \(\ell_v\) belongs to \(\mathcal R_i^\tau\), so the retraction used to define \(\kappa_i^\tau\) fixes it.  The score-image kernel then evaluates at the predecessor coordinates reconstructed from \(\ell_v\).  For each \(w\in[0,1]\), let \(\bar v_i^\tau(\ell_v,w)\) be this reconstructed point with target coordinate \(\pi(i)\) set to \(w\).  The order-recovery step places every transported parent of \(\pi(i)\) in \(B_i^\tau\), and the inverse predecessor-score map reconstructs those predecessor coordinates from \(\ell_v\).  Hence
\[
\bar v_i^\tau(\ell_v,w)_a=v^{(w)}_a
\qquad
\text{for every }a\in\operatorname{pa}_G(\pi(i)),
\]
where \(v^{(w)}\) is obtained from \(v\) by replacing coordinate \(\pi(i)\) by \(w\).  Parent locality of the mechanism gives
\[
p_{\pi(i)}\bigl(\bar v_i^\tau(\ell_v,w)\bigr)
=
p_{\pi(i)}\bigl(v^{(w)}\bigr).
\]
Substituting this identity into the lower-half-line evaluation of \(\kappa_i^\tau\) yields, for every \(t\in\mathbb R\),
\[
C_i^\star\bigl(t\mid L_{B_i^\tau}(f(v))\bigr)
=
\int_0^1
\mathbf 1\!\left\{
\log\frac{q_{\pi(i)}(w)}
{p_{\pi(i)}(v^{(w)})}\le t
\right\}
q_{\pi(i)}(w)\,dw .
\]
This is the second asserted clause, with \(C=C_i^\star\). % lean: equationElevenAmbientCDF_apply_of_latentState

\item Any two continuous conditional-CDF versions agree on the observed conditional-ratio support.  To see this, fix another continuous version \(C\).  For each fixed threshold \(t\), the defining conditional-CDF property gives
\[
C(t\mid\ell)=C_i(t\mid\ell)
\quad\text{for }\mu_i^\tau\text{-a.e. }\ell .
\]
Both sides are continuous in \(\ell\) on \(\operatorname{supp}\mu_i^\tau\).  Therefore, for every \(t\in\mathbb R\) and every \(\ell\in\operatorname{supp}\mu_i^\tau\), that is, for every \((t,\ell)\in\mathcal S_i^\tau=\mathbb R\times\operatorname{supp}\mu_i^\tau\),
\[
C(t\mid\ell)=C_i(t\mid\ell).
\]
Equivalently,
\[
C(z_1\mid z_2)=C_i(z_1\mid z_2)
\qquad
\text{for every }(z_1,z_2)\in\mathcal S_i^\tau .
\]
It remains to place realized predecessor scores in this support.  For every \(v\in[0,1]^n\),
\[
L_{B_i^\tau}(f(v))
\in
\operatorname{supp}\mu_i^\tau .
\]
Indeed, \(P^i\) is the pushforward by \(f\) of the target-\(\pi(i)\) intervention law; by strict positivity of the latent density on the cube and the shared diffeomorphic mixing, \(f(v)\) lies in the support of \(P^i\).  The predecessor-log-ratio projection is continuous on the observed support, so every neighborhood of \(L_{B_i^\tau}(f(v))\) has preimage with positive \(P^i\)-mass.  Applying the support uniqueness just proved to \(C_i^\star\), and then using the previous step, gives
\[
C_i\bigl(t\mid L_{B_i^\tau}(f(v))\bigr)
=
\int_0^1
\mathbf 1\!\left\{
\log\frac{q_{\pi(i)}(w)}
{p_{\pi(i)}(v^{(w)})}\le t
\right\}
q_{\pi(i)}(w)\,dw .
\]
This proves the third and fourth asserted clauses. % lean: observedConditionalRatioCDF_eq_on_support

\item It remains to evaluate the rank at the realized target log-ratio.  Fix \(v\in[0,1]^n\).  The latent ratio formula in \cref{obj:ass:one-perfect-intervention-per-node}, together with the selected log-ratio version used by the decoder, gives the pointwise log-ratio identity
\[
L_i(f(v))
=
\log\frac{q_{\pi(i)}(v_{\pi(i)})}{p_{\pi(i)}(v)} .
\]
By definition of the rank coordinate and by the fourth clause already proved,
\[
U_i(f(v))
=
\int_0^1
\mathbf 1\!\left\{
\log\frac{q_{\pi(i)}(w)}{p_{\pi(i)}(v^{(w)})}
\le
\log\frac{q_{\pi(i)}(v_{\pi(i)})}{p_{\pi(i)}(v)}
\right\}
q_{\pi(i)}(w)\,dw .
\]
Define, for this fixed \(v\in[0,1]^n\),
\[
\Gamma_{i,v}(w)
=
\log\frac{q_{\pi(i)}(w)}{p_{\pi(i)}(v^{(w)})},
\qquad
w\in[0,1].
\]
For every \(w\in[0,1]\), the point \(v^{(w)}\) remains in \([0,1]^n\).  Positivity and smoothness from \cref{obj:ass:positive-normalized-smooth-mechanisms} make \(\Gamma_{i,v}\) continuous on \([0,1]\) and differentiable in the interior, and its derivative is the own-coordinate log-ratio derivative at \(v^{(w)}\):
\[
\Gamma'_{i,v}(w)
=
\partial_{u}\nobreak\left.
\log\frac{q_{\pi(i)}(u)}{p_{\pi(i)}((v^{(w)})^{(u)})}
\right|_{u=w}.
\]
The fixed sign condition in \cref{obj:ass:fixed-own-derivative-sign} gives
\[
s_{\pi(i)}\,\Gamma'_{i,v}(w)>0,
\qquad 0<w<1.
\]
Thus \(\Gamma_{i,v}\) is strictly increasing on \([0,1]\) when \(s_{\pi(i)}=1\), and strictly decreasing on \([0,1]\) when \(s_{\pi(i)}=-1\).  Since
\[
\log\frac{q_{\pi(i)}(v_{\pi(i)})}{p_{\pi(i)}(v)}
=
\Gamma_{i,v}(v_{\pi(i)}),
\]
we have
\[
U_i(f(v))
=
\int_0^1
\mathbf 1\{\Gamma_{i,v}(w)\le \Gamma_{i,v}(v_{\pi(i)})\}
q_{\pi(i)}(w)\,dw .
\]
If \(s_{\pi(i)}=1\), strict increase makes the lower level set \([0,v_{\pi(i)}]\), so
\[
U_i(f(v))
=
\int_0^{v_{\pi(i)}}q_{\pi(i)}(w)\,dw
=
Q_{\pi(i)}(v_{\pi(i)}).
\]
If \(s_{\pi(i)}=-1\), strict decrease makes the lower level set \([v_{\pi(i)},1]\), and normalization of \(q_{\pi(i)}\) gives
\[
U_i(f(v))
=
\int_{v_{\pi(i)}}^1 q_{\pi(i)}(w)\,dw
=
1-Q_{\pi(i)}(v_{\pi(i)}).
\]
Thus, for every \(v\in[0,1]^n\),
\[
U_i(f(v))
=
\begin{cases}
Q_{\pi(i)}(v_{\pi(i)}), & s_{\pi(i)}=1,\\
1-Q_{\pi(i)}(v_{\pi(i)}), & s_{\pi(i)}=-1.
\end{cases}
\]
This proves the rank-coordinate clause. % lean: observedLawRankCoordinate_eq_of_equationEleven

\item The argument was for an arbitrary injective topological ordering \(\tau\) of \(H_D\) and an arbitrary \(i\in[n]\).  The five asserted clauses therefore hold for every such \(\tau\) and every \(i\). % lean: exactRatioDecoder_continuousRankClauses_of_order
\end{enumerate}
\end{proof}
