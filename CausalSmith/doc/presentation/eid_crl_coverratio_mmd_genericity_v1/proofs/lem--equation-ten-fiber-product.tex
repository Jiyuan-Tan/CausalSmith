\begin{proof}[Proof of \cref{obj:lem:equation-ten-fiber-product}]
Let \(i_\pi=\pi(i)\), and put \(A=A_i^\tau\).  For \(a\in[n]\), write
\[
v^{(w)}_a=
\begin{cases}
w, & a=i_\pi,\\
v_a, & a\ne i_\pi .
\end{cases}
\]

\begin{enumerate}
\item The intervention marginal factorization supplied by \cref{obj:lem:equation-ten-intervention-marginal} is used here in its ambient density-carrier form: for the arbitrary point \(v^{(w)}\in\mathbb R^n\),
\[
\begin{aligned}
&\int_{[0,1]^{[n]\setminus A}}
q_{i_\pi}\!\left(u^{v^{(w)},\tau}(r)_{i_\pi}\right)
\prod_{\ell\ne i_\pi}
p_\ell\!\left(u^{v^{(w)},\tau}(r)_\ell\mid
u^{v^{(w)},\tau}(r)_{\operatorname{pa}_G(\ell)}\right)\,dr  \\
&\qquad =
q_{i_\pi}\!\left(v^{(w)}_{i_\pi}\right)
\prod_{\ell\in A\setminus\{i_\pi\}}
p_\ell\!\left(v^{(w)}_\ell\mid
v^{(w)}_{\operatorname{pa}_G(\ell)}\right).
\end{aligned}
\]
The density carriers \(p_\ell\) and \(q_{i_\pi}\) are evaluated as their ambient functions on \(\mathbb R^n\) and \(\mathbb R\), respectively, so the coordinate replacement is admissible for every \(v\in\mathbb R^n\) and \(w\in\mathbb R\).  Since \(v^{(w)}_{i_\pi}=w\), the scalar factor on the right is \(q_{i_\pi}(w)\). % lean: equationTen_interventionMarginalDensity

\item It remains to identify the retained observational product after the coordinate replacement.  We claim that
\[
\prod_{\ell\in A\setminus\{i_\pi\}}
p_\ell\!\left(v^{(w)}_\ell\mid
v^{(w)}_{\operatorname{pa}_G(\ell)}\right)
=
\prod_{\ell\in A\setminus\{i_\pi\}}
p_\ell\!\left(v_\ell\mid v_{\operatorname{pa}_G(\ell)}\right).
\]
Fix \(\ell\in A\setminus\{i_\pi\}\).  By the definition
\[
A=\{i_\pi\}\cup \pi(B_i^\tau),
\qquad
B_i^\tau=\{j:\tau(j)<\tau(i)\},
\]
there is \(j\in B_i^\tau\) with \(\ell=\pi(j)\).  The own coordinate satisfies
\[
\ell\ne i_\pi,
\]
because \(\ell\) lies in \(A\setminus\{i_\pi\}\).  Now let \(a\in\operatorname{pa}_G(\ell)\).  If \(a=i_\pi\), then \(i_\pi\to \ell=\pi(j)\) is an edge of \(G\), so the transported-DAG order gives
\[
\tau(i)<\tau(j).
\]
But \(j\in B_i^\tau\) gives \(\tau(j)<\tau(i)\), a contradiction.  Hence
\[
a\ne i_\pi
\qquad\text{for every }a\in\operatorname{pa}_G(\ell).
\]
Thus \(v^{(w)}\) and \(v\) agree on \(\{\ell\}\cup\operatorname{pa}_G(\ell)\), and consequently
\[
p_\ell\!\left(v^{(w)}_\ell\mid
v^{(w)}_{\operatorname{pa}_G(\ell)}\right)
=
p_\ell\!\left(v_\ell\mid v_{\operatorname{pa}_G(\ell)}\right).
\]
Multiplying this equality over \(\ell\in A\setminus\{i_\pi\}\) proves the claimed product identity; when the index set is empty, both products are the empty product \(1\). % lean: equationTen_predecessorDensity_update_target

\item Substituting the product identity from the previous step into the marginal identity from the first step yields
\[
\begin{aligned}
&\int_{[0,1]^{[n]\setminus A_i^\tau}}
q_{\pi(i)}\!\left(u^{v^{(w)},\tau}(r)_{\pi(i)}\right)
\prod_{\ell\ne\pi(i)}
p_\ell\!\left(u^{v^{(w)},\tau}(r)_\ell\mid
u^{v^{(w)},\tau}(r)_{\operatorname{pa}_G(\ell)}\right)\,dr\\
&\qquad =
q_{\pi(i)}(w)
\prod_{\ell\in A_i^\tau\setminus\{\pi(i)\}}
p_\ell\!\left(v_\ell\mid v_{\operatorname{pa}_G(\ell)}\right),
\end{aligned}
\]
which is the asserted fiber product factorization. % lean: equationTen_interventionMarginalDensity_fiber_product
\end{enumerate}
\end{proof}
