\section{Appendix: proofs and verification}\label{sec:appendix}

This appendix collects the auxiliary statements that support the generic separation, population decoding, and sample-split confidence results. The first group records the explicit coordinate reflection and affine perturbation used to make a direct-edge ratio contrast nonzero inside a fixed-sign stratum.

\begin{definitionv}{synth_22}[Coordinate reflection map]
For \(\sigma,z\in\mathbb R\), define
\[
\operatorname{reflect}(\sigma,z)
=
\begin{cases}
z, & \sigma=1,\\
1-z, & \sigma\ne 1.
\end{cases}
\]
\end{definitionv}

\begin{definitionv}{def:affine-path}[Edge affine path]
For a finite labeled DAG \(G\) on \([n]\), a sign vector \(s\in\{-1,1\}^{[n]}\), a stratum point \(\theta=((p_\ell,q_\ell))_{\ell=1}^n\in\Theta_{G,s}\) as in \cref{obj:def:model-stratum}, an edge \(j\to i\) in \(G\), and \(t\in[0,1]\), define the reflected edge-specific sparse endpoint \(\theta^{\mathrm{sp}}=((p_\ell^{\mathrm{sp}},q_\ell^{\mathrm{sp}}))_{\ell=1}^n\) in \(G\) by
\[
p_\ell^{\mathrm{sp}}(v)=
\begin{cases}
1+\dfrac{1}{10}\,h(\operatorname{reflect}(s_j,v_j))\,h(\operatorname{reflect}(s_i,v_i)), & \ell=i,\\
1, & \ell\ne i,
\end{cases}
\qquad
q_\ell^{\mathrm{sp}}(z)=
\dfrac{4\exp(4\,\operatorname{reflect}(s_\ell,z))}{\exp(4)-1},
\]
where \(h(x)=2x-1\). The affine path \(\theta^t=((p_\ell^t,q_\ell^t))_{\ell=1}^n\) is the mechanism on \(G\) with
\[
p_\ell^t(v)=(1-t)p_\ell(v)+t\,p_\ell^{\mathrm{sp}}(v),
\qquad
q_\ell^t(z)=(1-t)q_\ell(z)+t\,q_\ell^{\mathrm{sp}}(z),
\qquad \ell\in[n].
\]
\end{definitionv}

\begin{definitionv}{synth_15}[Second-moment contrast]
For an observed world with environment laws \(P^0,\ldots,P^n\) and likelihood-ratio carriers \(R_i\), define, for \(j,i\in[n]\),
\[
F_{ji}
=
\int_{\mathcal X} R_i(x)^2\,dP^j(x)
-
\int_{\mathcal X} R_i(x)^2\,dP^0(x).
\]
\end{definitionv}

\begin{lemmav}{lem:analytic-edge-perturbation}[Analytic edge perturbation]
After the harmless target relabeling \(\pi=\operatorname{id}\), fix the ambient number \(n\) of scalar latent variables, a finite labeled DAG \(G\) on \([n]\), a sign vector \(s\in\{-1,1\}^{[n]}\), a mechanism \(\theta\in\Theta_{G,s}\) as in \cref{obj:def:model-stratum}, and an edge \(j\to i\) of \(G\). Assume that the identity-target canonical observed world generated by \(\theta\) satisfies \cref{obj:ass:one-perfect-intervention-per-node}. Let \(\theta^t\) be the edge-specific affine path of \cref{obj:def:affine-path}, and write \(F_{ji}(\eta)\) for the direct-edge second-moment contrast computed in the identity-target canonical observed world generated by a mechanism \(\eta\). Along this path,
\[
F_{ji}(\theta^t)
=
\int_{[0,1]^n}
\frac{(q_i^t(v_i))^2\bigl(q_j^t(v_j)-p_j^t(v_j\mid v_{\operatorname{pa}_G(j)})\bigr)
\prod_{\ell\ne i,j}p_\ell^t(v_\ell\mid v_{\operatorname{pa}_G(\ell)})}
{p_i^t(v_i\mid v_{\operatorname{pa}_G(i)})}
\,dv .
\]
There is an open set \(O\subseteq\mathbb R\) with \([0,1]\subseteq O\) such that:
\begin{itemize}
\item \textbf{(Integral identity.)} For every \(t\in O\), the displayed integral equals \(F_{ji}(\theta^t)\).
\item \textbf{(Analyticity.)} The map \(t\mapsto F_{ji}(\theta^t)\) is real analytic on a neighborhood of every point of \(O\).
\item \textbf{(Isolated zeros.)} If \(t\in O\) and \(F_{ji}(\theta^t)=0\), then there is \(\varepsilon>0\) such that every \(u\in(t-\varepsilon,t+\varepsilon)\) with \(u\ne t\) satisfies \(F_{ji}(\theta^u)\ne0\).
\end{itemize}
Moreover, \(F_{ji}(\theta^1)\ne0\). Finally, for every relative \(C^2\) neighborhood \(N\) of \(\theta\), there exists \(t\in(0,1)\) such that \(\theta^t\in N\), \(\theta^t\in\Theta_{G,s}\), and \(F_{ji}(\theta^t)\ne0\).
\end{lemmav}

The perturbation statement gives the local ingredient behind the density argument in \cref{obj:thm:generic-cover-separation}: the affine path stays in the same fixed-sign stratum near the starting mechanism and produces a nonzero direct-edge second-moment contrast. The companion zero statement records the population implication for labels whose transported targets have no ancestral relation, using the Gaussian mean-embedding discrepancy that defines the ratio graph in \cref{obj:def:population-decoder}.

\begin{lemmav}{lem:ratio-nonancestor-zero}[Nonancestor ratio discrepancy vanishes]
Fix an ambient number \(n\) of scalar latent variables, a finite labeled DAG \(G\) on \([n]\), a mechanism \(\theta\), and an observed world \(W\) with environment laws \(P^0,\ldots,P^n\), likelihood ratios \(R_i=dP^i/dP^0\), and target permutation \(\pi\).  Assume \cref{obj:ass:positive-normalized-smooth-mechanisms,obj:ass:shared-diffeomorphic-mixing,obj:ass:one-perfect-intervention-per-node}.  Let \(j,i\in[n]\) satisfy \(j\ne i\), and suppose that \(\pi(j)\) is not an ancestor of \(\pi(i)\) in \(G\).  Then the Gaussian-kernel discrepancy of ratio \(i\) between the observational environment and environment \(j\) vanishes:
\[
D_{ji}
=
\left\|
\int \phi(r)\,d\bigl((R_i)_{\#}P^0\bigr)(r)
-
\int \phi(r)\,d\bigl((R_i)_{\#}P^j\bigr)(r)
\right\|_{\mathcal H_k}
=0,
\]
where \(k(a,b)=\exp(-(a-b)^2)\) is the Gaussian kernel and \(\phi(a)=k(a,\cdot)\) is its feature map.
\end{lemmav}

The next statements isolate the triangular change of variables used by the conditional-rank decoder. They express predecessor log-ratios as coordinates for the already ordered predecessors and then factor the target intervention law conditional on those scores.

\begin{definitionv}{synth_29}[Triangular predecessor-score range and inverse]
Fix an observed world \(W\), a mechanism \(\theta\), and an injective topological order \(\tau:[n]\to\mathbb N\) for the observed ratio graph. For \(i\in[n]\), write
\[
B_i^\tau=\{j\in[n]:\tau(j)<\tau(i)\}
\]
for the predecessor labels of \(i\) in this order. The predecessor-score space is \(\mathbb R^{B_i^\tau}\), with coordinates indexed by \(j\in B_i^\tau\), and the predecessor latent cube is \([0,1]^{B_i^\tau}\).

For \(z\in[0,1]^{B_i^\tau}\), let \(\tilde z\in[0,1]^n\) be the latent vector that places \(z_j\) in coordinate \(\pi(j)\) for \(j\in B_i^\tau\) and fills all remaining coordinates by \(0\). The triangular predecessor-score map is
\[
\Lambda_i^\tau:[0,1]^{B_i^\tau}\longrightarrow \mathbb R^{B_i^\tau},
\qquad
\bigl(\Lambda_i^\tau(z)\bigr)_j
=
\log\frac{q_{\pi(j)}(\tilde z_{\pi(j)})}{p_{\pi(j)}(\tilde z)}
\quad (j\in B_i^\tau).
\]
Equivalently, when \(v\in[0,1]^n\) and \(z_j=v_{\pi(j)}\) for \(j\in B_i^\tau\),
\[
\Lambda_i^\tau(z)
=
\bigl(L_j(f(v))\bigr)_{j\in B_i^\tau}.
\]

The realized predecessor-score range is the image
\[
\mathcal R_i^\tau=\Lambda_i^\tau\bigl([0,1]^{B_i^\tau}\bigr)\subseteq \mathbb R^{B_i^\tau}.
\]
Under the positive smooth mechanism, shared mixing, one-intervention-per-node, and fixed own-derivative-sign conditions used in \(\cref{obj:lem:equation-eleven-kernel-factorization}\), \(\Lambda_i^\tau\) is injective and hence is a homeomorphism from \([0,1]^{B_i^\tau}\) onto \(\mathcal R_i^\tau\). Its inverse is written
\[
(\Lambda_i^\tau)^{-1}:\mathcal R_i^\tau\longrightarrow[0,1]^{B_i^\tau},
\]
so that for \(\ell\in\mathcal R_i^\tau\), \((\Lambda_i^\tau)^{-1}(\ell)\) is the unique predecessor latent vector \(z\in[0,1]^{B_i^\tau}\) with \(\Lambda_i^\tau(z)=\ell\), and
\[
\Lambda_i^\tau\circ(\Lambda_i^\tau)^{-1}=\operatorname{id}_{\mathcal R_i^\tau},
\qquad
(\Lambda_i^\tau)^{-1}\circ\Lambda_i^\tau=\operatorname{id}_{[0,1]^{B_i^\tau}}.
\]
The range \(\mathcal R_i^\tau\) is the continuous image of a compact cube, hence a nonempty compact Borel subset of \(\mathbb R^{B_i^\tau}\). Formulas indexed by an arbitrary score vector are totalized through a fixed Borel retraction
\[
\rho_i^\tau:\mathbb R^{B_i^\tau}\longrightarrow\mathcal R_i^\tau,
\qquad
\rho_i^\tau(\ell)=\ell\ \ (\ell\in\mathcal R_i^\tau),
\]
which sends every \(\ell\notin\mathcal R_i^\tau\) to one fixed point of \(\mathcal R_i^\tau\). Writing \(\rho_i^\tau\) for this retraction and \((\Lambda_i^\tau)^{-1}\) for the inverse keeps the two maps distinct: the composition \((\Lambda_i^\tau)^{-1}\circ\rho_i^\tau\) is defined on all of \(\mathbb R^{B_i^\tau}\), and off \(\mathcal R_i^\tau\) the convention carries no content, since every continuity and support claim is asserted on \(\mathcal R_i^\tau\).
\end{definitionv}

\begin{lemmav}{lem:predecessor-score-homeomorphism}[Predecessor score homeomorphism]
Let \(G\) be a DAG on \([n]\), let \(s\) be a sign vector on \([n]\), let \(\theta\) be a mechanism on \(G\), and let \(W\) be an observed world with observed laws \(P^0,\ldots,P^n\), likelihood ratios \(R_i\), shared observation map \(f\), and target permutation \(\pi\). Assume:
\begin{itemize}
\item \textbf{(Regular mechanism and observation model.)} The hypotheses of \cref{obj:ass:positive-normalized-smooth-mechanisms,obj:ass:shared-diffeomorphic-mixing,obj:ass:one-perfect-intervention-per-node,obj:ass:fixed-own-derivative-sign} hold for \(G,\theta,W\), with prescribed sign vector \(s\).
\item \textbf{(Observed ratio ordering.)} The map \(\tau:[n]\to\mathbb N\) is injective and is a topological ordering of the observable ratio-discrepancy graph \(H_D=\{j\to k:D_{jk}>0\}\) constructed from the observed laws with the Gaussian feature map in \cref{obj:def:population-decoder}: whenever \(D_{jk}>0\) and \(j\ne k\), \(\tau(j)<\tau(k)\).
\item \textbf{(Transported latent ordering.)} The same order also respects the transported latent DAG: whenever \(\pi(j)\to\pi(k)\) is an edge of \(G\), \(\tau(j)<\tau(k)\).
\end{itemize}
Fix \(i\in[n]\), set \(B_i^\tau=\{b:\tau(b)<\tau(i)\}\), and let
\[
\mathcal Z_i^\tau=\prod_{b\in B_i^\tau}[0,1].
\]
For \(\chi\in\mathcal Z_i^\tau\), let \(E_i^\tau(\chi)\in[0,1]^n\) be the zero-filled latent point with \(E_i^\tau(\chi)_{\pi(b)}=\chi_b\) for \(b\in B_i^\tau\) and all other coordinates equal to zero. Define \(\Lambda_i^\tau:\mathcal Z_i^\tau\to\mathbb R^{B_i^\tau}\) by
\[
\Lambda_i^\tau(\chi)_b
=
\log
\frac{
q_{\pi(b)}\!\left(E_i^\tau(\chi)_{\pi(b)}\right)
}{
p_{\pi(b)}\!\left(E_i^\tau(\chi)_{\pi(b)}\mid E_i^\tau(\chi)_{\operatorname{pa}_G(\pi(b))}\right)
},
\qquad b\in B_i^\tau,
\]
and write \(\mathcal R_i^\tau=\Lambda_i^\tau(\mathcal Z_i^\tau)\). Then \(\Lambda_i^\tau\) is a homeomorphism from \(\mathcal Z_i^\tau\) onto \(\mathcal R_i^\tau\). In particular, the inverse map \((\Lambda_i^\tau)^{-1}:\mathcal R_i^\tau\to\mathcal Z_i^\tau\) is continuous.
\end{lemmav}

The homeomorphism clause supplies the support-level continuity needed for conditional distribution transforms: predecessor scores identify predecessor latent coordinates on the realized range. The following two marginal identities then remove coordinates outside the retained ancestral set while preserving the factors that enter the target log-ratio.

\begin{lemmav}{lem:equation-ten-intervention-marginal}[Intervention marginal identity]
Fix \(i\in[n]\). Suppose that:
\begin{itemize}
\item \textbf{(Regular mechanisms.)} The mechanism satisfies \cref{obj:ass:positive-normalized-smooth-mechanisms}.
\item \textbf{(Ratio-graph order.)} The map \(\tau:[n]\to\mathbb N\) is injective and is a topological ordering of the observable ratio-discrepancy graph \(H_D\) in \cref{obj:def:population-decoder}: whenever \(D_{ji}>0\), \(\tau(j)<\tau(i)\).
\item \textbf{(Transported-DAG order.)} The same order respects the transported latent DAG: whenever \(\pi(j)\to\pi(i)\) is an edge of \(G\), \(\tau(j)<\tau(i)\).
\end{itemize}
Let
\[
B_i^\tau=\{j\in[n]:\tau(j)<\tau(i)\},
\qquad
A_i^\tau=\{\pi(i)\}\cup\pi(B_i^\tau).
\]
For \(v\in[0,1]^n\) and \(r\in[0,1]^{[n]\setminus A_i^\tau}\), let \(u^{v,\tau}(r)\in[0,1]^n\) be the point with
\[
u^{v,\tau}(r)_a=v_a \quad(a\in A_i^\tau),
\qquad
u^{v,\tau}(r)_a=r_a \quad(a\notin A_i^\tau).
\]
Then, for every \(v\in[0,1]^n\),
\[
\int_{[0,1]^{[n]\setminus A_i^\tau}}
q_{\pi(i)}\!\left(u^{v,\tau}(r)_{\pi(i)}\right)
\prod_{\ell\ne \pi(i)}
p_\ell\!\left(u^{v,\tau}(r)_\ell\mid
u^{v,\tau}(r)_{\operatorname{pa}_G(\ell)}\right)\,dr
=
q_{\pi(i)}(v_{\pi(i)})
\prod_{\ell\in A_i^\tau\setminus\{\pi(i)\}}
p_\ell\!\left(v_\ell\mid v_{\operatorname{pa}_G(\ell)}\right).
\]
\end{lemmav}

\begin{lemmav}{lem:equation-ten-fiber-product}[Fiber product factorization]
In the setting of \cref{obj:ass:positive-normalized-smooth-mechanisms}, fix an observed world with target permutation \(\pi\).  Let \(\tau\) be a numerical order on \([n]\), and fix \(i\in[n]\).  Assume:
\begin{itemize}
\item \textbf{(Ratio-graph order.)} The order \(\tau\) is injective and topological for the observable ratio graph \(H_D=\{j\to i:j\ne i,\ D_{ji}>0\}\) of \cref{obj:def:population-decoder}: whenever \(D_{ji}>0\), one has \(\tau(j)<\tau(i)\).
\item \textbf{(Transported-DAG order.)} The same order respects every transported latent edge: whenever \(\pi(j)\to\pi(i)\) is an edge of \(G\), one has \(\tau(j)<\tau(i)\).
\end{itemize}
Write
\[
B_i^\tau=\{j:\tau(j)<\tau(i)\},
\qquad
A_i^\tau=\{\pi(i)\}\cup \pi(B_i^\tau).
\]
For \(x\in\mathbb R^n\) and \(r\in[0,1]^{[n]\setminus A_i^\tau}\), let \(u^{x,\tau}(r)\) be the latent point whose coordinates in \(A_i^\tau\) agree with \(x\) and whose remaining coordinates are filled by \(r\).  For every \(v\in\mathbb R^n\) and every \(w\in\mathbb R\), if \(v^{(w)}\) denotes \(v\) with its \(\pi(i)\)-coordinate replaced by \(w\), then
\[
\int_{[0,1]^{[n]\setminus A_i^\tau}}
q_{\pi(i)}\!\left(u^{v^{(w)},\tau}(r)_{\pi(i)}\right)
\prod_{\ell\ne\pi(i)}
p_\ell\!\left(u^{v^{(w)},\tau}(r)_\ell\mid
u^{v^{(w)},\tau}(r)_{\operatorname{pa}_G(\ell)}\right)\,dr
=
q_{\pi(i)}(w)
\prod_{\ell\in A_i^\tau\setminus\{\pi(i)\}}
p_\ell\!\left(v_\ell\mid v_{\operatorname{pa}_G(\ell)}\right).
\]
\end{lemmav}

\begin{lemmav}{lem:equation-eleven-kernel-factorization}[Kernel factorization of scores]
Let \(n\) be fixed, let \(G\) be a finite labeled DAG on \([n]\), let \(s\) be a sign vector with entries in \(\{-1,1\}\), and let \(W\) be an observed world generated by a mechanism \(\theta\). Assume:
\begin{itemize}
\item \textbf{(Structural regularity.)} The mechanism and observed world satisfy \cref{obj:ass:positive-normalized-smooth-mechanisms,obj:ass:shared-diffeomorphic-mixing,obj:ass:one-perfect-intervention-per-node,obj:ass:fixed-own-derivative-sign} with sign vector \(s\).
\item \textbf{(Observed ratio order.)} The map \(\tau:[n]\to\mathbb N\) is injective and is a topological ordering of the observed Gaussian-MMD ratio graph \(H_D\) of \cref{obj:def:population-decoder}: whenever \(H_D\) contains \(j\to k\), \(\tau(j)<\tau(k)\).
\item \textbf{(Transported graph order.)} The same order respects the transported latent DAG: whenever \(\pi(j)\to\pi(k)\) is an edge of \(G\), \(\tau(j)<\tau(k)\).
\end{itemize}
Fix \(i\in[n]\), and put
\[
B_i^\tau=\{j\in[n]:\tau(j)<\tau(i)\}.
\]
For each predecessor log-ratio vector \(\ell\in\mathbb R^{B_i^\tau}\), let \(\kappa_i^\tau(\ell,\cdot)\) be the law on \(\mathbb R\) of
\[
\log\frac{q_{\pi(i)}(W_i)}{p_{\pi(i)}\!\left(\widetilde v_i^\tau(z_i^\tau(\ell),W_i)\right)},
\qquad W_i\sim q_{\pi(i)}(w)\,dw \text{ on }[0,1],
\]
where \(z_i^\tau=(\Lambda_i^\tau)^{-1}\circ\rho_i^\tau\) is the triangular predecessor-score inverse composed with the retraction onto the realized score range \(\mathcal R_i^\tau\), and \(\widetilde v_i^\tau(z,w)\) is the latent point whose \(\pi(j)\)-coordinate is \(z_j\) for \(j\in B_i^\tau\), whose \(\pi(i)\)-coordinate is \(w\), and whose remaining coordinates are zero. Then, under the target-\(\pi(i)\) observed environment law \(P^i\), the joint law of the predecessor log-ratios and the target log-ratio factorizes as
\[
(L_{B_i^\tau},L_i)_{\#}P^i
=
(L_{B_i^\tau})_{\#}P^i\otimes \kappa_i^\tau .
\]
\end{lemmav}

Together, these identities turn the target intervention environment into a product between predecessor scores and a one-dimensional kernel indexed by those scores. The continuous-rank clauses below state the corresponding conditional-CDF representation and the resulting rank coordinate used by the population decoder.

\begin{lemmav}{lem:exact-ratio-decoder-continuous-rank-clauses}[Continuous rank decoder clauses]
Let \(n\) be the ambient number of scalar latent variables, let \(G\) be the ambient finite labeled DAG on \([n]\), let \(\theta\) be the mechanism, let \(W\) be the observed world with shared observation map \(f\), observed laws \(P^0,\ldots,P^n\), likelihood ratios \(R_i=dP^i/dP^0\), log ratios \(L_i=\log R_i\), and target permutation \(\pi\), and let \(s\) be a sign vector with \(s_i\in\{-1,1\}\) for every \(i\). Assume:
\begin{itemize}
\item \textbf{(Mechanism regularity.)} The mechanism satisfies \cref{obj:ass:positive-normalized-smooth-mechanisms}.
\item \textbf{(Shared mixing.)} The observed world satisfies \cref{obj:ass:shared-diffeomorphic-mixing}.
\item \textbf{(Single-target interventions.)} The observed laws and likelihood ratios satisfy \cref{obj:ass:one-perfect-intervention-per-node}.
\item \textbf{(Own-derivative signs.)} The mechanism satisfies \cref{obj:ass:fixed-own-derivative-sign} with sign vector \(s\).
\item \textbf{(Ratio-order recovery.)} For the observable directed ratio-discrepancy graph \(H_D\) of \cref{obj:def:population-decoder}, built from the Gaussian kernel \(k(a,b)=\exp(-(a-b)^2)\), the transitive closures agree:
\[
\operatorname{TC}(H_D)=\operatorname{TC}(G^\pi).
\]
\end{itemize}
For any injective topological ordering \(\tau\) of \(H_D\), any \(i\in[n]\), and \(B_i^\tau=\{j:\tau(j)<\tau(i)\}\), call a function \(C\) a continuous conditional-CDF version of \(L_i\) given \(L_{B_i^\tau}\) when \(C\) is jointly Borel measurable, for every fixed \(t\in\mathbb R\) it represents the regular conditional probability \(P^i(L_i\le t\mid L_{B_i^\tau}=\ell)\) for \((L_{B_i^\tau})_{\#}P^i\)-almost every \(\ell\), the same identity holds for \((L_i,L_{B_i^\tau})_{\#}P^i\)-almost every \((t,\ell)\), and \(C\) is continuous on the observed conditional-ratio support
\[
\mathcal S_i^\tau=\mathbb R\times\operatorname{supp}\bigl((L_{B_i^\tau})_{\#}P^i\bigr).
\]
Then, for every such \(\tau\) and every \(i\in[n]\), with \(C_i\) denoting the law-selected conditional CDF used by \cref{obj:def:population-decoder}, the following hold:
\begin{itemize}
\item \(C_i\) is a continuous conditional-CDF version of \(L_i\) given \(L_{B_i^\tau}\).
\item There exists a continuous conditional-CDF version \(C\) such that, for all \(t\in\mathbb R\) and all \(v\in[0,1]^n\),
\[
C\bigl(t\mid L_{B_i^\tau}(f(v))\bigr)
=
\int_0^1
\mathbf 1\!\left\{\log\frac{q_{\pi(i)}(w)}
{p_{\pi(i)}\bigl(v^{(w)}\bigr)}\le t\right\}
q_{\pi(i)}(w)\,dw,
\]
where \(v^{(w)}\) is the latent point obtained from \(v\) by replacing coordinate \(\pi(i)\) by \(w\).
\item Every continuous conditional-CDF version \(C\) agrees with \(C_i\) on \(\mathcal S_i^\tau\):
\[
C(z_1\mid z_2)=C_i(z_1\mid z_2)
\qquad\text{for every }(z_1,z_2)\in\mathcal S_i^\tau.
\]
\item For all \(t\in\mathbb R\) and all \(v\in[0,1]^n\),
\[
C_i\bigl(t\mid L_{B_i^\tau}(f(v))\bigr)
=
\int_0^1
\mathbf 1\!\left\{\log\frac{q_{\pi(i)}(w)}
{p_{\pi(i)}\bigl(v^{(w)}\bigr)}\le t\right\}
q_{\pi(i)}(w)\,dw.
\]
\item The rank coordinate \(U_i=C_i(L_i\mid L_{B_i^\tau})\) returned by \cref{obj:def:population-decoder} satisfies, for every \(v\in[0,1]^n\),
\[
U_i(f(v))
=
\begin{cases}
Q_{\pi(i)}(v_{\pi(i)}), & s_{\pi(i)}=1,\\
1-Q_{\pi(i)}(v_{\pi(i)}), & s_{\pi(i)}=-1,
\end{cases}
\]
where \(Q_{\pi(i)}(x)=\int_0^x q_{\pi(i)}(u)\,du\).
\end{itemize}
\end{lemmav}

The last auxiliary bound supports the simultaneous confidence graph in \cref{obj:thm:simultaneous-confidence-edges}. It is a Hilbert-space empirical-mean concentration statement for bounded Gaussian-kernel feature maps, matching the kernel mean-embedding discrepancy used in maximum mean discrepancy testing \citep{GrettonEtAl2012KernelTwoSample,SchrabEtAl2023MMDAgg}.

\begin{lemmav}{lem:bounded-rkhs-empirical-mean}[Conditional RKHS mean bound]
Fix \(n\ge 1\), a labeled DAG \(G\) on \([n]\), a mechanism \(\theta\), an observed world with environment laws \(P^0,\ldots,P^n\), and a sample-split experiment \(S\). Let \(\mathcal H_k\) be a complete real Hilbert space equipped with a measurable unit-norm feature map \(\phi\) for the Gaussian kernel \(k(a,b)=\exp(-(a-b)^2)\).

Assume:
\begin{itemize}
\item \textbf{(Conditional evaluation sampling.)} The experiment satisfies the conditional evaluation-sampling conditions: the ambient and environment laws are probability measures; the complete training fold is measurable; the evaluation observations are measurable, independent within each environment, distributed according to their corresponding \(P^e\), and independent of the training fold; and the fitted ratio map is jointly measurable in the training fold and the evaluation point.
\item \textbf{(Evaluation sizes.)} Each environment has at least one evaluation observation, \(N_e\ge 1\) for every \(e\in\{0,\ldots,n\}\). Write
\[
\underline N=\min_{0\le e\le n}N_e .
\]
\item \textbf{(Familywise level.)} The error level satisfies \(\alpha\in(0,1)\).
\end{itemize}

For each training realization, let \(\widehat R_i\) denote the fitted ratio obtained from the complete training fold, and define
\[
\widehat \mu_{e i}
=
\frac{1}{N_e}\sum_{r=1}^{N_e}\phi\!\left(\widehat R_i(X_{er})\right),
\qquad
\mu_{e i}
=
\mathbb E_{X\sim P^e}\!\left[\phi\!\left(\widehat R_i(X)\right)\right].
\]
Then, conditionally on the complete training fold, the simultaneous event
\[
\forall i\in[n]\ \forall e\in\{0,\ldots,n\},\qquad
\left\|\widehat \mu_{e i}-\mu_{e i}\right\|_{\mathcal H_k}
\le
\frac{1+\sqrt{2\log\!\left(n(n+1)/\alpha\right)}}{\sqrt{\underline N}}
\]
has probability at least \(1-\alpha\). Equivalently, for every event \(B\) measurable with respect to the complete training fold,
\[
(1-\alpha)\mathbb P(B)
\le
\mathbb P\!\left(
B\cap
\left\{
\forall i\in[n]\ \forall e\in\{0,\ldots,n\},\
\left\|\widehat \mu_{e i}-\mu_{e i}\right\|_{\mathcal H_k}
\le
\frac{1+\sqrt{2\log\!\left(n(n+1)/\alpha\right)}}{\sqrt{\underline N}}
\right\}
\right).
\]
\end{lemmav}
