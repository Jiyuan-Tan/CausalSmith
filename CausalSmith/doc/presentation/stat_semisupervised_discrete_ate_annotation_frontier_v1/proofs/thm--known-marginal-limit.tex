\begin{proof}[Proof of \cref{obj:thm:known-marginal-limit}]
Let
\[
\mathcal Z_n:=([d]\times\{0,1\}^2)^n,
\qquad
\mathcal C_d:=[d]\times\{0,1\}.
\]
We use the compact coordinate space
\[
\Theta_{\epsilon,d}
=
\Bigl\{(\pi,\varrho,\omega):
\pi\in\mathsf{Prob}([d]),\
\varrho_x\in[\epsilon,1-\epsilon],\
\omega_{\alpha x}\in[0,1]\Bigr\}.
\]
The assumptions \(d\ge2\) and \(0<\epsilon<1/2\) make this space nonempty.  For
\(\theta=(\pi,\varrho,\omega)\in\Theta_{\epsilon,d}\), set
\[
P_\theta(x,\alpha,y)
=
\pi_x\varrho_x^\alpha(1-\varrho_x)^{1-\alpha}
\omega_{\alpha x}^y(1-\omega_{\alpha x})^{1-y},
\]
and define
\[
\lambda_\theta(z)
=
\prod_{i=1}^n P_\theta(z_i),
\qquad
\xi_\theta(x,\alpha)
=
\sum_{y\in\{0,1\}}P_\theta(x,\alpha,y),
\qquad
\upsilon_\theta
=
\sum_{x\in[d]}\pi_x(\omega_{1x}-\omega_{0x}).
\]
Also,
\[
-1
=
\sum_x\pi_x(-1)
\le
\sum_x\pi_x(\omega_{1x}-\omega_{0x})
\le
\sum_x\pi_x(1)
=
1,
\]
so \(\upsilon_\theta\in[-1,1]\) for every \(\theta\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.knownMarginalTarget_mem

\begin{enumerate}
\item\label{pf:known-marginal-limit:finite-side} Define the finite-side values
\[
\mathfrak R_m
=
\inf_{\varphi:\mathcal Z_n\times\mathcal C_d^m\to[-1,1]}
\sup_{\theta\in\Theta_{\epsilon,d}}
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\sum_{u\in\mathcal C_d^m}
\Bigl(\prod_{j=1}^m\xi_\theta(u_j)\Bigr)
\bigl(\varphi(z,u)-\upsilon_\theta\bigr)^2
\]
and
\[
\mathfrak R_\star
=
\inf_{\psi:\mathcal Z_n\times\mathsf{Prob}(\mathcal C_d)\to[-1,1]}
\sup_{\theta\in\Theta_{\epsilon,d}}
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\bigl(\psi(z,\xi_\theta)-\upsilon_\theta\bigr)^2 .
\]
With the preceding coordinates, \cref{obj:lem:finite-side-information-convergence} gives
\[
\lim_{m\to\infty}\mathfrak R_m=\mathfrak R_\star .
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.knownMarginalGeneric_tendsto

\item\label{pf:known-marginal-limit:ambient-exact} Let \(\widetilde{\mathfrak R}_\star\) be the measurable ambient exact-table value
\[
\widetilde{\mathfrak R}_\star
=
\inf_{\psi}
\sup_{\theta\in\Theta_{\epsilon,d}}
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\bigl(\psi(z,\xi_\theta)-\upsilon_\theta\bigr)^2,
\]
where the infimum ranges over all rules
\(\psi:\mathcal Z_n\times\mathbb R^{\mathcal C_d}\to[-1,1]\) whose section
\(w\mapsto\psi(z,w)\) is Borel measurable for each \(z\).  The target range established above gives
\(\upsilon_\theta\in[-1,1]\), and \cref{obj:lem:finite-side-information-convergence} supplies the convergence
\[
\lim_{m\to\infty}\mathfrak R_m=\mathfrak R_\star .
\]
The measurable ambient exact-table value is tied to this limiting sequence by two comparisons.  The exact-side value infimizes over all rules on \(\mathsf{Prob}(\mathcal C_d)\), while the ambient value infimizes over rules on \(\mathbb R^{\mathcal C_d}\) with Borel-measurable sections.  Restricting any ambient rule to \(\mathsf{Prob}(\mathcal C_d)\) gives an admissible exact-side rule, so the exact-side class is at least as large after restriction, and therefore
\[
\mathfrak R_\star\le \widetilde{\mathfrak R}_\star .
\]
Conversely, \cref{obj:lem:finite-side-measurable-comparison} gives
\(\widetilde{\mathfrak R}_\star\le\mathfrak R_m\) for every \(m\ge0\).
For completeness, its measurable conditional-average construction in the
present coordinates is as follows.  Fix an integer \(m\ge0\) and an empirical-side rule
\(\varphi:\mathcal Z_n\times\mathcal C_d^m\to[-1,1]\).  Define first the polynomial conditional average on the ambient table space by
\[
\psi_\varphi^\circ(z,w)
=
\sum_{u\in\mathcal C_d^m}
\Bigl(\prod_{j=1}^m w(u_j)\Bigr)\varphi(z,u),
\qquad
w\in\mathbb R^{\mathcal C_d}.
\]
This function is Borel measurable in \(w\) for each fixed \(z\).  Use its clipped version as the ambient exact-table rule,
\[
\psi_\varphi(z,w)
=
\min\{1,\max\{-1,\psi_\varphi^\circ(z,w)\}\},
\qquad
w\in\mathbb R^{\mathcal C_d}.
\]
Then \(\psi_\varphi:\mathcal Z_n\times\mathbb R^{\mathcal C_d}\to[-1,1]\) is admissible for \(\widetilde{\mathfrak R}_\star\).  When \(w=r\in\mathsf{Prob}(\mathcal C_d)\), the coefficients \(\prod_{j=1}^m r(u_j)\) form a probability vector on \(\mathcal C_d^m\), and since \(\varphi\) takes values in \([-1,1]\), the polynomial average already lies in \([-1,1]\).  Hence, on probability tables,
\[
\psi_\varphi(z,r)
=
\sum_{u\in\mathcal C_d^m}
\Bigl(\prod_{j=1}^m r(u_j)\Bigr)\varphi(z,u).
\]
For every \(\theta\) and \(z\), Jensen's inequality for the product law \(\xi_\theta^{\otimes m}\) gives
\[
\bigl(\psi_\varphi(z,\xi_\theta)-\upsilon_\theta\bigr)^2
=
\left(
\sum_{u\in\mathcal C_d^m}
\Bigl(\prod_{j=1}^m\xi_\theta(u_j)\Bigr)
\bigl(\varphi(z,u)-\upsilon_\theta\bigr)
\right)^2
\le
\sum_{u\in\mathcal C_d^m}
\Bigl(\prod_{j=1}^m\xi_\theta(u_j)\Bigr)
\bigl(\varphi(z,u)-\upsilon_\theta\bigr)^2 .
\]
Multiplying by \(\lambda_\theta(z)\), summing over \(z\), and then taking the supremum over \(\theta\) gives
\[
\sup_{\theta}
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\bigl(\psi_\varphi(z,\xi_\theta)-\upsilon_\theta\bigr)^2
\le
\sup_{\theta}
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\sum_{u\in\mathcal C_d^m}
\Bigl(\prod_{j=1}^m\xi_\theta(u_j)\Bigr)
\bigl(\varphi(z,u)-\upsilon_\theta\bigr)^2 .
\]
Taking the infimum over \(\varphi\) yields
\[
\widetilde{\mathfrak R}_\star\le \mathfrak R_m
\qquad\text{for every }m\ge0 .
\]
Taking the limit along the displayed convergence yields
\[
\widetilde{\mathfrak R}_\star\le \lim_{m\to\infty}\mathfrak R_m=\mathfrak R_\star .
\]
The two inequalities identify the ambient measurable exact-table value with the exact-side value:
\[
\widetilde{\mathfrak R}_\star=\mathfrak R_\star .
\]
% lean: Causalean.Stat.Minimax.FiniteSideInformation.measurableExactTableMinimaxValue_eq_of_tendsto

\item\label{pf:known-marginal-limit:known-risk} We use here the standing
theorem hypotheses \(d\ge2\) and \(0<\epsilon<1/2\), which are also the
parameter-range clauses in the definition of \(\mathcal M_{d,\epsilon}\).
Under these hypotheses, the compact coordinates parametrize the laws in
\(\mathcal M_{d,\epsilon}\) from \cref{obj:def:model-class}.  For every \(\theta\in\Theta_{\epsilon,d}\), the law \(P_\theta\) satisfies occupied-cell overlap because, whenever \(\pi_x>0\), its treatment propensity is
\[
\frac{\sum_y P_\theta(x,1,y)}{\sum_{\alpha,y}P_\theta(x,\alpha,y)}
=\varrho_x\in[\epsilon,1-\epsilon].
\]
Conversely, given any \(P\in\mathcal M_{d,\epsilon}\), choose coordinates by
\[
\pi_x=p_x,
\qquad
\varrho_x=
\begin{cases}
 e_x, & p_x>0,\\
 \epsilon, & p_x=0,
\end{cases}
\qquad
\omega_{\alpha x}=\mu_{\alpha x}.
\]
The value of \(\varrho_x\) in an empty cell is immaterial because every atom in that cell is multiplied by \(\pi_x=0\).  For occupied cells, the overlap condition places \(\varrho_x\) in \([\epsilon,1-\epsilon]\), and the binary outcome regressions place \(\omega_{\alpha x}\) in \([0,1]\).  These coordinates recover the atom probabilities,
\[
P(x,\alpha,y)
=
\pi_x\varrho_x^\alpha(1-\varrho_x)^{1-\alpha}
\omega_{\alpha x}^y(1-\omega_{\alpha x})^{1-y},
\]
with the empty-cell case giving zero on both sides.  Thus the map \(\theta\mapsto P_\theta\) is surjective onto the model class.

Under this parametrization, the three quantities entering the known-marginal risk are transported by
\[
P_\theta^{\otimes n}(z)=\lambda_\theta(z),
\qquad
(P_\theta)_{XA}(x,\alpha)=\xi_\theta(x,\alpha),
\qquad
\tau(P_\theta)=\upsilon_\theta .
\]
The last identity follows from \cref{obj:def:ate-functional}: on occupied cells the outcome regressions are \(\omega_{1x}\) and \(\omega_{0x}\), while empty cells have coefficient \(\pi_x=0\).  Hence, for every bounded measurable known-table rule \(\psi\),
\[
\int
\bigl(\psi(Z,(P_\theta)_{XA})-\tau(P_\theta)\bigr)^2
\,dP_\theta^{\otimes n}(Z)
=
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\bigl(\psi(z,\xi_\theta)-\upsilon_\theta\bigr)^2 .
\]
The known-table sample space in \cref{obj:def:known-marginal-risk} is
\(\mathcal Z_n\times\mathbb R^{\mathcal C_d}\).  Its bounded measurable
procedures are in bijection with the ambient procedures used in
\(\widetilde{\mathfrak R}_\star\).  Explicitly, a paper procedure \(f\)
is sent to
\[
\psi_f(z,w):=f(z,w),
\]
and an ambient procedure \(\psi\) is sent back to
\(f_\psi(z,w):=\psi(z,w)\).  Because \(\mathcal Z_n\) is finite and
discrete, measurability of every section \(w\mapsto\psi(z,w)\) is
equivalent to measurability of the uncurried map; the two assignments are
pointwise inverse and preserve the range \([-1,1]\).

For corresponding \(f\) and \(\psi_f\), evaluation at the known table
\(w=(P_\theta)_{XA}=\xi_\theta\), followed by expansion of the finite
labeled product law, gives
\[
\begin{aligned}
&\int
\bigl(f(Z,(P_\theta)_{XA})-\tau(P_\theta)\bigr)^2
\,dP_\theta^{\otimes n}(Z)\\
&\qquad =
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\bigl(\psi_f(z,\xi_\theta)-\upsilon_\theta\bigr)^2.
\end{aligned}
\]
Thus the estimator bijection preserves every coordinate risk exactly.
Reindexing the supremum over \(P\in\mathcal M_{d,\epsilon}\) by the
surjective coordinate map, and then taking the same infimum over the
bijective procedure classes, gives
\[
R_\epsilon^{\mathrm{known}}(n,d)
=
\widetilde{\mathfrak R}_\star .
\]
Together with the equality \(\widetilde{\mathfrak R}_\star=\mathfrak R_\star\) above, this gives
\[
R_\epsilon^{\mathrm{known}}(n,d)=\mathfrak R_\star .
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.knownMarginalRisk_eq_measurableValue

\item\label{pf:known-marginal-limit:empirical-risk} For each \(m\), the minimax risk in \cref{obj:def:minimax-risk} equals the empirical finite-side value:
\[
R_\epsilon(n,m,d)=\mathfrak R_m .
\]
First reindex the supremum over \(P\in\mathcal M_{d,\epsilon}\) by the surjective map \(\theta\mapsto P_\theta\).  The annotation sample space is the
finite product \(\mathcal Z_n\times\mathcal C_d^m\), so every real-valued
map on it is measurable.  For any measurable estimator \(T\) in the annotation experiment, write its two-sample mean-squared risk at \(P_\theta\) as
\[
\mathcal V_T(\theta)
=
\int
\bigl(T(Z,U)-\tau(P_\theta)\bigr)^2
\,d\bigl(P_\theta^{\otimes n}\otimes (P_\theta)_{XA}^{\otimes m}\bigr)(Z,U).
\]
Expanding both finite product measures, and using
\((P_\theta)_{XA}=\xi_\theta\),
\(P_\theta^{\otimes n}=\lambda_\theta\), and
\(\tau(P_\theta)=\upsilon_\theta\), gives the exact table identity
\[
\mathcal V_{T_\varphi}(\theta)
=
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\sum_{u\in\mathcal C_d^m}
\Bigl(\prod_{j=1}^m\xi_\theta(u_j)\Bigr)
\bigl(\varphi(z,u)-\upsilon_\theta\bigr)^2
\]
for \(T_\varphi(z,u):=\varphi(z,u)\).

Thus every empirical-side procedure \(\varphi:\mathcal Z_n\times\mathcal C_d^m\to[-1,1]\) induces the paper estimator
\[
T_\varphi(z,u)=\varphi(z,u),
\]
viewed as an estimator in the annotation experiment.  For this induced estimator, the preceding expansion gives equality of the coordinate risks for every \(\theta\), hence equality of the corresponding worst-case risks after the reindexing by \(\theta\).  Since the paper infimum ranges over all measurable estimators, including these induced estimators, taking the infimum over \(\varphi\) gives
\[
R_\epsilon(n,m,d)\le \mathfrak R_m .
\]
For the reverse inequality, let \(T\) be any measurable paper estimator and define the clipped empirical-side rule
\[
\varphi_T(z,u)
=
\min\{1,\max\{-1,T(z,u)\}\}.
\]
The transported identity \(\tau(P_\theta)=\upsilon_\theta\), together with
the target-range bound, gives the pointwise clipping inequality
\[
\bigl(\varphi_T(z,u)-\tau(P_\theta)\bigr)^2
\le
\bigl(T(z,u)-\tau(P_\theta)\bigr)^2
\qquad\text{for all }(z,u),\theta .
\]
Integrating this inequality against
\(P_\theta^{\otimes n}\otimes (P_\theta)_{XA}^{\otimes m}\) yields
\[
\int
\bigl(\varphi_T(Z,U)-\tau(P_\theta)\bigr)^2
\,d\bigl(P_\theta^{\otimes n}\otimes (P_\theta)_{XA}^{\otimes m}\bigr)(Z,U)
\le
\mathcal V_T(\theta).
\]
By the same finite product expansion as above, the left side is exactly the
empirical-side table risk of \(\varphi_T\) at \(\theta\).
The comparison of worst-case risks also requires a finite bound for the unclipped estimator.  Set
\[
B_T:=\sum_{(z,u)\in\mathcal Z_n\times\mathcal C_d^m}
\bigl(|T(z,u)|+1\bigr)^2 .
\]
Since \(|\upsilon_\theta|\le1\),
\[
\bigl(T(z,u)-\upsilon_\theta\bigr)^2
\le
\bigl(|T(z,u)|+1\bigr)^2
\le B_T
\]
for every \((z,u)\) and \(\theta\).  Hence the worst-case risk of \(T\) over the reindexed model is bounded, and the integrated clipping inequality implies
\[
\sup_{\theta\in\Theta_{\epsilon,d}}
\int
\bigl(\varphi_T(Z,U)-\tau(P_\theta)\bigr)^2
\,d\bigl(P_\theta^{\otimes n}\otimes (P_\theta)_{XA}^{\otimes m}\bigr)(Z,U)
\le
\sup_{\theta\in\Theta_{\epsilon,d}}\mathcal V_T(\theta).
\]
Taking infima over all measurable paper estimators \(T\) gives
\[
\mathfrak R_m\le R_\epsilon(n,m,d).
\]
The two inequalities prove \(R_\epsilon(n,m,d)=\mathfrak R_m\).
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.minimaxRisk_eq_empiricalValue

\item The preceding displayed identities give
\[
\lim_{m\to\infty}R_\epsilon(n,m,d)
=
\lim_{m\to\infty}\mathfrak R_m
=
\mathfrak R_\star
=
R_\epsilon^{\mathrm{known}}(n,d).
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.known_marginal_limit
\end{enumerate}
\end{proof}
