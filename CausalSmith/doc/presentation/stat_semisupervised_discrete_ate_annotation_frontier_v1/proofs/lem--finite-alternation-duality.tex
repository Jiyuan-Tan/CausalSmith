\begin{proof}[Proof of \cref{obj:lem:finite-alternation-duality}]
\begin{enumerate}
\item Set
\[
D_\epsilon=\frac{\kappa}{(1+\kappa)(2+\kappa)},\qquad
\gamma_\epsilon=\min\left\{\frac{\kappa}{4},\frac{D_\epsilon\kappa}{32}\right\},\qquad
c_\epsilon=\frac{D_\epsilon}{8}.
\]
Since \(0<\epsilon<1/2\), we have \(\kappa>0\). Hence \(D_\epsilon>0\), \(\gamma_\epsilon>0\), and \(c_\epsilon>0\). Also \(D_\epsilon<1\), because
\[
\kappa<(1+\kappa)(2+\kappa)
\]
for \(\kappa>0\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.finite_alternation_duality

\item Fix \(L\ge2\) and \(B>0\), and put
\[
a=\frac{\gamma_\epsilon B}{L^2}.
\]
Then \(a>0\). Since \(\gamma_\epsilon\le \kappa/4\) and \(L^2\ge4\),
\[
\frac{a}{\kappa}
=\frac{\gamma_\epsilon B}{\kappa L^2}
\le B.
\]
Moreover,
\[
\frac{a}{\kappa}\le \frac{B}{16},
\]
because \(\gamma_\epsilon\le \kappa/4\) and \(L^2\ge4\). In particular \(a/\kappa<B\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.finite_alternation_duality

\item Let
\[
I=\left[\frac{a}{\kappa},B\right],
\qquad
f(p)=\frac{p}{p+a}.
\]
The pole \(-a\) lies to the left of \(I\), since \(a>0\) and \(a/\kappa>0\). Thus \(f\) is continuous on \(I\). Let \(E\) be the best uniform error for approximating \(f\) on \(I\) by polynomials of degree at most \(L\). The finite alternation theorem supplies an alternation dual certificate: distinct nodes in \(I\), normalized weights annihilating all monomials of degree at most \(L\), and an associated degree-\(L\) best approximating polynomial. % lean: Causalean.Mathlib.Analysis.FinitePolynomialAlternationDuality.exists_alternationDualCertificate

\item We claim
\[
c_\epsilon\le E.
\]
Suppose instead that \(E<c_\epsilon\). Define
\[
p_-=\frac{a}{\kappa},\qquad p_+=\frac{2a}{\kappa}.
\]
Then \(p_-<p_+\), and both points belong to \(I\). Indeed \(p_-=a/\kappa\), while
\[
p_+=\frac{2\gamma_\epsilon B}{\kappa L^2}\le B
\]
because \(\gamma_\epsilon\le \kappa\) and \(L^2\ge4\). At these two points,
\[
f(p_-)=\frac{1}{1+\kappa},\qquad
f(p_+)=\frac{2}{2+\kappa},
\]
so
\[
f(p_+)-f(p_-)
=\frac{\kappa}{(1+\kappa)(2+\kappa)}
=D_\epsilon.
\]
Let \(Q\) be the best approximating polynomial from the alternation certificate. For every \(p\in I\),
\[
|f(p)-Q(p)|\le E.
\]
Consequently,
\[
D_\epsilon-2E
\le Q(p_+)-Q(p_-).
\]
By the mean-value theorem there is a point \(\xi\in[p_-,p_+]\subset I\) such that
\[
Q'(\xi)=\frac{Q(p_+)-Q(p_-)}{p_+-p_-},
\]
and therefore
\[
\frac{D_\epsilon-2E}{p_+-p_-}\le |Q'(\xi)|.
\]
Since \(E<c_\epsilon=D_\epsilon/8\), the numerator is positive. % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.finite_alternation_duality

\item On \(I\), the target satisfies \(0\le f(p)\le1\). Combining this with \(|f(p)-Q(p)|\le E\) gives
\[
|Q(p)|\le 1+E\qquad (p\in I),
\]
or equivalently
\[
\|Q\|_{\infty,I}\le 1+E.
\]
Markov's inequality on the interval \(I=[a/\kappa,B]\), applied to the degree-\(L\) polynomial \(Q\), gives
\[
\|Q'\|_{\infty,I}
\le
\frac{2L^2}{B-a/\kappa}\,\|Q\|_{\infty,I}
\le
\frac{2L^2}{B-a/\kappa}(1+E).
\]
Because \(a/\kappa\le B/16\), we have \(B/2\le B-a/\kappa\). Also \(E<c_\epsilon<1\). Hence
\[
|Q'(\xi)|
\le
\frac{2L^2}{B-a/\kappa}(1+E)
\le
\frac{2L^2}{B/2}\cdot2
=
\frac{8L^2}{B}.
\]
Combining this with the previous lower bound and using
\[
p_+-p_-=\frac{a}{\kappa}
=\frac{\gamma_\epsilon B}{\kappa L^2}
\]
yields
\[
\frac{(D_\epsilon-2E)\kappa L^2}{\gamma_\epsilon B}
\le
\frac{8L^2}{B}.
\]
After cancellation,
\[
(D_\epsilon-2E)\kappa\le 8\gamma_\epsilon.
\]
But \(E<D_\epsilon/8\), so \(D_\epsilon-2E>3D_\epsilon/4\), while
\[
8\gamma_\epsilon\le \frac{D_\epsilon\kappa}{4}.
\]
This is impossible. Therefore \(c_\epsilon\le E\). % lean: Causalean.Mathlib.Analysis.FinitePolynomialAlternationDuality.markov_derivative_Icc

\item Let the alternation certificate have nodes \(p_0,\ldots,p_{L+1}\in I\) and weights \(\alpha_0,\ldots,\alpha_{L+1}\). They satisfy
\[
\sum_{i=0}^{L+1}|\alpha_i|=1,
\qquad
\sum_{i=0}^{L+1}\alpha_i p_i^j=0\quad(0\le j\le L),
\]
and
\[
\left|\sum_{i=0}^{L+1}\alpha_i f(p_i)\right|=E.
\]
Choose
\[
s=
\begin{cases}
1,&\text{if }\sum_i\alpha_i f(p_i)\ge0,\\
-1,&\text{if }\sum_i\alpha_i f(p_i)<0,
\end{cases}
\qquad
w_i=s\alpha_i .
\]
Then the nodes remain distinct and lie in \(I\), and
\[
\sum_{i=0}^{L+1}|w_i|
=
\sum_{i=0}^{L+1}|\alpha_i|
=1.
\]
For every \(0\le j\le L\),
\[
\sum_{i=0}^{L+1}w_i p_i^j
=
s\sum_{i=0}^{L+1}\alpha_i p_i^j
=0.
\]
Finally,
\[
\sum_{i=0}^{L+1}w_i\frac{p_i}{p_i+a}
=
s\sum_{i=0}^{L+1}\alpha_i f(p_i)
=
\left|\sum_{i=0}^{L+1}\alpha_i f(p_i)\right|
=
E
\ge c_\epsilon.
\]
These nodes and weights give the required finite moment dual certificate on \([a/\kappa,B]\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.finite_alternation_duality
\end{enumerate}
\end{proof}
