\begin{proof}[Proof of \cref{obj:lem:finite-product-padding-identity}]
\begin{enumerate}
\item
Write
\[
\Lambda_{\mathrm{in}}
  :=\left(\bigotimes_{i\in I}\mu_i\right)\otimes \nu
  \quad\text{on }\mathsf Y^I\times\mathsf Y,
\qquad
\Lambda_{\mathrm{out}}
  :=\bigotimes_{j\in J}\mathsf H_j
  \quad\text{on }\mathsf Y^J .
\]
The map \(\mathsf A\) is measurable. Indeed, for each \(j\in J\), the \(j\)-th coordinate of
\(\mathsf A(z,y)\) is the second projection if \(j=r\), is the \(i\)-th coordinate projection
from \(z\) if \(j=f(i)\), and is the constant \(y_0\) on the remaining coordinates. The condition
\(r\ne f(i)\) separates the distinguished coordinate from the embedded ones, and injectivity of
\(f\) makes the embedded coordinate \(i\) unique. Since \(I\) and \(J\) are finite, coordinatewise
measurability gives measurability of \(\mathsf A\). It is therefore enough, by uniqueness of finite
product measures on measurable rectangles, to prove that for every family of measurable sets
\[
(\mathsf G_j)_{j\in J},
\qquad
\mathsf G_j\in\mathcal Y ,
\]
one has
\[
\Lambda_{\mathrm{in}}\!\left(\mathsf A^{-1}
  \{w\in\mathsf Y^J:\forall j\in J,\ w_j\in\mathsf G_j\}\right)
=
\prod_{j\in J}\mathsf H_j(\mathsf G_j).
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.pi_prod_map_finitePadding

\item
First suppose that there is an unused coordinate \(j_\circ\in J\) such that
\[
j_\circ\ne r,\qquad
j_\circ\ne f(i)\ \text{for every }i\in I,
\qquad
y_0\notin \mathsf G_{j_\circ}.
\]
For every \((z,y)\in\mathsf Y^I\times\mathsf Y\), the definition of \(\mathsf A\) gives
\[
\mathsf A(z,y)_{j_\circ}=y_0.
\]
Thus
\[
\mathsf A^{-1}\{w:\forall j,\ w_j\in\mathsf G_j\}=\varnothing,
\]
so the left-hand side is \(0\). On the right-hand side, the same coordinate has
\[
\mathsf H_{j_\circ}=\delta_{y_0},
\qquad
\mathsf H_{j_\circ}(\mathsf G_{j_\circ})
  =\delta_{y_0}(\mathsf G_{j_\circ})=0,
\]
because \(y_0\notin\mathsf G_{j_\circ}\). Hence
\[
\prod_{j\in J}\mathsf H_j(\mathsf G_j)=0,
\]
and the required rectangle identity holds in this case.
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.pi_prod_map_finitePadding

\item
It remains to consider the complementary case:
\[
\forall j\in J,\quad
j\ne r\ \text{and}\ \bigl(\forall i\in I,\ j\ne f(i)\bigr)
\quad\Longrightarrow\quad
y_0\in\mathsf G_j .
\]
Then the preimage of the rectangle is exactly
\[
\mathsf A^{-1}\{w:\forall j,\ w_j\in\mathsf G_j\}
=
\{z\in\mathsf Y^I:\forall i\in I,\ z_i\in\mathsf G_{f(i)}\}
\times \mathsf G_r .
\]
For the forward inclusion, membership in the rectangle gives the \(r\)-coordinate condition
\(y\in\mathsf G_r\), and for each \(i\in I\) it gives
\[
z_i=\mathsf A(z,y)_{f(i)}\in\mathsf G_{f(i)},
\]
using \(r\ne f(i)\) and injectivity of \(f\). Conversely, if \(z_i\in\mathsf G_{f(i)}\) for all
\(i\) and \(y\in\mathsf G_r\), then each coordinate \(j\in J\) is checked by the three alternatives
in the definition of \(\mathsf A\): the coordinate \(r\) gives \(y\), a coordinate \(f(i)\) gives
\(z_i\), and an unused coordinate gives \(y_0\in\mathsf G_j\) by the displayed complementary
condition.
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.pi_prod_map_finitePadding

\item
Using the preceding preimage identity and the finite product formula,
\[
\begin{aligned}
\Lambda_{\mathrm{in}}\!\left(\mathsf A^{-1}
  \{w:\forall j,\ w_j\in\mathsf G_j\}\right)
&=
\left(\bigotimes_{i\in I}\mu_i\right)
  \{z:\forall i,\ z_i\in\mathsf G_{f(i)}\}\,
  \nu(\mathsf G_r) \\
&=
\left(\prod_{i\in I}\mu_i(\mathsf G_{f(i)})\right)\nu(\mathsf G_r).
\end{aligned}
\]
The product over \(J\) has the same value. The distinguished coordinate contributes
\[
\mathsf H_r(\mathsf G_r)=\nu(\mathsf G_r),
\]
each embedded coordinate contributes, for \(i\in I\),
\[
\mathsf H_{f(i)}(\mathsf G_{f(i)})=\mu_i(\mathsf G_{f(i)}),
\]
again by \(r\ne f(i)\) and injectivity of \(f\), and every unused coordinate \(j\) contributes
\[
\mathsf H_j(\mathsf G_j)=\delta_{y_0}(\mathsf G_j)=1
\]
by the complementary condition. Since \(f\) is injective, the embedded coordinates are indexed
without repetition, and therefore
\[
\prod_{j\in J}\mathsf H_j(\mathsf G_j)
=
\nu(\mathsf G_r)\prod_{i\in I}\mu_i(\mathsf G_{f(i)}).
\]
This agrees with the preceding expression for the pushforward on every measurable rectangle, so
\[
\mathsf A_\#\Lambda_{\mathrm{in}}=\Lambda_{\mathrm{out}},
\]
which is the desired identity.
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.pi_prod_map_finitePadding
\end{enumerate}
\end{proof}
