\begin{proof}[Proof of \cref{obj:lem:hybrid-light-branch-mean-bound}]
Let
\[
\theta_x:=p_x(\mu_{1x}-\mu_{0x}),\qquad
\ell_x:=\mathbb E P_x,\qquad
h_x:=\mathbb E H_x,
\]
where \(\mu_{ax}=P(Y=1\mid X=x,A=a)\), with the usual harmless value when
\(s_{ax}=0\). Let \(S_B=\{x\in[d]:p_x\le B\}\). The cell masses are nonnegative and
\(\sum_{x\in[d]}p_x=1\), since \(P\) is a probability law. We prove the bound by first controlling
each summand and then summing over \(S_B\). % lean: sum_light_branchMean_sq_le

\emph{Step 1.} Fix \(x\in S_B\). The light-branch expectation equals
\[
\ell_x
=
q_{1x}\frac{p_x}{B}G_L(s_{1x}/B)
-
q_{0x}\frac{p_x}{B}G_L(s_{0x}/B).
\]
To verify the first equality, write the light polynomial as
\[
W_{a,L}(z_0,z_1)=\sum_{r_0,r_1}c_{a,r_0,r_1}z_0^{r_0}z_1^{r_1}.
\]
The factorial lift in \cref{obj:synth_4} is
\[
\widehat W_{a,L,x}
=\sum_{r_0,r_1}c_{a,r_0,r_1}
\frac{(K_{0x})_{r_0}(K_{1x})_{r_1}}{t^{r_0+r_1}}.
\]
Independence and the Poisson factorial-moment identity
\(\mathbb E[(K_{ax})_r]=(ts_{ax})^r\) therefore give
\(\mathbb E\widehat W_{a,L,x}=W_{a,L}(s_{0x},s_{1x})\).  Independently,
\(\mathbb E(S_{ax}/u)=q_{ax}\), which proves the displayed formula for \(\ell_x\).
The light Chebyshev estimate then gives
\[
|\ell_x-\theta_x|
\le
\frac{2B}{\epsilon L^2}.
\]
Indeed \(p_x\le B\), \(s_{0x}+s_{1x}=p_x\), and \cref{obj:ass:overlap} gives
\(s_{ax}\ge \epsilon p_x\) for each arm \(a\) whenever \(p_x>0\). If \(p_x=0\),
then \(s_{0x}=s_{1x}=q_{0x}=q_{1x}=0\), so the displayed bound is immediate. If \(p_x>0\), the
shifted Chebyshev certificate in \cref{obj:lem:chebyshev-factorial-certificate}, together with the
definitions of \(E_L\) and \(G_L\) in \cref{obj:def:mixed-estimator-handle}, gives, for
\(z_a=s_{ax}/B\in[0,1]\),
\[
0\le E_L(z_a)\le \frac{1}{L^2z_a},
\qquad
1-z_aG_L(z_a)=E_L(z_a).
\]
Writing \(q_{ax}=s_{ax}\mu_{ax}\), the light-branch error becomes a linear combination of the two
terms \(p_x\mu_{ax}E_L(s_{ax}/B)\). Since \(0\le\mu_{ax}\le1\) and
\(s_{ax}\ge\epsilon p_x\), each arm contributes at most \(B/(\epsilon L^2)\), giving the stated
two-arm bound. % lean: light_exact_mean_bias_le

\emph{Step 2.} The heavy inverse-count branch satisfies
\[
|h_x-\theta_x|
\le
2p_x\exp(-\epsilon t p_x).
\]
Write \(u_{\rm out}:=u\) for the outcome-block intensity.  For each arm
\(a\in\{0,1\}\), set
\[
\eta_{ax}:=
\mathbb E\!\left[
\frac{S_{ax}}{u_{\rm out}}\frac{K_{0x}+K_{1x}+1}{K_{ax}+1}
\right].
\]
Since the outcome pool is independent of the factorial pool and
\(\mathbb E S_{ax}=u_{\rm out}q_{ax}\),
\[
\eta_{ax}
=
q_{ax}\,
\mathbb E\!\left[\frac{K_{0x}+K_{1x}+1}{K_{ax}+1}\right].
\]
The armwise inverse-count expectation formula in
\cref{obj:lem:poisson-inverse-arm-mean-formula}, applied to the independent factorial counts,
gives
\[
\eta_{ax}
=
p_x\mu_{ax}-(p_x-s_{ax})\mu_{ax}\exp(-t s_{ax}).
\]
Consequently
\[
|\eta_{ax}-p_x\mu_{ax}|
=
(p_x-s_{ax})\mu_{ax}\exp(-t s_{ax})
\le
p_x\exp(-\epsilon t p_x),
\]
because \(0\le p_x-s_{ax}\le p_x\), \(0\le\mu_{ax}\le1\), and occupied-cell overlap gives
\(s_{ax}\ge\epsilon p_x\); when \(p_x=0\), both sides vanish. Since
\[
h_x-\theta_x=(\eta_{1x}-p_x\mu_{1x})-(\eta_{0x}-p_x\mu_{0x}),
\]
the triangle inequality gives the displayed heavy-branch bound. % lean: integral_hybridHeavyPools_bias_le

\emph{Step 3.} Combining the two preceding bounds,
\[
(\ell_x-h_x)^2
\le
\frac{8B^2}{\epsilon^2L^4}
+
8p_x^2\exp(-2\epsilon t p_x).
\]
To see this, write
\[
\ell_x-h_x=(\ell_x-\theta_x)-(h_x-\theta_x)
\]
and use \((r-s)^2\le 2r^2+2s^2\) with
\(|r|\le 2B/(\epsilon L^2)\) and
\(|s|\le 2p_x\exp(-\epsilon t p_x)\). % lean: sum_light_branchMean_sq_le

\emph{Step 4.} For every \(x\),
\[
p_x^2\exp(-2\epsilon t p_x)
\le
\frac{p_x}{2\epsilon e\,t}.
\]
Indeed,
\[
p_x^2\exp(-2\epsilon t p_x)
=
p_x\{p_x\exp(-(2\epsilon)t p_x)\}.
\]
The elementary envelope \(z\exp(-\alpha t z)\le(\alpha e t)^{-1}\), applied with
\(z=p_x\) and \(\alpha=2\epsilon\), gives
\[
p_x\exp(-(2\epsilon)t p_x)\le \frac{1}{2\epsilon e\,t}.
\]
Multiplying by the nonnegative factor \(p_x\) gives the displayed bound. The envelope itself follows from
\(e\,(\alpha t z)\le \exp(\alpha t z)\), multiplied by
\(\exp(-\alpha t z)\), and then divided by the positive number \(\alpha e t\). % lean: poisson_bias_envelope

\emph{Step 5.} Summing the pointwise bound over \(S_B\) gives
\[
\sum_{x\in S_B}(\ell_x-h_x)^2
\le
\sum_{x\in S_B}
\left\{
\frac{8B^2}{\epsilon^2L^4}
+
8\frac{p_x}{2\epsilon e\,t}
\right\}.
\]
Since \(|S_B|\le d\) and \(\sum_{x\in S_B}p_x\le\sum_{x\in[d]}p_x=1\),
\[
\sum_{x\in S_B}(\ell_x-h_x)^2
\le
\frac{8dB^2}{\epsilon^2L^4}
+
\frac{8}{2\epsilon e\,t}.
\]
Finally \(1\le e\), so \(8/(2\epsilon e\,t)\le 4/(\epsilon t)\). Substituting
\(\ell_x=\mathbb E P_x\) and \(h_x=\mathbb E H_x\) yields
\[
\sum_{x:\,p_x\le B}
\left(\mathbb E P_x-\mathbb E H_x\right)^2
\le
\frac{8dB^2}{\epsilon^2L^4}
+
\frac{4}{\epsilon t},
\]
as claimed. % lean: sum_light_branchMean_sq_le
\end{proof}
