\begin{proof}[Proof of \cref{obj:lem:common-marginal-recipe-scale-tail}]
\begin{enumerate}
\item Define the mass-tail event on the atom space by
\[
\mathsf{MassTail}
:=
\Bigl\{w:\frac12<|S(w)-1|\Bigr\}.
\]
The mass assumptions give
\[
\int S\,d\Gamma=1,
\qquad
\operatorname{Var}_{\Gamma}(S)\le CkBa.
\]
Since \(\Gamma\) is a probability law, Chebyshev's inequality at radius \(1/2\) yields
\[
\Gamma(\mathsf{MassTail})
=
\Gamma\Bigl\{w:\frac12<|S(w)-1|\Bigr\}
\le
\frac{\operatorname{Var}_{\Gamma}(S)}{(1/2)^2}
\le
4CkBa .
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.commonMarginal_mass_concentration

\item For a fixed branch \(b\in\{0,1\}\), set
\[
\mathsf{SmallScale}_b
:=
\Bigl\{P:\max\{s_{\mathcal R}(P),0\}<\frac12\Bigr\}.
\]
Because \(\Pi_b=\Gamma\circ P_b^{-1}\),
\[
\Pi_b(\mathsf{SmallScale}_b)
=
\Gamma\{w:P_b(w)\in\mathsf{SmallScale}_b\}.
\]
For \(\Gamma\)-almost every \(w\), the recipe supplies both identities
\[
s_{\mathcal R}(P_b(w))=S(w)
\qquad\text{and}\qquad
0<S(w).
\]
On this full-measure set, \(P_b(w)\in\mathsf{SmallScale}_b\) implies
\[
S(w)=\max\{S(w),0\}<\frac12.
\]
Hence \(S(w)-1<-1/2\), and therefore
\[
|S(w)-1|=1-S(w)>\frac12.
\]
Thus
\[
\{w:P_b(w)\in\mathsf{SmallScale}_b\}
\subseteq \mathsf{MassTail}
\quad\text{up to a \(\Gamma\)-null set,}
\]
and consequently
\[
\Pi_b(\mathsf{SmallScale}_b)\le \Gamma(\mathsf{MassTail}).
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.commonMarginal_prior_scale_tail

\item The preceding containment is applied to the two branch cases separately: for \(b=0\) it uses
\[
\Pi_0=\Gamma\circ P_0^{-1},
\]
and for \(b=1\) it uses
\[
\Pi_1=\Gamma\circ P_1^{-1}.
\]
Combining the appropriate branch identity with the bound from the first step gives, for each
\(b\in\{0,1\}\),
\[
\Pi_b\!\left\{P:\max\{s_{\mathcal R}(P),0\}<\frac12\right\}
\le
\Gamma(\mathsf{MassTail})
\le
4CkBa .
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.commonMarginal_prior_scale_tail
\end{enumerate}
\end{proof}
