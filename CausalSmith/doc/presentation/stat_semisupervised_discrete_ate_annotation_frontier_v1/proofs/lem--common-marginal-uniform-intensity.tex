\begin{proof}[Proof of \cref{obj:lem:common-marginal-uniform-intensity}]
Fix \(0<\epsilon<1/2\).
\begin{enumerate}
\item By \cref{obj:lem:finite-alternation-duality}, with
\[
\kappa_\epsilon:=\frac{1-2\epsilon}{\epsilon}>0,
\]
there are \(\gamma_\epsilon>0\) and \(\bar c_\epsilon>0\) such that every \(L\ge2\) and \(B>0\), with \(a=\gamma_\epsilon B/L^2\), admit a finite signed certificate on \([a/\kappa_\epsilon,B]\), annihilating monomials through degree \(L\), whose signed integral of \(p/(p+a)\) is at least \(\bar c_\epsilon\). The one-coordinate marked-Poisson moment-matching estimate supplies constants \(\bar b_\epsilon>0\), \(\bar C_\epsilon>0\), and \(\rho_\epsilon\in(0,1)\) with the following uniform property for the complete normalized certificate object that generates the coordinate mixtures. For every finite index set \(\mathcal I\) and every normalized finite signed moment certificate
\[
\mathfrak C=\{(p_\iota,\omega_\iota):\iota\in\mathcal I\}
\]
of order \(L\), meaning that its signed measure \(\sum_{\iota\in\mathcal I}\omega_\iota\delta_{p_\iota}\) has normalized variation \(\sum_{\iota\in\mathcal I}|\omega_\iota|=1\) and the certificate data include the moment equations through degree \(L\), if its variation measure is supported on \([a/\kappa_\epsilon,B]\) and the support scale satisfies \((u+v)B\le \bar b_\epsilon L\), then the two branch marked-Poisson predictive laws \(\mathsf Q_0,\mathsf Q_1\) associated with that same certificate obey
\[
\operatorname{TV}(\mathsf Q_0,\mathsf Q_1)
 \le \bar C_\epsilon\,u\,a\,\rho_\epsilon^L .
\] Set
\[
C_\epsilon:=\max\{\bar C_\epsilon,1\},\qquad
c_\epsilon:=\min\{\bar c_\epsilon,1/4\},\qquad
b_\epsilon:=\bar b_\epsilon .
\]
Let
\[
D_\epsilon
:=\max\left\{
2,\,
\frac{\max\{1,\log(8C_\epsilon 65536\,c_\epsilon)+2\}}
{-\log\rho_\epsilon}
\right\},
\qquad
s_\epsilon:=\frac1{65536\,C_\epsilon}.
\]
Define
\[
b_{0,\epsilon}
:=
\min\left\{
\frac{b_\epsilon}{131072},\,
\frac{c_\epsilon}{8\gamma_\epsilon},\,
\frac{s_\epsilon c_\epsilon^3}{8(D_\epsilon+1)}
\right\},
\]
\[
q_\epsilon:=\frac{c_\epsilon}{2\gamma_\epsilon b_{0,\epsilon}},
\qquad
r_\epsilon
:=
\left(
\max\left\{
1,\,
\frac{4(D_\epsilon+1)^2}{s_\epsilon c_\epsilon^2\gamma_\epsilon}
\right\}
\right)^2,
\]
and
\[
\delta_\epsilon
:=
\min\left\{
\frac{c_\epsilon^4}{16},\,
\left(\frac{c_\epsilon\gamma_\epsilon b_{0,\epsilon}}
{2(D_\epsilon+1)}\right)^2,\,
\frac{1}{65536\,r_\epsilon}
\right\}.
\]
All these constants are positive, \(0<\rho_\epsilon<1\), and \(c_\epsilon\le1/4\). The first five displayed calibrations follow immediately from the three minima and the definitions: \(131072\,b_{0,\epsilon}\le b_\epsilon\), \(D_\epsilon\ge2\),
\[
q_\epsilon\gamma_\epsilon b_{0,\epsilon}
=\frac{c_\epsilon}{2}\le c_\epsilon,\qquad
q_\epsilon=\frac{c_\epsilon}{2\gamma_\epsilon b_{0,\epsilon}},\qquad
q_\epsilon\ge4,
\]
and
\[
\delta_\epsilon r_\epsilon\le\frac1{65536},\qquad
\delta_\epsilon\le\frac{c_\epsilon^4}{16},\qquad
\delta_\epsilon\le
\left(\frac{c_\epsilon\gamma_\epsilon b_{0,\epsilon}}
{2(D_\epsilon+1)}\right)^2 .
\]
Moreover,
\[
8(D_\epsilon+1)b_{0,\epsilon}
\le
\frac{c_\epsilon^3}{65536\,C_\epsilon},
\]
and the same bound on \(b_{0,\epsilon}\) gives
\[
\frac{4(D_\epsilon+1)}
{(1/(65536\,C_\epsilon))c_\epsilon^2\gamma_\epsilon}
\le q_\epsilon .
\]
The definition of \(r_\epsilon\) gives
\[
\left\{
\frac{4(D_\epsilon+1)^2}
{(1/(65536\,C_\epsilon))c_\epsilon^2\gamma_\epsilon}
\right\}^2
\le r_\epsilon .
\]
Finally, put \(\lambda_\epsilon:=-\log\rho_\epsilon>0\) and
\(M_\epsilon:=8C_\epsilon 65536\,c_\epsilon\). Since
\[
\lambda_\epsilon D_\epsilon
\ge \max\{1,\log M_\epsilon+2\},
\]
for every \(n\ge1\),
\[
\rho_\epsilon^{\lceil D_\epsilon\log(en)\rceil}
\le \frac1{M_\epsilon n}.
\]
Thus
\[
C_\epsilon(65536n)c_\epsilon
\rho_\epsilon^{\lceil D_\epsilon\log(en)\rceil}
\le
\frac{C_\epsilon 65536\,n\,c_\epsilon}{8C_\epsilon 65536\,c_\epsilon\,n}
=\frac18 .
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_uniform_intensity

\item Now fix \(n,m,L,d,k,u,v,B,a\) satisfying the hypotheses in the statement. By the bandwidth slack,
\[
(u+v)B\le b_\epsilon L=\bar b_\epsilon L,
\]
and \(a=\gamma_\epsilon B/L^2>0\). Applying \cref{obj:lem:finite-alternation-duality} at this \(L,B\) gives distinct points \(p_j\in[a/\kappa_\epsilon,B]\) and weights \(\omega_j\) with
\[
\sum_j|\omega_j|=1,\qquad
\sum_j \omega_j p_j^\ell=0\quad(0\le \ell\le L),
\qquad
c_\epsilon\le
\sum_j\omega_j\frac{p_j}{p_j+a}.
\]
Let
\[
\sigma:=\sum_j\omega_j\delta_{p_j},
\qquad
|\sigma|:=\sum_j|\omega_j|\delta_{p_j},
\]
and define the polar sign on all \(p\in\mathbb R\) by
\[
h(p):=\sum_j {\bf 1}_{\{p=p_j\}}\operatorname{sgn}(\omega_j).
\]
Then \(|\sigma|\) is a probability measure, \(|h(p)|\le1\), and
\[
\sigma=h\,|\sigma|,\qquad
c_\epsilon\le \int \frac{p}{p+a}\,d\sigma(p).
\]
Also, every certificate node satisfies \(p_j\ge a/\kappa_\epsilon>0\), so \(0\) is not one of the nodes. Hence the displayed definition gives \(h(0)=0\).
Define the zero-inflated scalar law
\[
\nu(E)
:=
\int_E \frac{a}{p+a}\,d|\sigma|(p)
+
\left(1-\int \frac{a}{p+a}\,d|\sigma|(p)\right){\bf 1}_E(0).
\]
Since \(p\ge a/\kappa_\epsilon>0\) on the support of \(|\sigma|\),
\[
0\le \frac{a}{p+a}\le 1
\quad |\sigma|\text{-a.e.}
\]
and \(|\sigma|\) has total mass one. Hence
\[
0\le \int \frac{a}{p+a}\,d|\sigma|(p)\le 1,
\]
so the coefficient of \(\delta_0\) is nonnegative and \(\nu\) is a probability measure supported on
\(\{0\}\cup[a/\kappa_\epsilon,B]\). Its mean \(\bar p:=\int p\,d\nu(p)\) is computed directly from the definition of \(\nu\): the atom at zero contributes nothing, and therefore
\[
\bar p
=
\int p\frac{a}{p+a}\,d|\sigma|(p)
=
a\int \frac{p}{p+a}\,d|\sigma|(p).
\]
Because \(0\le p/(p+a)\le1\) on the support, this gives
\[
0\le\bar p\le a .
\]
Let \(\mathsf G:=\nu^{\otimes k}\). For every vector \(\boldsymbol\xi\in\mathbb R^k\), set
\[
\widehat\xi_i
:=
\begin{cases}
\xi_i,& \xi_i\in\{p_j\}_j,\\
0,& \xi_i\notin\{p_j\}_j .
\end{cases}
\]
Under \(\mathsf G\), \(\widehat{\boldsymbol\xi}=\boldsymbol\xi\) almost surely, and the range of \(\widehat{\boldsymbol\xi}\) is finite. Choose one reservoir cell \(x_\star\in[d]\) and distinct rare cells \(x_1,\ldots,x_k\in[d]\setminus\{x_\star\}\), possible because \(k+1\le d\). Put
\[
p_\star:=1-k\bar p .
\]
Since \(k a\le c_\epsilon\le1/4\), one has \(p_\star\ge3/4>0\). Define the raw cell masses and raw total mass by
\[
\Omega_{\boldsymbol\xi}(x)
:=
p_\star{\bf 1}_{\{x=x_\star\}}
+\sum_{i=1}^k \widehat\xi_i{\bf 1}_{\{x=x_i\}},
\qquad
S(\boldsymbol\xi)
:=
p_\star+\sum_{i=1}^k\widehat\xi_i .
\]
Then \(S(\boldsymbol\xi)>0\) and
\[
S(\boldsymbol\xi)=\sum_{x=1}^d\Omega_{\boldsymbol\xi}(x).
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_uniform_intensity

\item Define the propensity field on all cells by
\[
e_{\boldsymbol\xi}(x)
:=
\begin{cases}
1/2, & x=x_\star,\\
1/2, & x\ne x_\star\ \text{and}\ \sum_i\widehat\xi_i{\bf 1}_{\{x=x_i\}}=0,\\
\displaystyle
\epsilon\,
\frac{\sum_i\widehat\xi_i{\bf 1}_{\{x=x_i\}}+a}
{\sum_i\widehat\xi_i{\bf 1}_{\{x=x_i\}}},
& \text{otherwise}.
\end{cases}
\]
If the last line applies, the denominator is some nonzero \(\widehat\xi_i\in[a/\kappa_\epsilon,B]\), so
\[
\epsilon
\le
\epsilon\left(1+\frac{a}{\widehat\xi_i}\right)
\le
\epsilon(1+\kappa_\epsilon)
=1-\epsilon .
\]
The other cases give \(e_{\boldsymbol\xi}(x)=1/2\in[\epsilon,1-\epsilon]\). For branches \(b\in\{0,1\}\), define control outcome means to be zero and treated outcome means by
\[
\mu_{1,\boldsymbol\xi}^{(0)}(x_i)
:=
\frac{1-h(\widehat\xi_i)}2,\qquad
\mu_{1,\boldsymbol\xi}^{(1)}(x_i)
:=
\frac{1+h(\widehat\xi_i)}2,
\]
with \(\mu_{1,\boldsymbol\xi}^{(b)}(x_\star)=0\) and
\(\mu_{1,\boldsymbol\xi}^{(b)}(x)=0\) on unused cells. Since \(|h|\le1\), all outcome means lie in \([0,1]\). Let \(P_b(\boldsymbol\xi)\) be the law on \([d]\times\{0,1\}^2\) with cell mass
\[
p_{\boldsymbol\xi}(x):=\frac{\Omega_{\boldsymbol\xi}(x)}{S(\boldsymbol\xi)},
\]
propensity \(e_{\boldsymbol\xi}(x)\), zero control outcome mean, and treated outcome mean \(\mu_{1,\boldsymbol\xi}^{(b)}(x)\). This is a probability law because each raw mass is nonnegative and
\[
\sum_{x=1}^d p_{\boldsymbol\xi}(x)
=
\frac{\sum_x\Omega_{\boldsymbol\xi}(x)}{S(\boldsymbol\xi)}=1 .
\]
For the reservoir cell, the raw mass is \(p_\star>0\) and the propensity is \(1/2\), which lies in \([\epsilon,1-\epsilon]\). For an unused cell, the raw mass is zero, so the occupied-cell overlap condition has no constraint; the totalized propensity there is \(1/2\). For a rare cell \(x_i\), if \(\widehat\xi_i=0\) then its raw mass is zero. If \(\widehat\xi_i\ne0\), then \(\widehat\xi_i\in[a/\kappa_\epsilon,B]\), and the calculation above gives
\[
\epsilon\le e_{\boldsymbol\xi}(x_i)
=\epsilon\left(1+\frac a{\widehat\xi_i}\right)
\le 1-\epsilon .
\]
The control outcome means are zero, and the treated outcome means are either zero or of the form \((1\pm h(\widehat\xi_i))/2\), hence lie in \([0,1]\). Thus every occupied cell satisfies the overlap bounds and all outcome means are legal binary means, so \(P_b(\boldsymbol\xi)\in\mathcal M_{d,\epsilon}\) by \cref{obj:def:model-class}. Define
\[
\Pi_b:=(P_b)_\#\mathsf G,\qquad b\in\{0,1\}.
\]
These are the two branch priors of the recipe, and the finite range of \(\widehat{\boldsymbol\xi}\) places the recipe in the finite-atom construction regime. The labeled and auxiliary intensities stored in the recipe are, by construction, \(u\) and \(v\). Since \(p_{\boldsymbol\xi}(x)\) and \(e_{\boldsymbol\xi}(x)\) are independent of \(b\),
\[
(P\mapsto P_{XA})_\#\Pi_0
=
(P\mapsto P_{XA})_\#\Pi_1 .
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_uniform_intensity

\item The raw total mass has mean one:
\[
\int S(\boldsymbol\xi)\,d\mathsf G(\boldsymbol\xi)
=
p_\star+\sum_{i=1}^k\int \xi_i\,d\mathsf G(\boldsymbol\xi)
=
1-k\bar p+k\bar p
=
1 .
\]
For branch \(b\), the raw target is
\[
T_b(\boldsymbol\xi)
:=
\sum_{i=1}^k
\widehat\xi_i\,
\frac{1+(-1)^{1-b}h(\widehat\xi_i)}2 .
\]
It satisfies the raw version of \cref{obj:def:ate-functional}. Indeed, the control outcome means are zero, so
\[
\tau(P_b(\boldsymbol\xi))
=\sum_{x=1}^d
\frac{\Omega_{\boldsymbol\xi}(x)}{S(\boldsymbol\xi)}
\mu_{1,\boldsymbol\xi}^{(b)}(x).
\]
Multiplying by \(S(\boldsymbol\xi)\) cancels the normalization. The reservoir and unused cells contribute zero, and the rare-cell terms give
\[
S(\boldsymbol\xi)\,\tau(P_b(\boldsymbol\xi))
=
\sum_{i=1}^k
\widehat\xi_i\,
\frac{1+(-1)^{1-b}h(\widehat\xi_i)}2
=T_b(\boldsymbol\xi).
\]
Moreover,
\[
\int
\left[
p\,\frac{1+h(p)}2
-
p\,\frac{1-h(p)}2
\right]d\nu(p)
=
\int p h(p)\,d\nu(p).
\]
The atom of \(\nu\) at zero contributes zero. On the nonzero support, \(d\nu=(a/(p+a))\,d|\sigma|\) and \(d\sigma=h\,d|\sigma|\), so
\[
\int p h(p)\,d\nu(p)
=
a\int \frac{p}{p+a}\,d\sigma(p).
\]
Therefore the raw prior centers \(\theta_b:=\int T_b\,d\mathsf G\) obey
\[
\theta_1-\theta_0
=
k a\int \frac{p}{p+a}\,d\sigma(p).
\]
The alternation gap gives
\[
c_\epsilon\le \int \frac{p}{p+a}\,d\sigma(p),
\]
so the displayed difference is nonnegative and at least \(c_\epsilon k a\). Hence
\[
c_\epsilon k a\le |\theta_1-\theta_0|.
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_uniform_intensity

\item The support and mean bounds for \(\nu\) give
\[
\operatorname{Var}_{\mathsf G}(S)
=
\operatorname{Var}_{\mathsf G}\left(\sum_{i=1}^k\xi_i\right)
\le
\sum_{i=1}^k\mathbb E_\nu[p^2]
\le
kB\,\mathbb E_\nu[p]
\le
kBa
\le C_\epsilon kBa .
\]
For each branch define
\[
g_b(p):=
p\,\frac{1+(-1)^{1-b}h(p)}2 .
\]
Then \(0\le g_b(p)\le p\le B\) on the support of \(\nu\), and
\[
T_b(\boldsymbol\xi)=\sum_{i=1}^k g_b(\xi_i)
\quad\text{almost surely under }\mathsf G .
\]
Consequently,
\[
\operatorname{Var}_{\mathsf G}(T_b)
\le
\sum_{i=1}^k\mathbb E_\nu[g_b(p)^2]
\le
kB\,\mathbb E_\nu[g_b(p)]
\le
kB\,\mathbb E_\nu[p]
\le
kBa
\le
C_\epsilon kBa .
\]
Taking the maximum over \(b=0,1\) gives the asserted raw target-variance bound. The same bounded-support estimates also give the required square integrability of \(S\) and of both \(T_b\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_uniform_intensity

\item It remains to bound the prior-predictive distance in the raw Poisson experiment. For a single rare cell, let \(\mathcal Z_{\mathrm{rare}}\) be the finite product space consisting of the two labeled treated-outcome counts, the auxiliary treated count, and the labeled and auxiliary aggregate control counts, and let \(z_{\varnothing}\) be its all-zero element. For \(b\in\{0,1\}\) and \(p\in\mathbb R\), define the raw rare-cell kernel by
\[
\mathsf Q^{\mathrm{raw}}_{b,p}
:=
\begin{cases}
\delta_{z_{\varnothing}}, & p=0,\\
\mathsf S_{\alpha_{u,v}}\mathsf M_{b,p}, & p\ne0,
\end{cases}
\qquad
\alpha_{u,v}:=\frac{u}{u+v}\in[0,1],
\]
where \(\mathsf M_{b,p}\) is the branch-\(b\) marked-Poisson law associated with the certificate at latent mass \(p\), and \(\mathsf S_{\alpha_{u,v}}\) is the fixed splitting kernel that separates the marked treated counts into labeled and auxiliary pieces and carries along the corresponding control-count components. Let
\[
\mathsf Q_b:=\int \mathsf Q^{\mathrm{raw}}_{b,p}\,d\nu(p),
\qquad
\widetilde{\mathsf Q}_b
:=\int \mathsf S_{\alpha_{u,v}}\mathsf M_{b,p}\,d\nu(p).
\]
For every measurable event \(E\subseteq\mathcal Z_{\mathrm{rare}}\) and every \(p\in\mathbb R\),
\[
\mathsf Q^{\mathrm{raw}}_{1,p}(E)-\mathsf Q^{\mathrm{raw}}_{0,p}(E)
=
(\mathsf S_{\alpha_{u,v}}\mathsf M_{1,p})(E)
-(\mathsf S_{\alpha_{u,v}}\mathsf M_{0,p})(E).
\]
Indeed, this is the definition when \(p\ne0\). When \(p=0\), the left side is zero because both raw kernels are \(\delta_{z_{\varnothing}}\). The right side is also zero: as shown above, \(0\) is not a certificate node, so \(h(0)=0\), and the two branch marked-Poisson laws agree at the zero fiber before the fixed splitting is applied. Integrating the displayed identity with respect to \(\nu\) gives, for every measurable \(E\),
\[
(\mathsf Q_1-\mathsf Q_0)(E)
=
(\widetilde{\mathsf Q}_1-\widetilde{\mathsf Q}_0)(E).
\]
Thus the signed finite measures \(\mathsf Q_1-\mathsf Q_0\) and \(\widetilde{\mathsf Q}_1-\widetilde{\mathsf Q}_0\) are equal, and their total variations are equal. Hence
\[
\operatorname{TV}(\mathsf Q_0,\mathsf Q_1)
=
\operatorname{TV}(\widetilde{\mathsf Q}_0,\widetilde{\mathsf Q}_1).
\]
Data processing under the splitting kernel and the one-coordinate marked-Poisson estimate give
\[
\operatorname{TV}(\mathsf Q_0,\mathsf Q_1)
\le
\bar C_\epsilon\,u\,a\,\rho_\epsilon^L .
\]
The certificate used here is precisely the finite-node certificate constructed above: its node set
is finite, its signed weights have total variation one, all nodes lie in
\([a/\kappa_\epsilon,B]\), and its moments through degree \(L\) vanish. Thus the normalized
finite-certificate and support hypotheses of the one-coordinate estimate are all discharged by the
displayed construction, rather than by an additional regularity convention.
Independence of the \(k\) rare coordinates yields
\[
\operatorname{TV}\!\left(\mathsf Q_0^{\otimes k},\mathsf Q_1^{\otimes k}\right)
\le
k\,\operatorname{TV}(\mathsf Q_0,\mathsf Q_1)
\le
\bar C_\epsilon\,u\,k\,a\,\rho_\epsilon^L .
\]
It remains to identify this product experiment with the raw Poisson experiment generated by the constructed laws. Let \(\mathsf R_b(\boldsymbol\xi)\) be the conditional raw Poisson law generated by \(P_b(\boldsymbol\xi)\) at raw scale \(S(\boldsymbol\xi)\). Let
\[
\mathcal Y_{\mathrm{cell}}
:=
\left\{y:\{0,1\}\to(\mathbb N\times\mathbb N)\times\mathbb N\right\}
\]
be the single-cell raw-count space; \(y(\alpha)=((y^{\mathrm{s}}_\alpha,y^{\mathrm{f}}_\alpha),y^{\mathrm{aux}}_\alpha)\) records labeled successes, labeled failures, and auxiliary counts in arm \(\alpha\). Let \(z_{\mathrm{cell},0}\in\mathcal Y_{\mathrm{cell}}\) be the all-zero cell observation, and let
\[
\mathfrak r:\mathcal Z_{\mathrm{rare}}\to\mathcal Y_{\mathrm{cell}}
\]
send the five rare-cell counts to the two arm-indexed cell observations: in the treated arm it records the treated success count, treated failure count, and auxiliary treated count, while in the control arm it records zero successes, the labeled control count, and the auxiliary control count. The common reservoir law is
\[
\mathsf Q_\star
:=
\bigotimes_{\alpha\in\{0,1\}}
\left
\{\operatorname{Poi}(0)\otimes
\operatorname{Poi}\left(\frac{u p_\star}{2}\right)\otimes
\operatorname{Poi}\left(\frac{v p_\star}{2}\right)\right\}.
\]
For fixed \(\boldsymbol\xi\), define rare-coordinate cell laws by
\[
\mathsf H_{b,\boldsymbol\xi,i}
:=
(\mathsf Q^{\mathrm{raw}}_{b,\widehat\xi_i})_\#\mathfrak r,
\qquad i=1,\ldots,k,
\]
and define a zero law \(\mathsf H_0:=\delta_{z_{\mathrm{cell},0}}\). For \(\boldsymbol y=(y_1,\ldots,y_k)\in\mathcal Y_{\mathrm{cell}}^k\) and \(y_\star\in\mathcal Y_{\mathrm{cell}}\), define the deterministic assembly map by
\[
\bigl[\mathsf A(\boldsymbol y,y_\star)\bigr](x,\alpha)
:=
\begin{cases}
y_\star(\alpha), & x=x_\star,\\
y_i(\alpha), & x=x_i\text{ for the unique }i,\\
z_{\mathrm{cell},0}(\alpha), & x\notin\{x_\star,x_1,\ldots,x_k\}.
\end{cases}
\]
The middle line is unambiguous because the rare cells are distinct and none is the reservoir. By \cref{obj:lem:finite-product-padding-identity}, applied with the embedding \(i\mapsto x_i\), distinguished coordinate \(x_\star\), coordinate laws \(\mathsf H_{b,\boldsymbol\xi,i}\), reservoir law \(\mathsf Q_\star\), and filler \(z_{\mathrm{cell},0}\),
\[
\mathsf A_\#\left(\left(\bigotimes_{i=1}^k
\mathsf H_{b,\boldsymbol\xi,i}\right)\otimes\mathsf Q_\star\right)
=
\bigotimes_{x=1}^d \mathsf J_{b,\boldsymbol\xi,x},
\]
where
\[
\mathsf J_{b,\boldsymbol\xi,x}
:=
\begin{cases}
\mathsf Q_\star, & x=x_\star,\\
\mathsf H_{b,\boldsymbol\xi,i}, & x=x_i\text{ for the unique }i,\\
\mathsf H_0, & x\notin\{x_\star,x_1,\ldots,x_k\}.
\end{cases}
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.pi_prod_map_finitePadding
Equivalently, if \(\mathsf A_{\mathrm{rare}}\) denotes the map
\[
\mathsf A_{\mathrm{rare}}(\boldsymbol z,z_\star)
:=
\mathsf A\bigl((\mathfrak r(z_1),\ldots,\mathfrak r(z_k)),z_\star\bigr),
\]
then
\[
(\mathsf A_{\mathrm{rare}})_\#\left(\left(\bigotimes_{i=1}^k
\mathsf Q^{\mathrm{raw}}_{b,\widehat\xi_i}\right)\otimes\mathsf Q_\star\right)
=
\bigotimes_{x=1}^d \mathsf J_{b,\boldsymbol\xi,x},
\]
because pushforward by \(\mathfrak r\) in each rare coordinate turns \(\bigotimes_i\mathsf Q^{\mathrm{raw}}_{b,\widehat\xi_i}\) into \(\bigotimes_i\mathsf H_{b,\boldsymbol\xi,i}\).
More explicitly, coordinatewise pushforward gives
\[
\left(\bigotimes_{i=1}^k
\mathsf Q^{\mathrm{raw}}_{b,\widehat\xi_i}\right)_\#
\bigl[(z_i)_i\mapsto(\mathfrak r(z_i))_i\bigr]
=
\bigotimes_{i=1}^k\mathsf H_{b,\boldsymbol\xi,i}.
\]
Taking the product with \(\mathsf Q_\star\), then composing this coordinate map with
\(\mathsf A\), gives \(\mathsf A_{\mathrm{rare}}\). Functoriality of pushforward therefore turns
the left side of the padding identity into the asserted assembled rare-coordinate law.
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.fiveProduct_map_rareObservationToCell

We now compare these coordinate laws with the conditional raw Poisson law cell by cell. At the reservoir, raw scale multiplication cancels the normalization:
\[
S(\boldsymbol\xi)\,p_{\boldsymbol\xi}(x_\star)=p_\star,
\qquad e_{\boldsymbol\xi}(x_\star)=\frac12,
\qquad \mu^{(b)}_{1,\boldsymbol\xi}(x_\star)=\mu^{(b)}_{0,\boldsymbol\xi}(x_\star)=0.
\]
Thus each arm has raw treatment-covariate mass \(p_\star/2\), zero labeled-success rate, labeled-failure rate \(u p_\star/2\), and auxiliary rate \(v p_\star/2\). Therefore the reservoir coordinate law of \(\mathsf R_b(\boldsymbol\xi)\) is
\[
\mathsf J_{b,\boldsymbol\xi,x_\star}=\mathsf Q_\star .
\]
At a rare cell \(x_i\), put \(p=\widehat\xi_i\). If \(p=0\), then
\[
S(\boldsymbol\xi)\,p_{\boldsymbol\xi}(x_i)=0,
\]
so all labeled and auxiliary Poisson rates at this cell are zero. The conditional raw Poisson coordinate law is \(\delta_{z_{\mathrm{cell},0}}\). The raw rare-cell kernel is also \(\delta_{z_{\varnothing}}\), and \(\mathfrak r(z_{\varnothing})=z_{\mathrm{cell},0}\), hence
\[
(\mathsf Q^{\mathrm{raw}}_{b,0})_\#\mathfrak r
=\delta_{z_{\mathrm{cell},0}}
=\mathsf J_{b,\boldsymbol\xi,x_i} .
\]
If \(p>0\), define
\[
s_{\mathrm{tr}}(p):=\epsilon(p+a),
\qquad
s_{\mathrm{ct}}(p):=p-s_{\mathrm{tr}}(p),
\qquad
\eta_1:=1,
\quad
\eta_0:=-1.
\]
The construction gives
\[
S(\boldsymbol\xi)\,p_{\boldsymbol\xi}(x_i)=p,
\qquad
e_{\boldsymbol\xi}(x_i)=\frac{s_{\mathrm{tr}}(p)}{p},
\qquad
\mu^{(b)}_{1,\boldsymbol\xi}(x_i)=\frac{1+\eta_b h(p)}2,
\qquad
\mu^{(b)}_{0,\boldsymbol\xi}(x_i)=0.
\]
Consequently the raw treatment-covariate masses in the treated and control arms are
\[
p\,e_{\boldsymbol\xi}(x_i)=s_{\mathrm{tr}}(p),
\qquad
p\{1-e_{\boldsymbol\xi}(x_i)\}=s_{\mathrm{ct}}(p).
\]
The conditional raw Poisson coordinate law at \(x_i\) is therefore the product law on \(\mathcal Y_{\mathrm{cell}}\) with treated-arm rates
\[
\frac{u s_{\mathrm{tr}}(p)(1+\eta_b h(p))}{2},
\qquad
\frac{u s_{\mathrm{tr}}(p)(1-\eta_b h(p))}{2},
\qquad
v s_{\mathrm{tr}}(p)
\]
for labeled successes, labeled failures, and auxiliary counts, and with control-arm rates
\[
0,
\qquad
u s_{\mathrm{ct}}(p),
\qquad
v s_{\mathrm{ct}}(p)
\]
for labeled successes, labeled failures, and auxiliary counts. The split marked rare-cell law \(\mathsf Q^{\mathrm{raw}}_{b,p}\) is the fivefold product law whose coordinates have exactly the nonzero rates just displayed: treated success, treated failure, auxiliary treated, labeled control, and auxiliary control. Applying \(\mathfrak r\) places the first three counts in the treated arm and the last two in the control arm, with zero control successes. Product Poisson laws are determined by their coordinate rates, so
\[
(\mathsf Q^{\mathrm{raw}}_{b,p})_\#\mathfrak r
=
\mathsf J_{b,\boldsymbol\xi,x_i}
\qquad(p=\widehat\xi_i>0).
\]
Combining the two rare-cell cases gives
\[
(\mathsf Q^{\mathrm{raw}}_{b,\widehat\xi_i})_\#\mathfrak r
=
\mathsf J_{b,\boldsymbol\xi,x_i}
\qquad(1\le i\le k).
\]
At an unused cell the raw mass is zero, so all rates are zero and its coordinate law is
\[
\mathsf J_{b,\boldsymbol\xi,x}=\mathsf H_0 .
\]
Thus the coordinate laws of the conditional raw Poisson experiment are precisely the padded product coordinate laws: reservoir law at \(x_\star\), the pushed-forward rare laws at the rare cells, and the zero law elsewhere. Since conditional raw Poisson counts are independent across cells,
\[
\mathsf R_b(\boldsymbol\xi)
=
\bigotimes_{x=1}^d \mathsf J_{b,\boldsymbol\xi,x}.
\]
The preceding coordinate identification, together with the padding identity 
\[
\mathsf A_\#\left(\left(\bigotimes_{i=1}^k
\mathsf H_{b,\boldsymbol\xi,i}\right)\otimes\mathsf Q_\star\right)
=
\bigotimes_{x=1}^d \mathsf J_{b,\boldsymbol\xi,x},
\]
which follows from \cref{obj:lem:finite-product-padding-identity}, yields the pointwise fiber identity
\[
\mathsf R_b(\boldsymbol\xi)
=
(\mathsf A_{\mathrm{rare}})_\#\left(\left(\bigotimes_{i=1}^k
\mathsf Q^{\mathrm{raw}}_{b,\widehat\xi_i}\right)\otimes\mathsf Q_\star\right).
\]
This equality uses more than the abstract padding identity: the reservoir calculation fixes the
distinguished coordinate, the two rare-cell cases identify every pushed-forward rare law, and the
unused-cell calculation supplies the Dirac fillers. Independence of the raw Poisson counts then
identifies the conditional experiment with the resulting finite product. Hence every coordinate
law on the two sides is the same before finite-product uniqueness is invoked.
The identity holds for every \(\boldsymbol\xi\). Integrating it against \(\mathsf G\) and using \(\widehat{\boldsymbol\xi}=\boldsymbol\xi\) almost surely gives
\[
\int \mathsf R_b(\boldsymbol\xi)\,d\mathsf G(\boldsymbol\xi)
=
(\mathsf A_{\mathrm{rare}})_\#\left(
\left[\int \bigotimes_{i=1}^k
\mathsf Q^{\mathrm{raw}}_{b,\xi_i}\,d\nu^{\otimes k}(\boldsymbol\xi)\right]
\otimes\mathsf Q_\star\right).
\]
For cylinder events \(E_1\times\cdots\times E_k\subseteq\mathcal Z_{\mathrm{rare}}^k\), independence under \(\nu^{\otimes k}\) gives
\[
\int \prod_{i=1}^k
\mathsf Q^{\mathrm{raw}}_{b,\xi_i}(E_i)\,d\nu^{\otimes k}(\boldsymbol\xi)
=
\prod_{i=1}^k \int
\mathsf Q^{\mathrm{raw}}_{b,p}(E_i)\,d\nu(p)
=
\prod_{i=1}^k \mathsf Q_b(E_i),
\]
because each map \(p\mapsto\mathsf Q^{\mathrm{raw}}_{b,p}(E_i)\) is measurable and takes values in
\([0,1]\). Finite Fubini therefore factors the integral of their product under
\(\nu^{\otimes k}\) into the product of the coordinate integrals. Both sides define probability
measures on the finite product space, and measurable rectangles form a generating \(\pi\)-system;
the \(\pi\)-\(\lambda\) uniqueness theorem extends the displayed equality from rectangles to the
whole product \(\sigma\)-algebra. Thus
\[
\int \bigotimes_{i=1}^k
\mathsf Q^{\mathrm{raw}}_{b,\xi_i}\,d\nu^{\otimes k}(\boldsymbol\xi)
=
\mathsf Q_b^{\otimes k}.
\]
Consequently the branch-
\(b\) raw prior-predictive law is
\[
\mathsf R_b
:=
\int \mathsf R_b(\boldsymbol\xi)\,d\mathsf G(\boldsymbol\xi)
=
(\mathsf A_{\mathrm{rare}})_\#\left(\mathsf Q_b^{\otimes k}\otimes\mathsf Q_\star\right).
\]
Appending the common reservoir law and applying the deterministic assembly map cannot increase total variation, so
\[
\operatorname{TV}(\mathsf R_0,\mathsf R_1)
\le
\operatorname{TV}(\mathsf Q_0^{\otimes k},\mathsf Q_1^{\otimes k})
\le
\bar C_\epsilon\,u\,k\,a\,\rho_\epsilon^L
\le
C_\epsilon\,u\,k\,a\,\rho_\epsilon^L .
\]
The constructed recipe with priors \(\Pi_0,\Pi_1\) therefore satisfies all displayed calibration, support, common-table, mean, separation, variance, and mixture-distance conclusions. % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_uniform_intensity
\end{enumerate}
\end{proof}
