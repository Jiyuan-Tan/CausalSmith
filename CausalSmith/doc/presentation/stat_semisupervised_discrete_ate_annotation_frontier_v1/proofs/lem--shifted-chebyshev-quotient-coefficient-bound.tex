\begin{proof}[Proof of \cref{obj:lem:shifted-chebyshev-quotient-coefficient-bound}]
Let \(L\ge 1\). We use the notation for \(H_L,E_L,G_L\) from \cref{obj:def:mixed-estimator-handle}. The phrase ``polynomial continuation at zero'' is used through the quotient identities
\[
zE_L(z)=H_L(z),\qquad zG_L(z)=1-E_L(z),
\]
with equality in \(\mathbb R[z]\); the cancellations below use only that multiplication by the nonzero indeterminate \(z\) is injective on \(\mathbb R[z]\).

\begin{enumerate}
\item Define, for every \(\alpha\in\mathbb N\),
\[
\theta_{L,\alpha}
=
(-1)^\alpha
\frac{2^{2\alpha+3}}{L(L+\alpha+2)}
\binom{L+\alpha+2}{2\alpha+4},
\]
and put
\[
\widetilde G_L(z)
=
\sum_{0\le \alpha<L-1}\theta_{L,\alpha}z^\alpha .
\]
We first derive the shifted-Chebyshev coefficient expansion used below.
Write \(Q_r(z)=T_r(1-2z)\).  The initial polynomials and the defining
Chebyshev recurrence give
\[
Q_0(z)=1,\qquad Q_1(z)=1-2z,\qquad
Q_{r+2}(z)=2(1-2z)Q_{r+1}(z)-Q_r(z).
\]
For \(r\in\mathbb N\) and \(\nu\in\mathbb Z\), define
\[
\beta_{r,\nu}:=
\begin{cases}
1, & \nu=0,\\[0.3em]
(-1)^\nu\dfrac{r}{r+\nu}
\binom{r+\nu}{2\nu}4^\nu, & 1\le\nu\le r,\\[0.8em]
0, & \nu<0\text{ or }r<\nu .
\end{cases}
\]
Comparing coefficients in the displayed recurrence reduces the induction to
\[
\beta_{r+2,\nu}
=2\beta_{r+1,\nu}-4\beta_{r+1,\nu-1}-\beta_{r,\nu}.
\]
For \(1\le\nu\le r\), write
\[
A_{r,\nu}:=\frac{r}{r+\nu}\binom{r+\nu}{2\nu}.
\]
Set \(A_{r,0}:=1\).  For \(\nu\ge1\), the factorial form is
\[
A_{r,\nu}
=\frac{r(r+\nu-1)!}{(2\nu)!(r-\nu)!},
\]
After cancelling the common factor
\((-1)^\nu4^\nu\), the coefficient recurrence is
\[
A_{r+2,\nu}
=2A_{r+1,\nu}+A_{r+1,\nu-1}-A_{r,\nu}.
\]
For \(1\le\nu\le r\), multiply this equality by
\((2\nu)!(r+2-\nu)!/(r+\nu-1)!\).  It becomes
\[
\begin{aligned}
&(r+2)(r+\nu+1)(r+\nu)\\
&\quad =
2(r+1)(r+\nu)(r+2-\nu)
+(r+1)(2\nu)(2\nu-1)\\
&\qquad\quad
-r(r+2-\nu)(r+1-\nu),
\end{aligned}
\]
and expanding both sides proves it.  At \(\nu=0\), the term
\(\beta_{r+1,-1}\) is zero by the outside-range convention, and the identity
becomes \(1=2\cdot1-0-1\).  At \(\nu=r+1\) and \(\nu=r+2\), the same
zero convention leaves, respectively,
\[
\begin{aligned}
(-1)^{r+1}(r+2)4^{r+1}
&=2\left((-1)^{r+1}\frac{4^{r+1}}2\right)
-4\left((-1)^r(r+1)4^r\right),\\
(-1)^{r+2}\frac{4^{r+2}}2
&=-4\left((-1)^{r+1}\frac{4^{r+1}}2\right),
\end{aligned}
\]
so the recurrence holds there as well.  Since the formula agrees with \(Q_0,Q_1\),
induction proves, for every real \(z\),
\[
T_L(1-2z)
=
1+
\sum_{1\le \nu\le L}
(-1)^\nu\frac{L}{L+\nu}
\binom{L+\nu}{2\nu}4^\nu z^\nu .
\]
The \(\nu=1\) term is \(-2L^2z\).  For the remaining terms write
\(\nu=\alpha+2\).  Then
\[
(-1)^{\alpha+2}\frac{L}{L+\alpha+2}
\binom{L+\alpha+2}{2\alpha+4}4^{\alpha+2}z^{\alpha+2}
=
2L^2\theta_{L,\alpha}z^{\alpha+2},
\]
so, with \(Q_L(z)=T_L(1-2z)\),
\[
Q_L(z)
=
1-2L^2z+2L^2z^2\widetilde G_L(z).
\]
Using these quotient identities, since \(Q_L(0)=1\), the polynomial
\[
H_L(z)=\frac{1-Q_L(z)}{2L^2}
\]
has zero constant term. Moreover \(Q_L'(0)=-2T_L'(1)=-2L^2\), so \(H_L'(0)=1\), and hence the quotient identity for \(E_L\) gives \(E_L(0)=1\). Thus \(1-E_L\) has zero constant term, as required for the quotient identity defining \(G_L\). Using the displayed formula for \(Q_L\),
\[
H_L(z)
=\frac{1-Q_L(z)}{2L^2}
=z-z^2\widetilde G_L(z)
=z(1-z\widetilde G_L(z)).
\]
Thus
\[
zE_L(z)=z(1-z\widetilde G_L(z)).
\]
Because multiplication by the nonzero indeterminate \(z\) is cancellative in \(\mathbb R[z]\),
\[
E_L(z)=1-z\widetilde G_L(z).
\]
Consequently
\[
zG_L(z)=1-E_L(z)=z\widetilde G_L(z),
\]
and the same cancellation gives
\[
G_L(z)=\widetilde G_L(z)
\]
as polynomials. % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.chebG_eq_explicit

\item Since
\[
\widetilde G_L(z)=\sum_{0\le \alpha<L-1}\theta_{L,\alpha}z^\alpha,
\]
every coefficient of \(\widetilde G_L\) outside the range \(0\le \alpha<L-1\) is zero. Hence
\[
\|G_L\|_1
=
\|\widetilde G_L\|_1
=
\sum_{\alpha\in\operatorname{supp}(\widetilde G_L)}
   |[z^\alpha]\widetilde G_L|
\le
\sum_{0\le \alpha<L-1}|\theta_{L,\alpha}|.
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.explicitG_support_subset

\item The last sum is the positive coefficient sum at \(1\):
\[
\sum_{0\le \alpha<L-1}|\theta_{L,\alpha}|
=
\sum_{0\le \alpha<L-1}|\theta_{L,\alpha}|\,1^\alpha .
\]
The coefficient formula also gives
\[
|\theta_{L,\alpha}|
=
\frac{2^{2\alpha+3}}{L(L+\alpha+2)}
\binom{L+\alpha+2}{2\alpha+4},
\]
because \(L>0\) and all factors except \((-1)^\alpha\) are nonnegative. % lean: CausalSmith.Stat.DiscreteAteMinimaxLoggap.gpos

\item It remains to bound this positive coefficient sum. Write
\[
\Theta_L^+
=
\sum_{0\le \alpha<L-1}|\theta_{L,\alpha}|.
\]
This is the positive coefficient sum at the argument \(1\). The sign identity
\((-1)^\alpha\theta_{L,\alpha}=|\theta_{L,\alpha}|\) gives
\[
\Theta_L^+
=
\sum_{0\le \alpha<L-1}\theta_{L,\alpha}(-1)^\alpha
=
\widetilde G_L(-1).
\]
Substituting \(z=-1\) in the identity
\[
Q_L(z)=1-2L^2z+2L^2z^2\widetilde G_L(z)
\]
and using \(Q_L(-1)=T_L(3)\) gives
\[
T_L(3)=1+2L^2+2L^2\Theta_L^+ .
\]
Since \(\Theta_L^+\ge0\) and \(L\ge1\),
\[
\Theta_L^+
\le
2L^2\Theta_L^+
\le
1+2L^2+2L^2\Theta_L^+
=
T_L(3).
\]
It remains to bound \(T_L(3)\). Put \(a_\nu=T_\nu(3)\). The Chebyshev recurrence gives
\[
a_0=1,\qquad a_1=3,\qquad
a_{\nu+2}=6a_{\nu+1}-a_\nu .
\]
Let \(\lambda_+=3+\sqrt8\) and \(\lambda_-=3-\sqrt8\).  Since
\(\lambda_++\lambda_-=6\) and \(\lambda_+\lambda_-=1\), both roots satisfy
\(x^2=6x-1\).  Consequently
\[
b_\nu:=\frac{\lambda_+^\nu+\lambda_-^\nu}{2}
\]
satisfies \(b_{\nu+2}=6b_{\nu+1}-b_\nu\), while
\[
b_0=1=a_0,\qquad b_1=\frac{\lambda_++\lambda_-}{2}=3=a_1.
\]
Induction in the common recurrence gives \(a_\nu=b_\nu\), so
\[
a_\nu
=
\frac{(3+\sqrt8)^\nu+(3-\sqrt8)^\nu}{2},
\]
and in particular \(a_\nu\ge0\), because \(2<\sqrt8<3\) makes both
\(\lambda_+\) and \(\lambda_-\) positive. A two-step induction now gives \(a_\nu\le6^\nu\): the cases \(\nu=0,1\) are immediate, and if the bound holds for \(\nu\) and \(\nu+1\), then
\[
a_{\nu+2}=6a_{\nu+1}-a_\nu
\le 6a_{\nu+1}
\le 6\cdot 6^{\nu+1}
=6^{\nu+2}.
\]
Thus \(T_L(3)=a_L\le6^L\), and therefore
\[
\sum_{0\le \alpha<L-1}|\theta_{L,\alpha}|
=
\Theta_L^+
\le 6^L .
\]
% lean: CausalSmith.Stat.DiscreteAteMinimaxLoggap.gpos_bound

\item Combining the preceding inequalities yields
\[
\|G_L\|_1
\le
\sum_{0\le \alpha<L-1}|\theta_{L,\alpha}|
\le
6^L,
\]
which proves the claim. % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.coefficientL1_chebG_le
\end{enumerate}
\end{proof}
