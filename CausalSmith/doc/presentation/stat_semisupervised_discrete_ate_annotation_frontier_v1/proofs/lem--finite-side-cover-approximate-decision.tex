\begin{proof}[Proof of \cref{obj:lem:finite-side-cover-approximate-decision}]
\begin{enumerate}
\item For a decision \(b:\mathsf Z\to[l,u]\), write
\[
R_\theta(b)
=
\sum_{z\in\mathsf Z}\Lambda_\theta(z)\bigl(b(z)-\upsilon_\theta\bigr)^2 .
\]
Since \(b(z),\upsilon_\theta\in[l,u]\), each squared term is at most \((u-l)^2\), and since \(\Lambda_\theta\) is a probability vector,
\[
0\le R_\theta(b)\le (u-l)^2
\qquad\text{for every }\theta\in\Theta .
\]
In particular the constant decision \(b_0(z)=l\) is admissible. % lean: finiteSquaredRisk_bounds

\item Fix \(\theta_0\in\Theta\). Let
\[
\mathcal F(\theta_0)
=
\{\theta\in\Theta:\Xi_\theta=\Xi_{\theta_0}\}
\]
and let
\[
V_{\mathrm{fib}}(\theta_0)
=
\inf_{\beta:\mathsf Z\to[l,u]}\ \sup_{\theta\in\mathcal F(\theta_0)} R_\theta(\beta).
\]
The fiber is nonempty because it contains \(\theta_0\). By the previous bound, the worst-case risk of the constant decision \(b_0\) over this fiber is at most \((u-l)^2\), hence \(V_{\mathrm{fib}}(\theta_0)\le (u-l)^2\). Thus the set of all fiber values is bounded above.

For an exact-side rule \(\psi:\mathsf Z\times\mathsf{Prob}(\mathsf C)\to[l,u]\), write
\[
R^{\mathrm{ex}}_\theta(\psi)
=
\sum_{z\in\mathsf Z}\Lambda_\theta(z)
\bigl(\psi(z,\Xi_\theta)-\upsilon_\theta\bigr)^2,
\qquad
\mathfrak R_\star
=
\inf_\psi\sup_{\theta\in\Theta}R^{\mathrm{ex}}_\theta(\psi).
\]
For any such \(\psi\), define the fiber decision
\[
\beta_{\psi,\theta_0}(z)=\psi(z,\Xi_{\theta_0}).
\]
If \(\theta\in\mathcal F(\theta_0)\), then \(\Xi_\theta=\Xi_{\theta_0}\), so
\[
R_\theta(\beta_{\psi,\theta_0})=R^{\mathrm{ex}}_\theta(\psi).
\]
Consequently
\[
V_{\mathrm{fib}}(\theta_0)
\le
\sup_{\theta\in\mathcal F(\theta_0)}R_\theta(\beta_{\psi,\theta_0})
\le
\sup_{\theta\in\Theta}R^{\mathrm{ex}}_\theta(\psi).
\]
Taking the infimum over \(\psi\) gives
\[
V_{\mathrm{fib}}(\theta_0)\le \mathfrak R_\star,
\qquad
\sup_{\vartheta\in\Theta}V_{\mathrm{fib}}(\vartheta)\le \mathfrak R_\star.
\]

For the reverse inequality, set
\[
\mathfrak s=\sup_{\vartheta\in\Theta}V_{\mathrm{fib}}(\vartheta).
\]
Let \(\mathfrak a\) be any real number with \(\mathfrak s<\mathfrak a\). Then every fiber value is strictly below \(\mathfrak a\). Since the fiber risks are nonnegative, the strict infimum property gives, for each \(\theta_0\), a decision \(\beta_{\theta_0}:\mathsf Z\to[l,u]\) such that
\[
\sup_{\theta\in\mathcal F(\theta_0)}R_\theta(\beta_{\theta_0})<\mathfrak a.
\]
Define a representative map on the whole simplex by
\[
\operatorname{rep}(\nu)=
\begin{cases}
\text{one }\theta\in\Theta\text{ with }\Xi_\theta=\nu, & \text{if such a }\theta\text{ exists},\\
\text{an arbitrary point of }\Theta, & \text{otherwise}.
\end{cases}
\]
This is defined for every \(\nu\in\mathsf{Prob}(\mathsf C)\) because \(\Theta\) is nonempty. Now assemble the exact-side rule
\[
\psi_{\mathfrak a}(z,\nu)=\beta_{\operatorname{rep}(\nu)}(z).
\]
For each \(\theta\in\Theta\), the representative \(\operatorname{rep}(\Xi_\theta)\) has side law \(\Xi_\theta\), so \(\theta\in\mathcal F(\operatorname{rep}(\Xi_\theta))\). Hence
\[
R^{\mathrm{ex}}_\theta(\psi_{\mathfrak a})
=
R_\theta\bigl(\beta_{\operatorname{rep}(\Xi_\theta)}\bigr)
<\mathfrak a.
\]
Therefore
\[
\mathfrak R_\star
\le
\sup_{\theta\in\Theta}R^{\mathrm{ex}}_\theta(\psi_{\mathfrak a})
\le \mathfrak a.
\]
Since this holds for every \(\mathfrak a>\mathfrak s\),
\[
\mathfrak R_\star\le\mathfrak s.
\]
Combining the two inequalities yields
\[
\mathfrak R_\star=\sup_{\vartheta\in\Theta}V_{\mathrm{fib}}(\vartheta),
\]
and in particular
\[
V_{\mathrm{fib}}(\theta_0)\le \mathfrak R_\star .
\]
% lean: exactSideMinimaxValue_eq_iSup_fiber

\item For \(\alpha>0\), define the closed side-law neighborhood value
\[
V_\alpha(\theta_0)
=
\inf_{\beta:\mathsf Z\to[l,u]}\ 
\sup_{\theta:\,D_\infty(\Xi_\theta,\Xi_{\theta_0})\le \alpha} R_\theta(\beta).
\]
The required shrinking-neighborhood fact is the reciprocal-radius convergence
\[
V_{1/(N+1)}(\theta_0)\longrightarrow V_{\mathrm{fib}}(\theta_0)
\qquad\text{as }N\to\infty .
\]
Indeed, for every \(\alpha\ge0\), the exact fiber \(\mathcal F(\theta_0)\) is contained in the closed \(\alpha\)-neighborhood, and therefore
\[
V_{\mathrm{fib}}(\theta_0)\le V_\alpha(\theta_0).
\]
This gives the lower side of the limit. For the upper side, fix a real number \(\mathfrak b>V_{\mathrm{fib}}(\theta_0)\) and put
\[
\mathfrak c=\frac{V_{\mathrm{fib}}(\theta_0)+\mathfrak b}{2},
\qquad
V_{\mathrm{fib}}(\theta_0)<\mathfrak c<\mathfrak b.
\]
Since the fiber risks are nonnegative, the strict infimum property gives a decision \(\beta^\sharp:\mathsf Z\to[l,u]\) with
\[
\sup_{\theta\in\mathcal F(\theta_0)}R_\theta(\beta^\sharp)<\mathfrak c,
\]
so
\[
\theta\in\mathcal F(\theta_0)
\quad\Longrightarrow\quad
R_\theta(\beta^\sharp)<\mathfrak c.
\]
For this fixed decision the map \(\theta\mapsto R_\theta(\beta^\sharp)\) is continuous, because \(\mathsf Z\) is finite and the coordinates \(\Lambda_\theta(z)\) and \(\upsilon_\theta\) are continuous. Hence the set
\[
\mathcal B=
\{\theta\in\Theta:\mathfrak c\le R_\theta(\beta^\sharp)\}
\]
is closed, and compactness of \(\Theta\) makes \(\mathcal B\) compact. The side-law map \(\theta\mapsto\Xi_\theta\) is continuous, so its image \(\Xi(\mathcal B)=\{\Xi_\theta:\theta\in\mathcal B\}\) is compact and therefore closed in the finite simplex. The preceding fiber inequality implies
\[
\Xi_{\theta_0}\notin \Xi(\mathcal B),
\]
for otherwise some \(\theta\in\mathcal B\) would satisfy \(\Xi_\theta=\Xi_{\theta_0}\) and hence both \(\mathfrak c\le R_\theta(\beta^\sharp)\) and \(R_\theta(\beta^\sharp)<\mathfrak c\). Since a closed set misses \(\Xi_{\theta_0}\), there is \(\alpha_0>0\) such that
\[
D_\infty(\Xi_\theta,\Xi_{\theta_0})<\alpha_0
\quad\Longrightarrow\quad
R_\theta(\beta^\sharp)<\mathfrak c .
\]
For all sufficiently large \(N\), \(1/(N+1)<\alpha_0\). For such \(N\), every \(\theta\) with
\[
D_\infty(\Xi_\theta,\Xi_{\theta_0})\le \frac1{N+1}
\]
satisfies \(R_\theta(\beta^\sharp)<\mathfrak c\), and therefore
\[
V_{1/(N+1)}(\theta_0)
\le
\sup_{\theta:\,D_\infty(\Xi_\theta,\Xi_{\theta_0})\le 1/(N+1)}
R_\theta(\beta^\sharp)
\le \mathfrak c
<\mathfrak b.
\]
Together with \(V_{\mathrm{fib}}(\theta_0)\le V_{1/(N+1)}(\theta_0)\), this proves the displayed convergence.

Apply the convergence with \(\mathfrak b=V_{\mathrm{fib}}(\theta_0)+\delta/2\). Choose an integer \(N_{\theta_0}\ge0\) with
\[
V_{1/(N_{\theta_0}+1)}(\theta_0)
<
V_{\mathrm{fib}}(\theta_0)+\delta/2 .
\]
Set
\[
r(\theta_0)=\frac1{N_{\theta_0}+1}>0 .
\]
Together with \(V_{\mathrm{fib}}(\theta_0)\le\mathfrak R_\star\), this gives
\[
V_{r(\theta_0)}(\theta_0)<\mathfrak R_\star+\delta .
\]
Since all risks are nonnegative, the range of worst-case risks over admissible decisions is bounded below by \(0\). Hence the strict inequality for the infimum yields a decision
\[
\beta_{\theta_0}:\mathsf Z\to[l,u]
\]
such that
\[
\sup_{\theta:\,D_\infty(\Xi_\theta,\Xi_{\theta_0})\le r(\theta_0)}
R_\theta(\beta_{\theta_0})
<
\mathfrak R_\star+\delta .
\]
Consequently, whenever
\[
D_\infty(\Xi_\theta,\Xi_{\theta_0})<r(\theta_0),
\]
the same \(\theta\) belongs to the closed neighborhood, and therefore
\[
R_\theta(\beta_{\theta_0})\le \mathfrak R_\star+\delta .
\]
% lean: localMinimaxValue_tendsto_fiber

\item The map \(\theta\mapsto\Xi_\theta\) is continuous because each coordinate \(\theta\mapsto\Xi_\theta(c)\) is continuous and \(\mathsf C\) is finite. For each \(\theta_0\), define
\[
U_{\theta_0}
=
\left\{\theta\in\Theta:
D_\infty(\Xi_\theta,\Xi_{\theta_0})<\frac{r(\theta_0)}2
\right\}.
\]
Each \(U_{\theta_0}\) is open, and \(\theta_0\in U_{\theta_0}\). Thus \(\{U_{\theta_0}:\theta_0\in\Theta\}\) is an open cover of \(\Theta\). Compactness gives finitely many centers
\[
\theta^\circ_1,\ldots,\theta^\circ_k
\]
such that these sets still cover \(\Theta\). Since \(\Theta\) is nonempty, this finite subcover has \(k\ge1\). % lean: IsCompact.elim_finite_subcover

\item For each selected center set
\[
\operatorname{rad}_j=r(\theta^\circ_j),
\qquad
\beta_j=\beta_{\theta^\circ_j}.
\]
Define
\[
\eta=\min_{1\le j\le k}\frac{\operatorname{rad}_j}{2}.
\]
Every \(\operatorname{rad}_j\) is positive, so \(\eta>0\), and
\[
\eta\le \frac{\operatorname{rad}_j}{2}<\operatorname{rad}_j
\qquad\text{for every }j.
\]
If \(\theta\in\Theta\), the finite subcover gives an index \(j\) with
\[
D_\infty(\Xi_\theta,\Xi_{\theta^\circ_j})
<
\frac{\operatorname{rad}_j}{2}.
\]
Since \(\eta\le\operatorname{rad}_j/2\), this implies
\[
D_\infty(\Xi_\theta,\Xi_{\theta^\circ_j})
<
\operatorname{rad}_j-\eta .
\]
Finally, if for some \(j\) and \(\theta\),
\[
D_\infty(\Xi_\theta,\Xi_{\theta^\circ_j})<\operatorname{rad}_j,
\]
then the local guarantee from the construction at \(\theta^\circ_j\) gives
\[
\sum_{z\in\mathsf Z}\Lambda_\theta(z)
\bigl(\beta_j(z)-\upsilon_\theta\bigr)^2
=
R_\theta(\beta_j)
\le \mathfrak R_\star+\delta .
\]
These are exactly the required centers, decisions, radii, and margin. % lean: finiteCover_approximateDecision
\end{enumerate}
\end{proof}
