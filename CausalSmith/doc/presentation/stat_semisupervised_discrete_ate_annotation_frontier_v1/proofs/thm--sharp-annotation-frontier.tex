\begin{proof}[Proof of \cref{obj:thm:sharp-annotation-frontier}]
\begin{enumerate}
\item By \cref{obj:thm:common-marginal-converse}, choose a constant \(c_\epsilon>0\) such that, for every \(n\ge1\), \(m\ge0\), and \(d\ge2\),
\[
c_\epsilon r_\epsilon(n,m,d)\le R_\epsilon(n,m,d).
\]
By \cref{obj:thm:uniform-mixed-upper}, choose \(C_\epsilon^{\mathrm{up}}>0\) such that
\[
\sup_{P\in\mathcal M_{d,\epsilon}}
E_{P^{\otimes n}\otimes P_{XA}^{\otimes m}}
\!\left[
\left\{\widehat\tau^{\mathrm{mix}}_{n,m,d}-\tau(P)\right\}^2
\right]
\le
C_\epsilon^{\mathrm{up}} r_\epsilon(n,m,d)
\]
for the same range of \(n,m,d\).  Set
\[
C_\epsilon:=\max\{c_\epsilon,C_\epsilon^{\mathrm{up}}\}.
\]
Then \(0<c_\epsilon\le C_\epsilon\).  The estimator
\(\widehat\tau^{\mathrm{mix}}_{n,m,d}\) is measurable, so the definition of
\(R_\epsilon(n,m,d)\) in \cref{obj:def:minimax-risk} gives
\[
R_\epsilon(n,m,d)
\le
\sup_{P\in\mathcal M_{d,\epsilon}}
E_{P^{\otimes n}\otimes P_{XA}^{\otimes m}}
\!\left[
\left\{\widehat\tau^{\mathrm{mix}}_{n,m,d}-\tau(P)\right\}^2
\right].
\]
Combining the last two displays, and using \(r_\epsilon(n,m,d)>0\), yields
\[
R_\epsilon(n,m,d)
\le
C_\epsilon^{\mathrm{up}} r_\epsilon(n,m,d)
\le
C_\epsilon r_\epsilon(n,m,d).
\]
Together with the lower bound from \cref{obj:thm:common-marginal-converse}, this proves the finite-sample bracket. % lean: sharp_annotation_frontier

\item The unannotated endpoint is the specialization \(m=0\) of \cref{obj:def:frontier-rate-handle}.  Since \(N=n\) in that case,
\[
r_\epsilon(n,0,d)
=
\min\!\left\{1,\frac1n+\frac{d^2}{n^2\log^2(en)}\right\}.
\]
% lean: sharp_annotation_frontier

\item The known-marginal limit is exactly the conclusion of \cref{obj:thm:known-marginal-limit}: for fixed \(n\ge1\) and \(d\ge2\),
\[
\lim_{m\to\infty} R_\epsilon(n,m,d)
=
R_\epsilon^{\mathrm{known}}(n,d).
\]
% lean: sharp_annotation_frontier

\item It remains to record the sequence consequences.  Fix sequences \((m_n)\) and \((d_n)\) with \(d_n\ge2\), and set
\[
\rho_n:=R_\epsilon(n,m_n,d_n),\qquad
q_n:=r_\epsilon(n,m_n,d_n),
\]
\[
\rho_n^{0}:=R_\epsilon(n,0,d_n),\qquad
q_n^{0}:=r_\epsilon(n,0,d_n).
\]
The finite-sample bracket gives, for all \(n\ge1\),
\[
c_\epsilon q_n\le \rho_n\le C_\epsilon q_n,
\qquad
c_\epsilon q_n^{0}\le \rho_n^{0}\le C_\epsilon q_n^{0}.
\]
Since all four sequences are nonnegative, these two-sided bounds imply
\[
(\rho_n)=\Theta(q_n),
\qquad
(\rho_n^{0})=\Theta(q_n^{0}).
\]
Indeed, the upper inequalities give the \(O\)-bounds in one direction, and the lower inequalities give
\(q_n\le c_\epsilon^{-1}\rho_n\) and
\(q_n^{0}\le c_\epsilon^{-1}\rho_n^{0}\). % lean: positive_two_sided_bounds_isTheta

With
\[
N_n=n+m_n,\qquad
b_n:=\frac{d_n}{N_n\ell_n},
\]
\cref{obj:def:frontier-rate-handle} gives
\[
q_n=\min\left\{1,\frac1n+b_n^2\right\}.
\]
Because \(n^{-1}\to0\), this display satisfies
\[
q_n\to0
\quad\Longleftrightarrow\quad
b_n\to0.
\]
The implication from right to left follows by continuity of \(x\mapsto\min\{1,x\}\); the implication from left to right follows because eventually \(q_n<1/2\), hence
\[
q_n=\frac1n+b_n^2
\]
eventually, so \(b_n^2\to0\).  Using \((\rho_n)=\Theta(q_n)\), we obtain
\[
R_\epsilon(n,m_n,d_n)\to0
\quad\Longleftrightarrow\quad
\frac{d_n}{N_n\ell_n}\to0.
\]
% lean: annotation_frontier_consistency_iff

Next,
\[
q_n=O(n^{-1})
\quad\Longleftrightarrow\quad
b_n^2=O(n^{-1})
\quad\Longleftrightarrow\quad
d_n=O\!\left(\frac{N_n\ell_n}{\sqrt n}\right).
\]
For the first equivalence, the term \(n^{-1}\) is already present in \(q_n\), and the same eventual equality
\(q_n=n^{-1}+b_n^2\) applies after the \(O(n^{-1})\) bound forces \(q_n\to0\).  The second equivalence is obtained by taking square roots of nonnegative sequences and multiplying by \(N_n\ell_n\).  Transferring through \((\rho_n)=\Theta(q_n)\) gives
\[
R_\epsilon(n,m_n,d_n)=O(n^{-1})
\quad\Longleftrightarrow\quad
d_n=O\!\left(\frac{N_n\ell_n}{\sqrt n}\right).
\]
% lean: annotation_frontier_parametric_iff

Finally put
\[
a_n:=\frac{d_n}{n\ell_n}.
\]
Then
\[
q_n=\min\left\{1,\frac1n+b_n^2\right\},
\qquad
q_n^{0}=\min\left\{1,\frac1n+a_n^2\right\},
\qquad
0\le b_n\le a_n .
\]
By \cref{obj:lem:annotation-frontier-strict-improvement}, the numerical strict-improvement
criterion for these two rates is
\[
q_n=o(q_n^{0})
\]
if and only if both
\[
\frac{d_n}{\sqrt n\,\ell_n}\to\infty
\]
and
\[
\frac{b_n}{\min\{1,a_n\}}\to0.
\]
To see this, observe first that \(q_n\ge n^{-1}\), so \(q_n=o(q_n^{0})\) forces
\(n^{-1}=o(q_n^{0})\), which is equivalent to
\(a_n\sqrt n=d_n/(\sqrt n\,\ell_n)\to\infty\).  Under this condition the floor term \(n^{-1}\) is negligible in \(q_n^{0}\), and
\[
q_n^{0}\asymp \min\{1,a_n^2\}.
\]
Since \(0\le b_n\le a_n\), the remaining requirement
\(\min\{1,n^{-1}+b_n^2\}=o(q_n^{0})\) is equivalent to
\[
b_n^2=o(\min\{1,a_n^2\}),
\]
which, for nonnegative sequences, is the same as
\[
\frac{b_n}{\min\{1,a_n\}}\to0.
\]
Substituting the definitions of \(a_n\) and \(b_n\), this last condition is exactly
\[
\frac{d_n/(N_n\ell_n)}
{\min\{1,d_n/(n\ell_n)\}}
\to0.
\]
The two-sided bounds \((\rho_n)=\Theta(q_n)\) and \((\rho_n^{0})=\Theta(q_n^{0})\) transfer
\(q_n=o(q_n^{0})\) precisely to
\[
R_\epsilon(n,m_n,d_n)
=
o\!\left(R_\epsilon(n,0,d_n)\right),
\]
which proves the stated strict-improvement equivalence. % lean: annotation_frontier_strict_improvement_iff
\end{enumerate}
\end{proof}
