\begin{proof}[Proof of \cref{obj:lem:explicit-chebyshev-calibration}]
Fix \(L\ge2\), so \(L>0\).
\begin{enumerate}
\item Write
\[
Q_L(z)=1+\sum_{j=1}^L q_{L,j}z^j,
\qquad q_{L,j}=[z^j]Q_L .
\]
The shifted Chebyshev expansion used for \(Q_L(z)=T_L(1-2z)\) gives
\[
\|Q_L\|_1
 =1+\sum_{j=1}^L |q_{L,j}|
 =T_L(3).
\]
The standard growth bound for Chebyshev polynomials at \(3\) gives \(T_L(3)\le 6^L\), and hence
\[
\|Q_L\|_1\le 6^L\le 7^L .
\]
For the quotient polynomial, \cref{obj:lem:shifted-chebyshev-quotient-coefficient-bound} gives
\[
\|G_L\|_1\le 6^L .
\]
Since \(L\ge2\),
\[
2\|G_L\|_1
 \le 2\cdot 6^L
 \le 2^L7^L
 =14^L .
\]
This proves the two coefficient bounds. % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.coefficientL1_chebQ_le; CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.coefficientL1_chebG_le

\item Fix \(a\in\{0,1\}\), \(t>0\), \(B>0\), and \(s_0,s_1\ge0\), and assume \(L\le tB\). Put
\[
\mathscr D=tB,\qquad p=s_0+s_1,\qquad \eta=1+\frac pB .
\]
Then \(\mathscr D>0\), \(\mathscr D(s_b/B)=ts_b\) for \(b=0,1\), and \(\eta\ge1\). Let
\[
\mathscr I_L=\{j\in\mathbb N:[z^j]G_L\ne0\}.
\]
To identify this index set, write
\[
Q_L(z)=1+q_{L,1}z+\sum_{r=2}^L q_{L,r}z^r,
\qquad q_{L,1}=-2L^2.
\]
The definitions of \(H_L\) and \(E_L\) give the scaled identity
\[
zE_L(z)=\frac{1-Q_L(z)}{2L^2}.
\]
Using the displayed value of \(q_{L,1}\), this identity gives
\[
E_L(z)=1-\sum_{r=2}^L\frac{q_{L,r}}{2L^2}z^{r-1},
\]
and the quotient definition of \(G_L\) gives
\[
G_L(z)=\frac{1-E_L(z)}z
=\sum_{r=2}^L\frac{q_{L,r}}{2L^2}z^{r-2}.
\]
Thus
\[
\mathscr I_L\subseteq\{0,\ldots,L-2\},
\]
with the assertion empty if the displayed sum has no nonzero terms, and every \(j\in\mathscr I_L\) satisfies \(j+1\le L\). For \(K=(K_0,K_1)\), with \(K_a=K_1\) when \(a=1\) and \(K_a=K_0\) when \(a=0\), define
\[
X_j(K)
=
\frac{(K_a)_{j+1}}{\mathscr D^{j+1}}
+
\frac{K_{1-a}(K_a)_j}{\mathscr D^{j+1}},
\qquad j\in\mathscr I_L .
\]
If \(\mathscr I_L\) is empty, the following sum is zero. With \(K_b\sim\operatorname{Poi}(ts_b)\) independently, the identities \(\mathscr D(s_b/B)=ts_b\) give the factorial lift as
\[
\widehat W_{a,L}
=
\sum_{j\in\mathscr I_L} [z^j]G_L\,X_j(K).
\]

For the moment bound, first use the one-dimensional Poisson falling-factorial estimate: if
\(\mathfrak N_z\sim\operatorname{Poi}(\mathscr D z)\), \(z\ge0\), and \(r,v\le L\), then
\[
\mathbb E\frac{(\mathfrak N_z)_r(\mathfrak N_z)_v}{\mathscr D^{r+v}}
=
\sum_{h=0}^{\min(r,v)}
\binom rh\binom vh h!\,
\frac{(\mathscr D z)^{r+v-h}}{\mathscr D^{r+v}}.
\]
For \(0\le h\le\min(r,v)\), the corresponding summand satisfies
\[
\binom rh\binom vh h!\,
\frac{(\mathscr D z)^{r+v-h}}{\mathscr D^{r+v}}
\le
\frac{(rv)^h}{h!\mathscr D^h}(1+z)^{r+v}
\le
\frac{L^h}{h!}(1+z)^{r+v},
\]
because \(rv/\mathscr D\le L\), which follows from \(r,v\le L\) and \(L\le\mathscr D\), and because \(z^{r+v-h}\le(1+z)^{r+v}\). Therefore
\[
\mathbb E\frac{(\mathfrak N_z)_r(\mathfrak N_z)_v}{\mathscr D^{r+v}}
\le
\sum_{h=0}^{L}\frac{L^h}{h!}(1+z)^{r+v}
\le
3^L(1+z)^{r+v},
\]
where the last inequality is the elementary finite-series bound \(\sum_{h=0}^L L^h/h!\le3^L\).

This estimate gives the same-arm square bound. For every \(0\le r\le L\), integrating out the unused arm in the product law and applying the preceding estimate with \(z=s_a/B\) and orders \((r,r)\) gives
\[
\mathbb E\left(\frac{(K_a)_r}{\mathscr D^r}\right)^2
\le
3^L\left(1+\frac{s_a}{B}\right)^{2r}.
\]
Since \(1+s_a/B\le\eta\), \(\eta\ge1\), and \(2r\le2L\),
\[
\mathbb E\left(\frac{(K_a)_r}{\mathscr D^r}\right)^2
\le 3^L\eta^{2L}.
\]
In particular, for \(j\in\mathscr I_L\), using \(j+1\le L\),
\[
\mathbb E\left(\frac{(K_a)_{j+1}}{\mathscr D^{j+1}}\right)^2
\le
3^L\eta^{2L}.
\]

The cross-arm square bound follows by the product structure. For \(j\in\mathscr I_L\), independence and \((K_{1-a})_1=K_{1-a}\) give
\[
\mathbb E\left(\frac{K_{1-a}(K_a)_j}{\mathscr D^{j+1}}\right)^2
=
\mathbb E\left(\frac{(K_a)_j}{\mathscr D^j}\right)^2
\mathbb E\left(\frac{(K_{1-a})_1}{\mathscr D}\right)^2.
\]
Applying the one-dimensional estimate to arm \(a\) with orders \((j,j)\) and to arm \(1-a\) with orders \((1,1)\) yields
\[
\mathbb E\left(\frac{K_{1-a}(K_a)_j}{\mathscr D^{j+1}}\right)^2
\le
3^L\left(1+\frac{s_a}{B}\right)^{2j}
\cdot
3^L\left(1+\frac{s_{1-a}}{B}\right)^2.
\]
Now \(1+s_a/B\le\eta\), \(1+s_{1-a}/B\le\eta\), \(\eta\ge1\), and \(2j+2\le2L\). Hence
\[
\left(1+\frac{s_a}{B}\right)^{2j}
\left(1+\frac{s_{1-a}}{B}\right)^2
\le
\eta^{2j+2}
\le
\eta^{2L},
\]
and therefore
\[
\mathbb E\left(\frac{K_{1-a}(K_a)_j}{\mathscr D^{j+1}}\right)^2
\le
9^L\eta^{2L}.
\]
Since \((\xi+\upsilon)^2\le2\xi^2+2\upsilon^2\),
\[
\mathbb E X_j(K)^2
\le
2\cdot 3^L\eta^{2L}+2\cdot 9^L\eta^{2L}
\le
36^L\eta^{2L}
=
\bigl(6^L\eta^L\bigr)^2 .
\]
The finite-sum \(L^2\) inequality
\[
\mathbb E\left(\sum_{j\in\mathscr I_L} c_jX_j\right)^2
\le
\left(\sum_{j\in\mathscr I_L}|c_j|\right)^2
\max_{j\in\mathscr I_L}\mathbb E X_j^2
\]
therefore yields
\[
\mathbb E\widehat W_{a,L}^2
\le
\|G_L\|_1^2\bigl(6^L\eta^L\bigr)^2
\le
(6^L)^2\bigl(6^L\eta^L\bigr)^2
=
1296^L\eta^{2L}.
\]
Because \(1296\le7056\),
\[
\mathbb E\widehat W_{a,L}^2
\le
7056^L\left(1+\frac pB\right)^{2L}.
\]
If \(p\le B\), then \(\eta\le2\), and hence
\[
\mathbb E\widehat W_{a,L}^2
\le
1296^L2^{2L}
=
5184^L
\le
7056^L .
\]
This proves both second-moment assertions. % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.armMonomial_sum_sq_le; CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.explicit_chebyshev_calibration

\item Let \(z\in[0,1]\). The scaled identity for the continuation is
\[
zE_L(z)=(2L^2)^{-1}\{1-Q_L(z)\}=
\frac{1-Q_L(z)}{2L^2}.
\]
At \(z=0\), the derivative identity \(T_L'(1)=L^2\) gives \(E_L(0)=1\). For \(z>0\), put
\[
\theta=\arccos(1-2z).
\]
Then \(Q_L(z)=\cos(L\theta)\), so \(|Q_L(z)|\le1\). Hence \(1-Q_L(z)\ge0\), and the scaled identity gives \(E_L(z)\ge0\). Also
\[
1-Q_L(z)
=
1-\cos(L\theta)
\le
L^2(1-\cos\theta)
=
2L^2z,
\]
so \(zE_L(z)\le z\), whence \(E_L(z)\le1\). Finally, \(|Q_L(z)|\le1\) gives \(1-Q_L(z)\le2\), and the normalization by \(2L^2\) in the scaled identity gives
\[
zE_L(z)
=
\frac{1-Q_L(z)}{2L^2}
\le
\frac1{L^2},
\]
and division by \(z>0\) gives
\[
E_L(z)\le\frac1{L^2z}.
\]
The quotient definition gives the polynomial identity
\[
zG_L(z)+E_L(z)=1
\]
for every real \(z\), and therefore on \([0,1]\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.chebE_interval_bounds; CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.chebG_defining_identity
\end{enumerate}
\end{proof}
