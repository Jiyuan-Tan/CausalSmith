\begin{proof}[Proof of \cref{obj:lem:annotation-frontier-strict-improvement}]
\begin{enumerate}
\item
All limits are taken as \(n\to\infty\), and changing finitely many initial values is immaterial.  Put
\[
\iota_n:=\frac1n,\qquad
\alpha_n:=\frac{d_n}{n\ell_n},\qquad
\beta_n:=\frac{d_n}{N_n\ell_n}.
\]
For every sufficiently large \(n\), \(\ell_n=\log(en)>0\), \(0<\iota_n\le 1\), \(\alpha_n>0\), \(\beta_n\ge0\), and \(\beta_n\le\alpha_n\).  The last inequality uses \(N_n=n+m_n\ge n\), while positivity of \(\alpha_n\) uses \(d_n\ge2\).
By \cref{obj:def:frontier-rate-handle},
\[
r_\epsilon(n,m_n,d_n)=\min\{1,\iota_n+\beta_n^2\},
\qquad
r_\epsilon(n,0,d_n)=\min\{1,\iota_n+\alpha_n^2\}.
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.annotation_frontier_strict_improvement_iff

\item
We use the following numerical reduction.  Suppose sequences
\(\iota_n,\alpha_n,\beta_n\) satisfy, eventually,
\[
0<\iota_n\le1,\qquad 0<\alpha_n,\qquad 0\le\beta_n\le\alpha_n.
\]
Define
\[
\mathcal R_n:=\min\{1,\iota_n+\beta_n^2\},\qquad
\mathcal R^0_n:=\min\{1,\iota_n+\alpha_n^2\},\qquad
\mathcal H_n:=\min\{1,\alpha_n\}.
\]
Then
\[
\mathcal R_n=o(\mathcal R^0_n)
\quad\Longleftrightarrow\quad
\frac{\iota_n}{\mathcal R^0_n}\to0
\ \text{ and }\
\frac{\beta_n}{\mathcal H_n}\to0.
\]
Indeed, if \(\mathcal R_n=o(\mathcal R^0_n)\), then
\(\mathcal R_n/\mathcal R^0_n\to0\).  Since
\(\iota_n\le\mathcal R_n\) and \(\mathcal R^0_n>0\) eventually,
\[
0\le\frac{\iota_n}{\mathcal R^0_n}
\le
\frac{\mathcal R_n}{\mathcal R^0_n},
\]
so \(\iota_n/\mathcal R^0_n\to0\).  Also
\(\mathcal R_n/\mathcal R^0_n<1\) eventually; hence \(\mathcal R_n<1\) eventually, and therefore
\[
\mathcal R_n=\iota_n+\beta_n^2
\]
eventually.  Thus
\[
0\le
\frac{\beta_n^2}{\mathcal R^0_n}
\le
\frac{\mathcal R_n}{\mathcal R^0_n},
\]
so \(\beta_n^2/\mathcal R^0_n\to0\).  To replace \(\mathcal R^0_n\) by \(\mathcal H_n^2\), observe that
\[
\mathcal H_n^2=\min\{1,\alpha_n^2\},
\]
and hence
\[
\mathcal H_n^2\le \mathcal R^0_n\le \iota_n+\mathcal H_n^2.
\]
Writing \(\mathcal Q_n:=\iota_n/\mathcal R^0_n\), we have \(\mathcal Q_n\to0\), and whenever
\(\mathcal Q_n<1\),
\[
0\le
\frac{\iota_n}{\mathcal H_n^2}
\le
\frac{\mathcal Q_n}{1-\mathcal Q_n}.
\]
Consequently \(\iota_n/\mathcal H_n^2\to0\), and the displayed two-sided bound gives
\[
\frac{\mathcal R^0_n}{\mathcal H_n^2}\to1.
\]
Therefore
\[
\frac{\beta_n^2}{\mathcal H_n^2}
=
\frac{\beta_n^2}{\mathcal R^0_n}\,
\frac{\mathcal R^0_n}{\mathcal H_n^2}
\to0,
\]
and taking square roots, using \(\beta_n\ge0\) and \(\mathcal H_n>0\), yields
\(\beta_n/\mathcal H_n\to0\).

Conversely, assume
\[
\frac{\iota_n}{\mathcal R^0_n}\to0,
\qquad
\frac{\beta_n}{\mathcal H_n}\to0.
\]
Then \(\beta_n^2/\mathcal H_n^2\to0\), and since
\(\mathcal H_n^2\le\mathcal R^0_n\),
\[
0\le
\frac{\beta_n^2}{\mathcal R^0_n}
\le
\frac{\beta_n^2}{\mathcal H_n^2}
\to0.
\]
Finally,
\[
0\le
\frac{\mathcal R_n}{\mathcal R^0_n}
\le
\frac{\iota_n+\beta_n^2}{\mathcal R^0_n}
=
\frac{\iota_n}{\mathcal R^0_n}
+
\frac{\beta_n^2}{\mathcal R^0_n}
\to0,
\]
which is exactly \(\mathcal R_n=o(\mathcal R^0_n)\).
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.strict_min_rates_iff_ratios

\item
We also need the supervised floor equivalence
\[
\frac{(1/n)}{r_\epsilon(n,0,d_n)}\to0
\quad\Longleftrightarrow\quad
\frac{d_n}{\sqrt n\,\ell_n}\to\infty.
\]
Set
\[
\sigma_n:=\frac{d_n}{\sqrt n\,\ell_n}.
\]
For \(n\ge1\),
\[
r_\epsilon(n,0,d_n)=\min\{1,\iota_n+\alpha_n^2\},
\]
and direct algebra gives
\[
\frac{\iota_n}{\iota_n+\alpha_n^2}
=
\frac1{1+\sigma_n^2}.
\]
Assume first that \(\iota_n/r_\epsilon(n,0,d_n)\to0\).  Given a real threshold \(Y\), put
\[
Y_+:=\max\{Y,0\}+1.
\]
Eventually
\[
\frac{\iota_n}{r_\epsilon(n,0,d_n)}
<
\frac1{1+Y_+^2}.
\]
Since \(r_\epsilon(n,0,d_n)\le \iota_n+\alpha_n^2\),
\[
\frac1{1+\sigma_n^2}
=
\frac{\iota_n}{\iota_n+\alpha_n^2}
\le
\frac{\iota_n}{r_\epsilon(n,0,d_n)}
<
\frac1{1+Y_+^2}.
\]
Because \(\sigma_n\ge0\), this forces \(\sigma_n\ge Y_+\), and hence \(\sigma_n\ge Y\) eventually.  Thus \(\sigma_n\to\infty\).

Conversely, if \(\sigma_n\to\infty\), then \(\iota_n\to0\) and \(1/\sigma_n^2\to0\).  With
\[
\mathcal H_n:=\min\{1,\alpha_n\},
\]
the same identity \(\mathcal H_n^2=\min\{1,\alpha_n^2\}\) gives
\[
\mathcal H_n^2\le r_\epsilon(n,0,d_n).
\]
Therefore
\[
\frac{\iota_n}{r_\epsilon(n,0,d_n)}
\le
\frac{\iota_n}{\mathcal H_n^2}.
\]
If \(\alpha_n\le1\), then
\[
\frac{\iota_n}{\mathcal H_n^2}
=
\frac{\iota_n}{\alpha_n^2}
=
\frac1{\sigma_n^2};
\]
if \(\alpha_n>1\), then \(\mathcal H_n=1\) and
\[
\frac{\iota_n}{\mathcal H_n^2}
=
\iota_n
\le
\iota_n+\frac1{\sigma_n^2}.
\]
Thus, in all cases,
\[
0\le
\frac{\iota_n}{r_\epsilon(n,0,d_n)}
\le
\iota_n+\frac1{\sigma_n^2}\to0,
\]
which proves the equivalence.
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.supervised_floor_ratio_iff

\item
Applying the reduction from the second step to the sequences in the first step yields
\[
r_\epsilon(n,m_n,d_n)=o\!\left(r_\epsilon(n,0,d_n)\right)
\]
if and only if
\[
\frac{\iota_n}{r_\epsilon(n,0,d_n)}\to0
\qquad\text{and}\qquad
\frac{\beta_n}{\min\{1,\alpha_n\}}\to0.
\]
Substituting the definitions of \(\alpha_n\) and \(\beta_n\), the second condition is exactly
\[
\frac{d_n/(N_n\ell_n)}
{\min\{1,d_n/(n\ell_n)\}}
\to0.
\]
By the supervised floor equivalence from the third step, the first condition is exactly
\[
\frac{d_n}{\sqrt n\,\ell_n}\to\infty.
\]
Combining the two equivalences proves the claimed characterization.
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.annotation_frontier_strict_improvement_iff
\end{enumerate}
\end{proof}
