\begin{proof}[Proof of \cref{obj:prop:binary-alphabet-parametric-rate}]
Fix \(0<\epsilon<1/2\).
\begin{enumerate}
\item By \cref{obj:lem:parametric-label-floor}, choose \(c_{\mathrm{lab}}>0\) such that, for every \(n\ge1\), \(m\ge0\), and \(d\ge2\),
\[
\frac{c_{\mathrm{lab}}}{n}\le R_\epsilon(n,m,d).
\]
By \cref{obj:thm:inverse-count-baseline}, choose \(C_{\mathrm{IC}}>0\) such that, for every \(n\ge1\), \(m\ge0\), and \(d\ge2\),
\[
R_\epsilon(n,m,d)
\le
C_{\mathrm{IC}}
\min\left\{
1,\frac1n+\frac{d^2}{(n+m)^2}
\right\}.
\]
Set
\[
c_\epsilon:=c_{\mathrm{lab}},
\qquad
C_\epsilon:=\max\{c_{\mathrm{lab}},5C_{\mathrm{IC}}\}.
\]
Then \(0<c_\epsilon\le C_\epsilon\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.binary_alphabet_parametric_rate

\item For \(d=2\), the lower bound supplied by \cref{obj:lem:parametric-label-floor} gives, for every \(n\ge1\) and \(m\ge0\),
\[
\frac{c_\epsilon}{n}
=
\frac{c_{\mathrm{lab}}}{n}
\le
R_\epsilon(n,m,2).
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.parametric_label_floor

\item It remains to bound the right-hand side.  Fix \(n\ge1\) and \(m\ge0\), and put
\[
N:=n+m .
\]
Then \(N\ge n\ge1\), hence \(N>0\), and
\[
n\le N\le N^2,
\]
because \(N\ge1\).  Therefore
\[
\frac{4}{N^2}\le \frac4n .
\]
Using \cref{obj:thm:inverse-count-baseline} with \(d=2\),
\[
\begin{aligned}
R_\epsilon(n,m,2)
&\le
C_{\mathrm{IC}}
\min\left\{
1,\frac1n+\frac{4}{N^2}
\right\} \\
&\le
C_{\mathrm{IC}}
\left(\frac1n+\frac{4}{N^2}\right) \\
&\le
C_{\mathrm{IC}}\frac5n
=
\frac{5C_{\mathrm{IC}}}{n}
\le
\frac{C_\epsilon}{n}.
\end{aligned}
\]
This proves the asserted two-sided bound for all \(n\ge1\) and \(m\ge0\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.inverse_count_baseline
\end{enumerate}
\end{proof}
