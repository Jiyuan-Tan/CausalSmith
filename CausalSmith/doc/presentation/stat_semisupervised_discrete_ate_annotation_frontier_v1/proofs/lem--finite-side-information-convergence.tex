\begin{proof}[Proof of \cref{obj:lem:finite-side-information-convergence}]
Let
\[
\mathsf Z_n=([d]\times\{0,1\}^2)^n,
\qquad
\mathsf C_d=[d]\times\{0,1\}.
\]
For \(\theta\in\Theta_{\epsilon,d}\), write
\[
\Lambda_\theta(z)=\lambda_\theta(\{z\}),\qquad
\Xi_\theta(c)=\xi_\theta(c),\qquad c\in\mathsf C_d .
\]
The assumptions \(d\ge2\) and \(0<\epsilon<1/2\) make \([d]\) nonempty and make
\([\epsilon,1-\epsilon]\) nonempty. The coordinate map
\[
(\pi,\varrho,\omega)
\longmapsto
\Bigl(\pi,\ (\varrho_x)_{x\in[d]},\ (\omega_{\alpha x})_{(\alpha,x)\in\{0,1\}\times[d]}\Bigr)
\]
identifies \(\Theta_{\epsilon,d}\), with its coordinate topology, with the finite product
\[
\mathsf{Prob}([d])
\times
[\epsilon,1-\epsilon]^{[d]}
\times
[0,1]^{\{0,1\}\times[d]} .
\]
Here \(\mathsf{Prob}([d])\) is the closed simplex
\[
\left\{\pi\in[0,1]^{[d]}:\sum_{x\in[d]}\pi_x=1\right\},
\]
which is compact as a closed subset of the finite cube \([0,1]^{[d]}\). The two interval powers are finite products of compact intervals. Hence \(\Theta_{\epsilon,d}\) is nonempty and compact.

For each \(\theta\), \(\Lambda_\theta\) is a probability law on \(\mathsf Z_n\), because it is the
\(n\)-fold product of the one-record law \(P_\theta\). Also \(\Xi_\theta\) is a probability law on
\(\mathsf C_d\). Indeed, every summand is nonnegative, and the normalization expands as
\[
\begin{aligned}
\sum_{x\in[d]}\sum_{\alpha\in\{0,1\}}\Xi_\theta(x,\alpha)
&=
\sum_{x\in[d]}\sum_{\alpha\in\{0,1\}}\sum_{y\in\{0,1\}}
\pi_x\varrho_x^\alpha(1-\varrho_x)^{1-\alpha}
\omega_{\alpha x}^y(1-\omega_{\alpha x})^{1-y}\\
&=
\sum_{x\in[d]}\pi_x
\left(\sum_{\alpha\in\{0,1\}}\varrho_x^\alpha(1-\varrho_x)^{1-\alpha}
\left[\sum_{y\in\{0,1\}}\omega_{\alpha x}^y(1-\omega_{\alpha x})^{1-y}\right]\right)\\
&=
\sum_{x\in[d]}\pi_x
\left((1-\varrho_x)+\varrho_x\right)
=1 .
\end{aligned}
\]
The maps \(\theta\mapsto\Lambda_\theta(z)\), \(\theta\mapsto\Xi_\theta(c)\), and
\(\theta\mapsto\upsilon_\theta\) are continuous, since each is obtained from finitely many coordinate
projections by finite sums and products. Finally,
\[
-1\le \omega_{1x}-\omega_{0x}\le1
\quad\text{for every }x,
\]
and the weights \(\pi_x\) are nonnegative with total mass one, so
\[
-1
=
\sum_x\pi_x(-1)
\le
\sum_x\pi_x(\omega_{1x}-\omega_{0x})
=
\upsilon_\theta
\le
\sum_x\pi_x(1)
=
1 .
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.knownMarginalAuxProb_simplex

With this notation, the empirical-side minimax value at auxiliary size \(m\) is
\[
\mathfrak R_m
=
\inf_{\varphi:\mathsf Z_n\times\mathsf C_d^m\to[-1,1]}
\sup_{\theta\in\Theta_{\epsilon,d}}
\sum_{z\in\mathsf Z_n}\Lambda_\theta(z)
\sum_{u\in\mathsf C_d^m}\Xi_\theta^{\otimes m}(u)
\bigl(\varphi(z,u)-\upsilon_\theta\bigr)^2,
\]
and the exact-side minimax value is
\[
\mathfrak R_\star
=
\inf_{\psi:\mathsf Z_n\times\mathsf{Prob}(\mathsf C_d)\to[-1,1]}
\sup_{\theta\in\Theta_{\epsilon,d}}
\sum_{z\in\mathsf Z_n}\Lambda_\theta(z)
\bigl(\psi(z,\Xi_\theta)-\upsilon_\theta\bigr)^2 .
\]
We now use the finite side-information convergence argument for these data. First, for any
empirical estimator \(\varphi:\mathsf Z_n\times\mathsf C_d^m\to[-1,1]\), define the corresponding
exact-side estimator
\[
\overline\varphi(z,\nu)
=
\sum_{u\in\mathsf C_d^m}\nu^{\otimes m}(u)\,\varphi(z,u),
\qquad
z\in\mathsf Z_n,\quad \nu\in\mathsf{Prob}(\mathsf C_d).
\]
Convexity of \(v\mapsto(v-\upsilon_\theta)^2\) gives, for every \(\theta\),
\[
\int
\bigl(\overline\varphi(Z,\Xi_\theta)-\upsilon_\theta\bigr)^2\,d\lambda_\theta(Z)
\le
\int
\bigl(\varphi(Z,U)-\upsilon_\theta\bigr)^2\,
d\bigl(\lambda_\theta\otimes\Xi_\theta^{\otimes m}\bigr)(Z,U).
\]
Taking suprema over \(\theta\) and then infima over \(\varphi\) yields
\[
\mathfrak R_\star\le \mathfrak R_m
\qquad\text{for every }m\ge0 .
\]
% lean: CausalSmith.Substrate.FiniteSideInformationMinimaxConvergence.exactSideMinimaxValue_le_empiricalSideMinimaxValue

It remains to prove the reverse asymptotic inequality. Put
\[
D_\infty(\nu,\nu')=\max_{c\in\mathsf C_d}|\nu(c)-\nu'(c)|
\]
for probability laws on \(\mathsf C_d\). Fix \(\delta>0\). With
\(\Theta=\Theta_{\epsilon,d}\), \(\mathsf Z=\mathsf Z_n\), \(\mathsf C=\mathsf C_d\),
\(\Lambda_\theta=\lambda_\theta\), \(\Xi_\theta=\xi_\theta\), \(l=-1\), and \(u=1\),
the compactness, simplex, continuity, and target-range checks above give the analytic hypotheses of
\cref{obj:lem:finite-side-cover-approximate-decision}. Its remaining structural hypotheses are
immediate but worth recording: \(\mathsf Z_n\) is a finite power of a finite set,
\(\mathsf C_d\) is finite and nonempty because \(d\ge2\), and \(-1\le1\). Thus every hypothesis of
\cref{obj:lem:finite-side-cover-approximate-decision} holds. Hence, for the present \(\delta\), there are
an integer \(k\ge1\), centers \(\theta^\circ_1,\ldots,\theta^\circ_k\in\Theta_{\epsilon,d}\),
bounded decisions \(\beta_1,\ldots,\beta_k:\mathsf Z_n\to[-1,1]\), radii
\(\operatorname{rad}_1,\ldots,\operatorname{rad}_k\), and a margin \(\eta>0\) such that
\[
\eta<\operatorname{rad}_j\qquad(1\le j\le k),
\]
\[
\forall\theta\in\Theta_{\epsilon,d}\quad
\exists j\in\{1,\ldots,k\},\qquad
D_\infty(\Xi_\theta,\Xi_{\theta^\circ_j})
<
\operatorname{rad}_j-\eta,
\]
and, for every \(j\in\{1,\ldots,k\}\) and every \(\theta\in\Theta_{\epsilon,d}\),
\[
D_\infty(\Xi_\theta,\Xi_{\theta^\circ_j})<\operatorname{rad}_j
\quad\Longrightarrow\quad
\int
\bigl(\beta_j(Z)-\upsilon_\theta\bigr)^2\,d\lambda_\theta(Z)
\le
\mathfrak R_\star+\delta .
\]
% lean: CausalSmith.Substrate.FiniteSideInformationMinimaxConvergence.finiteCover_approximateDecision

For \(m\ge1\) and \(u\in\mathsf C_d^m\), let \(\widehat\Xi_u\in\mathsf{Prob}(\mathsf C_d)\) be the
empirical law of \(u\), and put \(\mathcal J:=\{1,\ldots,k\}\). Since \(k\ge1\), this set is
nonempty. Choose a selector \(\sigma:\mathsf{Prob}(\mathsf C_d)\to\mathcal J\) as follows: if at
least one \(\jmath\in\mathcal J\) satisfies
\[
D_\infty(\nu,\Xi_{\theta^\circ_{\jmath}})
<
\operatorname{rad}_{\jmath}-\eta/2,
\]
let \(\sigma(\nu)\) be the least such index; if there is no such index, set
\(\sigma(\nu)=1\). Define
\[
\Phi_{\delta,m}(z,u)=\beta_{\sigma(\widehat\Xi_u)}(z).
\]
On the event
\[
G_{\theta,m}
=
\left\{
u\in\mathsf C_d^m:
D_\infty(\widehat\Xi_u,\Xi_\theta)<\eta/2
\right\},
\]
choose a cover index \(j\in\mathcal J\) with
\[
D_\infty(\Xi_\theta,\Xi_{\theta_j^\circ})
<\operatorname{rad}_j-\eta.
\]
The good-event bound and the triangle inequality give
\[
D_\infty(\widehat\Xi_u,\Xi_{\theta_j^\circ})
<\eta/2+\operatorname{rad}_j-\eta
=\operatorname{rad}_j-\eta/2,
\]
so \(j\) is eligible for the selector. Hence
\[
D_\infty\bigl(\widehat\Xi_u,
\Xi_{\theta^\circ_{\sigma(\widehat\Xi_u)}}\bigr)
<\operatorname{rad}_{\sigma(\widehat\Xi_u)}-\eta/2.
\]
Applying the triangle inequality once more, now through \(\widehat\Xi_u\), yields
\[
D_\infty\bigl(\Xi_\theta,\Xi_{\theta^\circ_{\sigma(\widehat\Xi_u)}}\bigr)
<
\eta/2+\operatorname{rad}_{\sigma(\widehat\Xi_u)}-\eta/2
=
\operatorname{rad}_{\sigma(\widehat\Xi_u)}.
\]
Therefore
\[
\int
\bigl(\Phi_{\delta,m}(Z,U)-\upsilon_\theta\bigr)^2\,
d(\lambda_\theta\otimes\Xi_\theta^{\otimes m})(Z,U)
\le
\mathfrak R_\star+\delta
+
4\,\Xi_\theta^{\otimes m}(G_{\theta,m}^{c}),
\]
because all estimates and all targets lie in \([-1,1]\). For the finite-alphabet concentration step, fix \(\theta\). For \(c\in\mathsf C_d\) and
\(u=(u_1,\ldots,u_m)\in\mathsf C_d^m\), write
\[
N_c(u)=\sum_{i=1}^m\mathbf 1\{u_i=c\},
\qquad
\widehat\Xi_u(c)=\frac{N_c(u)}m .
\]
Under \(\Xi_\theta^{\otimes m}\), the variables \(\mathbf 1\{U_i=c\}\) are independent Bernoulli variables with mean \(\Xi_\theta(c)\). Thus
\[
\mathbb E_\theta\widehat\Xi_U(c)=\Xi_\theta(c)
\]
and, since a Bernoulli variance is at most \(1/4\),
\[
\operatorname{Var}_\theta\!\left(\widehat\Xi_U(c)\right)
=
\frac{1}{m^2}\sum_{i=1}^m
\operatorname{Var}_\theta\!\left(\mathbf 1\{U_i=c\}\right)
\le
\frac{1}{4m} .
\]
Chebyshev's inequality gives, for every \(t>0\),
\[
\Xi_\theta^{\otimes m}\left(
\left|\widehat\Xi_U(c)-\Xi_\theta(c)\right|
\ge t
\right)
\le
\frac{1}{4mt^2} .
\]
If \(D_\infty(\widehat\Xi_U,\Xi_\theta)\ge\eta/2\), then the defining identity
\[
D_\infty(\widehat\Xi_U,\Xi_\theta)
=
\max_{c\in\mathsf C_d}
\left|\widehat\Xi_U(c)-\Xi_\theta(c)\right|
\]
implies that some \(c\in\mathsf C_d\) satisfies
\[
\left|\widehat\Xi_U(c)-\Xi_\theta(c)\right|
\ge \eta/2 .
\]
Hence the bad event satisfies
\[
G_{\theta,m}^c
=
\left\{D_\infty(\widehat\Xi_U,\Xi_\theta)\ge\eta/2\right\}
\subseteq
\bigcup_{c\in\mathsf C_d}
\left\{
\left|\widehat\Xi_U(c)-\Xi_\theta(c)\right|
\ge \eta/2
\right\}.
\]
The union bound over the finite alphabet gives
\[
\Xi_\theta^{\otimes m}(G_{\theta,m}^{c})
\le
\sum_{c\in\mathsf C_d}
\Xi_\theta^{\otimes m}\left(
\left|\widehat\Xi_U(c)-\Xi_\theta(c)\right|
\ge \eta/2
\right)
\le
\frac{|\mathsf C_d|}{m\eta^2} .
\]
The right-hand side is independent of \(\theta\) and tends to zero, hence
\[
\sup_{\theta\in\Theta_{\epsilon,d}}
\Xi_\theta^{\otimes m}(G_{\theta,m}^{c})
\le
\frac{|\mathsf C_d|}{m\eta^2}
\longrightarrow0 .
\]
% lean: CausalSmith.Substrate.FiniteSideInformationMinimaxConvergence.empiricalL1_tail
Taking the supremum over \(\theta\) and then using the displayed estimator
\(\Phi_{\delta,m}\) in the minimax infimum yields
\[
\limsup_{m\to\infty}\mathfrak R_m\le \mathfrak R_\star+\delta .
\]
Since \(\delta>0\) was arbitrary,
\[
\limsup_{m\to\infty}\mathfrak R_m\le \mathfrak R_\star .
\]
% lean: CausalSmith.Substrate.FiniteSideInformationMinimaxConvergence.empiricalSideMinimax_limsup_le_exact

Combining \(\mathfrak R_\star\le\mathfrak R_m\) for every \(m\) with the last display gives
\[
\liminf_{m\to\infty}\mathfrak R_m\ge\mathfrak R_\star
\qquad\text{and}\qquad
\limsup_{m\to\infty}\mathfrak R_m\le\mathfrak R_\star .
\]
Thus \(\lim_{m\to\infty}\mathfrak R_m=\mathfrak R_\star\). % lean: CausalSmith.Substrate.FiniteSideInformationMinimaxConvergence.finiteSideInfo_minimax_tendsto
\end{proof}
