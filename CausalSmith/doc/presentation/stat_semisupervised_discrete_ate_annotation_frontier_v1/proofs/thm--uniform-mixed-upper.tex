\begin{proof}[Proof of \cref{obj:thm:uniform-mixed-upper}]
Fix \(0<\epsilon<1/2\).

\begin{enumerate}
\item By \cref{obj:lem:fixed-sample-hybrid-transfer}, choose a universal constant \(C_{\mathrm{tr}}>0\) such that, for every \(n\ge1\), \(m\ge0\), \(d\ge2\), and \(P\in\mathcal M_{d,\epsilon}\),
\[
\mathbb E_{P^{\otimes n}\otimes P_{XA}^{\otimes m}}
\!\left[
\left\{\widehat\tau^{\mathrm{mix}}_{n,m,d}-\tau(P)\right\}^2
\right]
\le
2\,\mathcal R^{\mathrm{Pois}}_{\mathrm{hyb}}(P;n,m,\epsilon)
+\frac{C_{\mathrm{tr}}}{n}.
\]
By \cref{obj:lem:poisson-hybrid-risk}, choose \(C_{\mathrm{Pois},\epsilon}>0\), depending only on \(\epsilon\), such that, for the same range of parameters and laws,
\[
\mathcal R^{\mathrm{Pois}}_{\mathrm{hyb}}(P;n,m,\epsilon)
\le
C_{\mathrm{Pois},\epsilon}\,r_\epsilon(n,m,d).
\]
Set
\[
C_\epsilon:=2C_{\mathrm{Pois},\epsilon}+C_{\mathrm{tr}}.
\]
Then \(C_\epsilon>0\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.uniform_mixed_upper

\item It remains to verify the displayed bound uniformly over \(P\). Fix \(n\ge1\), \(m\ge0\), and \(d\ge2\). For any \(P\in\mathcal M_{d,\epsilon}\), the two estimates above give
\[
\begin{aligned}
\mathbb E_{P^{\otimes n}\otimes P_{XA}^{\otimes m}}
\!\left[
\left\{\widehat\tau^{\mathrm{mix}}_{n,m,d}-\tau(P)\right\}^2
\right]
&\le
2\,\mathcal R^{\mathrm{Pois}}_{\mathrm{hyb}}(P;n,m,\epsilon)
+\frac{C_{\mathrm{tr}}}{n}  \\
&\le
2C_{\mathrm{Pois},\epsilon}\,r_\epsilon(n,m,d)
+C_{\mathrm{tr}}\,r_\epsilon(n,m,d).
\end{aligned}
\]
The last inequality uses the elementary rate comparison
\[
\frac1n\le r_\epsilon(n,m,d),
\]
which follows directly from \cref{obj:def:frontier-rate-handle}, since
\[
\frac1n
\le
\min\!\left\{
1,\frac1n+\frac{d^2}{(n+m)^2\ell_n^2}
\right\}
=
r_\epsilon(n,m,d)
\]
for \(n\ge1\). Therefore
\[
\mathbb E_{P^{\otimes n}\otimes P_{XA}^{\otimes m}}
\!\left[
\left\{\widehat\tau^{\mathrm{mix}}_{n,m,d}-\tau(P)\right\}^2
\right]
\le
(2C_{\mathrm{Pois},\epsilon}+C_{\mathrm{tr}})\,r_\epsilon(n,m,d)
=
C_\epsilon\,r_\epsilon(n,m,d).
\] % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.uniform_mixed_upper

\item Since the preceding bound holds for every \(P\in\mathcal M_{d,\epsilon}\), taking the supremum over the model class yields
\[
\sup_{P\in\mathcal M_{d,\epsilon}}
\mathbb E_{P^{\otimes n}\otimes P_{XA}^{\otimes m}}
\!\left[
\left\{\widehat\tau^{\mathrm{mix}}_{n,m,d}-\tau(P)\right\}^2
\right]
\le
C_\epsilon\,r_\epsilon(n,m,d).
\]
The rate is positive for \(n\ge1\) by \cref{obj:def:frontier-rate-handle}, so the right-hand side is a well-defined finite real number. The asserted uniform bound follows for all \(n\ge1\), \(m\ge0\), and \(d\ge2\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.uniform_mixed_upper
\end{enumerate}
\end{proof}
