\begin{proof}[Proof of \cref{obj:lem:calibrated-envelope-absorption}]
\begin{enumerate}
\item Let \(H_\epsilon\) be the calibration constant appearing in \cref{obj:def:mixed-estimator-handle}, viewed as a real number, and set
\[
K_\epsilon(C_0)
=
4096H_\epsilon 512^2+4096H_\epsilon^2 512+64C_0
+\frac{8192H_\epsilon^2 512^2}{\epsilon^2}
+\frac{128}{\epsilon}+414
+2\left(\frac{64H_\epsilon 512}{\epsilon}\right)^2+72 .
\]
We take
\[
C=1+K_\epsilon(C_0).
\]
Since \(\epsilon>0\), \(C_0>0\), and \(H_\epsilon\ge0\), every displayed summand in \(K_\epsilon(C_0)\) is nonnegative except for none, and hence \(C>0\). % lean: calibrated_envelope_absorption_exists

\item Fix \(n,m,d\in\mathbb N\) with \(n\ge1\), \(d\ge2\), and \(\mathsf C_\epsilon(n,m)\). Write
\[
N=n+m,\qquad \ell_n=\log(en),\qquad A=A_\star^L,\qquad \eta=n^{-12},
\]
where \(L=L(n)\), and define the comparison scale
\[
\Xi
=
\frac1n+\frac{d^2}{N^2\ell_n^2}.
\]
All quantities in the displayed inequalities below are understood as real numbers. The sample-size assumptions give
\[
1\le n,\qquad n\le N,\qquad 1\le d.
\]
Moreover,
\[
\ell_n=\log(en)=1+\log n\ge1,
\]
because \(n\ge1\). From \(L\ge2\) in \cref{obj:def:mixed-estimator-handle}, \(L>0\), and \(A=A_\star^L\ge0\). The calibration inequalities also give \(u>0\), \(t_p>0\), \(t>0\), and \(B>0\): indeed \(u/n\ge1/32\), \(t_p/N\ge1/32\), \(t/N\ge1/32\), and \(8\le t_pB\). % lean: calibrationPredicate_positive_parameters, Ldeg_ge_two

\item The block-proportion part of \(\mathsf C_\epsilon(n,m)\) supplies
\[
\frac{n}{32}\le u,
\qquad
\frac{N}{32}\le t .
\]
The scale definition \(B=H_\epsilon L/\min\{t_p,t\}\), together with
\[
\frac{N}{32}\le t_p,\qquad \frac{N}{32}\le t,
\]
gives
\[
B
=
\frac{H_\epsilon L}{\min\{t_p,t\}}
\le
\frac{32H_\epsilon L}{N}.
\]
The same calculation with the upper block bounds \(t_p\le3N/32\) and \(t\le3N/32\) yields the full auxiliary estimate
\[
\frac{32H_\epsilon L}{3N}\le B\le \frac{32H_\epsilon L}{N}.
\]
Only the upper bound is used below. % lean: Bscale_bounds

\item Let \(b_n\) be the binary-length calibration quantity from \cref{obj:def:mixed-estimator-handle}. The logarithmic estimate gives
\[
\ell_n\le b_n.
\]
The degree calibration inequality is
\[
\frac{c^\circ b_n}{2}\le L,
\qquad c^\circ=\frac1{256},
\]
so \(b_n\le512L\). Therefore
\[
\ell_n\le512L.
\]
The polynomial-size condition \(L^6A_\star^{2L}\le n\) gives
\[
(L^3A_\star^L)^2=L^6A_\star^{2L}\le n.
\]
Finally, since \(n\ge1\),
\[
0\le \eta=n^{-12}\le \frac1n .
\]
% lean: logEN_le_binLen, moment_tests_square_forms, calibrated_envelope_absorption_exists

\item Put
\[
\Theta=L^3A=L^3A_\star^L.
\]
From \(\Theta^2\le n\), the elementary inequality \((x-y)^2\ge0\), applied with
\[
x=\frac{\Theta}{n},
\qquad
y=\frac{d}{N\ell_n},
\]
gives
\[
\frac{\Theta d}{Nn\ell_n}
\le
\frac12\left\{\left(\frac{\Theta}{n}\right)^2
+\left(\frac{d}{N\ell_n}\right)^2\right\}
\le
\frac12\left(\frac1n+\frac{d^2}{N^2\ell_n^2}\right)
=
\frac{\Xi}{2}.
\]
Also, from \(\ell_n\le512L\) and positivity of \(\ell_n,L\),
\[
\frac1L\le\frac{512}{\ell_n},
\qquad
\frac1{L^2}\le\frac{512^2}{\ell_n^2}.
\]
% lean: normalized_young, inverse_schedule, inverse_square_schedule

\item The reciprocal block bounds are
\[
u^{-1}\le\frac{32}{n},
\qquad
t^{-1}\le\frac{32}{n},
\]
because \(u\ge n/32\) and \(t\ge N/32\ge n/32\). Hence
\[
C_0(u^{-1}+t^{-1})
\le
64C_0\frac1n,
\]
\[
\frac4{\epsilon t}
=
\frac4\epsilon\,t^{-1}
\le
\frac{128}{\epsilon}\frac1n,
\]
and
\[
414\eta\le414\frac1n.
\]
Since \(1/n\le\Xi\), these estimates combine to
\[
C_0(u^{-1}+t^{-1})+\frac4{\epsilon t}+414\eta
\le
\left(64C_0+\frac{128}{\epsilon}+414\right)\Xi .
\]
% lean: hybrid_deterministic_envelope_le

\item The two core normalized products obey
\[
\frac{\Theta d}{NnL^2}
\le
\frac{\Theta d}{Nn}\frac{512^2}{\ell_n^2}
\le
512^2\frac{\Theta d}{Nn\ell_n}
\le
512^2\frac{\Xi}{2},
\]
where the middle step uses \(\ell_n\ge1\), and
\[
\frac{\Theta d}{N^2L}
\le
\frac{\Theta d}{N^2}\frac{512}{\ell_n}
\le
512\frac{\Theta d}{Nn\ell_n}
\le
512\frac{\Xi}{2},
\]
where \(N\ge n\) is used in the second inequality. % lean: hybrid_deterministic_envelope_le

\item The first light-envelope term satisfies
\[
4A\frac{\min\{1,dB\}}{u}
\le
4A\frac{dB}{u}
\le
4A\frac{d(32H_\epsilon L/N)}{n/32}
=
4096H_\epsilon\frac{\Theta d}{NnL^2}
\le
(4096H_\epsilon512^2)\frac{\Xi}{2}.
\]
The second light-envelope term satisfies
\[
4AdB^2
\le
4Ad\left(\frac{32H_\epsilon L}{N}\right)^2
=
4096H_\epsilon^2\frac{\Theta d}{N^2L}
\le
(4096H_\epsilon^2 512)\frac{\Xi}{2}.
\]
Adding the two bounds and using \(\Xi\ge0\) gives
\[
4A\left\{\frac{\min\{1,dB\}}u+dB^2\right\}
\le
(4096H_\epsilon512^2+4096H_\epsilon^2 512)\Xi .
\]
% lean: hybrid_deterministic_envelope_le

\item Since \(d\ge1\),
\[
d\le d^2.
\]
Consequently,
\[
\frac{8dB^2}{\epsilon^2L^4}
\le
\frac{8d}{\epsilon^2L^4}
\left(\frac{32H_\epsilon L}{N}\right)^2
=
\frac{8192H_\epsilon^2}{\epsilon^2}
\frac{d}{N^2L^2}.
\]
Using \(1/L^2\le512^2/\ell_n^2\) and \(d\le d^2\),
\[
\frac{d}{N^2L^2}
\le
\frac{512^2d^2}{N^2\ell_n^2}.
\]
Because \(d^2/(N^2\ell_n^2)\le\Xi\), we obtain
\[
\frac{8dB^2}{\epsilon^2L^4}
\le
\frac{8192H_\epsilon^2 512^2}{\epsilon^2}\,\Xi .
\]
% lean: hybrid_deterministic_envelope_le

\item For the squared bias term,
\[
\frac{2dB}{\epsilon L^2}
\le
\frac{2d}{\epsilon L^2}\frac{32H_\epsilon L}{N}
=
\frac{64H_\epsilon}{\epsilon}\frac{d}{NL}
\le
\frac{64H_\epsilon512}{\epsilon}\frac{d}{N\ell_n}.
\]
Both sides are nonnegative, so squaring gives
\[
\left(\frac{2dB}{\epsilon L^2}\right)^2
\le
\left(\frac{64H_\epsilon512}{\epsilon}\right)^2
\frac{d^2}{N^2\ell_n^2}
\le
\left(\frac{64H_\epsilon512}{\epsilon}\right)^2\Xi .
\]
Also \(0\le\eta\le1/n\le1\), hence
\[
(6\eta)^2=36\eta^2\le36\eta\le\frac{36}{n}.
\]
The inequality \((x+y)^2\le2x^2+2y^2\), applied to
\[
x=\frac{2dB}{\epsilon L^2},
\qquad
y=6\eta,
\]
therefore yields
\[
\left(\frac{2dB}{\epsilon L^2}+6\eta\right)^2
\le
\left\{
2\left(\frac{64H_\epsilon512}{\epsilon}\right)^2+72
\right\}\Xi .
\]
% lean: sq_add_le_two, hybrid_deterministic_envelope_le

\item Combining the light-envelope, reciprocal, variance-polynomial, and squared-bias estimates gives
\[
\begin{aligned}
&4A\left\{\frac{\min\{1,dB\}}u+dB^2\right\}
+C_0(u^{-1}+t^{-1})
+\frac{8dB^2}{\epsilon^2L^4}
+\frac4{\epsilon t}
+414\eta
+\left(\frac{2dB}{\epsilon L^2}+6\eta\right)^2  \\
&\qquad\le
K_\epsilon(C_0)\,\Xi .
\end{aligned}
\]
Since \(0\le\Xi\) and \(K_\epsilon(C_0)\le C\), the right-hand side is at most \(C\Xi\). Substituting back
\[
A=A_\star^L,\qquad \eta=n^{-12},\qquad
\Xi=\frac1n+\frac{d^2}{N^2\ell_n^2},
\]
with \(N=n+m\) and \(\ell_n=\log(en)\) from \cref{obj:def:frontier-rate-handle}, is precisely the asserted inequality. % lean: hybrid_deterministic_envelope_le, calibrated_envelope_absorption_exists
\end{enumerate}
\end{proof}
