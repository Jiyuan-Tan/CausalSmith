\begin{proof}[Proof of \cref{obj:thm:common-marginal-converse}]
\begin{enumerate}
\item Fix \(0<\epsilon<1/2\).  By \cref{obj:lem:common-marginal-canonical-certificate-family}, there are positive numerical constants
\[
C_{\mathrm{fix}}>0,\qquad c_{\mathrm{risk}}>0,
\]
and, at this \(\epsilon\), calibration constants as in \cref{obj:def:common-marginal-prior-handle}, together with
\[
\rho\in(0,1),\qquad c_{\mathrm{floor}}>0,
\qquad c_{\mathrm{floor}}\le c_{\mathrm{risk}}.
\]
Write \(C_d>0\) for the dimension constant in this calibration, and write \(c_{\mathrm{gap}}>0\) for the calibration's dual-gap constant.  For every \(n,m,d\) with \(n\ge1\) and \(d\ge2\), \cref{obj:lem:common-marginal-canonical-certificate-family} supplies a handle \(H\) satisfying
\[
\mathcal L_{\Pi_0(H)}(P_{XA})
=
\mathcal L_{\Pi_1(H)}(P_{XA})
\]
and the floor certificate
\[
c_{\mathrm{floor}}\frac1n\le R_\epsilon(n,m,d).
\]
For this handle, put
\[
D(n,m,d):=
\min\left\{1,\frac{d^2}{(n+m)^2\log(en)^2}\right\}
\]
and
\[
\Delta_H:=c_{\mathrm{gap}}\,k(H)\,a(H),
\]
where \(k(H)\) and \(a(H)\) are the rare-cell count and shift in \cref{obj:def:common-marginal-prior-handle}.  In the finite-atom branch, the same cited certificate gives the nonparametric dimension comparison
\[
C_d\,D(n,m,d)\le \Delta_H^2
\]
and the fixed-sample lower bound attached to the canonical handle,
\[
c_{\mathrm{floor}}\Delta_H^2\le R_\epsilon(n,m,d).
\]
The displayed lower bound follows once the handle's common-marginal witness, total-variation bound, center separation, and variance absorption satisfy the hypotheses of \cref{obj:lem:fixed-sample-common-marginal-transfer}.
In the complementary branch, \cref{obj:lem:common-marginal-canonical-certificate-family} gives directly
\[
C_dD(n,m,d)\le \frac{c_{\mathrm{floor}}}{n}.
\] % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_canonical_certificate_family

\item Define
\[
c_{\mathrm{lb}}:=
\min\left\{
\frac{c_{\mathrm{floor}}}{2},
\frac{C_d}{2},
\frac{c_{\mathrm{floor}}C_d}{2}
\right\}.
\]
Since \(c_{\mathrm{floor}}>0\) and \(C_d>0\), this constant is positive. % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_converse

\item Fix \(n,m,d\) with \(n\ge1\) and \(d\ge2\), and take the handle \(H\) from the certificate package.  Put
\[
Q(n,m,d):=
\frac{d^2}{(n+m)^2\log(en)^2},
\qquad
D(n,m,d)=\min\{1,Q(n,m,d)\}.
\]
Both \(Q(n,m,d)\) and \(D(n,m,d)\) are nonnegative.  The frontier rate in \cref{obj:def:frontier-rate-handle} satisfies
\[
r_\epsilon(n,m,d)
=
\min\left\{1,\frac1n+Q(n,m,d)\right\}
\le
\frac1n+D(n,m,d).
\]
Indeed, if \(Q(n,m,d)\le1\), the right side is \(1/n+Q(n,m,d)\), which dominates the minimum.  If \(Q(n,m,d)>1\), then \(D(n,m,d)=1\), and the left side is at most \(1\le 1/n+1\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_converse

\item Suppose first that \(H\) is in the finite-atom branch.  Since \(c_{\mathrm{floor}}>0\), multiplying the finite-branch dimension comparison by \(c_{\mathrm{floor}}\) and then using the transferred fixed-sample lower bound gives
\[
c_{\mathrm{floor}}C_dD(n,m,d)
\le
c_{\mathrm{floor}}\Delta_H^2
\le
R_\epsilon(n,m,d).
\]
Together with the floor inequality,
\[
c_{\mathrm{floor}}\frac1n\le R_\epsilon(n,m,d),
\]
this supplies the two lower bounds used in the averaging step.  Using the definition of \(c_{\mathrm{lb}}\) and the bound from the previous step,
\[
\begin{aligned}
c_{\mathrm{lb}} r_\epsilon(n,m,d)
&\le
c_{\mathrm{lb}}\left(\frac1n+D(n,m,d)\right)\\
&=
c_{\mathrm{lb}}\frac1n+c_{\mathrm{lb}} D(n,m,d)\\
&\le
\frac{c_{\mathrm{floor}}}{2}\frac1n
+\frac{c_{\mathrm{floor}}C_d}{2}D(n,m,d)\\
&=
\frac12\left(
c_{\mathrm{floor}}\frac1n
+
c_{\mathrm{floor}}C_dD(n,m,d)
\right)\\
&\le R_\epsilon(n,m,d).
\end{aligned}
\]
Thus the asserted lower bound holds in the finite-atom branch. % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_converse

\item It remains to treat the complementary branch.  In this branch, the certificate clause in \cref{obj:lem:common-marginal-canonical-certificate-family} gives directly
\[
C_dD(n,m,d)
\le
\frac{c_{\mathrm{floor}}}{n}
=
c_{\mathrm{floor}}\frac1n.
\]
Together with the floor inequality
\[
c_{\mathrm{floor}}\frac1n\le R_\epsilon(n,m,d),
\]
we obtain
\[
\begin{aligned}
c_{\mathrm{lb}} r_\epsilon(n,m,d)
&\le
c_{\mathrm{lb}}\left(\frac1n+D(n,m,d)\right)\\
&=
c_{\mathrm{lb}}\frac1n+c_{\mathrm{lb}} D(n,m,d)\\
&\le
\frac{c_{\mathrm{floor}}}{2}\frac1n
+\frac{C_d}{2}D(n,m,d)\\
&\le
\frac12\left(
c_{\mathrm{floor}}\frac1n+
c_{\mathrm{floor}}\frac1n
\right)\\
&=
c_{\mathrm{floor}}\frac1n\\
&\le R_\epsilon(n,m,d).
\end{aligned}
\]
This proves the risk lower bound in the complementary branch as well. % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_converse

\item Combining the two branches, for every \(n,m,d\) with \(n\ge1\), \(m\ge0\), and \(d\ge2\), the handle \(H\) supplies priors \(\Pi_0(H)\) and \(\Pi_1(H)\) with common random treatment-covariate marginal law and satisfies
\[
c_{\mathrm{lb}} r_\epsilon(n,m,d)\le R_\epsilon(n,m,d).
\]
Taking the theorem's lower-bound constant \(c_\epsilon\) to be \(c_{\mathrm{lb}}\), and using \(c_{\mathrm{lb}}>0\) with the fixed calibration constants chosen above, gives the asserted conclusion. % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_converse
\end{enumerate}
\end{proof}
