\begin{proof}[Proof of \cref{obj:lem:poisson-inverse-marginal-energy-scalar}]
\begin{enumerate}
\item First derive the scalar bidirectional energy bound.  The lower and upper intensity bounds, together with \(0<\epsilon\le1\) and \(w\ge0\), imply
\[
0\le \lambda_0,\qquad 0\le \lambda_1,
\]
and, for the denominator attached to \(\lambda_1\),
\[
\epsilon(1+w)=\epsilon+\epsilon w\le 1+\lambda_1,
\qquad
1+w\le \epsilon^{-1}(1+\lambda_1),
\qquad
w\le \epsilon^{-1}(1+\lambda_1).
\]
Using also \(\lambda_0\le w\), we have
\[
(1+w)\lambda_0
   \le (1+w)w
   \le \epsilon^{-2}(1+\lambda_1)^2,
\]
and hence
\[
\lambda_0 r_{2,1}
   \le 8\,{\lambda_0\over (1+\lambda_1)^2}
   \le {8\epsilon^{-2}\over 1+w}.
\]
Similarly,
\[
(1+w)\lambda_1\lambda_0
 \le \epsilon^{-2}(1+\lambda_1)^4,
\qquad
(1+w)\lambda_1\lambda_0^2
 \le \epsilon^{-3}(1+\lambda_1)^4,
\]
so the fourth-moment rate gives
\[
\lambda_1(\lambda_0+\lambda_0^2)r_{4,1}
 \le
 {96(\epsilon^{-2}+\epsilon^{-3})\over 1+w}.
\]
Combining these two estimates yields
\[
\lambda_1(\lambda_0+\lambda_0^2)r_{4,1}
 +\lambda_0 r_{2,1}
 \le {c_{\mathrm{mar}}\over 1+w}.
\]
The same calculation with the roles of \(0\) and \(1\) interchanged gives
\[
\lambda_0(\lambda_1+\lambda_1^2)r_{4,0}
 +\lambda_1 r_{2,0}
 \le {c_{\mathrm{mar}}\over 1+w}.
\]
Therefore
\[
\bigl\{\lambda_1(\lambda_0+\lambda_0^2)r_{4,1}
 +\lambda_0 r_{2,1}\bigr\}
+
\bigl\{\lambda_0(\lambda_1+\lambda_1^2)r_{4,0}
 +\lambda_1 r_{2,0}\bigr\}
\le {2c_{\mathrm{mar}}\over 1+w}.
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.poisson_inverse_bidirectional_poincare_energy_scalar

\item Define the three real quantities
\[
\eta_0=q_0u^{-1}+q_0^2,\qquad
\eta_1=q_1u^{-1}+q_1^2,\qquad
\eta_p=pu^{-1}+p^2 .
\]
Since \(u>0\), we have \(u^{-1}\ge0\).  For each displayed outcome mass,
\[
\eta_p-\eta_0
  =(p-q_0)(u^{-1}+p+q_0)\ge0,
\qquad
\eta_p-\eta_1
  =(p-q_1)(u^{-1}+p+q_1)\ge0,
\]
because \(0\le q_0,q_1\le p\).  Thus
\[
\eta_0\le \eta_p,\qquad \eta_1\le \eta_p,
\qquad \eta_p\ge0.
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.poisson_inverse_marginal_energy_with_outcome_rates

\item All coefficients multiplying \(\eta_0\) and \(\eta_1\) below are nonnegative, by the nonnegativity of the intensities, the moment rates, and \(\lambda_a+\lambda_a^2\).  Hence the comparison from the previous step gives
\[
\begin{aligned}
&\lambda_1\bigl\{\eta_1(\lambda_0+\lambda_0^2)r_{4,1}
 +\eta_0 r_{2,0}\bigr\}
+\lambda_0\bigl\{\eta_0(\lambda_1+\lambda_1^2)r_{4,0}
 +\eta_1 r_{2,1}\bigr\} \\
&\qquad\le
\eta_p\Bigl[
\bigl\{\lambda_1(\lambda_0+\lambda_0^2)r_{4,1}
 +\lambda_0 r_{2,1}\bigr\}
+
\bigl\{\lambda_0(\lambda_1+\lambda_1^2)r_{4,0}
 +\lambda_1 r_{2,0}\bigr\}
\Bigr].
\end{aligned}
\]
Applying the scalar bidirectional bound from Step 1 and using \(\eta_p\ge0\), we obtain
\[
\lambda_1\bigl\{(q_1u^{-1}+q_1^2)(\lambda_0+\lambda_0^2)r_{4,1}
 +(q_0u^{-1}+q_0^2)r_{2,0}\bigr\}
+\lambda_0\bigl\{(q_0u^{-1}+q_0^2)(\lambda_1+\lambda_1^2)r_{4,0}
 +(q_1u^{-1}+q_1^2)r_{2,1}\bigr\}
\le
(pu^{-1}+p^2){2c_{\mathrm{mar}}\over 1+w}.
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.poisson_inverse_marginal_energy_with_outcome_rates
\end{enumerate}
\end{proof}
