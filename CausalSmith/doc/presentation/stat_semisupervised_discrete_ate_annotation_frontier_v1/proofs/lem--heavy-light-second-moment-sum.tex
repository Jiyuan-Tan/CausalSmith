\begin{proof}[Proof of \cref{obj:lem:heavy-light-second-moment-sum}]
Let
\[
\mathsf{Heavy}_B=\{x\in[d]:p_x>B\},
\qquad
p_x=\mathbb P_P\{X=x\}.
\]
The quantities \(u,t_p,t,L,B,k_0\) are those fixed in \cref{obj:def:mixed-estimator-handle}. For each \(x\), put
\[
\Xi_x
=
\int
\left[
\frac{S_{1x}}u\widehat W_{1,L,x}
-
\frac{S_{0x}}u\widehat W_{0,L,x}
\right]^2\,d\mu_x .
\]
Thus the left-hand side is \(\sum_{x\in\mathsf{Heavy}_B}\pi_x\Xi_x\).

\begin{enumerate}
\item The cell masses satisfy
\[
0\le p_x\le1
\quad\text{for every }x\in[d],
\qquad
\sum_{x\in[d]}p_x=1.
\]
Hence, for the finite set \(\mathsf{Heavy}_B\),
\[
\sum_{x\in\mathsf{Heavy}_B}p_x\le1,
\qquad
\sum_{x\in\mathsf{Heavy}_B}p_x^2
\le
\sum_{x\in\mathsf{Heavy}_B}p_x
\le1.
\]
The calibration predicate gives \(u,t_p,t,B>0\), \(L\ge2\), \(L\le tB\), and \(1/32\le u/n\). Since \(n\ge1\),
\[
u\ge \frac n{32}\ge \frac1{32},
\qquad
u^{-1}\le32.
\]
These inequalities also cover the case \(\mathsf{Heavy}_B=\varnothing\), where the sums are zero. % lean: cellMass_sum_eq_one_annotation

\item Fix \(x\in\mathsf{Heavy}_B\), and define
\[
y_x=\frac{p_x}{B}.
\]
Because \(B>0\) and \(p_x>B\), \(y_x\ge1\). Also \(p_x=By_x\). The pilot count is
\[
J_x=J_{0x}+J_{1x},
\qquad
J_{ax}\sim\operatorname{Poi}(t_ps_{ax}),
\]
with independent arms and \(s_{0x}+s_{1x}=p_x\). Since \(k_0=\lfloor t_pB/4\rfloor\) and \(t_pB\ge0\),
\[
k_0\le \frac{t_pB}{4}<\frac{t_pp_x}{4}.
\]
The standard Poisson lower-tail bound therefore gives
\[
\pi_x
=
\mathbb P\{J_x\le k_0\}
\le
\exp\!\left(-\frac{t_pp_x}{4}\right).
\]
For the light-polynomial second moment, \cref{obj:lem:explicit-chebyshev-calibration} supplies the factorial-lift bound
\[
\mathbb E\!\left[\widehat W_{a,L,x}^{\,2}\right]
\le
A_\star^L\left(1+\frac{p_x}{B}\right)^{2L},
\qquad a\in\{0,1\}.
\]
Writing \(q_{ax}=\mathbb P_P\{X=x,A=a,Y=1\}\), independence of \(S_{ax}\) and the factorial counts gives
\[
\mathbb E\!\left[
\left(\frac{S_{ax}}u\widehat W_{a,L,x}\right)^2
\right]
=
\left(\frac{q_{ax}}u+q_{ax}^2\right)
\mathbb E\!\left[\widehat W_{a,L,x}^{\,2}\right].
\]
Since \(0\le q_{ax}\le p_x\), and using \((r-s)^2\le2r^2+2s^2\),
\[
\Xi_x
\le
4\left(\frac{p_x}{u}+p_x^2\right)
A_\star^L\left(1+\frac{p_x}{B}\right)^{2L}.
\]
Consequently,
\[
\pi_x\Xi_x
\le
4\left(\frac{p_x}{u}+p_x^2\right)
A_\star^L(1+y_x)^{2L}
\exp\!\left(-\frac{t_pp_x}{4}\right).
\] % lean: calibrated_heavy_cell_light_secondMoment_le

\item The definition \(B=H_\epsilon L/\min(t_p,t)\), together with positivity of \(t_p,t\), gives
\[
H_\epsilon L\le t_pB.
\]
Therefore
\[
\exp\!\left(-\frac{t_pp_x}{4}\right)
\le
\exp\!\left(-\frac{H_\epsilon L y_x}{4}\right).
\]
The numerical choice of \(H_\epsilon\) in \cref{obj:def:mixed-estimator-handle} includes
\[
15+\frac{24}{c^\circ}\le \frac{H_\epsilon}{4}.
\]
Using \(\log A_\star\le13\), \(\log(1+y_x)\le y_x\), and \(y_x\ge1\),
\[
\begin{aligned}
\log\!\left[
A_\star^L(1+y_x)^{2L}
\exp\!\left(-\frac{H_\epsilon L y_x}{4}\right)
\right]
&=
L\log A_\star+2L\log(1+y_x)-\frac{H_\epsilon L y_x}{4} \\
&\le
15Ly_x-\frac{H_\epsilon L y_x}{4} \\
&\le
-\frac{24Ly_x}{c^\circ}.
\end{aligned}
\]
Thus
\[
A_\star^L(1+y_x)^{2L}
\exp\!\left(-\frac{t_pp_x}{4}\right)
\le
\exp\!\left(-\frac{24Ly_x}{c^\circ}\right).
\]
The calibration predicate also gives \(c^\circ b_n/2\le L\), where \(b_n=1+\lceil\log_2 n\rceil\). Since \(y_x\ge1\),
\[
12b_n\le \frac{24Ly_x}{c^\circ}.
\]
Finally \(b_n\) dominates \(\log n\) for \(n\ge1\), so
\[
\exp(-12b_n)\le n^{-12}.
\]
Combining the last three displays yields
\[
A_\star^L(1+y_x)^{2L}
\exp\!\left(-\frac{t_pp_x}{4}\right)
\le n^{-12}.
\]
Substituting this into the bound from the previous step gives, for every \(x\in\mathsf{Heavy}_B\),
\[
\pi_x\Xi_x
\le
4\left(\frac{p_x}{u}+p_x^2\right)n^{-12}.
\] % lean: factorial_growth_mul_heavy_exp_le

\item Summing the preceding cellwise inequality over \(x\in\mathsf{Heavy}_B\),
\[
\sum_{x\in\mathsf{Heavy}_B}\pi_x\Xi_x
\le
4\left(
\frac{\sum_{x\in\mathsf{Heavy}_B}p_x}{u}
+
\sum_{x\in\mathsf{Heavy}_B}p_x^2
\right)n^{-12}.
\]
By the bounds from the first step,
\[
\frac{\sum_{x\in\mathsf{Heavy}_B}p_x}{u}
\le
u^{-1}
\le32,
\qquad
\sum_{x\in\mathsf{Heavy}_B}p_x^2\le1.
\]
Therefore
\[
\sum_{x\in\mathsf{Heavy}_B}\pi_x\Xi_x
\le
4(32+1)n^{-12}
=
132\,n^{-12}.
\]
This is the claimed bound. % lean: sum_heavy_light_secondMoment_le
\end{enumerate}
\end{proof}
