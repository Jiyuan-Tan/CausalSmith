\begin{proof}[Proof of \cref{obj:lem:hybrid-heavy-branch-mean-bound}]
Let
\[
\mathcal H=\{x\in[d]:p_x>B\},\qquad
E^L_x=\mathbb E_{\mu_x}P_x,\qquad
E^H_x=\mathbb E_{\mu_x}H_x .
\]
The calibration inequalities give \(u>0\), \(t_p>0\), \(t>0\), and \(B>0\). Also \(0\le p_x\le1\) for every \(x\), and \(\sum_xp_x=1\), hence
\[
\sum_{x\in\mathcal H}p_x\le1 .
\]
% lean: calibrationPredicate_positive_parameters, cellMass_mem_unitInterval, cellMass_sum_eq_one_annotation

Fix \(x\in\mathcal H\). We first establish the calibrated pilot-tail estimate
\[
\pi_x\le n^{-12}.
\]
For \(a\in\{0,1\}\), write \(s_{ax}\) for the treatment-covariate mass in arm \(a\), so \(p_x=s_{0x}+s_{1x}\). Under the pilot law, \(J_x=J_{0x}+J_{1x}\) has Poisson mean
\[
\lambda_x=t_p(s_{0x}+s_{1x})=t_pp_x,
\]
because the two pilot-arm counts are independent Poisson variables. The definition of \(k_0\) gives \(k_0\le t_pB/4\), and \(x\in\mathcal H\) gives \(B<p_x\). Since \(t_p>0\),
\[
k_0<\frac{t_pp_x}{4}=\frac{\lambda_x}{4}.
\]
For a Poisson variable with mean \(\lambda_x\), the following lower-tail estimate applies whenever the cutoff is below one quarter of the mean:
\[
\mathbb P\{J_x\le k_0\}\le \exp\{-\lambda_x/4\}.
\]
Indeed, for every \(\alpha>0\), Markov's inequality applied to \(e^{-\alpha J_x}\) gives
\[
\mathbb P\{J_x\le k_0\}
\le
\exp\{\lambda_x(e^{-\alpha}-1)+\alpha k_0\}.
\]
If \(k_0=0\), this is \(e^{-\lambda_x}\le e^{-\lambda_x/4}\). If \(k_0\ge1\), take \(\alpha=\log(\lambda_x/k_0)\). With \(\zeta_x=k_0/\lambda_x\le1/4\), the exponent becomes
\[
\lambda_x\{-1+\zeta_x-\zeta_x\log\zeta_x\}
\le -\lambda_x/4,
\]
since \(z-z\log z\le 1/4+(\log 4)/4<3/4\) on \(0\le z\le1/4\). Therefore
\[
\pi_x=\mathbb P\{J_x\le k_0\}\le \exp\{-t_pp_x/4\}.
\]
Let \(h_n=1+\lceil\log_2 n\rceil\) be the binary-length calibration quantity in \cref{obj:def:mixed-estimator-handle}. The tail calibration inequality
\[
15L+12h_n\le \frac{t_pB}{4},
\]
together with \(p_x>B\), yields
\[
\frac{t_pp_x}{4}\ge 12h_n.
\]
Thus
\[
\pi_x\le \exp\{-12h_n\}\le n^{-12},
\]
where the last inequality follows from \(n\ge1\) and \(h_n\ge \log n\).
% lean: calibrated_heavy_cell_pilot_le

The heavy inverse-count mean satisfies
\[
|E^H_x|\le 3p_x .
\]
To prove this, first write
\[
\theta_x=p_x(\mu_{1x}-\mu_{0x}),
\]
where, for \(a\in\{0,1\}\), \(q_{ax}\) is the outcome-marked mass, \(s_{ax}\) is the treatment-covariate mass, and the arm outcome regression is characterized by
\[
q_{ax}=s_{ax}\mu_{ax},\qquad 0\le \mu_{ax}\le 1 .
\]
For the arm-specific inverse-count means set
\[
\overline H_{a,x}
=
\mathbb E_{\mu_x}\!\left[
\frac{S_{ax}}{u}\,
\frac{K_{0x}+K_{1x}+1}{K_{ax}+1}
\right],\qquad a\in\{0,1\}.
\]
By the definition of the heavy cell contribution,
\[
E^H_x=\overline H_{1,x}-\overline H_{0,x}.
\]
We next compute each arm mean. Since \(S_{ax}\) is independent of \((K_{0x},K_{1x})\) and \(\mathbb E S_{ax}=u q_{ax}\),
\[
\overline H_{a,x}
=
q_{ax}\,
\mathbb E\!\left[\frac{K_{0x}+K_{1x}+1}{K_{ax}+1}\right].
\]
For the chosen arm \(a\), the fraction inside the expectation is
\[
\frac{K_{0x}+K_{1x}+1}{K_{ax}+1}
=
1+\frac{K_{1-a,x}}{K_{ax}+1},
\]
where \(1-a\) denotes the opposite arm. Therefore \cref{obj:lem:poisson-inverse-arm-mean-formula}, applied with this arm and its opposite arm, gives the exact identity
\[
\overline H_{a,x}
=
 p_x\mu_{ax}-(p_x-s_{ax})\mu_{ax}\exp\{-t s_{ax}\}.
\]
Since \(x\in\mathcal H\), the cell is occupied. Hence \(P\in\mathcal M_{d,\epsilon}\) and \cref{obj:def:model-class,obj:ass:overlap} give, for both arms,
\[
\epsilon p_x\le s_{ax}\le p_x .
\]
Using this, \(0\le\mu_{ax}\le1\), and \(t>0\), the exact identity yields the armwise envelope
\[
\begin{aligned}
|\overline H_{a,x}-p_x\mu_{ax}|
&=\left|-(p_x-s_{ax})\mu_{ax}\exp\{-t s_{ax}\}\right|\\
&=(p_x-s_{ax})\mu_{ax}\exp\{-t s_{ax}\}\\
&\le p_x\exp\{-\epsilon t p_x\}.
\end{aligned}
\]
The first equality uses the displayed mean formula, the second uses nonnegativity of every factor, and the last step uses \(p_x-s_{ax}\le p_x\) and \(\exp\{-t s_{ax}\}\le\exp\{-\epsilon t p_x\}\). Hence
\[
\begin{aligned}
|E^H_x-\theta_x|
&=\left|\bigl(\overline H_{1,x}-p_x\mu_{1x}\bigr)
      -\bigl(\overline H_{0,x}-p_x\mu_{0x}\bigr)\right|\\
&\le |\overline H_{1,x}-p_x\mu_{1x}|
   + |\overline H_{0,x}-p_x\mu_{0x}|\\
&\le 2p_x\exp\{-\epsilon t p_x\}.
\end{aligned}
\]
% lean: integral_hybridHeavyPools_bias_le, poisson_inverse_arm_mean_formula, poisson_inverse_arm_bias_envelope
Since \(0\le p_x\le1\) and \(\mu_{1x},\mu_{0x}\in[0,1]\),
\[
|\theta_x|\le p_x .
\]
Also, because \(\epsilon>0\), \(t>0\), and \(p_x\ge0\),
\[
\exp\{-\epsilon t p_x\}\le1.
\]
Thus
\[
|E^H_x|
\le |E^H_x-\theta_x|+|\theta_x|
\le 2p_x+p_x
=3p_x .
\]
% lean: abs_integral_hybridHeavyPools_le_three_cellMass

Now \(\eta_x\le1\), \(\pi_x\ge0\), and \(\eta_x\ge0\). Using
\[
(E^L_x-E^H_x)^2\le 2(E^L_x)^2+2(E^H_x)^2
\]
and the preceding heavy-mean bound,
\[
(E^H_x)^2\le 9p_x^2\le 9p_x,
\]
where the last inequality uses \(0\le p_x\le1\). Hence
\[
\begin{aligned}
\pi_x\eta_x(E^L_x-E^H_x)^2
&\le \pi_x(E^L_x-E^H_x)^2\\
&\le 2\pi_x(E^L_x)^2+2\pi_x(E^H_x)^2\\
&\le 2\pi_x(E^L_x)^2+18p_x n^{-12}.
\end{aligned}
\]
% lean: sq_sub_le_two_sq_add_two_sq

Summing this pointwise estimate over \(x\in\mathcal H\) gives
\[
\sum_{x\in\mathcal H}\pi_x\eta_x(E^L_x-E^H_x)^2
\le
2\sum_{x\in\mathcal H}\pi_x(E^L_x)^2
+
18n^{-12}\sum_{x\in\mathcal H}p_x .
\]
% lean: sum_heavy_branchMean_sq_le

It remains to bound the first sum. For each \(x\), nonnegativity of variance under \(\mu_x\) gives
\[
(E^L_x)^2
\le
\mathbb E_{\mu_x}\!\left[P_x^2\right].
\]
Consequently,
\[
\sum_{x\in\mathcal H}\pi_x(E^L_x)^2
\le
\sum_{x\in\mathcal H}\pi_x\,\mathbb E_{\mu_x}\!\left[P_x^2\right].
\]
The heavy-light second-moment bound in \cref{obj:lem:heavy-light-second-moment-sum}, applied with the same values of \(u,t_p,t,L,B,k_0\), gives
\[
\sum_{x\in\mathcal H}\pi_x\,\mathbb E_{\mu_x}\!\left[P_x^2\right]
\le 132n^{-12}.
\]
% lean: sum_heavy_light_secondMoment_le, hybridLightPools_memLp

Combining the last two displays with \(\sum_{x\in\mathcal H}p_x\le1\),
\[
\sum_{x\in\mathcal H}\pi_x\eta_x(E^L_x-E^H_x)^2
\le
2\cdot132\,n^{-12}+18\,n^{-12}
=
282\,n^{-12}.
\]
This is the desired bound.
% lean: sum_heavy_branchMean_sq_le
\end{proof}
