\begin{proof}[Proof of \cref{obj:thm:known-marginal-boundary}]
Fix \(0<\epsilon<1/2\).  We prove the result with
\[
c_\epsilon=\frac1{100},
\qquad
C_\epsilon=2\epsilon^{-1}.
\]
The first constant is positive, and \(c_\epsilon\le C_\epsilon\) because \(0<\epsilon<1/2\) gives
\[
\frac1{100}\le \frac{2}{\epsilon}
\quad\Longleftrightarrow\quad
\frac{\epsilon}{100}\le 2 .
\]
It remains to verify the two risk inequalities uniformly over \(n\ge1\) and \(d\ge2\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.known_marginal_boundary

\begin{enumerate}
\item
For the lower bound, put
\[
\delta:=\frac{2}{5\sqrt n}.
\]
Then \(\delta\ge0\), \(\delta\le2/5\), and
\[
\delta^2=\frac{4}{25n}.
\]
Consider two laws on \([d]\times\{0,1\}^2\).  Under \(P_{\circ}\), the covariate is uniform on \([d]\), the propensity is \(1/2\), and both outcome regressions equal \(1/2\).  Under \(P_{\delta}\), the covariate is again uniform, the propensity is \(1/2\), the control regression is \(1/2\), and the treated regression is \(1/2+\delta\).  Since \(\delta\le2/5\), the perturbed treated regression lies in \([0,1]\).  Both laws belong to \(\mathcal M_{d,\epsilon}\) from \cref{obj:def:model-class}, because their occupied-cell propensity is \(1/2\), which satisfies the overlap condition in \cref{obj:ass:overlap}. % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.parametricFloorPertLaw_model

\item
The two laws have the same supplied treatment-covariate table:
\[
(P_{\circ})_{XA}(x,a)=\frac{1}{2d}=(P_{\delta})_{XA}(x,a),
\qquad x\in[d],\ a\in\{0,1\}.
\]
Their ATE values, computed from \cref{obj:def:ate-functional}, are
\[
\tau(P_{\circ})=0,
\qquad
\tau(P_{\delta})
=\sum_{x=1}^d \frac1d\Bigl(\frac12+\delta-\frac12\Bigr)
=\delta .
\]
Thus, with the testing half-separation
\[
\eta_{\mathrm{test}}:=\frac{\delta}{2},
\]
we have
\[
2\eta_{\mathrm{test}}=\delta
=\bigl|\tau(P_{\circ})-\tau(P_{\delta})\bigr|.
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.parametricFloorPertLaw_ate

\item
The product laws are sufficiently close in total variation.  For one labeled observation, with the perturbed law compared to the null law, the chi-squared increment is
\[
\chi_1^2
:=
\frac{(1/2)\delta^2}{(1/2)(1-1/2)}
=2\delta^2
=\frac{8}{25n}.
\]
Thus
\[
n\chi_1^2=\frac{8}{25}\le \log 2 .
\]
The product chi-squared divergence tensorizes as
\[
1+\chi^2\!\left(P_{\delta}^{\otimes n}\Vert P_{\circ}^{\otimes n}\right)
=\bigl(1+\chi_1^2\bigr)^n .
\]
Since \(1+u\le e^u\) for \(u\ge0\), the preceding calibration gives
\[
1+\chi^2\!\left(P_{\delta}^{\otimes n}\Vert P_{\circ}^{\otimes n}\right)
\le
\exp(n\chi_1^2)
\le
\exp(\log 2)=2,
\]
and hence
\[
\chi^2\!\left(P_{\delta}^{\otimes n}\Vert P_{\circ}^{\otimes n}\right)
\le 1.
\]
The perturbed product law is absolutely continuous with respect to the null product law because every one-observation atom under the null has positive mass.  Therefore the standard Cauchy--Schwarz comparison between total variation and chi-squared divergence yields
\[
\operatorname{TV}\!\left(P_{\circ}^{\otimes n},P_{\delta}^{\otimes n}\right)
=
\operatorname{TV}\!\left(P_{\delta}^{\otimes n},P_{\circ}^{\otimes n}\right)
\le
\frac12
\sqrt{\chi^2\!\left(P_{\delta}^{\otimes n}\Vert P_{\circ}^{\otimes n}\right)}
\le \frac12 .
\]
% lean: Causalean.Estimation.MinimaxATE.Parametric.tvDist_productLaw_le_half

\item
Let \(T\in\mathcal T^{\mathrm{known}}_{n,d}\) be any known-marginal estimator, and define
\[
f(\mathcal L_n):=
T\!\left(\mathcal L_n,(P_{\circ})_{XA}\right)
=
T\!\left(\mathcal L_n,(P_{\delta})_{XA}\right),
\]
where the equality uses the common supplied table.  Since the two targets are separated by \(2\eta_{\mathrm{test}}\), for every sample point at least one of
\[
\bigl|f-\tau(P_{\circ})\bigr|\ge \eta_{\mathrm{test}},
\qquad
\bigl|f-\tau(P_{\delta})\bigr|\ge \eta_{\mathrm{test}}
\]
holds.  The total-variation bound therefore gives
\[
\max\!\left\{
P_{\circ}^{\otimes n}\!\left(\bigl|f-\tau(P_{\circ})\bigr|\ge \eta_{\mathrm{test}}\right),
P_{\delta}^{\otimes n}\!\left(\bigl|f-\tau(P_{\delta})\bigr|\ge \eta_{\mathrm{test}}\right)
\right\}
\ge \frac{1-\operatorname{TV}(P_{\circ}^{\otimes n},P_{\delta}^{\otimes n})}{2}
\ge \frac14 .
\]
For either law \(P_\star\in\{P_{\circ},P_{\delta}\}\),
\[
\eta_{\mathrm{test}}^2\,
P_\star^{\otimes n}\!\left(\bigl|f-\tau(P_\star)\bigr|\ge \eta_{\mathrm{test}}\right)
\le
\mathbb E_{P_\star^{\otimes n}}
\!\left[\bigl(T(\mathcal L_n,(P_\star)_{XA})-\tau(P_\star)\bigr)^2\right].
\]
Taking the larger of the two risks, then the supremum over \(\mathcal M_{d,\epsilon}\) and the infimum in \cref{obj:def:known-marginal-risk}, yields
\[
R_\epsilon^{\mathrm{known}}(n,d)
\ge
\frac{\eta_{\mathrm{test}}^2}{4}
=
\frac{\delta^2}{16}
=
\frac{1}{100n}.
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.knownMarginalRisk_ge_two_model_tv

\item
For the upper bound, take any \(P\in\mathcal M_{d,\epsilon}\).  For a supplied table \(\Lambda=(\Lambda_{x,a})_{x,a}\), define the known-marginal score on a record \((x,a,y)\) by
\[
\psi_{\Lambda}(x,a,y)
:=
\begin{cases}
(\Lambda_{x,0}+\Lambda_{x,1})/\Lambda_{x,1}, & y=1,\ a=1,\\
-(\Lambda_{x,0}+\Lambda_{x,1})/\Lambda_{x,0}, & y=1,\ a=0,\\
0, & y=0,
\end{cases}
\]
with any ratio having denominator \(0\) assigned value \(0\).  The raw estimator is
\[
\overline\psi_{\Lambda}(\mathcal L_n)
:=
\frac1n\sum_{i=1}^n \psi_{\Lambda}(X_i,A_i,Y_i),
\]
and the estimator used in the risk is its projection to \([-1,1]\).  This projection is measurable on the finite sample space and weakly decreases squared loss because \(\tau(P)\in[-1,1]\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.exists_clipped_knownMarginalEstimator

\item
Now specialize the table to the true table \(P_{XA}\).  Write
\[
p_x=P(X=x),\qquad
s_{ax}=P(X=x,A=a),\qquad
q_{ax}=P(X=x,A=a,Y=1).
\]
If \(p_x>0\), overlap gives
\[
\epsilon p_x\le s_{1x},
\qquad
\epsilon p_x\le s_{0x};
\]
if \(p_x=0\), then \(s_{0x}=s_{1x}=q_{0x}=q_{1x}=0\).  Hence
\[
\mathbb E_P[\psi_{P_{XA}}(X,A,Y)]
=
\sum_x\left(q_{1x}\frac{p_x}{s_{1x}}-q_{0x}\frac{p_x}{s_{0x}}\right)
=
\sum_x p_x(\mu_{1x}-\mu_{0x})
=
\tau(P),
\]
with zero-mass cells contributing \(0\).  Similarly,
\[
\begin{aligned}
\mathbb E_P[\psi_{P_{XA}}(X,A,Y)^2]
&=
\sum_x\left\{
q_{1x}\left(\frac{p_x}{s_{1x}}\right)^2
+
q_{0x}\left(\frac{p_x}{s_{0x}}\right)^2
\right\} \\
&\le
\sum_x\left\{
\frac{p_x^2}{s_{1x}}
+
\frac{p_x^2}{s_{0x}}
\right\}
\le
\sum_x 2\epsilon^{-1}p_x
=
2\epsilon^{-1}.
\end{aligned}
\]
Thus
\[
\operatorname{Var}_P\!\left(\psi_{P_{XA}}(X,A,Y)\right)
\le 2\epsilon^{-1}.
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.knownMarginalScore_variance_le

\item
By the labeled product sampling in \cref{obj:ass:labeled-iid},
\[
\mathbb E_{P^{\otimes n}}\!\left[\overline\psi_{P_{XA}}(\mathcal L_n)\right]
=\tau(P),
\qquad
\operatorname{Var}_{P^{\otimes n}}\!\left(\overline\psi_{P_{XA}}(\mathcal L_n)\right)
\le
\frac{2\epsilon^{-1}}{n}.
\]
After clipping, the squared risk is bounded by the same quantity.  Taking the supremum over \(P\in\mathcal M_{d,\epsilon}\) and then the infimum over estimators in \cref{obj:def:known-marginal-risk} gives
\[
R_\epsilon^{\mathrm{known}}(n,d)
\le
\frac{2\epsilon^{-1}}{n}
=
\frac{C_\epsilon}{n}.
\]
Together with the lower bound, this proves the claimed inequalities for all \(n\ge1\) and \(d\ge2\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.knownMarginalRisk_upper
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:uniform-mixed-upper}]
Fix \(0<\epsilon<1/2\).

\begin{enumerate}
\item By \cref{obj:lem:fixed-sample-hybrid-transfer}, choose a universal constant \(C_{\mathrm{tr}}>0\) such that, for every \(n\ge1\), \(m\ge0\), \(d\ge2\), and \(P\in\mathcal M_{d,\epsilon}\),
\[
\mathbb E_{P^{\otimes n}\otimes P_{XA}^{\otimes m}}
\!\left[
\left\{\widehat\tau^{\mathrm{mix}}_{n,m,d}-\tau(P)\right\}^2
\right]
\le
2\,\mathcal R^{\mathrm{Pois}}_{\mathrm{hyb}}(P;n,m,\epsilon)
+\frac{C_{\mathrm{tr}}}{n}.
\]
By \cref{obj:lem:poisson-hybrid-risk}, choose \(C_{\mathrm{Pois},\epsilon}>0\), depending only on \(\epsilon\), such that, for the same range of parameters and laws,
\[
\mathcal R^{\mathrm{Pois}}_{\mathrm{hyb}}(P;n,m,\epsilon)
\le
C_{\mathrm{Pois},\epsilon}\,r_\epsilon(n,m,d).
\]
Set
\[
C_\epsilon:=2C_{\mathrm{Pois},\epsilon}+C_{\mathrm{tr}}.
\]
Then \(C_\epsilon>0\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.uniform_mixed_upper

\item It remains to verify the displayed bound uniformly over \(P\). Fix \(n\ge1\), \(m\ge0\), and \(d\ge2\). For any \(P\in\mathcal M_{d,\epsilon}\), the two estimates above give
\[
\begin{aligned}
\mathbb E_{P^{\otimes n}\otimes P_{XA}^{\otimes m}}
\!\left[
\left\{\widehat\tau^{\mathrm{mix}}_{n,m,d}-\tau(P)\right\}^2
\right]
&\le
2\,\mathcal R^{\mathrm{Pois}}_{\mathrm{hyb}}(P;n,m,\epsilon)
+\frac{C_{\mathrm{tr}}}{n}  \\
&\le
2C_{\mathrm{Pois},\epsilon}\,r_\epsilon(n,m,d)
+C_{\mathrm{tr}}\,r_\epsilon(n,m,d).
\end{aligned}
\]
The last inequality uses the elementary rate comparison
\[
\frac1n\le r_\epsilon(n,m,d),
\]
which follows directly from \cref{obj:def:frontier-rate-handle}, since
\[
\frac1n
\le
\min\!\left\{
1,\frac1n+\frac{d^2}{(n+m)^2\ell_n^2}
\right\}
=
r_\epsilon(n,m,d)
\]
for \(n\ge1\). Therefore
\[
\mathbb E_{P^{\otimes n}\otimes P_{XA}^{\otimes m}}
\!\left[
\left\{\widehat\tau^{\mathrm{mix}}_{n,m,d}-\tau(P)\right\}^2
\right]
\le
(2C_{\mathrm{Pois},\epsilon}+C_{\mathrm{tr}})\,r_\epsilon(n,m,d)
=
C_\epsilon\,r_\epsilon(n,m,d).
\] % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.uniform_mixed_upper

\item Since the preceding bound holds for every \(P\in\mathcal M_{d,\epsilon}\), taking the supremum over the model class yields
\[
\sup_{P\in\mathcal M_{d,\epsilon}}
\mathbb E_{P^{\otimes n}\otimes P_{XA}^{\otimes m}}
\!\left[
\left\{\widehat\tau^{\mathrm{mix}}_{n,m,d}-\tau(P)\right\}^2
\right]
\le
C_\epsilon\,r_\epsilon(n,m,d).
\]
The rate is positive for \(n\ge1\) by \cref{obj:def:frontier-rate-handle}, so the right-hand side is a well-defined finite real number. The asserted uniform bound follows for all \(n\ge1\), \(m\ge0\), and \(d\ge2\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.uniform_mixed_upper
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:common-marginal-converse}]
\begin{enumerate}
\item Fix \(0<\epsilon<1/2\).  By \cref{obj:lem:common-marginal-canonical-certificate-family}, there are positive numerical constants
\[
C_{\mathrm{fix}}>0,\qquad c_{\mathrm{risk}}>0,
\]
and, at this \(\epsilon\), calibration constants as in \cref{obj:def:common-marginal-prior-handle}, together with
\[
\rho\in(0,1),\qquad c_{\mathrm{floor}}>0,
\qquad c_{\mathrm{floor}}\le c_{\mathrm{risk}}.
\]
Write \(C_d>0\) for the dimension constant in this calibration, and write \(c_{\mathrm{gap}}>0\) for the calibration's dual-gap constant.  For every \(n,m,d\) with \(n\ge1\) and \(d\ge2\), \cref{obj:lem:common-marginal-canonical-certificate-family} supplies a handle \(H\) satisfying
\[
\mathcal L_{\Pi_0(H)}(P_{XA})
=
\mathcal L_{\Pi_1(H)}(P_{XA})
\]
and the floor certificate
\[
c_{\mathrm{floor}}\frac1n\le R_\epsilon(n,m,d).
\]
For this handle, put
\[
D(n,m,d):=
\min\left\{1,\frac{d^2}{(n+m)^2\log(en)^2}\right\}
\]
and
\[
\Delta_H:=c_{\mathrm{gap}}\,k(H)\,a(H),
\]
where \(k(H)\) and \(a(H)\) are the rare-cell count and shift in \cref{obj:def:common-marginal-prior-handle}.  In the finite-atom branch, the same cited certificate gives the nonparametric dimension comparison
\[
C_d\,D(n,m,d)\le \Delta_H^2
\]
and the fixed-sample lower bound attached to the canonical handle,
\[
c_{\mathrm{floor}}\Delta_H^2\le R_\epsilon(n,m,d).
\]
The displayed lower bound follows once the handle's common-marginal witness, total-variation bound, center separation, and variance absorption satisfy the hypotheses of \cref{obj:lem:fixed-sample-common-marginal-transfer}.
In the complementary branch, \cref{obj:lem:common-marginal-canonical-certificate-family} gives directly
\[
C_dD(n,m,d)\le \frac{c_{\mathrm{floor}}}{n}.
\] % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_canonical_certificate_family

\item Define
\[
c_{\mathrm{lb}}:=
\min\left\{
\frac{c_{\mathrm{floor}}}{2},
\frac{C_d}{2},
\frac{c_{\mathrm{floor}}C_d}{2}
\right\}.
\]
Since \(c_{\mathrm{floor}}>0\) and \(C_d>0\), this constant is positive. % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_converse

\item Fix \(n,m,d\) with \(n\ge1\) and \(d\ge2\), and take the handle \(H\) from the certificate package.  Put
\[
Q(n,m,d):=
\frac{d^2}{(n+m)^2\log(en)^2},
\qquad
D(n,m,d)=\min\{1,Q(n,m,d)\}.
\]
Both \(Q(n,m,d)\) and \(D(n,m,d)\) are nonnegative.  The frontier rate in \cref{obj:def:frontier-rate-handle} satisfies
\[
r_\epsilon(n,m,d)
=
\min\left\{1,\frac1n+Q(n,m,d)\right\}
\le
\frac1n+D(n,m,d).
\]
Indeed, if \(Q(n,m,d)\le1\), the right side is \(1/n+Q(n,m,d)\), which dominates the minimum.  If \(Q(n,m,d)>1\), then \(D(n,m,d)=1\), and the left side is at most \(1\le 1/n+1\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_converse

\item Suppose first that \(H\) is in the finite-atom branch.  Since \(c_{\mathrm{floor}}>0\), multiplying the finite-branch dimension comparison by \(c_{\mathrm{floor}}\) and then using the transferred fixed-sample lower bound gives
\[
c_{\mathrm{floor}}C_dD(n,m,d)
\le
c_{\mathrm{floor}}\Delta_H^2
\le
R_\epsilon(n,m,d).
\]
Together with the floor inequality,
\[
c_{\mathrm{floor}}\frac1n\le R_\epsilon(n,m,d),
\]
this supplies the two lower bounds used in the averaging step.  Using the definition of \(c_{\mathrm{lb}}\) and the bound from the previous step,
\[
\begin{aligned}
c_{\mathrm{lb}} r_\epsilon(n,m,d)
&\le
c_{\mathrm{lb}}\left(\frac1n+D(n,m,d)\right)\\
&=
c_{\mathrm{lb}}\frac1n+c_{\mathrm{lb}} D(n,m,d)\\
&\le
\frac{c_{\mathrm{floor}}}{2}\frac1n
+\frac{c_{\mathrm{floor}}C_d}{2}D(n,m,d)\\
&=
\frac12\left(
c_{\mathrm{floor}}\frac1n
+
c_{\mathrm{floor}}C_dD(n,m,d)
\right)\\
&\le R_\epsilon(n,m,d).
\end{aligned}
\]
Thus the asserted lower bound holds in the finite-atom branch. % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_converse

\item It remains to treat the complementary branch.  In this branch, the certificate clause in \cref{obj:lem:common-marginal-canonical-certificate-family} gives directly
\[
C_dD(n,m,d)
\le
\frac{c_{\mathrm{floor}}}{n}
=
c_{\mathrm{floor}}\frac1n.
\]
Together with the floor inequality
\[
c_{\mathrm{floor}}\frac1n\le R_\epsilon(n,m,d),
\]
we obtain
\[
\begin{aligned}
c_{\mathrm{lb}} r_\epsilon(n,m,d)
&\le
c_{\mathrm{lb}}\left(\frac1n+D(n,m,d)\right)\\
&=
c_{\mathrm{lb}}\frac1n+c_{\mathrm{lb}} D(n,m,d)\\
&\le
\frac{c_{\mathrm{floor}}}{2}\frac1n
+\frac{C_d}{2}D(n,m,d)\\
&\le
\frac12\left(
c_{\mathrm{floor}}\frac1n+
c_{\mathrm{floor}}\frac1n
\right)\\
&=
c_{\mathrm{floor}}\frac1n\\
&\le R_\epsilon(n,m,d).
\end{aligned}
\]
This proves the risk lower bound in the complementary branch as well. % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_converse

\item Combining the two branches, for every \(n,m,d\) with \(n\ge1\), \(m\ge0\), and \(d\ge2\), the handle \(H\) supplies priors \(\Pi_0(H)\) and \(\Pi_1(H)\) with common random treatment-covariate marginal law and satisfies
\[
c_{\mathrm{lb}} r_\epsilon(n,m,d)\le R_\epsilon(n,m,d).
\]
Taking the theorem's lower-bound constant \(c_\epsilon\) to be \(c_{\mathrm{lb}}\), and using \(c_{\mathrm{lb}}>0\) with the fixed calibration constants chosen above, gives the asserted conclusion. % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.common_marginal_converse
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:sharp-annotation-frontier}]
\begin{enumerate}
\item By \cref{obj:thm:common-marginal-converse}, choose a constant \(c_\epsilon>0\) such that, for every \(n\ge1\), \(m\ge0\), and \(d\ge2\),
\[
c_\epsilon r_\epsilon(n,m,d)\le R_\epsilon(n,m,d).
\]
By \cref{obj:thm:uniform-mixed-upper}, choose \(C_\epsilon^{\mathrm{up}}>0\) such that
\[
\sup_{P\in\mathcal M_{d,\epsilon}}
E_{P^{\otimes n}\otimes P_{XA}^{\otimes m}}
\!\left[
\left\{\widehat\tau^{\mathrm{mix}}_{n,m,d}-\tau(P)\right\}^2
\right]
\le
C_\epsilon^{\mathrm{up}} r_\epsilon(n,m,d)
\]
for the same range of \(n,m,d\).  Set
\[
C_\epsilon:=\max\{c_\epsilon,C_\epsilon^{\mathrm{up}}\}.
\]
Then \(0<c_\epsilon\le C_\epsilon\).  The estimator
\(\widehat\tau^{\mathrm{mix}}_{n,m,d}\) is measurable, so the definition of
\(R_\epsilon(n,m,d)\) in \cref{obj:def:minimax-risk} gives
\[
R_\epsilon(n,m,d)
\le
\sup_{P\in\mathcal M_{d,\epsilon}}
E_{P^{\otimes n}\otimes P_{XA}^{\otimes m}}
\!\left[
\left\{\widehat\tau^{\mathrm{mix}}_{n,m,d}-\tau(P)\right\}^2
\right].
\]
Combining the last two displays, and using \(r_\epsilon(n,m,d)>0\), yields
\[
R_\epsilon(n,m,d)
\le
C_\epsilon^{\mathrm{up}} r_\epsilon(n,m,d)
\le
C_\epsilon r_\epsilon(n,m,d).
\]
Together with the lower bound from \cref{obj:thm:common-marginal-converse}, this proves the finite-sample bracket. % lean: sharp_annotation_frontier

\item The unannotated endpoint is the specialization \(m=0\) of \cref{obj:def:frontier-rate-handle}.  Since \(N=n\) in that case,
\[
r_\epsilon(n,0,d)
=
\min\!\left\{1,\frac1n+\frac{d^2}{n^2\log^2(en)}\right\}.
\]
% lean: sharp_annotation_frontier

\item The known-marginal limit is exactly the conclusion of \cref{obj:thm:known-marginal-limit}: for fixed \(n\ge1\) and \(d\ge2\),
\[
\lim_{m\to\infty} R_\epsilon(n,m,d)
=
R_\epsilon^{\mathrm{known}}(n,d).
\]
% lean: sharp_annotation_frontier

\item It remains to record the sequence consequences.  Fix sequences \((m_n)\) and \((d_n)\) with \(d_n\ge2\), and set
\[
\rho_n:=R_\epsilon(n,m_n,d_n),\qquad
q_n:=r_\epsilon(n,m_n,d_n),
\]
\[
\rho_n^{0}:=R_\epsilon(n,0,d_n),\qquad
q_n^{0}:=r_\epsilon(n,0,d_n).
\]
The finite-sample bracket gives, for all \(n\ge1\),
\[
c_\epsilon q_n\le \rho_n\le C_\epsilon q_n,
\qquad
c_\epsilon q_n^{0}\le \rho_n^{0}\le C_\epsilon q_n^{0}.
\]
Since all four sequences are nonnegative, these two-sided bounds imply
\[
(\rho_n)=\Theta(q_n),
\qquad
(\rho_n^{0})=\Theta(q_n^{0}).
\]
Indeed, the upper inequalities give the \(O\)-bounds in one direction, and the lower inequalities give
\(q_n\le c_\epsilon^{-1}\rho_n\) and
\(q_n^{0}\le c_\epsilon^{-1}\rho_n^{0}\). % lean: positive_two_sided_bounds_isTheta

With
\[
N_n=n+m_n,\qquad
b_n:=\frac{d_n}{N_n\ell_n},
\]
\cref{obj:def:frontier-rate-handle} gives
\[
q_n=\min\left\{1,\frac1n+b_n^2\right\}.
\]
Because \(n^{-1}\to0\), this display satisfies
\[
q_n\to0
\quad\Longleftrightarrow\quad
b_n\to0.
\]
The implication from right to left follows by continuity of \(x\mapsto\min\{1,x\}\); the implication from left to right follows because eventually \(q_n<1/2\), hence
\[
q_n=\frac1n+b_n^2
\]
eventually, so \(b_n^2\to0\).  Using \((\rho_n)=\Theta(q_n)\), we obtain
\[
R_\epsilon(n,m_n,d_n)\to0
\quad\Longleftrightarrow\quad
\frac{d_n}{N_n\ell_n}\to0.
\]
% lean: annotation_frontier_consistency_iff

Next,
\[
q_n=O(n^{-1})
\quad\Longleftrightarrow\quad
b_n^2=O(n^{-1})
\quad\Longleftrightarrow\quad
d_n=O\!\left(\frac{N_n\ell_n}{\sqrt n}\right).
\]
For the first equivalence, the term \(n^{-1}\) is already present in \(q_n\), and the same eventual equality
\(q_n=n^{-1}+b_n^2\) applies after the \(O(n^{-1})\) bound forces \(q_n\to0\).  The second equivalence is obtained by taking square roots of nonnegative sequences and multiplying by \(N_n\ell_n\).  Transferring through \((\rho_n)=\Theta(q_n)\) gives
\[
R_\epsilon(n,m_n,d_n)=O(n^{-1})
\quad\Longleftrightarrow\quad
d_n=O\!\left(\frac{N_n\ell_n}{\sqrt n}\right).
\]
% lean: annotation_frontier_parametric_iff

Finally put
\[
a_n:=\frac{d_n}{n\ell_n}.
\]
Then
\[
q_n=\min\left\{1,\frac1n+b_n^2\right\},
\qquad
q_n^{0}=\min\left\{1,\frac1n+a_n^2\right\},
\qquad
0\le b_n\le a_n .
\]
By \cref{obj:lem:annotation-frontier-strict-improvement}, the numerical strict-improvement
criterion for these two rates is
\[
q_n=o(q_n^{0})
\]
if and only if both
\[
\frac{d_n}{\sqrt n\,\ell_n}\to\infty
\]
and
\[
\frac{b_n}{\min\{1,a_n\}}\to0.
\]
To see this, observe first that \(q_n\ge n^{-1}\), so \(q_n=o(q_n^{0})\) forces
\(n^{-1}=o(q_n^{0})\), which is equivalent to
\(a_n\sqrt n=d_n/(\sqrt n\,\ell_n)\to\infty\).  Under this condition the floor term \(n^{-1}\) is negligible in \(q_n^{0}\), and
\[
q_n^{0}\asymp \min\{1,a_n^2\}.
\]
Since \(0\le b_n\le a_n\), the remaining requirement
\(\min\{1,n^{-1}+b_n^2\}=o(q_n^{0})\) is equivalent to
\[
b_n^2=o(\min\{1,a_n^2\}),
\]
which, for nonnegative sequences, is the same as
\[
\frac{b_n}{\min\{1,a_n\}}\to0.
\]
Substituting the definitions of \(a_n\) and \(b_n\), this last condition is exactly
\[
\frac{d_n/(N_n\ell_n)}
{\min\{1,d_n/(n\ell_n)\}}
\to0.
\]
The two-sided bounds \((\rho_n)=\Theta(q_n)\) and \((\rho_n^{0})=\Theta(q_n^{0})\) transfer
\(q_n=o(q_n^{0})\) precisely to
\[
R_\epsilon(n,m_n,d_n)
=
o\!\left(R_\epsilon(n,0,d_n)\right),
\]
which proves the stated strict-improvement equivalence. % lean: annotation_frontier_strict_improvement_iff
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:known-marginal-limit}]
Let
\[
\mathcal Z_n:=([d]\times\{0,1\}^2)^n,
\qquad
\mathcal C_d:=[d]\times\{0,1\}.
\]
We use the compact coordinate space
\[
\Theta_{\epsilon,d}
=
\Bigl\{(\pi,\varrho,\omega):
\pi\in\mathsf{Prob}([d]),\
\varrho_x\in[\epsilon,1-\epsilon],\
\omega_{\alpha x}\in[0,1]\Bigr\}.
\]
The assumptions \(d\ge2\) and \(0<\epsilon<1/2\) make this space nonempty.  For
\(\theta=(\pi,\varrho,\omega)\in\Theta_{\epsilon,d}\), set
\[
P_\theta(x,\alpha,y)
=
\pi_x\varrho_x^\alpha(1-\varrho_x)^{1-\alpha}
\omega_{\alpha x}^y(1-\omega_{\alpha x})^{1-y},
\]
and define
\[
\lambda_\theta(z)
=
\prod_{i=1}^n P_\theta(z_i),
\qquad
\xi_\theta(x,\alpha)
=
\sum_{y\in\{0,1\}}P_\theta(x,\alpha,y),
\qquad
\upsilon_\theta
=
\sum_{x\in[d]}\pi_x(\omega_{1x}-\omega_{0x}).
\]
Also,
\[
-1
=
\sum_x\pi_x(-1)
\le
\sum_x\pi_x(\omega_{1x}-\omega_{0x})
\le
\sum_x\pi_x(1)
=
1,
\]
so \(\upsilon_\theta\in[-1,1]\) for every \(\theta\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.knownMarginalTarget_mem

\begin{enumerate}
\item\label{pf:known-marginal-limit:finite-side} Define the finite-side values
\[
\mathfrak R_m
=
\inf_{\varphi:\mathcal Z_n\times\mathcal C_d^m\to[-1,1]}
\sup_{\theta\in\Theta_{\epsilon,d}}
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\sum_{u\in\mathcal C_d^m}
\Bigl(\prod_{j=1}^m\xi_\theta(u_j)\Bigr)
\bigl(\varphi(z,u)-\upsilon_\theta\bigr)^2
\]
and
\[
\mathfrak R_\star
=
\inf_{\psi:\mathcal Z_n\times\mathsf{Prob}(\mathcal C_d)\to[-1,1]}
\sup_{\theta\in\Theta_{\epsilon,d}}
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\bigl(\psi(z,\xi_\theta)-\upsilon_\theta\bigr)^2 .
\]
With the preceding coordinates, \cref{obj:lem:finite-side-information-convergence} gives
\[
\lim_{m\to\infty}\mathfrak R_m=\mathfrak R_\star .
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.knownMarginalGeneric_tendsto

\item\label{pf:known-marginal-limit:ambient-exact} Let \(\widetilde{\mathfrak R}_\star\) be the measurable ambient exact-table value
\[
\widetilde{\mathfrak R}_\star
=
\inf_{\psi}
\sup_{\theta\in\Theta_{\epsilon,d}}
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\bigl(\psi(z,\xi_\theta)-\upsilon_\theta\bigr)^2,
\]
where the infimum ranges over all rules
\(\psi:\mathcal Z_n\times\mathbb R^{\mathcal C_d}\to[-1,1]\) whose section
\(w\mapsto\psi(z,w)\) is Borel measurable for each \(z\).  The target range established above gives
\(\upsilon_\theta\in[-1,1]\), and \cref{obj:lem:finite-side-information-convergence} supplies the convergence
\[
\lim_{m\to\infty}\mathfrak R_m=\mathfrak R_\star .
\]
The measurable ambient exact-table value is tied to this limiting sequence by two comparisons.  The exact-side value infimizes over all rules on \(\mathsf{Prob}(\mathcal C_d)\), while the ambient value infimizes over rules on \(\mathbb R^{\mathcal C_d}\) with Borel-measurable sections.  Restricting any ambient rule to \(\mathsf{Prob}(\mathcal C_d)\) gives an admissible exact-side rule, so the exact-side class is at least as large after restriction, and therefore
\[
\mathfrak R_\star\le \widetilde{\mathfrak R}_\star .
\]
Conversely, \cref{obj:lem:finite-side-measurable-comparison} gives
\(\widetilde{\mathfrak R}_\star\le\mathfrak R_m\) for every \(m\ge0\).
For completeness, its measurable conditional-average construction in the
present coordinates is as follows.  Fix an integer \(m\ge0\) and an empirical-side rule
\(\varphi:\mathcal Z_n\times\mathcal C_d^m\to[-1,1]\).  Define first the polynomial conditional average on the ambient table space by
\[
\psi_\varphi^\circ(z,w)
=
\sum_{u\in\mathcal C_d^m}
\Bigl(\prod_{j=1}^m w(u_j)\Bigr)\varphi(z,u),
\qquad
w\in\mathbb R^{\mathcal C_d}.
\]
This function is Borel measurable in \(w\) for each fixed \(z\).  Use its clipped version as the ambient exact-table rule,
\[
\psi_\varphi(z,w)
=
\min\{1,\max\{-1,\psi_\varphi^\circ(z,w)\}\},
\qquad
w\in\mathbb R^{\mathcal C_d}.
\]
Then \(\psi_\varphi:\mathcal Z_n\times\mathbb R^{\mathcal C_d}\to[-1,1]\) is admissible for \(\widetilde{\mathfrak R}_\star\).  When \(w=r\in\mathsf{Prob}(\mathcal C_d)\), the coefficients \(\prod_{j=1}^m r(u_j)\) form a probability vector on \(\mathcal C_d^m\), and since \(\varphi\) takes values in \([-1,1]\), the polynomial average already lies in \([-1,1]\).  Hence, on probability tables,
\[
\psi_\varphi(z,r)
=
\sum_{u\in\mathcal C_d^m}
\Bigl(\prod_{j=1}^m r(u_j)\Bigr)\varphi(z,u).
\]
For every \(\theta\) and \(z\), Jensen's inequality for the product law \(\xi_\theta^{\otimes m}\) gives
\[
\bigl(\psi_\varphi(z,\xi_\theta)-\upsilon_\theta\bigr)^2
=
\left(
\sum_{u\in\mathcal C_d^m}
\Bigl(\prod_{j=1}^m\xi_\theta(u_j)\Bigr)
\bigl(\varphi(z,u)-\upsilon_\theta\bigr)
\right)^2
\le
\sum_{u\in\mathcal C_d^m}
\Bigl(\prod_{j=1}^m\xi_\theta(u_j)\Bigr)
\bigl(\varphi(z,u)-\upsilon_\theta\bigr)^2 .
\]
Multiplying by \(\lambda_\theta(z)\), summing over \(z\), and then taking the supremum over \(\theta\) gives
\[
\sup_{\theta}
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\bigl(\psi_\varphi(z,\xi_\theta)-\upsilon_\theta\bigr)^2
\le
\sup_{\theta}
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\sum_{u\in\mathcal C_d^m}
\Bigl(\prod_{j=1}^m\xi_\theta(u_j)\Bigr)
\bigl(\varphi(z,u)-\upsilon_\theta\bigr)^2 .
\]
Taking the infimum over \(\varphi\) yields
\[
\widetilde{\mathfrak R}_\star\le \mathfrak R_m
\qquad\text{for every }m\ge0 .
\]
Taking the limit along the displayed convergence yields
\[
\widetilde{\mathfrak R}_\star\le \lim_{m\to\infty}\mathfrak R_m=\mathfrak R_\star .
\]
The two inequalities identify the ambient measurable exact-table value with the exact-side value:
\[
\widetilde{\mathfrak R}_\star=\mathfrak R_\star .
\]
% lean: Causalean.Stat.Minimax.FiniteSideInformation.measurableExactTableMinimaxValue_eq_of_tendsto

\item\label{pf:known-marginal-limit:known-risk} We use here the standing
theorem hypotheses \(d\ge2\) and \(0<\epsilon<1/2\), which are also the
parameter-range clauses in the definition of \(\mathcal M_{d,\epsilon}\).
Under these hypotheses, the compact coordinates parametrize the laws in
\(\mathcal M_{d,\epsilon}\) from \cref{obj:def:model-class}.  For every \(\theta\in\Theta_{\epsilon,d}\), the law \(P_\theta\) satisfies occupied-cell overlap because, whenever \(\pi_x>0\), its treatment propensity is
\[
\frac{\sum_y P_\theta(x,1,y)}{\sum_{\alpha,y}P_\theta(x,\alpha,y)}
=\varrho_x\in[\epsilon,1-\epsilon].
\]
Conversely, given any \(P\in\mathcal M_{d,\epsilon}\), choose coordinates by
\[
\pi_x=p_x,
\qquad
\varrho_x=
\begin{cases}
 e_x, & p_x>0,\\
 \epsilon, & p_x=0,
\end{cases}
\qquad
\omega_{\alpha x}=\mu_{\alpha x}.
\]
The value of \(\varrho_x\) in an empty cell is immaterial because every atom in that cell is multiplied by \(\pi_x=0\).  For occupied cells, the overlap condition places \(\varrho_x\) in \([\epsilon,1-\epsilon]\), and the binary outcome regressions place \(\omega_{\alpha x}\) in \([0,1]\).  These coordinates recover the atom probabilities,
\[
P(x,\alpha,y)
=
\pi_x\varrho_x^\alpha(1-\varrho_x)^{1-\alpha}
\omega_{\alpha x}^y(1-\omega_{\alpha x})^{1-y},
\]
with the empty-cell case giving zero on both sides.  Thus the map \(\theta\mapsto P_\theta\) is surjective onto the model class.

Under this parametrization, the three quantities entering the known-marginal risk are transported by
\[
P_\theta^{\otimes n}(z)=\lambda_\theta(z),
\qquad
(P_\theta)_{XA}(x,\alpha)=\xi_\theta(x,\alpha),
\qquad
\tau(P_\theta)=\upsilon_\theta .
\]
The last identity follows from \cref{obj:def:ate-functional}: on occupied cells the outcome regressions are \(\omega_{1x}\) and \(\omega_{0x}\), while empty cells have coefficient \(\pi_x=0\).  Hence, for every bounded measurable known-table rule \(\psi\),
\[
\int
\bigl(\psi(Z,(P_\theta)_{XA})-\tau(P_\theta)\bigr)^2
\,dP_\theta^{\otimes n}(Z)
=
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\bigl(\psi(z,\xi_\theta)-\upsilon_\theta\bigr)^2 .
\]
The known-table sample space in \cref{obj:def:known-marginal-risk} is
\(\mathcal Z_n\times\mathbb R^{\mathcal C_d}\).  Its bounded measurable
procedures are in bijection with the ambient procedures used in
\(\widetilde{\mathfrak R}_\star\).  Explicitly, a paper procedure \(f\)
is sent to
\[
\psi_f(z,w):=f(z,w),
\]
and an ambient procedure \(\psi\) is sent back to
\(f_\psi(z,w):=\psi(z,w)\).  Because \(\mathcal Z_n\) is finite and
discrete, measurability of every section \(w\mapsto\psi(z,w)\) is
equivalent to measurability of the uncurried map; the two assignments are
pointwise inverse and preserve the range \([-1,1]\).

For corresponding \(f\) and \(\psi_f\), evaluation at the known table
\(w=(P_\theta)_{XA}=\xi_\theta\), followed by expansion of the finite
labeled product law, gives
\[
\begin{aligned}
&\int
\bigl(f(Z,(P_\theta)_{XA})-\tau(P_\theta)\bigr)^2
\,dP_\theta^{\otimes n}(Z)\\
&\qquad =
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\bigl(\psi_f(z,\xi_\theta)-\upsilon_\theta\bigr)^2.
\end{aligned}
\]
Thus the estimator bijection preserves every coordinate risk exactly.
Reindexing the supremum over \(P\in\mathcal M_{d,\epsilon}\) by the
surjective coordinate map, and then taking the same infimum over the
bijective procedure classes, gives
\[
R_\epsilon^{\mathrm{known}}(n,d)
=
\widetilde{\mathfrak R}_\star .
\]
Together with the equality \(\widetilde{\mathfrak R}_\star=\mathfrak R_\star\) above, this gives
\[
R_\epsilon^{\mathrm{known}}(n,d)=\mathfrak R_\star .
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.knownMarginalRisk_eq_measurableValue

\item\label{pf:known-marginal-limit:empirical-risk} For each \(m\), the minimax risk in \cref{obj:def:minimax-risk} equals the empirical finite-side value:
\[
R_\epsilon(n,m,d)=\mathfrak R_m .
\]
First reindex the supremum over \(P\in\mathcal M_{d,\epsilon}\) by the surjective map \(\theta\mapsto P_\theta\).  The annotation sample space is the
finite product \(\mathcal Z_n\times\mathcal C_d^m\), so every real-valued
map on it is measurable.  For any measurable estimator \(T\) in the annotation experiment, write its two-sample mean-squared risk at \(P_\theta\) as
\[
\mathcal V_T(\theta)
=
\int
\bigl(T(Z,U)-\tau(P_\theta)\bigr)^2
\,d\bigl(P_\theta^{\otimes n}\otimes (P_\theta)_{XA}^{\otimes m}\bigr)(Z,U).
\]
Expanding both finite product measures, and using
\((P_\theta)_{XA}=\xi_\theta\),
\(P_\theta^{\otimes n}=\lambda_\theta\), and
\(\tau(P_\theta)=\upsilon_\theta\), gives the exact table identity
\[
\mathcal V_{T_\varphi}(\theta)
=
\sum_{z\in\mathcal Z_n}\lambda_\theta(z)
\sum_{u\in\mathcal C_d^m}
\Bigl(\prod_{j=1}^m\xi_\theta(u_j)\Bigr)
\bigl(\varphi(z,u)-\upsilon_\theta\bigr)^2
\]
for \(T_\varphi(z,u):=\varphi(z,u)\).

Thus every empirical-side procedure \(\varphi:\mathcal Z_n\times\mathcal C_d^m\to[-1,1]\) induces the paper estimator
\[
T_\varphi(z,u)=\varphi(z,u),
\]
viewed as an estimator in the annotation experiment.  For this induced estimator, the preceding expansion gives equality of the coordinate risks for every \(\theta\), hence equality of the corresponding worst-case risks after the reindexing by \(\theta\).  Since the paper infimum ranges over all measurable estimators, including these induced estimators, taking the infimum over \(\varphi\) gives
\[
R_\epsilon(n,m,d)\le \mathfrak R_m .
\]
For the reverse inequality, let \(T\) be any measurable paper estimator and define the clipped empirical-side rule
\[
\varphi_T(z,u)
=
\min\{1,\max\{-1,T(z,u)\}\}.
\]
The transported identity \(\tau(P_\theta)=\upsilon_\theta\), together with
the target-range bound, gives the pointwise clipping inequality
\[
\bigl(\varphi_T(z,u)-\tau(P_\theta)\bigr)^2
\le
\bigl(T(z,u)-\tau(P_\theta)\bigr)^2
\qquad\text{for all }(z,u),\theta .
\]
Integrating this inequality against
\(P_\theta^{\otimes n}\otimes (P_\theta)_{XA}^{\otimes m}\) yields
\[
\int
\bigl(\varphi_T(Z,U)-\tau(P_\theta)\bigr)^2
\,d\bigl(P_\theta^{\otimes n}\otimes (P_\theta)_{XA}^{\otimes m}\bigr)(Z,U)
\le
\mathcal V_T(\theta).
\]
By the same finite product expansion as above, the left side is exactly the
empirical-side table risk of \(\varphi_T\) at \(\theta\).
The comparison of worst-case risks also requires a finite bound for the unclipped estimator.  Set
\[
B_T:=\sum_{(z,u)\in\mathcal Z_n\times\mathcal C_d^m}
\bigl(|T(z,u)|+1\bigr)^2 .
\]
Since \(|\upsilon_\theta|\le1\),
\[
\bigl(T(z,u)-\upsilon_\theta\bigr)^2
\le
\bigl(|T(z,u)|+1\bigr)^2
\le B_T
\]
for every \((z,u)\) and \(\theta\).  Hence the worst-case risk of \(T\) over the reindexed model is bounded, and the integrated clipping inequality implies
\[
\sup_{\theta\in\Theta_{\epsilon,d}}
\int
\bigl(\varphi_T(Z,U)-\tau(P_\theta)\bigr)^2
\,d\bigl(P_\theta^{\otimes n}\otimes (P_\theta)_{XA}^{\otimes m}\bigr)(Z,U)
\le
\sup_{\theta\in\Theta_{\epsilon,d}}\mathcal V_T(\theta).
\]
Taking infima over all measurable paper estimators \(T\) gives
\[
\mathfrak R_m\le R_\epsilon(n,m,d).
\]
The two inequalities prove \(R_\epsilon(n,m,d)=\mathfrak R_m\).
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.minimaxRisk_eq_empiricalValue

\item The preceding displayed identities give
\[
\lim_{m\to\infty}R_\epsilon(n,m,d)
=
\lim_{m\to\infty}\mathfrak R_m
=
\mathfrak R_\star
=
R_\epsilon^{\mathrm{known}}(n,d).
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.known_marginal_limit
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:inverse-count-baseline}]
\begin{enumerate}
\item By \cref{obj:lem:poisson-inverse-count-risk}, choose \(C_0>0\), depending only on \(\epsilon\), such that for every \(d\), every \(u,t>0\), and every \(P\in\mathcal M_{d,\epsilon}\),
\[
\operatorname{Var}_P(Z^H)\le C_0(u^{-1}+t^{-1}),
\qquad
\bigl|E_P Z^H-\tau(P)\bigr|\le C_0\min\{1,d/t\}.
\]
Hence, in the independent Poisson count experiment,
\[
E_P\{Z^H-\tau(P)\}^2
=
\operatorname{Var}_P(Z^H)+\{E_PZ^H-\tau(P)\}^2
\le
C_0(u^{-1}+t^{-1})
+
C_0^2\min\{1,d/t\}^2 .
\]
Set
\[
C_\epsilon=100000(1+C_0+C_0^2)>0 .
\]
% lean: poisson_inverse_count_sqRisk_bound

\item Fix \(n\ge1\), \(m\ge0\), and \(d\ge2\).  Let
\[
M_0=\lfloor n/3\rfloor,\qquad
n_p=\left\lfloor\frac{n-M_0}{2}\right\rfloor,\qquad
n_f=n-M_0-n_p,\qquad
m_f=m-\lfloor m/2\rfloor ,
\]
and put
\[
u=\frac{M_0}{8},\qquad
t=\frac{n_f+m_f}{8},\qquad
\mathfrak q=\frac1n+\frac{d^2}{(n+m)^2}.
\]
The estimator is a finite sum of measurable prefix statistics with deterministic weights, so it is measurable.  The class \(\mathcal M_{d,\epsilon}\) is nonempty: the law with uniform covariate distribution, propensity \(1/2\), and equal outcome regressions in the two arms satisfies \cref{obj:ass:overlap} because \(0<\epsilon<1/2\). % lean: inverse_count_baseline

\item First suppose \(24\le n\).  The block arithmetic gives
\[
n\le4M_0,\qquad n+m\le4(n_f+m_f),
\]
and therefore \(M_0>0\), \(n_f+m_f>0\), \(u>0\), and \(t>0\).  Consequently
\[
\frac n{32}\le u,\qquad
\frac{n+m}{32}\le t,
\]
so
\[
u^{-1}\le\frac{32}{n},
\qquad
t^{-1}\le\frac{32}{n+m}\le\frac{32}{n}.
\]
Also,
\[
\frac dt\le \frac{32d}{n+m},
\]
and hence
\[
\min\{1,d/t\}^2
\le (d/t)^2
\le
1024\,\frac{d^2}{(n+m)^2}.
\]
% lean: inverse_count_baseline

\item For the Poisson prefix overflow terms, the elementary bound
\[
\Pr\{\operatorname{Poi}(\Lambda_{\mathrm{tail}}/8)>\Lambda_{\mathrm{tail}}\}
\le \frac8{\Lambda_{\mathrm{tail}}},\qquad \Lambda_{\mathrm{tail}}\ge1,
\]
is obtained as follows.  Bernstein's upper-tail inequality for a Poisson variable with mean \(\Lambda_{\mathrm{tail}}/8\), applied with deviation parameter \(\Lambda_{\mathrm{tail}}/4\), yields
\[
\Pr\{\operatorname{Poi}(\Lambda_{\mathrm{tail}}/8)>\Lambda_{\mathrm{tail}}\}
\le \exp(-\Lambda_{\mathrm{tail}}/4).
\]
Since \(\exp(x)\ge 1+x\),
\[
\exp(-\Lambda_{\mathrm{tail}}/4)=\frac1{\exp(\Lambda_{\mathrm{tail}}/4)}
\le \frac1{\Lambda_{\mathrm{tail}}/4}
=\frac4{\Lambda_{\mathrm{tail}}}
\le\frac8{\Lambda_{\mathrm{tail}}} .
\]
Applying this with \(\Lambda_{\mathrm{tail}}=M_0\) and \(\Lambda_{\mathrm{tail}}=n_f+m_f\) gives
\[
\Pr\{\operatorname{Poi}(u)>M_0\}\le \frac8{M_0},
\qquad
\Pr\{\operatorname{Poi}(t)>n_f+m_f\}\le \frac8{n_f+m_f}.
\]
Together with \(n\le4M_0\) and \(n+m\le4(n_f+m_f)\), this implies
\[
\frac8{M_0}\le \frac{32}{n},
\qquad
\frac8{n_f+m_f}\le \frac{32}{n+m}\le \frac{32}{n}.
\]
% lean: poisson_eighth_overflow_le_inv

\item Fix \(P\in\mathcal M_{d,\epsilon}\).  Under the probability law \(\mathbb P^{\mathrm{pref}}_{P,u,t}\), draw two independent finite prefixes: an outcome-marked prefix from \(P\) with Poisson length of mean \(u\), and a treatment--covariate prefix from \(P_{XA}\) with Poisson length of mean \(t\).  If \(S_{ax}\) is the number of outcome-marked prefix records with \((X,A,Y)=(x,a,1)\) and \(K_{ax}\) is the number of treatment--covariate prefix records with \((X,A)=(x,a)\), set
\[
\widehat\tau^{\mathrm{IC,pref}}_{u,t}
=
\operatorname{clip}_{[-1,1]}
\left[
\sum_{x=1}^d
\left\{
\frac{S_{1x}}u\frac{K_{0x}+K_{1x}+1}{K_{1x}+1}
-
\frac{S_{0x}}u\frac{K_{0x}+K_{1x}+1}{K_{0x}+1}
\right\}
\right],
\]
and define
\[
\mathcal R_{\mathrm{pref}}(P;u,t)
:=
E_{\mathbb P^{\mathrm{pref}}_{P,u,t}}
\left[
\left\{
\widehat\tau^{\mathrm{IC,pref}}_{u,t}-\tau(P)
\right\}^2
\right].
\]
The prefix transfer in \cref{obj:lem:inverse-count-poisson-prefix-transfer}, applied with the present block sizes and intensities, gives
\[
E_P\!\left[
\left\{\widehat\tau^{\mathrm{IC}}_{n,m,d}-\tau(P)\right\}^2
\right]
\le
\mathcal R_{\mathrm{pref}}(P;u,t)
+
4\left[
\Pr\{\operatorname{Poi}(u)>M_0\}
+
\Pr\{\operatorname{Poi}(t)>n_f+m_f\}
\right].
\]
Here the fixed-sample estimator is the finite Rao--Blackwell average over admissible prefix lengths in \cref{obj:synth_5}; the displayed inequality accounts for the two overflow events, and the coefficient \(4\) is the squared diameter of \([-1,1]\) for the clipped statistic and the target \(\tau(P)\). % lean: inverseCountEstimator_risk_le_poissonPrefixes

\item The Poisson-prefix counts have the Poisson count law by \cref{obj:lem:poisson-prefix-count-law}: for every \(x\in[d]\) and \(a\in\{0,1\}\),
\[
S_{ax}\sim\operatorname{Poi}(u q_{ax}),
\qquad
K_{ax}\sim\operatorname{Poi}(t s_{ax}),
\]
and these coordinates are mutually independent.  Thus the pushforward of \(\mathbb P^{\mathrm{pref}}_{P,u,t}\) under the count map is exactly the product law used in \cref{obj:lem:poisson-inverse-count-risk}.  We first record the range of the target.  The cell masses satisfy \(p_x\ge0\) and \(\sum_{x=1}^d p_x=1\), and the binary-outcome regressions satisfy \(0\le\mu_{ax}\le1\).  Hence
\[
\left|\tau(P)\right|
=
\left|\sum_{x=1}^d p_x(\mu_{1x}-\mu_{0x})\right|
\le
\sum_{x=1}^d p_x\,|\mu_{1x}-\mu_{0x}|
\le
\sum_{x=1}^d p_x
=1 .
\]
Thus \(\tau(P)\in[-1,1]\).  Since projection onto \([-1,1]\) cannot increase distance from a point of \([-1,1]\), clipping gives
\[
\mathcal R_{\mathrm{pref}}(P;u,t)
\le
E_P\{Z^H-\tau(P)\}^2 .
\]
Combining this with Step 1 gives
\[
\mathcal R_{\mathrm{pref}}(P;u,t)
\le
C_0(u^{-1}+t^{-1})
+
C_0^2\min\{1,d/t\}^2 .
\]
% lean: finitePrefixInverseCountStatistic_risk_le_poissonCount

\item The preceding displays imply
\[
E_P\!\left[
\left\{\widehat\tau^{\mathrm{IC}}_{n,m,d}-\tau(P)\right\}^2
\right]
\le
\left(64C_0+1024C_0^2+256\right)
\left\{
\frac1n+\frac{d^2}{(n+m)^2}
\right\}.
\]
Indeed, the variance contribution is bounded by
\[
C_0(u^{-1}+t^{-1})\le 64C_0\,\frac1n,
\]
the squared-bias contribution by
\[
C_0^2\min\{1,d/t\}^2
\le
1024C_0^2\,\frac{d^2}{(n+m)^2},
\]
and the overflow contribution by
\[
4\left(\frac8{M_0}+\frac8{n_f+m_f}\right)
\le
256\,\frac1n .
\]
Since
\[
64C_0+1024C_0^2+256\le C_\epsilon,
\]
the preceding display gives the large-sample estimate
\[
E_P\!\left[
\left\{\widehat\tau^{\mathrm{IC}}_{n,m,d}-\tau(P)\right\}^2
\right]
\le
C_\epsilon\mathfrak q .
\]
If \(\mathfrak q\le1\), then \(\min\{1,\mathfrak q\}=\mathfrak q\), and hence
\[
E_P\!\left[
\left\{\widehat\tau^{\mathrm{IC}}_{n,m,d}-\tau(P)\right\}^2
\right]
\le
C_\epsilon\min\{1,\mathfrak q\}.
\]
When \(\mathfrak q>1\), both the estimator and \(\tau(P)\) lie in \([-1,1]\), so
\[
E_P\!\left[
\left\{\widehat\tau^{\mathrm{IC}}_{n,m,d}-\tau(P)\right\}^2
\right]\le4
\le C_\epsilon\min\{1,\mathfrak q\}.
\]
% lean: inverseCountEstimator_mse_le_four

\item It remains to cover \(n<24\).  Then \(n\le23\), and
\[
\frac1{23}\le \frac1n\le \mathfrak q,
\qquad
\frac1{23}\le \min\{1,\mathfrak q\}.
\]
Again using the boundedness of the estimator and of \(\tau(P)\),
\[
E_P\!\left[
\left\{\widehat\tau^{\mathrm{IC}}_{n,m,d}-\tau(P)\right\}^2
\right]\le4
\le
C_\epsilon\min\{1,\mathfrak q\},
\]
because \(C_\epsilon\ge92\). % lean: inverse_count_baseline

\item Taking the supremum over \(P\in\mathcal M_{d,\epsilon}\) in the preceding bound gives
\[
\sup_{P\in\mathcal M_{d,\epsilon}}
E_{P^{\otimes n}\otimes P_{XA}^{\otimes m}}
\left[
\left\{
\widehat\tau^{\mathrm{IC}}_{n,m,d}-\tau(P)
\right\}^2
\right]
\le
C_\epsilon
\min\left\{
1,\frac1n+\frac{d^2}{(n+m)^2}
\right\}.
\]
Finally, \(\widehat\tau^{\mathrm{IC}}_{n,m,d}\) is an admissible measurable estimator in the class defining \(R_\epsilon(n,m,d)\) in \cref{obj:def:minimax-risk}.  Therefore
\[
R_\epsilon(n,m,d)
\le
C_\epsilon
\min\left\{
1,\frac1n+\frac{d^2}{(n+m)^2}
\right\}.
\]
% lean: inverse_count_baseline
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:binary-alphabet-parametric-rate}]
Fix \(0<\epsilon<1/2\).
\begin{enumerate}
\item By \cref{obj:lem:parametric-label-floor}, choose \(c_{\mathrm{lab}}>0\) such that, for every \(n\ge1\), \(m\ge0\), and \(d\ge2\),
\[
\frac{c_{\mathrm{lab}}}{n}\le R_\epsilon(n,m,d).
\]
By \cref{obj:thm:inverse-count-baseline}, choose \(C_{\mathrm{IC}}>0\) such that, for every \(n\ge1\), \(m\ge0\), and \(d\ge2\),
\[
R_\epsilon(n,m,d)
\le
C_{\mathrm{IC}}
\min\left\{
1,\frac1n+\frac{d^2}{(n+m)^2}
\right\}.
\]
Set
\[
c_\epsilon:=c_{\mathrm{lab}},
\qquad
C_\epsilon:=\max\{c_{\mathrm{lab}},5C_{\mathrm{IC}}\}.
\]
Then \(0<c_\epsilon\le C_\epsilon\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.binary_alphabet_parametric_rate

\item For \(d=2\), the lower bound supplied by \cref{obj:lem:parametric-label-floor} gives, for every \(n\ge1\) and \(m\ge0\),
\[
\frac{c_\epsilon}{n}
=
\frac{c_{\mathrm{lab}}}{n}
\le
R_\epsilon(n,m,2).
\]
% lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.parametric_label_floor

\item It remains to bound the right-hand side.  Fix \(n\ge1\) and \(m\ge0\), and put
\[
N:=n+m .
\]
Then \(N\ge n\ge1\), hence \(N>0\), and
\[
n\le N\le N^2,
\]
because \(N\ge1\).  Therefore
\[
\frac{4}{N^2}\le \frac4n .
\]
Using \cref{obj:thm:inverse-count-baseline} with \(d=2\),
\[
\begin{aligned}
R_\epsilon(n,m,2)
&\le
C_{\mathrm{IC}}
\min\left\{
1,\frac1n+\frac{4}{N^2}
\right\} \\
&\le
C_{\mathrm{IC}}
\left(\frac1n+\frac{4}{N^2}\right) \\
&\le
C_{\mathrm{IC}}\frac5n
=
\frac{5C_{\mathrm{IC}}}{n}
\le
\frac{C_\epsilon}{n}.
\end{aligned}
\]
This proves the asserted two-sided bound for all \(n\ge1\) and \(m\ge0\). % lean: CausalSmith.Stat.SemisupervisedDiscreteAteAnnotationFrontier.inverse_count_baseline
\end{enumerate}
\end{proof}
