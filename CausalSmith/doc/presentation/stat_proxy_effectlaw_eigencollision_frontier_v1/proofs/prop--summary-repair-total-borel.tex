\begin{proof}[Proof of \cref{obj:prop:summary-repair-total-borel}]
\begin{enumerate}
\item Write
\[
\mathcal K=\overline{\mathscr S}^{\,d_S}
       =\operatorname{cl}_{d_S}\{S(P):P\in\mathcal M\}.
\]
By \cref{obj:prop:summary-closure-compact}, \(\mathcal K\) is compact.  Also \(2\le k\) gives \(0<k\), and \cref{obj:def:effect-radius} with \(L\ge 1\) and \(0<\sigma_0\le 1\) gives \(L_\tau\ge 0\).  These two inequalities supply the positivity data needed for the atomic-law fallback \(\delta_0\).  From \cref{obj:thm:gap-free-positive-measure-modulus} we take a continuous, hence Borel, extension
\[
\overline F:\mathcal K\longrightarrow \mathcal P_k([-L_\tau,L_\tau])
\]
satisfying, for every admissible model law \(P\),
\[
\overline F(S(P))=\nu_P .
\]
\hfill{}% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.summary_repair_total_borel

\item First suppose \(\mathcal K=\varnothing\).  Define the selector on the full five-block summary space by
\[
\Pi(s)=0\qquad\text{for every summary }s .
\]
This map is Borel.  The extension condition over model summaries is vacuous, because any admissible \(P\) would give
\[
S(P)\in\mathscr S\subseteq \mathcal K,
\]
contradicting \(\mathcal K=\varnothing\).  The nonempty-\(\mathcal K\) nearest-point clause in \cref{obj:def:summary-space-repair} likewise has a false premise.  Thus these choices define admissible repair data \(R\).  For every observed sample \(\mathsf o_{1:n}\in O_{d_x,d_z}^n\), \cref{obj:def:summary-space-repair} gives
\[
\widehat\nu_n(\mathsf o_{1:n})
=\delta_0,
\]
so \(\widehat\nu_n\) is the constant Borel map.  In the same case, \cref{obj:def:wasserstein-confidence-set} gives the retained set
\[
\{\delta_0\},
\]
which is nonempty. \hfill{}% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.summary_repair_total_borel

\item Now suppose \(\mathcal K\ne\varnothing\).  Let
\[
d_{\mathrm{sum}}=4d_zd_x+d_x
\]
be the number of real coordinates in a five-block summary, and let
\[
\Psi_{\mathrm{sum}}(s)
=\bigl(s.M_0,s.M_1,s.N_0,s.N_1,s.m_X\bigr)\in\mathbb R^{d_{\mathrm{sum}}}
\]
be the coordinate homeomorphism from the summary space to Euclidean space.  Set
\[
\mathcal K^\Psi=\Psi_{\mathrm{sum}}(\mathcal K)
\]
and, for \(x_{\mathrm{Euc}},y_{\mathrm{Euc}}\in\mathbb R^{d_{\mathrm{sum}}}\), define the transported loss
\[
\ell_\Psi(x_{\mathrm{Euc}},y_{\mathrm{Euc}})
=
d_S\!\left(\Psi_{\mathrm{sum}}^{-1}(y_{\mathrm{Euc}}),
           \Psi_{\mathrm{sum}}^{-1}(x_{\mathrm{Euc}})\right).
\]
The set \(\mathcal K^\Psi\) is nonempty and compact, and \(\ell_\Psi\) is continuous because \(\Psi_{\mathrm{sum}}\) is a homeomorphism and \cref{obj:lem:summary-metric-continuity} gives continuity of \(d_S\).  Applying \cref{obj:lem:borel-nearest-point-selector} to \((\mathcal K^\Psi,\ell_\Psi)\), then transporting the selector back through \(\Psi_{\mathrm{sum}}\), gives a Borel map
\[
\Pi:\text{summary space}\longrightarrow \text{summary space}
\]
such that, for every summary \(s\),
\[
\Pi(s)\in\mathcal K
\quad\text{and}\quad
\forall q_{\mathrm{sum}}\in\mathcal K,\qquad
d_S(\Pi(s),s)\le d_S(q_{\mathrm{sum}},s).
\]
Together with the \(\overline F\) from the first step, this defines repair data \(R\) as in \cref{obj:def:summary-space-repair}; the fallback clause has premise \(\mathcal K=\varnothing\), which is false in the present case. \hfill{}% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.compactLoss_selector_of_homeomorph

\item For \(\mathcal K\ne\varnothing\), the repaired estimator is
\[
\widehat\nu_n(\mathsf o_{1:n})
=
\overline F\!\left(\Pi(\widehat S_n(\mathsf o_{1:n}))\right),
\qquad
\mathsf o_{1:n}\in O_{d_x,d_z}^n.
\]
The empirical summary map \(\widehat S_n\) is Borel by hypothesis, \(\Pi\) is Borel by the preceding step, and \(\overline F\) is Borel because it is continuous.  Hence \(\widehat\nu_n\) is Borel measurable. \hfill{}% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.summary_repair_total_borel

\item It remains in the nonempty case to verify nonemptiness of the retained summary-inversion set.  By \cref{obj:def:summary-radius},
\[
r_{n,\alpha}
=
C_0L\sqrt{\frac{\log(C_0/\alpha)}{n}} .
\]
Since \(n\ge 1\), \(0<\alpha<1/2\), \(C_0\ge 1\), and \(L\ge 1\), one has \(C_0/\alpha>1\), hence \(\log(C_0/\alpha)>0\), and therefore
\[
0<r_{n,\alpha}
\qquad\text{and}\qquad
0\le 2r_{n,\alpha}.
\]
For a fixed sample \(\mathsf o_{1:n}\), put
\[
q_{\mathrm{ret}}=\Pi(\widehat S_n(\mathsf o_{1:n})) .
\]
The selector property gives \(q_{\mathrm{ret}}\in\mathcal K\), and the defining formula for \(d_S\) gives
\[
d_S(q_{\mathrm{ret}},q_{\mathrm{ret}})=0\le 2r_{n,\alpha}.
\]
Moreover,
\[
\widehat\nu_n(\mathsf o_{1:n})
=
\overline F(q_{\mathrm{ret}}).
\]
Thus \(q_{\mathrm{ret}}\) witnesses membership of \(\widehat\nu_n(\mathsf o_{1:n})\) in the displayed retained set, so that set is nonempty for every sample. \hfill{}% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.summaryRadius_pos

\item The two cases \(\mathcal K=\varnothing\) and \(\mathcal K\ne\varnothing\) cover all possibilities.  In each case the constructed repair data \(R\) has Borel selector \(\Pi\), continuous extension \(\overline F\), Borel repaired estimator \(\widehat\nu_n\), the asserted \(\delta_0\) fallback when \(\mathcal K=\varnothing\), and a nonempty retained summary-inversion set for every sample. \hfill{}% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.summary_repair_total_borel
\end{enumerate}
\end{proof}
