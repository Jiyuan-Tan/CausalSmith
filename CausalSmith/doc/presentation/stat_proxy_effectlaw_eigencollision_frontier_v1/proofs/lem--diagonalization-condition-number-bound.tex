\begin{proof}[Proof of \cref{obj:lem:diagonalization-condition-number-bound}]
\begin{enumerate}
\item Write
\[
E_{r,c}:=\operatorname{Ent}_{r,c}.
\]
For these constants we use the following finite-dimensional entrywise comparison. If \(\mathsf A_{\mathrm{ent}}\in\mathbb R^{r\times c}\), \(0\le M_{\mathrm{ent}}\), and \(|(\mathsf A_{\mathrm{ent}})_{ab}|\le M_{\mathrm{ent}}\) for all entries, then
\[
\|\mathsf A_{\mathrm{ent}}\|_{\mathrm{op}}\le E_{r,c}M_{\mathrm{ent}} .
\]
Indeed, writing \(e_j^{(c)}\) for the standard coordinate vectors and \(J_c\) for the coordinate identification in the definition of \(\operatorname{Ent}_{r,c}\), each column satisfies
\[
\|\mathsf A_{\mathrm{ent}}e_j^{(c)}\|_2\le rM_{\mathrm{ent}} .
\]
Thus, for every \(x\in\mathbb R^c\),
\[
\begin{aligned}
\|\mathsf A_{\mathrm{ent}}x\|_2
&\le \sum_{j=1}^c |(J_cx)_j|\,\|\mathsf A_{\mathrm{ent}}e_j^{(c)}\|_2 \\
&\le c\,r\,M_{\mathrm{ent}}\|J_cx\|_2 \\
&\le E_{r,c}M_{\mathrm{ent}}\|x\|_2,
\end{aligned}
\]
which gives the displayed operator-norm bound. In particular \(E_{r,c}\ge0\). Since the columns of \(V\) are orthonormal, every coordinate of \(V\) has absolute value at most one, and therefore
\[
\|V\|_{\mathrm{op}}\le E_{d_x,k},
\qquad
\|V^\top\|_{\mathrm{op}}\le E_{k,d_x}.
\]
The same comparison gives
\[
\|I_{d_x}\|_{\mathrm{op}}\le E_{d_x,d_x}.
\]
% lean: matrixNorm_le_entryBound

\item Let \(\Upsilon\) and \(\Upsilon^{-1}\) be the coordinate matrix and coordinate inverse from the thin factorization \(B=V\Upsilon\). Since \(V^\top V=I_k\),
\[
\Upsilon=V^\top B .
\]
Thus, for coordinate indices \(r,u\),
\[
|\Upsilon_{ru}|
=
\left|\sum_{i=1}^{d_x}V_{ir}B_{iu}\right|
\le
\sum_{i=1}^{d_x}|V_{ir}|\,|B_{iu}|
\le
\sum_{i=1}^{d_x}L
=
d_xL .
\]
Consequently,
\[
\|\Upsilon^\top\|_{\mathrm{op}}\le E_{k,k}(d_xL),
\qquad
\|(\Upsilon^{-1})^\top\|_{\mathrm{op}}\le E_{k,k}\sigma_0^{-1}.
\]
% lean: ThinSignalFactorization.coord_entry_bound

\item Define the forward and backward ambient maps
\[
\Phi_{\mathrm{forw}}
:=
V(\Upsilon^{-1})^\top V^\top+(I_{d_x}-VV^\top),
\qquad
\Phi_{\mathrm{back}}
:=
V\Upsilon^\top V^\top+(I_{d_x}-VV^\top).
\]
Using submultiplicativity, the triangle inequality, and the bounds above,
\[
\begin{aligned}
\|\Phi_{\mathrm{forw}}\|_{\mathrm{op}}
&\le
\|V(\Upsilon^{-1})^\top V^\top\|_{\mathrm{op}}
+
\|I_{d_x}-VV^\top\|_{\mathrm{op}}\\
&\le
E_{d_x,k}E_{k,k}E_{k,d_x}\sigma_0^{-1}
+
\bigl(E_{d_x,d_x}+E_{d_x,k}E_{k,d_x}\bigr)
=
K_{\mathrm{forw}},
\end{aligned}
\]
and similarly
\[
\begin{aligned}
\|\Phi_{\mathrm{back}}\|_{\mathrm{op}}
&\le
\|V\Upsilon^\top V^\top\|_{\mathrm{op}}
+
\|I_{d_x}-VV^\top\|_{\mathrm{op}}\\
&\le
E_{d_x,k}E_{k,k}E_{k,d_x}(d_xL)
+
\bigl(E_{d_x,d_x}+E_{d_x,k}E_{k,d_x}\bigr)
=
K_{\mathrm{back}} .
\end{aligned}
\]
Both constants are nonnegative because \(L\ge0\), \(\sigma_0>0\), and \(E_{r,c}\ge0\). % lean: ThinSignalFactorization.forward_norm_bound; ThinSignalFactorization.backward_norm_bound

\item Let \((q_j)_{j=1}^{d_x}\) be the ambient orthonormal extension of the signal columns used in the associated real diagonalization, and let the diagonalizing basis vectors be
\[
s_j:=\Phi_{\mathrm{forw}}q_j,\qquad j=1,\ldots,d_x .
\]
The backward map is the inverse of the forward map. To see this, put \(P_\perp:=I_{d_x}-VV^\top\). Then
\[
V^\top P_\perp=0,\qquad P_\perp V=0,\qquad P_\perp^2=P_\perp,
\]
and the identities \(\Upsilon\Upsilon^{-1}=I_k\) and \(\Upsilon^{-1}\Upsilon=I_k\) give
\[
\Phi_{\mathrm{forw}}\Phi_{\mathrm{back}}
=
VV^\top+P_\perp
=
I_{d_x},
\qquad
\Phi_{\mathrm{back}}\Phi_{\mathrm{forw}}
=
VV^\top+P_\perp
=
I_{d_x}.
\]
Hence the inverse-coordinate matrix of the basis \((s_j)_j\) is computed by first applying \(\Phi_{\mathrm{back}}\) and then taking coordinates in the orthonormal basis \((q_j)_j\). % lean: ThinSignalFactorization.forward_mul_backward; ThinSignalFactorization.backward_mul_forward; ThinSignalFactorization.linearEquiv

\item Let \(\mathsf{Basis}_{\mathrm{diag}}\) be the matrix with columns \(s_j\), and let \(\mathsf{Basis}_{\mathrm{diag}}^{-1}\) be its coordinate inverse. For every pair of ambient indices \(i,j\),
\[
|(\mathsf{Basis}_{\mathrm{diag}})_{ij}|
=
|(s_j)_i|
\le
\|s_j\|_2
=
\|\Phi_{\mathrm{forw}}q_j\|_2
\le
\|\Phi_{\mathrm{forw}}\|_{\mathrm{op}}\|q_j\|_2
\le
K_{\mathrm{forw}},
\]
because \(\|q_j\|_2=1\). Therefore
\[
\|\mathsf{Basis}_{\mathrm{diag}}\|_{\mathrm{op}}
\le
E_{d_x,d_x}K_{\mathrm{forw}} .
\]
Likewise, for the inverse-coordinate entries,
\[
|(\mathsf{Basis}_{\mathrm{diag}}^{-1})_{ij}|
=
|\langle q_i,\Phi_{\mathrm{back}}e_j\rangle|
\le
\|q_i\|_2\|\Phi_{\mathrm{back}}e_j\|_2
\le
K_{\mathrm{back}},
\]
so
\[
\|\mathsf{Basis}_{\mathrm{diag}}^{-1}\|_{\mathrm{op}}
\le
E_{d_x,d_x}K_{\mathrm{back}} .
\]
The condition number of the associated real diagonalization is the product of these two operator norms, hence
\[
\operatorname{cond}
\le
\bigl(E_{d_x,d_x}K_{\mathrm{forw}}\bigr)
\bigl(E_{d_x,d_x}K_{\mathrm{back}}\bigr)
=
E_{d_x,d_x}^{\,2}K_{\mathrm{forw}}K_{\mathrm{back}}
=
\kappa_\star .
\]
% lean: ThinSignalFactorization.diagonalization_conditionNumber_le
\end{enumerate}
\end{proof}
