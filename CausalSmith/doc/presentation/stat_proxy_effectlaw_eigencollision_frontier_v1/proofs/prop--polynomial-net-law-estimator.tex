\begin{proof}[Proof of \cref{obj:prop:polynomial-net-law-estimator}]
Set
\[
s_0:=\pi_0\sigma_0^2,\qquad
L_\tau:=\frac{4L\sqrt{d_z}}{\sigma_0},
\]
and order singular values nonincreasingly.

\begin{enumerate}
\item Let \(H_n,q_n,\ell_n\) and
\[
\mathcal A_n=\mathcal V_n\times\mathcal R_n\times\mathcal P_n\times\mathcal T_n
\]
be the structured lattice of \cref{obj:synth_32}.  Its four factors have cardinalities bounded by fixed-parameter constants times
\[
H_n^{d_xk},\qquad H_n^{k^2},\qquad H_n^{k-1},\qquad H_n^k .
\]
Since \(n\ge1\), the lattice height satisfies
\[
H_n\le C_H\sqrt n
\]
for a finite \(C_H=C_H(k,d_x,\pi_0,\sigma_0)\).  Hence there is a constant
\(C_\#>0\), depending only on the displayed parameters, such that
\[
|\mathcal A_n|
\le C_\# n^{(d_xk+k^2+2k-1)/2}.
\]
In the fixed-dimensional exact-real model of \cref{obj:synth_16}, the work is the empirical-summary pass plus one fixed-size objective evaluation for each candidate, so
\[
\operatorname{Work}_n
\le
C_\#\left\{n+n^{(d_xk+k^2+2k-1)/2}\right\}.
\] % lean: prescribed_latticeOperationCount_polynomial_bound

\item Choose \(C_0\ge1\) from \cref{obj:lem:uniform-summary-concentration}.  Thus, for every \(P\in\mathcal M\), \(n\ge1\), and \(0<\eta<1/2\),
\[
Q_P^{(n)}
\left\{
d_S(\widehat S_n,S(P))
>
C_0L\sqrt{\frac{\log(C_0/\eta)}{n}}
\right\}
\le\eta .
\]
The assumptions imply \(d_z>0\), \(L_\tau>0\), and \(s_0>0\).  Therefore
\[
0<\frac{8L_\tau}{s_0}
\le
C_{\mathrm{lat}}
=
\max\!\left\{\frac{8L_\tau}{s_0},
(L_\tau+K_D+LK_f)B_{\mathrm{lat}}\right\}.
\]
Define
\[
C_{\mathrm{tail}}
:=
\max\!\left\{C_0,\mathrm e,C_{\mathrm{lat}}(C_0L+1)\right\},
\qquad
C:=\max\{C_\#,C_{\mathrm{tail}}\}.
\]
Then \(C>0\). % lean: prescribedLatticeConstant_pos

\item Fix \(n\ge1\).  The lattice factors are finite and nonempty: canonical coordinate columns furnish an element of \(\mathcal V_n\), a rounded scalar multiple of \(I_k\) furnishes an element of \(\mathcal R_n\), the inequality \(k\ell_n\le H_n\) furnishes an element of \(\mathcal P_n\), and \(0\in\mathcal T_n\).  Enumerate \(\mathcal A_n\) in duplicate-free lexicographic order as
\[
\vartheta_i=(V_i,R_i,p_i,\tau_i),
\qquad i=1,\ldots,|\mathcal A_n|.
\]
For a summary \(s\), put
\[
i_n(s):=
\min_{\mathrm{lex}}\operatorname*{arg\,min}_{1\le i\le|\mathcal A_n|}
J_n(\vartheta_i;s),
\qquad
\Lambda_n(s):=\lambda_{\vartheta_{i_n(s)}} ,
\]
where \(J_n\) is the criterion in \cref{obj:synth_37}.  Thresholded singular-value pseudoinversion is Borel, each score \(s\mapsto J_n(\vartheta_i;s)\) is Borel, and a finite first-minimizer rule over Borel scores is Borel.  Hence
\[
\widehat\lambda_n(\omega):=\Lambda_n(\widehat S_n(\omega))
\]
is a total Borel empirical-summary rule, with \(\widehat S_n\) as in \cref{obj:def:empirical-summary}, and it is the first minimizer prescribed by \cref{obj:def:structured-lattice-program}.  Every coordinate of every \(p_i\in\mathcal P_n\) is at least \(\pi_0\), so aggregating coordinates at repeated effect locations gives
\[
\operatorname{AtomFloor}(\pi_0,\widehat\lambda_n(\omega))
\]
for every sample \(\omega\). % lean: structuredLatticeEstimator_exists

\item Fix \(P\in\mathcal M\), and put
\[
G_P:=\operatorname{stack}(M_0(P),M_1(P)).
\]
\Cref{obj:lem:model-compressed-spectral-certificate} gives \(\operatorname{rank}(G_P)=k\) and the threshold characterization at \(S(P)\).  In addition, \cref{obj:lem:stacked-proxy-moment-signal-margin} gives the population signal margin; since the singular values are nonincreasing,
\[
\sigma_j(G_P)\ge s_0\quad (1\le j\le k),
\qquad
\sigma_j(G_P)=0\quad (j>k).
\]
In particular, \(G_P\) has exactly its first \(k\) singular values at least \(s_0/2\).  Now let \(s\) satisfy
\[
d_S(s,S(P))<s_0/2,
\]
and set \(G_s:=\operatorname{stack}(M_0(s),M_1(s))\).  The stacked-block comparison gives
\[
\|G_s-G_P\|_{\mathrm{op}}\le d_S(s,S(P))<s_0/2.
\]
Weyl's inequality yields
\[
|\sigma_j(G_s)-\sigma_j(G_P)|<s_0/2
\qquad\text{for every }j.
\]
Thus, for \(1\le j\le k\),
\[
\sigma_j(G_s)>s_0/2,
\]
while, for \(j>k\),
\[
\sigma_j(G_s)<s_0/2.
\]
This proves that \(G_s\) also has exactly its first \(k\) singular values at least \(s_0/2\) and all remaining singular values below \(s_0/2\). % lean: model_thresholdRecoversDimension_of_dS_lt

\item Fix a sample \(\omega\), and write
\[
s_\omega:=\widehat S_n(\omega),
\qquad
\varepsilon_{\omega,P}:=d_S(s_\omega,S(P)).
\]
Choose the population signal tuple
\[
B(P)=V_{\mathrm{pop}}R_{\mathrm{pop}}^\top,\qquad
p_{\mathrm{pop}}=(P(U=u))_{u=1}^k,\qquad
\tau_{\mathrm{pop}}=(\tau_u(P))_{u=1}^k .
\]
By \cref{obj:def:target-feature,obj:def:latent-mass}, and because the latent classes partition the record space with positive class masses,
\[
m_X(P)
=E_P[X]
=\sum_{u=1}^k P(U=u)E_P[X\mid U=u]
=B(P)p_{\mathrm{pop}}.
\]
The chosen signal factorization \(B(P)=V_{\mathrm{pop}}R_{\mathrm{pop}}^\top\) therefore gives
\[
m_X(P)=V_{\mathrm{pop}}R_{\mathrm{pop}}^\top p_{\mathrm{pop}}.
\]
Using \cref{obj:ass:anchor}, the first target-proxy coordinate has conditional mean one in every latent class:
\[
B(P)^\top e_1
=
\bigl(E_P[X_1\mid U=u]\bigr)_{u=1}^k
=\mathbf1_k.
\]
Since \(B(P)^\top=R_{\mathrm{pop}}V_{\mathrm{pop}}^\top\), this yields
\[
R_{\mathrm{pop}}V_{\mathrm{pop}}^\top e_1=\mathbf1_k.
\]
The proxy and outcome moment factorizations from
\cref{obj:lem:observed-proxy-moment-factorization,obj:lem:observed-outcome-proxy-moment-factorization} give, for each arm \(t\),
\[
M_t(P)=A_t\Gamma_tB(P)^\top,
\qquad
N_t(P)=A_t\Gamma_t\operatorname{diag}(\mu_{tu})B(P)^\top .
\]
The proxy-rank margin in \cref{obj:ass:proxy-rank-margin} supplies injectivity of the linear maps represented by \(A_t\) and \(B(P)\).  By \cref{obj:lem:latent-arm-weight-invertibility}, \(\Gamma_t\) is injective, so \(A_t\Gamma_t\) is injective.  Applying \cref{obj:lem:moore-penrose-product-cancellation} with \(C=A_t\Gamma_t\), \(B=B(P)\), and \(\mu=(\mu_{tu})_u\) yields
\[
M_t(P)^\dagger N_t(P)
=
\{B(P)^\dagger\}^\top\operatorname{diag}(\mu_{tu})B(P)^\top .
\]
Subtracting the two arms and using \(\tau_{\mathrm{pop},u}=\mu_{1u}-\mu_{0u}\), together with \(B(P)=V_{\mathrm{pop}}R_{\mathrm{pop}}^\top\), gives
\[
\mathcal D_P:=
M_1(P)^\dagger N_1(P)-M_0(P)^\dagger N_0(P)
=
V_{\mathrm{pop}}R_{\mathrm{pop}}^{-1}
\operatorname{diag}(\tau_{\mathrm{pop}})
R_{\mathrm{pop}}V_{\mathrm{pop}}^\top .
\]
Moreover,
\[
\sigma_k(R_{\mathrm{pop}})\ge\sigma_0,\qquad
\|R_{\mathrm{pop}}\|_{\mathrm{op}}\le\sqrt{k}L,
\]
\[
p_{\mathrm{pop},u}\ge2\pi_0,\qquad
\sum_{u=1}^k p_{\mathrm{pop},u}=1,\qquad
|\tau_{\mathrm{pop},u}|\le L_\tau ,
\]
where the effect envelope is supplied by \cref{obj:lem:latent-effect-envelope}.  The associated quotient law is
\[
\nu_P=\sum_{u=1}^k p_{\mathrm{pop},u}\delta_{\tau_{\mathrm{pop},u}} .
\] % lean: population_structured_tuple_exists

\item Recall that \(q_n=H_n^{-1}\).  The definition of \(H_n\) gives the four mesh inequalities
\[
\sqrt{d_xk}\,q_n\le\frac14,
\qquad
kq_n\le\frac{\sigma_0}{2},
\qquad
kq_n\le\sqrt{k}L,
\qquad
\lceil\pi_0^{-1}\rceil\le H_n .
\]
For the signal basis \(V_{\mathrm{pop}}\), each entry has absolute value at most one because its columns are orthonormal.  Coordinatewise clipped rounding therefore gives a grid matrix \(V_{n,\mathrm{raw}}\) with
\[
(V_{n,\mathrm{raw}})_{ij}=q_n z_{ij}\quad (z_{ij}\in\mathbb Z),
\qquad |(V_{n,\mathrm{raw}})_{ij}|\le1,
\qquad
\|V_{n,\mathrm{raw}}-V_{\mathrm{pop}}\|_{\mathrm{op}}
\le \sqrt{d_xk}\,q_n .
\]
Weyl's inequality and the first mesh bound give
\[
\sigma_k(V_{n,\mathrm{raw}})\ge1-\sqrt{d_xk}\,q_n\ge\frac34\ge\frac12 .
\]
Let \(V_n^\circ\) be the polar factor \(V_{n,\mathrm{raw}}(V_{n,\mathrm{raw}}^\top V_{n,\mathrm{raw}})^{-1/2}\).  The polar-factor perturbation estimate at an orthonormal basis yields
\[
\|V_n^\circ-V_{\mathrm{pop}}\|_{\mathrm{op}}
\le4\sqrt{d_xk}\,q_n=c_Vq_n .
\]
Next, since \(\|R_{\mathrm{pop}}\|_{\mathrm{op}}\le\sqrt{k}L\), coordinatewise clipped rounding in the radius \(2\sqrt{k}L\) gives \(R_n^\circ\) with grid entries and
\[
\|R_n^\circ-R_{\mathrm{pop}}\|_{\mathrm{op}}\le kq_n .
\]
Using \(kq_n\le\sigma_0/2\) and Weyl's inequality,
\[
\sigma_k(R_n^\circ)\ge\sigma_k(R_{\mathrm{pop}})-kq_n
\ge\frac{\sigma_0}{2},
\]
and using \(kq_n\le\sqrt{k}L\),
\[
\|R_n^\circ\|_{\mathrm{op}}
\le \|R_{\mathrm{pop}}\|_{\mathrm{op}}+\|R_n^\circ-R_{\mathrm{pop}}\|_{\mathrm{op}}
\le2\sqrt{k}L .
\]
For the weight vector, set \(b_u:=\lfloor H_np_{\mathrm{pop},u}\rfloor\).  Since \(p_{\mathrm{pop},u}\ge2\pi_0\) and \(\lceil\pi_0^{-1}\rceil\le H_n\), the quantities \(x:=\pi_0H_n\) and \(y_u:=H_np_{\mathrm{pop},u}\) satisfy \(1\le x\) and \(2x\le y_u\).  Because \(y_u< b_u+1\), this implies \(x\le b_u\), hence
\[
\lceil\pi_0H_n\rceil\le b_u .
\]
The floor vector satisfies \(\sum_u b_u\le H_n\) and \(H_n\le\sum_u b_u+k\).  Choose a subset of indices of size \(H_n-\sum_u b_u\) and add one to \(b_u\) on that subset, obtaining integers \(a_u\) with
\[
\sum_{u=1}^k a_u=H_n,
\qquad
\lceil\pi_0H_n\rceil\le a_u,
\qquad
\left|\frac{a_u}{H_n}-p_{\mathrm{pop},u}\right|\le q_n .
\]
Thus \(p_n^\circ:=(a_u/H_n)_{u=1}^k\) belongs to \(\mathcal P_n\), and summing the squared coordinate errors gives
\[
\|p_n^\circ-p_{\mathrm{pop}}\|_2
=\left\{\sum_{u=1}^k\left(\frac{a_u}{H_n}-p_{\mathrm{pop},u}\right)^2\right\}^{1/2}
\le\sqrt{k}q_n .
\]
Finally, since \(|\tau_{\mathrm{pop},u}|\le L_\tau\), coordinatewise clipped rounding on the effect grid gives \(\tau_n^\circ\in\mathcal T_n\) with
\[
|\tau_{n,u}^\circ-\tau_{\mathrm{pop},u}|\le q_n
\qquad(1\le u\le k).
\]
Combining these four rounded components gives a well-formed lattice comparator
\[
\vartheta_n^\circ=(V_n^\circ,R_n^\circ,p_n^\circ,\tau_n^\circ)\in\mathcal A_n
\]
satisfying
\[
\|V_n^\circ-V_{\mathrm{pop}}\|_{\mathrm{op}}\le c_Vq_n,\qquad
\|R_n^\circ-R_{\mathrm{pop}}\|_{\mathrm{op}}\le kq_n,
\]
\[
\|p_n^\circ-p_{\mathrm{pop}}\|_2\le\sqrt{k}q_n,\qquad
|\tau_{n,u}^\circ-\tau_{\mathrm{pop},u}|\le q_n .
\] % lean: population_structuredLatticeComparator_exists

\item Assume first that
\[
\varepsilon_{\omega,P}<s_0/4 .
\]
For each treatment arm,
\[
\|M_t(s_\omega)-M_t(P)\|_{\mathrm{op}}\le\varepsilon_{\omega,P},
\qquad
\|N_t(s_\omega)-N_t(P)\|_{\mathrm{op}}\le\varepsilon_{\omega,P}.
\]
The population armwise proxy block has rank \(k\), kth singular value at least \(s_0\), and \(\|N_t(P)\|_{\mathrm{op}}\le L\).  If \(\widetilde M_t(s_\omega)\) denotes the rank-\(k\) truncation retained at threshold \(s_0/2\), singular-value perturbation gives
\[
(M_t(s_\omega))^\dagger_{\ge s_0/2}=\widetilde M_t(s_\omega)^\dagger,
\quad
\sigma_k(\widetilde M_t(s_\omega))\ge3s_0/4,
\quad
\|\widetilde M_t(s_\omega)-M_t(P)\|_{\mathrm{op}}
\le2\varepsilon_{\omega,P}.
\]
Define the thresholded empirical compressed operator, in the notation of \cref{obj:synth_37}, by
\[
\widehat D(s_\omega):=
M_1(s_\omega)^\dagger{}_{\ge s_0/2}N_1(s_\omega)
-
M_0(s_\omega)^\dagger{}_{\ge s_0/2}N_0(s_\omega).
\]
The equal-rank Moore--Penrose perturbation identity gives
\[
\|\mathcal U^\dagger-\mathcal V^\dagger\|_{\mathrm{op}}
\le
3\max\{\|\mathcal U^\dagger\|_{\mathrm{op}}^2,\|\mathcal V^\dagger\|_{\mathrm{op}}^2\}
\|\mathcal U-\mathcal V\|_{\mathrm{op}},
\]
and therefore
\[
\|\widehat D(s_\omega)-\mathcal D_P\|_{\mathrm{op}}
\le A_D\varepsilon_{\omega,P},
\qquad
\|m_X(s_\omega)-m_X(P)\|_2\le\varepsilon_{\omega,P}.
\]
For the comparator \(\vartheta_n^\circ\), the mean residual decomposes as
\[
V_n^\circ(R_n^\circ)^\top p_n^\circ
-
V_{\mathrm{pop}}R_{\mathrm{pop}}^\top p_{\mathrm{pop}}
=
(V_n^\circ-V_{\mathrm{pop}})(R_n^\circ)^\top p_n^\circ
+
V_{\mathrm{pop}}(R_n^\circ-R_{\mathrm{pop}})^\top p_n^\circ
+
V_{\mathrm{pop}}R_{\mathrm{pop}}^\top(p_n^\circ-p_{\mathrm{pop}}),
\]
so the bounds \(\|V_{\mathrm{pop}}\|_{\mathrm{op}}\le1\), \(\|p_n^\circ\|_2\le1\),
\(\|R_n^\circ\|_{\mathrm{op}}\le2\sqrt{k}L\), and
\(\|R_{\mathrm{pop}}\|_{\mathrm{op}}\le\sqrt{k}L\) give
\[
\|V_n^\circ(R_n^\circ)^\top p_n^\circ
-
V_{\mathrm{pop}}R_{\mathrm{pop}}^\top p_{\mathrm{pop}}\|_2
\le c_mq_n .
\]
Similarly,
\[
R_n^\circ(V_n^\circ)^\top e_1
-
R_{\mathrm{pop}}V_{\mathrm{pop}}^\top e_1
=
(R_n^\circ-R_{\mathrm{pop}})V_{\mathrm{pop}}^\top e_1
+
R_n^\circ(V_n^\circ-V_{\mathrm{pop}})^\top e_1,
\]
and \(\|e_1\|_2=1\), \(\|V_{\mathrm{pop}}\|_{\mathrm{op}}\le1\), and
\(\|R_n^\circ\|_{\mathrm{op}}\le2\sqrt{k}L\) yield
\[
\|R_n^\circ(V_n^\circ)^\top e_1-\mathbf 1_k\|_2\le c_bq_n .
\]
For the operator residual, write
\[
\begin{aligned}
&D_{\vartheta_n^\circ}-\mathcal D_P \\
&=(V_n^\circ-V_{\mathrm{pop}})(R_n^\circ)^{-1}\operatorname{diag}(\tau_n^\circ)R_n^\circ(V_n^\circ)^\top\\
&\quad+V_{\mathrm{pop}}\{(R_n^\circ)^{-1}-R_{\mathrm{pop}}^{-1}\}\operatorname{diag}(\tau_n^\circ)R_n^\circ(V_n^\circ)^\top\\
&\quad+V_{\mathrm{pop}}R_{\mathrm{pop}}^{-1}\operatorname{diag}(\tau_n^\circ-\tau_{\mathrm{pop}})R_n^\circ(V_n^\circ)^\top\\
&\quad+V_{\mathrm{pop}}R_{\mathrm{pop}}^{-1}\operatorname{diag}(\tau_{\mathrm{pop}})(R_n^\circ-R_{\mathrm{pop}})(V_n^\circ)^\top\\
&\quad+V_{\mathrm{pop}}R_{\mathrm{pop}}^{-1}\operatorname{diag}(\tau_{\mathrm{pop}})R_{\mathrm{pop}}(V_n^\circ-V_{\mathrm{pop}})^\top .
\end{aligned}
\]
The inverse identity
\[
(R_n^\circ)^{-1}-R_{\mathrm{pop}}^{-1}
=
(R_n^\circ)^{-1}(R_{\mathrm{pop}}-R_n^\circ)R_{\mathrm{pop}}^{-1}
\]
and the singular-value bounds give
\[
\|(R_n^\circ)^{-1}\|_{\mathrm{op}}\le\frac{2}{\sigma_0},
\qquad
\|R_{\mathrm{pop}}^{-1}\|_{\mathrm{op}}\le\frac1{\sigma_0},
\qquad
\|(R_n^\circ)^{-1}-R_{\mathrm{pop}}^{-1}\|_{\mathrm{op}}
\le\frac{2kq_n}{\sigma_0^2}.
\]
Bounding the five displayed summands with
\(\|\operatorname{diag}(\tau_n^\circ)\|_{\mathrm{op}}\le L_\tau\),
\(\|\operatorname{diag}(\tau_{\mathrm{pop}})\|_{\mathrm{op}}\le L_\tau\), and
\(\|\operatorname{diag}(\tau_n^\circ-\tau_{\mathrm{pop}})\|_{\mathrm{op}}\le q_n\)
gives
\[
\|D_{\vartheta_n^\circ}-\mathcal D_P\|_{\mathrm{op}}\le c_Dq_n.
\]
Consequently
\[
J_n(\vartheta_n^\circ;s_\omega)
\le
(A_D+1)\varepsilon_{\omega,P}+c_{\mathrm{grid}}q_n .
\]
Let
\[
\widehat\vartheta_n(\omega)
=(\widehat V_n,\widehat R_n,\widehat p_n,\widehat\tau_n)
\]
be the first minimizer.  Minimality and the nonnegativity of the three terms in \(J_n\) imply
\[
\|D_{\widehat\vartheta_n}-\mathcal D_P\|_{\mathrm{op}}
\le(2A_D+1)\varepsilon_{\omega,P}+c_{\mathrm{grid}}q_n,
\]
\[
\|\widehat V_n\widehat R_n^\top\widehat p_n-m_X(P)\|_2
\le(A_D+2)\varepsilon_{\omega,P}+c_{\mathrm{grid}}q_n,
\]
and
\[
\|\widehat R_n\widehat V_n^\top e_1-\mathbf1_k\|_2
\le(A_D+1)\varepsilon_{\omega,P}+c_{\mathrm{grid}}q_n.
\]
Since
\[
B_{\mathrm{lat}}=2(A_D+1)+c_{\mathrm{grid}},
\]
all three residuals are bounded by
\[
B_{\mathrm{lat}}(\varepsilon_{\omega,P}+q_n).
\] % lean: selected_structuredLattice_population_residuals_le

\item Let \(\varphi_\omega\) be the attaining Kantorovich--Rubinstein potential for
\((\widehat\lambda_n(\omega),\nu_P)\) supplied by
\cref{obj:lem:kantorovich-rubinstein-attaining-potential}.  Thus
\[
\varphi_\omega(0)=0,\qquad \operatorname{Lip}(\varphi_\omega)\le1,
\]
and
\[
W_1(\widehat\lambda_n(\omega),\nu_P)
=
\left|
\int\varphi_\omega\,d\widehat\lambda_n(\omega)
-
\int\varphi_\omega\,d\nu_P
\right|.
\]
The selected structured operator and \(\mathcal D_P\), extended by zero on the orthogonal complement of their signal spaces, have real diagonalizations with spectra in \([-L_\tau,L_\tau]\).  Put
\[
\kappa:=\frac{4\sqrt{k}L}{\sigma_0}.
\]
For either signal factorization with coordinate matrix \(R_{\mathrm{coord}}\), the ambient eigenbasis uses the signal coordinates and an orthonormal basis on the orthogonal complement.  Hence its condition number is bounded by
\[
\operatorname{cond}\le
\max\{\|R_{\mathrm{coord}}^{-1}\|_{\mathrm{op}},1\}
\max\{\|R_{\mathrm{coord}}\|_{\mathrm{op}},1\}.
\]
For the selected lattice tuple, the defining lattice constraints give
\[
\sigma_k(\widehat R_n)\ge\sigma_0/2,
\qquad
\|\widehat R_n\|_{\mathrm{op}}\le2\sqrt{k}L,
\qquad
|\widehat\tau_{n,u}|\le L_\tau .
\]
Therefore
\[
\|\widehat R_n^{-1}\|_{\mathrm{op}}\le\frac{2}{\sigma_0},
\qquad
\max\{\|\widehat R_n^{-1}\|_{\mathrm{op}},1\}\le\frac{2}{\sigma_0},
\qquad
\max\{\|\widehat R_n\|_{\mathrm{op}},1\}\le2\sqrt{k}L,
\]
where the last two inequalities use \(0<\sigma_0\le1\), \(2\le k\), and \(1\le L\).  Thus the selected diagonalization has condition number at most \(\kappa\).  For the population tuple,
\[
\sigma_k(R_{\mathrm{pop}})\ge\sigma_0,
\qquad
\|R_{\mathrm{pop}}\|_{\mathrm{op}}\le\sqrt{k}L,
\qquad
|\tau_{\mathrm{pop},u}|\le L_\tau,
\]
so the same calculation, with
\[
\|R_{\mathrm{pop}}^{-1}\|_{\mathrm{op}}\le\frac1{\sigma_0}\le\frac2{\sigma_0},
\qquad
\max\{\|R_{\mathrm{pop}}\|_{\mathrm{op}},1\}\le2\sqrt{k}L,
\]
gives the condition-number bound \(\kappa\) for the population diagonalization as well.
For the diagonalizations
\[
D_{\widehat\vartheta_n}=\mathcal Z_{\mathrm{hat}}\Lambda_{\mathrm{hat}}\mathcal Z_{\mathrm{hat}}^{-1},
\qquad
\mathcal D_P=\mathcal Z_{\mathrm{pop}}\Lambda_{\mathrm{pop}}\mathcal Z_{\mathrm{pop}}^{-1},
\]
the divided-difference identity gives
\[
\{\varphi_\omega(\Lambda_{\mathrm{hat}})\mathcal Z_{\mathrm{hat}}^{-1}\mathcal Z_{\mathrm{pop}}
-\mathcal Z_{\mathrm{hat}}^{-1}\mathcal Z_{\mathrm{pop}}\varphi_\omega(\Lambda_{\mathrm{pop}})\}_{ij}
=
d_{ij}\{\Lambda_{\mathrm{hat}}\mathcal Z_{\mathrm{hat}}^{-1}\mathcal Z_{\mathrm{pop}}
-\mathcal Z_{\mathrm{hat}}^{-1}\mathcal Z_{\mathrm{pop}}\Lambda_{\mathrm{pop}}\}_{ij},
\qquad |d_{ij}|\le1,
\]
with \(d_{ij}=0\) when the two eigenvalues coincide.  The coordinate perturbation in the last display is
\[
\Lambda_{\mathrm{hat}}\mathcal Z_{\mathrm{hat}}^{-1}\mathcal Z_{\mathrm{pop}}
-\mathcal Z_{\mathrm{hat}}^{-1}\mathcal Z_{\mathrm{pop}}\Lambda_{\mathrm{pop}}
=
\mathcal Z_{\mathrm{hat}}^{-1}
(D_{\widehat\vartheta_n}-\mathcal D_P)
\mathcal Z_{\mathrm{pop}}.
\]
Since \(|d_{ij}|\le1\), each column of the Schur product has Euclidean norm bounded by the corresponding column norm of this coordinate perturbation.  The fixed-dimensional column estimate therefore gives
\[
\left\|
\varphi_\omega(\Lambda_{\mathrm{hat}})\mathcal Z_{\mathrm{hat}}^{-1}\mathcal Z_{\mathrm{pop}}
-\mathcal Z_{\mathrm{hat}}^{-1}\mathcal Z_{\mathrm{pop}}\varphi_\omega(\Lambda_{\mathrm{pop}})
\right\|_{\mathrm{op}}
\le
\sqrt{d_x}\,
\left\|
\mathcal Z_{\mathrm{hat}}^{-1}
(D_{\widehat\vartheta_n}-\mathcal D_P)
\mathcal Z_{\mathrm{pop}}
\right\|_{\mathrm{op}}.
\]
Multiplying by the exterior diagonalizer factors yields
\[
\begin{aligned}
\|\varphi_\omega(D_{\widehat\vartheta_n})-\varphi_\omega(\mathcal D_P)\|_{\mathrm{op}}
&\le
\sqrt{d_x}\,
\|\mathcal Z_{\mathrm{hat}}\|_{\mathrm{op}}
\|\mathcal Z_{\mathrm{hat}}^{-1}\|_{\mathrm{op}}
\|D_{\widehat\vartheta_n}-\mathcal D_P\|_{\mathrm{op}}
\|\mathcal Z_{\mathrm{pop}}\|_{\mathrm{op}}
\|\mathcal Z_{\mathrm{pop}}^{-1}\|_{\mathrm{op}} \\
&\le
\sqrt{d_x}\,\kappa^2
\|D_{\widehat\vartheta_n}-\mathcal D_P\|_{\mathrm{op}}.
\end{aligned}
\]
Because \(\kappa=4\sqrt{k}L/\sigma_0\), this is
\[
\|\varphi_\omega(D_{\widehat\vartheta_n})-\varphi_\omega(\mathcal D_P)\|_{\mathrm{op}}
\le K_f\|D_{\widehat\vartheta_n}-\mathcal D_P\|_{\mathrm{op}},
\qquad
K_f=\frac{16\sqrt{d_x}\,kL^2}{\sigma_0^2}.
\]
For the selected law, put
\[
\chi_{\widehat\vartheta,u}:=(\widehat R_n\widehat V_n^\top e_1)_u .
\]
Since \(\widehat V_n^\top\widehat V_n=I_k\) and \(\widehat R_n\widehat R_n^{-1}=I_k\), the functional calculus of the selected structured operator gives
\[
\varphi_\omega(D_{\widehat\vartheta_n})
=
\widehat V_n\widehat R_n^{-1}
\operatorname{diag}\{\varphi_\omega(\widehat\tau_{n,u}):1\le u\le k\}
\widehat R_n\widehat V_n^\top,
\]
because \(\varphi_\omega(0)=0\) on the orthogonal complement.  Therefore
\[
\begin{aligned}
&(\widehat V_n\widehat R_n^\top\widehat p_n)^\top
\varphi_\omega(D_{\widehat\vartheta_n})e_1 \\
&\quad=
\widehat p_n^\top\widehat R_n\widehat V_n^\top
\widehat V_n\widehat R_n^{-1}
\operatorname{diag}\{\varphi_\omega(\widehat\tau_{n,u})\}
\widehat R_n\widehat V_n^\top e_1 \\
&\quad=
\widehat p_n^\top
\operatorname{diag}\{\varphi_\omega(\widehat\tau_{n,u})\}
\widehat R_n\widehat V_n^\top e_1
=
\sum_{u=1}^k\widehat p_{n,u}\varphi_\omega(\widehat\tau_{n,u})\chi_{\widehat\vartheta,u}.
\end{aligned}
\]
The labelled integral of \(\widehat\lambda_n(\omega)\) is
\[
\int\varphi_\omega\,d\widehat\lambda_n(\omega)
=
\sum_{u=1}^k\widehat p_{n,u}\varphi_\omega(\widehat\tau_{n,u}),
\]
and hence
\[
\int\varphi_\omega\,d\widehat\lambda_n(\omega)
-
(\widehat V_n\widehat R_n^\top\widehat p_n)^\top
\varphi_\omega(D_{\widehat\vartheta_n})e_1
=
\sum_{u=1}^k
\widehat p_{n,u}\varphi_\omega(\widehat\tau_{n,u})(1-\chi_{\widehat\vartheta,u}).
\]
Because \(\|\widehat p_n\|_2\le1\) and
\[
|\varphi_\omega(\widehat\tau_{n,u})|
=
|\varphi_\omega(\widehat\tau_{n,u})-\varphi_\omega(0)|
\le|\widehat\tau_{n,u}|
\le L_\tau,
\]
Cauchy's inequality yields
\[
\left|
\int\varphi_\omega\,d\widehat\lambda_n(\omega)
-
(\widehat V_n\widehat R_n^\top\widehat p_n)^\top
\varphi_\omega(D_{\widehat\vartheta_n})e_1
\right|
\le
L_\tau\|\widehat R_n\widehat V_n^\top e_1-\mathbf1_k\|_2 .
\]
For the population tuple, functional calculus gives
\[
\varphi_\omega(\mathcal D_P)
=
V_{\mathrm{pop}}R_{\mathrm{pop}}^{-1}
\operatorname{diag}\{\varphi_\omega(\tau_{\mathrm{pop}})\}
R_{\mathrm{pop}}V_{\mathrm{pop}}^\top ,
\]
since \(\varphi_\omega(0)=0\) on the orthogonal complement.  Using
\(m_X(P)=V_{\mathrm{pop}}R_{\mathrm{pop}}^\top p_{\mathrm{pop}}\),
\(V_{\mathrm{pop}}^\top V_{\mathrm{pop}}=I_k\), and
\(R_{\mathrm{pop}}V_{\mathrm{pop}}^\top e_1=\mathbf1_k\), we obtain
\[
m_X(P)^\top\varphi_\omega(\mathcal D_P)e_1
=
\sum_{u=1}^k p_{\mathrm{pop},u}\varphi_\omega(\tau_{\mathrm{pop},u})
=
\int\varphi_\omega\,d\nu_P .
\]
Together with \(\|m_X(P)\|_2\le L\) and
\[
\|\varphi_\omega(D_{\widehat\vartheta_n})e_1\|_2
\le
\kappa L_\tau
=
K_D,
\]
the residual bounds from the previous step give, in the small-error case,
\[
W_1(\widehat\lambda_n(\omega),\nu_P)
\le
(L_\tau+K_D+LK_f)B_{\mathrm{lat}}(\varepsilon_{\omega,P}+q_n).
\]
If instead \(\varepsilon_{\omega,P}\ge s_0/4\), both laws are supported on
\([-L_\tau,L_\tau]\), so the diameter bound from \cref{obj:def:wasserstein-distance} gives
\[
W_1(\widehat\lambda_n(\omega),\nu_P)
\le2L_\tau
\le\frac{8L_\tau}{s_0}\varepsilon_{\omega,P}.
\]
Finally \(q_n\le n^{-1/2}\).  Combining the two cases with the definition of \(C_{\mathrm{lat}}\) proves, for every sample,
\[
W_1(\widehat\lambda_n(\omega),\nu_P)
\le
C_{\mathrm{lat}}
\left\{
d_S(\widehat S_n(\omega),S(P))+\frac1{\sqrt n}
\right\}.
\] % lean: prescribedEstimator_wass1_le

\item The choice \(C\ge C_\#\) gives the displayed candidate-count and work bounds with the same \(C\).  For the tail bound, fix \(0<\eta<1/2\).  Since
\[
C\ge C_0,\qquad C\ge\mathrm e,\qquad C\ge C_{\mathrm{lat}}(C_0L+1),
\]
we have
\[
\log(C_0/\eta)\le\log(C/\eta),
\qquad
1\le\log(C/\eta),
\qquad
\frac1{\sqrt n}\le\sqrt{\frac{\log(C/\eta)}{n}}.
\]
On the event
\[
d_S(\widehat S_n,S(P))
\le
C_0L\sqrt{\frac{\log(C_0/\eta)}{n}},
\]
the deterministic inequality gives
\[
W_1(\widehat\lambda_n,\nu_P)
\le
C_{\mathrm{lat}}(C_0L+1)
\sqrt{\frac{\log(C/\eta)}{n}}
\le
C\sqrt{\frac{\log(C/\eta)}{n}}.
\]
Therefore
\[
\left\{
W_1(\widehat\lambda_n,\nu_P)>
C\sqrt{\frac{\log(C/\eta)}{n}}
\right\}
\subseteq
\left\{
d_S(\widehat S_n,S(P))>
C_0L\sqrt{\frac{\log(C_0/\eta)}{n}}
\right\}.
\]
Applying \cref{obj:lem:uniform-summary-concentration} under \(Q_P^{(n)}\) yields
\[
Q_P^{(n)}
\left\{
W_1(\widehat\lambda_n,\nu_P)>
C\sqrt{\frac{\log(C/\eta)}{n}}
\right\}
\le\eta .
\] % lean: polynomial_lattice_law_estimator
\end{enumerate}
\end{proof}
