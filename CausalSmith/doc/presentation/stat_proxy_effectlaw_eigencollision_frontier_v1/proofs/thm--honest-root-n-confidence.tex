\begin{proof}[Proof of \cref{obj:thm:honest-root-n-confidence}]
\begin{enumerate}
\item By \cref{obj:lem:uniform-summary-concentration}, choose \(C_0\ge1\) such that, for every \(L\ge1\), \(n\ge1\), \(P\in\mathcal M\), and \(0<\alpha<1/2\),
\[
Q_P^{(n)}
\left\{
d_S(\widehat S_n,S(P))>
C_0L\sqrt{\frac{\log(C_0/\alpha)}{n}}
\right\}
\le \alpha .
\]
Fix \(L\ge1\), and put
\[
L_\tau=\frac{4L\sqrt{d_z}}{\sigma_0}.
\]
Let
\[
\mathcal K=\overline{\{S(P):P\in\mathcal M\}}
\]
be the summary closure from \cref{obj:def:summary-closure}.  By \cref{obj:thm:gap-free-positive-measure-modulus}, choose \(C_{\mathrm{mod}}>0\) and a continuous extension
\[
\overline F:\mathcal K\to \mathcal P_k([-L_\tau,L_\tau])
\]
such that
\[
W_1\{\overline F(\chi),\overline F(\chi')\}
\le C_{\mathrm{mod}}d_S(\chi,\chi')
\qquad(\chi,\chi'\in\mathcal K),
\]
and
\[
\overline F(S(P))=\nu_P
\qquad(P\in\mathcal M).
\]
By \cref{obj:prop:polynomial-net-law-estimator}, the prescribed structured-lattice constant \(C_{\mathrm{lat}}\) is positive.  For each \(n\ge1\), \cref{obj:prop:polynomial-net-law-estimator} supplies a measurable structured-lattice estimator \(\widehat\lambda_n\), generated by the finite exhaustive duplicate-free lexicographic list of well-formed lattice candidates and by the first minimizer of \(J_n(\vartheta;\widehat S_n)\), such that, for every sample \(\mathsf o\in O_{d_x,d_z}^{n}\) and every \(P\in\mathcal M\),
\[
W_1\bigl(\widehat\lambda_n(\mathsf o),\nu_P\bigr)
\le
C_{\mathrm{lat}}
\left\{
d_S(\widehat S_n(\mathsf o),S(P))+(\sqrt n)^{-1}
\right\}.
\]
The same cited result gives, for every sample,
\[
\operatorname{AtomFloor}\bigl(\pi_0,\widehat\lambda_n(\mathsf o)\bigr).
\]
For \(0<\alpha<1/2\), define
\[
r_{n,\alpha}
=
C_0L\sqrt{\frac{\log(C_0/\alpha)}{n}},
\qquad
R_{n,\alpha}
=
C_{\mathrm{lat}}\left(r_{n,\alpha}+(\sqrt n)^{-1}\right).
\]
The parameter domain gives \(r_{n,\alpha}>0\), and \(C_{\mathrm{lat}}>0\) gives \(R_{n,\alpha}>0\).  % lean: honest_root_n_confidence

\item Construct the repair selector \(\Pi\).  By \cref{obj:prop:summary-closure-compact}, \(\mathcal K\) is compact.  If \(\mathcal K=\varnothing\), set
\[
\Pi(s)=0
\qquad\text{for every ambient summary }s.
\]
If \(\mathcal K\ne\varnothing\), then \cref{obj:lem:summary-metric-continuity} and \cref{obj:lem:borel-nearest-point-selector} give a Borel map \(\Pi\) such that, for every ambient summary \(s\),
\[
\Pi(s)\in\mathcal K,
\qquad
d_S(s,\Pi(s))\le d_S(s,q)\quad(q\in\mathcal K).
\]
Together with \(\overline F\), this is the repair datum of \cref{obj:def:summary-space-repair}; the induced repaired summary-to-law map is measurable, and the displayed Lipschitz bound for \(\overline F\) is retained on \(\mathcal K\).  % lean: summaryRepair_with_modulus

\item Fix a sample \(\mathsf o\in O_{d_x,d_z}^{n}\).  If \(\mathcal K=\varnothing\), then \(\mathcal C_{n,\alpha}(\mathsf o)=\{\delta_0\}\), hence \(\mathcal C_{n,\alpha}(\mathsf o)\) is nonempty.  If \(\mathcal K\ne\varnothing\), then \(\Pi(\widehat S_n(\mathsf o))\in\mathcal K\), and
\[
d_S\{\Pi(\widehat S_n(\mathsf o)),\Pi(\widehat S_n(\mathsf o))\}
=0
\le 2r_{n,\alpha}.
\]
Thus
\[
\overline F\{\Pi(\widehat S_n(\mathsf o))\}\in\mathcal C_{n,\alpha}(\mathsf o),
\]
so \(\mathcal C_{n,\alpha}(\mathsf o)\) is nonempty.  The algorithmic set is nonempty because
\[
\operatorname{AtomFloor}\bigl(\pi_0,\widehat\lambda_n(\mathsf o)\bigr),
\qquad
W_1\{\widehat\lambda_n(\mathsf o),\widehat\lambda_n(\mathsf o)\}=0\le R_{n,\alpha}.
\]
By \cref{obj:def:wasserstein-confidence-set}, \(m_\star=\pi_0\) and membership in \(\mathcal C^{\mathrm{alg}}_{n,\alpha}\) is membership in the atom-floor subclass together with
\[
W_1(\nu,\widehat\lambda_n(\mathsf o))\le R_{n,\alpha}.
\]
By \cref{obj:def:wasserstein-distance}, this last inequality is equivalent to the existence of a finite nonnegative transport plan from \(\nu\) to \(\widehat\lambda_n(\mathsf o)\) whose absolute-distance cost is at most \(R_{n,\alpha}\).  This gives the asserted finite constrained transport representation.  % lean: confidenceSets_constrainedRepresentation

\item Fix \(P\in\mathcal M\).  Since \(S(P)\in\mathcal K\), the closure \(\mathcal K\) is nonempty.  Define the bad event
\[
\mathsf{Bad}_P
=
\left\{
\mathsf o\in O_{d_x,d_z}^{n}:
r_{n,\alpha}<d_S(\widehat S_n(\mathsf o),S(P))
\right\}.
\]
The concentration bound gives
\[
Q_P^{(n)}(\mathsf{Bad}_P)\le \alpha .
\]
The empirical summary is Borel by \cref{obj:def:empirical-summary}, and \((s,q)\mapsto d_S(s,q)\) is continuous by \cref{obj:lem:summary-metric-continuity}; hence \(\mathsf{Bad}_P\) is measurable.  Therefore
\[
Q_P^{(n)}(\mathsf{Bad}_P^{\,c})
=
1-Q_P^{(n)}(\mathsf{Bad}_P)
\ge 1-\alpha .
\]
For every \(\mathsf o\in\mathsf{Bad}_P^{\,c}\),
\[
d_S(\widehat S_n(\mathsf o),S(P))\le r_{n,\alpha}.
\]
Nearest-summary optimality gives
\[
d_S\{\widehat S_n(\mathsf o),\Pi(\widehat S_n(\mathsf o))\}
\le
d_S\{\widehat S_n(\mathsf o),S(P)\}
\le r_{n,\alpha},
\]
and the triangle inequality gives
\[
\begin{aligned}
d_S\{S(P),\Pi(\widehat S_n(\mathsf o))\}
&\le
d_S\{S(P),\widehat S_n(\mathsf o)\}
+
d_S\{\widehat S_n(\mathsf o),\Pi(\widehat S_n(\mathsf o))\}  \\
&\le 2r_{n,\alpha}.
\end{aligned}
\]
Since \(\overline F(S(P))=\nu_P\), the summary \(S(P)\) itself witnesses
\[
\nu_P\in\mathcal C_{n,\alpha}(\mathsf o).
\]
% lean: honest_root_n_confidence

\item On the same sample \(\mathsf o\in\mathsf{Bad}_P^{\,c}\), \cref{obj:prop:polynomial-net-law-estimator} gives
\[
W_1\{\widehat\lambda_n(\mathsf o),\nu_P\}
\le
C_{\mathrm{lat}}\{r_{n,\alpha}+(\sqrt n)^{-1}\}
=
R_{n,\alpha}.
\]
By symmetry of \(W_1\),
\[
W_1\{\nu_P,\widehat\lambda_n(\mathsf o)\}
\le R_{n,\alpha}.
\]
It remains to verify the atom floor for \(\nu_P\).  For each latent class \(u\), \cref{obj:ass:latent-arm-positivity} as included in \cref{obj:def:model-class} gives
\[
P(U=u,T=0)\ge\pi_0,
\qquad
P(U=u,T=1)\ge\pi_0.
\]
The two treatment cells partition the latent class, so
\[
p_u=P(U=u)=P(U=u,T=0)+P(U=u,T=1)\ge2\pi_0\ge\pi_0,
\]
where \(p_u\) is the latent mass from \cref{obj:def:latent-mass}.  For the raw latent law
\[
\sum_{u=1}^{k}p_u\delta_{\tau_u},
\]
the aggregate mass at any distinct support value \(\zeta\) is
\[
\sum_{u:\tau_u=\zeta}p_u\ge\pi_0,
\]
because the displayed sum contains at least one latent class.  Passing to the quotient law \(\nu_P\) of \cref{obj:def:quotient-law} only aggregates equal atom locations, so \cref{obj:def:atom-floor} gives
\[
\operatorname{AtomFloor}(\pi_0,\nu_P).
\]
Thus
\[
\nu_P\in\mathcal C^{\mathrm{alg}}_{n,\alpha}(\mathsf o).
\]
Combining the two memberships on \(\mathsf{Bad}_P^{\,c}\),
\[
1-\alpha
\le
Q_P^{(n)}
\left\{
\nu_P\in
\mathcal C_{n,\alpha}\cap\mathcal C^{\mathrm{alg}}_{n,\alpha}
\right\}.
\]
% lean: quotientLaw_atomFloor

\item Fix a sample \(\mathsf o\).  For the theoretical diameter, take any
\[
\xi_1,\xi_2\in\mathcal C_{n,\alpha}(\mathsf o).
\]
If \(\mathcal K=\varnothing\), then both elements equal the zero-law fallback, so
\[
W_1(\xi_1,\xi_2)=0\le4C_{\mathrm{mod}}r_{n,\alpha}.
\]
If \(\mathcal K\ne\varnothing\), then by \cref{obj:def:wasserstein-confidence-set} there are \(\chi_1,\chi_2\in\mathcal K\) such that
\[
\xi_j=\overline F(\chi_j),
\qquad
d_S\{\chi_j,\Pi(\widehat S_n(\mathsf o))\}\le2r_{n,\alpha}
\quad(j=1,2).
\]
By symmetry and the triangle inequality,
\[
\begin{aligned}
d_S(\chi_1,\chi_2)
&\le
d_S\{\chi_1,\Pi(\widehat S_n(\mathsf o))\}
+
d_S\{\Pi(\widehat S_n(\mathsf o)),\chi_2\}  \\
&\le4r_{n,\alpha}.
\end{aligned}
\]
The Lipschitz bound for \(\overline F\) gives
\[
W_1(\xi_1,\xi_2)
\le
C_{\mathrm{mod}}d_S(\chi_1,\chi_2)
\le
4C_{\mathrm{mod}}r_{n,\alpha}.
\]
Taking the supremum over all pairs in the nonempty set \(\mathcal C_{n,\alpha}(\mathsf o)\) yields
\[
\operatorname{diam}_{W_1}(\mathcal C_{n,\alpha})
\le
4C_{\mathrm{mod}}r_{n,\alpha}.
\]
% lean: honest_root_n_confidence

\item For the algorithmic diameter, take any
\[
\xi_1,\xi_2\in\mathcal C^{\mathrm{alg}}_{n,\alpha}(\mathsf o).
\]
By \cref{obj:def:wasserstein-confidence-set},
\[
W_1\{\xi_1,\widehat\lambda_n(\mathsf o)\}\le R_{n,\alpha},
\qquad
W_1\{\xi_2,\widehat\lambda_n(\mathsf o)\}\le R_{n,\alpha}.
\]
By symmetry and the triangle inequality,
\[
\begin{aligned}
W_1(\xi_1,\xi_2)
&\le
W_1\{\xi_1,\widehat\lambda_n(\mathsf o)\}
+
W_1\{\widehat\lambda_n(\mathsf o),\xi_2\}  \\
&\le2R_{n,\alpha}.
\end{aligned}
\]
Taking the supremum over all pairs in the nonempty set \(\mathcal C^{\mathrm{alg}}_{n,\alpha}(\mathsf o)\) gives
\[
\operatorname{diam}_{W_1}(\mathcal C^{\mathrm{alg}}_{n,\alpha})
\le
2R_{n,\alpha}.
\]
% lean: honest_root_n_confidence
\end{enumerate}
\end{proof}
