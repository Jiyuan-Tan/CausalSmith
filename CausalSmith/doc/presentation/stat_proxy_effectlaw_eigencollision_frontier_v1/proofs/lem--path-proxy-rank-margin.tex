\begin{proof}[Proof of \cref{obj:lem:path-proxy-rank-margin}]
\begin{enumerate}
\item \Cref{obj:synth_14,obj:def:reference-feature,obj:def:target-feature} give the feature matrices of \(P_{g,h}\). Under \(|h|\le 1/100\), the denominators below are positive, and
\[
A_0(h)=
\begin{pmatrix}
1&1\\[0.2em]
\dfrac{35h+9}{10(5h+3)}&\dfrac{7}{10}
\end{pmatrix},
\qquad
A_1(h)=
\begin{pmatrix}
1&1\\[0.2em]
\dfrac{225h+28}{20(15h+4)}&\dfrac{3}{4}
\end{pmatrix},
\]
while
\[
B(h)=
\begin{pmatrix}
1&1\\[0.2em]
\dfrac15&\dfrac{12/25-h/5}{3/5-h}
\end{pmatrix}.
\]
Thus the matrices \(A_0,A_1,B\) in \cref{obj:ass:proxy-rank-margin}, formed from \(P_{g,h}\), are \(A_0(h),A_1(h),B(h)\), respectively. % lean: path_referenceFeature, path_targetFeature

\item The undisplaced reference features have singular-value slack \(1/5\). At \(h=0\),
\[
A_0(0)=
\begin{pmatrix}1&1\\[0.2em]3/10&7/10\end{pmatrix},
\qquad
A_1(0)=
\begin{pmatrix}1&1\\[0.2em]7/20&3/4\end{pmatrix}.
\]
For every \(\upsilon=(\upsilon_1,\upsilon_2)^\top\in\mathbb R^2\),
\[
100\left(\|A_0(0)\upsilon\|_2^2-\frac1{25}\|\upsilon\|_2^2\right)
=
105\upsilon_1^2+242\upsilon_1\upsilon_2+145\upsilon_2^2
=
\frac{146(3\upsilon_1-7\upsilon_2)^2+(549\upsilon_1+641\upsilon_2)^2}{2883}\ge0,
\]
and
\[
400\left(\|A_1(0)\upsilon\|_2^2-\frac1{25}\|\upsilon\|_2^2\right)
=
433\upsilon_1^2+1010\upsilon_1\upsilon_2+609\upsilon_2^2
=
\frac{2168(7\upsilon_1-15\upsilon_2)^2+(5015\upsilon_1+5919\upsilon_2)^2}{58329}\ge0.
\]
The variational characterization of the least singular value therefore gives
\[
\sigma_{\min}(A_0(0))\ge\frac15,
\qquad
\sigma_{\min}(A_1(0))\ge\frac15 .
\] % lean: baseReferenceFeature_strictRank, signalMinSingular_lower_fin_two

\item Define the reference-feature entry displacements
\[
\chi_0(h):=\frac{35h+9}{10(5h+3)}-\frac3{10},
\qquad
\chi_1(h):=\frac{225h+28}{20(15h+4)}-\frac7{20}.
\]
Then
\[
A_0(h)=A_0(0)+\begin{pmatrix}0&0\\ \chi_0(h)&0\end{pmatrix},
\qquad
A_1(h)=A_1(0)+\begin{pmatrix}0&0\\ \chi_1(h)&0\end{pmatrix}.
\]
Since \(|h|\le1/100\),
\[
10(5h+3)\ge \frac{59}{2},
\qquad
20(15h+4)\ge 77,
\]
and the algebraic identities
\[
\chi_0(h)=\frac{20h}{10(5h+3)},
\qquad
\chi_1(h)=\frac{120h}{20(15h+4)}
\]
give
\[
|\chi_0(h)|\le \frac{20/100}{59/2}=\frac{2}{295}\le\frac1{50},
\qquad
|\chi_1(h)|\le \frac{120/100}{77}=\frac{6}{385}\le\frac1{50}.
\]
For any real \(\chi\),
\[
\left\|
\begin{pmatrix}0&0\\ \chi&0\end{pmatrix}
\right\|_{\mathrm{op}}\le |\chi|,
\]
because its action sends \(\upsilon\) to \((0,\chi\upsilon_1)^\top\). Weyl's singular-value perturbation inequality therefore yields, for \(t=0,1\),
\[
\sigma_{\min}(A_t(h))
\ge
\sigma_{\min}(A_t(0))-\frac1{50}
\ge
\frac15-\frac1{50}
\ge
\frac1{10}.
\] % lean: pathReferenceFeature_entry_displacement, path_lowerLeft_opNorm, singular_value_weyl, pathReferenceFeature_rankMargin

\item The target feature is handled in the same perturbative form, starting from
\[
B(0)=
\begin{pmatrix}1&1\\[0.2em]1/5&4/5\end{pmatrix}.
\]
For every \(\upsilon\in\mathbb R^2\),
\[
25\left(\|B(0)\upsilon\|_2^2-\frac1{25}\|\upsilon\|_2^2\right)
=
25\upsilon_1^2+58\upsilon_1\upsilon_2+40\upsilon_2^2
=
\frac{159(3\upsilon_1-4\upsilon_2)^2+(187\upsilon_1+236\upsilon_2)^2}{1456}\ge0,
\]
so
\[
\sigma_{\min}(B(0))\ge\frac15.
\]
With
\[
\chi_B(h):=\frac{12/25-h/5}{3/5-h}-\frac45,
\]
we have
\[
B(h)=B(0)+\begin{pmatrix}0&0\\0&\chi_B(h)\end{pmatrix},
\qquad
\chi_B(h)=\frac{(3/5)h}{3/5-h}.
\]
The bound \(|h|\le1/100\) gives \(3/5-h\ge59/100\), hence
\[
|\chi_B(h)|
\le
\frac{(3/5)(1/100)}{59/100}
=
\frac{3}{295}
\le
\frac1{50}.
\]
Also
\[
\left\|
\begin{pmatrix}0&0\\0&\chi\end{pmatrix}
\right\|_{\mathrm{op}}\le|\chi|
\]
for any real \(\chi\), since its action sends \(\upsilon\) to \((0,\chi\upsilon_2)^\top\). Another application of Weyl's inequality gives
\[
\sigma_{\min}(B(h))
\ge
\sigma_{\min}(B(0))-\frac1{50}
\ge
\frac15-\frac1{50}
\ge
\frac1{10}.
\] % lean: pathTargetFeature_zero_strictRank, signalMinSingular_lower_fin_two, pathTargetFeature_entry_displacement, path_lowerRight_opNorm, singular_value_weyl, pathTargetFeature_rankMargin

\item Combining the three estimates,
\[
\min\{\sigma_{\min}(A_0),\sigma_{\min}(A_1),\sigma_{\min}(B)\}
=
\min\{\sigma_{\min}(A_0(h)),\sigma_{\min}(A_1(h)),\sigma_{\min}(B(h))\}
\ge \frac1{10}.
\]
This is the proxy-rank margin of \cref{obj:ass:proxy-rank-margin} with \(\sigma_0=1/10\). % lean: path_proxyRankMargin
\end{enumerate}
\end{proof}
