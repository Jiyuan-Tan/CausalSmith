\begin{proof}[Proof of \cref{obj:lem:witness-latent-mass}]
Fix \(\varepsilon\) with \(0\le \varepsilon\le 1/8\).  For a Bernoulli success
parameter \(\theta\in\mathbb R\) and \(b\in\{0,1\}\), put
\[
\beta_\theta(b):=
\begin{cases}
1-\theta,& b=0,\\
\theta,& b=1.
\end{cases}
\]
For the two witness classes in the paper labels, define
\[
\mathfrak p_{\mathrm U}(1)=\frac25,\qquad
\mathfrak p_{\mathrm U}(2)=\frac35,
\qquad
\mathfrak p_{\mathrm T}(1)=\frac25,\qquad
\mathfrak p_{\mathrm T}(2)=\frac35,
\]
\[
\mathfrak p_{\mathrm X}(1)=\frac15,\qquad
\mathfrak p_{\mathrm X}(2)=\frac45,
\]
\[
\mathfrak p_{\mathrm Z}(0,1)=\frac3{10},\quad
\mathfrak p_{\mathrm Z}(0,2)=\frac7{10},\quad
\mathfrak p_{\mathrm Z}(1,1)=\frac7{20},\quad
\mathfrak p_{\mathrm Z}(1,2)=\frac34,
\]
and
\[
\mathfrak p_{\mathrm Y_1,\varepsilon}(1)=\frac12-\varepsilon,
\qquad
\mathfrak p_{\mathrm Y_1,\varepsilon}(2)=\frac12+\varepsilon .
\]
Thus \cref{obj:def:two-class-witness} assigns to the atom with
\[
(U,T,X_2,Z_2,Y(0),Y(1))
=
(v,t_{\mathrm b},x_{\mathrm b},z_{\mathrm b},
 y_{\mathrm b,0},y_{\mathrm b,1})
\]
the elementary mass
\[
\mathsf{wt}_\varepsilon
(v,t_{\mathrm b},x_{\mathrm b},z_{\mathrm b},y_{\mathrm b,0},y_{\mathrm b,1})
=
\mathfrak p_{\mathrm U}(v)\,
\beta_{\mathfrak p_{\mathrm T}(v)}(t_{\mathrm b})\,
\beta_{\mathfrak p_{\mathrm X}(v)}(x_{\mathrm b})\,
\beta_{\mathfrak p_{\mathrm Z}(t_{\mathrm b},v)}(z_{\mathrm b})\,
\beta_{1/4}(y_{\mathrm b,0})\,
\beta_{\mathfrak p_{\mathrm Y_1,\varepsilon}(v)}(y_{\mathrm b,1}).
\]
The bounds on \(\varepsilon\) place every Bernoulli parameter above in
\([0,1]\), so these elementary masses are nonnegative.  By
\cref{obj:def:latent-mass} and the finite witness-law expansion, for
\(u\in\{1,2\}\),
\[
p_u(P_{\varepsilon}^{\mathrm{wit}})
=
\sum_{v=1}^{2}\sum_{t_{\mathrm b},x_{\mathrm b},z_{\mathrm b},
y_{\mathrm b,0},y_{\mathrm b,1}\in\{0,1\}}
\mathsf{wt}_\varepsilon
(v,t_{\mathrm b},x_{\mathrm b},z_{\mathrm b},y_{\mathrm b,0},y_{\mathrm b,1})
\,\mathbf 1\{v=u\}.
\] % lean: latentMass, witnessLaw_real, measurableSet_witness_latentClass, witnessWeight_nonneg

The indicator restricts the finite sum to the \(v=u\) slice, hence
\[
p_u(P_{\varepsilon}^{\mathrm{wit}})
=
\sum_{t_{\mathrm b},x_{\mathrm b},z_{\mathrm b},
y_{\mathrm b,0},y_{\mathrm b,1}\in\{0,1\}}
\mathsf{wt}_\varepsilon
(u,t_{\mathrm b},x_{\mathrm b},z_{\mathrm b},y_{\mathrm b,0},y_{\mathrm b,1}).
\] % lean: witness_sum_restrict_latentClass

For fixed \(u\) and \(t_{\mathrm b}\), summing over the nuisance Bernoulli
coordinates gives
\[
\begin{aligned}
&\sum_{x_{\mathrm b},z_{\mathrm b},y_{\mathrm b,0},y_{\mathrm b,1}\in\{0,1\}}
\mathsf{wt}_\varepsilon
(u,t_{\mathrm b},x_{\mathrm b},z_{\mathrm b},y_{\mathrm b,0},y_{\mathrm b,1})
\\
&\quad =
\mathfrak p_{\mathrm U}(u)\,
\beta_{\mathfrak p_{\mathrm T}(u)}(t_{\mathrm b})
\left(\sum_{x_{\mathrm b}}\beta_{\mathfrak p_{\mathrm X}(u)}(x_{\mathrm b})\right)
\left(\sum_{z_{\mathrm b}}\beta_{\mathfrak p_{\mathrm Z}(t_{\mathrm b},u)}(z_{\mathrm b})\right)
\left(\sum_{y_{\mathrm b,0}}\beta_{1/4}(y_{\mathrm b,0})\right)
\left(\sum_{y_{\mathrm b,1}}\beta_{\mathfrak p_{\mathrm Y_1,\varepsilon}(u)}(y_{\mathrm b,1})\right)
\\
&\quad =
\mathfrak p_{\mathrm U}(u)\,
\beta_{\mathfrak p_{\mathrm T}(u)}(t_{\mathrm b}),
\end{aligned}
\]
because \(\sum_{b\in\{0,1\}}\beta_\theta(b)=(1-\theta)+\theta=1\). % lean: witnessWeight_sum_nuisance, sum_bernoulliMass

Therefore
\[
p_u(P_{\varepsilon}^{\mathrm{wit}})
=
\sum_{t_{\mathrm b}\in\{0,1\}}
\mathfrak p_{\mathrm U}(u)\,
\beta_{\mathfrak p_{\mathrm T}(u)}(t_{\mathrm b})
=
\mathfrak p_{\mathrm U}(u)
\left\{(1-\mathfrak p_{\mathrm T}(u))+\mathfrak p_{\mathrm T}(u)\right\}
=
\mathfrak p_{\mathrm U}(u).
\]
Evaluating this identity at the two paper-labeled classes yields
\[
p_1(P_{\varepsilon}^{\mathrm{wit}})=\mathfrak p_{\mathrm U}(1)=\frac25,
\qquad
p_2(P_{\varepsilon}^{\mathrm{wit}})=\mathfrak p_{\mathrm U}(2)=\frac35.
\] % lean: witness_latentMass

The displayed values are independent of \(\varepsilon\), and they are distinct since
\[
\frac25\ne \frac35 .
\] % lean: witness_latentMass
\end{proof}
