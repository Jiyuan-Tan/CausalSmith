\begin{proof}[Proof of \cref{obj:thm:published-vmw-converse-transfer}]
\begin{enumerate}
\item Apply \cref{obj:thm:matching-local-lower-bounds} with
\[
c_{\mathrm{loc}}=\frac14,
\]
which lies in the domain of \cref{obj:def:local-kl-radius}.  This gives constants \(a,c\in\mathbb R\), and an auxiliary positive Kullback--Leibler constant, such that
\[
0<a\le \frac18,
\qquad
0<c,
\]
and all conclusions of \cref{obj:thm:matching-local-lower-bounds} hold for every \(n\ge1\).  In the present specialization,
\[
L_\tau
=\frac{4L\sqrt{d_z}}{\sigma_0}
=\frac{4\cdot2\sqrt2}{1/10}.
\]

We record the published-scope implication used for the endpoint laws.  Let \(P\in\mathcal M\) for
\[
k=d_x=d_z=2,\qquad L=2,\qquad \pi_0=\sigma_0=\frac1{10}.
\]
The restrictions in \cref{obj:def:model-class} supply the qualitative proxy separations, causal consistency, armwise latent ignorability, anchor normalization, boundedness, latent-arm positivity, and proxy-rank margin.  The proxy-rank margin gives
\[
\sigma_2(A_0)\ge \frac1{10},
\qquad
\sigma_2(A_1)\ge \frac1{10},
\qquad
\sigma_2(B)\ge \frac1{10},
\]
so the three proxy feature matrices have full column rank.  Latent-arm positivity gives
\[
P(U=u,T=t)\ge \frac1{10}>0,
\qquad
u\in\{1,2\},\quad t\in\{0,1\}.
\]
Thus \(P\) satisfies the qualitative structural conditions in \cref{obj:synth_11}.  The cited VMW model-scope correspondence, \citep[Assumptions 1--2 and Section 4.2]{VirkMazaheriWu2026}, identifies these qualitative conditions with membership in the published VMW model predicate attached to \cref{obj:def:published-vmw-regimes}.

It remains to verify VMW Assumption 4 in \cref{obj:synth_18}.  Put
\[
M_{\mathrm{stack}}:=\operatorname{stack}(M_0(P),M_1(P)).
\]
Choose a singular-value decomposition
\[
M_{\mathrm{stack}}=\sum_j s_j\ell_j r_j^\top,
\qquad
s_1\ge s_2\ge\cdots .
\]
By \cref{obj:prop:observed-vmw-margin-inclusion},
\[
s_2(M_{\mathrm{stack}})=\sigma_2(M_{\mathrm{stack}})
\ge \pi_0\sigma_0^2=\frac1{1000}>0.
\]
Let \(V=(r_1,r_2)\).  Then the columns of \(V\) are orthonormal and satisfy
\[
M_{\mathrm{stack}}^\top M_{\mathrm{stack}}r_j=s_j^2r_j,
\qquad j=1,2,
\]
so \(V\) is a population top-two right singular basis.  Since \(s_j>0\),
\[
r_j=s_j^{-1}M_{\mathrm{stack}}^\top\ell_j,
\qquad j=1,2.
\]
For \(y=(y_0,y_1)\in\mathbb R^{d_z}\oplus\mathbb R^{d_z}\),
\[
M_{\mathrm{stack}}^\top y=M_0(P)^\top y_0+M_1(P)^\top y_1.
\]
Hence each \(r_j\) lies in the signal space generated by the two armwise row spaces.  The factorization in \cref{obj:prop:observed-vmw-margin-inclusion} gives rank at most \(2\) for this signal space, while \(s_2(M_{\mathrm{stack}})>0\) gives rank at least \(2\).  Therefore the span of \(V\) is exactly the signal space.  Applying the compressed-margin conclusion of \cref{obj:prop:observed-vmw-margin-inclusion} to this basis gives
\[
\sigma_2\{M_t(P)V\}\ge \pi_0\sigma_0^2=\frac1{1000},
\qquad t\in\{0,1\}.
\]
Together with
\[
\|X\|_2\le2,\qquad
\|ZX^\top\|_{\mathrm{op}}\le2,\qquad
\|YZX^\top\|_{\mathrm{op}}\le2,
\qquad
P(T=t)\ge2\pi_0=\frac15,
\]
these are the finite envelope, treatment-arm, top-right-basis, stacked-margin, and compressed-margin clauses of \cref{obj:synth_18}, with
\[
L_X=L_{ZX}=L_{YZX}=2,
\qquad
\pi=\frac15,
\qquad
\sigma=\frac1{1000}.
\]
Thus every such \(P\in\mathcal M\) satisfies VMW Assumption 4. % lean: matching_local_lower_bounds, ucvmwModel_publishedQualitativeConditions, publishedTopRightSignalBasis_exists, concreteVMWAssumption4_of_ucvmwModel

\item Fix \(n\ge1\).  The quotient-law certificate in \cref{obj:thm:matching-local-lower-bounds}, together with \cref{obj:def:local-quotient-experiment}, supplies probability laws
\[
P_0^{\mathrm{wit}},
\qquad
P_{a/\sqrt n}^{\mathrm{wit}},
\]
belonging to \(\mathcal M\) with the fixed parameters above.  By the implication from the preceding step, both endpoint laws belong to the published VMW model and satisfy VMW Assumption 4.

At the collision endpoint, \cref{obj:lem:witness-latent-effect} gives
\[
\tau_1(P_0^{\mathrm{wit}})=\frac14,
\qquad
\tau_2(P_0^{\mathrm{wit}})=\frac14.
\]
The cited recovery-regime correspondence, \citep[Assumption 3, lines 261--263; Assumption 4, lines 321--337; and Theorem 7.2, lines 350--381]{VirkMazaheriWu2026}, identifies the separated recovery predicate attached to \cref{obj:def:published-vmw-regimes}, for \(k=d_x=d_z=2\), with the conjunction in \cref{obj:synth_13}, including spectral separation of the two latent effects.  The displayed equality violates that spectral-separation clause, so \(P_0^{\mathrm{wit}}\) lies outside the published separated recovery regime. % lean: published_vmw_converse_transfer

\item Let \(\mathcal V_n^\nu\) satisfy the quotient-class hypotheses, and choose
\[
P_{\nu,0}^{\mathcal V},P_{\nu,1}^{\mathcal V}\in\mathcal V_n^\nu
\]
with
\[
P_{\nu,0}^{\mathcal V}=P_0^{\mathrm{wit}},
\qquad
P_{\nu,1}^{\mathcal V}=P_{a/\sqrt n}^{\mathrm{wit}}.
\]
For any estimator \(\widetilde\nu_n\) with support radius \(L_\tau\), the quotient-risk conclusion of \cref{obj:thm:matching-local-lower-bounds} gives an endpoint
\[
P_{\nu,\star}\in
\left\{P_0^{\mathrm{wit}},P_{a/\sqrt n}^{\mathrm{wit}}\right\}
\]
such that
\[
\frac{c}{\sqrt n}
\le
E_{Q_{P_{\nu,\star}}^{(n)}}\!\left[
W_1\!\left(\widetilde\nu_n,\nu_{P_{\nu,\star}}\right)
\right].
\]
If \(P_{\nu,\star}=P_0^{\mathrm{wit}}\), use \(P_{\nu,0}^{\mathcal V}\); if
\(P_{\nu,\star}=P_{a/\sqrt n}^{\mathrm{wit}}\), use \(P_{\nu,1}^{\mathcal V}\).  Equality as full-data laws identifies the observed product law and the quotient law, hence
\[
\sup_{P\in\mathcal V_n^\nu}
E_{Q_P^{(n)}}\!\left[
W_1\!\left(\widetilde\nu_n,\nu_P\right)
\right]
\ge
\frac{c}{\sqrt n}.
\]
This is the quotient lower bound. % lean: published_vmw_converse_transfer

\item Fix \(g\) in the gap-scale domain, so
\[
0<g\le\frac14,
\]
and set
\[
h(n,g):=
a\min\left\{1,\frac{1}{\sqrt n\,g}\right\}.
\]
The labeled-path certificate in \cref{obj:thm:matching-local-lower-bounds} supplies the tangent-amplitude condition
\[
h(n,g)\in[-1,1],
\]
and places
\[
P_{g,0},
\qquad
P_{g,h(n,g)}
\]
in the local labeled-weight experiment \(\mathcal L_{n,g}^p\).  By \cref{obj:def:local-weight-experiment}, both laws belong to the gap stratum \(\mathcal M(g)\), and therefore to \(\mathcal M\) with the fixed parameters.  The implication from the first step gives published VMW model membership and VMW Assumption 4 for both endpoint laws.

Let \(P_{\mathrm{end}}\) be either \(P_{g,0}\) or \(P_{g,h(n,g)}\).  Since \(P_{\mathrm{end}}\in\mathcal M(g)\), \cref{obj:def:gap-stratum} supplies the distinct-effects condition of \cref{obj:ass:distinct-effects}.  Latent-arm positivity gives
\[
P_{\mathrm{end}}(U=u)
\ge
P_{\mathrm{end}}(U=u,T=0)
\ge \frac1{10}>0,
\qquad u\in\{1,2\}.
\]
Thus both latent classes have positive mass, and the distinct-effects condition yields
\[
\tau_1(P_{\mathrm{end}})\ne \tau_2(P_{\mathrm{end}}).
\]
This is the two-class spectral-separation condition in \cref{obj:synth_13}. % lean: gapStratum_publishedSpectralSeparation, published_vmw_converse_transfer

\item Let \(\mathcal V_{n,g}^p\) satisfy the labeled-weight comparator-class hypotheses, and choose
\[
P_{p,0}^{\mathcal V},P_{p,1}^{\mathcal V}\in\mathcal V_{n,g}^p
\]
with
\[
P_{p,0}^{\mathcal V}=P_{g,0},
\qquad
P_{p,1}^{\mathcal V}=P_{g,h(n,g)}.
\]
For any ordered-weight estimator \(\widetilde p_n\), the ordered-weight conclusion of \cref{obj:thm:matching-local-lower-bounds} gives an endpoint
\[
P_{p,\star}\in
\left\{P_{g,0},P_{g,h(n,g)}\right\}
\]
such that
\[
c\min\left\{1,\frac{1}{\sqrt n\,g}\right\}
\le
E_{Q_{P_{p,\star}}^{(n)}}\!\left[
\left\|\widetilde p_n-p^{\uparrow}(P_{p,\star})\right\|_1
\right].
\]
If \(P_{p,\star}=P_{g,0}\), use \(P_{p,0}^{\mathcal V}\); if
\(P_{p,\star}=P_{g,h(n,g)}\), use \(P_{p,1}^{\mathcal V}\).  Equality as full-data laws preserves the observed product law and the ordered-mass target, hence
\[
\sup_{P\in\mathcal V_{n,g}^p}
E_{Q_P^{(n)}}\!\left[
\left\|\widetilde p_n-p^{\uparrow}(P)\right\|_1
\right]
\ge
c\min\left\{1,\frac{1}{\sqrt n\,g}\right\}.
\]
This proves the labeled-weight lower bound. % lean: published_vmw_converse_transfer
\end{enumerate}
\end{proof}
