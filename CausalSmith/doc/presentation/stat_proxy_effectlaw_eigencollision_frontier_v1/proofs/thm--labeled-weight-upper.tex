\begin{proof}[Proof of \cref{obj:thm:labeled-weight-upper}]
\begin{enumerate}
\item
Let \(C_{\mathrm{quo}}>0\) be the constant supplied by \cref{obj:thm:collision-uniform-root-n}.  Define
\[
\chi_{\mathrm{tail}}:=\max\{C_{\mathrm{quo}},2\},\qquad
\beta_{\mathrm{mean}}
:=\chi_{\mathrm{tail}}\sqrt{\log(2\chi_{\mathrm{tail}})}
+\frac{\chi_{\mathrm{tail}}^2\sqrt{\pi}}{2},
\]
\[
\kappa_{\mathrm{ord}}:=8+\frac{16}{\pi_0},
\qquad
C:=\max\{2,\kappa_{\mathrm{ord}}\beta_{\mathrm{mean}}\}.
\]
The hypotheses give \(\chi_{\mathrm{tail}}\ge2\), \(\log(2\chi_{\mathrm{tail}})>0\), \(\beta_{\mathrm{mean}}>0\), and \(\kappa_{\mathrm{ord}}>0\), hence \(C>0\).  Fix \(n\ge1\).  By \cref{obj:thm:collision-uniform-root-n}, choose the measurable nearest-summary quotient-law estimator \(\widehat\nu_n\) from \cref{obj:def:summary-space-repair}.  Set
\[
\widehat p_n^{\uparrow}(\omega):=
p^{\uparrow}\!\left(\widehat\nu_n(\omega)\right),
\qquad \omega\in O_{d_x,d_z}^n ,
\]
which is exactly the ordered-weight estimator of \cref{obj:def:labeled-weight-estimator}.  Its measurability follows from the measurability of \(\widehat\nu_n\) and the Borel ordered-mass map; its values lie in the probability simplex by the ordered-mass construction, including its barycentric fallback branch. % lean: labeled_weight_upper

\item
Fix \(g\) with \(0<g\le1/4\), and fix \(P\in\mathcal M(g)\).  Write
\[
Z_P(\omega):=
W_1\!\left(\widehat\nu_n(\omega),\nu_P\right),
\qquad \omega\in O_{d_x,d_z}^n ,
\]
where \(\nu_P\) is the quotient latent-effect law of \cref{obj:def:quotient-law}.  The support radius \(L_\tau=4L\sqrt{d_z}/\sigma_0\) gives
\[
0\le Z_P(\omega)\le 2L_\tau
\]
for every \(\omega\), so \(Z_P\) is integrable under \(Q_P^{(n)}\).  For every \(0<\eta<1/2\), the tail bound in \cref{obj:thm:collision-uniform-root-n} and the inequality \(C_{\mathrm{quo}}\le\chi_{\mathrm{tail}}\) give
\[
Q_P^{(n)}\!\left\{
\chi_{\mathrm{tail}}\sqrt{\frac{\log(\chi_{\mathrm{tail}}/\eta)}{n}}
< Z_P
\right\}
\le \eta .
\]
Indeed, the displayed event is contained in the corresponding collision-uniform event because
\[
C_{\mathrm{quo}}\sqrt{\frac{\log(C_{\mathrm{quo}}/\eta)}{n}}
\le
\chi_{\mathrm{tail}}\sqrt{\frac{\log(\chi_{\mathrm{tail}}/\eta)}{n}} .
\]
To integrate this tail, put
\[
b_{\mathrm{mean}}:=
\chi_{\mathrm{tail}}\sqrt{\frac{\log(2\chi_{\mathrm{tail}})}{n}},
\qquad
\varrho_{\mathrm{mean}}:=\frac{n}{\chi_{\mathrm{tail}}^2}.
\]
For \(x>b_{\mathrm{mean}}\), define
\[
\eta_x:=\chi_{\mathrm{tail}}\exp(-\varrho_{\mathrm{mean}}x^2).
\]
Then \(0<\eta_x<1/2\) and
\[
\chi_{\mathrm{tail}}\sqrt{\frac{\log(\chi_{\mathrm{tail}}/\eta_x)}{n}}=x.
\]
Thus
\[
Q_P^{(n)}\{Z_P>x\}\le
\chi_{\mathrm{tail}}\exp(-\varrho_{\mathrm{mean}}x^2),
\qquad x>b_{\mathrm{mean}}.
\]
The layer-cake identity and the Gaussian tail bound yield
\[
E_{Q_P^{(n)}}[Z_P]
\le
b_{\mathrm{mean}}
+\chi_{\mathrm{tail}}\int_{b_{\mathrm{mean}}}^{\infty}
\exp(-\varrho_{\mathrm{mean}}x^2)\,dx
\le
\frac{\beta_{\mathrm{mean}}}{\sqrt n}.
\] % lean: integral_le_of_sqrt_log_tail

\item
Let
\[
\nu_P^{\mathrm{slot}}
:=
\bigl((p_u(P))_{u=1}^k,(\tau_u(P))_{u=1}^k\bigr)
\]
be the \(k\)-slot representative of \(\nu_P\).  By \cref{obj:lem:ordered-masses-measure-invariance},
\[
p^{\uparrow}(P)=p^{\uparrow}\!\left(\nu_P^{\mathrm{slot}}\right),
\]
because \(\nu_P^{\mathrm{slot}}\) and \(\nu_P\) represent the same aggregated atomic law.  The gap-stratum assumptions give, for every latent class \(u\),
\[
\pi_0\le P(U=u,T=0)\le P(U=u)=p_u(P),
\]
and \cref{obj:def:gap-stratum} gives distinct latent effects and
\[
\frac g2\le |\tau_u(P)-\tau_v(P)|\qquad (u\ne v).
\]
Consequently, if \(\xi=\sum_{a=1}^k \widetilde p_a\delta_{\widetilde\tau_a}\) is any valid \(k\)-slot atomic law and
\[
W_1\!\left(\nu_P^{\mathrm{slot}},\xi\right)
<
\frac{\pi_0(g/2)}{4},
\]
then
\[
\sum_{j=1}^k
\left|p_j^{\uparrow}(\xi)-p_j^{\uparrow}(\nu_P^{\mathrm{slot}})\right|
\le
\frac{8}{g}\,
W_1\!\left(\nu_P^{\mathrm{slot}},\xi\right).
\]
For this last implication, take an optimal transport plan \(\gamma\).  Each true atom \(\tau_u(P)\) must have a positive-mass atom \(\widetilde\tau_{\varphi(u)}\) of \(\xi\) within distance \(g/8\); otherwise the mass \(p_u(P)\ge\pi_0\) transported from that atom alone would cost at least \(\pi_0(g/2)/4\).  The map \(\varphi\) is injective, hence bijective, because two \(g/8\)-close images of distinct \(g/2\)-separated true atoms would violate the triangle inequality.  It also preserves the increasing order.  Therefore
\[
\sum_{j=1}^k
\left|p_j^{\uparrow}(\xi)-p_j^{\uparrow}(\nu_P^{\mathrm{slot}})\right|
=
\sum_{u=1}^k
\left|\widetilde p_{\varphi(u)}-p_u(P)\right|.
\]
With
\[
\operatorname{diag}_{\gamma}:=\sum_{u=1}^k\gamma_{u,\varphi(u)},
\qquad
\operatorname{off}_{\gamma}:=1-\operatorname{diag}_{\gamma},
\]
the row and column marginal identities give
\[
\sum_{u=1}^k
\left|\widetilde p_{\varphi(u)}-p_u(P)\right|
\le 2\,\operatorname{off}_{\gamma}.
\]
All off-matching mass moves at least \(g/4\), so
\[
\frac g4\,\operatorname{off}_{\gamma}
\le W_1\!\left(\nu_P^{\mathrm{slot}},\xi\right),
\]
and the displayed local stability bound follows. % lean: gapStratum_orderedMasses_l1_le

\item
Apply the preceding bound to the representative \(\xi_\omega\) of \(\widehat\nu_n(\omega)\).  Since \(\xi_\omega\) and \(\widehat\nu_n(\omega)\) represent the same law,
\[
W_1\!\left(\nu_P^{\mathrm{slot}},\xi_\omega\right)=Z_P(\omega).
\]
If
\[
Z_P(\omega)<\frac{\pi_0(g/2)}4,
\]
then
\[
\sum_{j=1}^k
\left|\widehat p_{n,j}^{\uparrow}(\omega)-p_j^{\uparrow}(P)\right|
\le
\frac{8 Z_P(\omega)}{g}
\le
\frac{\kappa_{\mathrm{ord}} Z_P(\omega)}{g}.
\]
If instead \(Z_P(\omega)\ge \pi_0(g/2)/4\), the simplex diameter gives
\[
\sum_{j=1}^k
\left|\widehat p_{n,j}^{\uparrow}(\omega)-p_j^{\uparrow}(P)\right|
\le 2
\le
\frac{16}{\pi_0}\frac{Z_P(\omega)}{g}
\le
\frac{\kappa_{\mathrm{ord}} Z_P(\omega)}{g}.
\]
Thus, for every sample \(\omega\),
\[
\sum_{j=1}^k
\left|\widehat p_{n,j}^{\uparrow}(\omega)-p_j^{\uparrow}(P)\right|
\le
\frac{\kappa_{\mathrm{ord}} Z_P(\omega)}{g}.
\] % lean: labeled_weight_upper

\item
Integrating the samplewise inequality and using the mean bound for \(Z_P\),
\[
E_{Q_P^{(n)}}\!\left[
\left\|\widehat p_n^{\uparrow}-p^{\uparrow}(P)\right\|_1
\right]
\le
\frac{\kappa_{\mathrm{ord}}}{g}E_{Q_P^{(n)}}[Z_P]
\le
\kappa_{\mathrm{ord}}\beta_{\mathrm{mean}}\,
(\sqrt n\,g)^{-1}.
\]
Independently, both \(\widehat p_n^{\uparrow}(\omega)\) and \(p^{\uparrow}(P)\) lie in the simplex, hence
\[
\left\|\widehat p_n^{\uparrow}(\omega)-p^{\uparrow}(P)\right\|_1\le2
\]
for every \(\omega\), and therefore
\[
E_{Q_P^{(n)}}\!\left[
\left\|\widehat p_n^{\uparrow}-p^{\uparrow}(P)\right\|_1
\right]\le2 .
\] % lean: simplex_l1_le_two

\item
If \((\sqrt n\,g)^{-1}\le1\), the preceding rate bound and
\(\kappa_{\mathrm{ord}}\beta_{\mathrm{mean}}\le C\) give
\[
E_{Q_P^{(n)}}\!\left[
\left\|\widehat p_n^{\uparrow}-p^{\uparrow}(P)\right\|_1
\right]
\le
C(\sqrt n\,g)^{-1}
=
C\min\{1,(\sqrt n\,g)^{-1}\}.
\]
If \(1\le(\sqrt n\,g)^{-1}\), the simplex-diameter bound and \(2\le C\) give
\[
E_{Q_P^{(n)}}\!\left[
\left\|\widehat p_n^{\uparrow}-p^{\uparrow}(P)\right\|_1
\right]
\le
C
=
C\min\{1,(\sqrt n\,g)^{-1}\}.
\]
The constant \(C\) depends only on \(k,d_x,d_z,L,\pi_0,\sigma_0\), and the construction above supplies the required estimator for every \(n\ge1\). % lean: labeled_weight_upper
\end{enumerate}
\end{proof}
