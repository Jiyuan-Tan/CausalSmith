\begin{proof}[Proof of \cref{obj:lem:witness-observed-arm-mass}]
Fix \(\varepsilon\) with \(0\le\varepsilon\le 1/8\) and fix \(t\in\{0,1\}\).

\begin{enumerate}
\item Define the observed arm set
\[
\mathsf{Arm}_t:=\{(T,X,Z,Y)\in O_{2,2}:T=t\}.
\]
For a full-data record \(w\), let \(O(w)=(T(w),X(w),Z(w),Y(w))\).  Then
\[
O^{-1}(\mathsf{Arm}_t)=\{w:T(w)=t\}.
\]
The coordinate map \(O\) is measurable and \(\mathsf{Arm}_t\) is measurable, so \(\{w:T(w)=t\}\) is measurable.  By \cref{obj:def:observed-margin},
\[
(P_{\varepsilon}^{\mathrm{wit}})_O(\mathsf{Arm}_t)
=
P_{\varepsilon}^{\mathrm{wit}}\{w:T(w)=t\}.
\]
% lean: measurableSet_obsArm, obsMap_measurable, obs_witness_arm_mass

\item For \(q\in\mathbb R\) and \(r\in\{0,1\}\), put
\[
\beta(q,r):=
\begin{cases}
q, & r=1,\\
1-q, & r=0.
\end{cases}
\]
The finite witness law in \cref{obj:def:two-class-witness} assigns to the atom indexed by
\[
u\in\{1,2\},
\qquad
s,x,z,y_0,y_1\in\{0,1\}
\]
the real weight
\[
\begin{aligned}
\mathfrak w_{\varepsilon}(u,s,x,z,y_0,y_1)
&=
\mathfrak p_u\,
\beta(\mathfrak p_u,s)\,
\beta(\mathfrak x_u,x)\,
\beta(\mathfrak z_{s,u},z)\,
\beta(1/4,y_0)\,
\beta(\mathfrak y_{u,\varepsilon},y_1),
\end{aligned}
\]
where
\[
\mathfrak p_1=\frac25,\quad
\mathfrak p_2=\frac35,\quad
\mathfrak x_1=\frac15,\quad
\mathfrak x_2=\frac45,
\]
\[
\mathfrak z_{1,1}=\frac7{20},\quad
\mathfrak z_{1,2}=\frac34,\quad
\mathfrak z_{0,1}=\frac3{10},\quad
\mathfrak z_{0,2}=\frac7{10},
\]
and
\[
\mathfrak y_{1,\varepsilon}=\frac12-\varepsilon,\qquad
\mathfrak y_{2,\varepsilon}=\frac12+\varepsilon .
\]
The bounds on \(\varepsilon\) give
\[
0\le \mathfrak y_{u,\varepsilon}\le 1\qquad (u=1,2),
\]
and all the other displayed Bernoulli parameters also lie in \([0,1]\).  Hence every
\(\mathfrak w_{\varepsilon}(u,s,x,z,y_0,y_1)\) is nonnegative.  Therefore the real mass of the full-data arm event is the ordinary finite sum
\[
P_{\varepsilon}^{\mathrm{wit}}\{w:T(w)=t\}
=
\sum_{u=1}^{2}
\sum_{s,x,z,y_0,y_1\in\{0,1\}}
\mathfrak w_{\varepsilon}(u,s,x,z,y_0,y_1)\,\mathbf 1\{s=t\}.
\]
% lean: witnessLaw_real, witnessWeight_nonneg

\item Selecting the treatment coordinate in the finite sum gives
\[
\sum_{u=1}^{2}
\sum_{s,x,z,y_0,y_1\in\{0,1\}}
\mathfrak w_{\varepsilon}(u,s,x,z,y_0,y_1)\,\mathbf 1\{s=t\}
=
\sum_{u=1}^{2}
\sum_{x,z,y_0,y_1\in\{0,1\}}
\mathfrak w_{\varepsilon}(u,t,x,z,y_0,y_1).
\]
This is the two-case calculation \(t=0\) and \(t=1\): exactly one value of \(s\) has indicator one.
% lean: obs_witness_arm_mass

\item For each fixed \(u\in\{1,2\}\) and \(t\in\{0,1\}\),
\[
\begin{aligned}
&\sum_{x,z,y_0,y_1\in\{0,1\}}
\mathfrak w_{\varepsilon}(u,t,x,z,y_0,y_1)\\
&\quad =
\mathfrak p_u\beta(\mathfrak p_u,t)
\left(\sum_{x\in\{0,1\}}\beta(\mathfrak x_u,x)\right)
\left(\sum_{z\in\{0,1\}}\beta(\mathfrak z_{t,u},z)\right)
\left(\sum_{y_0\in\{0,1\}}\beta(1/4,y_0)\right)
\left(\sum_{y_1\in\{0,1\}}\beta(\mathfrak y_{u,\varepsilon},y_1)\right).
\end{aligned}
\]
Since \(\beta(q,0)+\beta(q,1)=1\) for every \(q\), the four nuisance factors sum to one, and hence
\[
\sum_{x,z,y_0,y_1\in\{0,1\}}
\mathfrak w_{\varepsilon}(u,t,x,z,y_0,y_1)
=
\mathfrak p_u\beta(\mathfrak p_u,t).
\]
% lean: witnessWeight_sum_nuisance

\item Combining the preceding identities,
\[
(P_{\varepsilon}^{\mathrm{wit}})_O(T=t)
=
\sum_{u=1}^{2}\mathfrak p_u\beta(\mathfrak p_u,t).
\]
If \(t=0\), then
\[
\sum_{u=1}^{2}\mathfrak p_u\beta(\mathfrak p_u,0)
=
\frac25\frac35+\frac35\frac25
=
\frac{12}{25}.
\]
If \(t=1\), then
\[
\sum_{u=1}^{2}\mathfrak p_u\beta(\mathfrak p_u,1)
=
\frac25\frac25+\frac35\frac35
=
\frac{13}{25}.
\]
Thus
\[
(P_{\varepsilon}^{\mathrm{wit}})_O(T=t)
=
\begin{cases}
12/25, & t=0,\\
13/25, & t=1.
\end{cases}
\]
% lean: obs_witness_arm_mass
\end{enumerate}
\end{proof}
