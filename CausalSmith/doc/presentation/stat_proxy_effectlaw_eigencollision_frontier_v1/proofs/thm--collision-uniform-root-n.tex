\begin{proof}[Proof of \cref{obj:thm:collision-uniform-root-n}]
\begin{enumerate}
\item Let
\[
\mathscr S=\{S(P'):P'\in\mathcal M\},
\qquad
\mathcal K=\overline{\mathscr S}
\]
in the finite-dimensional summary space equipped with \(d_S\).  By
\cref{obj:lem:uniform-summary-concentration}, choose a simultaneous
concentration constant \(C_0\ge 1\).  By
\cref{obj:prop:polynomial-net-law-estimator}, choose a positive lattice
tail constant \(C_{\mathrm{tail}}\) associated with the structured-lattice
estimator.  By \cref{obj:thm:gap-free-positive-measure-modulus}, choose
\(C_{\mathrm{mod}}>0\) and a continuous extension
\[
\overline F:\mathcal K\to \mathcal P_k([-L_\tau,L_\tau])
\]
such that
\[
W_1\{\overline F(s),\overline F(s')\}
\le
C_{\mathrm{mod}}\,d_S(s,s'),
\qquad s,s'\in\mathcal K,
\]
and
\[
\overline F(S(Q))=\nu_Q,
\qquad Q\in\mathcal M .
\]
Set
\[
C=
\max\{2C_{\mathrm{tail}},\,
      \max(2C_0,\,4C_{\mathrm{mod}}C_0L)\}.
\]
Then \(C>0\), and \(C\) depends on
\((k,d_x,d_z,L,\pi_0,\sigma_0)\).% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.collision_uniform_root_n

\item Fix \(n\ge1\).  The empirical summary \(\widehat S_n\) is a Borel
sample map: its coordinates are finite sums of Borel coordinate maps,
treatment-arm indicators, products, and the total empty-arm normalizers in
\cref{obj:def:empirical-summary}.  Applying
\cref{obj:prop:polynomial-net-law-estimator} at this \(n\), choose the
structured-lattice estimator \(\widehat\lambda_n\).  Its finite list contains
each well-formed candidate once, in lexicographic order, and the returned law is
\[
\widehat\lambda_n(\omega)=\lambda_{\widehat\vartheta_n(\omega)},
\]
where \(\widehat\vartheta_n(\omega)\) is the first lexicographic minimizer of
\(J_n(\vartheta;\widehat S_n(\omega))\) over the structured lattice in
\cref{obj:def:structured-lattice-program}, with threshold
\(\pi_0\sigma_0^2/2\).  The same result gives Borel measurability of
\(\widehat\lambda_n\) and, for every \(P\in\mathcal M\) and every
\(0<\zeta<1/2\),
\[
Q_P^{(n)}
\left\{
W_1(\widehat\lambda_n,\nu_P)>
C_{\mathrm{tail}}
\sqrt{\frac{\log(C_{\mathrm{tail}}/\zeta)}{n}}
\right\}
\le \zeta .
\]% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.polynomial_lattice_law_estimator

\item \Cref{obj:prop:summary-closure-compact} gives compactness of
\(\mathcal K\).  If \(\mathcal K=\varnothing\), take the selector
\[
\Pi(s)=0
\]
for every ambient summary \(s\), and define the repaired estimator by the
zero-law branch in \cref{obj:def:summary-space-repair}.  If
\(\mathcal K\ne\varnothing\), identify the finite-dimensional summary space
with Euclidean space.  The loss \((s,q)\mapsto d_S(q,s)\) is continuous by
\cref{obj:lem:summary-metric-continuity}, so
\cref{obj:lem:borel-nearest-point-selector} gives a Borel map \(\Pi\) such
that
\[
\Pi(s)\in\mathcal K,
\qquad
d_S(\Pi(s),s)\le d_S(q,s)
\quad\text{for all }q\in\mathcal K .
\]
With this selector, set
\[
\widehat\nu_n(\omega)
=
\begin{cases}
\delta_0, & \mathcal K=\varnothing,\\
\overline F\{\Pi(\widehat S_n(\omega))\}, & \mathcal K\ne\varnothing .
\end{cases}
\]
Continuity of \(\overline F\), Borel measurability of \(\Pi\), and Borel
measurability of \(\widehat S_n\) make \(\widehat\nu_n\) a total Borel
estimator into \(\mathcal P_k([-L_\tau,L_\tau])\).% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.compactLoss_selector_of_homeomorph

\item Fix \(P\in\mathcal M\).  Then \(S(P)\in\mathscr S\subseteq\mathcal K\),
so \(\mathcal K\ne\varnothing\).  For every sample \(\omega\), nearest-summary
optimality gives
\[
d_S\{\Pi(\widehat S_n(\omega)),\widehat S_n(\omega)\}
\le
d_S\{S(P),\widehat S_n(\omega)\}
=
d_S\{\widehat S_n(\omega),S(P)\}.
\]
By the triangle inequality for \(d_S\),
\[
\begin{aligned}
d_S\{\Pi(\widehat S_n(\omega)),S(P)\}
&\le
d_S\{\Pi(\widehat S_n(\omega)),\widehat S_n(\omega)\}
+d_S\{\widehat S_n(\omega),S(P)\}  \\
&\le
2d_S\{\widehat S_n(\omega),S(P)\}.
\end{aligned}
\]
Since \(\overline F(S(P))=\nu_P\), the Lipschitz extension bound gives
\[
\begin{aligned}
W_1(\widehat\nu_n(\omega),\nu_P)
&\le
C_{\mathrm{mod}}\,
d_S\{\Pi(\widehat S_n(\omega)),S(P)\}  \\
&\le
2C_{\mathrm{mod}}\,d_S\{\widehat S_n(\omega),S(P)\}.
\end{aligned}
\]% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.collision_uniform_root_n

\item Fix \(0<\eta<1/2\), and define
\[
\Theta_\lambda
=
C_{\mathrm{tail}}
\sqrt{\frac{\log(C_{\mathrm{tail}}/(\eta/2))}{n}},
\qquad
\Theta_S
=
C_0L
\sqrt{\frac{\log(C_0/(\eta/2))}{n}},
\]
\[
\Theta_C
=
C\sqrt{\frac{\log(C/\eta)}{n}} .
\]
The definition of \(C\) implies
\[
2C_{\mathrm{tail}}\le C,
\qquad
2C_0\le C,
\qquad
4C_{\mathrm{mod}}C_0L\le C .
\]
Hence
\[
\frac{C_{\mathrm{tail}}}{\eta/2}
=
\frac{2C_{\mathrm{tail}}}{\eta}
\le
\frac{C}{\eta},
\qquad
\frac{C_0}{\eta/2}
=
\frac{2C_0}{\eta}
\le
\frac{C}{\eta}.
\]
Monotonicity of \(\log\) and of the square root gives
\[
\Theta_\lambda\le\Theta_C
\]
and
\[
2C_{\mathrm{mod}}\Theta_S\le\Theta_C .
\]
Therefore
\[
\begin{aligned}
&\left\{
\max\!\left[
W_1(\widehat\lambda_n,\nu_P),
W_1(\widehat\nu_n,\nu_P)
\right]>\Theta_C
\right\}  \\
&\qquad\subseteq
\{W_1(\widehat\lambda_n,\nu_P)>\Theta_\lambda\}
\cup
\{d_S(\widehat S_n,S(P))>\Theta_S\}.
\end{aligned}
\]
Indeed, on the complement of the right-hand union, the lattice error is at most
\(\Theta_\lambda\le\Theta_C\), while the repaired-summary error is at most
\(2C_{\mathrm{mod}}\Theta_S\le\Theta_C\).% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.collision_uniform_root_n

\item Apply the lattice tail bound with \(\zeta=\eta/2\) and
\cref{obj:lem:uniform-summary-concentration} with \(\zeta=\eta/2\).  The
displayed set inclusion and the union bound yield
\[
\begin{aligned}
&Q_P^{(n)}
\left\{
\max\!\left[
W_1(\widehat\lambda_n,\nu_P),
W_1(\widehat\nu_n,\nu_P)
\right]>
C\sqrt{\frac{\log(C/\eta)}{n}}
\right\} \\
&\qquad\le
Q_P^{(n)}\{W_1(\widehat\lambda_n,\nu_P)>\Theta_\lambda\}
+
Q_P^{(n)}\{d_S(\widehat S_n,S(P))>\Theta_S\} \\
&\qquad\le
\eta/2+\eta/2
=\eta .
\end{aligned}
\]
The arm-count conclusion is also supplied by
\cref{obj:lem:uniform-summary-concentration}: for each
\(t\in\{0,1\}\),
\[
Q_P^{(n)}\{N_{t,n}=0\}\le (1-k\pi_0)^n .
\]% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.collision_uniform_root_n
\end{enumerate}
\end{proof}
