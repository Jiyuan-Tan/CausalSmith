\begin{proof}[Proof of \cref{obj:lem:observed-outcome-proxy-moment-factorization}]
\emph{Step 1.} Fix \(t\in\{0,1\}\).  For \(1\le u\le k\), put
\[
\mathsf U_u:=\{U=u\},\qquad
\mathsf R_t:=\{T=t\},\qquad
\mathsf H_{u,t}:=\mathsf U_u\cap\mathsf R_t .
\]
For a full-data event \(\mathfrak e\) and a real-valued measurable function \(\varphi\), write
\[
\mathbb E_{\mathfrak e}^{P}[\varphi]
:=
\{P(\mathfrak e)\}^{-1}\int_{\mathfrak e}\varphi\,dP,
\]
with the total convention \(0^{-1}=0\).  The latent-arm positivity assumption gives, for every \(u\),
\[
0<\pi_0\le P(\mathsf H_{u,t})\le P(\mathsf U_u).
\]
Also, \cref{obj:lem:arm-mass-lower-bound} gives
\[
k\pi_0\le P(\mathsf R_t),
\]
and since \(2\le k\) and \(0<\pi_0\),
\[
0<k\pi_0\le P(\mathsf R_t).
\]
Thus all normalizations used below have positive denominators on each latent class, each latent-treatment cell, and the treatment arm.
% lean: latentCell_pos_of_latentArmPositivity
% lean: arm_mass_lower_of_latentArmPositivity

\emph{Step 2.} Let \(P_u^\circ\) be the normalized restriction of \(P\) to \(\mathsf U_u\), equivalently the probability law with density \(P(\mathsf U_u)^{-1}\mathbf 1_{\mathsf U_u}\) with respect to \(P\).  Latent ignorability in \cref{obj:ass:latent-ignorability} gives independence of \(Y(t)\) and \(T\) under \(P_u^\circ\).  Therefore
\[
\int_{\mathsf R_t}Y(t)\,dP_u^\circ
=
P_u^\circ(\mathsf R_t)\int Y(t)\,dP_u^\circ,
\qquad
P_u^\circ(\mathsf R_t)=\frac{P(\mathsf H_{u,t})}{P(\mathsf U_u)} .
\]
Using the normalized-restriction formula on both sides,
\[
\frac{1}{P(\mathsf U_u)}\int_{\mathsf H_{u,t}}Y(t)\,dP
=
\frac{P(\mathsf H_{u,t})}{P(\mathsf U_u)}
\frac{1}{P(\mathsf U_u)}\int_{\mathsf U_u}Y(t)\,dP .
\]
The positive denominators from Step 1 allow cancellation, giving
\[
\mathbb E_{\mathsf H_{u,t}}^{P}[Y(t)]
=
\mathbb E_{\mathsf U_u}^{P}[Y(t)]
=
\mu_{tu},
\]
where the last equality is \cref{obj:def:latent-mean}.  On \(\mathsf R_t\), consistency in \cref{obj:ass:consistency} gives \(Y=Y(t)\) almost surely, hence
\[
\mathbb E_{\mathsf H_{u,t}}^{P}[Y]
=
\mathbb E_{\mathsf H_{u,t}}^{P}[Y(t)]
=
\mu_{tu}.
\]
% lean: conditionalMean_potential_latentCell
% lean: conditionalMean_observed_eq_potential

\emph{Step 3.} Fix coordinates \(1\le \iota_z\le d_z\) and \(1\le \iota_x\le d_x\).  Target-proxy separation in \cref{obj:ass:target-proxy-separation}, applied under \(P_u^\circ\), gives independence of \(X\) and \((Y,T)\) conditional on \(U=u\).  Define
\[
\psi_t(y,s):=y\,\mathbf 1\{s=t\},
\qquad (y,s)\in\mathbb R\times\{0,1\}.
\]
The product identity under \(P_u^\circ\) gives
\[
\int X_{\iota_x}\psi_t(Y,T)\,dP_u^\circ
=
\left(\int X_{\iota_x}\,dP_u^\circ\right)
\left(\int \psi_t(Y,T)\,dP_u^\circ\right).
\]
The three terms are
\[
\int X_{\iota_x}\psi_t(Y,T)\,dP_u^\circ
=
\frac{1}{P(\mathsf U_u)}
\int_{\mathsf H_{u,t}}X_{\iota_x}Y\,dP,
\]
\[
\int X_{\iota_x}\,dP_u^\circ
=
\frac{1}{P(\mathsf U_u)}
\int_{\mathsf U_u}X_{\iota_x}\,dP,
\qquad
\int \psi_t(Y,T)\,dP_u^\circ
=
\frac{1}{P(\mathsf U_u)}
\int_{\mathsf H_{u,t}}Y\,dP .
\]
Multiplying by \(P(\mathsf U_u)/P(\mathsf H_{u,t})\) and using Step 1 yields
\[
\mathbb E_{\mathsf H_{u,t}}^{P}[X_{\iota_x}Y]
=
\mathbb E_{\mathsf U_u}^{P}[X_{\iota_x}]\,
\mathbb E_{\mathsf H_{u,t}}^{P}[Y].
\]
By \cref{obj:def:target-feature} and Step 2,
\[
\mathbb E_{\mathsf H_{u,t}}^{P}[X_{\iota_x}Y]
=
B(\iota_x,u)\,\mu_{tu}.
\]
% lean: conditionalMean_target_mul_observed

\emph{Step 4.} Let \(P_{u,t}^\circ\) be the normalized restriction of \(P\) to \(\mathsf H_{u,t}\).  Reference-proxy separation in \cref{obj:ass:reference-proxy-separation}, applied under \(P_{u,t}^\circ\), gives independence of \(Z\) and \((X,Y)\) on the latent-treatment cell.  Applying the product identity to \(Z_{\iota_z}\) and \(X_{\iota_x}Y\),
\[
\int Z_{\iota_z}X_{\iota_x}Y\,dP_{u,t}^\circ
=
\left(\int Z_{\iota_z}\,dP_{u,t}^\circ\right)
\left(\int X_{\iota_x}Y\,dP_{u,t}^\circ\right).
\]
Equivalently,
\[
\mathbb E_{\mathsf H_{u,t}}^{P}[Z_{\iota_z}X_{\iota_x}Y]
=
\mathbb E_{\mathsf H_{u,t}}^{P}[Z_{\iota_z}]\,
\mathbb E_{\mathsf H_{u,t}}^{P}[X_{\iota_x}Y].
\]
Combining this identity with Step 3 and \cref{obj:def:reference-feature} gives the cellwise factorization
\[
\mathbb E_{\mathsf H_{u,t}}^{P}[YZ_{\iota_z}X_{\iota_x}]
=
A_t(\iota_z,u)\,\mu_{tu}\,B(\iota_x,u).
\]
% lean: conditionalMean_reference_targetOutcome
% lean: latentCell_outcomeProxy_factorization

\emph{Step 5.} Set
\[
\varphi_{\iota_z,\iota_x}:=YZ_{\iota_z}X_{\iota_x}.
\]
The bounded outcome-proxy product condition in \cref{obj:ass:bounded-outcome-proxy-product} implies
\[
|\varphi_{\iota_z,\iota_x}|
=
|YZ_{\iota_z}X_{\iota_x}|
\le
\|YZX^\top\|_{\mathrm{op}}
\le L
\]
almost surely, since every matrix entry is bounded by the operator norm.  Thus \(\varphi_{\iota_z,\iota_x}\) is integrable on every latent-treatment cell.  The treatment arm decomposes as the disjoint finite union
\[
\mathsf R_t=\bigsqcup_{u=1}^{k}\mathsf H_{u,t},
\]
so
\[
\int_{\mathsf R_t}\varphi_{\iota_z,\iota_x}\,dP
=
\sum_{u=1}^{k}\int_{\mathsf H_{u,t}}\varphi_{\iota_z,\iota_x}\,dP .
\]
Dividing by the positive arm mass from Step 1 and rewriting each cell integral as a conditional mean gives
\[
\mathbb E_{\mathsf R_t}^{P}[\varphi_{\iota_z,\iota_x}]
=
\sum_{u=1}^{k}
\frac{P(\mathsf H_{u,t})}{P(\mathsf R_t)}
\mathbb E_{\mathsf H_{u,t}}^{P}[\varphi_{\iota_z,\iota_x}].
\]
Substituting Step 4,
\[
\mathbb E_{\mathsf R_t}^{P}[YZ_{\iota_z}X_{\iota_x}]
=
\sum_{u=1}^{k}
A_t(\iota_z,u)\,
\frac{P(U=u,T=t)}{P(T=t)}\,
\mu_{tu}\,
B(\iota_x,u).
\]
% lean: conditionalMean_fullDataArm_eq_sum_latentCell

\emph{Step 6.} By \cref{obj:def:observed-margin}, \(P_O\) is the pushforward of \(P\) by \(O=(T,X,Z,Y)\).  Hence the observed-arm conditional mean agrees with the corresponding full-data arm conditional mean:
\[
E_{P_O}\!\left[YZ_{\iota_z}X_{\iota_x}\mid T=t\right]
=
\mathbb E_{\mathsf R_t}^{P}[YZ_{\iota_z}X_{\iota_x}].
\]
The armwise \(YZX^\top\) moment notation in \cref{obj:def:observable-population-moments} identifies the \((\iota_z,\iota_x)\) entry of \(N_t(P)\) with this same full-data conditional moment.  Therefore
\[
N_t(P)(\iota_z,\iota_x)
=
\sum_{u=1}^{k}
A_t(\iota_z,u)\,
\frac{P(U=u,T=t)}{P(T=t)}\,
\mu_{tu}\,
B(\iota_x,u).
\]
With
\[
\Gamma_t
=
\operatorname{diag}\left\{
\frac{P(U=u,T=t)}{P(T=t)}:1\le u\le k
\right\},
\]
the \((\iota_z,\iota_x)\) entry of
\[
A_t\,\Gamma_t\,\operatorname{diag}\{\mu_{tu}:1\le u\le k\}\,B^\top
\]
is exactly the displayed sum.  Since \(\iota_z\) and \(\iota_x\) were arbitrary,
\[
N_t(P)
=
A_t\,\Gamma_t\,\operatorname{diag}\{\mu_{tu}:1\le u\le k\}\,B^\top .
\]
% lean: conditionalMean_obsArm_eq_fullDataArm
% lean: observedOutcomeProxyMoment_factorization
\end{proof}
