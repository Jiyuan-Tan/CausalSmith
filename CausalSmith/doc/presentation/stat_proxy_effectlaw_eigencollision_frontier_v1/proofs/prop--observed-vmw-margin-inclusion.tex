\begin{proof}[Proof of \cref{obj:prop:observed-vmw-margin-inclusion}]
Fix \(P\in\mathcal M\) satisfying the displayed hypotheses, and set
\[
L_\tau:=\frac{4L\sqrt{d_z}}{\sigma_0}.
\]
All model restrictions used below are the components of \(\mathcal M\) in
\cref{obj:def:model-class}.

\begin{enumerate}
\item For each treatment arm \(t\in\{0,1\}\), \cref{obj:lem:arm-mass-lower-bound} gives
\[
P(T=t)\ge k\pi_0 .
\]
Since \(k\ge2\) and \(\pi_0>0\), this arm probability is positive.  Define
\[
\Gamma_t
:=
\operatorname{diag}\left\{
\frac{P(U=u,T=t)}{P(T=t)}:1\le u\le k
\right\}.
\]
The diagonal entries are
\[
\frac{P(U=u,T=t)}{P(T=t)}
=
P(U=u\mid T=t),
\]
so \cref{obj:lem:observed-proxy-moment-factorization} yields
\[
M_t(P)=A_t\Gamma_tB^\top
=
A_t\operatorname{diag}\{P(U=u\mid T=t):1\le u\le k\}B^\top .
\]
% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.observedProxyMoment_factorization

\item Latent-arm positivity gives, for every latent class \(u\),
\[
\frac{P(U=u,T=t)}{P(T=t)}
\ge
P(U=u,T=t)
\ge
\pi_0,
\]
because \(P(T=t)\le1\).  Hence, for every \(\beta\in\mathbb R^k\),
\[
\|\Gamma_t\beta\|_2^2
=
\sum_{u=1}^k \Gamma_t(u,u)^2\beta_u^2
\ge
\pi_0^2\sum_{u=1}^k\beta_u^2
=
\pi_0^2\|\beta\|_2^2,
\]
and therefore
\[
\sigma_k(\Gamma_t)\ge\pi_0 .
\]
The proxy-rank margin in \cref{obj:ass:proxy-rank-margin} gives
\[
\sigma_k(A_t)\ge\sigma_0,
\qquad
\sigma_k(B)\ge\sigma_0 .
\]
Using the product lower bound for full-column-rank factors in the factorization above,
\[
\sigma_k\{M_t(P)\}
=
\sigma_k(A_t\Gamma_tB^\top)
\ge
\sigma_k(A_t)\sigma_k(\Gamma_t)\sigma_k(B)
\ge
\pi_0\sigma_0^2 .
\]
% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.observedProxyMoment_minSingular_of_factorization

\item Let
\[
\mathcal H(P):=
\operatorname{row}\!\left(\operatorname{stack}(M_0(P),M_1(P))\right)
\subseteq\mathbb R^{d_x},
\]
and let \(V\in\mathbb R^{d_x\times k}\) have orthonormal columns spanning \(\mathcal H(P)\).  For the fixed arm \(t\), the row space of \(M_t(P)\) is contained in \(\mathcal H(P)\):
\[
\operatorname{range}\{M_t(P)^\top\}\subseteq \mathcal H(P)=\operatorname{range}(V).
\]
The lower bound from the previous step gives \(\sigma_k\{M_t(P)\}>0\), so this row space has dimension at least \(k\).  Since \(\operatorname{range}(V)\) has dimension \(k\), the two spaces are equal.  The map \(r\mapsto Vr\) is an isometry from \(\mathbb R^k\) onto this row space, and composing \(M_t(P)\) with that isometry preserves its positive singular values.  Thus
\[
\sigma_k\{M_t(P)V\}
=
\sigma_k\{M_t(P)\},
\qquad
\pi_0\sigma_0^2\le \sigma_k\{M_t(P)V\}.
\]
% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.observedProxyMoment_compression_margin

\item The stacked proxy signal margin is supplied by
\cref{obj:lem:stacked-proxy-moment-signal-margin}.  Applied with the present model membership and numerical hypotheses, it gives
\[
\pi_0\sigma_0^2
\le
\sigma_k\!\left(\operatorname{stack}(M_0(P),M_1(P))\right),
\]
with the displayed one-based singular-value indexing of the statement.
% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.stackedProxyMoment_minSingular

\item By \cref{obj:def:observable-population-moments},
\[
M_t(P)=E_P[ZX^\top\mid T=t],
\qquad
N_t(P)=E_P[YZX^\top\mid T=t],
\qquad
m_X(P)=E_P[X].
\]
The envelope components of \(\mathcal M\) give
\[
\|ZX^\top\|_{\mathrm{op}}\le L,
\qquad
\|YZX^\top\|_{\mathrm{op}}\le L,
\qquad
\|X\|_2\le L
\quad P\text{-a.s.}
\]
Since \(P(T=t)>0\), Jensen's inequality for the two conditional matrix means and for the unconditional vector mean yields
\[
\|M_t(P)\|_{\mathrm{op}}\le L,
\qquad
\|N_t(P)\|_{\mathrm{op}}\le L,
\qquad
\|m_X(P)\|_2\le L .
\]
% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.observedSummary_envelopes_of_model

\item Fix \(u\in\{1,\ldots,k\}\) and \(t\in\{0,1\}\), and put
\[
\mathsf{Cell}_{u,t}:=\{U=u,T=t\},
\qquad
\mathsf{Class}_u:=\{U=u\}.
\]
Latent-arm positivity gives
\[
P(\mathsf{Cell}_{u,t})\ge\pi_0>0.
\]
Let \(e_u^{(k)}\) be the \(u\)th canonical vector in \(\mathbb R^k\).  The proxy-rank margin implies
\[
\|A_te_u^{(k)}\|_2
\ge
\sigma_k(A_t)\|e_u^{(k)}\|_2
\ge
\sigma_0 .
\]
Since \(d_z\ge k\ge2\), some coordinate \(\jmath_{u,t}\in\{1,\ldots,d_z\}\) satisfies
\[
\left|E_P[Z_{\jmath_{u,t}}\mid \mathsf{Cell}_{u,t}]\right|
=
|A_t(\jmath_{u,t},u)|
\ge
\frac{\sigma_0}{\sqrt{d_z}} .
\]
Define
\[
\zeta_{u,t}:=\frac{\sigma_0}{2\sqrt{d_z}},
\qquad
\mathsf{Large}_{u,t}:=\{|Z_{\jmath_{u,t}}|\ge \zeta_{u,t}\}.
\]
If \(P(\mathsf{Large}_{u,t}\mid\mathsf{Cell}_{u,t})=0\), then
\[
\left|E_P[Z_{\jmath_{u,t}}\mid \mathsf{Cell}_{u,t}]\right|
\le
E_P[|Z_{\jmath_{u,t}}|\mid \mathsf{Cell}_{u,t}]
\le
\zeta_{u,t}
<
\frac{\sigma_0}{\sqrt{d_z}},
\]
a contradiction.  Hence
\[
P(\mathsf{Large}_{u,t}\mid\mathsf{Cell}_{u,t})>0 .
\]
By \cref{obj:lem:proxy-coordinate-envelopes},
\[
|YZ_{\jmath_{u,t}}|\le L
\quad P\text{-a.s. on }\mathsf{Cell}_{u,t}.
\]
The reference-proxy separation component in \cref{obj:ass:reference-proxy-separation} gives \(Z_{\jmath_{u,t}}\perp Y\mid \mathsf{Cell}_{u,t}\).  Therefore positive conditional probability for
\[
\mathsf{Bad}_{u,t}:=
\left\{|Y|>\frac{L}{\zeta_{u,t}}\right\}
\]
would give
\[
P(\mathsf{Large}_{u,t}\cap\mathsf{Bad}_{u,t}\mid\mathsf{Cell}_{u,t})
=
P(\mathsf{Large}_{u,t}\mid\mathsf{Cell}_{u,t})
P(\mathsf{Bad}_{u,t}\mid\mathsf{Cell}_{u,t})
>0.
\]
On this intersection,
\[
L<|Y|\,|Z_{\jmath_{u,t}}|=|YZ_{\jmath_{u,t}}|,
\]
contradicting the product envelope.  Thus
\[
|Y|
\le
\frac{L}{\zeta_{u,t}}
=
\frac{2L\sqrt{d_z}}{\sigma_0}
=
\frac{L_\tau}{2}
\quad P\text{-a.s. conditional on }\mathsf{Cell}_{u,t}.
\]
On \(\{T=t\}\), the consistency component in \cref{obj:ass:consistency} gives \(Y=Y(t)\) almost surely.  Inside the conditional law given \(\mathsf{Class}_u\), the event \(\{T=t\}\) has positive probability, and the latent-ignorability component in \cref{obj:ass:latent-ignorability} gives \(Y(t)\perp T\mid U\).  If
\[
\left\{|Y(t)|>\frac{L_\tau}{2}\right\}
\]
had positive conditional probability given \(\mathsf{Class}_u\), independence would give a positive-probability violation on \(\mathsf{Cell}_{u,t}\), contradicting the displayed cell bound.  Hence
\[
|Y(t)|\le\frac{L_\tau}{2}
\quad P\text{-a.s. conditional on }U=u .
\]
Finally, \cref{obj:lem:latent-mean-envelope,obj:lem:latent-effect-envelope} give
\[
|\mu_{tu}|\le \frac{L_\tau}{2},
\qquad
|\tau_u|\le L_\tau .
\]
% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.ae_abs_potential_le_on_latentClass

\item It remains to identify membership in the published VMW parameter class.  The scope in \cref{obj:def:published-vmw-regimes}, together with the cited VMW correspondence recorded in \cref{obj:synth_11} and \citep{VirkMazaheriWu2026}, identifies \(\mathsf{Model}_{\mathrm{VMW}}(k,d_x,d_z,P)\) with the qualitative structural conditions summarized there.  Those conditions are supplied as follows.  The proxy independences, consistency, and armwise latent ignorability are components of \(\mathcal M\).  The proxy-rank margin and \(\sigma_0>0\) give full column rank:
\[
\sigma_k(A_0)\ge\sigma_0>0,\qquad
\sigma_k(A_1)\ge\sigma_0>0,\qquad
\sigma_k(B)\ge\sigma_0>0.
\]
Latent-arm positivity gives
\[
P(U=u,T=t)\ge\pi_0>0
\]
for every \(u\) and \(t\), and hence the corresponding latent treatment probabilities are strictly positive.  Together with \(2\le k\), \(k\le d_x\), and \(k\le d_z\), these are exactly the qualitative conditions for \(P\in\mathsf{Model}_{\mathrm{VMW}}(k,d_x,d_z)\).
% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.observed_vmw_margin_inclusion
\end{enumerate}
\end{proof}
