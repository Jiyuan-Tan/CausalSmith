\begin{proof}[Proof of \cref{obj:lem:ordered-masses-measure-invariance}]
Write the two labelled representatives as
\[
\nu=\bigl((\varpi^\nu_{\mathfrak i})_{\mathfrak i\in I_k},
(\chi^\nu_{\mathfrak i})_{\mathfrak i\in I_k}\bigr),
\qquad
\xi=\bigl((\varpi^\xi_{\mathfrak i})_{\mathfrak i\in I_k},
(\chi^\xi_{\mathfrak i})_{\mathfrak i\in I_k}\bigr),
\qquad
I_k:=\{0,\ldots,k-1\}.
\]
Validity means
\[
0\le \varpi^\nu_{\mathfrak i},\quad
0\le \varpi^\xi_{\mathfrak i},\quad
\sum_{\mathfrak i\in I_k}\varpi^\nu_{\mathfrak i}=1,\quad
\sum_{\mathfrak i\in I_k}\varpi^\xi_{\mathfrak i}=1,
\]
with all labelled atoms lying in the prescribed compact interval.  For any labelled representative
\(\zeta=((\varpi^\zeta_{\mathfrak i}),(\chi^\zeta_{\mathfrak i}))\), put
\[
\operatorname{supp}_+(\zeta)
:=
\{\chi^\zeta_{\mathfrak i}:\mathfrak i\in I_k,\ 0<\varpi^\zeta_{\mathfrak i}\}.
\]
If \(k=0\), there are no vector coordinates to check.

\begin{enumerate}
\item\label{pf:ordered-masses:aggregate}
The equality of represented probability measures gives equality of aggregate mass at every real point:
\[
\sum_{\mathfrak i:\,\chi^\nu_{\mathfrak i}=\mathfrak x}\varpi^\nu_{\mathfrak i}
=
\sum_{\mathfrak j:\,\chi^\xi_{\mathfrak j}=\mathfrak x}\varpi^\xi_{\mathfrak j}
\qquad\text{for every }\mathfrak x\in\mathbb R .
\]
Indeed, by \cref{obj:def:atomic-law-class,obj:synth_28}, the represented measure assigns \(\{\mathfrak x\}\) the aggregate of the positive parts of the labelled weights at \(\mathfrak x\); validity gives \(\max\{\varpi^\nu_{\mathfrak i},0\}=\varpi^\nu_{\mathfrak i}\) and \(\max\{\varpi^\xi_{\mathfrak j},0\}=\varpi^\xi_{\mathfrak j}\).  Evaluating the common represented measure on \(\{\mathfrak x\}\) gives the displayed identity.
% lean: AtomicLaw.ProbabilityLaw.MeasureEquivalent.aggregate_weight

\item\label{pf:ordered-masses:support}
Consequently,
\[
\operatorname{supp}_+(\nu)=\operatorname{supp}_+(\xi).
\]
If \(\mathfrak x=\chi^\nu_{\mathfrak i}\) and \(0<\varpi^\nu_{\mathfrak i}\), then the aggregate mass of \(\nu\) at \(\mathfrak x\) is positive.  By \cref{pf:ordered-masses:aggregate}, the aggregate mass of \(\xi\) at \(\mathfrak x\) is positive, and nonnegativity of the labelled weights yields some \(\mathfrak j\in I_k\) with
\[
\chi^\xi_{\mathfrak j}=\mathfrak x,
\qquad
0<\varpi^\xi_{\mathfrak j}.
\]
This proves one inclusion, and the reverse inclusion follows by interchanging \(\nu\) and \(\xi\).
% lean: support_eq_of_measureEquivalent

\item\label{pf:ordered-masses:rank}
For a representative \(\zeta\), define the rank of a labelled slot by
\[
\operatorname{rank}_{\zeta}(\mathfrak i)
:=
\#\{\mathfrak q\in I_k:\chi^\zeta_{\mathfrak q}<\chi^\zeta_{\mathfrak i}\}.
\]
The counted set excludes \(\mathfrak i\), so
\[
\operatorname{rank}_{\zeta}(\mathfrak i)<k .
\]
Moreover, if \(\chi^\zeta_{\mathfrak i}<\chi^\zeta_{\mathfrak j}\), then
\[
\operatorname{rank}_{\zeta}(\mathfrak i)
<
\operatorname{rank}_{\zeta}(\mathfrak j),
\]
because the first counted set is contained in the second and \(\mathfrak i\) belongs to the second but not the first.  Hence, whenever
\(\mathfrak i\mapsto\chi^\zeta_{\mathfrak i}\) is injective, the map
\(\mathfrak i\mapsto\operatorname{rank}_{\zeta}(\mathfrak i)\) is injective, and therefore bijective on the finite set \(I_k\).
% lean: atomicRank_lt; atomicRank_lt_of_atom_lt; atomicRank_injective; rank_bijective

\item\label{pf:ordered-masses:full-support}
The following full-support criterion holds:
\[
\#\operatorname{supp}_+(\zeta)=k
\quad\Longleftrightarrow\quad
\left[
(\forall \mathfrak i\in I_k,\ 0<\varpi^\zeta_{\mathfrak i})
\ \text{and}\
\mathfrak i\mapsto\chi^\zeta_{\mathfrak i}\ \text{is injective}
\right].
\]
For the forward implication, \(\operatorname{supp}_+(\zeta)\) is the image of
\(\{\mathfrak i\in I_k:0<\varpi^\zeta_{\mathfrak i}\}\) under
\(\mathfrak i\mapsto\chi^\zeta_{\mathfrak i}\), so
\[
\#\operatorname{supp}_+(\zeta)
\le
\#\{\mathfrak i\in I_k:0<\varpi^\zeta_{\mathfrak i}\}
\le k .
\]
If the left side equals \(k\), both inequalities are equalities; thus every labelled weight is positive, and equality in the image-cardinality bound gives injectivity of the labelled locations.  The reverse implication follows because all \(k\) labelled slots are then positive and distinct.
% lean: full_support_iff

\item\label{pf:ordered-masses:apply-rank}
Suppose first that
\[
\#\operatorname{supp}_+(\nu)=k .
\]
By \cref{pf:ordered-masses:support},
\[
\#\operatorname{supp}_+(\xi)=k .
\]
Then \cref{pf:ordered-masses:full-support} gives
\[
0<\varpi^\nu_{\mathfrak i},
\quad
0<\varpi^\xi_{\mathfrak i}
\qquad(\mathfrak i\in I_k),
\]
and both labelled-location maps are injective.  For either such full-support representative \(\zeta\), \cref{obj:synth_26} gives
\[
\operatorname{orderedMasses}(\zeta)_{\operatorname{rank}_{\zeta}(\mathfrak i)}
=
\sum_{\mathfrak q:\,
\operatorname{rank}_{\zeta}(\mathfrak q)=\operatorname{rank}_{\zeta}(\mathfrak i)}
\varpi^\zeta_{\mathfrak q}
=
\varpi^\zeta_{\mathfrak i},
\]
where the last equality uses the rank injectivity from \cref{pf:ordered-masses:rank}.
% lean: orderedMasses_apply_rank

\item\label{pf:ordered-masses:weight-match}
Fix a coordinate \(\mathfrak r\in I_k\).  Since the rank map for \(\nu\) is bijective by \cref{pf:ordered-masses:rank}, choose \(\mathfrak i\in I_k\) such that
\[
\mathfrak r=\operatorname{rank}_{\nu}(\mathfrak i).
\]
The support equality in \cref{pf:ordered-masses:support} gives \(\mathfrak j\in I_k\) with
\[
\chi^\xi_{\mathfrak j}=\chi^\nu_{\mathfrak i},
\qquad
0<\varpi^\xi_{\mathfrak j}.
\]
Using \cref{pf:ordered-masses:aggregate} at \(\mathfrak x=\chi^\nu_{\mathfrak i}\), and using injectivity of both labelled-location maps to collapse the two sums, yields
\[
\varpi^\nu_{\mathfrak i}
=
\sum_{\mathfrak q:\,\chi^\nu_{\mathfrak q}=\chi^\nu_{\mathfrak i}}
\varpi^\nu_{\mathfrak q}
=
\sum_{\mathfrak q:\,\chi^\xi_{\mathfrak q}=\chi^\nu_{\mathfrak i}}
\varpi^\xi_{\mathfrak q}
=
\varpi^\xi_{\mathfrak j}.
\]
% lean: orderedMasses_eq_of_measureEquivalent

\item\label{pf:ordered-masses:rank-match}
For each \(\mathfrak q\in I_k\), \cref{pf:ordered-masses:support} supplies a slot
\(\psi(\mathfrak q)\in I_k\) such that
\[
\chi^\xi_{\psi(\mathfrak q)}=\chi^\nu_{\mathfrak q}.
\]
The map \(\psi\) is injective: if \(\psi(\mathfrak q_1)=\psi(\mathfrak q_2)\), then
\(\chi^\nu_{\mathfrak q_1}=\chi^\nu_{\mathfrak q_2}\), and injectivity of the locations of \(\nu\) gives
\(\mathfrak q_1=\mathfrak q_2\).  Thus \(\psi\) is bijective.  Since also
\(\chi^\xi_{\mathfrak j}=\chi^\nu_{\mathfrak i}\), for every \(\mathfrak q\in I_k\),
\[
\chi^\nu_{\mathfrak q}<\chi^\nu_{\mathfrak i}
\quad\Longleftrightarrow\quad
\chi^\xi_{\psi(\mathfrak q)}<\chi^\xi_{\mathfrak j}.
\]
The bijection \(\psi\) therefore identifies the two sets counted by the ranks, so
\[
\operatorname{rank}_{\nu}(\mathfrak i)
=
\operatorname{rank}_{\xi}(\mathfrak j).
\]
Combining this rank equality with \cref{pf:ordered-masses:apply-rank,pf:ordered-masses:weight-match} gives
\[
\operatorname{orderedMasses}(\nu)_{\mathfrak r}
=
\varpi^\nu_{\mathfrak i}
=
\varpi^\xi_{\mathfrak j}
=
\operatorname{orderedMasses}(\xi)_{\mathfrak r}.
\]
Since \(\mathfrak r\) was arbitrary, the ordered-mass vectors are equal in the full-support case.
% lean: orderedMasses_eq_of_measureEquivalent

\item\label{pf:ordered-masses:fallback}
It remains to consider
\[
\#\operatorname{supp}_+(\nu)\ne k .
\]
By \cref{pf:ordered-masses:support},
\[
\#\operatorname{supp}_+(\xi)\ne k .
\]
Using \cref{pf:ordered-masses:full-support}, the positivity-and-injectivity condition in \cref{obj:synth_26} fails for both representatives.  Hence both ordered-mass vectors take the barycentric fallback value:
\[
\operatorname{orderedMasses}(\nu)_{\mathfrak r}
=
k^{-1}
=
\operatorname{orderedMasses}(\xi)_{\mathfrak r}
\qquad\text{for every }\mathfrak r\in I_k .
\]
Thus the ordered-mass vectors are equal in the complementary case as well.
% lean: orderedMasses_eq_of_measureEquivalent
\end{enumerate}

The two cases prove
\[
\operatorname{orderedMasses}(\nu)=\operatorname{orderedMasses}(\xi),
\]
as required.
% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.orderedMasses_eq_of_measureEquivalent
\end{proof}
