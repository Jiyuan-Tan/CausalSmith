\begin{proof}[Proof of \cref{obj:lem:latent-mean-envelope}]
Fix \(u\in\{1,\ldots,k\}\) and \(t\in\{0,1\}\). Define
\[
\mathsf{Lat}_u:=\{U=u\},
\qquad
\mathsf{Cell}_{u,a}:=\{U=u,\ T=a\}\quad(a\in\{0,1\}).
\]
\begin{enumerate}
\item The condition in \cref{obj:ass:latent-arm-positivity}, together with \(\pi_0>0\), gives
\[
P(\mathsf{Cell}_{u,a})\ge \pi_0>0
\qquad\text{for each }a\in\{0,1\}.
\]
In particular \(P(\mathsf{Cell}_{u,0})>0\). Since
\[
\mathsf{Cell}_{u,0}\subseteq \mathsf{Lat}_u,
\]
we have
\[
P(\mathsf{Lat}_u)>0 .
\]
Let \(P_u:=P(\,\cdot\mid U=u)\). % lean: latentCell_pos_of_latentArmPositivity

\item We first bound the observed outcome on the latent treatment cell. Put
\[
\theta_z:=\frac{\sigma_0}{2\sqrt{d_z}}.
\]
Since \(2\le k\le d_z\) gives \(\sqrt{d_z}>0\), and since \(\sigma_0>0\), we have \(\theta_z>0\). The rank condition in \cref{obj:ass:proxy-rank-margin} gives
\[
\sigma_0\le \sigma_k(A_t).
\]
Applying the least-singular-value lower bound to the \(u\)th canonical vector gives
\[
\sigma_0
\le
\left(\sum_{j=1}^{d_z} A_t(j,u)^2\right)^{1/2}.
\]
Hence there exists \(j_\star\in\{1,\ldots,d_z\}\) such that
\[
\frac{\sigma_0}{\sqrt{d_z}}\le |A_t(j_\star,u)|.
\]
Let
\[
P_{u,t}^{\mathrm{cell}}:=P(\,\cdot\mid U=u,T=t),
\qquad
\mathsf{Ref}_{j_\star}:=\{|Z_{j_\star}|\ge \theta_z\}.
\]
Then
\[
P_{u,t}^{\mathrm{cell}}(\mathsf{Ref}_{j_\star})>0.
\]
Indeed, if this probability were zero, then
\[
|A_t(j_\star,u)|
=
\left|E_P[Z_{j_\star}\mid U=u,T=t]\right|
\le
E_P[|Z_{j_\star}|\mid U=u,T=t]
\le
\theta_z
<
\frac{\sigma_0}{\sqrt{d_z}},
\]
contradicting the displayed coordinate lower bound. % lean: exists_reference_coordinate_positive_event

\item Under \(P_{u,t}^{\mathrm{cell}}\), \cref{obj:ass:reference-proxy-separation} makes \(Z_{j_\star}\) independent of \(Y\). Also, \cref{obj:lem:proxy-coordinate-envelopes} supplies
\[
|Y Z_{j_\star}|\le L
\quad\text{almost surely under }P_{u,t}^{\mathrm{cell}}.
\]
Define
\[
\mathsf{Out}_{j_\star}
:=
\left\{|Y|>\frac{L}{\theta_z}\right\}.
\]
On \(\mathsf{Ref}_{j_\star}\cap\mathsf{Out}_{j_\star}\),
\[
|Y Z_{j_\star}|>L,
\]
and therefore
\[
P_{u,t}^{\mathrm{cell}}(\mathsf{Ref}_{j_\star}\cap\mathsf{Out}_{j_\star})=0.
\]
Independence gives
\[
0
=
P_{u,t}^{\mathrm{cell}}(\mathsf{Ref}_{j_\star})
P_{u,t}^{\mathrm{cell}}(\mathsf{Out}_{j_\star}).
\]
Since the first factor is positive,
\[
P_{u,t}^{\mathrm{cell}}(\mathsf{Out}_{j_\star})=0.
\]
Thus
\[
|Y|\le \frac{L}{\theta_z}
=
\frac{2L\sqrt{d_z}}{\sigma_0}
=
\frac{L_\tau}{2}
\quad\text{almost surely under }P(\,\cdot\mid U=u,T=t),
\]
where the last equality uses \cref{obj:def:effect-radius}. % lean: ae_abs_observedOutcome_le_on_latentCell

\item Now work under \(P_u=P(\,\cdot\mid U=u)\), and define
\[
\mathsf{Arm}_t:=\{T=t\}.
\]
The preceding positivity gives
\[
P_u(\mathsf{Arm}_t)
=
\frac{P(\mathsf{Cell}_{u,t})}{P(\mathsf{Lat}_u)}
>0.
\]
By \cref{obj:ass:latent-ignorability}, \(Y(t)\) is independent of \(T\), hence independent of \(\mathsf{Arm}_t\), under \(P_u\). On \(\mathsf{Arm}_t\), \cref{obj:ass:consistency} gives \(Y=Y(t)\) almost surely, and the cell bound from the previous step gives
\[
|Y(t)|\le \frac{L_\tau}{2}
\quad\text{almost surely on }\mathsf{Arm}_t\text{ under }P_u.
\]
With
\[
\mathsf{PotLarge}_t:=\left\{|Y(t)|>\frac{L_\tau}{2}\right\},
\]
we therefore have
\[
P_u(\mathsf{PotLarge}_t\cap\mathsf{Arm}_t)=0.
\]
Independence gives
\[
0
=
P_u(\mathsf{PotLarge}_t)P_u(\mathsf{Arm}_t),
\]
and the positive arm probability yields
\[
P_u(\mathsf{PotLarge}_t)=0.
\]
Consequently
\[
|Y(t)|\le \frac{L_\tau}{2}
\quad\text{almost surely under }P_u. % lean: ae_abs_potential_le_on_latentClass
\]

Finally, by \cref{obj:def:latent-mean},
\[
\mu_{tu}=E_{P_u}[Y(t)].
\]
The almost-sure bound just proved gives integrability and
\[
|\mu_{tu}|
=
\left|E_{P_u}[Y(t)]\right|
\le
E_{P_u}[|Y(t)|]
\le
\frac{L_\tau}{2}.
\]
This is the desired bound. % lean: latentMean_abs_le_of_model
\end{enumerate}
\end{proof}
