\begin{proof}[Proof of \cref{obj:thm:matching-local-lower-bounds}]
\begin{enumerate}
\item Define
\[
a:=\min\left\{\frac1{3200},\frac{\sqrt{c_{\mathrm{loc}}}}{3200}\right\},
\qquad
c:=\frac a8,
\qquad
C:=16000 .
\]
Since \(0<c_{\mathrm{loc}}<1\),
\[
0<a\le \frac1{3200}\le\frac18,\qquad c>0,\qquad C>0,
\]
and
\[
16000a^2\le c_{\mathrm{loc}},
\qquad
16000a^2\le \frac14\le\frac25 .
\]
Fix \(n\ge1\), and put
\[
\varepsilon_n:=\frac{a}{\sqrt n}.
\]
Then \(0\le\varepsilon_n\le1/8\), and
\[
16000\varepsilon_n^2
=
\frac{16000a^2}{n}
\le
\frac{c_{\mathrm{loc}}}{n}.
\]

Let
\[
\Omega_{\mathrm{vis}}:=\{0,1\}^4,
\qquad
\zeta=(t,x,z,y)\in\Omega_{\mathrm{vis}},
\]
and let
\[
\phi(t,x,z,y):=(t,(1,x)^\top,(1,z)^\top,y)\in O_{2,2},
\]
with \(O_{2,2}\) as in \cref{obj:def:observed-record-space}.  For \(q\in[0,1]\) write
\[
\operatorname{Bern}_q(1)=q,\qquad
\operatorname{Bern}_q(0)=1-q .
\]
For \(0\le\varepsilon\le1/8\), define the finite visible witness cell mass by
\[
\psi^{\mathrm{wit}}_{\varepsilon}(t,x,z,y)
=
\sum_{u=1}^{2}
p_u\,\operatorname{Bern}_{\theta_u}(t)\,
\operatorname{Bern}_{\beta_u}(x)\,
\operatorname{Bern}_{\alpha_{t,u}}(z)
\begin{cases}
\operatorname{Bern}_{1/4}(y),& t=0,\\
\operatorname{Bern}_{1/2+\operatorname{sgn}_u\varepsilon}(y),& t=1,
\end{cases}
\]
where
\[
(p_1,p_2)=\left(\frac25,\frac35\right),\quad
(\theta_1,\theta_2)=\left(\frac25,\frac35\right),\quad
(\beta_1,\beta_2)=\left(\frac15,\frac45\right),
\]
\[
(\alpha_{0,1},\alpha_{0,2})=\left(\frac3{10},\frac7{10}\right),\quad
(\alpha_{1,1},\alpha_{1,2})=\left(\frac7{20},\frac34\right),\quad
(\operatorname{sgn}_1,\operatorname{sgn}_2)=(-1,1).
\]
The pushforward of this finite law by \(\phi\) is \((P_{\varepsilon}^{\mathrm{wit}})_O\), the observed-data marginal of \cref{obj:def:observed-margin}.  At \(\varepsilon=0\), every visible cell is bounded below by the first-class contribution:
\[
\psi^{\mathrm{wit}}_{0}(\zeta)
\ge
\frac25\cdot\frac25\cdot\frac15\cdot\frac14\cdot\frac14
=
\frac1{500}
\ge
\frac1{1000}.
\]
Only the treated outcome factor varies with \(\varepsilon\), and hence
\[
\left|\psi^{\mathrm{wit}}_{\varepsilon}(\zeta)
-\psi^{\mathrm{wit}}_{0}(\zeta)\right|
\le \varepsilon .
\]
For Kullback--Leibler divergence as in \cref{obj:synth_6}, the elementary finite-carrier comparison is
\[
D_{\mathrm{KL}}(\mathfrak r\Vert\mathfrak s)
\le
\sum_{\zeta}\frac{(\mathfrak r(\zeta)-\mathfrak s(\zeta))^2}{\mathfrak s(\zeta)}
\]
whenever \(\mathfrak r\) and \(\mathfrak s\) are probability vectors on a finite carrier and
\(\mathfrak s(\zeta)>0\) for every \(\zeta\); this follows from \(\log x\le x-1\).  Therefore
\[
D_{\mathrm{KL}}\!\left((P_{\varepsilon}^{\mathrm{wit}})_O\Vert(P_0^{\mathrm{wit}})_O\right)
\le
\sum_{\zeta\in\Omega_{\mathrm{vis}}}
\frac{\{\psi^{\mathrm{wit}}_{\varepsilon}(\zeta)-\psi^{\mathrm{wit}}_{0}(\zeta)\}^2}
{\psi^{\mathrm{wit}}_{0}(\zeta)}
\le
16\cdot1000\,\varepsilon^2
=
16000\varepsilon^2,
\]
where the first inequality also uses data processing under \(\phi\).  Taking
\(\varepsilon=\varepsilon_n\), \cref{obj:prop:two-class-witness-valid} gives
\(P_0^{\mathrm{wit}},P_{\varepsilon_n}^{\mathrm{wit}}\in\mathcal M\), and the displayed KL bound places both laws in \(\mathcal L_n^\nu\) from \cref{obj:def:local-quotient-experiment}. % lean: witnessLaw_observed_kl_bound

\item We use the following two-point expected-risk inequality.  Let \(\mathsf R_1\ll\mathsf R_0\) be one-observation laws with integrable log-likelihood ratio and suppose
\[
nD_{\mathrm{KL}}(\mathsf R_1\Vert\mathsf R_0)\le\frac25 .
\]
Tensorization and Pinsker's inequality give
\[
\left\|\mathsf R_1^{\otimes n}-\mathsf R_0^{\otimes n}\right\|_{\mathrm{TV}}
\le
\sqrt{\frac12D_{\mathrm{KL}}(\mathsf R_1^{\otimes n}\Vert\mathsf R_0^{\otimes n})}
\le\frac12 .
\]
For an estimator \(\widehat\theta\) in a metric space with targets \(\Theta_1,\Theta_0\), set
\[
\Delta:=d(\Theta_1,\Theta_0),
\qquad
\mathcal B_1:=\left\{d(\widehat\theta,\Theta_1)\ge\frac{\Delta}{2}\right\},
\qquad
\mathcal B_0:=\left\{d(\widehat\theta,\Theta_0)\ge\frac{\Delta}{2}\right\}.
\]
The triangle inequality gives \(\mathcal B_1^{c}\subseteq\mathcal B_0\) when \(\Delta>0\), and the case \(\Delta=0\) is immediate.  Thus
\[
\mathsf R_1^{\otimes n}(\mathcal B_1)+\mathsf R_0^{\otimes n}(\mathcal B_0)
\ge
1-\left\|\mathsf R_1^{\otimes n}-\mathsf R_0^{\otimes n}\right\|_{\mathrm{TV}}
\ge\frac12,
\]
so one of the two probabilities is at least \(1/4\).  Since
\[
\frac{\Delta}{2}\,\Pr\!\left\{d(\widehat\theta,\Theta_i)\ge\frac{\Delta}{2}\right\}
\le
E\,d(\widehat\theta,\Theta_i),
\]
one of the two expected risks is at least \(\Delta/8\).  For simplex-valued estimators the same argument applied to one coordinate gives
\[
\frac{|p_1(j)-p_0(j)|}{8}
\le
E_{\mathsf R_1^{\otimes n}}\!\left[\|\widehat p-p_1\|_1\right]
\quad\text{or}\quad
\frac{|p_1(j)-p_0(j)|}{8}
\le
E_{\mathsf R_0^{\otimes n}}\!\left[\|\widehat p-p_0\|_1\right],
\]
because \(|\widehat p(j)-p_i(j)|\le\|\widehat p-p_i\|_1\).  The \(W_1\) loss is bounded on the stated compact support radius, and the simplex \(\ell^1\) loss is bounded by \(2\), so the expectations used below are finite. % lean: twoPoint_expectedMetricRisk_lower

\item Apply the two-point inequality with
\[
\mathsf R_1=(P_{\varepsilon_n}^{\mathrm{wit}})_O,
\qquad
\mathsf R_0=(P_0^{\mathrm{wit}})_O .
\]
The KL calculation gives
\[
nD_{\mathrm{KL}}(\mathsf R_1\Vert\mathsf R_0)
\le
16000a^2
\le
\frac25 .
\]
By \cref{obj:prop:two-class-witness-valid},
\[
\nu_{P_0^{\mathrm{wit}}}=\delta_{1/4},
\qquad
\nu_{P_{\varepsilon_n}^{\mathrm{wit}}}
=
\frac25\delta_{1/4-\varepsilon_n}
+
\frac35\delta_{1/4+\varepsilon_n}.
\]
Every coupling from the point mass at \(1/4\) to the second law transports total mass one over distance \(\varepsilon_n\).  Hence, using \cref{obj:def:wasserstein-distance},
\[
W_1\!\left(\nu_{P_0^{\mathrm{wit}}},\nu_{P_{\varepsilon_n}^{\mathrm{wit}}}\right)
=
\varepsilon_n
=
\frac a{\sqrt n}.
\]
Thus every measurable quotient-law estimator \(\widetilde\nu_n\) has one of the two local witness laws satisfying
\[
E_{Q_P^{(n)}}\!\left[W_1(\widetilde\nu_n,\nu_P)\right]
\ge
\frac{\varepsilon_n}{8}
=
\frac{c}{\sqrt n}.
\]
This proves the quotient-law risk assertion. % lean: witness_quotientLaw_wass1

\item Fix \(g\in(0,1/4]\), and define
\[
h:=h(n,g):=a\min\left\{1,\frac1{\sqrt n\,g}\right\},
\]
the calibrated displacement of \cref{obj:synth_31}.  Since \(n\ge1\) and \(g>0\),
\[
x_{n,g}:=\sqrt n\,g>0,
\qquad
0<\min\{1,x_{n,g}^{-1}\}\le1.
\]
Together with \(a>0\), this gives
\[
0<h\le a\le\frac1{3200}\le\frac1{100},
\qquad
|h|\le\frac1{100}.
\]
In particular \(h\in[-1,1]\), so \(h\) lies in the tangent-amplitude domain.  Moreover,
\[
x_{n,g}\min\{1,x_{n,g}^{-1}\}\le1,
\]
and therefore
\[
n g^2 h^2
=
a^2\left(x_{n,g}\min\{1,x_{n,g}^{-1}\}\right)^2
\le
a^2 .
\]
Consequently
\[
16000g^2h^2\le\frac{c_{\mathrm{loc}}}{n}.
\]
The hypotheses of \cref{obj:lem:path-local-weight-experiment} are satisfied for \(P_{g,h}\), and also for \(P_{g,0}\).  Hence
\[
P_{g,h},P_{g,0}\in\mathcal L_{n,g}^{p}.
\] % lean: calibratedDisplacement_sample_signal_sq_le

\item We prove the labelled-path KL and ordered-mass certificates.  For \((t,x,z,y)\in\Omega_{\mathrm{vis}}\), define
\[
\psi^{\mathrm{path}}_{g,h}(t,x,z,y)
=
\begin{cases}
\displaystyle
\sum_{u=1}^{2}\sum_{y_0=0}^{1}
P_{g,h}(U=u,T=1,X_2=x,Z_2=z,Y(0)=y_0,Y(1)=y),
& t=1,\\[1.2em]
\displaystyle
\sum_{u=1}^{2}\sum_{y_1=0}^{1}
P_{g,h}(U=u,T=0,X_2=x,Z_2=z,Y(0)=y,Y(1)=y_1),
& t=0.
\end{cases}
\]
The pushforward of this finite law by \(\phi\) is \((P_{g,h})_O\).  At \(h=0\), \cref{obj:synth_14} gives \(P_{g,0}=P_{g/2}^{\mathrm{wit}}\).  For \(t=0\), the first latent class contributes at least
\[
\frac6{25}\cdot\frac15\cdot\frac3{10}\cdot\frac14
=
\frac{18}{5000}
\ge
\frac1{1000}.
\]
For \(t=1\), because \(0\le g\le1/4\), both treated outcome probabilities lie in \([3/8,5/8]\), and the first latent class contributes at least
\[
\frac4{25}\cdot\frac15\cdot\frac7{20}\cdot\frac38
=
\frac{21}{5000}
\ge
\frac1{1000}.
\]
Therefore
\[
\psi^{\mathrm{path}}_{g,0}(t,x,z,y)\ge\frac1{1000}
\qquad
\text{for all }(t,x,z,y)\in\Omega_{\mathrm{vis}} .
\]

For \(|h|\le1/100\), all denominators appearing in \cref{obj:synth_14} are bounded away from zero; for example
\[
2/5+h\ge39/100,\qquad
3/5-h\ge59/100,\qquad
15h+4\ge77/20,\qquad
5h+3\ge59/20 .
\]
Substituting the displayed path probabilities of \cref{obj:synth_14}, summing out the inactive potential outcome, and reducing the four \((x,z)\) cases gives
\[
\sum_{u=1}^{2}P_{g,h}(U=u,T=1,X_2=x,Z_2=z)
=
\sum_{u=1}^{2}P_{g,0}(U=u,T=1,X_2=x,Z_2=z),
\]
and
\[
\begin{aligned}
&\sum_{u=1}^{2}\operatorname{sgn}_u
P_{g,h}(U=u,T=1,X_2=x,Z_2=z)
-
\sum_{u=1}^{2}\operatorname{sgn}_u
P_{g,0}(U=u,T=1,X_2=x,Z_2=z)  \\
&\hspace{3em}=
-2\kappa_{xz}h,
\end{aligned}
\]
where \(\operatorname{sgn}_1=-1\), \(\operatorname{sgn}_2=1\), and
\[
\kappa_{11}=\frac9{100},\qquad
\kappa_{10}=\frac3{100},\qquad
\kappa_{01}=\frac9{25},\qquad
\kappa_{00}=\frac3{25}.
\]
Since the treated outcome mass is
\[
\operatorname{Bern}_{1/2+\operatorname{sgn}_u g/2}(y)
=
\frac12+(2y-1)\frac{\operatorname{sgn}_u g}{2},
\]
the visible-cell displacement is
\[
\psi^{\mathrm{path}}_{g,h}(t,x,z,y)-\psi^{\mathrm{path}}_{g,0}(t,x,z,y)
=
\mathbf 1\{t=1\}(1-2y)\kappa_{xz}gh .
\]
Thus
\[
\left|\psi^{\mathrm{path}}_{g,h}(t,x,z,y)-\psi^{\mathrm{path}}_{g,0}(t,x,z,y)\right|
\le g|h|.
\]
The same finite KL--chi-square bound, now applied with the KL divergence of \cref{obj:synth_6}, and data processing under \(\phi\) give
\[
D_{\mathrm{KL}}\!\left((P_{g,h})_O\Vert(P_{g,0})_O\right)
\le
\sum_{\zeta\in\Omega_{\mathrm{vis}}}
\frac{\{\psi^{\mathrm{path}}_{g,h}(\zeta)-\psi^{\mathrm{path}}_{g,0}(\zeta)\}^2}
{\psi^{\mathrm{path}}_{g,0}(\zeta)}
\le
16000g^2h^2
=
C g^2h^2.
\]

The latent effects on the path are
\[
\tau_1(P_{g,h})=\frac14-\frac g2,
\qquad
\tau_2(P_{g,h})=\frac14+\frac g2,
\]
so their order is fixed for \(g>0\).  Also \(|h|\le1/100\) gives
\[
\frac25+h>0,\qquad \frac35-h>0.
\]
Using \cref{obj:def:ordered-mass-target},
\[
p^{\uparrow}(P_{g,h})=\left(\frac25+h,\frac35-h\right),
\qquad
p^{\uparrow}(P_{g,0})=\left(\frac25,\frac35\right),
\]
and hence
\[
\sum_i\left|p_i^{\uparrow}(P_{g,h})-p_i^{\uparrow}(P_{g,0})\right|
=
|h|+|-h|
=
2|h|.
\] % lean: pathLaw_observed_kl_bound

\item Apply the coordinate form of the two-point inequality to
\[
\mathsf R_1=(P_{g,h})_O,
\qquad
\mathsf R_0=(P_{g,0})_O,
\]
and to the ordered coordinate corresponding to the lower effect.  The preceding bounds give
\[
nD_{\mathrm{KL}}(\mathsf R_1\Vert\mathsf R_0)
\le
16000\,n g^2h^2
\le
16000a^2
\le
\frac25,
\]
and
\[
\left|p^{\uparrow}_1(P_{g,h})-p^{\uparrow}_1(P_{g,0})\right|=h .
\]
Therefore every measurable simplex-valued ordered-weight estimator \(\widetilde p_n\) has one of the two local laws satisfying
\[
E_{Q_P^{(n)}}\!\left[\|\widetilde p_n-p^{\uparrow}(P)\|_1\right]
\ge
\frac h8
=
\frac a8\min\left\{1,\frac1{\sqrt n\,g}\right\}
=
c\min\left\{1,\frac1{\sqrt n\,g}\right\}.
\]
The quotient-law certificate follows from the witness KL calculation with \(\varepsilon=\varepsilon_n\):
\[
D_{\mathrm{KL}}\!\left((P_{a/\sqrt n}^{\mathrm{wit}})_O\Vert(P_0^{\mathrm{wit}})_O\right)
\le
16000\frac{a^2}{n}
\le
\frac{16000}{n}
=
\frac Cn,
\]
and from
\[
\frac c{\sqrt n}
\le
\frac a{\sqrt n}
=
W_1\!\left(\nu_{P_0^{\mathrm{wit}}},\nu_{P_{a/\sqrt n}^{\mathrm{wit}}}\right).
\]
The labelled-path membership, KL bound, tangent-amplitude statement, and ordered-mass identity were established above for every \(g\in(0,1/4]\).  All four asserted conclusions hold with the constants \(a,c,C\) fixed at the start. % lean: matching_local_lower_bounds
\end{enumerate}
\end{proof}
