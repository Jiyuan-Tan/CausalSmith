\begin{proof}[Proof of \cref{obj:prop:two-class-witness-valid}]
\begin{enumerate}
\item Fix \(0\le \varepsilon\le 1/8\).  For \(\mathfrak b\in\{0,1\}\), write
\[
\operatorname{Ber}_{\theta}(\mathfrak b)
=
\begin{cases}
\theta,& \mathfrak b=1,\\
1-\theta,& \mathfrak b=0.
\end{cases}
\]
The law in \cref{obj:def:two-class-witness} is the finite law on points
\[
X=(1,\mathfrak x)^\top,\qquad
Z=(1,\mathfrak z)^\top,\qquad
Y(0)=\mathfrak y_0,\qquad
Y(1)=\mathfrak y_1,\qquad
Y=Y(T),
\]
where \(\mathfrak t,\mathfrak x,\mathfrak z,\mathfrak y_0,\mathfrak y_1\in\{0,1\}\).  Its elementary mass at
\((u,\mathfrak t,\mathfrak x,\mathfrak z,\mathfrak y_0,\mathfrak y_1)\), for \(u\in\{1,2\}\), is
\[
\theta^U_u
\operatorname{Ber}_{\theta^T_u}(\mathfrak t)
\operatorname{Ber}_{\theta^X_u}(\mathfrak x)
\operatorname{Ber}_{\theta^Z_{u,\mathfrak t}}(\mathfrak z)
\operatorname{Ber}_{1/4}(\mathfrak y_0)
\operatorname{Ber}_{\theta^1_u}(\mathfrak y_1),
\]
with
\[
(\theta^U_1,\theta^U_2)=\left(\frac25,\frac35\right),\quad
(\theta^T_1,\theta^T_2)=\left(\frac25,\frac35\right),\quad
(\theta^X_1,\theta^X_2)=\left(\frac15,\frac45\right),
\]
\[
\theta^Z_{1,0}=\frac3{10},\quad
\theta^Z_{2,0}=\frac7{10},\quad
\theta^Z_{1,1}=\frac7{20},\quad
\theta^Z_{2,1}=\frac34,\quad
\theta^1_1=\frac12-\varepsilon,\quad
\theta^1_2=\frac12+\varepsilon .
\]
The perturbation range gives
\[
\frac38\le \frac12-\varepsilon\le \frac12,
\qquad
\frac12\le \frac12+\varepsilon\le \frac58,
\]
so every displayed Bernoulli factor is a probability mass function.  Summing successively over
\(\mathfrak y_1,\mathfrak y_0,\mathfrak z,\mathfrak x,\mathfrak t\) leaves \(\theta^U_u\), and
\(\theta^U_1+\theta^U_2=1\).  Thus \(P_{\varepsilon}^{\mathrm{wit}}\) is a probability law. % lean: witnessLaw_isProbabilityMeasure

\item The parameter domain in \cref{obj:def:model-class} is satisfied because
\[
2\le 2,\qquad 2\le 2,\qquad 2\le 2,\qquad
1\le 2,\qquad
0<\frac1{10}\le \frac1{2\cdot2},\qquad
0<\frac1{10}\le 1 .
\]
The finite product form gives the conditional factorization requirements.  For bounded measurable
test functions \(\psi_Z,\psi_{XY},\psi_X,\psi_{YT},\psi_{Y_t},\psi_T\),
\[
E[\psi_Z(Z)\psi_{XY}(X,Y)\mid U=u,T=t]
=
E[\psi_Z(Z)\mid U=u,T=t]\,
E[\psi_{XY}(X,Y)\mid U=u,T=t],
\]
\[
E[\psi_X(X)\psi_{YT}(Y,T)\mid U=u]
=
E[\psi_X(X)\mid U=u]\,
E[\psi_{YT}(Y,T)\mid U=u],
\]
and
\[
E[\psi_{Y_t}(Y(t))\psi_T(T)\mid U=u]
=
E[\psi_{Y_t}(Y(t))\mid U=u]\,
E[\psi_T(T)\mid U=u].
\]
These identities give \cref{obj:ass:reference-proxy-separation,obj:ass:target-proxy-separation,obj:ass:latent-ignorability}.  The construction gives \(Y=Y(T)\) and \(X_1=1\) almost surely, which are \cref{obj:ass:consistency,obj:ass:anchor}.  Since \(X,Z\in\{(1,0)^\top,(1,1)^\top\}\) and \(Y\in\{0,1\}\),
\[
\|X\|_2\le \sqrt2\le 2,\qquad
\|ZX^\top\|_{\mathrm{op}}\le \|Z\|_2\|X\|_2\le 2,\qquad
\|YZX^\top\|_{\mathrm{op}}\le 2,
\]
so \cref{obj:ass:bounded-x,obj:ass:bounded-proxy-product,obj:ass:bounded-outcome-proxy-product} hold with \(L=2\).  The latent-arm masses are
\[
P_{\varepsilon}^{\mathrm{wit}}(U=1,T=0)=\frac6{25},\quad
P_{\varepsilon}^{\mathrm{wit}}(U=1,T=1)=\frac4{25},\quad
P_{\varepsilon}^{\mathrm{wit}}(U=2,T=0)=\frac6{25},\quad
P_{\varepsilon}^{\mathrm{wit}}(U=2,T=1)=\frac9{25},
\]
and each is at least \(1/10\), giving \cref{obj:ass:latent-arm-positivity}.  The feature matrices are
\[
B=\begin{pmatrix}1&1\\[2pt]1/5&4/5\end{pmatrix},\qquad
A_0=\begin{pmatrix}1&1\\[2pt]3/10&7/10\end{pmatrix},\qquad
A_1=\begin{pmatrix}1&1\\[2pt]7/20&3/4\end{pmatrix}.
\]
For every \(\upsilon=(\upsilon_1,\upsilon_2)\in\mathbb R^2\),
\[
\frac1{100}\|\upsilon\|_2^2\le \|B\upsilon\|_2^2,\qquad
\frac1{100}\|\upsilon\|_2^2\le \|A_0\upsilon\|_2^2,\qquad
\frac1{100}\|\upsilon\|_2^2\le \|A_1\upsilon\|_2^2.
\]
Indeed, after multiplying the three differences by \(100,100,400\), respectively, the differences equal
\[
103\left(\upsilon_1+\frac{116}{103}\upsilon_2\right)^2
+\frac{3333}{103}\upsilon_2^2,
\]
\[
108\left(\upsilon_1+\frac{121}{108}\upsilon_2\right)^2
+\frac{1343}{108}\upsilon_2^2,
\]
and
\[
445\left(\upsilon_1+\frac{505}{445}\upsilon_2\right)^2
+\frac{4264}{89}\upsilon_2^2.
\]
Thus the smallest singular value of each of \(B,A_0,A_1\) is at least \(1/10\), proving \cref{obj:ass:proxy-rank-margin}.  Hence
\[
P_{\varepsilon}^{\mathrm{wit}}\in\mathcal M
\quad\text{with}\quad
L=2,\quad \pi_0=\frac1{10},\quad \sigma_0=\frac1{10}.
\] % lean: witness_ucvmwModel

\item \Cref{obj:lem:witness-observed-arm-mass} gives
\[
P_{\varepsilon}^{\mathrm{wit}}(T=0)=\frac{12}{25}=0.48,\qquad
P_{\varepsilon}^{\mathrm{wit}}(T=1)=\frac{13}{25}=0.52 .
\]
Also, \cref{obj:lem:witness-target-feature-determinant,obj:lem:witness-reference-feature-determinant} give
\[
\det(B)=\frac35=0.6,\qquad
\det(A_0)=\det(A_1)=\frac25=0.4 .
\]
It remains to compute the observable proxy moment determinants.  For \(\iota,\jmath\in\{1,2\}\) and
\(\mathfrak t\in\{0,1\}\), put
\[
x_1(\mathfrak x)=1,\qquad x_2(\mathfrak x)=\mathfrak x,\qquad
z_1(\mathfrak z)=1,\qquad z_2(\mathfrak z)=\mathfrak z .
\]
After summing out \(\mathfrak y_0\) and \(\mathfrak y_1\), \cref{obj:def:observable-population-moments} gives the conditional-normalized finite sum
\[
M_{\mathfrak t}(P_{\varepsilon}^{\mathrm{wit}})_{\iota\jmath}
=
\frac{
\sum_{u=1}^{2}\sum_{\mathfrak x,\mathfrak z\in\{0,1\}}
\theta^U_u\operatorname{Ber}_{\theta^T_u}(\mathfrak t)
\operatorname{Ber}_{\theta^X_u}(\mathfrak x)
\operatorname{Ber}_{\theta^Z_{u,\mathfrak t}}(\mathfrak z)
z_\iota(\mathfrak z)x_\jmath(\mathfrak x)}
{\sum_{u=1}^{2}\theta^U_u\operatorname{Ber}_{\theta^T_u}(\mathfrak t)} .
\]
Since the \(\mathfrak x\)- and \(\mathfrak z\)-sums are Bernoulli means, this is
\[
M_{\mathfrak t}(P_{\varepsilon}^{\mathrm{wit}})_{\iota\jmath}
=
\sum_{u=1}^{2}
P_{\varepsilon}^{\mathrm{wit}}(U=u\mid T=\mathfrak t)\,
A_{\mathfrak t}(\iota,u)B(\jmath,u).
\]
For \(\mathfrak t=0\), the conditional latent weights are \((1/2,1/2)\), and for
\(\mathfrak t=1\) they are \((4/13,9/13)\).  Therefore
\[
M_0(P_{\varepsilon}^{\mathrm{wit}})
=
\begin{pmatrix}
1&1/2\\[2pt]
1/2&31/100
\end{pmatrix},
\qquad
M_1(P_{\varepsilon}^{\mathrm{wit}})
=
\begin{pmatrix}
1&8/13\\[2pt]
163/260&142/325
\end{pmatrix}.
\]
Taking \(2\times2\) determinants,
\[
\det(M_0(P_{\varepsilon}^{\mathrm{wit}}))
=
\frac{31}{100}-\frac14
=
\frac3{50}
=
0.06,
\]
and
\[
\det(M_1(P_{\varepsilon}^{\mathrm{wit}}))
=
\frac{142}{325}-\frac8{13}\frac{163}{260}
=
\frac{216}{4225}
=
\frac25\cdot\frac35\cdot\frac{36}{169}.
\] % lean: witness_observedProxyMoment_det

\item \Cref{obj:lem:witness-latent-mass} gives
\[
p_1(P_{\varepsilon}^{\mathrm{wit}})=\frac25,\qquad
p_2(P_{\varepsilon}^{\mathrm{wit}})=\frac35,
\]
and \cref{obj:lem:witness-latent-effect} gives
\[
\tau_1(P_{\varepsilon}^{\mathrm{wit}})=\frac14-\varepsilon,\qquad
\tau_2(P_{\varepsilon}^{\mathrm{wit}})=\frac14+\varepsilon .
\]
Since \(\nu_P\) in \cref{obj:def:quotient-law} places latent mass \(p_u\) at latent effect \(\tau_u\),
\[
\nu_{P_{\varepsilon}^{\mathrm{wit}}}
=
\frac25\,\delta_{1/4-\varepsilon}
+
\frac35\,\delta_{1/4+\varepsilon}
=
0.4\,\delta_{0.25-\varepsilon}
+
0.6\,\delta_{0.25+\varepsilon}.
\] % lean: quotientLaw

\item At \(\varepsilon=0\), the singular-value inequalities in the second step give injectivity, hence
nonsingularity, of \(B,A_0,A_1\).  The determinant identities in the third step give nonsingularity of
\(M_0(P_0^{\mathrm{wit}})\) and \(M_1(P_0^{\mathrm{wit}})\).  The quotient-law identity in the fourth step becomes
\[
\nu_{P_0^{\mathrm{wit}}}
=
\frac25\,\delta_{1/4}+\frac35\,\delta_{1/4}
=
\delta_{1/4}
=
\delta_{0.25}.
\]
Finally, \cref{obj:lem:witness-latent-mass} gives
\[
p_1(P_0^{\mathrm{wit}})=\frac25,\qquad
p_2(P_0^{\mathrm{wit}})=\frac35,
\]
and these two latent-class masses are distinct. % lean: two_class_witness_valid
\end{enumerate}
\end{proof}
