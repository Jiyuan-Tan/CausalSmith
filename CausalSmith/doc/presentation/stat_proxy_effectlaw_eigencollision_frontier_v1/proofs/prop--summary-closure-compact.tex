\begin{proof}[Proof of \cref{obj:prop:summary-closure-compact}]
\begin{enumerate}
\item First record the coordinate estimate used to pass from operator envelopes to coordinate boxes.  For any \(H\in\mathbb R^{d_z\times d_x}\), any \(\iota_z\in\{1,\ldots,d_z\}\), and any \(\iota_x\in\{1,\ldots,d_x\}\),
\[
        |H_{\iota_z\iota_x}|
        =
        \|(H e_{\iota_x})_{\iota_z}\|
        \le
        \|H e_{\iota_x}\|_2
        \le
        \|H\|_{\mathrm{op}}\|e_{\iota_x}\|_2
        =
        \|H\|_{\mathrm{op}},
\]
where \(e_{\iota_x}\) is the \(\iota_x\)th canonical basis vector of \(\mathbb R^{d_x}\). % lean: matrixEntry_abs_le_operatorNorm

\item Define the finite coordinate boxes
\[
\mathcal H_{\mathrm{mat}}(L)
=
\Bigl\{H\in\mathbb R^{d_z\times d_x}:
        -L\le H_{\iota_z\iota_x}\le L
        \text{ for all }\iota_z,\iota_x\Bigr\},
\]
\[
\mathcal H_X(L)
=
\Bigl\{h\in\mathbb R^{d_x}:
        -L\le h_{\iota_x}\le L
        \text{ for all }\iota_x\Bigr\},
\]
and
\[
\mathcal H_S(L)
=
\mathcal H_{\mathrm{mat}}(L)^4\times \mathcal H_X(L).
\]
Each factor is a finite product of closed intervals, hence compact, and therefore \(\mathcal H_S(L)\) is compact.  For a five-block summary \(s=(M_0,M_1,N_0,N_1,m_X)\), define the coordinate map
\[
        \Phi(s)=(M_0,M_1,N_0,N_1,m_X).
\]
This map is a homeomorphism from the space of five-block summaries onto this finite product coordinate space.  Thus
\[
        \Phi^{-1}\bigl(\mathcal H_S(L)\bigr)
\]
is compact in the summary topology. % lean: summarySpaceBox_compact

\item Let \(P\in\mathcal M\).  For \(t\in\{0,1\}\), put
\[
        \mathsf{Arm}_t(P)=\{T=t\}.
\]
By \cref{obj:ass:latent-arm-positivity}, together with \(2\le k\) and \(0<\pi_0\), the arm \(\mathsf{Arm}_t(P)\) has positive mass.  Let \(P^{\mathrm{arm}}_t\) be the probability law obtained by restricting \(P\) to \(\mathsf{Arm}_t(P)\) and normalizing by this positive mass.  By \cref{obj:def:observable-population-moments}, the armwise blocks in \(S(P)\) are conditional means; on this positive arm, equivalently,
\[
        M_t(P)=\int ZX^\top\,dP^{\mathrm{arm}}_t,
        \qquad
        N_t(P)=\int YZX^\top\,dP^{\mathrm{arm}}_t,
\]
and \(m_X(P)=\int X\,dP\).  By \cref{obj:ass:bounded-proxy-product}, the proxy-product envelope restricts to \(P^{\mathrm{arm}}_t\): \(\|ZX^\top\|_{\mathrm{op}}\le L\) \(P^{\mathrm{arm}}_t\)-almost surely.  Since \(1\le L\), the matrix-valued integral is integrable and
\[
\begin{aligned}
\|M_t(P)\|_{\mathrm{op}}
&=\left\|\int ZX^\top\,dP^{\mathrm{arm}}_t\right\|_{\mathrm{op}}  \\
&\le \int \|ZX^\top\|_{\mathrm{op}}\,dP^{\mathrm{arm}}_t
 \le \int L\,dP^{\mathrm{arm}}_t
 = L .
\end{aligned}
\]
The same normalized-restriction argument with \cref{obj:ass:bounded-outcome-proxy-product} gives
\[
        \|N_t(P)\|_{\mathrm{op}}\le L .
\]
Likewise, \cref{obj:ass:bounded-x} gives
\[
\|m_X(P)\|_2
=\left\|\int X\,dP\right\|_2
\le
\int \|X\|_2\,dP
\le L .
\]
Hence every coordinate of \(m_X(P)\) has absolute value at most \(L\).  Combining these inequalities with the coordinate estimate from the first step and the summary tuple in \cref{obj:def:observable-summary} gives
\[
        S(P)\in \Phi^{-1}\bigl(\mathcal H_S(L)\bigr).
\]
Since \(P\in\mathcal M\) was arbitrary,
\[
        \mathscr S\subseteq \Phi^{-1}\bigl(\mathcal H_S(L)\bigr).
\] % lean: admissibleImage_subset_summarySpaceBox

\item The same block envelopes give the stated uniform \(d_S\)-bound.  For \(s=S(P)\in\mathscr S\),
\[
\begin{aligned}
d_S(s,0)
&=
\|M_0(P)\|_{\mathrm{op}}
+\|M_1(P)\|_{\mathrm{op}}
+\|N_0(P)\|_{\mathrm{op}}
+\|N_1(P)\|_{\mathrm{op}}
+\Bigl(\sum_{\iota_x} m_X(P)_{\iota_x}^2\Bigr)^{1/2}.
\end{aligned}
\]
For the final term,
\[
\Bigl(\sum_{\iota_x} m_X(P)_{\iota_x}^2\Bigr)^{1/2}
=
\Bigl\|\sum_{\iota_x} m_X(P)_{\iota_x} e_{\iota_x}\Bigr\|_2
\le
\sum_{\iota_x}|m_X(P)_{\iota_x}|
\le
d_x L.
\]
Therefore
\[
        d_S(S(P),0)\le 4L+d_xL=(4+d_x)L.
\]
Taking the scalar bound
\[
        B_{\mathrm{sum}}=(4+d_x)L
\]
and then instantiating the existential bound with \(B_{\mathrm{sum}}\) proves the boundedness claim. % lean: admissibleImage_dS_bounded

\item Since \(\Phi^{-1}(\mathcal H_S(L))\) is compact and closed and contains \(\mathscr S\), it also contains
\[
        \mathcal K=\overline{\mathscr S}.
\]
The set \(\mathcal K\) is closed by definition as a closure, so it is a closed subset of the compact set \(\Phi^{-1}(\mathcal H_S(L))\).  Hence \(\mathcal K\) is compact. % lean: summaryClosure_compact_of_block_bounds

\item Finally,
\[
        \mathcal K\neq\varnothing
        \quad\Longleftrightarrow\quad
        \mathscr S\neq\varnothing,
\]
because a closure has the same nonemptiness as the set it closes.  By the definition
\[
        \mathscr S=\{S(P):P\in\mathcal M\},
\]
the condition \(\mathscr S\neq\varnothing\) is equivalent to \(\mathcal M\neq\varnothing\): a law \(P\in\mathcal M\) gives \(S(P)\in\mathscr S\), and any element of \(\mathscr S\) is \(S(P)\) for some \(P\in\mathcal M\).  Thus \(\mathcal K\) is nonempty if and only if \(\mathcal M\) is nonempty. % lean: summary_closure_compact
\end{enumerate}
\end{proof}
