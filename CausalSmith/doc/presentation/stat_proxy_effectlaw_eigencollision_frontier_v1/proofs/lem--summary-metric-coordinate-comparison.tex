\begin{proof}[Proof of \cref{obj:lem:summary-metric-coordinate-comparison}]
Write
\[
\varrho:=\rho_{\mathrm{sum}}(s,s').
\]
For \(d\in\mathbb N\), put
\[
[d]_0:=\{0,1,\ldots,d-1\};
\]
when \(d=0\), this set is empty and the finite sums below have value zero.  Write
\[
s=(s_1,s_2,s_3,s_4,s_5),
\qquad
s'=(s'_1,s'_2,s'_3,s'_4,s'_5),
\]
where \(s_1,\ldots,s_4\) and \(s'_1,\ldots,s'_4\) are \(d_z\times d_x\) matrix
blocks and \(s_5,s'_5\in\mathbb R^{d_x}\), as in
\cref{obj:def:observable-summary}.

\begin{enumerate}
\item Since \(\rho_{\mathrm{sum}}\) is the Euclidean distance on the full
coordinate vector, each coordinate-block projection is nonexpansive.  Hence,
for each \(\ell=1,2,3,4\),
\[
\left(
  \sum_{\mathfrak i\in[d_z]_0,\ \mathfrak j\in[d_x]_0}
  \bigl((s_\ell)_{\mathfrak i\mathfrak j}
        -(s'_\ell)_{\mathfrak i\mathfrak j}\bigr)^2
\right)^{1/2}
\le \varrho,
\]
and
\[
\left(
  \sum_{\mathfrak j\in[d_x]_0}
  \bigl((s_5)_{\mathfrak j}-(s'_5)_{\mathfrak j}\bigr)^2
\right)^{1/2}
\le \varrho .
\]
In particular,
\[
\bigl|(s_\ell)_{\mathfrak i\mathfrak j}
      -(s'_\ell)_{\mathfrak i\mathfrak j}\bigr|
\le \varrho
\quad
(\ell=1,2,3,4;\ \mathfrak i\in[d_z]_0,\ \mathfrak j\in[d_x]_0),
\]
and
\[
\bigl|(s_5)_{\mathfrak j}-(s'_5)_{\mathfrak j}\bigr|
\le \varrho
\quad(\mathfrak j\in[d_x]_0).
\]
% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.dS_le_dist_mul

\item We use the following finite-dimensional entrywise estimate.  Let
\(\Psi\in\mathbb R^{r\times c}\), let \(\zeta\in\mathbb R\) satisfy
\(0\le\zeta\), and suppose
\[
|\Psi_{\mathfrak i\mathfrak j}|\le \zeta
\quad
(\mathfrak i\in[r]_0,\ \mathfrak j\in[c]_0).
\]
For the canonical vector \(e^{(c)}_{\mathfrak j}\in\mathbb R^c\),
\[
\|\Psi e^{(c)}_{\mathfrak j}\|_2
\le
\sum_{\mathfrak i\in[r]_0}|\Psi_{\mathfrak i\mathfrak j}|
\le r\,\zeta .
\]
Thus, for every \(\upsilon\in\mathbb R^c\) with \(\|\upsilon\|_2\le1\),
\[
\begin{aligned}
\|\Psi\upsilon\|_2
&=
\left\|
\sum_{\mathfrak j\in[c]_0}
(J_c\upsilon)_{\mathfrak j}\,\Psi e^{(c)}_{\mathfrak j}
\right\|_2  \\
&\le
\sum_{\mathfrak j\in[c]_0}
|(J_c\upsilon)_{\mathfrak j}|\,
\|\Psi e^{(c)}_{\mathfrak j}\|_2  \\
&\le
c\,r\,\|J_c\|_{\mathrm{op}}\,\zeta .
\end{aligned}
\]
Using the ordinary matrix action and operator-norm convention in
\cref{obj:synth_30,obj:synth_35}, taking the supremum over the Euclidean
unit ball gives
\[
\|\Psi\|_{\mathrm{op}}
\le
\operatorname{Ent}_{r,c}\,\zeta,
\qquad
\operatorname{Ent}_{r,c}=c\,r\,\|J_c\|_{\mathrm{op}} .
\]
% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.matrixNorm_le_entryBound

\item Apply the preceding estimate with
\[
r=d_z,\qquad c=d_x,\qquad \zeta=\varrho
\]
to the four matrix differences \(s_\ell-s'_\ell\), \(\ell=1,2,3,4\).  The
coordinate bounds from the first step yield
\[
\|s_\ell-s'_\ell\|_{\mathrm{op}}
\le
\operatorname{Ent}_{d_z,d_x}\,\varrho,
\qquad \ell=1,2,3,4 .
\]
% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.dS_le_dist_mul

\item For the mean block, expand in the canonical basis of \(\mathbb R^{d_x}\):
\[
s_5-s'_5
=
\sum_{\mathfrak j\in[d_x]_0}
\bigl((s_5)_{\mathfrak j}-(s'_5)_{\mathfrak j}\bigr)e^{(d_x)}_{\mathfrak j}.
\]
The coordinate bounds and the triangle inequality give
\[
\|s_5-s'_5\|_2
\le
\sum_{\mathfrak j\in[d_x]_0}
\bigl|(s_5)_{\mathfrak j}-(s'_5)_{\mathfrak j}\bigr|
\le
d_x\,\varrho .
\]
% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.eucNorm_le_card_mul_bound

\item Substituting the four matrix-block estimates and the mean-block estimate
into the definition of \(d_S\) in \cref{obj:def:observable-summary} gives
\[
\begin{aligned}
d_S(s,s')
&=
\sum_{\ell=1}^{4}\|s_\ell-s'_\ell\|_{\mathrm{op}}
+\|s_5-s'_5\|_2 \\
&\le
4\,\operatorname{Ent}_{d_z,d_x}\,\varrho+d_x\,\varrho \\
&=
\bigl(4\operatorname{Ent}_{d_z,d_x}+d_x\bigr)
\rho_{\mathrm{sum}}(s,s') .
\end{aligned}
\]
% lean: CausalSmith.Stat.ProxyEffectlawEigencollisionFrontier.dS_le_dist_mul
\end{enumerate}
\end{proof}
