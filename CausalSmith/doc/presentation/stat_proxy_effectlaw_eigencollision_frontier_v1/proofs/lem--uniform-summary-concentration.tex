\begin{proof}[Proof of \cref{obj:lem:uniform-summary-concentration}]
Fix \(k,d_x,d_z,\pi_0,\sigma_0\) satisfying the displayed hypotheses.  Let
\(\mathcal I_{d_x,d_z}\) be the finite disjoint union of the following scalar
summary coordinates:
\[
\{\operatorname{prox},\operatorname{out}\}\times\{0,1\}
\times\{1,\ldots,d_z\}\times\{1,\ldots,d_x\},
\qquad
\operatorname{xcoord}\times\{1,\ldots,d_x\},
\qquad
\operatorname{arm}\times\{0,1\}.
\]
For \(O=(T,X,Z,Y)\in O_{d_x,d_z}\) from
\cref{obj:def:observed-record-space}, define
\[
\varphi_{\operatorname{prox},t,\mathfrak z,\mathfrak x}(O)
:=
\mathbf 1\{T=t\}Z_{\mathfrak z}X_{\mathfrak x},
\qquad
\varphi_{\operatorname{out},t,\mathfrak z,\mathfrak x}(O)
:=
\mathbf 1\{T=t\}YZ_{\mathfrak z}X_{\mathfrak x},
\]
\[
\varphi_{\operatorname{xcoord},\mathfrak x}(O):=X_{\mathfrak x},
\qquad
\varphi_{\operatorname{arm},t}(O):=\mathbf 1\{T=t\}.
\]
Set
\[
\varsigma_{\iota}(L):=
\begin{cases}
L, & \iota\text{ is a proxy, outcome, or target-proxy coordinate},\\
1, & \iota\text{ is an arm coordinate}.
\end{cases}
\]
The bounded target-proxy, bounded proxy-product, and bounded outcome-proxy
components of \cref{obj:def:model-class}, namely
\cref{obj:ass:bounded-x,obj:ass:bounded-proxy-product,obj:ass:bounded-outcome-proxy-product},
give, for every \(\iota\in\mathcal I_{d_x,d_z}\),
\[
|\varphi_\iota(O)|\le \varsigma_\iota(L)
\quad\text{for \(P_O\)-almost every }O,
\]
where \(P_O\) is the observed-data marginal of
\cref{obj:def:observed-margin}.  % lean: summaryCoordStat_ae_bound

Let \(\kappa_{d_z,d_x}\ge0\) be a finite coordinate-to-operator-norm constant
such that, for every \(d_z\times d_x\) real matrix \(H\) and every
\(u_{\mathrm{coord}}\ge0\),
\[
\max_{\mathfrak z,\mathfrak x}|H_{\mathfrak z\mathfrak x}|
\le u_{\mathrm{coord}}
\quad\Longrightarrow\quad
\|H\|_{\mathrm{op}}\le \kappa_{d_z,d_x}u_{\mathrm{coord}}.
\]
Such a constant exists because, if each coordinate of a vector in
\(\mathbb R^r\) has absolute value at most \(u_{\mathrm{coord}}\), then its
Euclidean norm is at most \(r u_{\mathrm{coord}}\); applying this to each
column and then taking the supremum over unit input vectors gives a finite
constant depending only on \(d_z,d_x\).  We shall also use the corresponding
vector estimate
\[
\max_{\mathfrak x}|v_{\mathfrak x}|\le u_{\mathrm{coord}}
\quad\Longrightarrow\quad
\|v\|_2\le d_xu_{\mathrm{coord}} .
\]  % lean: eucNorm_le_card_mul_bound, matrixNorm_le_entryBound

Put
\[
\chi_{\mathrm{sum}}:=|\mathcal I_{d_x,d_z}|=4d_zd_x+d_x+2,
\]
\[
K_{\mathrm{sm}}:=\frac{16\kappa_{d_z,d_x}}{k\pi_0}+d_x,
\qquad
K_{\mathrm{lg}}:=4(\kappa_{d_z,d_x}+1)+d_x+1,
\]
and choose
\[
C_0:=
\max\left\{\chi_{\mathrm{sum}}+1,\ 2K_{\mathrm{sm}},\
\frac{4K_{\mathrm{lg}}}{k\pi_0}\right\}.
\]
Then \(1\le C_0\).  Since \(2\le k\) and \(0<\pi_0\), the product
\(k\pi_0\) is positive, so all denominators in this definition are positive.  % lean: uniform_summary_concentration

Now fix \(L\ge1\), \(n\ge1\), \(P\in\mathcal M\), and
\(0<\eta<1/2\).  Define
\[
\zeta_{\eta,n}:=
\sqrt{\frac{\log(C_0/\eta)}{n}},
\qquad
\beta_{\eta,n}:=2\zeta_{\eta,n}.
\]
For each \(\iota\in\mathcal I_{d_x,d_z}\), Hoeffding's inequality under
\(Q_P^{(n)}=P_O^{\otimes n}\) gives
\[
Q_P^{(n)}
\left\{
\varsigma_\iota(L)\beta_{\eta,n}
\le
\left|
\frac1n\sum_{\mathfrak r=1}^n\varphi_\iota(O_{\mathfrak r})
-
\int \varphi_\iota\,dP_O
\right|
\right\}
\le
2\left(\frac{\eta}{C_0}\right)^2 .
\]
Indeed, \(\varphi_\iota\in[-\varsigma_\iota(L),\varsigma_\iota(L)]\)
\(P_O\)-almost surely, the interval length is \(2\varsigma_\iota(L)\), and the
choice \(\beta_{\eta,n}=2\sqrt{\log(C_0/\eta)/n}\) reduces the exponential
factor to \(\exp\{-2\log(C_0/\eta)\}\).  % lean: product_hoeffding, summaryCoord_deviation_probability

Let
\[
\mathcal B_{\mathrm{dev}}
:=
\left\{\omega:\exists\iota\in\mathcal I_{d_x,d_z},\
\varsigma_\iota(L)\beta_{\eta,n}
\le
\left|
\frac1n\sum_{\mathfrak r=1}^n\varphi_\iota(O_{\mathfrak r})
-
\int \varphi_\iota\,dP_O
\right|\right\}.
\]
By the union bound and \(\chi_{\mathrm{sum}}\le C_0\),
\[
Q_P^{(n)}(\mathcal B_{\mathrm{dev}})
\le
\chi_{\mathrm{sum}}\,2\left(\frac{\eta}{C_0}\right)^2
\le
\frac{2\eta^2}{C_0}
\le
2\eta^2
\le
\eta .
\]
Also set
\[
\mathcal B_{\mathrm{supp}}
:=
\left\{\omega:\exists \mathfrak r,\exists\iota\in\mathcal I_{d_x,d_z},\
|\varphi_\iota(O_{\mathfrak r})|>\varsigma_\iota(L)\right\}.
\]
The \(P_O\)-almost-sure coordinate envelope transported to the product
experiment gives
\[
Q_P^{(n)}(\mathcal B_{\mathrm{supp}})=0 .
\]  % lean: measureReal_iUnion_fintype_le, sample_summaryCoordSupport_ae

The population summary blocks obey
\[
\|M_t(P)\|_{\mathrm{op}}\le L,\qquad
\|N_t(P)\|_{\mathrm{op}}\le L,\qquad
\|m_X(P)\|_2\le L .
\]
For the armwise blocks, \cref{obj:lem:arm-mass-lower-bound} gives positive
arm probabilities, and Jensen's inequality applied to the conditional moments
in \cref{obj:def:observable-population-moments} transfers the almost-sure
bounds on \(ZX^\top\) and \(YZX^\top\).  For \(m_X(P)\), Jensen's inequality
applies directly to the unconditional target-proxy mean.  % lean: observedSummary_envelopes_of_model

It remains to prove a deterministic implication off the two bad events.  Write
\[
\theta_t:=P_O\{O:T=t\},
\qquad
\widehat\theta_t(\omega):=\frac{N_{t,n}(\omega)}{n}.
\]
By \cref{obj:lem:arm-mass-lower-bound}, transported through
\cref{obj:def:observed-margin},
\[
k\pi_0\le \theta_t
\qquad(t\in\{0,1\}).
\]
First suppose that
\[
\beta_{\eta,n}<\frac{k\pi_0}{2}.
\]
On \(\mathcal B_{\mathrm{dev}}^c\),
\[
|\widehat\theta_t-\theta_t|<\beta_{\eta,n},
\qquad
\widehat\theta_t>\frac{k\pi_0}{2}.
\]
For \(\bullet\in\{\operatorname{prox},\operatorname{out}\}\), put
\[
\overline\Phi^\bullet_{t,\mathfrak z,\mathfrak x}(\omega)
:=
\frac1n\sum_{\mathfrak r=1}^n
\varphi_{\bullet,t,\mathfrak z,\mathfrak x}(O_{\mathfrak r}).
\]
The coordinate integral identities are
\[
\int\varphi_{\operatorname{prox},t,\mathfrak z,\mathfrak x}\,dP_O
=
\theta_t\,M_t(P)_{\mathfrak z\mathfrak x},
\qquad
\int\varphi_{\operatorname{out},t,\mathfrak z,\mathfrak x}\,dP_O
=
\theta_t\,N_t(P)_{\mathfrak z\mathfrak x}.
\]
Because \(\widehat\theta_t>0\), the total empty-arm convention in
\cref{obj:def:empirical-summary-primitives} gives
\[
(\widehat M_{t,n}-M_t(P))_{\mathfrak z\mathfrak x}
=
\frac{
\{\overline\Phi^{\operatorname{prox}}_{t,\mathfrak z,\mathfrak x}
-\theta_tM_t(P)_{\mathfrak z\mathfrak x}\}
+
(\theta_t-\widehat\theta_t)M_t(P)_{\mathfrak z\mathfrak x}
}{\widehat\theta_t},
\]
and the same identity with \(\widehat N_{t,n},N_t(P)\) and
\(\overline\Phi^{\operatorname{out}}\).  Since each population entry is bounded
by \(L\), the preceding displays imply, for both matrix types,
\[
\left|
(\widehat M_{t,n}-M_t(P))_{\mathfrak z\mathfrak x}
\right|
\le
\frac{4L\beta_{\eta,n}}{k\pi_0},
\qquad
\left|
(\widehat N_{t,n}-N_t(P))_{\mathfrak z\mathfrak x}
\right|
\le
\frac{4L\beta_{\eta,n}}{k\pi_0}.
\]
The matrix norm conversion and the mean-coordinate bounds
\[
|(\widehat m_{X,n}-m_X(P))_{\mathfrak x}|<L\beta_{\eta,n}
\]
therefore yield
\[
d_S(\widehat S_n,S(P))
\le
\left(\frac{16\kappa_{d_z,d_x}}{k\pi_0}+d_x\right)L\beta_{\eta,n}
=
2K_{\mathrm{sm}}L\zeta_{\eta,n}
\le
C_0L\zeta_{\eta,n}.
\]  % lean: summaryCoordStat_arm_sampleMean, summaryCoordStat_integral, empiricalArmMatrix_entry_error, empSummary_error_of_small_deviations

Now suppose that
\[
\frac{k\pi_0}{2}\le \beta_{\eta,n}.
\]
On \(\mathcal B_{\mathrm{supp}}^c\), every empirical arm matrix has entries
bounded by \(L\): if \(N_{t,n}=0\), the convention in
\cref{obj:def:empirical-summary-primitives} gives the zero matrix, and if
\(N_{t,n}>0\), the matrix is an average of entries whose absolute values are at
most \(L\).  Hence each empirical matrix block has operator norm at most
\(\kappa_{d_z,d_x}L\), and
\[
\|\widehat m_{X,n}\|_2\le d_xL .
\]
Combining these empirical bounds with the population envelopes gives
\[
d_S(\widehat S_n,S(P))
\le
\{4(\kappa_{d_z,d_x}+1)+d_x+1\}L
=
K_{\mathrm{lg}}L .
\]
Since \(k\pi_0/2\le\beta_{\eta,n}=2\zeta_{\eta,n}\),
\[
\frac{k\pi_0}{4}\le \zeta_{\eta,n},
\]
and therefore
\[
K_{\mathrm{lg}}L
\le
\frac{4K_{\mathrm{lg}}}{k\pi_0}L\zeta_{\eta,n}
\le
C_0L\zeta_{\eta,n}.
\]  % lean: abs_fin_average_le, empiricalArmMatrix_entry_bound, empSummary_error_on_support

The two cases imply
\[
\left\{
d_S(\widehat S_n,S(P))>
C_0L\sqrt{\frac{\log(C_0/\eta)}{n}}
\right\}
\subseteq
\mathcal B_{\mathrm{dev}}\cup\mathcal B_{\mathrm{supp}}.
\]
Consequently,
\[
Q_P^{(n)}
\left\{
d_S(\widehat S_n,S(P))>
C_0L\sqrt{\frac{\log(C_0/\eta)}{n}}
\right\}
\le
Q_P^{(n)}(\mathcal B_{\mathrm{dev}})
+
Q_P^{(n)}(\mathcal B_{\mathrm{supp}})
\le
\eta .
\]  % lean: uniform_summary_concentration

Finally fix \(t\in\{0,1\}\).  The event \(N_{t,n}=0\) is the event that all
\(n\) observed records have \(T\ne t\).  Since
\(Q_P^{(n)}=P_O^{\otimes n}\),
\[
Q_P^{(n)}\{N_{t,n}=0\}
=
\bigl(P_O\{O:T\ne t\}\bigr)^n
=
\bigl(1-P_O\{O:T=t\}\bigr)^n .
\]
The lower bound from \cref{obj:lem:arm-mass-lower-bound}, again through
\cref{obj:def:observed-margin}, gives
\[
k\pi_0\le P_O\{O:T=t\}\le 1.
\]
Thus
\[
0\le 1-P_O\{O:T=t\}\le 1-k\pi_0,
\]
and monotonicity of \(x\mapsto x^n\) on \([0,\infty)\) gives
\[
\bigl(1-P_O\{O:T=t\}\bigr)^n
\le
(1-k\pi_0)^n .
\]
This proves both asserted conclusions.  % lean: uniform_summary_concentration, arm_mass_lower_of_latentArmPositivity
\end{proof}
