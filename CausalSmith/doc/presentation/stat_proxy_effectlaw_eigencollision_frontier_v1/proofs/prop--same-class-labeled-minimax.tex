\begin{proof}[Proof of \cref{obj:prop:same-class-labeled-minimax}]
For \(n\ge 1\) and \(0<g\le 1/4\), set
\[
\beta_{n,g}:=\min\left\{1,\frac{1}{\sqrt n\,g}\right\}.
\]
Let \(\mathsf{Est}_n\) be the class of measurable simplex-valued labelled weight estimators based on \(n\) observed records in the specialized model
\[
k=d_x=d_z=2,\qquad L=2,\qquad \pi_0=\frac1{10},\qquad \sigma_0=\frac1{10},
\]
and define
\[
\mathsf{Risk}_{n,g}
:=
\inf_{\widetilde p_n\in\mathsf{Est}_n}
\sup_{P\in\mathcal M(g)}
E_{Q_P^{(n)}}\!\left[
\left\|\widetilde p_n-p^{\uparrow}(P)\right\|_1
\right].
\]

\begin{enumerate}
\item Apply \cref{obj:thm:matching-local-lower-bounds} with \(c_{\mathrm{loc}}=1/4\). It gives a constant \(\kappa_{\mathrm{lo}}>0\) such that, for every \(n\ge1\), every \(0<g\le1/4\), and every \(\widetilde p_n\in\mathsf{Est}_n\), there is a law \(P\in\mathcal L_{n,g}^{p}\) satisfying
\[
\kappa_{\mathrm{lo}}\beta_{n,g}
\le
E_{Q_P^{(n)}}\!\left[
\left\|\widetilde p_n-p^{\uparrow}(P)\right\|_1
\right].
\]
By \cref{obj:def:local-weight-experiment}, \(\mathcal L_{n,g}^{p}\subseteq\mathcal M(g)\). Hence, for every admissible \(\widetilde p_n\),
\[
\sup_{P\in\mathcal M(g)}
E_{Q_P^{(n)}}\!\left[
\left\|\widetilde p_n-p^{\uparrow}(P)\right\|_1
\right]
\ge
\kappa_{\mathrm{lo}}\beta_{n,g},
\]
and taking the infimum over \(\widetilde p_n\in\mathsf{Est}_n\) yields
\[
\mathsf{Risk}_{n,g}\ge \kappa_{\mathrm{lo}}\beta_{n,g}.
\]
% lean: matching_local_lower_bounds

\item Apply \cref{obj:thm:labeled-weight-upper} with
\[
k=d_x=d_z=2,\qquad L=2,\qquad \pi_0=\frac1{10},\qquad \sigma_0=\frac1{10}.
\]
It gives a constant \(\kappa_{\mathrm{up}}>0\) such that, for every \(n\ge1\), there is an estimator
\[
\widehat p_n^{\uparrow}\in\mathsf{Est}_n
\]
satisfying
\[
E_{Q_P^{(n)}}\!\left[
\left\|\widehat p_n^{\uparrow}-p^{\uparrow}(P)\right\|_1
\right]
\le
\kappa_{\mathrm{up}}\beta_{n,g}
\]
for every \(0<g\le1/4\) and every \(P\in\mathcal M(g)\). The target is the ordered aggregate mass vector of the quotient law \(\nu_P\) from \cref{obj:def:quotient-law}: \cref{obj:def:ordered-mass-target} defines it as \(p^{\uparrow}(P)\), and \cref{obj:lem:ordered-masses-measure-invariance} makes this ordered vector depend only on the represented probability law. Therefore
\[
\mathsf{Risk}_{n,g}
\le
\sup_{P\in\mathcal M(g)}
E_{Q_P^{(n)}}\!\left[
\left\|\widehat p_n^{\uparrow}-p^{\uparrow}(P)\right\|_1
\right]
\le
\kappa_{\mathrm{up}}\beta_{n,g}.
\]
% lean: labeled_weight_upper

\item Define
\[
c:=\min\left\{\kappa_{\mathrm{lo}},\frac{\kappa_{\mathrm{up}}}{2}\right\},
\qquad
C:=\kappa_{\mathrm{up}}.
\]
Since \(\kappa_{\mathrm{lo}}>0\) and \(\kappa_{\mathrm{up}}>0\),
\[
0<c
\quad\text{and}\quad
c\le\frac{\kappa_{\mathrm{up}}}{2}<\kappa_{\mathrm{up}}=C,
\]
so \(0<c<C\). Combining the bounds from the preceding two steps gives, for every \(n\ge1\) and \(0<g\le1/4\),
\[
c\min\left\{1,\frac{1}{\sqrt n\,g}\right\}
\le
\mathsf{Risk}_{n,g}
\le
C\min\left\{1,\frac{1}{\sqrt n\,g}\right\}.
\]
This is the asserted minimax inequality.
% lean: same_class_labeled_minimax
\end{enumerate}
\end{proof}
