\begin{proof}[Proof of \cref{obj:lem:path-model-membership}]
\begin{enumerate}
\item Write the two latent classes as \(u=1,2\), and write \(t,x,z,y_0,y_1\in\{0,1\}\) for the binary coordinates in the finite support of the path.  Put
\[
p_1=\frac25+h,\qquad p_2=\frac35-h,
\]
and
\[
\begin{array}{c|cc}
&u=1&u=2\\ \hline
w_1(u)&\frac4{25}+\frac{3h}{5}&\frac9{25}-\frac{3h}{5}\\[0.25em]
w_0(u)&\frac6{25}+\frac{2h}{5}&\frac6{25}-\frac{2h}{5}.
\end{array}
\]
For a Bernoulli coordinate \(r\in\{0,1\}\) and success probability \(\theta\), set
\[
\operatorname{bm}(r;\theta)
=
\begin{cases}
1-\theta,& r=0,\\
\theta,& r=1.
\end{cases}
\]
The path assigns the atom \((u,t,x,z,y_0,y_1)\) mass
\[
p_u\,
\operatorname{bm}\!\left(t;\frac{w_1(u)}{p_u}\right)
\operatorname{bm}(x;\beta_u)
\operatorname{bm}(z;\alpha_{t,u})
\operatorname{bm}\!\left(y_0;\frac14\right)
\operatorname{bm}(y_1;\eta_u),
\]
where
\[
\beta_1=\frac15,\qquad
\beta_2=\frac{12/25-h/5}{3/5-h},
\]
\[
\alpha_{1,1}=\frac{225h+28}{20(15h+4)},\quad
\alpha_{1,2}=\frac34,\quad
\alpha_{0,1}=\frac{35h+9}{10(5h+3)},\quad
\alpha_{0,2}=\frac7{10},
\]
and
\[
\eta_1=\frac12-\frac g2,\qquad \eta_2=\frac12+\frac g2 .
\]
Since \(|h|\le 1/100\),
\[
\frac25+h\ge\frac{39}{100},\qquad
\frac35-h\ge\frac{59}{100},\qquad
15h+4\ge\frac{77}{20},\qquad
5h+3\ge\frac{59}{20}.
\]
A direct check of the two latent classes gives
\[
0\le \frac{w_1(u)}{p_u}\le1,\qquad
0\le \beta_u\le1,\qquad
0\le \alpha_{t,u}\le1,
\]
and \(0\le\eta_u\le1\) follows from \(0\le g\le1/4\).  Thus every displayed atom mass is nonnegative.  Summing first over the Bernoulli success and failure values gives
\[
\sum_{u,t,x,z,y_0,y_1}
p_u\,
\operatorname{bm}\!\left(t;\frac{w_1(u)}{p_u}\right)
\operatorname{bm}(x;\beta_u)
\operatorname{bm}(z;\alpha_{t,u})
\operatorname{bm}\!\left(y_0;\frac14\right)
\operatorname{bm}(y_1;\eta_u)
=
p_1+p_2=1 .
\]
Hence \(P_{g,h}\) is a probability law on the full-data record space with \(k=d_x=d_z=2\). % lean: pathLaw_isProbabilityMeasure

\item The numerical core domain in \cref{obj:def:model-class} is
\[
2\le2,\qquad 2\le2,\qquad 2\le2,\qquad 1\le2,\qquad
0<\frac1{10}\le\frac1{2\cdot2},\qquad
0<\frac1{10}\le1 .
\]
% lean: path_ucvmwModel

\item For \cref{obj:ass:reference-proxy-separation}, fix a latent-arm cell \((U=u,T=t)\).  For bounded measurable test functions \(\varphi_Z\) of \(Z\) and \(\psi_{XY}\) of \((X,Y)\), the finite conditional sums over the atom weights factor as
\[
E_{P_{g,h}}\!\left[
\varphi_Z(Z)\psi_{XY}(X,Y)\mid U=u,T=t
\right]
=
E_{P_{g,h}}\!\left[
\varphi_Z(Z)\mid U=u,T=t
\right]
E_{P_{g,h}}\!\left[
\psi_{XY}(X,Y)\mid U=u,T=t
\right].
\]
Indeed, conditional on \((U=u,T=t)\), the factor depending on \(z\) is
\(\operatorname{bm}(z;\alpha_{t,u})\), while the factors determining \(X\) and \(Y\) are functions of \(x,y_0,y_1\) and the fixed arm \(t\).  The normalizing denominator is the same finite latent-arm mass in all three conditional means, so expanding the four binary nuisance sums gives the displayed product identity. % lean: path_referenceProxySeparation

\item For \cref{obj:ass:target-proxy-separation}, fix \(U=u\).  For bounded measurable test functions \(\varphi_X\) of \(X\) and \(\psi_{YT}\) of \((Y,T)\),
\[
E_{P_{g,h}}\!\left[
\varphi_X(X)\psi_{YT}(Y,T)\mid U=u
\right]
=
E_{P_{g,h}}\!\left[
\varphi_X(X)\mid U=u
\right]
E_{P_{g,h}}\!\left[
\psi_{YT}(Y,T)\mid U=u
\right].
\]
After conditioning on \(U=u\), summing over \(z\) uses
\[
\sum_{z=0}^1\operatorname{bm}(z;\alpha_{t,u})=1
\]
for each arm \(t\).  The remaining finite sum separates the \(x\)-factor \(\operatorname{bm}(x;\beta_u)\) from the factors depending on \(t,y_0,y_1\), and hence on \((Y,T)\). % lean: path_targetProxySeparation

\item For \cref{obj:ass:latent-ignorability}, fix a treatment state \(r\in\{0,1\}\) and a latent class \(u\).  For bounded measurable test functions \(\varphi_r\) of the potential outcome \(Y(r)\) and \(\psi_T\) of \(T\),
\[
E_{P_{g,h}}\!\left[
\varphi_r(Y(r))\psi_T(T)\mid U=u
\right]
=
E_{P_{g,h}}\!\left[
\varphi_r(Y(r))\mid U=u
\right]
E_{P_{g,h}}\!\left[
\psi_T(T)\mid U=u
\right].
\]
This is the same finite-product calculation: the Bernoulli factor for \(Y(r)\) depends on \(u\) and \(r\), while the Bernoulli factor for \(T\) is \(\operatorname{bm}(t;w_1(u)/p_u)\). % lean: path_armwiseLatentIgnorability

\item Every support point of \(P_{g,h}\) records the observed outcome as \(Y=Y(T)\), which gives \cref{obj:ass:consistency}.  Every support point also has \(X=(1,x)^\top\), which gives \(X_1=1\) and hence \cref{obj:ass:anchor}. % lean: path_consistency path_anchor

\item The boundedness conditions hold supportwise.  Since \(X=(1,x)^\top\) and \(Z=(1,z)^\top\) with \(x,z\in\{0,1\}\),
\[
\|X\|_2\le\sqrt2\le2.
\]
For every vector \(v\),
\[
\|ZX^\top v\|_2
=
\|Z\|_2\,|X^\top v|
\le
\|Z\|_2\,\|X\|_2\,\|v\|_2
\le
2\|v\|_2,
\]
so \(\|ZX^\top\|_{\mathrm{op}}\le2\).  Since \(Y\in\{0,1\}\),
\[
\|YZX^\top\|_{\mathrm{op}}\le \|ZX^\top\|_{\mathrm{op}}\le2 .
\]
Thus \cref{obj:ass:bounded-x,obj:ass:bounded-proxy-product,obj:ass:bounded-outcome-proxy-product} hold with \(L=2\). % lean: path_boundedX path_boundedProxyProduct path_boundedOutcomeProxyProduct

\item The latent-arm cell masses are precisely the four displayed values \(w_t(u)\).  From \(|h|\le1/100\),
\[
\frac4{25}+\frac{3h}{5}\ge\frac4{25}-\frac3{500}\ge\frac1{10},\qquad
\frac9{25}-\frac{3h}{5}\ge\frac9{25}-\frac3{500}\ge\frac1{10},
\]
and
\[
\frac6{25}+\frac{2h}{5}\ge\frac6{25}-\frac1{250}\ge\frac1{10},\qquad
\frac6{25}-\frac{2h}{5}\ge\frac6{25}-\frac1{250}\ge\frac1{10}.
\]
Therefore
\[
\min_{u\in\{1,2\},\,t\in\{0,1\}}
P_{g,h}(U=u,T=t)
\ge
\frac1{10},
\]
which is \cref{obj:ass:latent-arm-positivity} with \(\pi_0=1/10\). % lean: path_latentCell_mass path_latentArmPositivity

\item By \cref{obj:lem:path-proxy-rank-margin},
\[
\min\{\sigma_2(A_0),\sigma_2(A_1),\sigma_2(B)\}\ge\frac1{10}
\]
for the reference- and target-proxy feature matrices formed from \(P_{g,h}\).  Hence \cref{obj:ass:proxy-rank-margin} holds with \(\sigma_0=1/10\). % lean: path_proxyRankMargin

\item Combining the core domain with the model restrictions just verified gives
\[
P_{g,h}\in\mathcal M
\]
for \(k=d_x=d_z=2\), \(L=2\), \(\pi_0=1/10\), and \(\sigma_0=1/10\), as required by \cref{obj:def:model-class}. % lean: path_ucvmwModel
\end{enumerate}
\end{proof}
