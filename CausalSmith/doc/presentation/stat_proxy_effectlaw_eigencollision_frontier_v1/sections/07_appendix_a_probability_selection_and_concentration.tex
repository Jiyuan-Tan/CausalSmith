\section{Appendix A: probability, selection, and concentration}

This appendix collects the auxiliary statements used by the population, estimation, confidence, and lower-bound arguments. The first group supplies measurable selection and finite-sample probability control for the observable summary space introduced in \cref{obj:def:observable-summary}. The compact-selector statement is the measurable nearest-point device used by the repair map in \cref{obj:def:summary-space-repair}; it is the form of measurable extrema needed for a compact feasible summary set, in the spirit of \citet{BrownPurves1973}.

\begin{lemmav}{lem:borel-nearest-point-selector}[Compact loss selector]
For every \(d\in\mathbb N\), every set \(\mathcal K\subseteq \mathbb R^d\), and every loss \(d_S:\mathbb R^d\times\mathbb R^d\to\mathbb R\), suppose that
\begin{itemize}
\item \textbf{(Nonempty feasible set.)} \(\mathcal K\) is nonempty.
\item \textbf{(Compact feasible set.)} \(\mathcal K\) is compact.
\item \textbf{(Continuous loss.)} The map \((s,q)\mapsto d_S(s,q)\) is continuous.
\end{itemize}
Then there exists a Borel measurable map \(\Pi:\mathbb R^d\to\mathbb R^d\) such that, for every \(s\in\mathbb R^d\),
\[
\Pi(s)\in\mathcal K
\quad\text{and}\quad
d_S(s,\Pi(s))\le d_S(s,q)\ \text{for every }q\in\mathcal K .
\]
\end{lemmav}

The selection lemma turns compactness of \(\mathcal K\) into a Borel rule, so the nearest-summary construction remains a sample-measurable estimator after applying the empirical summary map. The remaining probability statements in this subsection record the arm-mass and concentration primitives that support the total empty-arm convention.

\begin{lemmav}{lem:arm-mass-lower-bound}[Arm mass lower bound]
Let \(P\) be a full-data probability law in the setting of \cref{obj:def:model-class}. Suppose \(P\) satisfies \cref{obj:ass:latent-arm-positivity} with margin \(\pi_0\). Then, for each treatment arm \(t\in\{0,1\}\),
\[
k\pi_0 \le P(T=t).
\]
\end{lemmav}

The lower bound converts the latent-arm margin into an observed-arm margin. It is the elementary probability input behind the empty-arm probability bound in \cref{obj:lem:uniform-summary-concentration} and behind the observed VMW margin statement in \cref{obj:prop:observed-vmw-margin-inclusion}.

The next statements collect the moment factorizations and envelope calculations that connect the full-data restrictions to the observable five-block summary.

\begin{lemmav}{lem:observed-proxy-moment-factorization}[Observed proxy moment factorization]
Let \(P\) be a full-data probability law carrying \((U,T,X,Z,Y(0),Y(1),Y)\), with fixed target- and reference-proxy dimensions \(d_x,d_z\), in the uniformly conditioned proxy causal model class \(\mathcal M\) of \cref{obj:def:model-class}. Suppose that:
\begin{itemize}
\item \textbf{(Latent cardinality.)} The number \(k\) of latent classes satisfies \(2\le k\).
\item \textbf{(Target dimension.)} The target-proxy dimension satisfies \(k\le d_x\).
\item \textbf{(Boundedness radius.)} The uniform boundedness radius satisfies \(1\le L\), with the bounded-target-proxy condition as in \cref{obj:ass:bounded-x}.
\item \textbf{(Latent-arm margin.)} The latent-arm positivity margin satisfies \(\pi_0>0\), with latent-arm positivity as in \cref{obj:ass:latent-arm-positivity}.
\item \textbf{(Proxy separation.)} The reference- and target-proxy separation conditions hold as in \cref{obj:ass:reference-proxy-separation,obj:ass:target-proxy-separation}.
\end{itemize}
Then, for each treatment arm \(t\in\{0,1\}\), the observable armwise proxy moment \(M_t(P)\) from \cref{obj:def:observable-population-moments} factors as
\[
M_t(P)=A_t\Gamma_t B^\top,
\]
where \(A_t\) is the reference-proxy feature matrix of \cref{obj:def:reference-feature}, \(B\) is the target-proxy feature matrix of \cref{obj:def:target-feature}, and
\[
\Gamma_t
=
\operatorname{diag}\left\{
\frac{P(U=u,T=t)}{P(T=t)}:1\le u\le k
\right\}.
\]
\end{lemmav}

\begin{lemmav}{lem:proxy-coordinate-envelopes}[Coordinate envelope bounds]
Let \(P\in\mathcal M\) be a full-data probability law carrying \((U,T,X,Z,Y(0),Y(1),Y)\), with \(\mathcal M\) as in \cref{obj:def:model-class}.  Suppose that
\begin{itemize}
\item \textbf{(Latent cardinality.)} \(2\le k\).
\item \textbf{(Target-proxy dimension.)} \(k\le d_x\).
\item \textbf{(Envelope conditions.)} \(P\) satisfies \cref{obj:ass:bounded-x,obj:ass:bounded-proxy-product,obj:ass:bounded-outcome-proxy-product}.
\item \textbf{(Anchor normalization.)} \(P\) satisfies \cref{obj:ass:anchor}.
\end{itemize}
Then the model envelope \(L\) bounds the target proxy, the reference proxy, and the outcome--reference-proxy products \(P\)-almost surely:
\[
\Pr_P\{\,|X_i|\le L\text{ for every }1\le i\le d_x\,\}=1,
\]
\[
\Pr_P\{\,|Z_j|\le L\text{ for every }1\le j\le d_z\,\}=1,
\]
and
\[
\Pr_P\{\,|Y Z_j|\le L\text{ for every }1\le j\le d_z\,\}=1.
\]
\end{lemmav}

\begin{lemmav}{lem:latent-mean-envelope}[Latent mean envelope]
Let \(P\in\mathcal M\) be a full-data probability law carrying \((U,T,X,Z,Y(0),Y(1),Y)\), as in \cref{obj:def:model-class}.  Suppose that
\begin{itemize}
\item \textbf{(Latent cardinality.)} \(2\le k\).
\item \textbf{(Proxy dimensions.)} \(k\le d_x\) and \(k\le d_z\).
\item \textbf{(Model constants.)} \(1\le L\), \(\pi_0>0\), and \(\sigma_0>0\).
\item \textbf{(Model components.)} The model class includes the bounded target-proxy condition of \cref{obj:ass:bounded-x}, the latent-arm positivity condition of \cref{obj:ass:latent-arm-positivity}, and the proxy-rank margin condition of \cref{obj:ass:proxy-rank-margin}.
\end{itemize}
Then, for every latent class \(u\in\{1,\ldots,k\}\) and every treatment state \(t\in\{0,1\}\), the latent conditional potential-outcome mean \(\mu_{tu}\) of \cref{obj:def:latent-mean} satisfies
\[
|\mu_{tu}|\le \frac{L_\tau}{2},
\]
where \(L_\tau=4L\sqrt{d_z}/\sigma_0\) is the derived effect-support radius from \cref{obj:def:effect-radius}.
\end{lemmav}

\begin{lemmav}{lem:latent-effect-envelope}[Latent effect envelope]
Let \(k,d_x,d_z\in\mathbb N\), let \(L,\pi_0,\sigma_0\in\mathbb R\), let \(P\) be a full-data probability law carrying \((U,T,X,Z,Y(0),Y(1),Y)\), and fix a latent class \(u\in\{1,\ldots,k\}\). Suppose that
\begin{itemize}
\item \textbf{(Dimensions.)} \(2\le k\), \(k\le d_x\), and \(k\le d_z\).
\item \textbf{(Margins.)} \(1\le L\), \(\pi_0>0\), and \(\sigma_0>0\).
\item \textbf{(Model membership.)} \(P\in\mathcal M\) for the parameters \((k,d_x,d_z,L,\pi_0,\sigma_0)\), as in \cref{obj:def:model-class}, with the bounded-proxy, latent-arm-positivity, and proxy-rank-margin components recorded in \cref{obj:ass:bounded-x,obj:ass:latent-arm-positivity,obj:ass:proxy-rank-margin}.
\end{itemize}
Then the latent-class treatment effect \(\tau_u\), formed from the latent means in \cref{obj:def:latent-mean,obj:def:latent-effect}, satisfies
\[
|\tau_u|\le L_\tau=\frac{4L\sqrt{d_z}}{\sigma_0},
\]
where \(L_\tau\) is the derived effect-support radius in \cref{obj:def:effect-radius}.
\end{lemmav}

\begin{lemmav}{lem:witness-latent-mass}[Witness latent masses]
For every collision-witness effect displacement \(\varepsilon\in\mathbb R\) with \(0\le \varepsilon\le 1/8\), let \(P_{\varepsilon}^{\mathrm{wit}}\) be the two-class witness law of \cref{obj:def:two-class-witness}.  The latent-class masses \(p_u:=P_{\varepsilon}^{\mathrm{wit}}(U=u)\) from \cref{obj:def:latent-mass} are
\[
p_1=\frac25,
\qquad
p_2=\frac35 .
\]
Equivalently, these masses are constant in \(\varepsilon\) over this range and are distinct.
\end{lemmav}

\begin{definitionv}{synth_26}[Ordered aggregate mass map]
For a labelled representative \(\xi=\sum_{i=1}^k w_i\delta_{x_i}\in\mathcal P_{\le k}([-L_\tau,L_\tau])\), define \(\operatorname{orderedMasses}(\xi)\in\mathbb R^k\) as follows. If every labelled weight is positive and the labelled atoms are distinct, then its \(j\)th coordinate is the mass of the atom whose support rank is \(j\):
\[
\operatorname{orderedMasses}(\xi)_j
:=\sum_{i=1}^k w_i\mathbf 1\!\left\{\#\{\ell:x_\ell<x_i\}=j\right\},
\qquad j=0,\ldots,k-1.
\]
If some labelled weight is zero or two labelled atoms coincide, set \(\operatorname{orderedMasses}(\xi):=(1/k,\ldots,1/k)\).
\end{definitionv}

\begin{lemmav}{lem:ordered-masses-measure-invariance}[Ordered masses are representative-invariant]
Fix \(k\in\mathbb N\) and a real support radius.  Let \(\nu\) and \(\xi\) be valid labelled atomic representatives with \(k\) slots in the corresponding compact interval, as in \cref{obj:def:atomic-law-class}.  If \(\nu\) and \(\xi\) represent the same aggregated probability measure, then their ordered-mass vectors coincide:
\[
\operatorname{orderedMasses}(\nu)=\operatorname{orderedMasses}(\xi).
\]
\end{lemmav}

\begin{lemmav}{lem:observed-outcome-proxy-moment-factorization}[Outcome proxy moment factorization]
Let \(P\), a full-data probability law carrying \((U,T,X,Z,Y(0),Y(1),Y)\), belong to the model class \(\mathcal M\) in \cref{obj:def:model-class}. Suppose that
\begin{itemize}
\item \textbf{(Latent classes.)} \(2\le k\).
\item \textbf{(Positivity margin.)} The latent-arm positivity margin satisfies \(\pi_0>0\).
\item \textbf{(Model conditions.)} The reference- and target-proxy separation, consistency, latent ignorability, latent-arm positivity, bounded outcome-proxy product, and bounded target-proxy conditions in \cref{obj:ass:reference-proxy-separation,obj:ass:target-proxy-separation,obj:ass:consistency,obj:ass:latent-ignorability,obj:ass:latent-arm-positivity,obj:ass:bounded-outcome-proxy-product,obj:ass:bounded-x} hold.
\end{itemize}
Then, for each treatment arm \(t\in\{0,1\}\),
\[
N_t(P)
=
A_t\,\Gamma_t\,\operatorname{diag}\{\mu_{tu}:1\le u\le k\}\,B^\top,
\qquad
\Gamma_t
=
\operatorname{diag}\left\{\frac{P(U=u,T=t)}{P(T=t)}:1\le u\le k\right\}.
\]
Here \(N_t(P)\) is the observable armwise \(YZX^\top\) population moment in \cref{obj:def:observable-population-moments}, \(A_t\) is the reference-proxy feature matrix in \cref{obj:def:reference-feature}, \(B\) is the target-proxy feature matrix in \cref{obj:def:target-feature}, and \(\mu_{tu}\) is the latent conditional potential-outcome mean in \cref{obj:def:latent-mean}.
\end{lemmav}

\begin{lemmav}{lem:latent-arm-weight-invertibility}[Latent-arm weight injectivity]
Let \(P\) be a full-data probability law carrying \((U,T,X,Z,Y(0),Y(1),Y)\), and let \(\pi_0>0\). Suppose \(P\) satisfies latent-arm positivity with margin \(\pi_0\), as in \cref{obj:ass:latent-arm-positivity}. Then, for each treatment arm \(t\in\{0,1\}\), the diagonal matrix
\[
\Gamma_t
=
\operatorname{diag}\left\{
\frac{P(U=u,T=t)}{P(T=t)}
: u\in\{1,\ldots,k\}
\right\}
\]
defines an injective linear map on \(\mathbb R^k\).
\end{lemmav}

\begin{lemmav}{lem:moore-penrose-product-cancellation}[Moore--Penrose product cancellation]
Let \(C\in\mathbb R^{\mathrm{rows}\times r}\), \(B\in\mathbb R^{\mathrm{cols}\times r}\), and \(\mu\in\mathbb R^r\). Suppose that the linear maps represented by \(C\) and by \(B\), each with domain \(\mathbb R^r\), are injective. Then
\[
(CB^\top)^\dagger\bigl(C\operatorname{diag}(\mu)B^\top\bigr)
=
(B^\dagger)^\top \operatorname{diag}(\mu)B^\top .
\]
\end{lemmav}

\begin{lemmav}{lem:kantorovich-rubinstein-attaining-potential}[Attaining KR potential]
Let \(\nu\) and \(\xi\) be valid represented atomic laws with finite labelled slot sets: their weights are nonnegative and sum to one.  Let \(\varphi_{\nu,\xi}\) be the cumulative-distribution-gap sign Kantorovich--Rubinstein potential selected for the pair \((\nu,\xi)\).  Then \(\varphi_{\nu,\xi}\) is one-Lipschitz, is normalized by
\[
\varphi_{\nu,\xi}(0)=0,
\]
and attains the Wasserstein distance of \cref{obj:def:wasserstein-distance}:
\[
W_1(\nu,\xi)
=
\left|
  \int \varphi_{\nu,\xi}\,d\nu
  -
  \int \varphi_{\nu,\xi}\,d\xi
\right|.
\]
Moreover, for every \(\epsilon\in\mathbb R\), if every one-Lipschitz function \(\varphi:\mathbb R\to\mathbb R\) satisfying \(\varphi(0)=0\) obeys
\[
\left|
  \int \varphi\,d\nu
  -
  \int \varphi\,d\xi
\right|
\le \epsilon,
\]
then
\[
W_1(\nu,\xi)\le \epsilon.
\]
\end{lemmav}

\begin{lemmav}{lem:summary-metric-coordinate-comparison}[Coordinate metric comparison]
Let \(d_x\) and \(d_z\) be fixed target- and reference-proxy dimensions, and let \(s,s'\) be elements of the five-block observable summary space in \cref{obj:def:observable-summary}. Equip this space with its coordinate Euclidean distance \(\rho_{\mathrm{sum}}\). Then
\[
d_S(s,s')
\le
\bigl(4\operatorname{Ent}_{d_z,d_x}+d_x\bigr)\rho_{\mathrm{sum}}(s,s'),
\]
where
\[
\operatorname{Ent}_{r,c}
:=
c\,r\,\|J_c\|_{\mathrm{op}},
\]
and \(J_c:\mathbb R^c\to(\{1,\ldots,c\}\to\mathbb R)\) is the canonical coordinate identification.
\end{lemmav}

\begin{lemmav}{lem:summary-metric-continuity}[Summary metric continuity]
For fixed target- and reference-proxy dimensions \(d_x,d_z\), let \(d_S\) be the summary metric on observable moment summaries from \cref{obj:def:observable-summary}. The map
\[
(s,q)\longmapsto d_S(s,q)
\]
from the product of the five-block summary space with itself to \(\mathbb R\) is continuous.
\end{lemmav}

\begin{lemmav}{lem:stacked-proxy-moment-signal-margin}[Stacked proxy signal margin]
Let \(P\), a full-data probability law carrying \((U,T,X,Z,Y(0),Y(1),Y)\), belong to the uniformly conditioned proxy causal model class \(\mathcal M\) in \cref{obj:def:model-class}. Suppose:
\begin{itemize}
\item \textbf{(Dimensions.)} \(2\le k\le d_x\), where \(k\) is the number of latent classes and \(d_x\) is the target-proxy dimension.
\item \textbf{(Radius.)} \(L\ge 1\), with the bounded target-proxy condition in \cref{obj:ass:bounded-x}.
\item \textbf{(Margins.)} \(\pi_0>0\) and \(\sigma_0>0\), with the latent-arm positivity and proxy-rank margin conditions in \cref{obj:ass:latent-arm-positivity,obj:ass:proxy-rank-margin}.
\end{itemize}
Then the vertically stacked observable proxy moment satisfies
\[
\pi_0\sigma_0^2
\le
\sigma_{k-1}\!\left\{\operatorname{stack}\bigl(M_0(P),M_1(P)\bigr)\right\},
\]
where \(\sigma_{k-1}\) uses zero-based singular-value indexing and \(M_0(P),M_1(P)\) are the armwise \(ZX^\top\) population moments from \cref{obj:def:observable-population-moments}.
\end{lemmav}

These witness calculations locate the explicit two-class comparisons inside the same model restrictions used for the upper bounds. The path-local statement calibrates the observed Kullback--Leibler radius to the product \(g^2h^2\), which is the scale used by \cref{obj:thm:matching-local-lower-bounds}.

The final two auxiliary statements concern ordered masses as functions of represented atomic laws. They ensure that, when the support is separated, the ordered-weight target depends on the aggregate measure and is invariant across labelled coordinate representations.

\begin{lemmav}{lem:model-compressed-spectral-certificate}[Compressed spectral certificate]
Let \(k,d_x,d_z\in\mathbb N\), let \(L,\pi_0,\sigma_0\in\mathbb R\), and let \(Q\) be a full-data probability law in the uniformly conditioned model class \(\mathcal M\) of \cref{obj:def:model-class}, with observable summary \(S(Q)\) as in \cref{obj:def:observable-summary}. Assume that \(Q\) satisfies \cref{obj:ass:anchor,obj:ass:proxy-rank-margin,obj:ass:latent-arm-positivity}. Let \(M_t(S(Q))\) be the \(t\)-arm \(ZX^\top\) moment block from \cref{obj:def:observable-population-moments}, and set
\[
G:=\operatorname{stack}(M_0(S(Q)),M_1(S(Q))).
\]
Then there exist a signal basis \(V\) spanning the signal row space of \(S(Q)\) and a real diagonalization of the compressed effect operator \(\Delta Q(Q)\) from \cref{obj:def:compressed-operator} such that:
\begin{enumerate}
\item for each arm \(t\in\{0,1\}\), the compressed map
\[
x\longmapsto M_t(S(Q))Vx
\]
is injective;
\item the stacked proxy moment has rank
\[
\operatorname{rank}(G)=k;
\]
\item for every index \(j\),
\[
\frac{\pi_0\sigma_0^2}{2}\le \sigma_j(G)
\quad\Longleftrightarrow\quad
j<k;
\]
\item the eigenvalue vector of the diagonalization is the latent-effect vector \((\tau_u(Q))_u\) from \cref{obj:def:latent-effect};
\item writing \(R\) for the diagonalizing basis matrix, the left and right spectral anchors satisfy
\[
a(Q)=R^{-\top}p(Q),
\qquad
R^{-1}c(Q)=\mathbf 1_k,
\]
where \(p(Q)=(p_u(Q))_u\) is the latent-mass vector from \cref{obj:def:latent-mass}.
\end{enumerate}
\end{lemmav}

\begin{lemmav}{lem:diagonalization-condition-number-bound}[Diagonalization condition bound]
Let \(d_x\ge 1\) and \(k\ge 1\).  Let \(B\in\mathbb R^{d_x\times k}\), and let \(F\) be a thin signal factorization of \(B\), written \(B=V\Upsilon\), with coordinate inverse \(\Upsilon^{-1}\).  Let \(\tau\in\mathbb R^k\).  Suppose that
\begin{itemize}
\item \textbf{(Feature bound.)} For a constant \(L\ge 0\) as in \cref{obj:ass:bounded-x}, \(|B_{iu}|\le L\) for every row \(i\) and latent index \(u\).
\item \textbf{(Coordinate-inverse bound.)} For a constant \(\sigma_0>0\) as in \cref{obj:ass:proxy-rank-margin}, \(|(\Upsilon^{-1})_{ru}|\le \sigma_0^{-1}\) for every pair of coordinate indices \(r,u\).
\end{itemize}
Define
\[
K_{\mathrm{forw}}
:=
\operatorname{Ent}_{d_x,k}\operatorname{Ent}_{k,k}\operatorname{Ent}_{k,d_x}\sigma_0^{-1}
+
\bigl(\operatorname{Ent}_{d_x,d_x}
+
\operatorname{Ent}_{d_x,k}\operatorname{Ent}_{k,d_x}\bigr),
\]
\[
K_{\mathrm{back}}
:=
\operatorname{Ent}_{d_x,k}\operatorname{Ent}_{k,k}\operatorname{Ent}_{k,d_x}(d_xL)
+
\bigl(\operatorname{Ent}_{d_x,d_x}
+
\operatorname{Ent}_{d_x,k}\operatorname{Ent}_{k,d_x}\bigr),
\]
and
\[
\kappa_\star
:=
\operatorname{Ent}_{d_x,d_x}^{\,2}K_{\mathrm{forw}}K_{\mathrm{back}},
\]
where \(\operatorname{Ent}_{r,c}\) is the entrywise-to-operator norm constant of \cref{obj:lem:summary-metric-coordinate-comparison}.  Then the real diagonalization associated with \(F\) and \(\tau\) has condition number at most \(\kappa_\star\).
\end{lemmav}

The definition matches the ordering convention in \cref{obj:def:ordered-mass-target} and \cref{obj:def:labeled-weight-estimator}. The invariance lemma records the compatibility needed to read ordered masses from any valid representative of the same aggregated atomic law.
