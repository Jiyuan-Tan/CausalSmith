\begin{proof}[Proof of \cref{obj:thm:minimax-risk}]
Fix the displayed regime constants.  We use seven steps.

\begin{enumerate}
\item Write \(R_n^\star\) for the observed minimax risk over the admissible estimator class used here,
\[
R_n^\star
=
\inf_{T_n}
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}
\left[
\left|T_n-\theta_{\delta_n}(P)\right|
\right],
\]
where the infimum ranges over observed-sample measurable estimators satisfying
\(T_n(z)\in[0,1]\) for every observed sample \(z\).  For a split \(B_n\), put
\[
\mathcal R_n^{\mathrm{TG}}(B_n)
=
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}
\left[
\left|\widehat\theta_{n,B_n}^{\mathrm{TG}}-\theta_{\delta_n}(P)\right|
\right].
\]
This is the finite-sample stabilized risk in \cref{obj:synth_5}, with the product
sample law interpreted as in \cref{obj:synth_16}.  \Cref{obj:lem:stabilized-risk-upper}
gives a constant \(C_{\mathrm{up}}>0\) such that, for every admissible deterministic
\((\delta_n)\) and \(B_\bullet\), for all sufficiently large \(n\),
\[
\mathcal R_n^{\mathrm{TG}}(B_n)
\le
C_{\mathrm{up}}
\left(n^{-1/2}+\delta_n^{\kappa+1}h_n^\beta\right).
\]
The estimator \(\widehat\theta_{n,B_n}^{\mathrm{TG}}\) is observed-sample measurable and
\([0,1]\)-valued. Hence it is an admissible competitor in the infimum defining
\(R_n^\star\), so
\[
R_n^\star\le \mathcal R_n^{\mathrm{TG}}(B_n).
\]
% lean: observedMinimaxRisk_le_stabilizedWorstRisk

\item We record the two-point device used for both lower bounds.  Let
\(H_0,H_1\in\mathcal M\), put \(\vartheta_j=\theta_\delta(H_j)\), and suppose
\[
0\le \Delta\le |\vartheta_1-\vartheta_0|,
\qquad
0\le\delta\le1,
\]
with well-posed product chi-squared divergence
\[
H_1^{\otimes n}\ll H_0^{\otimes n},
\qquad
\int\left(\frac{dH_1^{\otimes n}}{dH_0^{\otimes n}}-1\right)^2dH_0^{\otimes n}<\infty,
\]
and
\[
\chi^2(H_1^{\otimes n},H_0^{\otimes n})\le \chi,
\qquad
0\le\chi<4 .
\]
For any admissible \([0,1]\)-valued observed-sample estimator \(T_n\), the testing
inequality and
\(\|H_1^{\otimes n}-H_0^{\otimes n}\|_{\mathrm{TV}}
\le \frac12\sqrt{\chi}\) give
\[
\max_{j=0,1}
H_j^{\otimes n}
\left\{
|T_n-\vartheta_j|\ge \Delta/2
\right\}
\ge
\frac{1-\frac12\sqrt{\chi}}{2}.
\]
Multiplying by \(\Delta/2\), the larger of the two risks under \(H_0,H_1\) is at least
\[
\frac{1-\frac12\sqrt{\chi}}{4}\,\Delta .
\]
The supremum over \(P\in\mathcal M\) in the restricted minimax problem dominates these
two risks for each \(T_n\); taking the infimum over admissible \(T_n\) yields
\[
R_n^\star
\ge
\frac{1-\frac12\sqrt{\chi}}{4}\,\Delta .
\]
% lean: observedMinimaxRisk_lower_of_two_point

\item For the parametric component, use the canonical design
\[
P(X=x)=1/J,
\qquad
A\mid X=x\sim (\kappa+1)a^\kappa\,da,
\quad a\in[0,1],
\]
and Bernoulli regressions
\[
\mu_{x}^{P_{\mathrm r,0}}(a)=\frac12,
\qquad
\mu_{x}^{P_{\mathrm r,1}}(a)=\frac12+\varepsilon_n,
\qquad
\varepsilon_n=\frac14 n^{-1/2}.
\]
Both laws belong to \(\mathcal M\).  The stratum masses are \(1/J\), which meets the stratum-mass condition of \cref{obj:ass:stratum-mass} under the regime constraint \(p_{\min}\le1/J\); the canonical density \((\kappa+1)a^\kappa\) meets the conditional-density condition of \cref{obj:ass:conditional-density-law} and, by the regime constraint \(c_-\le\kappa+1\le c_+\), the polynomial-thinning envelope of \cref{obj:ass:polynomial-thinning}; and each regression is constant, so every Taylor remainder in \cref{obj:ass:holder-regression} vanishes and the Hölder condition holds for any \(L>0\).  Finally \(|\varepsilon_n|\le1/4\) for all \(n\ge1\), so both constant regressions take values in \([0,1]\).
The clamp target shifts by
\[
\left|
\theta_{\delta_n}(P_{\mathrm r,1})
-
\theta_{\delta_n}(P_{\mathrm r,0})
\right|
=\varepsilon_n .
\]
The product chi-squared divergence is well posed, and
\[
1+\chi^2(P_{\mathrm r,1}^{\otimes n},P_{\mathrm r,0}^{\otimes n})
=
(1+4\varepsilon_n^2)^n
\le
\exp(n\,4\varepsilon_n^2)
=
\exp(1/4).
\]
With \(\chi_{\mathrm r}=\exp(1/4)-1<4\), the two-point mixture bound \((1+4\varepsilon_n^2)^n\le\exp(1/4)\) gives a constant
\(c_{\mathrm r}>0\), depending only on the regime constants, such that
\[
c_{\mathrm r}n^{-1/2}\le R_n^\star
\]
for every admissible \((\delta_n)\) and all sufficiently large \(n\).
% lean: observedMinimaxRisk_eventually_ge_root

\item For the localized lower-bound component, use a fixed \(C^\infty\) dose bump
\(b_\circ:\mathbb R\to[0,1]\) centered at zero, with
\[
b_\circ(0)=1,
\qquad
0\le b_\circ(u)\le1,
\qquad
b_\circ(u)=0\quad\text{for }u\notin[-1,1].
\]
Let \(\ell=\ell(\beta)\).  Since \(\beta>0\), the number
\(\gamma_\beta=\beta-\ell\) belongs to \((0,1]\).  Because \(b_\circ\) is smooth and
compactly supported, there is a finite constant \(K_{\mathrm{bump}}\ge1\) such that
\[
\left|
b_\circ^{(\ell)}(u)-b_\circ^{(\ell)}(v)
\right|
\le
K_{\mathrm{bump}}|u-v|^{\gamma_\beta},
\qquad u,v\in\mathbb R.
\]
Choose \(a_{\mathrm{low}}>0\) so small that
\[
a_{\mathrm{low}}\le\frac14,
\qquad
a_{\mathrm{low}}K_{\mathrm{bump}}\le L .
\]
For any \(0\le\delta\le1\) and \(0<h\le1\), define
\[
q_{\delta,h}^{\mathrm{low}}(a)
=
a_{\mathrm{low}}h^\beta
b_\circ\!\left(\frac{a-\delta}{h}\right),
\qquad
f_{\delta,h}^{\mathrm{low}}(a)
=
\frac12+q_{\delta,h}^{\mathrm{low}}(a).
\]
Then \(f_{\delta,h}^{\mathrm{low}}\) is continuous on \([0,1]\), and
\(f_{\delta,h}^{\mathrm{low}}(a)\in[0,1]\) there because
\[
0\le q_{\delta,h}^{\mathrm{low}}(a)\le a_{\mathrm{low}}h^\beta\le a_{\mathrm{low}}\le\frac14 .
\]
For \(s,t\in[0,1]\), differentiating the scaled bump \(\ell\) times and using
\(\gamma_\beta=\beta-\ell\) gives
\[
\begin{aligned}
\left|
\frac{d^\ell}{da^\ell}q_{\delta,h}^{\mathrm{low}}(s)
-
\frac{d^\ell}{da^\ell}q_{\delta,h}^{\mathrm{low}}(t)
\right|
&\le
 a_{\mathrm{low}}h^{\beta-\ell}K_{\mathrm{bump}}
 \left|\frac{s-\delta}{h}-\frac{t-\delta}{h}\right|^{\gamma_\beta} \\
&=
 a_{\mathrm{low}}K_{\mathrm{bump}}|s-t|^{\beta-\ell}
\le
 L|s-t|^{\beta-\ell}.
\end{aligned}
\]
The one-dimensional Taylor remainder theorem on \([0,1]\) therefore yields
\[
\left|
f_{\delta,h}^{\mathrm{low}}(t)
-
\sum_{j=0}^{\ell}
\frac{(f_{\delta,h}^{\mathrm{low}})^{(j)}(s)}{j!}(t-s)^j
\right|
\le
L|t-s|^\beta,
\qquad s,t\in[0,1].
\]
Thus the localized Bernoulli regression satisfies the member condition in
\cref{obj:ass:holder-regression} uniformly over \(0\le\delta\le1\) and \(0<h\le1\).
Set
\[
\zeta_{\mathrm{low}}=8(\kappa+1)a_{\mathrm{low}}^2,
\qquad
\rho_\beta=2\beta+1,
\qquad
\tau_{\mathrm{low}}
=
\exp\!\left(-\frac{\zeta_{\mathrm{low}}+1}{\rho_\beta}\right).
\]
Then \(0<\tau_{\mathrm{low}}\le1\) and
\[
\zeta_{\mathrm{low}}\tau_{\mathrm{low}}^{2\beta+1}\le1.
\]
For the contracted bandwidth \(\tau_{\mathrm{low}}h_n\), form Bernoulli laws
\[
\mu_x^{Q_{\mathrm{low},0}}(a)=\frac12,
\qquad
\mu_x^{Q_{\mathrm{low},1}}(a)=\frac12+
 a_{\mathrm{low}}(\tau_{\mathrm{low}}h_n)^\beta
 b_\circ\!\left(\frac{a-\delta_n}{\tau_{\mathrm{low}}h_n}\right)
\]
on the same canonical design.  For all sufficiently large \(n\),
\cref{obj:lem:bandwidth-balance} gives
\[
0<h_n,
\qquad
h_n\le1-\bar\delta,
\qquad
n h_n^{2\beta+1}(\delta_n+h_n)^\kappa=1 .
\]
Together with \(\delta_n\in[0,\bar\delta]\), this gives \(\delta_n\in[0,1]\),
\(0<\tau_{\mathrm{low}}h_n\le h_n\le1\), and \(\delta_n+h_n\le1\).  The displayed
Taylor bound places both laws in \(\mathcal M\).  The target separation obeys
\[
a_{\mathrm{low}}\delta_n^{\kappa+1}(\tau_{\mathrm{low}}h_n)^\beta
\le
\left|
\theta_{\delta_n}(Q_{\mathrm{low},1})
-
\theta_{\delta_n}(Q_{\mathrm{low},0})
\right|.
\]
The one-observation chi-squared increment satisfies
\[
\int 4a_{\mathrm{low}}^2(\tau_{\mathrm{low}}h_n)^{2\beta}
 b_\circ\!\left(\frac{a-\delta_n}{\tau_{\mathrm{low}}h_n}\right)^2
\,dP_{X,A}(x,a)
\le
8(\kappa+1)a_{\mathrm{low}}^2
(\tau_{\mathrm{low}}h_n)^{2\beta+1}(\delta_n+\tau_{\mathrm{low}}h_n)^\kappa .
\]
Using the balance identity above, \(\tau_{\mathrm{low}}h_n\le h_n\), and \(\kappa\ge0\),
\[
n\int 4a_{\mathrm{low}}^2(\tau_{\mathrm{low}}h_n)^{2\beta}
 b_\circ\!\left(\frac{a-\delta_n}{\tau_{\mathrm{low}}h_n}\right)^2
\,dP_{X,A}(x,a)
\le
\zeta_{\mathrm{low}}\tau_{\mathrm{low}}^{2\beta+1}
\le1 .
\]
Let
\[
I_{\mathrm{low},n}
=
\int 4a_{\mathrm{low}}^2(\tau_{\mathrm{low}}h_n)^{2\beta}
 b_\circ\!\left(\frac{a-\delta_n}{\tau_{\mathrm{low}}h_n}\right)^2
\,dP_{X,A}(x,a)
\]
be the one-observation chi-squared increment for this Bernoulli perturbation.  The product law factorization gives
\[
1+\chi^2(Q_{\mathrm{low},1}^{\otimes n},Q_{\mathrm{low},0}^{\otimes n})
=
(1+I_{\mathrm{low},n})^n .
\]
Since \(I_{\mathrm{low},n}\ge0\) and \(nI_{\mathrm{low},n}\le1\), the inequality \(1+u\le e^u\) yields
\[
\chi^2(Q_{\mathrm{low},1}^{\otimes n},Q_{\mathrm{low},0}^{\otimes n})
\le
\exp(nI_{\mathrm{low},n})-1
\le e-1<4 .
\]
Applying the two-point bound gives a constant \(c_{\mathrm{loc}}>0\) such that
\[
c_{\mathrm{loc}}\delta_n^{\kappa+1}h_n^\beta
\le
R_n^\star
\]
for every admissible \((\delta_n)\) and all sufficiently large \(n\).
% lean: observedMinimaxRisk_eventually_ge_local

\item The global witness in the theorem uses the same canonical design and
\[
\mu_x^{P_0}(a)=\frac12,
\qquad
\mu_x^{P_1}(a)=\frac12+n^{-1/2}.
\]
For all sufficiently large \(n\), \(n^{-1/2}\le1/2\), so these are Bernoulli laws in
\(\mathcal M\), with shared \((X,A)\)-design, shared \(p_x\), and shared \(\pi_x\).
They satisfy
\[
\mu_x^{P_1}(a)=\mu_x^{P_0}(a)+n^{-1/2}
\]
for every \(x\) and \(a\in[0,1]\), and
\[
\frac12 n^{-1/2}
\le
\left|
\theta_{\delta_n}(P_1)-\theta_{\delta_n}(P_0)
\right|.
\]
The one-observation chi-squared increment is \(4n^{-1}\).  Hence the product divergence is
well posed and satisfies
\[
1+\chi^2(P_1^{\otimes n},P_0^{\otimes n})
=
(1+4n^{-1})^n
\le
\exp(4),
\]
so
\[
\chi^2(P_1^{\otimes n},P_0^{\otimes n})
\le
\exp(4).
\]
% lean: globalBernoulli_witness_eventually

\item For the localized witness appearing in the statement, use the same fixed bump
\(b_\circ\).  The amplitude construction above also permits a separate witness
amplitude: choose \(\eta_{\mathrm{wit}}>0\) such that
\[
\eta_{\mathrm{wit}}\le\frac14,
\qquad
\eta_{\mathrm{wit}}K_{\mathrm{bump}}\le L .
\]
For every \(0\le\delta\le1\) and \(0<h\le1\), the function
\[
f_{\delta,h}^{\mathrm{wit}}(t)
=
\frac12+\eta_{\mathrm{wit}} h^\beta b_\circ\!\left(\frac{t-\delta}{h}\right)
\]
is continuous on \([0,1]\), takes values in \([0,1]\), and the same scaled-derivative
calculation gives
\[
\left|
f_{\delta,h}^{\mathrm{wit}}(t)
-
\sum_{j=0}^{\ell}
\frac{(f_{\delta,h}^{\mathrm{wit}})^{(j)}(s)}{j!}(t-s)^j
\right|
\le
L|t-s|^\beta,
\qquad s,t\in[0,1].
\]
Set
\[
\zeta_{\mathrm{wit}}=8(\kappa+1)\eta_{\mathrm{wit}}^2,
\qquad
D_{\mathrm{wit}}=\exp(\zeta_{\mathrm{wit}}).
\]
For \(h_n\) itself, define
\[
\mu_x^{Q_0}(t)=\frac12,
\qquad
\mu_x^{Q_1}(t)=\frac12+
 \eta_{\mathrm{wit}} h_n^\beta b_\circ\!\left(\frac{t-\delta_n}{h_n}\right).
\]
For all sufficiently large \(n\), \cref{obj:lem:bandwidth-balance} gives
\[
0<h_n,
\qquad
h_n\le1-\bar\delta,
\qquad
n h_n^{2\beta+1}(\delta_n+h_n)^\kappa=1 .
\]
Together with \(\delta_n\in[0,\bar\delta]\), this gives \(\delta_n\in[0,1]\),
\(h_n\le1\), and \(\delta_n+h_n\le1\), so \(Q_0,Q_1\in\mathcal M\).  The pair has
Bernoulli outcomes, common design, common \(p_x\), common \(\pi_x\), and
\[
\mu_x^{Q_1}(t)-\mu_x^{Q_0}(t)
=
\eta_{\mathrm{wit}} h_n^\beta b_\circ\!\left(\frac{t-\delta_n}{h_n}\right)
\]
for every \(x\) and \(t\in[0,1]\).  Since \(b_\circ(0)=1\) and the perturbation is
nonnegative,
\[
\eta_{\mathrm{wit}}\delta_n^{\kappa+1}h_n^\beta
\le
\left|
\theta_{\delta_n}(Q_1)-\theta_{\delta_n}(Q_0)
\right|.
\]
The same integral calculation and the balance identity give
\[
n\int 4\eta_{\mathrm{wit}}^2h_n^{2\beta}
 b_\circ\!\left(\frac{t-\delta_n}{h_n}\right)^2
\,dP_{X,A}(x,t)
\le
\zeta_{\mathrm{wit}}.
\]
Let
\[
I_{\mathrm{wit},n}
=
\int 4\eta_{\mathrm{wit}}^2h_n^{2\beta}
 b_\circ\!\left(\frac{t-\delta_n}{h_n}\right)^2
\,dP_{X,A}(x,t)
\]
be the one-observation chi-squared increment.  The product law factorization gives
\[
1+\chi^2(Q_1^{\otimes n},Q_0^{\otimes n})
=
(1+I_{\mathrm{wit},n})^n .
\]
The perturbation is bounded by \(1/4\), so the Bernoulli likelihood ratio is finite and
square-integrable for one observation; absolute continuity and square integrability tensorize to
the product laws, giving a well-posed product chi-squared divergence.  Since
\(I_{\mathrm{wit},n}\ge0\) and \(nI_{\mathrm{wit},n}\le\zeta_{\mathrm{wit}}\), the inequality
\(1+u\le e^u\) gives
\[
\chi^2(Q_1^{\otimes n},Q_0^{\otimes n})
\le
\exp(nI_{\mathrm{wit},n})-1
\le
\exp(\zeta_{\mathrm{wit}})-1
\le
D_{\mathrm{wit}} .
\]
% lean: localizedBernoulli_witness_eventually

\item Now set
\[
c=
\frac12\min\{c_{\mathrm r},c_{\mathrm{loc}},1/2,\eta_{\mathrm{wit}}\},
\qquad
C=C_{\mathrm{up}}+D_{\mathrm{wit}}+\exp(4)+c+1 .
\]
The amplitude asserted in the theorem is \(\eta_{\mathrm{wit}}\).  Then \(0<c<C\), and \(c\le1/2\), \(c\le \eta_{\mathrm{wit}}\), \(2c\le c_{\mathrm r}\), and
\(2c\le c_{\mathrm{loc}}\).  Increasing the eventual index so that all preceding
bounds hold, put
\[
r_{0,n}=n^{-1/2},
\qquad
r_{1,n}=\delta_n^{\kappa+1}h_n^\beta,
\qquad
r_n=r_{0,n}+r_{1,n}.
\]
The two lower components imply
\[
R_n^\star\ge c_{\mathrm r}r_{0,n},
\qquad
R_n^\star\ge c_{\mathrm{loc}}r_{1,n}.
\]
Since \(r_{0,n},r_{1,n}\ge0\), these inequalities and the choices of \(c\) give
\[
R_n^\star
\ge
\max\{2c\,r_{0,n},2c\,r_{1,n}\}
\ge
\frac12(2c\,r_{0,n})+
\frac12(2c\,r_{1,n})
=
c r_n .
\]
Together with the admissibility and upper bounds above,
\[
c r_n
\le
R_n^\star
\le
\mathcal R_n^{\mathrm{TG}}(B_n)
\le
C r_n .
\]
By \cref{obj:synth_5,obj:synth_16}, \(\mathcal R_n^{\mathrm{TG}}(B_n)\) is the
finite-sample worst-case risk \(\operatorname{Risk}_n(B_n)\) of the stabilized estimator
computed under the observed i.i.d. product law.  The global witness constructed above satisfies
\[
c n^{-1/2}
\le
\left|\theta_{\delta_n}(P_1)-\theta_{\delta_n}(P_0)\right|,
\qquad
\chi^2(P_1^{\otimes n},P_0^{\otimes n})\le C,
\]
because \(c\le1/2\) and \(\exp(4)\le C\).  The localized witness constructed above
satisfies
\[
c\delta_n^{\kappa+1}h_n^\beta
\le
\left|\theta_{\delta_n}(Q_1)-\theta_{\delta_n}(Q_0)\right|,
\qquad
\chi^2(Q_1^{\otimes n},Q_0^{\otimes n})\le C,
\]
because \(c\le \eta_{\mathrm{wit}}\) and \(D_{\mathrm{wit}}\le C\).  These are the asserted eventual
bounds and witness properties.
% lean: clamp_minimax_risk
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:honest-length}]
\begin{enumerate}
\item Let \(\ell=\lceil\beta\rceil-1\) and
\[
t_\alpha=\sqrt{\frac{\log(12J/\alpha)}{2}} .
\]
The expected-length upper bound used below is the following eventual statement: there is a constant
\(C_{\mathrm{up}}>0\), depending only on the declared regime constants, such that for every admissible
threshold path and split path,
\[
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}
\!\left[\operatorname{len}(\mathrm{CI}_n)\right]
\le
C_{\mathrm{up}}
\left(n^{-1/2}+\delta_n^{\kappa+1}h_n^\beta\right)
\]
for all sufficiently large \(n\). To see this, use \cref{obj:def:honest-interval}. Intersecting the
symmetric interval with \([0,1]\) weakly shortens its length, and therefore
\[
\mathbb E_{P^{\otimes n}}\!\left[\operatorname{len}(\mathrm{CI}_n)\right]
\le
2\,\mathbb E_{P^{\otimes n}}\!\left[
 b_{0,n}+\sum_{x\in\mathcal X}D_{n,x}\right],
\]
where
\[
b_{0,n}=\frac{t_\alpha}{\sqrt{|I_{0,n}|}},
\qquad
b_{1,n}=\frac{t_\alpha}{\sqrt{|I_{1,n}|}},
\]
and \(D_{n,x}\) is the stratum contribution in \cref{obj:def:honest-interval}. The split lower bounds
in \cref{obj:synth_8} give \(b_{0,n},b_{1,n}\le 3t_\alpha n^{-1/2}\) once \(n\ge8\).

For a fixed law \(P\in\mathcal M\) and stratum \(x\), the integrated stratum contribution is bounded by
\[
\begin{aligned}
\mathbb E_{P^{\otimes n}}D_{n,x}
&\le
\left(\mathbb E_{P^{\otimes n}}\widehat b_x+b_{1,n}\right)
\left[ C_{\ell,1}Lh_n^\beta
+t_\alpha
\int
\left(\sum_{i\in I_{2,n}}w_{ix}(z)^2\right)^{1/2}
\,dP^{\otimes n}(z)
\right]
+b_{1,n}+P^{\otimes n}(\Omega_x^c) .
\end{aligned}
\]
Here \(C_{\ell,1}\) is the finite \(\ell_1\)-weight constant supplied on \(\Omega_x\) by
\cref{obj:lem:total-gram}. The displayed product bound is the sample-splitting step: the empirical atom
estimate uses \(I_{1,n}\), the local weights use \(I_{2,n}\), and the mixed term factors as
\[
\int \widehat b_x(z)
\left(\sum_{i\in I_{2,n}}w_{ix}(z)^2\right)^{1/2}
\,dP^{\otimes n}(z)
=
\left(\int\widehat b_x\,dP^{\otimes n}\right)
\left(\int
\left(\sum_{i\in I_{2,n}}w_{ix}^2\right)^{1/2}
\,dP^{\otimes n}\right).
\]
For the empirical atom envelope, define
\[
f_x(O)=\mathbf 1\{X=x,\ A\leq\delta_n\},
\qquad
m_x(P)=p_xq_x(\delta_n).
\]
Then \(\widehat b_x=|I_{1,n}|^{-1}\sum_{i\in I_{1,n}}f_x(O_i)\). The mean of one summand is
\[
\mathbb E_P f_x(O)
=P\{X=x,A\leq\delta_n\}
=p_x\int_0^{\delta_n}\pi_x(a)\,da
=p_xq_x(\delta_n)
=m_x(P).
\]
The second equality is the conditional-density component of \cref{obj:def:model-class}, and the third is the definition of \(q_x\) in \cref{obj:def:clamp-functional}.
Since \(0\leq f_x\leq1\), \(\operatorname{Var}_P\{f_x(O)\}\leq1/4\). Independence on the deterministic second block therefore yields
\[
\begin{aligned}
\mathbb E_{P^{\otimes n}}
\left|\widehat b_x-m_x(P)\right|
&\leq
\left\{\operatorname{Var}_{P^{\otimes n}}(\widehat b_x)\right\}^{1/2}  \\
&\leq
\frac12 |I_{1,n}|^{-1/2}.
\end{aligned}
\]
Moreover, \cref{obj:prop:pushforward-setup} gives
\(q_x(\delta_n)\leq c_+\delta_n^{\kappa+1}/(\kappa+1)\), and \(p_x\leq1\), so
\[
m_x(P)\leq \frac{c_+}{\kappa+1}\delta_n^{\kappa+1}.
\]
The triangle inequality gives the integrated envelope
\[
\mathbb E_{P^{\otimes n}}\widehat b_x
\le
\frac12 |I_{1,n}|^{-1/2}
+
\frac{c_+}{\kappa+1}\delta_n^{\kappa+1}.
\]
Also, \cref{obj:lem:weight-energy-balance} gives
\[
\int
\left(\sum_{i\in I_{2,n}}w_{ix}(z)^2\right)^{1/2}
\,dP^{\otimes n}(z)
\le C_{\mathrm{eng}}h_n^\beta
\]
for a finite constant \(C_{\mathrm{eng}}\), and \cref{obj:lem:total-gram} gives
\[
P^{\otimes n}(\Omega_x^c)
\le
C_{\mathrm{tg}}\exp\{-c_{\mathrm{tg}}n h_n(\delta_n+h_n)^\kappa\}.
\]
By \cref{obj:lem:bandwidth-balance}, for all sufficiently large \(n\),
\[
n h_n^{2\beta+1}(\delta_n+h_n)^\kappa=1,
\qquad
0<h_n,
\qquad
h_n\le1.
\]
The effective local sample size has the following lower bound. Put
\[
\gamma_{\mathrm{bal}}=2\beta+\kappa+1 .
\]
Since \(\delta_n\ge0\), \(h_n>0\), and \(\kappa\ge0\),
\[
h_n^{\gamma_{\mathrm{bal}}}
=h_n^{2\beta+1}h_n^\kappa
\le h_n^{2\beta+1}(\delta_n+h_n)^\kappa
=n^{-1}.
\]
Because \(\gamma_{\mathrm{bal}}>0\), this gives \(h_n\le n^{-1/\gamma_{\mathrm{bal}}}\). Multiplying the balance identity by \(h_n^{-2\beta}\) gives
\[
n h_n(\delta_n+h_n)^\kappa=h_n^{-2\beta},
\]
and therefore, since \(-2\beta<0\),
\[
n^{2\beta/(2\beta+\kappa+1)}
=\left(n^{-1/\gamma_{\mathrm{bal}}}\right)^{-2\beta}
\le h_n^{-2\beta}
=n h_n(\delta_n+h_n)^\kappa .
\]
Exponential decay along this polynomial scale is eventually bounded by \(n^{-1/2}\). Combining the
exponential-decay bound with the polynomial scale bound for \(n h_n(\delta_n+h_n)^\kappa\), and using
\(h_n^\beta\le1\), yields uniformly in \(P\in\mathcal M\) and \(x\in\mathcal X\),
\[
\mathbb E_{P^{\otimes n}}D_{n,x}
\le
C_{\mathrm{rad}}
\left(n^{-1/2}+\delta_n^{\kappa+1}h_n^\beta\right).
\]
Since \(J\) is fixed, summing over strata gives the displayed \(C_{\mathrm{up}}\) bound.

\item Put \(\eta=1-2\alpha>0\). The lower-bound input is a two-point inequality for expected length.  Because an arbitrary interval procedure need not have finite expected length, the worst-case length in this step is taken in \([0,\infty]\), with \(\int^{-}\) denoting the lower integral of a nonnegative measurable function; for the bounded intervals of interest this is the ordinary expected length.
If \(P_0,P_1\in\mathcal M\) satisfy
\[
\Delta\le |\theta_{\delta_n}(P_1)-\theta_{\delta_n}(P_0)|,
\qquad
P_1^{\otimes n}\ll P_0^{\otimes n},
\]
the likelihood-ratio square is \(P_0^{\otimes n}\)-integrable, and
\(\chi^2(P_1^{\otimes n},P_0^{\otimes n})\le\chi\), then every uniformly honest observed-sample
interval procedure \(C_n\) satisfies
\[
\max\!\left\{\left(1-2\alpha-\frac12\sqrt\chi\right)\Delta,0\right\}
\le
\sup\left\{
\int^{-}\operatorname{len}(C_n)\,dP^{\otimes n}: P\in\mathcal M
\right\},
\]
where the supremum and the lower integral are taken in \([0,\infty]\) and the nonnegative real number on the left is read as an element of \([0,\infty]\) as in \cref{obj:synth_11}. The comparisons in this step are made in \([0,\infty]\); the final paragraph returns to the ordinary expected-length notation of \cref{obj:synth_3,obj:synth_17} once the interval is known to be bounded. Indeed, coverage at \(P_0\) and
\(P_1\), transferred from \(P_1^{\otimes n}\) to \(P_0^{\otimes n}\) by
\[
d_{\mathrm{TV}}(P_1^{\otimes n},P_0^{\otimes n})
\le \frac12\sqrt{\chi},
\]
gives \(P_0^{\otimes n}\{\theta_{\delta_n}(P_0),\theta_{\delta_n}(P_1)\in C_n\}\ge
1-2\alpha-\frac12\sqrt\chi\). On that event the interval length is at least \(\Delta\), and integrating
this indicator gives the displayed extended-real bound.

For the root component, set
\[
\varepsilon_n=\tau_\alpha n^{-1/2},
\qquad
\tau_\alpha=\frac{\eta}{16},
\qquad
\chi_\alpha=\frac{\eta^2}{16}.
\]
Let \(\mathbb P^{\mathrm{root}}_{0,n}\) and \(\mathbb P^{\mathrm{root}}_{1,n}\) have the common design
\[
\mathbb P(X=x,A\in da)=\frac1J(\kappa+1)a^\kappa\mathbf 1\{0\leq a\leq1\}\,da,
\]
and, conditionally on \((X,A)=(x,a)\), let \(Y\) be Bernoulli with success probability
\[
\frac12+g^{\mathrm{root}}_{j,n}(x,a),
\qquad
g^{\mathrm{root}}_{0,n}(x,a)=0,\qquad
g^{\mathrm{root}}_{1,n}(x,a)=\varepsilon_n .
\]
For all sufficiently large \(n\), \(|\varepsilon_n|\leq1/2\). The two laws therefore have outcomes in \([0,1]\), the same density \(\pi_x(a)=(\kappa+1)a^\kappa\), the same masses \(p_x=1/J\), and constant Hölder regressions; the regime inequalities place both laws in \(\mathcal M\). Their clamp-target gap is exactly
\[
\begin{aligned}
\theta_{\delta_n}(\mathbb P^{\mathrm{root}}_{1,n})
-\theta_{\delta_n}(\mathbb P^{\mathrm{root}}_{0,n})
&=\varepsilon_n
\left[
\mathbb P(A>\delta_n)+\sum_{x\in\mathcal X}p_xq_x(\delta_n)
\right] \\
&=\varepsilon_n .
\end{aligned}
\]
For one observation the chi-square divergence is \(4\varepsilon_n^2\), hence independence gives
\[
1+\chi^2\!\left((\mathbb P^{\mathrm{root}}_{1,n})^{\otimes n},
(\mathbb P^{\mathrm{root}}_{0,n})^{\otimes n}\right)
=(1+4\varepsilon_n^2)^n.
\]
Since \(4n\varepsilon_n^2=\eta^2/64\),
\[
(1+4\varepsilon_n^2)^n
\le \exp(\eta^2/64)
\le 1+\eta^2/64+(\eta^2/64)^2
\le 1+\chi_\alpha .
\]
The two-point length inequality above, with \(\chi=\chi_\alpha\), yields
\[
\max\!\left\{
\left(\eta-\frac12\sqrt{\chi_\alpha}\right)\varepsilon_n,0
\right\}
\le L_n^\star .
\]
Thus, for some \(c_{\mathrm{root}}>0\),
\[
\max\!\left\{c_{\mathrm{root}}n^{-1/2},0\right\}
\le L_n^\star
\]
eventually.

For the local component, use the fixed compactly supported \(C^\infty\) bump \(\mathfrak b:\mathbb R\to\mathbb R\) from the same two-point construction, normalized by
\[
\mathfrak b(0)=1,
\qquad
0\leq\mathfrak b(u)\leq1,
\qquad
\mathfrak b(u)=0\ \text{whenever } |u|\geq1 .
\]
The smoothness calibration for this particular bump gives an amplitude
\(\mathfrak a_{\mathrm{bump}}\in(0,1/4]\), depending only on \((\beta,L)\), with the following uniform property. For every \(0\leq\delta\leq1\) and \(0<h\leq1\), define
\[
q_{\delta,h}^{\mathrm{bump}}(a)
=
\mathfrak a_{\mathrm{bump}}h^\beta
\mathfrak b\!\left(\frac{a-\delta}{h}\right).
\]
Then \(a\mapsto 1/2+q_{\delta,h}^{\mathrm{bump}}(a)\) is continuous on \([0,1]\), takes values in \([0,1]\), and satisfies the Hölder Taylor remainder
\[
\left|
\left(\frac12+q_{\delta,h}^{\mathrm{bump}}(t)\right)
-
\sum_{j=0}^{\ell}
\frac{\partial_{[0,1]}^j(1/2+q_{\delta,h}^{\mathrm{bump}})(s)}{j!}(t-s)^j
\right|
\le L|t-s|^\beta
\]
for all \(s,t\in[0,1]\). Indeed, applying the normalized Hölder gate of \(\mathfrak b\) with radius \(\min\{L,1\}\) and then shrinking the amplitude to be at most \(1/4\) gives both the displayed Taylor bound and the range constraint.

Set
\[
\zeta_{\mathrm{loc}}
=
\frac{\eta^2/64}{8(\kappa+1)\mathfrak a_{\mathrm{bump}}^2+1},
\qquad
h_{n,\mathrm{loc}}=\zeta_{\mathrm{loc}}h_n .
\]
Let \(\mathbb P^{\mathrm{loc}}_{0,n}\) and \(\mathbb P^{\mathrm{loc}}_{1,n}\) have the same common design as the root pair, and let their conditional Bernoulli success probabilities be \(1/2+g^{\mathrm{loc}}_{j,n}(x,a)\), where
\[
g^{\mathrm{loc}}_{0,n}(x,a)=0,\qquad
g^{\mathrm{loc}}_{1,n}(x,a)
=\mathfrak a_{\mathrm{bump}}h_{n,\mathrm{loc}}^\beta
\mathfrak b\!\left(\frac{a-\delta_n}{h_{n,\mathrm{loc}}}\right).
\]
For all sufficiently large \(n\), \cref{obj:lem:bandwidth-balance} gives \(0<h_n\le1\), hence \(0<h_{n,\mathrm{loc}}\le1\), and the amplitude calibration places both laws in \(\mathcal M\). The localized shift is nonnegative and equals \(\mathfrak a_{\mathrm{bump}}h_{n,\mathrm{loc}}^\beta\) at \(a=\delta_n\), so the collapsed atom contribution alone gives
\[
\left|
\theta_{\delta_n}(\mathbb P^{\mathrm{loc}}_{1,n})
-\theta_{\delta_n}(\mathbb P^{\mathrm{loc}}_{0,n})
\right|
\geq
\mathfrak a_{\mathrm{bump}}\delta_n^{\kappa+1}h_{n,\mathrm{loc}}^\beta
=
\mathfrak a_{\mathrm{bump}}\zeta_{\mathrm{loc}}^\beta
\delta_n^{\kappa+1}h_n^\beta .
\]
For one observation, write
\[
\mathcal I_n
=
\int 4\{g^{\mathrm{loc}}_{1,n}(x,a)\}^2\,d\mathbb P_{X,A}(x,a),
\]
where \(\mathbb P_{X,A}\) is the common design law. Since \(0\leq\mathfrak b\leq1\) and \(\mathfrak b\) vanishes outside \([-1,1]\),
\[
\mathcal I_n
\le
8(\kappa+1)\mathfrak a_{\mathrm{bump}}^2
h_{n,\mathrm{loc}}^{2\beta+1}(\delta_n+h_{n,\mathrm{loc}})^\kappa .
\]
Because \(0<\zeta_{\mathrm{loc}}\leq1\), \(h_{n,\mathrm{loc}}\leq h_n\). With \(2\beta+1>1\), \(\kappa\geq0\), and the balance identity from \cref{obj:lem:bandwidth-balance},
\[
\begin{aligned}
n\mathcal I_n
&\le
8(\kappa+1)\mathfrak a_{\mathrm{bump}}^2
\zeta_{\mathrm{loc}}^{2\beta+1}
\,n h_n^{2\beta+1}(\delta_n+h_n)^\kappa \\
&\le
\eta^2/64 .
\end{aligned}
\]
Independence gives
\[
1+\chi^2\!\left((\mathbb P^{\mathrm{loc}}_{1,n})^{\otimes n},
(\mathbb P^{\mathrm{loc}}_{0,n})^{\otimes n}\right)
=(1+\mathcal I_n)^n
\le \exp(\eta^2/64)
\le 1+\chi_\alpha .
\]
The two-point length inequality therefore gives, for some \(c_{\mathrm{loc}}>0\),
\[
\max\!\left\{c_{\mathrm{loc}}\delta_n^{\kappa+1}h_n^\beta,0\right\}
\le L_n^\star
\]
eventually. Taking half the smaller of \(c_{\mathrm{root}}\) and \(c_{\mathrm{loc}}\), and splitting into
the two cases according to which component is larger, gives a constant \(c_{\mathrm{min}}>0\) such that
\[
\max\!\left\{c_{\mathrm{min}}\left(n^{-1/2}+\delta_n^{\kappa+1}h_n^\beta\right),0\right\}
\le L_n^\star
\]
eventually.

The every-procedure version uses that a uniformly honest procedure is an admissible competitor in the
minimax honest-length criterion. Hence there is a positive constant \(c_{\mathrm{proc}}>0\) such that
every uniformly honest observed-sample interval procedure \(C_n\) satisfies
\[
\max\!\left\{c_{\mathrm{proc}}\left(n^{-1/2}+\delta_n^{\kappa+1}h_n^\beta\right),0\right\}
\le
\sup\left\{
\int^{-}\operatorname{len}(C_n)\,dP^{\otimes n}: P\in\mathcal M
\right\}
\]
in \(\mathbb R_{\ge0}\cup\{\infty\}\).

\item Define
\[
c=\frac12\min\{c_{\mathrm{min}},c_{\mathrm{proc}}\},
\qquad
C=C_{\mathrm{up}}+c+1 .
\]
Then \(c>0\), \(c<C\), \(c\le c_{\mathrm{min}}\), \(c\le c_{\mathrm{proc}}\), and \(C_{\mathrm{up}}\le C\).

\item Fix an admissible threshold path \((\delta_n)\) and split path \(B_\bullet\). By
\cref{obj:lem:bandwidth-balance}, for all sufficiently large \(n\),
\[
0<h_n,
\qquad
h_n\le 1-\bar\delta,
\qquad
n h_n^{2\beta+1}(\delta_n+h_n)^\kappa=1 .
\]
Together with \(0\le\delta_n\le\bar\delta\), this gives \(\delta_n+h_n\le1\). Hence the hypotheses of
\cref{obj:lem:stabilized-interval-honesty} hold. That result supplies observed-sample measurability of
\(\mathrm{CI}_n\) and the modelwise coverage inequality
\[
1-\alpha
\le
P^{\otimes n}\{\theta_{\delta_n}(P)\in \mathrm{CI}_n\}
\]
for every \(P\in\mathcal M\), so the stabilized interval is uniformly honest.
Taking the infimum over \(P\) gives the displayed worst-case coverage bound.

\item Put
\[
r_n=n^{-1/2}+\delta_n^{\kappa+1}h_n^\beta .
\]
Since \(n^{-1/2}>0\) and \(\delta_n^{\kappa+1}h_n^\beta\ge0\), we have \(r_n>0\). The upper ingredient and \(C_{\mathrm{up}}\le C\)
yield
\[
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}\!\left[\operatorname{len}(\mathrm{CI}_n)\right]
\le C r_n .
\]
The minimax lower ingredient and \(c\le c_{\mathrm{min}}\) yield the extended-real comparison
\[
\max\{c r_n,0\}\le L_n^\star .
\]

Since \(\mathrm{CI}_n\) is itself uniformly honest by the coverage step, the procedure lower bound
applies to this concrete interval. Combined with \(c\le c_{\mathrm{proc}}\), it gives
\[
\max\{c r_n,0\}
\le
\sup\left\{
\int^{-}\operatorname{len}(\mathrm{CI}_n)\,dP^{\otimes n}: P\in\mathcal M
\right\}
\]
in \([0,\infty]\). For the concrete stabilized interval the length is measurable, nonnegative, and
bounded by one, so its worst-case length in \([0,\infty]\) agrees with its ordinary worst-case
expected length, read in \([0,\infty]\) as in \cref{obj:synth_11}. Since \(c r_n>0\), this gives the
real lower bound
\[
c r_n
\le
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}\!\left[\operatorname{len}(\mathrm{CI}_n)\right].
\]

Finally, for every uniformly honest observed-sample interval procedure \(C_n\), the procedure lower
bound gives the extended-real statement
\[
\max\{c r_n,0\}
\le
\sup\left\{
\int^{-}\operatorname{len}(C_n)\,dP^{\otimes n}: P\in\mathcal M
\right\},
\]
with the supremum taken in \(\mathbb R_{\ge0}\cup\{\infty\}\). This is the arbitrary-procedure worst-length criterion used in the comparison. For interval procedures whose pointwise lengths lie in \([0,1]\) as in \cref{obj:synth_17}, the nonnegative lower integral equals the ordinary expected length under the positive-part embedding of \cref{obj:synth_11}, and the display specializes to the corresponding real expected-length lower bound.
These are the eventual length comparisons established by the argument.
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:phase-diagram}]
Let
\[
\Gamma=\beta\kappa+2\beta+\kappa+1,
\qquad
\Xi=2\beta+\kappa+1,
\qquad
\zeta=\frac{\Gamma}{2\beta+1}.
\]
The regime assumptions give \(\Gamma>0\), \(\Xi>0\), and \(\zeta>0\).

\begin{enumerate}
\item The two deterministic scales satisfy
\[
\frac{\delta_{\mathrm{crit},n}}{\delta_{\mathrm{edge},n}}
=
\frac{n^{-1/(2\Gamma)}}{n^{-1/\Xi}}
=
n^{1/\Xi-1/(2\Gamma)}
\]
for every \(n\ge1\).  The exponent is positive because
\[
2\Gamma-\Xi
=
2\beta\kappa+2\beta+\kappa+1>0.
\]
Hence \(n^{1/\Xi-1/(2\Gamma)}\to\infty\), which proves
\[
\frac{\delta_{\mathrm{crit},n}}{\delta_{\mathrm{edge},n}}\to\infty .
\]
% lean: critical_scale_div_edge_scale_tendsto_top

\item Fix a deterministic threshold sequence \((\delta_n)\) with
\(\delta_n\in[0,\bar\delta]\).  Let \(h_n\) be the information-balance bandwidth
from \cref{obj:def:bandwidth}, define
\[
a_n=\delta_n^{\kappa+1}h_n^\beta,
\]
and let \(r_n\) be the frontier rate in \cref{obj:def:frontier}, so that
\(r_n=n^{-1/2}+a_n\).  By \cref{obj:lem:bandwidth-balance}, for all sufficiently
large \(n\),
\[
h_n>0,
\qquad
h_n\le 1-\bar\delta,
\qquad
n h_n^{2\beta+1}(\delta_n+h_n)^\kappa=1 .
\]
The phase comparisons below use this deterministic balance relation directly.
In the regular regime \(\delta_n/\delta_{\mathrm{crit},n}\to0\), we also have,
for all sufficiently large \(n\),
\[
\frac{\delta_n}{\delta_{\mathrm{crit},n}}<1 .
\]
If \(\delta_n=0\), then \(a_n=0\) because \(\kappa+1>0\).  If \(\delta_n>0\), set
\[
\eta_n=\bigl(n\delta_n^\kappa\bigr)^{-1/(2\beta+1)} .
\]
Since \(\delta_n+h_n\ge\delta_n\) and \(\kappa\ge0\), the balance equation implies
\[
h_n\le \eta_n .
\]
Therefore
\[
a_n
=
\delta_n^{\kappa+1}h_n^\beta
\le
\delta_n^{\kappa+1}\eta_n^\beta .
\]
The exact power computation is
\[
\frac{\delta_n^{\kappa+1}\eta_n^\beta}{n^{-1/2}}
=
\left(\frac{\delta_n}{\delta_{\mathrm{crit},n}}\right)^\zeta
\le 1 .
\]
Thus \(0\le a_n\le n^{-1/2}\) eventually, and
\[
n^{-1/2}\le r_n=n^{-1/2}+a_n\le 2n^{-1/2}
\]
eventually.  Hence \(r_n\asymp n^{-1/2}\).
% lean: regular_frontier_asymp

\item Suppose \(\delta_n/\delta_{\mathrm{crit},n}\to c_0\) with \(c_0>0\).
By the scale separation \(\delta_{\mathrm{crit},n}/\delta_{\mathrm{edge},n}\to\infty\),
\[
\frac{\delta_n}{\delta_{\mathrm{edge},n}}
=
\frac{\delta_n}{\delta_{\mathrm{crit},n}}
\frac{\delta_{\mathrm{crit},n}}{\delta_{\mathrm{edge},n}}
\to\infty .
\]
For this above-edge regime, the balance equation gives the closed-form bandwidth
order.  Indeed, for all sufficiently large \(n\),
\(\delta_n>0\) and \(\delta_{\mathrm{edge},n}\le\delta_n\).  With
\(\eta_n=(n\delta_n^\kappa)^{-1/(2\beta+1)}\), the balance equation gives
\(h_n\le \eta_n\).  The inequality \(\delta_{\mathrm{edge},n}\le\delta_n\) gives
\(\eta_n\le\delta_n\), and hence \(\delta_n+h_n\le2\delta_n\).  Substituting this
upper bound into the same balance equation yields
\[
2^{-\kappa/(2\beta+1)} \eta_n\le h_n\le \eta_n .
\]
Consequently
\[
h_n\asymp \bigl(n\delta_n^\kappa\bigr)^{-1/(2\beta+1)} .
\]
Multiplying by the nonnegative factor \(\delta_n^{\kappa+1}\) and raising the
bandwidth comparison to the positive power \(\beta\) gives
\[
a_n\asymp
\delta_n^{\kappa+1}
\left\{\bigl(n\delta_n^\kappa\bigr)^{-1/(2\beta+1)}\right\}^{\beta}.
\]
The normalized interior term satisfies
\[
\frac{
\delta_n^{\kappa+1}
\left\{\bigl(n\delta_n^\kappa\bigr)^{-1/(2\beta+1)}\right\}^{\beta}
}{n^{-1/2}}
=
\left(\frac{\delta_n}{\delta_{\mathrm{crit},n}}\right)^\zeta
\to c_0^\zeta\in(0,\infty).
\]
Consequently \(a_n\asymp n^{-1/2}\).  Since
\(r_n=n^{-1/2}+a_n\) and both summands are nonnegative, also
\[
r_n\asymp n^{-1/2}.
\]
% lean: critical_atom_and_frontier_asymp

\item Suppose \(\delta_n/\delta_{\mathrm{crit},n}\to\infty\) and
\(\delta_n\to0\).  The scale separation
\(\delta_{\mathrm{crit},n}/\delta_{\mathrm{edge},n}\to\infty\) again yields
\[
\frac{\delta_n}{\delta_{\mathrm{edge},n}}\to\infty,
\]
so the same direct balance comparison used at the critical scale gives
\[
h_n\asymp \bigl(n\delta_n^\kappa\bigr)^{-1/(2\beta+1)} .
\]
Thus
\[
a_n\asymp
\delta_n^{\kappa+1}
\bigl(n\delta_n^\kappa\bigr)^{-\beta/(2\beta+1)} .
\]
Moreover,
\[
\frac{
\delta_n^{\kappa+1}
\bigl(n\delta_n^\kappa\bigr)^{-\beta/(2\beta+1)}
}{n^{-1/2}}
=
\left(\frac{\delta_n}{\delta_{\mathrm{crit},n}}\right)^\zeta
\to\infty .
\]
Hence the root term is eventually bounded by a constant multiple of the
displayed atom scale, and
\[
r_n
=
n^{-1/2}+a_n
\asymp
\delta_n^{\kappa+1}
\bigl(n\delta_n^\kappa\bigr)^{-\beta/(2\beta+1)} .
\]
% lean: supercritical_frontier_asymp

\item Suppose \(\delta_n\to\delta_0\) with \(\delta_0\in(0,\bar\delta]\).  Since
\[
\delta_{\mathrm{crit},n}=n^{-1/(2\Gamma)}\to0,
\]
we have
\[
\frac{\delta_n}{\delta_{\mathrm{crit},n}}\to\infty .
\]
Together with the scale separation
\(\delta_{\mathrm{crit},n}/\delta_{\mathrm{edge},n}\to\infty\), this places the sequence above the
edge scale.
The same direct balance comparison used in the supercritical case gives
\[
r_n\asymp
\delta_n^{\kappa+1}
\bigl(n\delta_n^\kappa\bigr)^{-\beta/(2\beta+1)} .
\]
For all sufficiently large \(n\), \(\delta_n>0\), and
\[
\delta_n^{\kappa+1}
\bigl(n\delta_n^\kappa\bigr)^{-\beta/(2\beta+1)}
=
n^{-\beta/(2\beta+1)}\delta_n^\zeta .
\]
Because \(\delta_n^\zeta\to\delta_0^\zeta\in(0,\infty)\), it follows that
\[
r_n\asymp n^{-\beta/(2\beta+1)} .
\]
% lean: fixed_positive_frontier_asymp

\item Fix \(n\) and \(P\in\mathcal M\).  At threshold \(0\),
\cref{obj:def:clamp-functional} gives, for every stratum \(x\),
\[
q_x(0)=\int_{[0,0]}\pi_x(a)\,\mathrm da=0 .
\]
The conditional-density condition in \cref{obj:ass:conditional-density-law}
gives
\[
P(A=0,X=x)=p_x\int_{\{0\}}\pi_x(a)\,\mathrm da=0
\]
for every stratum \(x\).  Summing over the finite stratum set yields
\(P(A=0)=0\), and the treatment support encoded by \cref{obj:def:model-class}
gives \(A>0\) almost surely.  Therefore the retained term in
\cref{obj:def:clamp-functional} satisfies
\[
\nu_0(P)
=
\int Y\,\mathbf 1\{A>0\}\,dP
=
\int Y\,dP .
\]
Since every \(q_x(0)\) is zero,
\[
\theta_0(P)
=
\nu_0(P)+\sum_x p_x q_x(0)\mu_x^P(0)
=
\int Y\,dP .
\]
Finally, \(\kappa+1>0\), so \cref{obj:def:frontier} gives
\[
r_n\big|_{\delta_n=0}
=
n^{-1/2}+0^{\kappa+1}h_n^\beta
=
n^{-1/2}.
\]
% lean: zero_threshold_atom_and_frontier, zero_threshold_functional

\item Fix \(n\), \(P\in\mathcal M\), and an admissible split \(B=(I_0,I_1,I_2)\)
as in \cref{obj:synth_8}.  The conditional-density condition in
\cref{obj:ass:conditional-density-law} gives \(P(A=0)=0\), so \(A>0\) under \(P\)
almost surely.  Since the product law in
\cref{obj:synth_16} has each observed coordinate distributed as \(P\), every
coordinate satisfies \(A_i>0\) under \(P^{\otimes n}\) almost surely, and the
finite intersection of these coordinate events still has probability one.  On
that event, the finite-block average is interpreted as the totalized expression
\[
\operatorname{Avg}_I(Y)
=
\begin{cases}
|I|^{-1}\sum_{i\in I}Y_i,& |I|>0,\\
0,& |I|=0,
\end{cases}
\]
which is the displayed product \((|I|)^{-1}\sum_{i\in I}Y_i\) with the
real-inverse convention \(0^{-1}=0\).  Thus
\[
\widehat b_x
=
|I_1|^{-1}\sum_{i\in I_1}\mathbf 1\{X_i=x,\ A_i\le0\}
=
0
\qquad\text{for every }x,
\]
while
\[
\widehat\nu_n
=
|I_0|^{-1}\sum_{i\in I_0}Y_i .
\]
Substituting these identities into \cref{obj:def:total-gram-estimator} gives,
\(P^{\otimes n}\)-almost surely,
\[
\widehat\theta_n
=
\Pi_{[0,1]}
\left(
|I_0|^{-1}\sum_{i\in I_0}Y_i
+
\sum_x 0\cdot \widetilde\mu_x
\right)
=
\Pi_{[0,1]}
\left(
|I_0|^{-1}\sum_{i\in I_0}Y_i
\right).
\]
Thus the threshold-zero total-Gram estimator agrees under the product sampling
law with the clamped block-zero outcome average.
% lean: zero_threshold_totalGramEstimator

\item Apply \cref{obj:thm:honest-length} to the deterministic sequence
\(\delta_n\equiv0\) and to a deterministic admissible split sequence
\(B_\bullet=(B_n)_{n\ge1}\).  The identity \(r_n=n^{-1/2}\) at
threshold zero gives eventual worst-case coverage
\[
\inf_{P\in\mathcal M}
P^{\otimes n}\!
\left\{\theta_0(P)\in \mathrm{CI}_{n,B_n}\right\}
\ge 1-\alpha,
\]
and constants \(0<c_{\mathrm{len}}\le C_{\mathrm{len}}<\infty\) such that,
eventually, the concrete stabilized length from \cref{obj:synth_5} satisfies
\[
c_{\mathrm{len}}\,n^{-1/2}
\le
\operatorname{Length}_n(B_n)
\le
C_{\mathrm{len}}\,n^{-1/2}.
\]
Likewise, applying \cref{obj:thm:minimax-risk} to the same zero sequence and
using \(r_n=n^{-1/2}\) gives constants
\(0<c_{\mathrm{risk}}\le C_{\mathrm{risk}}<\infty\) such that, eventually, the
concrete stabilized risk from \cref{obj:synth_5} satisfies
\[
c_{\mathrm{risk}}\,n^{-1/2}
\le
\operatorname{Risk}_n(B_n)
\le
C_{\mathrm{risk}}\,n^{-1/2}.
\]
Therefore the stabilized zero-threshold procedure has coverage at least
\(1-\alpha\) for all sufficiently large \(n\), and its stabilized worst-case
absolute-error risk and stabilized worst-case expected length at threshold
\(0\) are both \(\asymp n^{-1/2}\).
% lean: zero_threshold_stabilizedCoverage_eventually, zero_threshold_stabilizedWorstRisk_asymp, zero_threshold_stabilizedWorstLength_asymp
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:one-cell-calibration}]
\begin{enumerate}
\item Put
\[
\tau_{\delta,h}(a)=\max\{0,1-|a-\delta|/h\},
\qquad
\gamma_{\delta,h}(a)=h\,\tau_{\delta,h}(a),
\]
so that, on the eventual range where \(h>0\),
\[
m_0(a)=\frac12,
\qquad
m_{1,\delta,h}(a)=\frac12+\gamma_{\delta,h}(a),
\qquad
\pi(a)=2a .
\]
Let
\[
\zeta_n=((n)\delta_n)^{-1/3}.
\]
For the infimum construction in \cref{obj:def:bandwidth}, set
\[
F_n(h)=n h^3(\delta_n+h).
\]
The bounds \(0\le \delta_n\le\bar\delta\) and \(0<\bar\delta<1\) make \(F_n\) continuous and strictly increasing on \([0,1-\bar\delta]\), with \(F_n(0)=0\).  For all sufficiently large \(n\), \(F_n(1-\bar\delta)\ge1\), so the crossing set in \cref{obj:def:bandwidth} is a nonempty interval whose left endpoint is the positive solution of \(F_n(h)=1\).  Hence, for all sufficiently large \(n\),
\[
0<h_n\le 1-\bar\delta,
\qquad
n h_n^3(\delta_n+h_n)=1 .
\]
Since \(\delta_n/n^{-1/4}\to\infty\), eventually \(n^{-1/4}\le \delta_n\), hence \(0<\delta_n\) and
\[
n\delta_n^4\ge 1 .
\]
The balance equation gives
\[
1=n h_n^3(\delta_n+h_n)\ge n h_n^3\delta_n,
\]
and therefore \(h_n\le \zeta_n\).  The inequality \(n\delta_n^4\ge1\) is exactly
\[
\zeta_n=(n\delta_n)^{-1/3}\le \delta_n .
\]
Consequently \(h_n\le\delta_n\), and then
\[
1=n h_n^3(\delta_n+h_n)\le 2n\delta_n h_n^3,
\]
so
\[
2^{-1/3}\zeta_n\le h_n\le\zeta_n
\]
eventually.  Also \(\delta_n+h_n\le\bar\delta+(1-\bar\delta)=1\), and the same balance equation forces \(h_n\le1/4\) eventually, since \(h_n>1/4\) and \(n\ge257\) would imply
\[
1=n h_n^3(\delta_n+h_n)\ge n h_n^4>1 .
\]
Thus
\[
h_n\asymp (n\delta_n)^{-1/3}.
\]
This is the one-stratum \(\beta=\kappa=1\) far-edge instance of the bandwidth
comparison in \cref{obj:lem:bandwidth-phases}, specialized to the density
\(\pi(a)=2a\) and the condition \(\delta_n/n^{-1/4}\to\infty\).
% lean: oneCellBandwidth_far

\item For all sufficiently large \(n\), the support of
\(\gamma_{\delta_n,h_n}\) is contained in
\([\delta_n-h_n,\delta_n+h_n]\subset[0,1]\).  On the left and right halves of
this interval,
\[
\gamma_{\delta_n,h_n}(a)=a-\delta_n+h_n
\quad(\delta_n-h_n\le a\le\delta_n),
\]
and
\[
\gamma_{\delta_n,h_n}(a)=\delta_n+h_n-a
\quad(\delta_n\le a\le\delta_n+h_n).
\]
Hence
\[
\begin{aligned}
\int_0^1 \gamma_{\delta_n,h_n}(a)^2\,2a\,da
&=
\int_0^{h_n}2(\delta_n-u)(h_n-u)^2\,du
+\int_0^{h_n}2(\delta_n+u)(h_n-u)^2\,du  \\
&=
4\delta_n\int_0^{h_n}(h_n-u)^2\,du
=
\frac43\,\delta_n h_n^3 .
\end{aligned}
\]
% lean: oneCellTentSqIntegral_eq

\item Because \(0\le\tau_{\delta_n,h_n}\le1\) and \(h_n\le1/4\) eventually,
\[
|\gamma_{\delta_n,h_n}(a)|\le\frac14
\qquad (0\le a\le1).
\]
For every real \(g\) with \(|g|\le1/4\), the Bernoulli comparison used here is
\[
2g^2
\le
\operatorname{KL}\!\left(\operatorname{Bernoulli}\!\left(\frac12+g\right),
\operatorname{Bernoulli}\!\left(\frac12\right)\right)
\le
4g^2 .
\]
The lower bound follows from Pinsker applied to the one-point event
\(\{1\}\), and the upper bound follows from the exact Bernoulli KL formula on
the interval \([1/4,3/4]\).  Integrating this display against the nonnegative
density \(2a\) and using the local perturbation-energy computation gives
\[
\frac83\,n\delta_n h_n^3
\le
K_n
\le
\frac{16}{3}\,n\delta_n h_n^3
\]
eventually.  Therefore
\[
K_n\asymp n\delta_n h_n^3 .
\]
% lean: oneCellProductKL_bounds

\item The collapsed atom mass is
\[
\int_0^{\delta_n}2a\,da=\delta_n^2,
\]
and \(\gamma_{\delta_n,h_n}(\delta_n)=h_n\).  The retained-course contribution
is supported on \((\delta_n,\delta_n+h_n]\), so
\[
\begin{aligned}
\int_{\delta_n}^1
\gamma_{\delta_n,h_n}(a)\,2a\,da
&=
\int_{\delta_n}^{\delta_n+h_n}
(\delta_n+h_n-a)\,2a\,da  \\
&=
\int_0^{h_n}2(\delta_n+u)(h_n-u)\,du
=
\delta_n h_n^2+\frac13 h_n^3 .
\end{aligned}
\]
Thus
\[
\Delta_n
=
\delta_n^2h_n+\delta_nh_n^2+\frac13 h_n^3 .
\]
Since \(0<h_n\le\delta_n\) eventually,
\[
\delta_n^2h_n
\le
\Delta_n
\le
\frac73\,\delta_n^2h_n
\]
eventually, and hence
\[
\Delta_n\asymp \delta_n^2h_n .
\]
% lean: oneCellTargetSeparation_eq

\item Let
\[
\chi_n=n^{-1/10},
\qquad
\varrho_n=n^{-1/2},
\qquad
\upsilon_n=\delta_n^2\zeta_n
=\delta_n^2(n\delta_n)^{-1/3}.
\]
For all sufficiently large \(n\), \(\delta_n>0\), and direct exponent algebra
gives
\[
\frac{\upsilon_n}{\varrho_n}
=
\left(\frac{\delta_n}{\chi_n}\right)^{5/3}.
\]
Combining the bandwidth comparison from the first item with the target
comparison from the fourth item gives constants
\(\underline{\mathfrak h},\overline{\mathfrak h}>0\) such that eventually
\[
\underline{\mathfrak h}\,\upsilon_n
\le
\Delta_n
\le
\overline{\mathfrak h}\,\upsilon_n .
\]
If \(\delta_n\asymp\chi_n\), then for some
\(\underline{\mathfrak d},\overline{\mathfrak d}>0\),
\[
\underline{\mathfrak d}
\le
\frac{\delta_n}{\chi_n}
\le
\overline{\mathfrak d}
\]
eventually.  Since \(x\mapsto x^{5/3}\) is increasing on \([0,\infty)\),
\[
\underline{\mathfrak h}\,\underline{\mathfrak d}^{5/3}\varrho_n
\le
\Delta_n
\le
\overline{\mathfrak h}\,\overline{\mathfrak d}^{5/3}\varrho_n
\]
eventually, so \(\Delta_n\asymp n^{-1/2}\).

Conversely, assume \(\Delta_n\asymp\varrho_n\).  Then for some
\(\underline{\mathfrak s},\overline{\mathfrak s}>0\),
\[
\underline{\mathfrak s}\varrho_n
\le
\Delta_n
\le
\overline{\mathfrak s}\varrho_n
\]
eventually.  With
\[
\lambda_{\mathrm{lo}}
=
\left(\frac{\underline{\mathfrak s}}{\overline{\mathfrak h}}\right)^{3/5},
\qquad
\lambda_{\mathrm{hi}}
=
\left(\frac{\overline{\mathfrak s}}{\underline{\mathfrak h}}\right)^{3/5},
\]
the two-sided display for \(\Delta_n\), together with
\(\upsilon_n/\varrho_n=(\delta_n/\chi_n)^{5/3}\), yields
\[
\lambda_{\mathrm{lo}}
\le
\frac{\delta_n}{\chi_n}
\le
\lambda_{\mathrm{hi}}
\]
eventually.  Hence
\[
\lambda_{\mathrm{lo}}\chi_n
\le
\delta_n
\le
(\lambda_{\mathrm{lo}}+\lambda_{\mathrm{hi}})\chi_n
\]
eventually, which is \(\delta_n\asymp n^{-1/10}\).  This proves
\[
\delta_n\asymp n^{-1/10}
\quad\Longleftrightarrow\quad
\Delta_n\asymp n^{-1/2}.
\]
% lean: one_cell_calibration
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:causal-bridge}]
\emph{Stratum product identity and the cell formula.}
Fix \(x\in\mathcal X\).  From the regime constants and \cref{obj:def:full-data-model-class},
\[
p_x=P(X=x)=P^{\mathrm F}(X=x)\ge p_{\min}>0,
\qquad
0\le \kappa,\qquad 0\le c_+,\qquad \bar\delta\le 1 .
\]
Let \(\eta_x\) be the latent-coordinate marginal of the normalized full-data
stratum law \(p_x^{-1}P^{\mathrm F}|_{\{X=x\}}\).  The exchangeability member of
\cref{obj:def:full-data-model-class}, together with \cref{obj:ass:conditional-density-law}
for the observed margin, gives the product formula
\[
p_x^{-1}P^{\mathrm F}\{X=x,\ A\in da,\ U\in du\}
=
\pi_x(a)\mathbf 1_{[0,1]}(a)\,da\,\eta_x(du).
\]
Consequently, for every Borel \(B\subseteq[0,1]\), consistency on the common
full-data event gives
\[
\begin{aligned}
p_x^{-1}\int_{\{X=x,\ A\in B\}}Y\,dP^{\mathrm F}
&=
\int_B\int_{\mathcal U}g_x(a,u)\,\eta_x(du)\,\pi_x(a)\,da  \\
&=
\int_B m_x^{\mathrm F}(a)\pi_x(a)\,da .
\end{aligned}
\]
The measurability, boundedness, and restriction-to-\([0,1]\) bookkeeping in
this calculation is supplied by the jointly measurable structural response,
the simultaneous consistency event, and the bounded \([0,1]\)-valued response
range in \cref{obj:def:full-data-model-class}. % lean: fullDataResponseMean_setIntegral

\emph{Pointwise identification of the threshold-range regression.}
For every Borel \(B_\circ\subseteq[0,\bar\delta]\), the stratum product formula for
\(p_x^{-1}P^{\mathrm F}\) established for this stratum and equality of the observed
margin give
\[
\int_{B_\circ}m_x^{\mathrm F}(a)\pi_x(a)\,da
=
p_x^{-1}\int_{\{X=x,\ A\in B_\circ\}}Y\,dP
=
\int_{B_\circ}\mu_x^P(a)\pi_x(a)\,da,
\]
where the second equality is the conditional-expectation version property in
\cref{obj:ass:holder-regression} combined with \cref{obj:ass:conditional-density-law}.
The products are integrable on \([0,\bar\delta]\): both regression curves are
continuous there, and the polynomial envelope in
\cref{obj:ass:polynomial-thinning} bounds \(\pi_x(a)\) by \(c_+a^\kappa\) on
\([0,1]\).  Hence
\[
\int_{B_\circ}\{m_x^{\mathrm F}(a)-\mu_x^P(a)\}\pi_x(a)\,da=0
\qquad
\text{for all Borel }B_\circ\subseteq[0,\bar\delta].
\]
Since \(c_->0\) and \(\kappa\ge0\), the same thinning envelope gives
\[
\pi_x(a)>0
\qquad\text{for Lebesgue-a.e. }a\in(0,\bar\delta].
\]
Thus \(m_x^{\mathrm F}=\mu_x^P\) Lebesgue-a.e. on \((0,\bar\delta]\).  Both
functions are continuous on \([0,\bar\delta]\), so equality extends to every
\(a\in[0,\bar\delta]\):
\[
m_x^{\mathrm F}(a)=\mu_x^P(a).
\]
This also covers the endpoint \(a=0\) by right-continuity on the closed
threshold interval. % lean: fullDataResponseMean_eq_mu

\emph{Support preservation.}
Fix \(\delta\in[0,\bar\delta]\).  The observed margin \(P\) is in
\(\mathcal M\), and the regime constants satisfy the hypotheses of
\cref{obj:prop:policy-support}.  Therefore, for each \(x\in\mathcal X\),
\[
d_\delta(a)\in
\operatorname{supp}\{\pi_x(a)\,da\text{ on }[0,1]\}
\qquad
\text{for conditional-treatment-a.e. }a .
\] % lean: clamp_policy_support

\emph{Pathwise clamp decomposition.}
Because \(\delta\in[0,\bar\delta]\subseteq[0,1]\), the common consistency event
applies both at \(A\) and at \(\delta\).  On that event, if \(\delta<A\), then
\(d_\delta(A)=A\) and \(Y(d_\delta(A))=Y\); if \(A\le\delta\), then
\(d_\delta(A)=\delta\) and \(Y(d_\delta(A))=Y(\delta)\).  Hence
\[
Y(d_\delta(A))
=
Y\mathbf 1\{\delta<A\}
+
Y(\delta)\mathbf 1\{A\le\delta\}
\qquad P^{\mathrm F}\text{-a.s.}
\] % lean: causalClamp_pathwise

\emph{The atom contribution.}
Define, for a full-data state \(\omega\),
\[
\varphi_\delta^{> }(\omega)
=
Y(\omega)\mathbf 1\{\delta<A(\omega)\},
\qquad
\varphi_\delta^{\le }(\omega)
=
Y(\delta)(\omega)\mathbf 1\{A(\omega)\le\delta\}.
\]
The bounded outcome and potential-outcome range make these integrable.  The
observed-margin identity gives
\[
\int \varphi_\delta^{>}\,dP^{\mathrm F}
=
\int Y\mathbf 1\{\delta<A\}\,dP
=
\nu_\delta(P).
\]
For the lower-tail term, partition over the finite strata and use the stratum
product formula for \(p_x^{-1}P^{\mathrm F}\) at the fixed dose \(\delta\):
\[
\begin{aligned}
\int \varphi_\delta^{\le}\,dP^{\mathrm F}
&=
\sum_{x\in\mathcal X}
\int_{\{X=x,\ A\le\delta\}}Y(\delta)\,dP^{\mathrm F}  \\
&=
\sum_{x\in\mathcal X}
p_x\left(\int_0^\delta \pi_x(a)\,da\right)m_x^{\mathrm F}(\delta) \\
&=
\sum_{x\in\mathcal X}p_xq_x(\delta)m_x^{\mathrm F}(\delta).
\end{aligned}
\] % lean: fullData_clampAtom_integral_eq

\emph{Target equality.}
Integrating the pathwise clamp decomposition of \(Y(d_\delta(A))\) and substituting
the two integrated clamp contributions \(\int\varphi_\delta^{>}\,dP^{\mathrm F}\) and
\(\int\varphi_\delta^{\le}\,dP^{\mathrm F}\) yields
\[
\psi_\delta(P^{\mathrm F})
=
\nu_\delta(P)+
\sum_{x\in\mathcal X}p_xq_x(\delta)m_x^{\mathrm F}(\delta).
\]
Since \(\delta\in[0,\bar\delta]\), the identification
\(m_x^{\mathrm F}=\mu_x^P\) on \([0,\bar\delta]\) gives
\(m_x^{\mathrm F}(\delta)=\mu_x^P(\delta)\) for every stratum, and therefore
\[
\psi_\delta(P^{\mathrm F})
=
\nu_\delta(P)+
\sum_{x\in\mathcal X}p_xq_x(\delta)\mu_x^P(\delta)
=
\theta_\delta(P).
\]
All endpoint cases in the closed range \([0,\bar\delta]\), including
\(\delta=0\) and \(\delta=\bar\delta\), are covered by the same closed-interval
hypotheses used above. % lean: causal_bridge
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:observed-margin-surjectivity}]
\begin{enumerate}
\item The regime assumptions give \(J>0\), \(p_{\min}>0\), and \(\bar\delta\le 1\).  Fix \(P\in\mathcal M\).  Define the total regression extension
\[
m(x,a)=
\begin{cases}
\mu_x^P(a), & a\in[0,1],\\
0, & a\notin[0,1].
\end{cases}
\]
The model assumptions in \cref{obj:def:model-class} give that \(m\) is Borel measurable, \(m(x,a)\in[0,1]\) for every \((x,a)\), and
\[
\mathbb E_P[Y\mid X,A]=m(X,A)\qquad P\text{-a.s.}
\]
They also give continuity of \(a\mapsto m(x,a)\) on \([0,\bar\delta]\) for every \(x\), since \(\bar\delta\le1\), and positivity of every stratum mass:
\[
0<p_{\min}\le p_x=P\{X=x\}.
\]

Let \(\zeta=(x,a)\in\mathcal X\times\mathbb R\), let
\[
\mathsf Z(o)=(o_X,o_A),
\qquad
\nu=P\circ\mathsf Z^{-1},
\]
By \citet[Theorem 6.3]{Kallenberg2002Foundations}, applied to the standard Borel design and outcome spaces, let \(K^{\mathrm{raw}}_\zeta\) be a conditional-distribution kernel of \(Y\) given \(\mathsf Z=\zeta\), so that for all Borel \(\mathsf B\subseteq\mathcal X\times\mathbb R\) and \(\mathsf C\subseteq\mathbb R\),
\[
P\{\mathsf Z\in\mathsf B,\ Y\in\mathsf C\}
=
\int_{\mathsf B}K^{\mathrm{raw}}_\zeta(\mathsf C)\,d\nu(\zeta).
\]
Put
\[
\tau(y)=\min\{1,\max\{0,y\}\},
\qquad
K^0_\zeta=K^{\mathrm{raw}}_\zeta\circ\tau^{-1},
\qquad
\eta(\zeta)=\int y\,K^0_\zeta(dy).
\]
Since \(Y\in[0,1]\) \(P\)-a.s.,
\[
K^{\mathrm{raw}}_\zeta=K^0_\zeta
\quad\text{for }\nu\text{-a.e. }\zeta,
\]
and the conditional-expectation identity gives
\[
\eta(\mathsf Z)=m(\mathsf Z)\qquad P\text{-a.s.},
\]
equivalently \(\eta=m\) \(\nu\)-a.e.  Hence the Borel set
\[
E=\{\zeta:\eta(\zeta)\ne m(\zeta)\}
\]
satisfies \(\nu(E)=0\).  Define the repaired kernel
\[
K^\star_\zeta=
\begin{cases}
\delta_{m(\zeta)}, & \zeta\in E,\\
K^0_\zeta, & \zeta\in E^c.
\end{cases}
\]
Then \(K^\star_\zeta\) is a probability law supported on \([0,1]\) for every \(\zeta\), \(K^\star=K^{\mathrm{raw}}\) \(\nu\)-a.e., and
\[
\int y\,K^\star_\zeta(dy)=m(\zeta)\qquad\text{for every }\zeta.
\]

Let \(\lambda_{[0,1]}\) be Lebesgue probability measure on \([0,1]\).  Realize the repaired kernel by its generalized inverse: for every \(\zeta\) and \(\upsilon\in[0,1]\), let \(\mathfrak q(\zeta,\upsilon)\in[0,1]\) be
\[
\mathfrak q(\zeta,\upsilon)
=
\sup\Bigl(\{0\}\cup
\{s\in[0,1]:K^\star_\zeta([0,s])<\upsilon\}\Bigr).
\]  The strict superlevel sets of this supremum are countable unions over rational \(s\in[0,1]\), so \((\zeta,\upsilon)\mapsto\mathfrak q(\zeta,\upsilon)\) is Borel.  The right-continuity of \(t\mapsto K^\star_\zeta((-\infty,t])\) gives the generalized-inverse identity, for every \(t\in\mathbb R\),
\[
\lambda_{[0,1]}\{\upsilon:\mathfrak q(\zeta,\upsilon)\le t\}
=
K^\star_\zeta((-\infty,t])
\]
Therefore
\[
\mathfrak q(\zeta,\cdot)_\#\lambda_{[0,1]}=K^\star_\zeta.
\]

On the product space
\[
(\mathcal X\times\mathbb R)\times[0,1],
\qquad
\nu\otimes\lambda_{[0,1]},
\]
define
\[
g_x(a,\upsilon)=\mathfrak q((x,a),\upsilon),
\qquad
Y(a)=g_X(a,U),
\qquad
Y=g_X(A,U).
\]
Because \(K^\star=K^{\mathrm{raw}}\) \(\nu\)-a.e., the law of \((X,A,Y)\) is exactly \(P\).  The displayed structural identities \(Y(a)=g_X(a,U)\) and \(Y=g_X(A,U)\) also give the simultaneous consistency condition in \cref{obj:def:full-data-model-class}.  For bounded Borel \(\varphi:\mathbb R\to\mathbb R\) and \(\chi:[0,1]\to\mathbb R\), the product construction yields, for every \(x\),
\[
p_x
\int_{\{X=x\}}\varphi(A)\chi(U)\,dP^{\mathrm F}
=
\left(\int_{\{X=x\}}\varphi(A)\,dP^{\mathrm F}\right)
\left(\int_{\{X=x\}}\chi(U)\,dP^{\mathrm F}\right),
\]
which is the exchangeability condition in \cref{obj:def:full-data-model-class}.  Finally, for \(a\in[0,\bar\delta]\),
\[
m_x^{\mathrm F}(a)
=
p_x^{-1}\int_{\{X=x\}}g_x(a,U)\,dP^{\mathrm F}
=
\int_0^1\mathfrak q((x,a),\upsilon)\,d\lambda_{[0,1]}(\upsilon)
=
\int y\,K^\star_{(x,a)}(dy)
=
m(x,a)
=
\mu_x^P(a).
\]
Thus \(a\mapsto m_x^{\mathrm F}(a)\) is continuous on \([0,\bar\delta]\).  For the remainder of the proof, write
\[
\Gamma(P^{\mathrm F})
:=
\text{the observed }(X,A,Y)\text{-margin of the full-data law package }P^{\mathrm F}.
\]
The constructed package \(P^{\mathrm F}\) belongs to \(\mathcal M^{\mathrm F}\), and its observed margin is \(P\).  Hence
\[
\forall P\in\mathcal M\quad \exists P^{\mathrm F}\in\mathcal M^{\mathrm F}
\quad\text{with}\quad \Gamma(P^{\mathrm F})=P.
\]
This proves observed-margin surjectivity. % lean: observed_margin_surjectivity

\item Fix \(n\in\mathbb N\), including the empty-sample case \(n=0\), and fix \(\delta\in[0,\bar\delta]\).  For each \(P^{\mathrm F}\in\mathcal M^{\mathrm F}\), \cref{obj:prop:causal-bridge} gives the target identity
\[
\psi_\delta(P^{\mathrm F})
=
\theta_\delta\{\Gamma(P^{\mathrm F})\}.
\]
The causal criteria in \cref{obj:def:causal-frontier-criteria} evaluate every observed-sample estimator, coverage event, and length objective under the product law \(\Gamma(P^{\mathrm F})^{\otimes n}\) of the observed margin, in the notation of \cref{obj:synth_16}.  Hence, after the displayed target identity, each causal risk, coverage probability, and expected length depends on \(P^{\mathrm F}\) only through \(\Gamma(P^{\mathrm F})\). % lean: observed_margin_surjectivity

\item Let \(T_n\) be any observed-measurable estimator with values in \([0,1]\).  The target and sampling-law identity \(\psi_\delta(P^{\mathrm F})=\theta_\delta\{\Gamma(P^{\mathrm F})\}\) gives
\[
\int
\left|T_n(z)-\psi_\delta(P^{\mathrm F})\right|\,\mathrm d\Gamma(P^{\mathrm F})^{\otimes n}(z)
=
\int
\left|T_n(z)-\theta_\delta\{\Gamma(P^{\mathrm F})\}\right|\,\mathrm d\Gamma(P^{\mathrm F})^{\otimes n}(z).
\]
Taking suprema and using the surjectivity already proved,
\[
\sup_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\int
\left|T_n(z)-\psi_\delta(P^{\mathrm F})\right|\,\mathrm d\Gamma(P^{\mathrm F})^{\otimes n}(z)
=
\sup_{P\in\mathcal M}
\int
\left|T_n(z)-\theta_\delta(P)\right|\,\mathrm dP^{\otimes n}(z).
\]
The admissible estimator class is the same on both sides, so taking the infimum over \(T_n\) and using \cref{obj:def:causal-frontier-criteria,obj:synth_3} gives
\[
R_{n,\mathrm F}^{\star}(\delta)=R_n^\star(\delta).
\] % lean: observed_margin_surjectivity

\item Let \(\mathcal C_n\) be an observed-measurable interval procedure.  The same target and product-law identities give
\[
\Gamma(P^{\mathrm F})^{\otimes n}
\{z:\psi_\delta(P^{\mathrm F})\in\mathcal C_n(z)\}
=
\Gamma(P^{\mathrm F})^{\otimes n}
\{z:\theta_\delta(\Gamma(P^{\mathrm F}))\in\mathcal C_n(z)\}.
\]
Therefore, using surjectivity again,
\[
\inf_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\Gamma(P^{\mathrm F})^{\otimes n}
\{z:\psi_\delta(P^{\mathrm F})\in\mathcal C_n(z)\}
=
\inf_{P\in\mathcal M}
P^{\otimes n}\{z:\theta_\delta(P)\in\mathcal C_n(z)\}.
\]
Thus the uniformly honest interval procedures are exactly the same for the causal and observed criteria.  Their length objectives also agree:
\[
\sup_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\int \operatorname{len}\{\mathcal C_n(z)\}\,\mathrm d\Gamma(P^{\mathrm F})^{\otimes n}(z)
=
\sup_{P\in\mathcal M}
\int \operatorname{len}\{\mathcal C_n(z)\}\,\mathrm dP^{\otimes n}(z).
\]
Taking the infimum over the common feasible class and using \cref{obj:def:causal-frontier-criteria,obj:synth_3} gives
\[
L_{n,\mathrm F}^{\star}(\delta,\alpha)=L_n^\star(\delta,\alpha).
\] % lean: observed_margin_surjectivity
\end{enumerate}

\begin{proof}[Proof of \cref{obj:thm:causal-frontier-lift}]
\begin{enumerate}
\item For standard Borel spaces \(S,T\) and every probability law \(\varrho\) on
\(S\times T\), the usual disintegration gives a probability kernel
\(K_\varrho:S\rightsquigarrow T\) such that, for measurable
\(D\subseteq S\) and \(E\subseteq T\),
\[
\varrho(D\times E)
=
\int_D K_\varrho(s,E)\,d(\varrho\circ \operatorname{pr}_S^{-1})(s).
\]
This standard Borel disintegration supplies the regular conditional law needed
for the observed-margin comparison in \cref{obj:prop:observed-margin-surjectivity}.
Let
\[
\Gamma(P^{\mathrm F})
=
\text{the observed }(X,A,Y)\text{-margin of }P^{\mathrm F}.
\]
Under the regime conditions, \cref{obj:prop:observed-margin-surjectivity} gives
surjectivity of \(\Gamma\) from \(\mathcal M^{\mathrm F}\) onto \(\mathcal M\) and,
for every \(n\) and every \(\delta\in[0,\bar\delta]\),
\[
R_{n,\mathrm F}^\star=R_n^\star,
\qquad
L_{n,\mathrm F}^\star=L_n^\star .
\]
For every \(P^{\mathrm F}\in\mathcal M^{\mathrm F}\) and every such \(\delta\),
\cref{obj:prop:causal-bridge} gives the target identity
\[
\psi_\delta(P^{\mathrm F})
=
\theta_\delta\{\Gamma(P^{\mathrm F})\}.
\]
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.standardBorelRegularConditionalLaw_mathlib

\item Let \(c_{\mathrm{risk}},C_{\mathrm{risk}}\) be the constants supplied by
\cref{obj:thm:minimax-risk}, and let \(c_{\mathrm{len}},C_{\mathrm{len}}\)
be the constants supplied by \cref{obj:thm:honest-length}.  Set
\[
c=\min\{c_{\mathrm{risk}},c_{\mathrm{len}}\},
\qquad
C=\max\{C_{\mathrm{risk}},C_{\mathrm{len}}\}.
\]
Then \(0<c<C\).  Choose \(n_\star\) so that, for every \(n\ge n_\star\),
the eventual conclusions of \cref{obj:thm:minimax-risk},
\cref{obj:thm:honest-length}, and \cref{obj:lem:bandwidth-balance} hold, and
also \(n\ge4\).  Fix such an \(n\), and write
\[
\delta=\delta_n,
\qquad
h=h_n,
\qquad
r=r_n=n^{-1/2}+\delta^{\kappa+1}h^\beta .
\]
Since \(0\le\delta\le\bar\delta\), \cref{obj:lem:bandwidth-balance} gives
\[
0<h,
\qquad
h\le 1-\bar\delta,
\qquad
n h^{2\beta+1}(\delta+h)^\kappa=1,
\]
and hence
\[
\delta+h\le \bar\delta+(1-\bar\delta)=1.
\]
Moreover \(r\ge0\). % lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.causal_frontier_lift

\item The total-Gram estimator in \cref{obj:def:total-gram-estimator} is a
measurable function of the observed sample, so
\[
z\longmapsto
\widehat\theta_n(z)
\]
is observed-sample measurable.  The stabilized interval in
\cref{obj:def:honest-interval} has measurable endpoints built from the same
finite sums and local-polynomial weights.  The center is clipped to \([0,1]\),
and the displayed radius is nonnegative, so the clipped lower endpoint is no
larger than the clipped upper endpoint.  Thus \(\mathrm{CI}_n\) is an
observed-sample measurable interval. % lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.stabilizedInterval_observedMeasurable

risks, the identity
\(\psi_\delta(P^{\mathrm F})=\theta_\delta\{\Gamma(P^{\mathrm F})\}\), and the
surjectivity of \(\Gamma\) give
\[
\begin{aligned}
&\sup_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\mathbb E_{\Gamma(P^{\mathrm F})^{\otimes n}}
\left|\widehat\theta_n-\psi_\delta(P^{\mathrm F})\right|  \\
&\qquad =
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}
\left|\widehat\theta_n-\theta_\delta(P)\right|.
\end{aligned}
\]
Combining this equality with \cref{obj:thm:minimax-risk} and the choices
\(c\le c_{\mathrm{risk}}\), \(C_{\mathrm{risk}}\le C\) yields
\[
c r
\le
R_{n,\mathrm F}^\star
\le
\sup_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\mathbb E_{\Gamma(P^{\mathrm F})^{\otimes n}}
\left|\widehat\theta_n-\psi_\delta(P^{\mathrm F})\right|
\le
C r .
\]
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.causalStabilizedWorstRisk_eq_observed

\[
\inf_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\Gamma(P^{\mathrm F})^{\otimes n}
\{\psi_\delta(P^{\mathrm F})\in \mathrm{CI}_n\}
=
\inf_{P\in\mathcal M}
P^{\otimes n}\{\theta_\delta(P)\in \mathrm{CI}_n\}.
\]
The right-hand side is at least \(1-\alpha\) by
\cref{obj:thm:honest-length}, so
\[
\inf_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\Gamma(P^{\mathrm F})^{\otimes n}
\{\psi_\delta(P^{\mathrm F})\in \mathrm{CI}_n\}
\ge 1-\alpha .
\]
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.causalStabilizedCoverage_eq_observed

\item Expected length is transported by the same observed-margin argument.  Since
\(\mathrm{CI}_n\) is a function of the observed sample, its length depends on
\(P^{\mathrm F}\) only through \(\Gamma(P^{\mathrm F})\), and the surjectivity of
\(\Gamma\) gives
\[
\sup_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\mathbb E_{\Gamma(P^{\mathrm F})^{\otimes n}}
\operatorname{len}(\mathrm{CI}_n)
=
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}
\operatorname{len}(\mathrm{CI}_n).
\]
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.causalStabilizedWorstLength_eq_confidenceWorstLength

gives
\[
0\le \operatorname{len}(\mathrm{CI}_n(z))\le 1.
\]

\[
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}
\operatorname{len}(\mathrm{CI}_n)
\le
C_{\mathrm{len}} r
\le
C r,
\]
where the last inequality uses \(C_{\mathrm{len}}\le C\) and \(r\ge0\).  Combining
this bound with the equality of the causal and observed expected-length criteria gives
\[
\sup_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\mathbb E_{\Gamma(P^{\mathrm F})^{\otimes n}}
\operatorname{len}(\mathrm{CI}_n)
\le
C r.
\]
The same inequality may be viewed in \(\mathbb R_{\ge0}\cup\{\infty\}\) through
the order-preserving embedding of finite nonnegative real numbers; this is the
extended order used below for the minimax length criterion. % lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.confidenceWorstLength_stabilizedInterval_le_ofReal

\item Finally, \cref{obj:lem:stabilized-interval-honesty} makes
\(\mathrm{CI}_n\) an admissible honest observed-sample interval, since
\(n\ge4\), \(0<h\), \(\delta\in[0,\bar\delta]\), and \(\delta+h\le1\).  Hence,
by the definition of \(L_n^\star\) and the expected-length bound just proved,
\[
L_{n,\mathrm F}^\star
=
L_n^\star
\le
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}
\operatorname{len}(\mathrm{CI}_n)
\le
C r,
\]
with the two sides read in \(\mathbb R_{\ge0}\cup\{\infty\}\) by embedding the
finite nonnegative quantities as finite extended values.  The lower bound from
\cref{obj:thm:honest-length}, together with \(c\le c_{\mathrm{len}}\) and
\(r\ge0\), gives, in the same extended order,
\[
c r
\le
c_{\mathrm{len}} r
\le
L_n^\star
=
L_{n,\mathrm F}^\star .
\]
Substituting back \(\delta=\delta_n\), \(h=h_n\), and \(r=r_n\) gives all
displayed conclusions for every \(n\ge n_\star\). % lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.causal_frontier_lift
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:continuity-causal-bridge}]
Write \(P\) for the observed margin of \(P^{\mathrm F}\), and write \(p_x=P(X=x)=P^{\mathrm F}(X=x)\).  The hypotheses in \cref{obj:def:continuity-full-data-model-class} and the design constants give
\[
J>0,\qquad \kappa\ge0,\qquad c_->0,\qquad c_-\le \kappa+1\le c_+,
\qquad 0<p_{\min}\le 1/J,
\qquad 0<\bar\delta<1.
\]
In particular, \(p_x>0\) for every stratum, \(c_+\ge0\), and \(\bar\delta\le1\).

\begin{enumerate}
\item Fix \(x\in\mathcal X\).  Let \(\mu_{x,\mathrm{cont}}^P\) be the continuity-only regression extension selected in \cref{obj:synth_2}; the observed-margin component of \cref{obj:def:continuity-full-data-model-class} supplies the membership of \(P\) in \(\mathcal M_{\mathrm{cont}}\).  For every measurable \(B\subseteq[0,\bar\delta]\), the full-data product identity from consistency, exchangeability, and the conditional-density law gives
\[
\int_B m_x^{\mathrm F}(a)\pi_x(a)\,da
=
p_x^{-1}\int_{\{X=x,\ A\in B\}}Y\,dP^{\mathrm F}.
\]
Indeed, under the normalized full-data law restricted to \(X=x\), the treatment coordinate has the conditional law with density \(\pi_x(a)\,da\) on \([0,1]\).  Its latent-coordinate marginal is the pushforward
\[
\nu_x^{U}
:=
\bigl((O,u)\mapsto u\bigr)_{\#}
\left(p_x^{-1}\,P^{\mathrm F}\!\restriction\{(O,u):X(O)=x\}\right),
\]
equivalently, for every measurable \(C\subseteq\mathcal U\),
\[
\nu_x^{U}(C)=p_x^{-1}P^{\mathrm F}\{X=x,\ U\in C\}.
\]
With this normalization, \cref{obj:ass:consistency,obj:ass:exchangeability,obj:ass:conditional-density-law} give the product factorization of the treatment and latent coordinates in stratum \(x\).  Applying that factorization to \(\mathbf 1\{a\in B\}g_x(a,u)\), and using \cref{obj:synth_1}, identifies the stratum integral with
\[
\int_B\left(\int g_x(a,u)\,d\nu_x^{U}(u)\right)\pi_x(a)\,da
=
\int_B m_x^{\mathrm F}(a)\pi_x(a)\,da .
\]
The observed-margin identity gives
\[
\int_{\{X=x,\ A\in B\}}Y\,dP^{\mathrm F}
=
\int_{\{X=x,\ A\in B\}}Y\,dP .
\]
The regression-version property of \(\mu_{x,\mathrm{cont}}^P\) in \cref{obj:synth_2}, together with the conditional-density law, gives
\[
\int_{\{X=x,\ A\in B\}}Y\,dP
=
p_x\int_B \mu_{x,\mathrm{cont}}^P(a)\pi_x(a)\,da .
\]
Combining the three displays and using \(p_x>0\),
\[
\int_B m_x^{\mathrm F}(a)\pi_x(a)\,da
=
\int_B \mu_{x,\mathrm{cont}}^P(a)\pi_x(a)\,da
\qquad
(B\subseteq[0,\bar\delta]).
\]
The two products are integrable on \([0,\bar\delta]\): \cref{obj:synth_1,obj:ass:full-data-response-continuity,obj:synth_2} give continuity of the two mean functions on this compact interval, and \cref{obj:ass:polynomial-thinning} gives the envelope \(\pi_x(a)\le c_+a^\kappa\le c_+\) almost everywhere on \([0,\bar\delta]\).  Equality of all measurable set integrals therefore gives
\[
m_x^{\mathrm F}(a)\pi_x(a)
=
\mu_{x,\mathrm{cont}}^P(a)\pi_x(a)
\quad\text{for Lebesgue-a.e. }a\in[0,\bar\delta].
\]
For Lebesgue-a.e. \(a\in(0,\bar\delta]\), \cref{obj:ass:polynomial-thinning} also gives
\[
\pi_x(a)\ge c_-a^\kappa>0 .
\]
Since the endpoint \(0\) has Lebesgue measure zero, the two continuous functions \(m_x^{\mathrm F}\) and \(\mu_{x,\mathrm{cont}}^P\) agree Lebesgue-a.e. on \([0,\bar\delta]\).  Applying the continuous-version uniqueness principle in \cref{obj:lem:continuity-regression-extension-unique},
\[
m_x^{\mathrm F}(a)=\mu_{x,\mathrm{cont}}^P(a)
\qquad
\text{for every }a\in[0,\bar\delta].
\]
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.fullDataResponseMean_eq_contRegression

\item Fix \(\delta\in[0,\bar\delta]\).  Since \(\bar\delta\le1\), also \(\delta\in[0,1]\).  On the full-data carrier define
\[
\varphi_{\delta}^{\mathrm{ret}}(\omega)
=
\begin{cases}
Y(\omega),& \delta<A(\omega),\\
0,& \delta\ge A(\omega),
\end{cases}
\qquad
\varphi_{\delta}^{\mathrm{atom}}(\omega)
=
\begin{cases}
Y(\delta)(\omega),& A(\omega)\le\delta,\\
0,& A(\omega)>\delta.
\end{cases}
\]
The common consistency null set in \cref{obj:ass:consistency}, the observed-treatment support \(A\in[0,1]\) supplied by the observed-margin component of \cref{obj:def:continuity-full-data-model-class} and transferred to the full-data law, and \cref{obj:def:clamp-policy} give
\[
Y\!\left(d_\delta(A)\right)
=
\varphi_{\delta}^{\mathrm{ret}}
+
\varphi_{\delta}^{\mathrm{atom}}
\quad
P^{\mathrm F}\text{-a.s.}
\]
On the branch \(\delta<A\), treatment support puts \(A\) in \([0,1]\), the clamp policy returns \(A\), and consistency identifies \(Y(A)\) with the observed outcome \(Y\).  On the branch \(A\le\delta\), the clamp policy returns \(\delta\), so the displayed decomposition is the corresponding algebraic split.  The retained summand is measurable because it is built from the observed outcome and the measurable threshold set \(\{A>\delta\}\).  The atom summand is measurable because the potential-outcome process in \cref{obj:def:continuity-full-data-model-class} is jointly measurable and \(\{A\le\delta\}\) is measurable.  For integrability, the observed outcome support bounds \(\varphi_{\delta}^{\mathrm{ret}}\) by one almost surely, and \cref{obj:ass:consistency} identifies \(Y(\delta)\) with the \([0,1]\)-valued structural response \(g_X(\delta,U)\) on the consistency event; hence \(\varphi_{\delta}^{\mathrm{atom}}\) is also bounded by one almost surely.
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.causalClamp_pathwise

\item The retained summand depends only on the observed coordinates.  The observed-margin identity therefore gives
\[
\int \varphi_{\delta}^{\mathrm{ret}}\,dP^{\mathrm F}
=
\int Y\,\mathbf 1\{A>\delta\}\,dP
=
\nu_\delta(P),
\]
where \(\nu_\delta(P)\) is defined in \cref{obj:def:clamp-functional}.
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.fullData_observed_setIntegral

\item For the atom summand, decompose over strata and use the conditional product law for \(p_x^{-1}P^{\mathrm F}|_{\{X=x\}}\), now with the dose argument fixed at \(\delta\) and the treatment event \(A\le\delta\):
\[
\int \varphi_{\delta}^{\mathrm{atom}}\,dP^{\mathrm F}
=
\sum_{x\in\mathcal X}
p_x
\left(\int_{[0,\delta]}\pi_x(a)\,da\right)
m_x^{\mathrm F}(\delta).
\]
By \cref{obj:def:clamp-functional},
\[
q_x(\delta)=\int_{[0,\delta]}\pi_x(a)\,da,
\]
so
\[
\int \varphi_{\delta}^{\mathrm{atom}}\,dP^{\mathrm F}
=
\sum_{x\in\mathcal X}p_x q_x(\delta)m_x^{\mathrm F}(\delta).
\]
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.fullData_clampAtom_integral_eq

\item Combining the pathwise decomposition, integrability, and the two integral identities for the retained and atom summands, and using \cref{obj:def:causal-clamp-mean},
\[
\psi_\delta(P^{\mathrm F})
=
\nu_\delta(P)
+
\sum_{x\in\mathcal X}p_x q_x(\delta)m_x^{\mathrm F}(\delta).
\]
Applying the pointwise identity \(m_x^{\mathrm F}=\mu_{x,\mathrm{cont}}^P\) on \([0,\bar\delta]\) at \(a=\delta\) gives
\[
\psi_\delta(P^{\mathrm F})
=
\nu_\delta(P)
+
\sum_{x\in\mathcal X}p_x q_x(\delta)\mu_{x,\mathrm{cont}}^P(\delta).
\]
The right-hand side is exactly \(\theta_{\delta,\mathrm{cont}}(P)\) by \cref{obj:def:continuity-clamp-functional}.  Since \(\delta\in[0,\bar\delta]\) was arbitrary, the asserted equality holds throughout the declared threshold range.
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.continuity_causal_bridge
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:continuity-observed-margin-surjectivity}]
\begin{enumerate}
\item The continuity-regime hypotheses give
\[
J>0,\quad 0\le \kappa,\quad 0<c_-,\quad c_-\le \kappa+1\le c_+,\quad
0<p_{\min}\le \frac1J,\quad 0<\bar\delta<1,
\]
and also \(0<\alpha<1/2\).  The displayed inequalities are the continuity design constants needed in \cref{obj:prop:continuity-causal-bridge}.  The strict inequality \(0<p_{\min}\), together with \cref{obj:ass:stratum-mass}, gives positive mass to every stratum of any \(P\in\mathcal M_{\mathrm{cont}}\). %
% lean: continuity_observed_margin_surjectivity

\item Fix \(P\in\mathcal M_{\mathrm{cont}}\).  Put
\[
\mathsf d_{\mathrm{obs}}(o)=(X(o),A(o)),\qquad
\nu=P\circ\mathsf d_{\mathrm{obs}}^{-1},
\]
and choose a measurable conditional-mean version \(\mu_x:\mathbb R\to\mathbb R\) whose restriction to \([0,\bar\delta]\) is the continuous extension \(\mu_{x,\mathrm{cont}}^P\) in \cref{obj:synth_2}.  Take a Borel regular conditional response kernel
\[
\mathsf K^0_{x,a}(\cdot)=\Pr_P\{Y\in\cdot\mid X=x,A=a\}
\]
for \(Y\) given \((X,A)\).  Equivalently, for Borel \(H\subseteq\mathcal X\times\mathbb R\) and \(C\subseteq\mathbb R\),
\[
P\{\mathsf d_{\mathrm{obs}}(O)\in H,\ Y\in C\}
=
\int_H \mathsf K^0_{x,a}(C)\,\nu(d(x,a)).
\]
Let
\[
\operatorname{clip}(y)=\min\{\max\{y,0\},1\},\qquad
\mathsf K^1_{x,a}=(\operatorname{clip})_{\#}\mathsf K^0_{x,a},\qquad
\rho_P(x,a)=\int y\,\mathsf K^1_{x,a}(dy).
\]
Because \(Y\in[0,1]\) almost surely, the raw conditional kernel \(\mathsf K^0\) and its clipped pushforward \(\mathsf K^1\) agree for \(\nu\)-almost every \((x,a)\), so their conditional means agree there.  The conditional-distribution formula for conditional expectation, combined with the defining conditional-expectation version in \cref{obj:def:continuity-model-class}, gives directly
\[
\rho_P(X,A)=\mu_X(A)
\qquad P\text{-almost surely}.
\]
Define the total measurable response mean
\[
\eta_P(x,a)=
\begin{cases}
\operatorname{clip}\{\mu_x(a)\}, & a\in[0,\bar\delta],\\
\rho_P(x,a), & a\notin[0,\bar\delta].
\end{cases}
\]
Then \(\eta_P(x,a)\in[0,1]\) for every \((x,a)\), and \(a\mapsto\eta_P(x,a)\) is continuous on \([0,\bar\delta]\) for every \(x\).  Moreover,
\[
\eta_P(X,A)=\rho_P(X,A)=\mathbb E_P[Y\mid X,A]
\qquad P\text{-almost surely}.
\]
On the strip, the displayed almost-sure equality and \(\rho_P\in[0,1]\) show that clipping \(\mu_X(A)\) leaves the conditional mean unchanged.  Off the strip, the equality is the definition of \(\eta_P\). %
% lean: exists_fullData_cont_lift

\item Let
\[
\mathcal Z_\eta=
\left\{(x,a):\int y\,\mathsf K^1_{x,a}(dy)\ne \eta_P(x,a)\right\}
\]
and define
\[
\mathsf K^\eta_{x,a}
=
\begin{cases}
\delta_{\eta_P(x,a)}, & (x,a)\in\mathcal Z_\eta,\\
\mathsf K^1_{x,a}, & (x,a)\notin\mathcal Z_\eta .
\end{cases}
\]
The almost-sure agreement of \(\eta_P\) with the outcome mean of \(\mathsf K^1\) gives \(\nu(\mathcal Z_\eta)=0\), so replacing \(\mathsf K^1\) by \(\mathsf K^\eta\) preserves the joint observed law.  Also
\[
\int y\,\mathsf K^\eta_{x,a}(dy)=\eta_P(x,a),
\qquad
\mathsf K^\eta_{x,a}([0,1])=1
\]
for every \((x,a)\).  With \(\lambda\) denoting Lebesgue measure on \([0,1]\), define
\[
\mathsf Q_x(a,u)
=
\sup\left\{s\in[0,1]:\mathsf K^\eta_{x,a}([0,s])<u\right\},
\qquad u\in[0,1].
\]
The strict-superlevel sets of this supremum are countable unions over rational thresholds, so \((x,a,u)\mapsto\mathsf Q_x(a,u)\) is jointly Borel measurable.  The generalized-inverse identity on left rays gives
\[
(\mathsf Q_x(a,\cdot))_{\#}\lambda=\mathsf K^\eta_{x,a}.
\]
Construct \(P^{\mathrm F}\) on the latent carrier \(\mathcal U=[0,1]\) by first sampling \((x,a,u)\sim\nu\otimes\lambda\), taking the latent variable to be \(U=u\), and setting
\[
X=x,\qquad A=a,\qquad Y=\mathsf Q_x(a,u).
\]
Define the structural response maps and potential outcomes by
\[
g_x(b,u)=\mathsf Q_x(b,u),\qquad
Y(b)=g_X(b,U)=\mathsf Q_X(b,U),\qquad b\in[0,1].
\]
The simultaneous definition gives \cref{obj:ass:consistency}.  To verify \cref{obj:ass:exchangeability}, it suffices in this finite-stratum setting to check the bounded-Borel product-moment factorization used by the construction.  For every stratum \(x\), every bounded Borel \(\varphi:\mathbb R\to\mathbb R\), and every bounded Borel \(\chi:[0,1]\to\mathbb R\), let \(S_x=\{(x',a):x'=x\}\).  The product law \(\nu\otimes\lambda\) gives
\[
\int_{\{X=x\}} \varphi(A)\chi(U)\,dP^{\mathrm F}
=
\left(\int_{S_x}\varphi(a)\,\nu(d(x',a))\right)
\left(\int_0^1\chi(u)\,d\lambda(u)\right),
\]
\[
\int_{\{X=x\}} \varphi(A)\,dP^{\mathrm F}
=
\int_{S_x}\varphi(a)\,\nu(d(x',a)),
\qquad
\int_{\{X=x\}} \chi(U)\,dP^{\mathrm F}=\nu(S_x)\int_0^1\chi(u)\,d\lambda(u),
\]
and \(P^{\mathrm F}(X=x)=\nu(S_x)\).  Substitution yields
\[
P^{\mathrm F}(X=x)
\int_{\{X=x\}} \varphi(A)\chi(U)\,dP^{\mathrm F}
=
\left(\int_{\{X=x\}} \varphi(A)\,dP^{\mathrm F}\right)
\left(\int_{\{X=x\}} \chi(U)\,dP^{\mathrm F}\right).
\]
The observed margin is \(P\): the conditional law of \(Y\) given \((X,A)=(x,a)\) is \(\mathsf K^\eta_{x,a}\), this kernel equals \(\mathsf K^1\) for \(\nu\)-almost every \((x,a)\), and \(\mathsf K^1\) agrees \(\nu\)-almost surely with the original conditional response law because \(Y\in[0,1]\) \(P\)-almost surely.  Finally, for every stratum \(x\) and every \(a\in[0,\bar\delta]\), \cref{obj:synth_1} and the quantile identity give
\[
m_x^{\mathrm F}(a)
=
P^{\mathrm F}(X=x)^{-1}
\int_{\{X=x\}}\mathsf Q_x(a,U)\,dP^{\mathrm F}
=
\int_0^1 \mathsf Q_x(a,u)\,d\lambda(u)
=
\eta_P(x,a),
\]
where \(P^{\mathrm F}(X=x)>0\) follows from \(0<p_{\min}\) and \cref{obj:ass:stratum-mass}.  Since \(a\mapsto\eta_P(x,a)\) is continuous on \([0,\bar\delta]\), \cref{obj:ass:full-data-response-continuity} holds.  Therefore
\[
P^{\mathrm F}\in\mathcal M_{\mathrm{cont}}^{\mathrm F},
\qquad
P^{\mathrm F}_{\mathrm{obs}}=P,
\]
with the observed margin understood as in \cref{obj:synth_13}.  This proves surjectivity. %
% lean: exists_fullData_quantile_lift

\item Now fix \(n\in\mathbb N\) and \(\delta\in[0,\bar\delta]\).  For every \(P^{\mathrm F}\in\mathcal M_{\mathrm{cont}}^{\mathrm F}\), the design constants recorded above allow \cref{obj:prop:continuity-causal-bridge} to identify the causal and observed continuity targets:
\[
\psi_\delta(P^{\mathrm F})
=
\theta_{\delta,\mathrm{cont}}(P^{\mathrm F}_{\mathrm{obs}}).
\]
The observed product sample law in \cref{obj:synth_16} depends on \(P^{\mathrm F}\) only through the observed margin \(P^{\mathrm F}_{\mathrm{obs}}\).  Hence, for every observed-sample estimator \(\widehat\theta\) satisfying the measurability and range requirements in \cref{obj:def:continuity-frontier-criteria},
\[
\left\{
\int \left|\widehat\theta(z)-\psi_\delta(P^{\mathrm F})\right|\,
d(P^{\mathrm F}_{\mathrm{obs}})^{\otimes n}(z):
P^{\mathrm F}\in\mathcal M_{\mathrm{cont}}^{\mathrm F}
\right\}
=
\left\{
\int \left|\widehat\theta(z)-\theta_{\delta,\mathrm{cont}}(P)\right|\,
dP^{\otimes n}(z):
P\in\mathcal M_{\mathrm{cont}}
\right\}.
\]
The inclusion from left to right uses \(P^{\mathrm F}_{\mathrm{obs}}\in\mathcal M_{\mathrm{cont}}\); the inclusion from right to left uses surjectivity of the observed-margin map \(\Gamma\) onto \(\mathcal M_{\mathrm{cont}}\).  Taking the supremum of equal value sets and then the same infimum over observed-sample estimators in \cref{obj:def:continuity-frontier-criteria} yields
\[
R_{n,\mathrm{cont},\mathrm F}^\star
=
R_{n,\mathrm{cont}}^\star .
\]
The argument also covers \(n=0\), since it uses only equality of the defining value sets. %
% lean: continuity_observed_margin_surjectivity

\item The interval criteria are handled in the same way.  For every observed-sample interval procedure \(C_n\) satisfying the measurability requirement in \cref{obj:def:continuity-frontier-criteria}, the target identity and surjectivity give the equivalence
\[
\forall P^{\mathrm F}\in\mathcal M_{\mathrm{cont}}^{\mathrm F},\quad
1-\alpha\le
(P^{\mathrm F}_{\mathrm{obs}})^{\otimes n}
\!\left\{z:\psi_\delta(P^{\mathrm F})\in C_n(z)\right\}
\]
if and only if
\[
\forall P\in\mathcal M_{\mathrm{cont}},\quad
1-\alpha\le
P^{\otimes n}
\!\left\{z:\theta_{\delta,\mathrm{cont}}(P)\in C_n(z)\right\}.
\]
Using the pointwise interval length from \cref{obj:synth_17}, the same observed-margin argument gives equality of expected-length value sets:
\[
\left\{
\int \operatorname{len}\{C_n(z)\}\,d(P^{\mathrm F}_{\mathrm{obs}})^{\otimes n}(z):
P^{\mathrm F}\in\mathcal M_{\mathrm{cont}}^{\mathrm F}
\right\}
=
\left\{
\int \operatorname{len}\{C_n(z)\}\,dP^{\otimes n}(z):
P\in\mathcal M_{\mathrm{cont}}
\right\}.
\]
Taking the supremum over laws and then the infimum over honest interval procedures in \cref{obj:def:continuity-frontier-criteria} yields
\[
L_{n,\mathrm{cont},\mathrm F}^\star
=
L_{n,\mathrm{cont}}^\star .
\]
Together with the risk equality, this proves the two displayed identities for every \(n\in\mathbb N\) and every \(\delta\in[0,\bar\delta]\). %
% lean: continuity_observed_margin_surjectivity
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:continuity-only-frontier}]
\emph{1. Lower experiments for the observed criteria.}
For \(n\geq1\) and a threshold sequence write
\[
r_{n,\mathrm{root}}=(n:\mathbb R)^{-1/2},
\qquad
a_{n,\mathrm{atom}}=\delta_n^{\kappa+1},
\qquad
s_n=r_{n,\mathrm{root}}+a_{n,\mathrm{atom}},
\]
as in \cref{obj:def:continuity-frontier}.  The lower-bound witnesses all use the same design:
\[
\Pr(X=x)=\frac1J,\qquad
\pi_x(a)=(\kappa+1)a^\kappa,\quad 0\leq a\leq1 .
\]
Since \(0<J\), \(0\leq\kappa\), \(0<p_{\min}\leq1/J\), and
\(c_-\leq\kappa+1\leq c_+\), this design satisfies the stratum-mass and thinning requirements in
\cref{obj:def:continuity-model-class}.  For the root term, set
\[
Q_{\mathrm{root},0}(Y=1\mid X,A)=\frac12,\qquad
Q_{\mathrm{root},1}(Y=1\mid X,A)=\frac12+\varepsilon_n,
\qquad
\varepsilon_n=\frac14 r_{n,\mathrm{root}} .
\]
The two continuous regressions lie in \([0,1]\), their targets differ by
\[
\left|
\theta_{\delta_n,\mathrm{cont}}(Q_{\mathrm{root},1})
-\theta_{\delta_n,\mathrm{cont}}(Q_{\mathrm{root},0})
\right|
=\varepsilon_n,
\]
and the product chi-square divergence obeys
\[
1+\chi^2(Q_{\mathrm{root},1}^{\otimes n},Q_{\mathrm{root},0}^{\otimes n})
\leq \exp\{4n\varepsilon_n^2\}=\exp(1/4).
\]
For the atom term, let \(\varphi:\mathbb R\to[0,1]\) be the fixed smooth treatment bump with
\(\varphi(0)=1\) and \(\varphi(u)=0\) whenever \(|u|\geq1\).  Put \(a_\star=1/4\) and choose
\[
A_\star=8(\kappa+1)a_\star^2(\bar\delta+1)^\kappa,
\qquad
K_{\mathrm{risk}}=1+A_\star,
\qquad
h_{n,\mathrm{risk}}=\frac1{K_{\mathrm{risk}}n}.
\]
With
\[
q_{n,\mathrm{risk}}(a)=a_\star\,
\varphi\!\left(\frac{a-\delta_n}{h_{n,\mathrm{risk}}}\right),
\]
define \(Q_{\mathrm{atom},0}\) by regression \(1/2\) and \(Q_{\mathrm{atom},1}\) by regression
\(1/2+q_{n,\mathrm{risk}}(a)\).  The bump properties give a continuous perturbation with
\(0\leq q_{n,\mathrm{risk}}\leq a_\star\), support contained in
\([\delta_n-h_{n,\mathrm{risk}},\delta_n+h_{n,\mathrm{risk}}]\), and value
\(q_{n,\mathrm{risk}}(\delta_n)=a_\star\).  Therefore the atom contribution at the clamp point gives
\[
a_\star\,\delta_n^{\kappa+1}
\leq
\left|
\theta_{\delta_n,\mathrm{cont}}(Q_{\mathrm{atom},1})
-\theta_{\delta_n,\mathrm{cont}}(Q_{\mathrm{atom},0})
\right|.
\]
The same localization and boundedness give the design-integral estimate behind the Bernoulli
chi-square calculation,
\[
\mathbb E\!\bigl[4q_{n,\mathrm{risk}}(A)^2\bigr]
\leq
8(\kappa+1)a_\star^2h_{n,\mathrm{risk}}
(\delta_n+h_{n,\mathrm{risk}})^\kappa,
\]
under the displayed common design.  Since \(h_{n,\mathrm{risk}}\leq1\) and
\(\delta_n\leq\bar\delta\), the exponent is at most \(A_\star/K_{\mathrm{risk}}\leq1\), and hence
\[
1+\chi^2(Q_{\mathrm{atom},1}^{\otimes n},Q_{\mathrm{atom},0}^{\otimes n})
\leq
\exp\!\left\{n\,8(\kappa+1)a_\star^2h_{n,\mathrm{risk}}
(\delta_n+h_{n,\mathrm{risk}})^\kappa\right\}
\leq e .
\]
We use the following two-point risk bound.  For probability laws
\(\mathfrak Q_0,\mathfrak Q_1\) on the sample space and real target values
\(\vartheta_0,\vartheta_1\), every estimator \(\mathfrak t\) satisfies
\[
\max_{\iota\in\{0,1\}}
\mathbb E_{\mathfrak Q_\iota}|\mathfrak t-\vartheta_\iota|
\geq
\frac{|\vartheta_1-\vartheta_0|}{2}
\{1-\operatorname{TV}(\mathfrak Q_0,\mathfrak Q_1)\}.
\]
To verify this, fix a probability measure \(\Lambda\) dominating both laws and define the common-part measure
\[
\mathfrak Q_\wedge(E)=
\int_E
\min\left\{\frac{\mathrm d\mathfrak Q_0}{\mathrm d\Lambda},
          \frac{\mathrm d\mathfrak Q_1}{\mathrm d\Lambda}\right\}
\,\mathrm d\Lambda .
\]
Its total mass is \(1-\operatorname{TV}(\mathfrak Q_0,\mathfrak Q_1)\), and integration of
\(|\mathfrak t-\vartheta_0|+|\mathfrak t-\vartheta_1|\geq
|\vartheta_1-\vartheta_0|\) against \(\mathfrak Q_\wedge\) gives the display.  When
\(\mathfrak Q_1\ll\mathfrak Q_0\), Cauchy--Schwarz applied to the likelihood ratio gives
\[
\operatorname{TV}(\mathfrak Q_0,\mathfrak Q_1)
\leq \frac12\sqrt{\chi^2(\mathfrak Q_1,\mathfrak Q_0)}.
\]
For the root pair, the product bound gives \(\chi^2\leq e^{1/4}-1<4\); for the atom pair it gives
\(\chi^2\leq e-1<4\).  Hence the factors
\(1-\frac12\sqrt{\chi^2}\) are positive in both two-point applications.  The two target separations therefore yield constants
\(c_{\mathrm R,0},c_{\mathrm R,1}>0\) such that, eventually,
\[
c_{\mathrm R,0}r_{n,\mathrm{root}}\leq R_{n,\mathrm{cont}}^\star,
\qquad
c_{\mathrm R,1}a_{n,\mathrm{atom}}\leq R_{n,\mathrm{cont}}^\star .
\]
Taking \(c_{\mathrm R}=\min\{c_{\mathrm R,0},c_{\mathrm R,1}\}/2\) gives
\[
c_{\mathrm R}s_n\leq R_{n,\mathrm{cont}}^\star .
\]
For honest length, set
\[
\eta_\alpha=1-2\alpha,
\qquad
\chi_\alpha=\eta_\alpha^2/16,
\qquad
\tau_\alpha=\eta_\alpha/16 .
\]
For the root length witnesses, let
\(Q_{\mathrm{len},\mathrm{root},0}\) and \(Q_{\mathrm{len},\mathrm{root},1}\) be the global-shift pair with
\(\varepsilon_{n,\mathrm{len}}=\tau_\alpha r_{n,\mathrm{root}}\).  Then
\[
1+\chi^2(Q_{\mathrm{len},\mathrm{root},1}^{\otimes n},
Q_{\mathrm{len},\mathrm{root},0}^{\otimes n})
\leq
\exp(\eta_\alpha^2/64),
\qquad
\chi^2(Q_{\mathrm{len},\mathrm{root},1}^{\otimes n},
Q_{\mathrm{len},\mathrm{root},0}^{\otimes n})\leq \chi_\alpha .
\]
The second inequality follows from \(0<\eta_\alpha<1\),
\(e^x\leq1+x+x^2\) at \(x=\eta_\alpha^2/64\), and
\(x+x^2\leq\eta_\alpha^2/16\).  For the atom length witnesses, let
\(Q_{\mathrm{len},\mathrm{atom},0}\) and \(Q_{\mathrm{len},\mathrm{atom},1}\) be the localized-bump pair with the same amplitude \(a_\star\) and choose
\[
K_{\mathrm{len}}=\eta_\alpha^2+64A_\star,
\qquad
h_{n,\mathrm{len}}=\frac{\eta_\alpha^2}{K_{\mathrm{len}}n}.
\]
The localized-bump exponent satisfies
\[
n\,8(\kappa+1)a_\star^2h_{n,\mathrm{len}}
(\delta_n+h_{n,\mathrm{len}})^\kappa
\leq
\frac{A_\star\eta_\alpha^2}{K_{\mathrm{len}}}
\leq \frac{\eta_\alpha^2}{64},
\]
so the same exponential-to-polynomial calculation gives
\[
\chi^2(Q_{\mathrm{len},\mathrm{atom},1}^{\otimes n},
Q_{\mathrm{len},\mathrm{atom},0}^{\otimes n})\leq \chi_\alpha.
\]
For either length-witness pair, write the first law as \(\mathfrak Q_0\), the second as
\(\mathfrak Q_1\), and the two target values as \(\vartheta_0,\vartheta_1\).  If an interval covers
\(\vartheta_\iota\) under \(\mathfrak Q_\iota\) with probability at least \(1-\alpha\) for
\(\iota=0,1\), then total variation and the union bound give
\[
\mathfrak Q_0\{\vartheta_0,\vartheta_1\in C_n\}
\geq
1-2\alpha-\operatorname{TV}(\mathfrak Q_0,\mathfrak Q_1).
\]
Together with
\(\operatorname{TV}(\mathfrak Q_0,\mathfrak Q_1)\leq
\frac12\sqrt{\chi^2(\mathfrak Q_1,\mathfrak Q_0)}\), the displayed chi-square bounds give
\[
1-2\alpha-\frac12\sqrt{\chi^2(\mathfrak Q_1,\mathfrak Q_0)}
\geq
\eta_\alpha-\frac12\sqrt{\chi_\alpha}
=
\frac78\eta_\alpha>0 .
\]
On the event \(\{\vartheta_0,\vartheta_1\in C_n\}\), the interval length is at least
\(|\vartheta_1-\vartheta_0|\).  Thus there are \(c_{\mathrm L,0},c_{\mathrm L,1}>0\) with
\[
c_{\mathrm L,0}r_{n,\mathrm{root}}
\leq L_{n,\mathrm{cont}}^\star,
\qquad
c_{\mathrm L,1}a_{n,\mathrm{atom}}
\leq L_{n,\mathrm{cont}}^\star
\]
in the extended nonnegative-real order, where nonnegative real lower bounds are embedded as finite extended values.  With
\(c_{\mathrm L}=\min\{c_{\mathrm L,0},c_{\mathrm L,1}\}/2\), the combination is by the case split
\(c_{\mathrm L,0}r_{n,\mathrm{root}}\leq c_{\mathrm L,1}a_{n,\mathrm{atom}}\) or its complement, yielding
\[
c_{\mathrm L}s_n\leq L_{n,\mathrm{cont}}^\star .
\]
% lean: contMinimaxRisk_eventually_ge_frontier, contMinimaxLength_eventually_ge_frontier

\emph{2. Upper risk for the fixed fallback estimator.}
Fix \(P\in\mathcal M_{\mathrm{cont}}\), a sample split \(B_n=(I_0,I_1,I_2)\), and \(\delta=\delta_n\).  Define
\[
\widehat\nu_n
=
|I_0|^{-1}\sum_{i\in I_0}Y_i\mathbf 1\{A_i>\delta\},
\qquad
\widehat b_{n,x}
=
|I_1|^{-1}\sum_{i\in I_1}\mathbf 1\{X_i=x,\ A_i\leq\delta\},
\]
\[
\nu_\delta(P)=\mathbb E_P[Y\mathbf 1\{A>\delta\}],
\qquad
b_x(P,\delta)=p_xq_x(\delta),
\qquad
\mathfrak e_\delta=\frac{c_+\delta^{\kappa+1}}{\kappa+1}.
\]
The estimator in \cref{obj:def:continuity-fallback-estimator} is
\[
\widehat\theta_{n,\mathrm{cont}}
=
\Pi_{[0,1]}\!\left(\widehat\nu_n+\frac12\sum_x\widehat b_{n,x}\right),
\]
where \(\Pi_{[0,1]}(t)=\min\{1,\max\{0,t\}\}\).  By \cref{obj:def:continuity-clamp-functional}, the target atom contribution is
\[
A_{\delta,\mathrm{cont}}(P)
=
\sum_x b_x(P,\delta)\mu_{x,\mathrm{cont}}^P(\delta),
\qquad
\theta_{\delta,\mathrm{cont}}(P)=\nu_\delta(P)+A_{\delta,\mathrm{cont}}(P).
\]
Since \(\theta_{\delta,\mathrm{cont}}(P)\in[0,1]\), projection is contractive toward the target.  Adding and subtracting the fallback atom term gives
\[
\begin{aligned}
\left|\widehat\theta_{n,\mathrm{cont}}-\theta_{\delta,\mathrm{cont}}(P)\right|
&\leq
\left|\widehat\nu_n-\nu_\delta(P)\right|
+\left|\frac12\sum_x\widehat b_{n,x}
      -\sum_x b_x(P,\delta)\mu_{x,\mathrm{cont}}^P(\delta)\right|\\
&\leq
\left|\widehat\nu_n-\nu_\delta(P)\right|
+\frac12\sum_x\left|\widehat b_{n,x}-b_x(P,\delta)\right|
+\sum_x b_x(P,\delta)\left|\frac12-\mu_{x,\mathrm{cont}}^P(\delta)\right|\\
&\leq
\left|\widehat\nu_n-\nu_\delta(P)\right|
+\sum_x\left\{
\frac12\left|\widehat b_{n,x}-b_x(P,\delta)\right|
+\frac12\mathfrak e_\delta
\right\},
\end{aligned}
\]
Here
\[
0\leq b_x(P,\delta)=p_xq_x(\delta)
\leq \frac{c_+\delta^{\kappa+1}}{\kappa+1}
=\mathfrak e_\delta,
\]
by the thinning envelope integrated over \([0,\delta]\).  The retained and atom indicators are
\([0,1]\)-valued block averages, so Cauchy--Schwarz and the variance bound
\(\operatorname{Var}(V)\leq1/4\) for \(V\in[0,1]\) give
\[
\mathbb E_P\left|\widehat\nu_n-\nu_\delta(P)\right|
\leq \frac12 |I_0|^{-1/2},
\qquad
\mathbb E_P\left|\widehat b_{n,x}-b_x(P,\delta)\right|
\leq \frac12 |I_1|^{-1/2}.
\]
Integrating the pointwise bound therefore gives the explicit estimate
\[
\mathbb E_{P^{\otimes n}}
\left|\widehat\theta_{n,\mathrm{cont}}-\theta_{\delta,\mathrm{cont}}(P)\right|
\leq
\frac12 |I_0|^{-1/2}
+J\left\{\frac14 |I_1|^{-1/2}
+\frac12\frac{c_+\delta^{\kappa+1}}{\kappa+1}\right\}.
\]
For all \(n\geq8\), admissibility of \(B_n\) gives \(|I_j|^{-1/2}\leq3n^{-1/2}\) for \(j=0,1\).  Hence, uniformly over \(P\),
\[
\sup_{P\in\mathcal M_{\mathrm{cont}}}
\mathbb E_{P^{\otimes n}}
\left|\widehat\theta_{n,\mathrm{cont}}-\theta_{\delta_n,\mathrm{cont}}(P)\right|
\leq
C_{\mathrm R}s_n
\]
for a constant \(C_{\mathrm R}>0\) depending only on the declared constants.  The same estimator is an admissible competitor in \cref{obj:def:continuity-frontier-criteria}, so
\[
R_{n,\mathrm{cont}}^\star
\leq
\sup_{P\in\mathcal M_{\mathrm{cont}}}
\mathbb E_{P^{\otimes n}}
\left|\widehat\theta_{n,\mathrm{cont}}-\theta_{\delta_n,\mathrm{cont}}(P)\right|.
\]
% lean: contStabilizedWorstRisk_eventually_le_frontier, contFrontierCriteria_risk_le_stabilized

\emph{3. Coverage and length of the continuity interval.}
Let
\[
t_{0,\alpha}=\sqrt{\frac{\log(4/\alpha)}{2|I_0|}},
\qquad
t_{1,\alpha}=\sqrt{\frac{\log(4/\alpha)}{2|I_1|}},
\qquad
\widehat B_n=\sum_x\widehat b_{n,x}.
\]
The interval in \cref{obj:def:continuity-honest-interval} is
\[
\mathrm{CI}_{n,\mathrm{cont}}
=
\left[
\max\{0,\widehat\theta_{n,\mathrm{cont}}-(t_{0,\alpha}+t_{1,\alpha}+\widehat B_n/2)\},
\,
\min\{1,\widehat\theta_{n,\mathrm{cont}}+(t_{0,\alpha}+t_{1,\alpha}+\widehat B_n/2)\}
\right].
\]
The coverage argument uses the following unit-range specialization of \citep[Theorem 2]{Hoeffding1963}: an average of \(m\) independent \([0,1]\)-valued variables has two-sided deviation probability at radius \(t>0\) at most \(2\exp(-2mt^2)\).  The total empirical atom mass is itself one such average, because the strata partition the sample space:
\[
\widehat B_n
=\sum_x\widehat b_{n,x}
=|I_1|^{-1}\sum_{i\in I_1}\mathbf 1\{A_i\leq\delta\}.
\]
Its population mean is
\[
B_\delta(P)=\sum_x b_x(P,\delta)=P(A\leq\delta).
\]
Applied to the retained-course average and to this total lower-threshold atom average, with the radii in \cref{obj:def:continuity-honest-interval}, the Hoeffding bound gives
\[
P^{\otimes n}\!\left(
\left|\widehat\nu_n-\nu_\delta(P)\right|>t_{0,\alpha}
\right)\leq\frac\alpha2,
\qquad
P^{\otimes n}\!\left(
\left|\widehat B_n-B_\delta(P)\right|>t_{1,\alpha}
\right)\leq\frac\alpha2 .
\]
On the complement of the two events, set
\[
A_\delta(P)=\sum_x b_x(P,\delta)\mu_{x,\mathrm{cont}}^P(\delta).
\]
The regression values lie in \([0,1]\), so the unknown clamp atom contribution satisfies
\[
0\leq A_\delta(P)\leq B_\delta(P).
\]
Since the concentration event also gives
\(B_\delta(P)\leq \widehat B_n+t_{1,\alpha}\), the quantity
\(\widehat B_n/2-A_\delta(P)\) lies between
\(-\widehat B_n/2-t_{1,\alpha}\) and \(\widehat B_n/2\).  Hence
\[
\left|\frac12\widehat B_n-A_\delta(P)\right|
\leq t_{1,\alpha}+\widehat B_n/2.
\]
Using \cref{obj:def:continuity-clamp-functional} and the contraction of projection onto \([0,1]\)
toward \(\theta_{\delta,\mathrm{cont}}(P)\in[0,1]\), this gives
\[
\left|\widehat\theta_{n,\mathrm{cont}}-\theta_{\delta,\mathrm{cont}}(P)\right|
\leq t_{0,\alpha}+t_{1,\alpha}+\widehat B_n/2,
\]
so \(\theta_{\delta,\mathrm{cont}}(P)\in\mathrm{CI}_{n,\mathrm{cont}}\).  The union bound yields uniform coverage at least \(1-\alpha\).  Also,
\[
\operatorname{len}(\mathrm{CI}_{n,\mathrm{cont}})
\leq
2t_{0,\alpha}+2t_{1,\alpha}+\widehat B_n .
\]
Moreover, the displayed identity for \(\widehat B_n\) gives
\[
\mathbb E_{P^{\otimes n}}\widehat B_n
=B_\delta(P)
=\sum_x b_x(P,\delta)
\leq J\mathfrak e_\delta .
\]
Write \([u]_{\overline{\mathbb R}_+}\) for the canonical embedding of a nonnegative real \(u\) into the extended nonnegative reals.  Consequently, before the final uniform simplification one has the extended-real length bound
\[
\int^{+}\operatorname{len}(\mathrm{CI}_{n,\mathrm{cont}})\,dP^{\otimes n}
\leq
\left[
2\sqrt{\frac{\log(4/\alpha)}{2|I_0|}}
+2\sqrt{\frac{\log(4/\alpha)}{2|I_1|}}
+J\frac{c_+\delta^{\kappa+1}}{\kappa+1}
\right]_{\overline{\mathbb R}_+}.
\]
Since \(|I_1|^{-1/2}\geq0\), this immediately implies the deliberately weaker bound
\[
\int^{+}\operatorname{len}(\mathrm{CI}_{n,\mathrm{cont}})\,dP^{\otimes n}
\leq
\left[
2\sqrt{\frac{\log(4/\alpha)}{2|I_0|}}
+2\sqrt{\frac{\log(4/\alpha)}{2|I_1|}}
+J\left\{\frac12|I_1|^{-1/2}
+\frac{c_+\delta^{\kappa+1}}{\kappa+1}\right\}
\right]_{\overline{\mathbb R}_+}.
\]
The first display follows from the pointwise inequality
\(\operatorname{len}(\mathrm{CI}_{n,\mathrm{cont}})\leq2t_{0,\alpha}+2t_{1,\alpha}+\widehat B_n\) and the preceding expectation calculation; the second only enlarges the finite real upper bound by a nonnegative term.  With \(|I_j|^{-1/2}\leq3n^{-1/2}\) and
\(K_\alpha=\sqrt{\log(4/\alpha)/2}\), this gives a constant \(C_{\mathrm L}>0\) such that
\[
\sup_{P\in\mathcal M_{\mathrm{cont}}}
\mathbb E_{P^{\otimes n}}\operatorname{len}(\mathrm{CI}_{n,\mathrm{cont}})
\leq C_{\mathrm L}s_n
\]
and the same comparison holds in the extended nonnegative-real order after embedding the real bound as a finite extended value.
% lean: contStabilizedCoverage_ge, contStabilizedWorstLength_eventually_le_frontier

\emph{4. Transport to the causal continuity class.}
Let \(P^{\mathrm F}_{\mathrm{obs}}\) denote the observed margin in \cref{obj:synth_13}.  Since the estimator and interval are functions only of the observed coordinates,
for every nonnegative or integrable observed-sample functional \(H\),
\[
\mathbb E_{(P^{\mathrm F})^{\otimes n}}H(O_1,\ldots,O_n)
=
\mathbb E_{(P^{\mathrm F}_{\mathrm{obs}})^{\otimes n}}H(O_1,\ldots,O_n),
\]
and the analogous identity holds for observed-sample events.  The causal minimax criteria in
\cref{obj:def:continuity-frontier-criteria} are defined with the full-data product law.  For the observed-only estimator and interval used here, the preceding pushforward identity gives the corresponding fixed-procedure risk, coverage, and expected length as the same quantities computed under the observed-margin product law.  By \cref{obj:prop:continuity-causal-bridge}, every
\(P^{\mathrm F}\in\mathcal M_{\mathrm{cont}}^{\mathrm F}\) and every
\(\delta\in[0,\bar\delta]\) satisfy
\[
\psi_\delta(P^{\mathrm F})
=
\theta_{\delta,\mathrm{cont}}(P^{\mathrm F}_{\mathrm{obs}}).
\]
By \cref{obj:prop:continuity-observed-margin-surjectivity}, every
\(P\in\mathcal M_{\mathrm{cont}}\) is the observed margin of some member of
\(\mathcal M_{\mathrm{cont}}^{\mathrm F}\), and the observed and causal criteria in
\cref{obj:def:continuity-frontier-criteria} agree exactly:
\[
R_{n,\mathrm{cont},\mathrm F}^\star=R_{n,\mathrm{cont}}^\star,
\qquad
L_{n,\mathrm{cont},\mathrm F}^\star=L_{n,\mathrm{cont}}^\star .
\]
The bridge identity and observed-margin surjectivity also identify the fixed-procedure observed and causal worst-case quantities.  For the fallback estimator,
\[
\sup_{P^{\mathrm F}\in\mathcal M_{\mathrm{cont}}^{\mathrm F}}
\mathbb E_{(P^{\mathrm F}_{\mathrm{obs}})^{\otimes n}}
\left|\widehat\theta_{n,\mathrm{cont}}-\psi_{\delta_n}(P^{\mathrm F})\right|
=
\sup_{P\in\mathcal M_{\mathrm{cont}}}
\mathbb E_{P^{\otimes n}}
\left|\widehat\theta_{n,\mathrm{cont}}-\theta_{\delta_n,\mathrm{cont}}(P)\right|.
\]
For the interval,
\[
\inf_{P^{\mathrm F}\in\mathcal M_{\mathrm{cont}}^{\mathrm F}}
(P^{\mathrm F}_{\mathrm{obs}})^{\otimes n}
\{\psi_{\delta_n}(P^{\mathrm F})\in\mathrm{CI}_{n,\mathrm{cont}}\}
=
\inf_{P\in\mathcal M_{\mathrm{cont}}}
P^{\otimes n}
\{\theta_{\delta_n,\mathrm{cont}}(P)\in\mathrm{CI}_{n,\mathrm{cont}}\},
\]
and the expected lengths satisfy
\[
\sup_{P^{\mathrm F}\in\mathcal M_{\mathrm{cont}}^{\mathrm F}}
\mathbb E_{(P^{\mathrm F}_{\mathrm{obs}})^{\otimes n}}
\operatorname{len}(\mathrm{CI}_{n,\mathrm{cont}})
=
\sup_{P\in\mathcal M_{\mathrm{cont}}}
\mathbb E_{P^{\otimes n}}
\operatorname{len}(\mathrm{CI}_{n,\mathrm{cont}}).
\]
% lean: contFrontierCriteria_observed_causal_eq, contCausalStabilizedWorstRisk_eq_observed, contCausalStabilizedCoverage_eq_observed, contCausalStabilizedWorstLength_eq_observed

\emph{5. Choice of constants and eventual frontier bounds.}
Let \(c_{\mathrm R},c_{\mathrm L},C_{\mathrm R},C_{\mathrm L}\) be the constants obtained above and define
\[
c=\min\{c_{\mathrm R},c_{\mathrm L}\},
\qquad
C=\max\{C_{\mathrm R},C_{\mathrm L}\}+c+1 .
\]
Then \(0<c<C\).  Intersecting the eventual sets from the lower-risk, lower-length, upper-risk, and upper-length bounds with the sample-size threshold \(n\geq8\) gives, for every admissible threshold sequence and every admissible split sequence, all sufficiently large \(n\) satisfying; at those sample sizes the positivity of \(|I_0|\) and \(|I_1|\) used in the coverage and upper-bound arguments follows from split admissibility.
\[
c s_n\leq R_{n,\mathrm{cont}}^\star
\leq
\sup_{P\in\mathcal M_{\mathrm{cont}}}
\mathbb E_{P^{\otimes n}}
\left|\widehat\theta_{n,\mathrm{cont}}-\theta_{\delta_n,\mathrm{cont}}(P)\right|
\leq C s_n,
\]
the asserted observed coverage and length upper bounds, and, in the extended nonnegative-real order,
\[
c s_n\leq L_{n,\mathrm{cont}}^\star.
\]
The equalities and causal bounds follow from the transport identities above with the same \(c\), \(C\), estimator, and interval.

\emph{6. Elbow, zero threshold, and fixed positive thresholds.}
Since \(\kappa+1>0\), for \(n>0\),
\[
\left(n^{-1/(2(\kappa+1))}\right)^{\kappa+1}=n^{-1/2},
\]
and therefore
\[
s_n\bigl(\delta_n=n^{-1/(2(\kappa+1))}\bigr)=2n^{-1/2}\asymp n^{-1/2}.
\]
At \(\delta_n=0\), the identity \(0^{\kappa+1}=0\) gives
\[
s_n(0)=n^{-1/2}\asymp n^{-1/2}.
\]
If \(\delta_n\to\delta_0\in(0,\bar\delta]\), then continuity of \(u\mapsto u^{\kappa+1}\) at the positive point \(\delta_0\) gives
\[
\delta_n^{\kappa+1}\to\delta_0^{\kappa+1}>0 .
\]
The fixed-positive-threshold argument applies the atom-scale lower constructions directly.  For the risk criterion, there is a constant \(c_{\mathrm R,+}>0\) such that eventually
\[
c_{\mathrm R,+}\delta_n^{\kappa+1}
\leq R_{n,\mathrm{cont}}^\star .
\]
If \(R_{n,\mathrm{cont}}^\star\to0\), passing to limits in this eventual inequality would give
\(c_{\mathrm R,+}\delta_0^{\kappa+1}\leq0\), contradicting
\(c_{\mathrm R,+}\delta_0^{\kappa+1}>0\).  For honest length, the corresponding lower construction gives a constant \(c_{\mathrm L,+}>0\) with
\[
c_{\mathrm L,+}\delta_n^{\kappa+1}
\leq L_{n,\mathrm{cont}}^\star
\]
eventually in the extended nonnegative-real order.  Convergence of \(L_{n,\mathrm{cont}}^\star\) to zero would imply
\(c_{\mathrm L,+}\delta_0^{\kappa+1}\leq0\), again impossible.  Hence neither fixed-positive-threshold criterion tends to zero.
% lean: contFrontier_elbow_asymp, contFrontier_zero_asymp, contFrontierCriteria_fixed_positive_not_tendsto_zero, contFrontierLength_fixed_positive_not_tendsto_zero
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:pushforward-setup}]
Fix \(x\) and \(\delta\).  By \cref{obj:def:model-class}, the conditional treatment law in stratum \(x\) is represented on \([0,1]\) by the density \(\pi_x\).  By \cref{obj:def:clamp-policy}, the clamp map is measurable and satisfies
\[
d_\delta(a)=\delta\quad\text{for }a\leq\delta,
\qquad
d_\delta(a)=a\quad\text{for }a>\delta .
\]
Splitting the conditional treatment law at \((-\infty,\delta]\) and its complement gives
\[
P\{d_\delta(A)\in\cdot\mid X=x\}
=
P\{A\in\cdot,\ A>\delta\mid X=x\}
+
P\{A\leq\delta\mid X=x\}\,\delta_\delta .
\]% lean: clamp_pushforward_decomposition
Because this conditional treatment law is carried by \([0,1]\), the retained part is
\[
P\{A\in\cdot,\ A>\delta\mid X=x\}
=
P\{A\in\cdot,\ A\in(\delta,1]\mid X=x\}.
\]% lean: clamp_pushforward_decomposition
The same density representation, together with \(0\leq\delta\leq\bar\delta<1\), gives
\[
P\{A\leq\delta\mid X=x\}
=
\int_{[0,\delta]}\pi_x(a)\,da
=
q_x(\delta),
\]% lean: atomMass
where the last equality is the definition of \(q_x(\delta)\) in \cref{obj:def:clamp-functional}.  These identities give the asserted pushforward decomposition.

It remains to bound \(q_x(\delta)\).  The conditional-density and polynomial-thinning conditions in \cref{obj:def:model-class} give integrability of \(\pi_x\) on \([0,\delta]\) and, for Lebesgue-a.e. \(a\in[0,\delta]\),
\[
c_-a^\kappa\leq \pi_x(a)\leq c_+a^\kappa .
\]% lean: PolynomialThinning
Since \(\kappa\geq0\), the function \(a\mapsto a^\kappa\) is integrable on \([0,\delta]\).  Integrating the almost-everywhere inequalities yields
\[
c_-\int_0^\delta a^\kappa\,da
\leq
q_x(\delta)
\leq
c_+\int_0^\delta a^\kappa\,da .
\]% lean: integral_mono_ae
The regime condition \(\kappa\geq0\) implies \(-1<\kappa\), and with \(0\leq\delta\),
\[
\int_0^\delta a^\kappa\,da
=
\frac{\delta^{\kappa+1}}{\kappa+1}.
\]% lean: integral_rpow
Substitution gives
\[
\frac{c_-\,\delta^{\kappa+1}}{\kappa+1}
\leq
q_x(\delta)
\leq
\frac{c_+\,\delta^{\kappa+1}}{\kappa+1},
\]
as required.
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:policy-support}]
Fix the stratum \(x\), and let
\[
\mu_0=\operatorname{Leb}\!\restriction[0,1],
\qquad
\mu_x=\mu_0\text{ with density }\pi_x .
\]
Thus \(\mu_x\) is the conditional natural-treatment measure in stratum \(x\).  The endpoints are \(\mu_0\)-null, so
\[
\mu_0\text{-almost every }a\text{ lies in }(0,1).
\]% lean: clamp_policy_support
By \cref{obj:ass:polynomial-thinning}, for \(\mu_0\)-almost every \(a\in(0,1)\),
\[
\pi_x(a)\ge c_- a^\kappa>0,
\]% lean: PolynomialThinning
because \(c_->0\), \(\kappa\ge0\), and \(a^\kappa>0\) for \(a>0\).  Hence \(\pi_x>0\) \(\mu_0\)-almost everywhere.  Since \(\mu_x\) is obtained from \(\mu_0\) by this density, \(\mu_x\ll\mu_0\), and therefore
\[
\mu_x\text{-almost every }a\text{ lies in }(0,1).
\]% lean: withDensity_absolutelyContinuous
Fix such an \(a\), and put
\[
y=d_\delta(a)=\max\{a,\delta\},
\]
using \cref{obj:def:clamp-policy}.  Since \(0<a<1\), \(0\leq\delta\leq\bar\delta\), and \(\bar\delta<1\), we have
\[
0<y<1.
\]% lean: clamp_policy_support
Let \(U\subset\mathbb R\) be open with \(y\in U\).  Choose \(\varepsilon>0\) such that \(B(y,\varepsilon)\subset U\), and set
\[
r=\min\{\varepsilon,y,1-y\}.
\]
Then \(r>0\) and
\[
B(y,r)\subset U\cap[0,1].
\]% lean: Metric.isOpen_iff
Consequently,
\[
0<\operatorname{Leb}(B(y,r))\leq \mu_0(U).
\]% lean: Real.volume_Ioo
It remains to transfer this positivity from \(\mu_0\) to \(\mu_x\).  For a measure with density, the zero-set identity gives, for every open \(E\),
\[
\mu_x(E)=0
\quad\Longleftrightarrow\quad
\mu_0\bigl(E\cap\{z:\pi_x(z)>0\}\bigr)=0 .
\]% lean: withDensity_apply_eq_zero'
If \(\mu_x(U)=0\), this identity gives
\[
\mu_0\bigl(U\cap\{z:\pi_x(z)>0\}\bigr)=0.
\]
Because \(\pi_x>0\) \(\mu_0\)-almost everywhere, the set in this display agrees with \(U\) up to a \(\mu_0\)-null set, contradicting \(0<\mu_0(U)\).  Thus \(0<\mu_x(U)\).  Since \(U\) was arbitrary, \(d_\delta(a)\) is a support point of the conditional natural-treatment measure in stratum \(x\).  The choice of \(a\) was from a full \(\mu_x\)-measure set, which gives the claimed almost-everywhere support preservation.
\end{proof}
