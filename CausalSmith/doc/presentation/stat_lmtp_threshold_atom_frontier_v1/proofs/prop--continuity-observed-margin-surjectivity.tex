\begin{proof}[Proof of \cref{obj:prop:continuity-observed-margin-surjectivity}]
\begin{enumerate}
\item The continuity-regime hypotheses give
\[
J>0,\quad 0\le \kappa,\quad 0<c_-,\quad c_-\le \kappa+1\le c_+,\quad
0<p_{\min}\le \frac1J,\quad 0<\bar\delta<1,
\]
and also \(0<\alpha<1/2\).  The displayed inequalities are the continuity design constants needed in \cref{obj:prop:continuity-causal-bridge}.  The strict inequality \(0<p_{\min}\), together with \cref{obj:ass:stratum-mass}, gives positive mass to every stratum of any \(P\in\mathcal M_{\mathrm{cont}}\). %
% lean: continuity_observed_margin_surjectivity

\item Fix \(P\in\mathcal M_{\mathrm{cont}}\).  Put
\[
\mathsf d_{\mathrm{obs}}(o)=(X(o),A(o)),\qquad
\nu=P\circ\mathsf d_{\mathrm{obs}}^{-1},
\]
and choose a measurable conditional-mean version \(\mu_x:\mathbb R\to\mathbb R\) whose restriction to \([0,\bar\delta]\) is the continuous extension \(\mu_{x,\mathrm{cont}}^P\) in \cref{obj:synth_2}.  Take a Borel regular conditional response kernel
\[
\mathsf K^0_{x,a}(\cdot)=\Pr_P\{Y\in\cdot\mid X=x,A=a\}
\]
for \(Y\) given \((X,A)\).  Equivalently, for Borel \(H\subseteq\mathcal X\times\mathbb R\) and \(C\subseteq\mathbb R\),
\[
P\{\mathsf d_{\mathrm{obs}}(O)\in H,\ Y\in C\}
=
\int_H \mathsf K^0_{x,a}(C)\,\nu(d(x,a)).
\]
Let
\[
\operatorname{clip}(y)=\min\{\max\{y,0\},1\},\qquad
\mathsf K^1_{x,a}=(\operatorname{clip})_{\#}\mathsf K^0_{x,a},\qquad
\rho_P(x,a)=\int y\,\mathsf K^1_{x,a}(dy).
\]
Because \(Y\in[0,1]\) almost surely, the raw conditional kernel \(\mathsf K^0\) and its clipped pushforward \(\mathsf K^1\) agree for \(\nu\)-almost every \((x,a)\), so their conditional means agree there.  The conditional-distribution formula for conditional expectation, combined with the defining conditional-expectation version in \cref{obj:def:continuity-model-class}, gives directly
\[
\rho_P(X,A)=\mu_X(A)
\qquad P\text{-almost surely}.
\]
Define the total measurable response mean
\[
\eta_P(x,a)=
\begin{cases}
\operatorname{clip}\{\mu_x(a)\}, & a\in[0,\bar\delta],\\
\rho_P(x,a), & a\notin[0,\bar\delta].
\end{cases}
\]
Then \(\eta_P(x,a)\in[0,1]\) for every \((x,a)\), and \(a\mapsto\eta_P(x,a)\) is continuous on \([0,\bar\delta]\) for every \(x\).  Moreover,
\[
\eta_P(X,A)=\rho_P(X,A)=\mathbb E_P[Y\mid X,A]
\qquad P\text{-almost surely}.
\]
On the strip, the displayed almost-sure equality and \(\rho_P\in[0,1]\) show that clipping \(\mu_X(A)\) leaves the conditional mean unchanged.  Off the strip, the equality is the definition of \(\eta_P\). %
% lean: exists_fullData_cont_lift

\item Let
\[
\mathcal Z_\eta=
\left\{(x,a):\int y\,\mathsf K^1_{x,a}(dy)\ne \eta_P(x,a)\right\}
\]
and define
\[
\mathsf K^\eta_{x,a}
=
\begin{cases}
\delta_{\eta_P(x,a)}, & (x,a)\in\mathcal Z_\eta,\\
\mathsf K^1_{x,a}, & (x,a)\notin\mathcal Z_\eta .
\end{cases}
\]
The almost-sure agreement of \(\eta_P\) with the outcome mean of \(\mathsf K^1\) gives \(\nu(\mathcal Z_\eta)=0\), so replacing \(\mathsf K^1\) by \(\mathsf K^\eta\) preserves the joint observed law.  Also
\[
\int y\,\mathsf K^\eta_{x,a}(dy)=\eta_P(x,a),
\qquad
\mathsf K^\eta_{x,a}([0,1])=1
\]
for every \((x,a)\).  With \(\lambda\) denoting Lebesgue measure on \([0,1]\), define
\[
\mathsf Q_x(a,u)
=
\sup\left\{s\in[0,1]:\mathsf K^\eta_{x,a}([0,s])<u\right\},
\qquad u\in[0,1].
\]
The strict-superlevel sets of this supremum are countable unions over rational thresholds, so \((x,a,u)\mapsto\mathsf Q_x(a,u)\) is jointly Borel measurable.  The generalized-inverse identity on left rays gives
\[
(\mathsf Q_x(a,\cdot))_{\#}\lambda=\mathsf K^\eta_{x,a}.
\]
Construct \(P^{\mathrm F}\) on the latent carrier \(\mathcal U=[0,1]\) by first sampling \((x,a,u)\sim\nu\otimes\lambda\), taking the latent variable to be \(U=u\), and setting
\[
X=x,\qquad A=a,\qquad Y=\mathsf Q_x(a,u).
\]
Define the structural response maps and potential outcomes by
\[
g_x(b,u)=\mathsf Q_x(b,u),\qquad
Y(b)=g_X(b,U)=\mathsf Q_X(b,U),\qquad b\in[0,1].
\]
The simultaneous definition gives \cref{obj:ass:consistency}.  To verify \cref{obj:ass:exchangeability}, it suffices in this finite-stratum setting to check the bounded-Borel product-moment factorization used by the construction.  For every stratum \(x\), every bounded Borel \(\varphi:\mathbb R\to\mathbb R\), and every bounded Borel \(\chi:[0,1]\to\mathbb R\), let \(S_x=\{(x',a):x'=x\}\).  The product law \(\nu\otimes\lambda\) gives
\[
\int_{\{X=x\}} \varphi(A)\chi(U)\,dP^{\mathrm F}
=
\left(\int_{S_x}\varphi(a)\,\nu(d(x',a))\right)
\left(\int_0^1\chi(u)\,d\lambda(u)\right),
\]
\[
\int_{\{X=x\}} \varphi(A)\,dP^{\mathrm F}
=
\int_{S_x}\varphi(a)\,\nu(d(x',a)),
\qquad
\int_{\{X=x\}} \chi(U)\,dP^{\mathrm F}=\nu(S_x)\int_0^1\chi(u)\,d\lambda(u),
\]
and \(P^{\mathrm F}(X=x)=\nu(S_x)\).  Substitution yields
\[
P^{\mathrm F}(X=x)
\int_{\{X=x\}} \varphi(A)\chi(U)\,dP^{\mathrm F}
=
\left(\int_{\{X=x\}} \varphi(A)\,dP^{\mathrm F}\right)
\left(\int_{\{X=x\}} \chi(U)\,dP^{\mathrm F}\right).
\]
The observed margin is \(P\): the conditional law of \(Y\) given \((X,A)=(x,a)\) is \(\mathsf K^\eta_{x,a}\), this kernel equals \(\mathsf K^1\) for \(\nu\)-almost every \((x,a)\), and \(\mathsf K^1\) agrees \(\nu\)-almost surely with the original conditional response law because \(Y\in[0,1]\) \(P\)-almost surely.  Finally, for every stratum \(x\) and every \(a\in[0,\bar\delta]\), \cref{obj:synth_1} and the quantile identity give
\[
m_x^{\mathrm F}(a)
=
P^{\mathrm F}(X=x)^{-1}
\int_{\{X=x\}}\mathsf Q_x(a,U)\,dP^{\mathrm F}
=
\int_0^1 \mathsf Q_x(a,u)\,d\lambda(u)
=
\eta_P(x,a),
\]
where \(P^{\mathrm F}(X=x)>0\) follows from \(0<p_{\min}\) and \cref{obj:ass:stratum-mass}.  Since \(a\mapsto\eta_P(x,a)\) is continuous on \([0,\bar\delta]\), \cref{obj:ass:full-data-response-continuity} holds.  Therefore
\[
P^{\mathrm F}\in\mathcal M_{\mathrm{cont}}^{\mathrm F},
\qquad
P^{\mathrm F}_{\mathrm{obs}}=P,
\]
with the observed margin understood as in \cref{obj:synth_13}.  This proves surjectivity. %
% lean: exists_fullData_quantile_lift

\item Now fix \(n\in\mathbb N\) and \(\delta\in[0,\bar\delta]\).  For every \(P^{\mathrm F}\in\mathcal M_{\mathrm{cont}}^{\mathrm F}\), the design constants recorded above allow \cref{obj:prop:continuity-causal-bridge} to identify the causal and observed continuity targets:
\[
\psi_\delta(P^{\mathrm F})
=
\theta_{\delta,\mathrm{cont}}(P^{\mathrm F}_{\mathrm{obs}}).
\]
The observed product sample law in \cref{obj:synth_16} depends on \(P^{\mathrm F}\) only through the observed margin \(P^{\mathrm F}_{\mathrm{obs}}\).  Hence, for every observed-sample estimator \(\widehat\theta\) satisfying the measurability and range requirements in \cref{obj:def:continuity-frontier-criteria},
\[
\left\{
\int \left|\widehat\theta(z)-\psi_\delta(P^{\mathrm F})\right|\,
d(P^{\mathrm F}_{\mathrm{obs}})^{\otimes n}(z):
P^{\mathrm F}\in\mathcal M_{\mathrm{cont}}^{\mathrm F}
\right\}
=
\left\{
\int \left|\widehat\theta(z)-\theta_{\delta,\mathrm{cont}}(P)\right|\,
dP^{\otimes n}(z):
P\in\mathcal M_{\mathrm{cont}}
\right\}.
\]
The inclusion from left to right uses \(P^{\mathrm F}_{\mathrm{obs}}\in\mathcal M_{\mathrm{cont}}\); the inclusion from right to left uses surjectivity of the observed-margin map \(\Gamma\) onto \(\mathcal M_{\mathrm{cont}}\).  Taking the supremum of equal value sets and then the same infimum over observed-sample estimators in \cref{obj:def:continuity-frontier-criteria} yields
\[
R_{n,\mathrm{cont},\mathrm F}^\star
=
R_{n,\mathrm{cont}}^\star .
\]
The argument also covers \(n=0\), since it uses only equality of the defining value sets. %
% lean: continuity_observed_margin_surjectivity

\item The interval criteria are handled in the same way.  For every observed-sample interval procedure \(C_n\) satisfying the measurability requirement in \cref{obj:def:continuity-frontier-criteria}, the target identity and surjectivity give the equivalence
\[
\forall P^{\mathrm F}\in\mathcal M_{\mathrm{cont}}^{\mathrm F},\quad
1-\alpha\le
(P^{\mathrm F}_{\mathrm{obs}})^{\otimes n}
\!\left\{z:\psi_\delta(P^{\mathrm F})\in C_n(z)\right\}
\]
if and only if
\[
\forall P\in\mathcal M_{\mathrm{cont}},\quad
1-\alpha\le
P^{\otimes n}
\!\left\{z:\theta_{\delta,\mathrm{cont}}(P)\in C_n(z)\right\}.
\]
Using the pointwise interval length from \cref{obj:synth_17}, the same observed-margin argument gives equality of expected-length value sets:
\[
\left\{
\int \operatorname{len}\{C_n(z)\}\,d(P^{\mathrm F}_{\mathrm{obs}})^{\otimes n}(z):
P^{\mathrm F}\in\mathcal M_{\mathrm{cont}}^{\mathrm F}
\right\}
=
\left\{
\int \operatorname{len}\{C_n(z)\}\,dP^{\otimes n}(z):
P\in\mathcal M_{\mathrm{cont}}
\right\}.
\]
Taking the supremum over laws and then the infimum over honest interval procedures in \cref{obj:def:continuity-frontier-criteria} yields
\[
L_{n,\mathrm{cont},\mathrm F}^\star
=
L_{n,\mathrm{cont}}^\star .
\]
Together with the risk equality, this proves the two displayed identities for every \(n\in\mathbb N\) and every \(\delta\in[0,\bar\delta]\). %
% lean: continuity_observed_margin_surjectivity
\end{enumerate}
\end{proof}
