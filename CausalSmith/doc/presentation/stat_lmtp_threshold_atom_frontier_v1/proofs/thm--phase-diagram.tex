\begin{proof}[Proof of \cref{obj:thm:phase-diagram}]
Let
\[
\Gamma=\beta\kappa+2\beta+\kappa+1,
\qquad
\Xi=2\beta+\kappa+1,
\qquad
\zeta=\frac{\Gamma}{2\beta+1}.
\]
The regime assumptions give \(\Gamma>0\), \(\Xi>0\), and \(\zeta>0\).

\begin{enumerate}
\item The two deterministic scales satisfy
\[
\frac{\delta_{\mathrm{crit},n}}{\delta_{\mathrm{edge},n}}
=
\frac{n^{-1/(2\Gamma)}}{n^{-1/\Xi}}
=
n^{1/\Xi-1/(2\Gamma)}
\]
for every \(n\ge1\).  The exponent is positive because
\[
2\Gamma-\Xi
=
2\beta\kappa+2\beta+\kappa+1>0.
\]
Hence \(n^{1/\Xi-1/(2\Gamma)}\to\infty\), which proves
\[
\frac{\delta_{\mathrm{crit},n}}{\delta_{\mathrm{edge},n}}\to\infty .
\]
% lean: critical_scale_div_edge_scale_tendsto_top

\item Fix a deterministic threshold sequence \((\delta_n)\) with
\(\delta_n\in[0,\bar\delta]\).  Let \(h_n\) be the information-balance bandwidth
from \cref{obj:def:bandwidth}, define
\[
a_n=\delta_n^{\kappa+1}h_n^\beta,
\]
and let \(r_n\) be the frontier rate in \cref{obj:def:frontier}, so that
\(r_n=n^{-1/2}+a_n\).  By \cref{obj:lem:bandwidth-balance}, for all sufficiently
large \(n\),
\[
h_n>0,
\qquad
h_n\le 1-\bar\delta,
\qquad
n h_n^{2\beta+1}(\delta_n+h_n)^\kappa=1 .
\]
The phase comparisons below use this deterministic balance relation directly.
In the regular regime \(\delta_n/\delta_{\mathrm{crit},n}\to0\), we also have,
for all sufficiently large \(n\),
\[
\frac{\delta_n}{\delta_{\mathrm{crit},n}}<1 .
\]
If \(\delta_n=0\), then \(a_n=0\) because \(\kappa+1>0\).  If \(\delta_n>0\), set
\[
\eta_n=\bigl(n\delta_n^\kappa\bigr)^{-1/(2\beta+1)} .
\]
Since \(\delta_n+h_n\ge\delta_n\) and \(\kappa\ge0\), the balance equation implies
\[
h_n\le \eta_n .
\]
Therefore
\[
a_n
=
\delta_n^{\kappa+1}h_n^\beta
\le
\delta_n^{\kappa+1}\eta_n^\beta .
\]
The exact power computation is
\[
\frac{\delta_n^{\kappa+1}\eta_n^\beta}{n^{-1/2}}
=
\left(\frac{\delta_n}{\delta_{\mathrm{crit},n}}\right)^\zeta
\le 1 .
\]
Thus \(0\le a_n\le n^{-1/2}\) eventually, and
\[
n^{-1/2}\le r_n=n^{-1/2}+a_n\le 2n^{-1/2}
\]
eventually.  Hence \(r_n\asymp n^{-1/2}\).
% lean: regular_frontier_asymp

\item Suppose \(\delta_n/\delta_{\mathrm{crit},n}\to c_0\) with \(c_0>0\).
By the scale separation \(\delta_{\mathrm{crit},n}/\delta_{\mathrm{edge},n}\to\infty\),
\[
\frac{\delta_n}{\delta_{\mathrm{edge},n}}
=
\frac{\delta_n}{\delta_{\mathrm{crit},n}}
\frac{\delta_{\mathrm{crit},n}}{\delta_{\mathrm{edge},n}}
\to\infty .
\]
For this above-edge regime, the balance equation gives the closed-form bandwidth
order.  Indeed, for all sufficiently large \(n\),
\(\delta_n>0\) and \(\delta_{\mathrm{edge},n}\le\delta_n\).  With
\(\eta_n=(n\delta_n^\kappa)^{-1/(2\beta+1)}\), the balance equation gives
\(h_n\le \eta_n\).  The inequality \(\delta_{\mathrm{edge},n}\le\delta_n\) gives
\(\eta_n\le\delta_n\), and hence \(\delta_n+h_n\le2\delta_n\).  Substituting this
upper bound into the same balance equation yields
\[
2^{-\kappa/(2\beta+1)} \eta_n\le h_n\le \eta_n .
\]
Consequently
\[
h_n\asymp \bigl(n\delta_n^\kappa\bigr)^{-1/(2\beta+1)} .
\]
Multiplying by the nonnegative factor \(\delta_n^{\kappa+1}\) and raising the
bandwidth comparison to the positive power \(\beta\) gives
\[
a_n\asymp
\delta_n^{\kappa+1}
\left\{\bigl(n\delta_n^\kappa\bigr)^{-1/(2\beta+1)}\right\}^{\beta}.
\]
The normalized interior term satisfies
\[
\frac{
\delta_n^{\kappa+1}
\left\{\bigl(n\delta_n^\kappa\bigr)^{-1/(2\beta+1)}\right\}^{\beta}
}{n^{-1/2}}
=
\left(\frac{\delta_n}{\delta_{\mathrm{crit},n}}\right)^\zeta
\to c_0^\zeta\in(0,\infty).
\]
Consequently \(a_n\asymp n^{-1/2}\).  Since
\(r_n=n^{-1/2}+a_n\) and both summands are nonnegative, also
\[
r_n\asymp n^{-1/2}.
\]
% lean: critical_atom_and_frontier_asymp

\item Suppose \(\delta_n/\delta_{\mathrm{crit},n}\to\infty\) and
\(\delta_n\to0\).  The scale separation
\(\delta_{\mathrm{crit},n}/\delta_{\mathrm{edge},n}\to\infty\) again yields
\[
\frac{\delta_n}{\delta_{\mathrm{edge},n}}\to\infty,
\]
so the same direct balance comparison used at the critical scale gives
\[
h_n\asymp \bigl(n\delta_n^\kappa\bigr)^{-1/(2\beta+1)} .
\]
Thus
\[
a_n\asymp
\delta_n^{\kappa+1}
\bigl(n\delta_n^\kappa\bigr)^{-\beta/(2\beta+1)} .
\]
Moreover,
\[
\frac{
\delta_n^{\kappa+1}
\bigl(n\delta_n^\kappa\bigr)^{-\beta/(2\beta+1)}
}{n^{-1/2}}
=
\left(\frac{\delta_n}{\delta_{\mathrm{crit},n}}\right)^\zeta
\to\infty .
\]
Hence the root term is eventually bounded by a constant multiple of the
displayed atom scale, and
\[
r_n
=
n^{-1/2}+a_n
\asymp
\delta_n^{\kappa+1}
\bigl(n\delta_n^\kappa\bigr)^{-\beta/(2\beta+1)} .
\]
% lean: supercritical_frontier_asymp

\item Suppose \(\delta_n\to\delta_0\) with \(\delta_0\in(0,\bar\delta]\).  Since
\[
\delta_{\mathrm{crit},n}=n^{-1/(2\Gamma)}\to0,
\]
we have
\[
\frac{\delta_n}{\delta_{\mathrm{crit},n}}\to\infty .
\]
Together with the scale separation
\(\delta_{\mathrm{crit},n}/\delta_{\mathrm{edge},n}\to\infty\), this places the sequence above the
edge scale.
The same direct balance comparison used in the supercritical case gives
\[
r_n\asymp
\delta_n^{\kappa+1}
\bigl(n\delta_n^\kappa\bigr)^{-\beta/(2\beta+1)} .
\]
For all sufficiently large \(n\), \(\delta_n>0\), and
\[
\delta_n^{\kappa+1}
\bigl(n\delta_n^\kappa\bigr)^{-\beta/(2\beta+1)}
=
n^{-\beta/(2\beta+1)}\delta_n^\zeta .
\]
Because \(\delta_n^\zeta\to\delta_0^\zeta\in(0,\infty)\), it follows that
\[
r_n\asymp n^{-\beta/(2\beta+1)} .
\]
% lean: fixed_positive_frontier_asymp

\item Fix \(n\) and \(P\in\mathcal M\).  At threshold \(0\),
\cref{obj:def:clamp-functional} gives, for every stratum \(x\),
\[
q_x(0)=\int_{[0,0]}\pi_x(a)\,\mathrm da=0 .
\]
The conditional-density condition in \cref{obj:ass:conditional-density-law}
gives
\[
P(A=0,X=x)=p_x\int_{\{0\}}\pi_x(a)\,\mathrm da=0
\]
for every stratum \(x\).  Summing over the finite stratum set yields
\(P(A=0)=0\), and the treatment support encoded by \cref{obj:def:model-class}
gives \(A>0\) almost surely.  Therefore the retained term in
\cref{obj:def:clamp-functional} satisfies
\[
\nu_0(P)
=
\int Y\,\mathbf 1\{A>0\}\,dP
=
\int Y\,dP .
\]
Since every \(q_x(0)\) is zero,
\[
\theta_0(P)
=
\nu_0(P)+\sum_x p_x q_x(0)\mu_x^P(0)
=
\int Y\,dP .
\]
Finally, \(\kappa+1>0\), so \cref{obj:def:frontier} gives
\[
r_n\big|_{\delta_n=0}
=
n^{-1/2}+0^{\kappa+1}h_n^\beta
=
n^{-1/2}.
\]
% lean: zero_threshold_atom_and_frontier, zero_threshold_functional

\item Fix \(n\), \(P\in\mathcal M\), and an admissible split \(B=(I_0,I_1,I_2)\)
as in \cref{obj:synth_8}.  The conditional-density condition in
\cref{obj:ass:conditional-density-law} gives \(P(A=0)=0\), so \(A>0\) under \(P\)
almost surely.  Since the product law in
\cref{obj:synth_16} has each observed coordinate distributed as \(P\), every
coordinate satisfies \(A_i>0\) under \(P^{\otimes n}\) almost surely, and the
finite intersection of these coordinate events still has probability one.  On
that event, the finite-block average is interpreted as the totalized expression
\[
\operatorname{Avg}_I(Y)
=
\begin{cases}
|I|^{-1}\sum_{i\in I}Y_i,& |I|>0,\\
0,& |I|=0,
\end{cases}
\]
which is the displayed product \((|I|)^{-1}\sum_{i\in I}Y_i\) with the
real-inverse convention \(0^{-1}=0\).  Thus
\[
\widehat b_x
=
|I_1|^{-1}\sum_{i\in I_1}\mathbf 1\{X_i=x,\ A_i\le0\}
=
0
\qquad\text{for every }x,
\]
while
\[
\widehat\nu_n
=
|I_0|^{-1}\sum_{i\in I_0}Y_i .
\]
Substituting these identities into \cref{obj:def:total-gram-estimator} gives,
\(P^{\otimes n}\)-almost surely,
\[
\widehat\theta_n
=
\Pi_{[0,1]}
\left(
|I_0|^{-1}\sum_{i\in I_0}Y_i
+
\sum_x 0\cdot \widetilde\mu_x
\right)
=
\Pi_{[0,1]}
\left(
|I_0|^{-1}\sum_{i\in I_0}Y_i
\right).
\]
Thus the threshold-zero total-Gram estimator agrees under the product sampling
law with the clamped block-zero outcome average.
% lean: zero_threshold_totalGramEstimator

\item Apply \cref{obj:thm:honest-length} to the deterministic sequence
\(\delta_n\equiv0\) and to a deterministic admissible split sequence
\(B_\bullet=(B_n)_{n\ge1}\).  The identity \(r_n=n^{-1/2}\) at
threshold zero gives eventual worst-case coverage
\[
\inf_{P\in\mathcal M}
P^{\otimes n}\!
\left\{\theta_0(P)\in \mathrm{CI}_{n,B_n}\right\}
\ge 1-\alpha,
\]
and constants \(0<c_{\mathrm{len}}\le C_{\mathrm{len}}<\infty\) such that,
eventually, the concrete stabilized length from \cref{obj:synth_5} satisfies
\[
c_{\mathrm{len}}\,n^{-1/2}
\le
\operatorname{Length}_n(B_n)
\le
C_{\mathrm{len}}\,n^{-1/2}.
\]
Likewise, applying \cref{obj:thm:minimax-risk} to the same zero sequence and
using \(r_n=n^{-1/2}\) gives constants
\(0<c_{\mathrm{risk}}\le C_{\mathrm{risk}}<\infty\) such that, eventually, the
concrete stabilized risk from \cref{obj:synth_5} satisfies
\[
c_{\mathrm{risk}}\,n^{-1/2}
\le
\operatorname{Risk}_n(B_n)
\le
C_{\mathrm{risk}}\,n^{-1/2}.
\]
Therefore the stabilized zero-threshold procedure has coverage at least
\(1-\alpha\) for all sufficiently large \(n\), and its stabilized worst-case
absolute-error risk and stabilized worst-case expected length at threshold
\(0\) are both \(\asymp n^{-1/2}\).
% lean: zero_threshold_stabilizedCoverage_eventually, zero_threshold_stabilizedWorstRisk_asymp, zero_threshold_stabilizedWorstLength_asymp
\end{enumerate}
\end{proof}
