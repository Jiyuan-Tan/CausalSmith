\begin{proof}[Proof of \cref{obj:prop:observed-margin-surjectivity}]
\begin{enumerate}
\item The regime assumptions give \(J>0\), \(p_{\min}>0\), and \(\bar\delta\le 1\).  Fix \(P\in\mathcal M\).  Define the total regression extension
\[
m(x,a)=
\begin{cases}
\mu_x^P(a), & a\in[0,1],\\
0, & a\notin[0,1].
\end{cases}
\]
The model assumptions in \cref{obj:def:model-class} give that \(m\) is Borel measurable, \(m(x,a)\in[0,1]\) for every \((x,a)\), and
\[
\mathbb E_P[Y\mid X,A]=m(X,A)\qquad P\text{-a.s.}
\]
They also give continuity of \(a\mapsto m(x,a)\) on \([0,\bar\delta]\) for every \(x\), since \(\bar\delta\le1\), and positivity of every stratum mass:
\[
0<p_{\min}\le p_x=P\{X=x\}.
\]

Let \(\zeta=(x,a)\in\mathcal X\times\mathbb R\), let
\[
\mathsf Z(o)=(o_X,o_A),
\qquad
\nu=P\circ\mathsf Z^{-1},
\]
By \citet[Theorem 6.3]{Kallenberg2002Foundations}, applied to the standard Borel design and outcome spaces, let \(K^{\mathrm{raw}}_\zeta\) be a conditional-distribution kernel of \(Y\) given \(\mathsf Z=\zeta\), so that for all Borel \(\mathsf B\subseteq\mathcal X\times\mathbb R\) and \(\mathsf C\subseteq\mathbb R\),
\[
P\{\mathsf Z\in\mathsf B,\ Y\in\mathsf C\}
=
\int_{\mathsf B}K^{\mathrm{raw}}_\zeta(\mathsf C)\,d\nu(\zeta).
\]
Put
\[
\tau(y)=\min\{1,\max\{0,y\}\},
\qquad
K^0_\zeta=K^{\mathrm{raw}}_\zeta\circ\tau^{-1},
\qquad
\eta(\zeta)=\int y\,K^0_\zeta(dy).
\]
Since \(Y\in[0,1]\) \(P\)-a.s.,
\[
K^{\mathrm{raw}}_\zeta=K^0_\zeta
\quad\text{for }\nu\text{-a.e. }\zeta,
\]
and the conditional-expectation identity gives
\[
\eta(\mathsf Z)=m(\mathsf Z)\qquad P\text{-a.s.},
\]
equivalently \(\eta=m\) \(\nu\)-a.e.  Hence the Borel set
\[
E=\{\zeta:\eta(\zeta)\ne m(\zeta)\}
\]
satisfies \(\nu(E)=0\).  Define the repaired kernel
\[
K^\star_\zeta=
\begin{cases}
\delta_{m(\zeta)}, & \zeta\in E,\\
K^0_\zeta, & \zeta\in E^c.
\end{cases}
\]
Then \(K^\star_\zeta\) is a probability law supported on \([0,1]\) for every \(\zeta\), \(K^\star=K^{\mathrm{raw}}\) \(\nu\)-a.e., and
\[
\int y\,K^\star_\zeta(dy)=m(\zeta)\qquad\text{for every }\zeta.
\]

Let \(\lambda_{[0,1]}\) be Lebesgue probability measure on \([0,1]\).  Realize the repaired kernel by its generalized inverse: for every \(\zeta\) and \(\upsilon\in[0,1]\), let \(\mathfrak q(\zeta,\upsilon)\in[0,1]\) be
\[
\mathfrak q(\zeta,\upsilon)
=
\sup\Bigl(\{0\}\cup
\{s\in[0,1]:K^\star_\zeta([0,s])<\upsilon\}\Bigr).
\]  The strict superlevel sets of this supremum are countable unions over rational \(s\in[0,1]\), so \((\zeta,\upsilon)\mapsto\mathfrak q(\zeta,\upsilon)\) is Borel.  The right-continuity of \(t\mapsto K^\star_\zeta((-\infty,t])\) gives the generalized-inverse identity, for every \(t\in\mathbb R\),
\[
\lambda_{[0,1]}\{\upsilon:\mathfrak q(\zeta,\upsilon)\le t\}
=
K^\star_\zeta((-\infty,t])
\]
Therefore
\[
\mathfrak q(\zeta,\cdot)_\#\lambda_{[0,1]}=K^\star_\zeta.
\]

On the product space
\[
(\mathcal X\times\mathbb R)\times[0,1],
\qquad
\nu\otimes\lambda_{[0,1]},
\]
define
\[
g_x(a,\upsilon)=\mathfrak q((x,a),\upsilon),
\qquad
Y(a)=g_X(a,U),
\qquad
Y=g_X(A,U).
\]
Because \(K^\star=K^{\mathrm{raw}}\) \(\nu\)-a.e., the law of \((X,A,Y)\) is exactly \(P\).  The displayed structural identities \(Y(a)=g_X(a,U)\) and \(Y=g_X(A,U)\) also give the simultaneous consistency condition in \cref{obj:def:full-data-model-class}.  For bounded Borel \(\varphi:\mathbb R\to\mathbb R\) and \(\chi:[0,1]\to\mathbb R\), the product construction yields, for every \(x\),
\[
p_x
\int_{\{X=x\}}\varphi(A)\chi(U)\,dP^{\mathrm F}
=
\left(\int_{\{X=x\}}\varphi(A)\,dP^{\mathrm F}\right)
\left(\int_{\{X=x\}}\chi(U)\,dP^{\mathrm F}\right),
\]
which is the exchangeability condition in \cref{obj:def:full-data-model-class}.  Finally, for \(a\in[0,\bar\delta]\),
\[
m_x^{\mathrm F}(a)
=
p_x^{-1}\int_{\{X=x\}}g_x(a,U)\,dP^{\mathrm F}
=
\int_0^1\mathfrak q((x,a),\upsilon)\,d\lambda_{[0,1]}(\upsilon)
=
\int y\,K^\star_{(x,a)}(dy)
=
m(x,a)
=
\mu_x^P(a).
\]
Thus \(a\mapsto m_x^{\mathrm F}(a)\) is continuous on \([0,\bar\delta]\).  For the remainder of the proof, write
\[
\Gamma(P^{\mathrm F})
:=
\text{the observed }(X,A,Y)\text{-margin of the full-data law package }P^{\mathrm F}.
\]
The constructed package \(P^{\mathrm F}\) belongs to \(\mathcal M^{\mathrm F}\), and its observed margin is \(P\).  Hence
\[
\forall P\in\mathcal M\quad \exists P^{\mathrm F}\in\mathcal M^{\mathrm F}
\quad\text{with}\quad \Gamma(P^{\mathrm F})=P.
\]
This proves observed-margin surjectivity. % lean: observed_margin_surjectivity

\item Fix \(n\in\mathbb N\), including the empty-sample case \(n=0\), and fix \(\delta\in[0,\bar\delta]\).  For each \(P^{\mathrm F}\in\mathcal M^{\mathrm F}\), \cref{obj:prop:causal-bridge} gives the target identity
\[
\psi_\delta(P^{\mathrm F})
=
\theta_\delta\{\Gamma(P^{\mathrm F})\}.
\]
The causal criteria in \cref{obj:def:causal-frontier-criteria} evaluate every observed-sample estimator, coverage event, and length objective under the product law \(\Gamma(P^{\mathrm F})^{\otimes n}\) of the observed margin, in the notation of \cref{obj:synth_16}.  Hence, after the displayed target identity, each causal risk, coverage probability, and expected length depends on \(P^{\mathrm F}\) only through \(\Gamma(P^{\mathrm F})\). % lean: observed_margin_surjectivity

\item Let \(T_n\) be any observed-measurable estimator with values in \([0,1]\).  The target and sampling-law identity \(\psi_\delta(P^{\mathrm F})=\theta_\delta\{\Gamma(P^{\mathrm F})\}\) gives
\[
\int
\left|T_n(z)-\psi_\delta(P^{\mathrm F})\right|\,\mathrm d\Gamma(P^{\mathrm F})^{\otimes n}(z)
=
\int
\left|T_n(z)-\theta_\delta\{\Gamma(P^{\mathrm F})\}\right|\,\mathrm d\Gamma(P^{\mathrm F})^{\otimes n}(z).
\]
Taking suprema and using the surjectivity already proved,
\[
\sup_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\int
\left|T_n(z)-\psi_\delta(P^{\mathrm F})\right|\,\mathrm d\Gamma(P^{\mathrm F})^{\otimes n}(z)
=
\sup_{P\in\mathcal M}
\int
\left|T_n(z)-\theta_\delta(P)\right|\,\mathrm dP^{\otimes n}(z).
\]
The admissible estimator class is the same on both sides, so taking the infimum over \(T_n\) and using \cref{obj:def:causal-frontier-criteria,obj:synth_3} gives
\[
R_{n,\mathrm F}^{\star}(\delta)=R_n^\star(\delta).
\] % lean: observed_margin_surjectivity

\item Let \(\mathcal C_n\) be an observed-measurable interval procedure.  The same target and product-law identities give
\[
\Gamma(P^{\mathrm F})^{\otimes n}
\{z:\psi_\delta(P^{\mathrm F})\in\mathcal C_n(z)\}
=
\Gamma(P^{\mathrm F})^{\otimes n}
\{z:\theta_\delta(\Gamma(P^{\mathrm F}))\in\mathcal C_n(z)\}.
\]
Therefore, using surjectivity again,
\[
\inf_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\Gamma(P^{\mathrm F})^{\otimes n}
\{z:\psi_\delta(P^{\mathrm F})\in\mathcal C_n(z)\}
=
\inf_{P\in\mathcal M}
P^{\otimes n}\{z:\theta_\delta(P)\in\mathcal C_n(z)\}.
\]
Thus the uniformly honest interval procedures are exactly the same for the causal and observed criteria.  Their length objectives also agree:
\[
\sup_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\int \operatorname{len}\{\mathcal C_n(z)\}\,\mathrm d\Gamma(P^{\mathrm F})^{\otimes n}(z)
=
\sup_{P\in\mathcal M}
\int \operatorname{len}\{\mathcal C_n(z)\}\,\mathrm dP^{\otimes n}(z).
\]
Taking the infimum over the common feasible class and using \cref{obj:def:causal-frontier-criteria,obj:synth_3} gives
\[
L_{n,\mathrm F}^{\star}(\delta,\alpha)=L_n^\star(\delta,\alpha).
\] % lean: observed_margin_surjectivity
\end{enumerate}
\end{proof}
