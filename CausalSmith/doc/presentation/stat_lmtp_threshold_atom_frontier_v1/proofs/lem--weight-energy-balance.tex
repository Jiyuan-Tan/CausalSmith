\begin{proof}[Proof of \cref{obj:lem:weight-energy-balance}]
Set
\[
        p=\mathbb E_P\!\left[
        \mathbf 1\!\left\{X=x,\ 0\le\frac{A-\delta}{h}\le1\right\}\right],
        \qquad
        D=\frac{4(\ell+1)}{\lambda_\star^2},
        \qquad
        K=\frac{p_{\min}c_-}{2\cdot 2^\kappa}.
\]
Because \(h>0\), the event in the definition of \(p\) is
\(\{X=x,\ A\in[\delta,\delta+h]\}\).  Since
\(\delta\ge0\) and \(\delta+h\le1\), this window is contained in
\([0,1]\).  The conditional-density, stratum-mass, and polynomial-thinning
components of \cref{obj:def:model-class}, equivalently
\cref{obj:ass:conditional-density-law,obj:ass:stratum-mass,obj:ass:polynomial-thinning}, give
\[
\begin{aligned}
        p
        &=p_x\int_{[\delta,\delta+h]}\pi_x(a)\,da \\
        &\ge p_{\min}c_-\int_{\delta}^{\delta+h}a^\kappa\,da .
\end{aligned}
\]
For \(a\in[\delta+h/2,\delta+h]\), the assumptions
\(\delta\ge0\) and \(\kappa\ge0\) imply
\(a^\kappa\ge((\delta+h)/2)^\kappa\).  Hence
\[
        \int_{\delta}^{\delta+h}a^\kappa\,da
        \ge
        \int_{\delta+h/2}^{\delta+h}
        \left(\frac{\delta+h}{2}\right)^\kappa da
        =
        \left(\frac{\delta+h}{2}\right)^\kappa\frac h2 .
\]
Thus
\[
        p_{\min}c_-\left(\frac{\delta+h}{2}\right)^\kappa\frac{h}{2}
        \le p .
\]
Since \(\delta+h>0\),
\[
        \left(\frac{\delta+h}{2}\right)^\kappa
        =
        \frac{(\delta+h)^\kappa}{2^\kappa},
\]
and therefore
\[
        K\,h(\delta+h)^\kappa\le p .
\]
The constants satisfy \(K>0\), \(h>0\), and \((\delta+h)^\kappa>0\), so \(p>0\).
% lean: localWindowWeight_integral_lower

Let \(m=|I_2|\).  The admissible split-size condition in \cref{obj:synth_8}
gives \(|I_2|\ge\lfloor n/4\rfloor\), and together with \(n\ge8\) this yields
\[
        \frac n8\le m .
\]
Thus \(m>0\), and \(p>0\) gives \(mp>0\).  Let \(N_x(z)\)
be the local count in \cref{obj:def:local-design-notation}, evaluated for the
displayed \((\delta,h)\).  The count decomposition used for the weight energy is
\[
 \sum_{i\in I_2}w_{ix}(z)^2
 \le
 \frac{2D}{mp}
 +D\,\mathbf 1\!\left\{N_x(z)\le\frac{mp}{2}\right\}.
\]
Indeed, on \(\Omega_x\), \cref{obj:lem:good-gram-weights} gives
\[
        \sum_{i\in I_2}w_{ix}(z)^2\le \frac{D}{N_x(z)};
\]
on \(\Omega_x^c\), the weights in \cref{obj:def:local-design-notation} are zero.
Consequently the energy is at most \(D\) on the low-count event
\(N_x(z)\le mp/2\), while on its complement the good-design case gives
\(D/N_x(z)\le2D/(mp)\) and the \(\Omega_x^c\) case gives zero.
Under \cref{obj:synth_16,obj:ass:iid-sampling}, the variables defining
\(N_x\) on the third split block are independent Bernoulli variables with mean
\(p\).  The Bernoulli lower-tail bound gives
\[
 P^{\otimes n}\!\left\{z:N_x(z)\le\frac{mp}{2}\right\}
 \le
 2\exp\!\left(-\frac{mp}{20}\right).
\]
Integrating the pointwise decomposition gives
\[
\begin{aligned}
 \int\sum_{i\in I_2}w_{ix}(z)^2\,dP^{\otimes n}(z)
 &\le
 \frac{2D}{mp}
 +D\,P^{\otimes n}\!\left\{z:N_x(z)\le\frac{mp}{2}\right\} \\
 &\le
 \frac{2D}{mp}
 +2D\exp\!\left(-\frac{mp}{20}\right).
\end{aligned}
\]
Jensen's inequality for the square root therefore gives
\[
 \int\left(\sum_{i\in I_2}w_{ix}(z)^2\right)^{1/2}dP^{\otimes n}(z)
 \le
 \left\{
 \frac{2D}{mp}
 +2D\exp\!\left(-\frac{mp}{20}\right)
 \right\}^{1/2}.
\]
% lean: integral_sqrt_interceptWeight_energy_le

The balance condition rewrites the effective sample size as
\[
        n\,h(\delta+h)^\kappa=h^{-2\beta}.
\]
Together with \(m\ge n/8\) and \(p\ge K h(\delta+h)^\kappa\), this gives
\[
        mp\ge \frac K8 h^{-2\beta},
        \qquad
        (mp)^{-1}\le \frac8K h^{2\beta}.
\]
Also \(\exp(-t/20)\le20/t\) for \(t>0\), applied with \(t=mp\).  Therefore
\[
\begin{aligned}
 \frac{2D}{mp}+2D\exp\!\left(-\frac{mp}{20}\right)
 &\le
 42D(mp)^{-1}  \\
 &\le
 \frac{336D}{K}h^{2\beta}.
\end{aligned}
\]
Taking square roots and using \(h>0\) yields
\[
 \left\{
 \frac{2D}{mp}
 +2D\exp\!\left(-\frac{mp}{20}\right)
 \right\}^{1/2}
 \le
 \left(\frac{336D}{K}\right)^{1/2}h^\beta .
\]
% lean: balancedEffectiveSample_sqrt_le

Substituting the definitions of \(D\) and \(K\) gives exactly
\[
\int
\left(\sum_{i\in I_2} w_{ix}(z)^2\right)^{1/2}
\,dP^{\otimes n}(z)
\le
\left(
\frac{336\{4(\ell+1)/\lambda_\star^2\}}
{p_{\min}c_-/(2\cdot 2^\kappa)}
\right)^{1/2}
h^\beta .
\]
% lean: integral_sqrt_interceptWeight_energy_balanced_le
\end{proof}
