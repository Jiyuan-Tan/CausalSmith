\begin{proof}[Proof of \cref{obj:lem:stabilized-interval-honesty}]
Write \(h=h_n\), and let
\[
        t_\alpha=\sqrt{\frac{\log(12J/\alpha)}{2}},
        \qquad
        b_0=\frac{t_\alpha}{\sqrt{|I_0|}},
        \qquad
        b_1=\frac{t_\alpha}{\sqrt{|I_1|}} .
\]
The regime assumptions imply \(t_\alpha>0\), and the split-size and sample-size assumptions imply \(|I_0|,|I_1|>0\).  The center \(\widehat\theta_n\) in \cref{obj:def:total-gram-estimator} and every stratum contribution \(D_x\) in \cref{obj:def:honest-interval} are measurable functions of the observed sample.  Hence the endpoint maps
\[
        z\mapsto
        \max\!\left\{0,\widehat\theta_n(z)-b_0-\sum_xD_x(z)\right\},
        \qquad
        z\mapsto
        \min\!\left\{1,\widehat\theta_n(z)+b_0+\sum_xD_x(z)\right\}
\]
are measurable.  The radius \(b_0+\sum_xD_x\) is nonnegative and \(\widehat\theta_n(z)\in[0,1]\), so these endpoints are ordered and define an observed-sample measurable interval. % lean: honestInterval_observedMeasurable

Fix \(P\in\mathcal M\).  Define
\[
 f_0(O)=Y\mathbf 1\{A>\delta\},
 \qquad
 f_x(O)=\mathbf 1\{X=x,\ A\le\delta\}.
\]
By \cref{obj:def:clamp-functional} and the conditional-density component of \cref{obj:def:model-class},
\[
        \mathbb E_P f_0(O)=\nu_\delta(P),
        \qquad
        \mathbb E_P f_x(O)=p_xq_x(\delta).
\]
Let
\[
\mathcal B_0=
\left\{
\left|
\widehat\nu_n-\mathbb E_P f_0(O)
\right|>b_0
\right\},
\]
\[
\mathcal B_{1x}=
\left\{
\left|
\widehat b_x-P(X=x,A\le\delta)
\right|>b_1
\right\},
\]
and
\[
\mathcal B_{2x}=
\left\{
\Omega_x,
\ t_\alpha\left(\sum_{i\in I_2}w_{ix}^2\right)^{1/2}
\le
\left|
\sum_{i\in I_2}w_{ix}\{Y_i-\mu^P_{X_i}(A_i)\}
\right|
\right\}.
\]
Hoeffding's inequality on the first two split blocks gives
\[
        P^{\otimes n}(\mathcal B_0)
        \le \frac{\alpha}{6J},
        \qquad
        P^{\otimes n}(\mathcal B_{1x})
        \le \frac{\alpha}{6J}
        \quad\text{for every }x .
\]
The self-normalized weighted Hoeffding bound on the third block gives
\[
        P^{\otimes n}(\mathcal B_{2x})
        \le
        2e^{-2t_\alpha^2}
        \le
        \frac{\alpha}{6J}
        \quad\text{for every }x .
\]
% lean: honest_radius_tail_le
% lean: honest_weighted_tail_le
% lean: block_weighted_tail_on_good_gram

Let
\[
        \mathcal B
        =
        \mathcal B_0\cup
        \bigcup_x\mathcal B_{1x}\cup
        \bigcup_x\mathcal B_{2x}.
\]
Since \(J\ge1\),
\[
        P^{\otimes n}(\mathcal B)
        \le
        \frac{\alpha}{6J}
        +J\frac{\alpha}{6J}
        +J\frac{\alpha}{6J}
        \le \alpha .
\]
% lean: honestInterval_coverage_model

On \(\mathcal B^c\), the retained-course and atom-mass deviations satisfy
\[
        |\widehat\nu_n-\nu_\delta(P)|\le b_0,
        \qquad
        |\widehat b_x-p_xq_x(\delta)|\le b_1
        \quad\text{for every }x .
\]
In the notation of \cref{obj:def:local-design-notation}, set
\[
        \ell=\ell(\beta),
        \qquad
        \lambda_\star=\lambda_\star(\ell,\kappa,c_-,c_+).
\]
The regime assumptions give \(\ell\in\mathbb N\), \(\kappa\ge0\), \(c_->0\), and \(c_+>0\), so \cref{obj:lem:lambda-star-positive} yields
\[
        \lambda_\star>0 .
\]
% lean: totalGram_lambdaStar_pos
For a stratum \(x\) with \(\Omega_x\), the hypotheses \(h>0\), \(\lambda_\star>0\), and \(\Omega_x\) are therefore in force, and \cref{obj:lem:good-gram-weights} gives the reproduction identities
\[
        \sum_{i\in I_2}w_{ix}U_i^j=\mathbf 1\{j=0\},
        \qquad j=0,\ldots,\ell .
\]
With
\[
        T_x(a)=\sum_{j=0}^{\ell}
        \frac{(\mu_x^P)^{(j)}(\delta)}{j!}(a-\delta)^j,
\]
these identities imply
\[
        \sum_{i\in I_2}w_{ix}T_x(A_i)=\mu_x^P(\delta).
\]
The inactive-window weights are zero, and on active summands \(X_i=x\) and \(A_i\in[\delta,\delta+h]\subset[0,1]\).  The Hölder condition in \cref{obj:def:model-class} therefore yields
\[
\left|
\sum_{i\in I_2}w_{ix}\mu_{X_i}^P(A_i)-\mu_x^P(\delta)
\right|
\le
Lh^\beta\sum_{i\in I_2}|w_{ix}|.
\]
Since \(\mu_x^P(\delta)\in[0,1]\), projection onto \([0,1]\) is contractive around this target, and the decomposition
\[
\sum_{i\in I_2}w_{ix}Y_i-\mu_x^P(\delta)
=
\sum_{i\in I_2}w_{ix}\{Y_i-\mu_{X_i}^P(A_i)\}
+
\left(\sum_{i\in I_2}w_{ix}\mu_{X_i}^P(A_i)-\mu_x^P(\delta)\right)
\]
gives
\[
\left|
\widetilde\mu_x-\mu_x^P(\delta)
\right|
\le
Lh^\beta\sum_{i\in I_2}|w_{ix}|
+
\left|
\sum_{i\in I_2}w_{ix}\{Y_i-\mu^P_{X_i}(A_i)\}
\right|.
\]
On \(\mathcal B^c\cap\Omega_x\), the last absolute value is at most
\(t_\alpha(\sum_{i\in I_2}w_{ix}^2)^{1/2}\), so
\[
\left|
\widetilde\mu_x-\mu_x^P(\delta)
\right|
\le
B_x,
\]
where \(B_x\) is the local regression bias-plus-noise radius in \cref{obj:def:honest-interval}. % lean: localRegression_good_error_le_empirical

The two displayed deviation bounds imply the deterministic error bound
\[
        \left|
        \widehat\theta_n-\theta_\delta(P)
        \right|
        \le
        b_0+\sum_xD_x .
\]
To see the algebra, set \(a_x=p_xq_x(\delta)\) and \(\widehat a_x=\widehat b_x\).  The atom deviation gives
\[
        a_x\le \widehat a_x+b_1
        \quad\text{and}\quad
        |\widehat a_x-a_x|\le b_1 .
\]
Using \cref{obj:def:total-gram-estimator,obj:def:clamp-functional} and contraction of the projection around the target \(\theta_\delta(P)\in[0,1]\),
\[
\left|
\widehat\theta_n-\theta_\delta(P)
\right|
\le
|\widehat\nu_n-\nu_\delta(P)|
+
\sum_x
\left|
\widehat a_x\widetilde\mu_x-a_x\mu_x^P(\delta)
\right|.
\]
On \(\Omega_x\), the product decomposition
\[
\widehat a_x\widetilde\mu_x-a_x\mu_x^P(\delta)
=
(\widehat a_x-a_x)\widetilde\mu_x
+
a_x\{\widetilde\mu_x-\mu_x^P(\delta)\}
\]
and the bounds \(0\le \widetilde\mu_x\le1\), \(a_x\le\widehat a_x+b_1\), and
\(|\widetilde\mu_x-\mu_x^P(\delta)|\le B_x\) give
\[
\left|
\widehat a_x\widetilde\mu_x-a_x\mu_x^P(\delta)
\right|
\le
b_1+(\widehat b_x+b_1)B_x
=D_x .
\]
On \(\Omega_x^c\), both \(\widetilde\mu_x\) and \(\mu_x^P(\delta)\) lie in \([0,1]\), so
\[
\left|
\widehat a_x\widetilde\mu_x-a_x\mu_x^P(\delta)
\right|
\le
\max\{\widehat a_x,a_x\}
\le
\widehat b_x+b_1
=D_x .
\]
Together with \(|\widehat\nu_n-\nu_\delta(P)|\le b_0\), these are exactly the two branches of the interval contribution in \cref{obj:def:honest-interval}. % lean: totalGramEstimator_error_le_honestRadius

Because \(\theta_\delta(P)\in[0,1]\), the two displayed branches of the interval contribution place \(\theta_\delta(P)\) inside the clipped interval \(\mathrm{CI}_n(z)\) for every \(z\in\mathcal B^c\).  Hence
\[
 P^{\otimes n}\!\left\{
 \theta_\delta(P)\in \mathrm{CI}_n(O_1,\ldots,O_n)
 \right\}
 \ge
 P^{\otimes n}(\mathcal B^c)
 \ge
 1-\alpha .
\]
% lean: clampFunctional_mem_honestInterval_of_deviations
% lean: honestInterval_coverage_model

Since \(P\in\mathcal M\) was arbitrary, the same lower bound holds for every law in the stated model class, together with the measurability assertion proved above. % lean: stabilizedInterval_uniformCoverage
\end{proof}
