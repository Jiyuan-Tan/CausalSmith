\begin{proof}[Proof of \cref{obj:prop:causal-bridge}]
\emph{Stratum product identity and the cell formula.}
Fix \(x\in\mathcal X\).  From the regime constants and \cref{obj:def:full-data-model-class},
\[
p_x=P(X=x)=P^{\mathrm F}(X=x)\ge p_{\min}>0,
\qquad
0\le \kappa,\qquad 0\le c_+,\qquad \bar\delta\le 1 .
\]
Let \(\eta_x\) be the latent-coordinate marginal of the normalized full-data
stratum law \(p_x^{-1}P^{\mathrm F}|_{\{X=x\}}\).  The exchangeability member of
\cref{obj:def:full-data-model-class}, together with \cref{obj:ass:conditional-density-law}
for the observed margin, gives the product formula
\[
p_x^{-1}P^{\mathrm F}\{X=x,\ A\in da,\ U\in du\}
=
\pi_x(a)\mathbf 1_{[0,1]}(a)\,da\,\eta_x(du).
\]
Consequently, for every Borel \(B\subseteq[0,1]\), consistency on the common
full-data event gives
\[
\begin{aligned}
p_x^{-1}\int_{\{X=x,\ A\in B\}}Y\,dP^{\mathrm F}
&=
\int_B\int_{\mathcal U}g_x(a,u)\,\eta_x(du)\,\pi_x(a)\,da  \\
&=
\int_B m_x^{\mathrm F}(a)\pi_x(a)\,da .
\end{aligned}
\]
The measurability, boundedness, and restriction-to-\([0,1]\) bookkeeping in
this calculation is supplied by the jointly measurable structural response,
the simultaneous consistency event, and the bounded \([0,1]\)-valued response
range in \cref{obj:def:full-data-model-class}. % lean: fullDataResponseMean_setIntegral

\emph{Pointwise identification of the threshold-range regression.}
For every Borel \(B_\circ\subseteq[0,\bar\delta]\), the stratum product formula for
\(p_x^{-1}P^{\mathrm F}\) established for this stratum and equality of the observed
margin give
\[
\int_{B_\circ}m_x^{\mathrm F}(a)\pi_x(a)\,da
=
p_x^{-1}\int_{\{X=x,\ A\in B_\circ\}}Y\,dP
=
\int_{B_\circ}\mu_x^P(a)\pi_x(a)\,da,
\]
where the second equality is the conditional-expectation version property in
\cref{obj:ass:holder-regression} combined with \cref{obj:ass:conditional-density-law}.
The products are integrable on \([0,\bar\delta]\): both regression curves are
continuous there, and the polynomial envelope in
\cref{obj:ass:polynomial-thinning} bounds \(\pi_x(a)\) by \(c_+a^\kappa\) on
\([0,1]\).  Hence
\[
\int_{B_\circ}\{m_x^{\mathrm F}(a)-\mu_x^P(a)\}\pi_x(a)\,da=0
\qquad
\text{for all Borel }B_\circ\subseteq[0,\bar\delta].
\]
Since \(c_->0\) and \(\kappa\ge0\), the same thinning envelope gives
\[
\pi_x(a)>0
\qquad\text{for Lebesgue-a.e. }a\in(0,\bar\delta].
\]
Thus \(m_x^{\mathrm F}=\mu_x^P\) Lebesgue-a.e. on \((0,\bar\delta]\).  Both
functions are continuous on \([0,\bar\delta]\), so equality extends to every
\(a\in[0,\bar\delta]\):
\[
m_x^{\mathrm F}(a)=\mu_x^P(a).
\]
This also covers the endpoint \(a=0\) by right-continuity on the closed
threshold interval. % lean: fullDataResponseMean_eq_mu

\emph{Support preservation.}
Fix \(\delta\in[0,\bar\delta]\).  The observed margin \(P\) is in
\(\mathcal M\), and the regime constants satisfy the hypotheses of
\cref{obj:prop:policy-support}.  Therefore, for each \(x\in\mathcal X\),
\[
d_\delta(a)\in
\operatorname{supp}\{\pi_x(a)\,da\text{ on }[0,1]\}
\qquad
\text{for conditional-treatment-a.e. }a .
\] % lean: clamp_policy_support

\emph{Pathwise clamp decomposition.}
Because \(\delta\in[0,\bar\delta]\subseteq[0,1]\), the common consistency event
applies both at \(A\) and at \(\delta\).  On that event, if \(\delta<A\), then
\(d_\delta(A)=A\) and \(Y(d_\delta(A))=Y\); if \(A\le\delta\), then
\(d_\delta(A)=\delta\) and \(Y(d_\delta(A))=Y(\delta)\).  Hence
\[
Y(d_\delta(A))
=
Y\mathbf 1\{\delta<A\}
+
Y(\delta)\mathbf 1\{A\le\delta\}
\qquad P^{\mathrm F}\text{-a.s.}
\] % lean: causalClamp_pathwise

\emph{The atom contribution.}
Define, for a full-data state \(\omega\),
\[
\varphi_\delta^{> }(\omega)
=
Y(\omega)\mathbf 1\{\delta<A(\omega)\},
\qquad
\varphi_\delta^{\le }(\omega)
=
Y(\delta)(\omega)\mathbf 1\{A(\omega)\le\delta\}.
\]
The bounded outcome and potential-outcome range make these integrable.  The
observed-margin identity gives
\[
\int \varphi_\delta^{>}\,dP^{\mathrm F}
=
\int Y\mathbf 1\{\delta<A\}\,dP
=
\nu_\delta(P).
\]
For the lower-tail term, partition over the finite strata and use the stratum
product formula for \(p_x^{-1}P^{\mathrm F}\) at the fixed dose \(\delta\):
\[
\begin{aligned}
\int \varphi_\delta^{\le}\,dP^{\mathrm F}
&=
\sum_{x\in\mathcal X}
\int_{\{X=x,\ A\le\delta\}}Y(\delta)\,dP^{\mathrm F}  \\
&=
\sum_{x\in\mathcal X}
p_x\left(\int_0^\delta \pi_x(a)\,da\right)m_x^{\mathrm F}(\delta) \\
&=
\sum_{x\in\mathcal X}p_xq_x(\delta)m_x^{\mathrm F}(\delta).
\end{aligned}
\] % lean: fullData_clampAtom_integral_eq

\emph{Target equality.}
Integrating the pathwise clamp decomposition of \(Y(d_\delta(A))\) and substituting
the two integrated clamp contributions \(\int\varphi_\delta^{>}\,dP^{\mathrm F}\) and
\(\int\varphi_\delta^{\le}\,dP^{\mathrm F}\) yields
\[
\psi_\delta(P^{\mathrm F})
=
\nu_\delta(P)+
\sum_{x\in\mathcal X}p_xq_x(\delta)m_x^{\mathrm F}(\delta).
\]
Since \(\delta\in[0,\bar\delta]\), the identification
\(m_x^{\mathrm F}=\mu_x^P\) on \([0,\bar\delta]\) gives
\(m_x^{\mathrm F}(\delta)=\mu_x^P(\delta)\) for every stratum, and therefore
\[
\psi_\delta(P^{\mathrm F})
=
\nu_\delta(P)+
\sum_{x\in\mathcal X}p_xq_x(\delta)\mu_x^P(\delta)
=
\theta_\delta(P).
\]
All endpoint cases in the closed range \([0,\bar\delta]\), including
\(\delta=0\) and \(\delta=\bar\delta\), are covered by the same closed-interval
hypotheses used above. % lean: causal_bridge
\end{proof}
