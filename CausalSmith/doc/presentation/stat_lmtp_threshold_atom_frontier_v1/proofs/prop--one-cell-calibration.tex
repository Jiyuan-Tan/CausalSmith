\begin{proof}[Proof of \cref{obj:prop:one-cell-calibration}]
\begin{enumerate}
\item Put
\[
\tau_{\delta,h}(a)=\max\{0,1-|a-\delta|/h\},
\qquad
\gamma_{\delta,h}(a)=h\,\tau_{\delta,h}(a),
\]
so that, on the eventual range where \(h>0\),
\[
m_0(a)=\frac12,
\qquad
m_{1,\delta,h}(a)=\frac12+\gamma_{\delta,h}(a),
\qquad
\pi(a)=2a .
\]
Let
\[
\zeta_n=((n)\delta_n)^{-1/3}.
\]
For the infimum construction in \cref{obj:def:bandwidth}, set
\[
F_n(h)=n h^3(\delta_n+h).
\]
The bounds \(0\le \delta_n\le\bar\delta\) and \(0<\bar\delta<1\) make \(F_n\) continuous and strictly increasing on \([0,1-\bar\delta]\), with \(F_n(0)=0\).  For all sufficiently large \(n\), \(F_n(1-\bar\delta)\ge1\), so the crossing set in \cref{obj:def:bandwidth} is a nonempty interval whose left endpoint is the positive solution of \(F_n(h)=1\).  Hence, for all sufficiently large \(n\),
\[
0<h_n\le 1-\bar\delta,
\qquad
n h_n^3(\delta_n+h_n)=1 .
\]
Since \(\delta_n/n^{-1/4}\to\infty\), eventually \(n^{-1/4}\le \delta_n\), hence \(0<\delta_n\) and
\[
n\delta_n^4\ge 1 .
\]
The balance equation gives
\[
1=n h_n^3(\delta_n+h_n)\ge n h_n^3\delta_n,
\]
and therefore \(h_n\le \zeta_n\).  The inequality \(n\delta_n^4\ge1\) is exactly
\[
\zeta_n=(n\delta_n)^{-1/3}\le \delta_n .
\]
Consequently \(h_n\le\delta_n\), and then
\[
1=n h_n^3(\delta_n+h_n)\le 2n\delta_n h_n^3,
\]
so
\[
2^{-1/3}\zeta_n\le h_n\le\zeta_n
\]
eventually.  Also \(\delta_n+h_n\le\bar\delta+(1-\bar\delta)=1\), and the same balance equation forces \(h_n\le1/4\) eventually, since \(h_n>1/4\) and \(n\ge257\) would imply
\[
1=n h_n^3(\delta_n+h_n)\ge n h_n^4>1 .
\]
Thus
\[
h_n\asymp (n\delta_n)^{-1/3}.
\]
This is the one-stratum \(\beta=\kappa=1\) far-edge instance of the bandwidth
comparison in \cref{obj:lem:bandwidth-phases}, specialized to the density
\(\pi(a)=2a\) and the condition \(\delta_n/n^{-1/4}\to\infty\).
% lean: oneCellBandwidth_far

\item For all sufficiently large \(n\), the support of
\(\gamma_{\delta_n,h_n}\) is contained in
\([\delta_n-h_n,\delta_n+h_n]\subset[0,1]\).  On the left and right halves of
this interval,
\[
\gamma_{\delta_n,h_n}(a)=a-\delta_n+h_n
\quad(\delta_n-h_n\le a\le\delta_n),
\]
and
\[
\gamma_{\delta_n,h_n}(a)=\delta_n+h_n-a
\quad(\delta_n\le a\le\delta_n+h_n).
\]
Hence
\[
\begin{aligned}
\int_0^1 \gamma_{\delta_n,h_n}(a)^2\,2a\,da
&=
\int_0^{h_n}2(\delta_n-u)(h_n-u)^2\,du
+\int_0^{h_n}2(\delta_n+u)(h_n-u)^2\,du  \\
&=
4\delta_n\int_0^{h_n}(h_n-u)^2\,du
=
\frac43\,\delta_n h_n^3 .
\end{aligned}
\]
% lean: oneCellTentSqIntegral_eq

\item Because \(0\le\tau_{\delta_n,h_n}\le1\) and \(h_n\le1/4\) eventually,
\[
|\gamma_{\delta_n,h_n}(a)|\le\frac14
\qquad (0\le a\le1).
\]
For every real \(g\) with \(|g|\le1/4\), the Bernoulli comparison used here is
\[
2g^2
\le
\operatorname{KL}\!\left(\operatorname{Bernoulli}\!\left(\frac12+g\right),
\operatorname{Bernoulli}\!\left(\frac12\right)\right)
\le
4g^2 .
\]
The lower bound follows from Pinsker applied to the one-point event
\(\{1\}\), and the upper bound follows from the exact Bernoulli KL formula on
the interval \([1/4,3/4]\).  Integrating this display against the nonnegative
density \(2a\) and using the local perturbation-energy computation gives
\[
\frac83\,n\delta_n h_n^3
\le
K_n
\le
\frac{16}{3}\,n\delta_n h_n^3
\]
eventually.  Therefore
\[
K_n\asymp n\delta_n h_n^3 .
\]
% lean: oneCellProductKL_bounds

\item The collapsed atom mass is
\[
\int_0^{\delta_n}2a\,da=\delta_n^2,
\]
and \(\gamma_{\delta_n,h_n}(\delta_n)=h_n\).  The retained-course contribution
is supported on \((\delta_n,\delta_n+h_n]\), so
\[
\begin{aligned}
\int_{\delta_n}^1
\gamma_{\delta_n,h_n}(a)\,2a\,da
&=
\int_{\delta_n}^{\delta_n+h_n}
(\delta_n+h_n-a)\,2a\,da  \\
&=
\int_0^{h_n}2(\delta_n+u)(h_n-u)\,du
=
\delta_n h_n^2+\frac13 h_n^3 .
\end{aligned}
\]
Thus
\[
\Delta_n
=
\delta_n^2h_n+\delta_nh_n^2+\frac13 h_n^3 .
\]
Since \(0<h_n\le\delta_n\) eventually,
\[
\delta_n^2h_n
\le
\Delta_n
\le
\frac73\,\delta_n^2h_n
\]
eventually, and hence
\[
\Delta_n\asymp \delta_n^2h_n .
\]
% lean: oneCellTargetSeparation_eq

\item Let
\[
\chi_n=n^{-1/10},
\qquad
\varrho_n=n^{-1/2},
\qquad
\upsilon_n=\delta_n^2\zeta_n
=\delta_n^2(n\delta_n)^{-1/3}.
\]
For all sufficiently large \(n\), \(\delta_n>0\), and direct exponent algebra
gives
\[
\frac{\upsilon_n}{\varrho_n}
=
\left(\frac{\delta_n}{\chi_n}\right)^{5/3}.
\]
Combining the bandwidth comparison from the first item with the target
comparison from the fourth item gives constants
\(\underline{\mathfrak h},\overline{\mathfrak h}>0\) such that eventually
\[
\underline{\mathfrak h}\,\upsilon_n
\le
\Delta_n
\le
\overline{\mathfrak h}\,\upsilon_n .
\]
If \(\delta_n\asymp\chi_n\), then for some
\(\underline{\mathfrak d},\overline{\mathfrak d}>0\),
\[
\underline{\mathfrak d}
\le
\frac{\delta_n}{\chi_n}
\le
\overline{\mathfrak d}
\]
eventually.  Since \(x\mapsto x^{5/3}\) is increasing on \([0,\infty)\),
\[
\underline{\mathfrak h}\,\underline{\mathfrak d}^{5/3}\varrho_n
\le
\Delta_n
\le
\overline{\mathfrak h}\,\overline{\mathfrak d}^{5/3}\varrho_n
\]
eventually, so \(\Delta_n\asymp n^{-1/2}\).

Conversely, assume \(\Delta_n\asymp\varrho_n\).  Then for some
\(\underline{\mathfrak s},\overline{\mathfrak s}>0\),
\[
\underline{\mathfrak s}\varrho_n
\le
\Delta_n
\le
\overline{\mathfrak s}\varrho_n
\]
eventually.  With
\[
\lambda_{\mathrm{lo}}
=
\left(\frac{\underline{\mathfrak s}}{\overline{\mathfrak h}}\right)^{3/5},
\qquad
\lambda_{\mathrm{hi}}
=
\left(\frac{\overline{\mathfrak s}}{\underline{\mathfrak h}}\right)^{3/5},
\]
the two-sided display for \(\Delta_n\), together with
\(\upsilon_n/\varrho_n=(\delta_n/\chi_n)^{5/3}\), yields
\[
\lambda_{\mathrm{lo}}
\le
\frac{\delta_n}{\chi_n}
\le
\lambda_{\mathrm{hi}}
\]
eventually.  Hence
\[
\lambda_{\mathrm{lo}}\chi_n
\le
\delta_n
\le
(\lambda_{\mathrm{lo}}+\lambda_{\mathrm{hi}})\chi_n
\]
eventually, which is \(\delta_n\asymp n^{-1/10}\).  This proves
\[
\delta_n\asymp n^{-1/10}
\quad\Longleftrightarrow\quad
\Delta_n\asymp n^{-1/2}.
\]
% lean: one_cell_calibration
\end{enumerate}
\end{proof}
