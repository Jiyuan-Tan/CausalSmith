\begin{proof}[Proof of \cref{obj:prop:continuity-causal-bridge}]
Write \(P\) for the observed margin of \(P^{\mathrm F}\), and write \(p_x=P(X=x)=P^{\mathrm F}(X=x)\).  The hypotheses in \cref{obj:def:continuity-full-data-model-class} and the design constants give
\[
J>0,\qquad \kappa\ge0,\qquad c_->0,\qquad c_-\le \kappa+1\le c_+,
\qquad 0<p_{\min}\le 1/J,
\qquad 0<\bar\delta<1.
\]
In particular, \(p_x>0\) for every stratum, \(c_+\ge0\), and \(\bar\delta\le1\).

\begin{enumerate}
\item Fix \(x\in\mathcal X\).  Let \(\mu_{x,\mathrm{cont}}^P\) be the continuity-only regression extension selected in \cref{obj:synth_2}; the observed-margin component of \cref{obj:def:continuity-full-data-model-class} supplies the membership of \(P\) in \(\mathcal M_{\mathrm{cont}}\).  For every measurable \(B\subseteq[0,\bar\delta]\), the full-data product identity from consistency, exchangeability, and the conditional-density law gives
\[
\int_B m_x^{\mathrm F}(a)\pi_x(a)\,da
=
p_x^{-1}\int_{\{X=x,\ A\in B\}}Y\,dP^{\mathrm F}.
\]
Indeed, under the normalized full-data law restricted to \(X=x\), the treatment coordinate has the conditional law with density \(\pi_x(a)\,da\) on \([0,1]\).  Its latent-coordinate marginal is the pushforward
\[
\nu_x^{U}
:=
\bigl((O,u)\mapsto u\bigr)_{\#}
\left(p_x^{-1}\,P^{\mathrm F}\!\restriction\{(O,u):X(O)=x\}\right),
\]
equivalently, for every measurable \(C\subseteq\mathcal U\),
\[
\nu_x^{U}(C)=p_x^{-1}P^{\mathrm F}\{X=x,\ U\in C\}.
\]
With this normalization, \cref{obj:ass:consistency,obj:ass:exchangeability,obj:ass:conditional-density-law} give the product factorization of the treatment and latent coordinates in stratum \(x\).  Applying that factorization to \(\mathbf 1\{a\in B\}g_x(a,u)\), and using \cref{obj:synth_1}, identifies the stratum integral with
\[
\int_B\left(\int g_x(a,u)\,d\nu_x^{U}(u)\right)\pi_x(a)\,da
=
\int_B m_x^{\mathrm F}(a)\pi_x(a)\,da .
\]
The observed-margin identity gives
\[
\int_{\{X=x,\ A\in B\}}Y\,dP^{\mathrm F}
=
\int_{\{X=x,\ A\in B\}}Y\,dP .
\]
The regression-version property of \(\mu_{x,\mathrm{cont}}^P\) in \cref{obj:synth_2}, together with the conditional-density law, gives
\[
\int_{\{X=x,\ A\in B\}}Y\,dP
=
p_x\int_B \mu_{x,\mathrm{cont}}^P(a)\pi_x(a)\,da .
\]
Combining the three displays and using \(p_x>0\),
\[
\int_B m_x^{\mathrm F}(a)\pi_x(a)\,da
=
\int_B \mu_{x,\mathrm{cont}}^P(a)\pi_x(a)\,da
\qquad
(B\subseteq[0,\bar\delta]).
\]
The two products are integrable on \([0,\bar\delta]\): \cref{obj:synth_1,obj:ass:full-data-response-continuity,obj:synth_2} give continuity of the two mean functions on this compact interval, and \cref{obj:ass:polynomial-thinning} gives the envelope \(\pi_x(a)\le c_+a^\kappa\le c_+\) almost everywhere on \([0,\bar\delta]\).  Equality of all measurable set integrals therefore gives
\[
m_x^{\mathrm F}(a)\pi_x(a)
=
\mu_{x,\mathrm{cont}}^P(a)\pi_x(a)
\quad\text{for Lebesgue-a.e. }a\in[0,\bar\delta].
\]
For Lebesgue-a.e. \(a\in(0,\bar\delta]\), \cref{obj:ass:polynomial-thinning} also gives
\[
\pi_x(a)\ge c_-a^\kappa>0 .
\]
Since the endpoint \(0\) has Lebesgue measure zero, the two continuous functions \(m_x^{\mathrm F}\) and \(\mu_{x,\mathrm{cont}}^P\) agree Lebesgue-a.e. on \([0,\bar\delta]\).  Applying the continuous-version uniqueness principle in \cref{obj:lem:continuity-regression-extension-unique},
\[
m_x^{\mathrm F}(a)=\mu_{x,\mathrm{cont}}^P(a)
\qquad
\text{for every }a\in[0,\bar\delta].
\]
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.fullDataResponseMean_eq_contRegression

\item Fix \(\delta\in[0,\bar\delta]\).  Since \(\bar\delta\le1\), also \(\delta\in[0,1]\).  On the full-data carrier define
\[
\varphi_{\delta}^{\mathrm{ret}}(\omega)
=
\begin{cases}
Y(\omega),& \delta<A(\omega),\\
0,& \delta\ge A(\omega),
\end{cases}
\qquad
\varphi_{\delta}^{\mathrm{atom}}(\omega)
=
\begin{cases}
Y(\delta)(\omega),& A(\omega)\le\delta,\\
0,& A(\omega)>\delta.
\end{cases}
\]
The common consistency null set in \cref{obj:ass:consistency}, the observed-treatment support \(A\in[0,1]\) supplied by the observed-margin component of \cref{obj:def:continuity-full-data-model-class} and transferred to the full-data law, and \cref{obj:def:clamp-policy} give
\[
Y\!\left(d_\delta(A)\right)
=
\varphi_{\delta}^{\mathrm{ret}}
+
\varphi_{\delta}^{\mathrm{atom}}
\quad
P^{\mathrm F}\text{-a.s.}
\]
On the branch \(\delta<A\), treatment support puts \(A\) in \([0,1]\), the clamp policy returns \(A\), and consistency identifies \(Y(A)\) with the observed outcome \(Y\).  On the branch \(A\le\delta\), the clamp policy returns \(\delta\), so the displayed decomposition is the corresponding algebraic split.  The retained summand is measurable because it is built from the observed outcome and the measurable threshold set \(\{A>\delta\}\).  The atom summand is measurable because the potential-outcome process in \cref{obj:def:continuity-full-data-model-class} is jointly measurable and \(\{A\le\delta\}\) is measurable.  For integrability, the observed outcome support bounds \(\varphi_{\delta}^{\mathrm{ret}}\) by one almost surely, and \cref{obj:ass:consistency} identifies \(Y(\delta)\) with the \([0,1]\)-valued structural response \(g_X(\delta,U)\) on the consistency event; hence \(\varphi_{\delta}^{\mathrm{atom}}\) is also bounded by one almost surely.
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.causalClamp_pathwise

\item The retained summand depends only on the observed coordinates.  The observed-margin identity therefore gives
\[
\int \varphi_{\delta}^{\mathrm{ret}}\,dP^{\mathrm F}
=
\int Y\,\mathbf 1\{A>\delta\}\,dP
=
\nu_\delta(P),
\]
where \(\nu_\delta(P)\) is defined in \cref{obj:def:clamp-functional}.
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.fullData_observed_setIntegral

\item For the atom summand, decompose over strata and use the conditional product law for \(p_x^{-1}P^{\mathrm F}|_{\{X=x\}}\), now with the dose argument fixed at \(\delta\) and the treatment event \(A\le\delta\):
\[
\int \varphi_{\delta}^{\mathrm{atom}}\,dP^{\mathrm F}
=
\sum_{x\in\mathcal X}
p_x
\left(\int_{[0,\delta]}\pi_x(a)\,da\right)
m_x^{\mathrm F}(\delta).
\]
By \cref{obj:def:clamp-functional},
\[
q_x(\delta)=\int_{[0,\delta]}\pi_x(a)\,da,
\]
so
\[
\int \varphi_{\delta}^{\mathrm{atom}}\,dP^{\mathrm F}
=
\sum_{x\in\mathcal X}p_x q_x(\delta)m_x^{\mathrm F}(\delta).
\]
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.fullData_clampAtom_integral_eq

\item Combining the pathwise decomposition, integrability, and the two integral identities for the retained and atom summands, and using \cref{obj:def:causal-clamp-mean},
\[
\psi_\delta(P^{\mathrm F})
=
\nu_\delta(P)
+
\sum_{x\in\mathcal X}p_x q_x(\delta)m_x^{\mathrm F}(\delta).
\]
Applying the pointwise identity \(m_x^{\mathrm F}=\mu_{x,\mathrm{cont}}^P\) on \([0,\bar\delta]\) at \(a=\delta\) gives
\[
\psi_\delta(P^{\mathrm F})
=
\nu_\delta(P)
+
\sum_{x\in\mathcal X}p_x q_x(\delta)\mu_{x,\mathrm{cont}}^P(\delta).
\]
The right-hand side is exactly \(\theta_{\delta,\mathrm{cont}}(P)\) by \cref{obj:def:continuity-clamp-functional}.  Since \(\delta\in[0,\bar\delta]\) was arbitrary, the asserted equality holds throughout the declared threshold range.
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.continuity_causal_bridge
\end{enumerate}
\end{proof}
