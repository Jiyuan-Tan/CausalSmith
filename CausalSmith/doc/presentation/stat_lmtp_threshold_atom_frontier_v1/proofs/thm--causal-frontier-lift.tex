\begin{proof}[Proof of \cref{obj:thm:causal-frontier-lift}]
\begin{enumerate}
\item For standard Borel spaces \(S,T\) and every probability law \(\varrho\) on
\(S\times T\), the usual disintegration gives a probability kernel
\(K_\varrho:S\rightsquigarrow T\) such that, for measurable
\(D\subseteq S\) and \(E\subseteq T\),
\[
\varrho(D\times E)
=
\int_D K_\varrho(s,E)\,d(\varrho\circ \operatorname{pr}_S^{-1})(s).
\]
This standard Borel disintegration supplies the regular conditional law needed
for the observed-margin comparison in \cref{obj:prop:observed-margin-surjectivity}.
Let
\[
\Gamma(P^{\mathrm F})
=
\text{the observed }(X,A,Y)\text{-margin of }P^{\mathrm F}.
\]
Under the regime conditions, \cref{obj:prop:observed-margin-surjectivity} gives
surjectivity of \(\Gamma\) from \(\mathcal M^{\mathrm F}\) onto \(\mathcal M\) and,
for every \(n\) and every \(\delta\in[0,\bar\delta]\),
\[
R_{n,\mathrm F}^\star=R_n^\star,
\qquad
L_{n,\mathrm F}^\star=L_n^\star .
\]
For every \(P^{\mathrm F}\in\mathcal M^{\mathrm F}\) and every such \(\delta\),
\cref{obj:prop:causal-bridge} gives the target identity
\[
\psi_\delta(P^{\mathrm F})
=
\theta_\delta\{\Gamma(P^{\mathrm F})\}.
\]
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.standardBorelRegularConditionalLaw_mathlib

\item Let \(c_{\mathrm{risk}},C_{\mathrm{risk}}\) be the constants supplied by
\cref{obj:thm:minimax-risk}, and let \(c_{\mathrm{len}},C_{\mathrm{len}}\)
be the constants supplied by \cref{obj:thm:honest-length}.  Set
\[
c=\min\{c_{\mathrm{risk}},c_{\mathrm{len}}\},
\qquad
C=\max\{C_{\mathrm{risk}},C_{\mathrm{len}}\}.
\]
Then \(0<c<C\).  Choose \(n_\star\) so that, for every \(n\ge n_\star\),
the eventual conclusions of \cref{obj:thm:minimax-risk},
\cref{obj:thm:honest-length}, and \cref{obj:lem:bandwidth-balance} hold, and
also \(n\ge4\).  Fix such an \(n\), and write
\[
\delta=\delta_n,
\qquad
h=h_n,
\qquad
r=r_n=n^{-1/2}+\delta^{\kappa+1}h^\beta .
\]
Since \(0\le\delta\le\bar\delta\), \cref{obj:lem:bandwidth-balance} gives
\[
0<h,
\qquad
h\le 1-\bar\delta,
\qquad
n h^{2\beta+1}(\delta+h)^\kappa=1,
\]
and hence
\[
\delta+h\le \bar\delta+(1-\bar\delta)=1.
\]
Moreover \(r\ge0\). % lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.causal_frontier_lift

\item The total-Gram estimator in \cref{obj:def:total-gram-estimator} is a
measurable function of the observed sample, so
\[
z\longmapsto
\widehat\theta_n(z)
\]
is observed-sample measurable.  The stabilized interval in
\cref{obj:def:honest-interval} has measurable endpoints built from the same
finite sums and local-polynomial weights.  The center is clipped to \([0,1]\),
and the displayed radius is nonnegative, so the clipped lower endpoint is no
larger than the clipped upper endpoint.  Thus \(\mathrm{CI}_n\) is an
observed-sample measurable interval. % lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.stabilizedInterval_observedMeasurable

\item For the concrete estimator, the definitions of the full-data and observed
risks, the identity
\(\psi_\delta(P^{\mathrm F})=\theta_\delta\{\Gamma(P^{\mathrm F})\}\), and the
surjectivity of \(\Gamma\) give
\[
\begin{aligned}
&\sup_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\mathbb E_{\Gamma(P^{\mathrm F})^{\otimes n}}
\left|\widehat\theta_n-\psi_\delta(P^{\mathrm F})\right|  \\
&\qquad =
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}
\left|\widehat\theta_n-\theta_\delta(P)\right|.
\end{aligned}
\]
Combining this equality with \cref{obj:thm:minimax-risk} and the choices
\(c\le c_{\mathrm{risk}}\), \(C_{\mathrm{risk}}\le C\) yields
\[
c r
\le
R_{n,\mathrm F}^\star
\le
\sup_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\mathbb E_{\Gamma(P^{\mathrm F})^{\otimes n}}
\left|\widehat\theta_n-\psi_\delta(P^{\mathrm F})\right|
\le
C r .
\]
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.causalStabilizedWorstRisk_eq_observed

\item The same target transport gives the concrete coverage identity
\[
\inf_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\Gamma(P^{\mathrm F})^{\otimes n}
\{\psi_\delta(P^{\mathrm F})\in \mathrm{CI}_n\}
=
\inf_{P\in\mathcal M}
P^{\otimes n}\{\theta_\delta(P)\in \mathrm{CI}_n\}.
\]
The right-hand side is at least \(1-\alpha\) by
\cref{obj:thm:honest-length}, so
\[
\inf_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\Gamma(P^{\mathrm F})^{\otimes n}
\{\psi_\delta(P^{\mathrm F})\in \mathrm{CI}_n\}
\ge 1-\alpha .
\]
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.causalStabilizedCoverage_eq_observed

\item Expected length is transported by the same observed-margin argument.  Since
\(\mathrm{CI}_n\) is a function of the observed sample, its length depends on
\(P^{\mathrm F}\) only through \(\Gamma(P^{\mathrm F})\), and the surjectivity of
\(\Gamma\) gives
\[
\sup_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\mathbb E_{\Gamma(P^{\mathrm F})^{\otimes n}}
\operatorname{len}(\mathrm{CI}_n)
=
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}
\operatorname{len}(\mathrm{CI}_n).
\]
% lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.causalStabilizedWorstLength_eq_confidenceWorstLength
For every observed law \(P\), the clipping in \cref{obj:def:honest-interval}
gives
\[
0\le \operatorname{len}(\mathrm{CI}_n(z))\le 1.
\]
Thus the real expected-length bound from \cref{obj:thm:honest-length} gives
\[
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}
\operatorname{len}(\mathrm{CI}_n)
\le
C_{\mathrm{len}} r
\le
C r,
\]
where the last inequality uses \(C_{\mathrm{len}}\le C\) and \(r\ge0\).  Combining
this bound with the equality of the causal and observed expected-length criteria gives
\[
\sup_{P^{\mathrm F}\in\mathcal M^{\mathrm F}}
\mathbb E_{\Gamma(P^{\mathrm F})^{\otimes n}}
\operatorname{len}(\mathrm{CI}_n)
\le
C r.
\]
The same inequality may be viewed in \(\mathbb R_{\ge0}\cup\{\infty\}\) through
the order-preserving embedding of finite nonnegative real numbers; this is the
extended order used below for the minimax length criterion. % lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.confidenceWorstLength_stabilizedInterval_le_ofReal

\item Finally, \cref{obj:lem:stabilized-interval-honesty} makes
\(\mathrm{CI}_n\) an admissible honest observed-sample interval, since
\(n\ge4\), \(0<h\), \(\delta\in[0,\bar\delta]\), and \(\delta+h\le1\).  Hence,
by the definition of \(L_n^\star\) and the expected-length bound just proved,
\[
L_{n,\mathrm F}^\star
=
L_n^\star
\le
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}
\operatorname{len}(\mathrm{CI}_n)
\le
C r,
\]
with the two sides read in \(\mathbb R_{\ge0}\cup\{\infty\}\) by embedding the
finite nonnegative quantities as finite extended values.  The lower bound from
\cref{obj:thm:honest-length}, together with \(c\le c_{\mathrm{len}}\) and
\(r\ge0\), gives, in the same extended order,
\[
c r
\le
c_{\mathrm{len}} r
\le
L_n^\star
=
L_{n,\mathrm F}^\star .
\]
Substituting back \(\delta=\delta_n\), \(h=h_n\), and \(r=r_n\) gives all
displayed conclusions for every \(n\ge n_\star\). % lean: CausalSmith.Stat.LmtpThresholdAtomFrontier.causal_frontier_lift
\end{enumerate}
\end{proof}
