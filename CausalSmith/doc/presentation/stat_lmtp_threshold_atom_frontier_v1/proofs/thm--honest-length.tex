\begin{proof}[Proof of \cref{obj:thm:honest-length}]
\begin{enumerate}
\item Let \(\ell=\lceil\beta\rceil-1\) and
\[
t_\alpha=\sqrt{\frac{\log(12J/\alpha)}{2}} .
\]
The expected-length upper bound used below is the following eventual statement: there is a constant
\(C_{\mathrm{up}}>0\), depending only on the declared regime constants, such that for every admissible
threshold path and split path,
\[
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}
\!\left[\operatorname{len}(\mathrm{CI}_n)\right]
\le
C_{\mathrm{up}}
\left(n^{-1/2}+\delta_n^{\kappa+1}h_n^\beta\right)
\]
for all sufficiently large \(n\). To see this, use \cref{obj:def:honest-interval}. Intersecting the
symmetric interval with \([0,1]\) weakly shortens its length, and therefore
\[
\mathbb E_{P^{\otimes n}}\!\left[\operatorname{len}(\mathrm{CI}_n)\right]
\le
2\,\mathbb E_{P^{\otimes n}}\!\left[
 b_{0,n}+\sum_{x\in\mathcal X}D_{n,x}\right],
\]
where
\[
b_{0,n}=\frac{t_\alpha}{\sqrt{|I_{0,n}|}},
\qquad
b_{1,n}=\frac{t_\alpha}{\sqrt{|I_{1,n}|}},
\]
and \(D_{n,x}\) is the stratum contribution in \cref{obj:def:honest-interval}. The split lower bounds
in \cref{obj:synth_8} give \(b_{0,n},b_{1,n}\le 3t_\alpha n^{-1/2}\) once \(n\ge8\).

For a fixed law \(P\in\mathcal M\) and stratum \(x\), the integrated stratum contribution is bounded by
\[
\begin{aligned}
\mathbb E_{P^{\otimes n}}D_{n,x}
&\le
\left(\mathbb E_{P^{\otimes n}}\widehat b_x+b_{1,n}\right)
\left[ C_{\ell,1}Lh_n^\beta
+t_\alpha
\int
\left(\sum_{i\in I_{2,n}}w_{ix}(z)^2\right)^{1/2}
\,dP^{\otimes n}(z)
\right]
+b_{1,n}+P^{\otimes n}(\Omega_x^c) .
\end{aligned}
\]
Here \(C_{\ell,1}\) is the finite \(\ell_1\)-weight constant supplied on \(\Omega_x\) by
\cref{obj:lem:total-gram}. The displayed product bound is the sample-splitting step: the empirical atom
estimate uses \(I_{1,n}\), the local weights use \(I_{2,n}\), and the mixed term factors as
\[
\int \widehat b_x(z)
\left(\sum_{i\in I_{2,n}}w_{ix}(z)^2\right)^{1/2}
\,dP^{\otimes n}(z)
=
\left(\int\widehat b_x\,dP^{\otimes n}\right)
\left(\int
\left(\sum_{i\in I_{2,n}}w_{ix}^2\right)^{1/2}
\,dP^{\otimes n}\right).
\]
For the empirical atom envelope, define
\[
f_x(O)=\mathbf 1\{X=x,\ A\leq\delta_n\},
\qquad
m_x(P)=p_xq_x(\delta_n).
\]
Then \(\widehat b_x=|I_{1,n}|^{-1}\sum_{i\in I_{1,n}}f_x(O_i)\). The mean of one summand is
\[
\mathbb E_P f_x(O)
=P\{X=x,A\leq\delta_n\}
=p_x\int_0^{\delta_n}\pi_x(a)\,da
=p_xq_x(\delta_n)
=m_x(P).
\]
The second equality is the conditional-density component of \cref{obj:def:model-class}, and the third is the definition of \(q_x\) in \cref{obj:def:clamp-functional}.
Since \(0\leq f_x\leq1\), \(\operatorname{Var}_P\{f_x(O)\}\leq1/4\). Independence on the deterministic second block therefore yields
\[
\begin{aligned}
\mathbb E_{P^{\otimes n}}
\left|\widehat b_x-m_x(P)\right|
&\leq
\left\{\operatorname{Var}_{P^{\otimes n}}(\widehat b_x)\right\}^{1/2}  \\
&\leq
\frac12 |I_{1,n}|^{-1/2}.
\end{aligned}
\]
Moreover, \cref{obj:prop:pushforward-setup} gives
\(q_x(\delta_n)\leq c_+\delta_n^{\kappa+1}/(\kappa+1)\), and \(p_x\leq1\), so
\[
m_x(P)\leq \frac{c_+}{\kappa+1}\delta_n^{\kappa+1}.
\]
The triangle inequality gives the integrated envelope
\[
\mathbb E_{P^{\otimes n}}\widehat b_x
\le
\frac12 |I_{1,n}|^{-1/2}
+
\frac{c_+}{\kappa+1}\delta_n^{\kappa+1}.
\]
Also, \cref{obj:lem:weight-energy-balance} gives
\[
\int
\left(\sum_{i\in I_{2,n}}w_{ix}(z)^2\right)^{1/2}
\,dP^{\otimes n}(z)
\le C_{\mathrm{eng}}h_n^\beta
\]
for a finite constant \(C_{\mathrm{eng}}\), and \cref{obj:lem:total-gram} gives
\[
P^{\otimes n}(\Omega_x^c)
\le
C_{\mathrm{tg}}\exp\{-c_{\mathrm{tg}}n h_n(\delta_n+h_n)^\kappa\}.
\]
By \cref{obj:lem:bandwidth-balance}, for all sufficiently large \(n\),
\[
n h_n^{2\beta+1}(\delta_n+h_n)^\kappa=1,
\qquad
0<h_n,
\qquad
h_n\le1.
\]
The effective local sample size has the following lower bound. Put
\[
\gamma_{\mathrm{bal}}=2\beta+\kappa+1 .
\]
Since \(\delta_n\ge0\), \(h_n>0\), and \(\kappa\ge0\),
\[
h_n^{\gamma_{\mathrm{bal}}}
=h_n^{2\beta+1}h_n^\kappa
\le h_n^{2\beta+1}(\delta_n+h_n)^\kappa
=n^{-1}.
\]
Because \(\gamma_{\mathrm{bal}}>0\), this gives \(h_n\le n^{-1/\gamma_{\mathrm{bal}}}\). Multiplying the balance identity by \(h_n^{-2\beta}\) gives
\[
n h_n(\delta_n+h_n)^\kappa=h_n^{-2\beta},
\]
and therefore, since \(-2\beta<0\),
\[
n^{2\beta/(2\beta+\kappa+1)}
=\left(n^{-1/\gamma_{\mathrm{bal}}}\right)^{-2\beta}
\le h_n^{-2\beta}
=n h_n(\delta_n+h_n)^\kappa .
\]
Exponential decay along this polynomial scale is eventually bounded by \(n^{-1/2}\). Combining the
exponential-decay bound with the polynomial scale bound for \(n h_n(\delta_n+h_n)^\kappa\), and using
\(h_n^\beta\le1\), yields uniformly in \(P\in\mathcal M\) and \(x\in\mathcal X\),
\[
\mathbb E_{P^{\otimes n}}D_{n,x}
\le
C_{\mathrm{rad}}
\left(n^{-1/2}+\delta_n^{\kappa+1}h_n^\beta\right).
\]
Since \(J\) is fixed, summing over strata gives the displayed \(C_{\mathrm{up}}\) bound.

\item Put \(\eta=1-2\alpha>0\). The lower-bound input is a two-point inequality for expected length.  Because an arbitrary interval procedure need not have finite expected length, the worst-case length in this step is taken in \([0,\infty]\), with \(\int^{-}\) denoting the lower integral of a nonnegative measurable function; for the bounded intervals of interest this is the ordinary expected length.
If \(P_0,P_1\in\mathcal M\) satisfy
\[
\Delta\le |\theta_{\delta_n}(P_1)-\theta_{\delta_n}(P_0)|,
\qquad
P_1^{\otimes n}\ll P_0^{\otimes n},
\]
the likelihood-ratio square is \(P_0^{\otimes n}\)-integrable, and
\(\chi^2(P_1^{\otimes n},P_0^{\otimes n})\le\chi\), then every uniformly honest observed-sample
interval procedure \(C_n\) satisfies
\[
\max\!\left\{\left(1-2\alpha-\frac12\sqrt\chi\right)\Delta,0\right\}
\le
\sup\left\{
\int^{-}\operatorname{len}(C_n)\,dP^{\otimes n}: P\in\mathcal M
\right\},
\]
where the supremum and the lower integral are taken in \([0,\infty]\) and the nonnegative real number on the left is read as an element of \([0,\infty]\) as in \cref{obj:synth_11}. The comparisons in this step are made in \([0,\infty]\); the final paragraph returns to the ordinary expected-length notation of \cref{obj:synth_3,obj:synth_17} once the interval is known to be bounded. Indeed, coverage at \(P_0\) and
\(P_1\), transferred from \(P_1^{\otimes n}\) to \(P_0^{\otimes n}\) by
\[
d_{\mathrm{TV}}(P_1^{\otimes n},P_0^{\otimes n})
\le \frac12\sqrt{\chi},
\]
gives \(P_0^{\otimes n}\{\theta_{\delta_n}(P_0),\theta_{\delta_n}(P_1)\in C_n\}\ge
1-2\alpha-\frac12\sqrt\chi\). On that event the interval length is at least \(\Delta\), and integrating
this indicator gives the displayed extended-real bound.

For the root component, set
\[
\varepsilon_n=\tau_\alpha n^{-1/2},
\qquad
\tau_\alpha=\frac{\eta}{16},
\qquad
\chi_\alpha=\frac{\eta^2}{16}.
\]
Let \(\mathbb P^{\mathrm{root}}_{0,n}\) and \(\mathbb P^{\mathrm{root}}_{1,n}\) have the common design
\[
\mathbb P(X=x,A\in da)=\frac1J(\kappa+1)a^\kappa\mathbf 1\{0\leq a\leq1\}\,da,
\]
and, conditionally on \((X,A)=(x,a)\), let \(Y\) be Bernoulli with success probability
\[
\frac12+g^{\mathrm{root}}_{j,n}(x,a),
\qquad
g^{\mathrm{root}}_{0,n}(x,a)=0,\qquad
g^{\mathrm{root}}_{1,n}(x,a)=\varepsilon_n .
\]
For all sufficiently large \(n\), \(|\varepsilon_n|\leq1/2\). The two laws therefore have outcomes in \([0,1]\), the same density \(\pi_x(a)=(\kappa+1)a^\kappa\), the same masses \(p_x=1/J\), and constant Hölder regressions; the regime inequalities place both laws in \(\mathcal M\). Their clamp-target gap is exactly
\[
\begin{aligned}
\theta_{\delta_n}(\mathbb P^{\mathrm{root}}_{1,n})
-\theta_{\delta_n}(\mathbb P^{\mathrm{root}}_{0,n})
&=\varepsilon_n
\left[
\mathbb P(A>\delta_n)+\sum_{x\in\mathcal X}p_xq_x(\delta_n)
\right] \\
&=\varepsilon_n .
\end{aligned}
\]
For one observation the chi-square divergence is \(4\varepsilon_n^2\), hence independence gives
\[
1+\chi^2\!\left((\mathbb P^{\mathrm{root}}_{1,n})^{\otimes n},
(\mathbb P^{\mathrm{root}}_{0,n})^{\otimes n}\right)
=(1+4\varepsilon_n^2)^n.
\]
Since \(4n\varepsilon_n^2=\eta^2/64\),
\[
(1+4\varepsilon_n^2)^n
\le \exp(\eta^2/64)
\le 1+\eta^2/64+(\eta^2/64)^2
\le 1+\chi_\alpha .
\]
The two-point length inequality above, with \(\chi=\chi_\alpha\), yields
\[
\max\!\left\{
\left(\eta-\frac12\sqrt{\chi_\alpha}\right)\varepsilon_n,0
\right\}
\le L_n^\star .
\]
Thus, for some \(c_{\mathrm{root}}>0\),
\[
\max\!\left\{c_{\mathrm{root}}n^{-1/2},0\right\}
\le L_n^\star
\]
eventually.

For the local component, use the fixed compactly supported \(C^\infty\) bump \(\mathfrak b:\mathbb R\to\mathbb R\) from the same two-point construction, normalized by
\[
\mathfrak b(0)=1,
\qquad
0\leq\mathfrak b(u)\leq1,
\qquad
\mathfrak b(u)=0\ \text{whenever } |u|\geq1 .
\]
The smoothness calibration for this particular bump gives an amplitude
\(\mathfrak a_{\mathrm{bump}}\in(0,1/4]\), depending only on \((\beta,L)\), with the following uniform property. For every \(0\leq\delta\leq1\) and \(0<h\leq1\), define
\[
q_{\delta,h}^{\mathrm{bump}}(a)
=
\mathfrak a_{\mathrm{bump}}h^\beta
\mathfrak b\!\left(\frac{a-\delta}{h}\right).
\]
Then \(a\mapsto 1/2+q_{\delta,h}^{\mathrm{bump}}(a)\) is continuous on \([0,1]\), takes values in \([0,1]\), and satisfies the Hölder Taylor remainder
\[
\left|
\left(\frac12+q_{\delta,h}^{\mathrm{bump}}(t)\right)
-
\sum_{j=0}^{\ell}
\frac{\partial_{[0,1]}^j(1/2+q_{\delta,h}^{\mathrm{bump}})(s)}{j!}(t-s)^j
\right|
\le L|t-s|^\beta
\]
for all \(s,t\in[0,1]\). Indeed, applying the normalized Hölder gate of \(\mathfrak b\) with radius \(\min\{L,1\}\) and then shrinking the amplitude to be at most \(1/4\) gives both the displayed Taylor bound and the range constraint.

Set
\[
\zeta_{\mathrm{loc}}
=
\frac{\eta^2/64}{8(\kappa+1)\mathfrak a_{\mathrm{bump}}^2+1},
\qquad
h_{n,\mathrm{loc}}=\zeta_{\mathrm{loc}}h_n .
\]
Let \(\mathbb P^{\mathrm{loc}}_{0,n}\) and \(\mathbb P^{\mathrm{loc}}_{1,n}\) have the same common design as the root pair, and let their conditional Bernoulli success probabilities be \(1/2+g^{\mathrm{loc}}_{j,n}(x,a)\), where
\[
g^{\mathrm{loc}}_{0,n}(x,a)=0,\qquad
g^{\mathrm{loc}}_{1,n}(x,a)
=\mathfrak a_{\mathrm{bump}}h_{n,\mathrm{loc}}^\beta
\mathfrak b\!\left(\frac{a-\delta_n}{h_{n,\mathrm{loc}}}\right).
\]
For all sufficiently large \(n\), \cref{obj:lem:bandwidth-balance} gives \(0<h_n\le1\), hence \(0<h_{n,\mathrm{loc}}\le1\), and the amplitude calibration places both laws in \(\mathcal M\). The localized shift is nonnegative and equals \(\mathfrak a_{\mathrm{bump}}h_{n,\mathrm{loc}}^\beta\) at \(a=\delta_n\), so the collapsed atom contribution alone gives
\[
\left|
\theta_{\delta_n}(\mathbb P^{\mathrm{loc}}_{1,n})
-\theta_{\delta_n}(\mathbb P^{\mathrm{loc}}_{0,n})
\right|
\geq
\mathfrak a_{\mathrm{bump}}\delta_n^{\kappa+1}h_{n,\mathrm{loc}}^\beta
=
\mathfrak a_{\mathrm{bump}}\zeta_{\mathrm{loc}}^\beta
\delta_n^{\kappa+1}h_n^\beta .
\]
For one observation, write
\[
\mathcal I_n
=
\int 4\{g^{\mathrm{loc}}_{1,n}(x,a)\}^2\,d\mathbb P_{X,A}(x,a),
\]
where \(\mathbb P_{X,A}\) is the common design law. Since \(0\leq\mathfrak b\leq1\) and \(\mathfrak b\) vanishes outside \([-1,1]\),
\[
\mathcal I_n
\le
8(\kappa+1)\mathfrak a_{\mathrm{bump}}^2
h_{n,\mathrm{loc}}^{2\beta+1}(\delta_n+h_{n,\mathrm{loc}})^\kappa .
\]
Because \(0<\zeta_{\mathrm{loc}}\leq1\), \(h_{n,\mathrm{loc}}\leq h_n\). With \(2\beta+1>1\), \(\kappa\geq0\), and the balance identity from \cref{obj:lem:bandwidth-balance},
\[
\begin{aligned}
n\mathcal I_n
&\le
8(\kappa+1)\mathfrak a_{\mathrm{bump}}^2
\zeta_{\mathrm{loc}}^{2\beta+1}
\,n h_n^{2\beta+1}(\delta_n+h_n)^\kappa \\
&\le
\eta^2/64 .
\end{aligned}
\]
Independence gives
\[
1+\chi^2\!\left((\mathbb P^{\mathrm{loc}}_{1,n})^{\otimes n},
(\mathbb P^{\mathrm{loc}}_{0,n})^{\otimes n}\right)
=(1+\mathcal I_n)^n
\le \exp(\eta^2/64)
\le 1+\chi_\alpha .
\]
The two-point length inequality therefore gives, for some \(c_{\mathrm{loc}}>0\),
\[
\max\!\left\{c_{\mathrm{loc}}\delta_n^{\kappa+1}h_n^\beta,0\right\}
\le L_n^\star
\]
eventually. Taking half the smaller of \(c_{\mathrm{root}}\) and \(c_{\mathrm{loc}}\), and splitting into
the two cases according to which component is larger, gives a constant \(c_{\mathrm{min}}>0\) such that
\[
\max\!\left\{c_{\mathrm{min}}\left(n^{-1/2}+\delta_n^{\kappa+1}h_n^\beta\right),0\right\}
\le L_n^\star
\]
eventually.

The every-procedure version uses that a uniformly honest procedure is an admissible competitor in the
minimax honest-length criterion. Hence there is a positive constant \(c_{\mathrm{proc}}>0\) such that
every uniformly honest observed-sample interval procedure \(C_n\) satisfies
\[
\max\!\left\{c_{\mathrm{proc}}\left(n^{-1/2}+\delta_n^{\kappa+1}h_n^\beta\right),0\right\}
\le
\sup\left\{
\int^{-}\operatorname{len}(C_n)\,dP^{\otimes n}: P\in\mathcal M
\right\}
\]
in \(\mathbb R_{\ge0}\cup\{\infty\}\).

\item Define
\[
c=\frac12\min\{c_{\mathrm{min}},c_{\mathrm{proc}}\},
\qquad
C=C_{\mathrm{up}}+c+1 .
\]
Then \(c>0\), \(c<C\), \(c\le c_{\mathrm{min}}\), \(c\le c_{\mathrm{proc}}\), and \(C_{\mathrm{up}}\le C\).

\item Fix an admissible threshold path \((\delta_n)\) and split path \(B_\bullet\). By
\cref{obj:lem:bandwidth-balance}, for all sufficiently large \(n\),
\[
0<h_n,
\qquad
h_n\le 1-\bar\delta,
\qquad
n h_n^{2\beta+1}(\delta_n+h_n)^\kappa=1 .
\]
Together with \(0\le\delta_n\le\bar\delta\), this gives \(\delta_n+h_n\le1\). Hence the hypotheses of
\cref{obj:lem:stabilized-interval-honesty} hold. That result supplies observed-sample measurability of
\(\mathrm{CI}_n\) and the modelwise coverage inequality
\[
1-\alpha
\le
P^{\otimes n}\{\theta_{\delta_n}(P)\in \mathrm{CI}_n\}
\]
for every \(P\in\mathcal M\), so the stabilized interval is uniformly honest.
Taking the infimum over \(P\) gives the displayed worst-case coverage bound.

\item Put
\[
r_n=n^{-1/2}+\delta_n^{\kappa+1}h_n^\beta .
\]
Since \(n^{-1/2}>0\) and \(\delta_n^{\kappa+1}h_n^\beta\ge0\), we have \(r_n>0\). The upper ingredient and \(C_{\mathrm{up}}\le C\)
yield
\[
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}\!\left[\operatorname{len}(\mathrm{CI}_n)\right]
\le C r_n .
\]
The minimax lower ingredient and \(c\le c_{\mathrm{min}}\) yield the extended-real comparison
\[
\max\{c r_n,0\}\le L_n^\star .
\]

Since \(\mathrm{CI}_n\) is itself uniformly honest by the coverage step, the procedure lower bound
applies to this concrete interval. Combined with \(c\le c_{\mathrm{proc}}\), it gives
\[
\max\{c r_n,0\}
\le
\sup\left\{
\int^{-}\operatorname{len}(\mathrm{CI}_n)\,dP^{\otimes n}: P\in\mathcal M
\right\}
\]
in \([0,\infty]\). For the concrete stabilized interval the length is measurable, nonnegative, and
bounded by one, so its worst-case length in \([0,\infty]\) agrees with its ordinary worst-case
expected length, read in \([0,\infty]\) as in \cref{obj:synth_11}. Since \(c r_n>0\), this gives the
real lower bound
\[
c r_n
\le
\sup_{P\in\mathcal M}
\mathbb E_{P^{\otimes n}}\!\left[\operatorname{len}(\mathrm{CI}_n)\right].
\]

Finally, for every uniformly honest observed-sample interval procedure \(C_n\), the procedure lower
bound gives the extended-real statement
\[
\max\{c r_n,0\}
\le
\sup\left\{
\int^{-}\operatorname{len}(C_n)\,dP^{\otimes n}: P\in\mathcal M
\right\},
\]
with the supremum taken in \(\mathbb R_{\ge0}\cup\{\infty\}\). This is the arbitrary-procedure worst-length criterion used in the comparison. For interval procedures whose pointwise lengths lie in \([0,1]\) as in \cref{obj:synth_17}, the nonnegative lower integral equals the ordinary expected length under the positive-part embedding of \cref{obj:synth_11}, and the display specializes to the corresponding real expected-length lower bound.
These are the eventual length comparisons established by the argument.
\end{enumerate}
\end{proof}
