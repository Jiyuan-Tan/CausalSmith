\begin{proof}[Proof of \cref{obj:lem:total-gram}]
Set
\[
\ell=\lceil\beta\rceil-1,\qquad
\lambda=\lambda_\star(\ell,\kappa,c_-,c_+),\qquad
d=\ell+1 .
\]
The regime assumptions give \(\kappa\ge0\), \(c_->0\), and \(c_+>0\), so \(\lambda>0\) by \cref{obj:lem:lambda-star-positive}. % lean: totalGram_lambdaStar_pos

Define
\[
\eta=
\min\left\{
\frac1{20},
\frac{\lambda^2}{16d(4d+\lambda)}
\right\},
\qquad
K=\frac{p_{\min}c_-}{2\cdot2^\kappa},
\qquad
c=\frac{\eta K}{8},
\]
and
\[
A_{\mathrm{tail}}=2(1+d^2),\qquad
C=
A_{\mathrm{tail}}e^{8c}+\frac{2\sqrt d}{\lambda}+\frac{4d}{\lambda^2}+1 .
\]
The constants \(\eta,K,c,A_{\mathrm{tail}},C\) are positive and, together with \(\lambda\), depend only on the displayed regime constants. % lean: total_gram_stabilization

Fix \(P,n,B_n,\delta,x\) as in the statement, and write
\[
h=h_n,
\qquad
q(O)=\mathbf 1\left\{X=x,\ 0\le\frac{A-\delta}{h}\le1\right\},
\qquad
\phi_j(O)=
\mathbf 1\left\{0\le\frac{A-\delta}{h}\le1\right\}
\left(\frac{A-\delta}{h}\right)^j .
\]
The regime assumptions give \(\beta>0\), \(\kappa\ge0\), \(\delta\in[0,\bar\delta]\), and \(0<\bar\delta<1\).  The construction of \(h\) in \cref{obj:def:bandwidth} gives \(0<h\le1-\bar\delta\).  Indeed, if the crossing set in \cref{obj:def:bandwidth} is empty, then \(h=1-\bar\delta\).  If it is nonempty and \(z\) is a member, then \(n>0\), \(0<z\le1-\bar\delta\), \(\delta+z\le1\), and
\[
1\le n z^{2\beta+1}(\delta+z)^\kappa\le n z^{2\beta+1},
\]
so \(n^{-1/(2\beta+1)}\le z\).  Hence the infimum is bounded below by this positive number and above by any member of the set, giving \(0<h\le1-\bar\delta\).  Since \(\delta\le\bar\delta\), this also gives \(\delta+h\le1\). % lean: infoBandwidth_pos_le

Let
\[
p=\mathbb E_P q(O)
=
P\left\{X=x,\ 0\le\frac{A-\delta}{h}\le1\right\}.
\]
By \cref{obj:lem:local-window-mass,obj:lem:power-window-integral-lower} and the thinning and stratum-mass assumptions,
\[
p
=
p_x\int_{\delta}^{\delta+h}\pi_x(a)\,da
\ge
p_{\min}c_-\int_{\delta}^{\delta+h}a^\kappa\,da
\ge
p_{\min}c_-\frac h2\left(\frac{\delta+h}{2}\right)^\kappa
=
K\,h(\delta+h)^\kappa .
\]
% lean: localWindowWeight_integral_lower

For every \(b\in\mathbb R^{\ell+1}\), \cref{obj:lem:local-gram-coercivity} gives
\[
\lambda\,p\,\|b\|_2^2
\le
\sum_{j=0}^{\ell}\sum_{k=0}^{\ell}
b_jb_k\,\mathbb E_P\{q(O)\phi_j(O)\phi_k(O)\}.
\]
Moreover \(0\le q\le1\) and \(|\phi_j|\le1\) for every \(j\). % lean: localWindow_populationGram_coercive

Let \(m=|I_2|\), and let \(i_1,\ldots,i_m\) be the elements of \(I_2\) in increasing order.  For a block vector \(\omega=(\omega_1,\ldots,\omega_m)\), define
\[
N_{\mathrm{blk}}(\omega)=\sum_{r=1}^m q(\omega_r),\qquad
M_{\mathrm{blk},jk}(\omega)=\sum_{r=1}^m q(\omega_r)\phi_j(\omega_r)\phi_k(\omega_r),
\]
and put
\[
M_{jk}=\mathbb E_P\{q(O)\phi_j(O)\phi_k(O)\}.
\]
The deterministic ordered-block map
\[
T_{I_2}(O_1,\ldots,O_n)=(O_{i_1},\ldots,O_{i_m})
\]
satisfies, for every measurable block event \(E\),
\[
P^{\otimes n}\{T_{I_2}(O_1,\ldots,O_n)\in E\}=P^{\otimes m}(E),
\]
because the selected coordinates are independent and each has law \(P\).  Under this map,
\[
N_{\mathrm{blk}}(T_{I_2}z)
=
\sum_{i\in I_2}q(O_i)=N_x,
\]
and, for every \(j,k\),
\[
M_{\mathrm{blk},jk}(T_{I_2}z)
=
\sum_{i\in I_2}q(O_i)\phi_j(O_i)\phi_k(O_i)
=(G_x)_{jk},
\]
where the last equality is the entrywise definition of the local Gram matrix in \cref{obj:def:local-design-notation}.

Define the block good event
\[
\Gamma_x=
\left\{
\omega:
N_{\mathrm{blk}}(\omega)>0
\ \text{and}\ 
\sum_{j=0}^{\ell}\sum_{k=0}^{\ell}
 b_jb_k M_{\mathrm{blk},jk}(\omega)
\ge
\frac{\lambda N_{\mathrm{blk}}(\omega)}{2}\|b\|_2^2
\ \text{for every } b\in\mathbb R^d
\right\}.
\]
The preceding identities give the exact event identity
\[
\Omega_x=\{z:T_{I_2}z\in\Gamma_x\}.
\]
Indeed, \(q\) is the indicator of the stratum-window event, \(\phi_j\) is its local monomial coordinate, and \(\lambda=\lambda_\star(\ell,\kappa,c_-,c_+)\), so \(N_{\mathrm{blk}}\) and \(M_{\mathrm{blk}}\) are exactly the count and quadratic form appearing in \cref{obj:def:local-design-notation}. % lean: localizedGramGood_iff_goodGramEvent

If \(n\ge8\), the split condition gives \(m\ge n/8\), hence \(m>0\).  The lower bound on \(p\) gives \(p>0\).  Applying the localized empirical Gram concentration bound to the \(m\) independent block coordinates, with the population coercivity above and with \(0\le q\le1\) and \(|\phi_j|\le1\), gives
\[
P^{\otimes m}(\Gamma_x^c)
\le
A_{\mathrm{tail}}\exp\{-mp\eta\}.
\]
% lean: Causalean.Stat.Concentration.localized_empiricalGram_coercive_of_pos
By the ordered-block product identity and \(\Omega_x=\{z:T_{I_2}z\in\Gamma_x\}\),
\[
P^{\otimes n}(\Omega_x^c)
\le
A_{\mathrm{tail}}\exp\{-mp\eta\}.
\]
Combining \(m\ge n/8\) with the lower bound on \(p\),
\[
c\,n\,h(\delta+h)^\kappa
\le
m\,p\,\eta .
\]
Hence
\[
P^{\otimes n}(\Omega_x^c)
\le
A_{\mathrm{tail}}\exp\{-mp\eta\}
\le
A_{\mathrm{tail}}\exp\{-c\,n\,h(\delta+h)^\kappa\}
\le
C\exp\{-c\,n\,h(\delta+h)^\kappa\}.
\]
% lean: total_gram_stabilization

If \(n<8\), then \(h\le1\), \(\delta+h\le1\), and \(\kappa\ge0\) imply
\[
n\,h(\delta+h)^\kappa\le8.
\]
Thus
\[
1\le
A_{\mathrm{tail}}e^{8c}\exp\{-c\,n\,h(\delta+h)^\kappa\}
\le
C\exp\{-c\,n\,h(\delta+h)^\kappa\},
\]
and the same probability bound follows from \(P^{\otimes n}(\Omega_x^c)\le1\). % lean: total_gram_stabilization

On \(\Omega_x\), \cref{obj:lem:good-gram-weights} gives, for every \(0\le j\le\ell\),
\[
\sum_{i\in I_2}w_{ix}U_i^j
=
\mathbf 1\{j=0\},
\]
which is the displayed reproduction identity.  The same result also gives
\[
\sum_{i\in I_2}|w_{ix}|
\le
\frac{2\sqrt d}{\lambda}
\le C,
\qquad
\sum_{i\in I_2}w_{ix}^2
\le
\frac{4d}{\lambda^2N_x}
\le
\frac{C}{N_x}.
\]
% lean: goodGram_weight_controls
\end{proof}
