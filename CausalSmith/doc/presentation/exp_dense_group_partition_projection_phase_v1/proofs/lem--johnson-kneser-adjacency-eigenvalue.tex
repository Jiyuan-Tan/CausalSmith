\begin{proof}[Proof of \cref{obj:lem:johnson-kneser-adjacency-eigenvalue}]
Write
\[
m_{\Upsilon}(\Gamma):=\mathbf 1\{\Upsilon\subseteq \Gamma\},
\qquad
V_{\le q}:=\operatorname{span}\{m_{\Upsilon}:|\Upsilon|\le q\}\subseteq L^2(\Omega_{n,M}),
\]
for finite \(\Upsilon\subseteq\{1,\ldots,n\}\) and \(q\ge 0\). Also write
\[
\delta_{\Upsilon}(\Gamma):=\mathbf 1\{\Upsilon\cap\Gamma=\varnothing\}.
\]
The Johnson harmonic construction used in \cref{obj:def:johnson-decomposition} has
\[
\mathcal H_{n,0}=V_{\le 0},
\qquad
\mathcal H_{n,j}=V_{\le j}\cap V_{\le j-1}^{\perp}\quad(j\ge 1),
\]
where orthogonality is with respect to the uniform-slice inner product. The hypothesis \(2M\le n\) gives \(M\le n\), so \(\Omega_{n,M}\) is nonempty.

\begin{enumerate}
\item First take \(k=0\). Since \(V_{\le 0}\) is spanned by \(m_{\varnothing}\), membership \(f\in\mathcal H_{n,0}=V_{\le 0}\) gives a scalar \(\eta\in\mathbb R\) such that
\[
f=\eta\,m_{\varnothing}.
\]
For every \(\Gamma\in\Omega_{n,M}\),
\[
(\mathrm{Adj}_{n,M}m_{\varnothing})(\Gamma)
=
\#\{\Delta\in\Omega_{n,M}:\Gamma\cap\Delta=\varnothing\}
=
\binom{n-M}{M}.
\]
Indeed, after \(\Gamma\) is fixed, the admissible \(\Delta\)'s are precisely the \(M\)-subsets of the \(n-M\) points outside \(\Gamma\). Thus
\[
\mathrm{Adj}_{n,M}f
=
\eta\binom{n-M}{M}m_{\varnothing}
=
(-1)^0\binom{n-M}{M}f,
\]
which is the asserted formula in degree zero. % lean: mem_degreeAtMost_zero_iff, kneserAdjacency_inclusionMonomial_eq

\item Now suppose \(k\ge 1\). Define the scalar
\[
\alpha_k:=(-1)^k\binom{n-M-k}{M-k}
\]
and the residual
\[
\rho:=\mathrm{Adj}_{n,M}f-\alpha_k f.
\]
We first show
\[
\rho\in V_{\le k-1}.
\]
It is enough to check this on the spanning monomials \(m_{\Upsilon}\) with \(|\Upsilon|\le k\). If \(|\Upsilon|<k\), then \(m_{\Upsilon}\in V_{\le k-1}\), and the adjacency operator preserves this lower filtration level:
\[
\mathrm{Adj}_{n,M}m_{\Upsilon}\in V_{\le k-1}.
\]
Hence \(\mathrm{Adj}_{n,M}m_{\Upsilon}-\alpha_km_{\Upsilon}\in V_{\le k-1}\).

It remains to consider \(|\Upsilon|=k\). For every \(\Gamma\in\Omega_{n,M}\),
\[
(\mathrm{Adj}_{n,M}m_{\Upsilon})(\Gamma)
=
\binom{n-M-k}{M-k}\,\delta_{\Upsilon}(\Gamma).
\]
When \(\Gamma\cap\Upsilon\ne\varnothing\), no disjoint \(\Delta\) can contain \(\Upsilon\). When \(\Gamma\cap\Upsilon=\varnothing\), such \(\Delta\)'s are obtained by adjoining \(M-k\) further points from the complement of \(\Gamma\cup\Upsilon\), which has size \(n-M-k\). Inclusion-exclusion gives
\[
\delta_{\Upsilon}
=
\sum_{\Theta\subseteq\Upsilon}(-1)^{|\Theta|}m_{\Theta}.
\]
Therefore
\[
\mathrm{Adj}_{n,M}m_{\Upsilon}-\alpha_km_{\Upsilon}
=
\binom{n-M-k}{M-k}
\sum_{\Theta\subsetneq\Upsilon}(-1)^{|\Theta|}m_{\Theta}\in V_{\le k-1},
\]
because every proper subset \(\Theta\subsetneq\Upsilon\) has \(|\Theta|\le k-1\). By linearity over the span \(V_{\le k}\), the same conclusion holds for every \(f\in V_{\le k}\), in particular for the present \(f\in\mathcal H_{n,k}\subseteq V_{\le k}\). % lean: kneserAdjacency_sub_eigen_mem_lower

\item Since \(f\in\mathcal H_{n,k}=V_{\le k}\cap V_{\le k-1}^{\perp}\), for every \(g\in V_{\le k-1}\),
\[
\langle f,g\rangle_n=0.
\]
The adjacency operator preserves \(V_{\le k-1}\), so \(\mathrm{Adj}_{n,M}g\in V_{\le k-1}\). It is also self-adjoint for the uniform-slice inner product:
\[
\langle \mathrm{Adj}_{n,M}u,v\rangle_n
=
\langle u,\mathrm{Adj}_{n,M}v\rangle_n
\qquad
(u,v\in L^2(\Omega_{n,M})),
\]
which follows by interchanging the two finite sums and using symmetry of disjointness. Hence, for every \(g\in V_{\le k-1}\),
\[
\begin{aligned}
\langle \rho,g\rangle_n
&=
\langle \mathrm{Adj}_{n,M}f,g\rangle_n-\alpha_k\langle f,g\rangle_n  \\
&=
\langle f,\mathrm{Adj}_{n,M}g\rangle_n-\alpha_k\langle f,g\rangle_n
=
0.
\end{aligned}
\]
Thus \(\rho\) is orthogonal to \(V_{\le k-1}\). % lean: sliceInner_kneserAdjacency, kneserAdjacency_mem_degreeAtMost

\item Combining the previous two steps, \(\rho\in V_{\le k-1}\) and \(\rho\perp V_{\le k-1}\). Taking \(g=\rho\) gives
\[
\langle \rho,\rho\rangle_n=0.
\]
Since \(M\le n\),
\[
\langle \rho,\rho\rangle_n
=
\binom nM^{-1}\sum_{\Gamma\in\Omega_{n,M}}\rho(\Gamma)^2
\]
with \(\binom nM>0\), so the uniform-slice inner product is positive definite. Therefore \(\rho=0\), i.e.
\[
\mathrm{Adj}_{n,M}f=\alpha_k f
=
(-1)^k\binom{n-M-k}{M-k}f.
\]
This is the desired identity for \(k\ge 1\), and together with the degree-zero case proves the lemma. % lean: sliceInner_eq, kneserAdjacency_eigen
\end{enumerate}
\end{proof}
