\begin{proof}[Proof of \cref{obj:lem:classical-johnson-decomposition}]
Assume \(2M\le n\). Then \(M\le n\), so \(\Omega_{n,M}\) is the finite uniform slice with
\[
|\Omega_{n,M}|=\binom nM .
\]
For a slice function \(\varphi:\Omega_{n,M}\to\mathbb R\), write
\[
\operatorname{mean}_n(\varphi)
:=
\frac{1}{|\Omega_{n,M}|}\sum_{\omega\in\Omega_{n,M}}\varphi(\omega).
\]

\begin{enumerate}
\item For \(0\le u\le M\), let \(\mathcal F_{\le u}\subseteq L^2(\Omega_{n,M})\) be the span of the inclusion monomials
\[
\omega\longmapsto \prod_{i\in T}{\bf 1}_{\{i\in\omega\}},
\qquad
T\subseteq\{1,\ldots,n\},\quad |T|\le u .
\]
The Johnson harmonic layers are constructed by
\[
\mathcal H_{n,0}=\mathcal F_{\le 0},
\qquad
\mathcal H_{n,u+1}
=
\mathcal F_{\le u+1}\cap \mathcal F_{\le u}^{\perp}.
\]
Since \(\mathcal F_{\le u}\subseteq \mathcal F_{\le u+1}\), finite-dimensional orthogonal decomposition gives
\[
\mathcal F_{\le u+1}
=
\mathcal F_{\le u}\oplus \mathcal H_{n,u+1}.
\]
Iterating from \(u=0\) to \(M-1\),
\[
\mathcal F_{\le M}
=
\bigoplus_{u=0}^{M}\mathcal H_{n,u}.
\]
Moreover \(\mathcal F_{\le M}=L^2(\Omega_{n,M})\): for every fixed
\(\omega_0\in\Omega_{n,M}\),
\[
{\bf 1}_{\{\omega_0\}}(\omega)
=
\prod_{i\in\omega_0}{\bf 1}_{\{i\in\omega\}},
\]
because \(|\omega|=|\omega_0|=M\). Thus point masses belong to \(\mathcal F_{\le M}\), and they span all slice functions.

Therefore the orthogonal projections onto the layers satisfy, for every
\(\varphi:\Omega_{n,M}\to\mathbb R\),
\[
\varphi
=
\sum_{u=0}^{M}\Pi_{n,u}\varphi .
\]
The degree-zero space \(\mathcal H_{n,0}\) is the space of constant functions. Its orthogonal projection is the constant function with value \(\operatorname{mean}_n(\varphi)\), since for every real scalar \(a_{\mathrm{const}}\),
\[
\left\langle
\varphi-(\omega\mapsto \operatorname{mean}_n(\varphi)),
\,(\omega\mapsto a_{\mathrm{const}})
\right\rangle_n
=
a_{\mathrm{const}}
\left(
\operatorname{mean}_n(\varphi)-\operatorname{mean}_n(\varphi)
\right)
=0 .
\]
Hence
\[
\Pi_{n,0}\varphi=\omega\mapsto \operatorname{mean}_n(\varphi).
\]
Evaluating the full decomposition at \(A\in\Omega_{n,M}\) and subtracting the degree-zero component gives
\[
\sum_{u=1}^{M}\Pi_{n,u}\varphi(A)
=
\varphi(A)-\operatorname{mean}_n(\varphi).
\]
Applied to \(\varphi=f\), this is the centered projection identity with the algebraic uniform mean. % lean: sum_positive_harmonicProjection_eq_center

\item The averaging operator \(\mathbb E_n\) is the uniform-slice expectation, so for every \(f:\Omega_{n,M}\to\mathbb R\),
\[
\mathbb E_n f
=
\frac{1}{|\Omega_{n,M}|}\sum_{\omega\in\Omega_{n,M}}f(\omega)
=
\operatorname{mean}_n(f).
\]
Substituting this equality into the preceding display yields
\[
\sum_{u=1}^{M}\Pi_{n,u}f(A)
=
f(A)-\mathbb E_n f .
\]
When \(M=0\), the same formula reads \(0=f(A)-\mathbb E_n f\), because the slice has one point and the degree-zero projection is \(f\). % lean: canonicalJohnsonOrthogonalDecomposition

\item It remains to check distinct-degree orthogonality. For any \(k\in\{0,\ldots,M\}\) and any slice function \(f\),
\[
\Pi_{n,k}f\in\mathcal H_{n,k},
\]
because \(\Pi_{n,k}\) is the orthogonal projection onto \(\mathcal H_{n,k}\). If \(k<\ell\), then
\[
\mathcal H_{n,\ell}\subseteq \mathcal F_{\le \ell-1}^{\perp},
\qquad
\mathcal H_{n,k}\subseteq \mathcal F_{\le k}\subseteq \mathcal F_{\le \ell-1},
\]
and therefore every element of \(\mathcal H_{n,k}\) is orthogonal to every element of \(\mathcal H_{n,\ell}\). The case \(\ell<k\) is symmetric. Thus, whenever \(k\ne \ell\),
\[
\left\langle \Pi_{n,k}f,\Pi_{n,\ell}g\right\rangle_n=0 .
\]
This proves the pairwise orthogonality assertion. % lean: johnsonHarmonic_pairwise_orthogonal
\end{enumerate}
\end{proof}
