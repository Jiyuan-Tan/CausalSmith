\section{Appendix: proofs and auxiliary constructions}

This appendix collects the auxiliary statements used to support the main rate calculations. The first group records the observable filtration and the finite-prefix quantities that enter the signed-depth likelihood comparison.

\begin{definitionv}{def:filter-quantities}[Filter quantities \(\mathcal F_{t-1}^{\mathrm{obs}},\ell_t,q_t^v,a_t^{(Q)}\)]
Define
\[
\mathcal F_{t-1}^{\mathrm{obs}}
:=
\sigma(A_0,Y_0,\ldots,A_{t-1},Y_{t-1}),
\]
\[
\ell_t:=\min\{Q,t\},
\]
\[
q_t^v
:=
\Pr_v(J_t=Q\mid\mathcal F_{t-1}^{\mathrm{obs}}),
\]
and
\[
a_t^{(Q)}
:=
L^{-(Q-\ell_t)}
\prod_{j=t-\ell_t}^{t-1}\mathbf 1\{A_j=1\},
\]
where the empty product is equal to one.
\end{definitionv}

The quantities in \cref{obj:def:filter-quantities} separate two sources of information in the observed history. The posterior \(q_t^v\) captures the chance that the hidden depth has reached its terminal value, while the factor \(a_t^{(Q)}\) records the visible action pattern needed for that event under the rare-action construction. This separation is the device that makes the observed likelihood calculation finite-prefix and observable.

The upper-bound argument begins with local second-moment and covariance controls for the PHIW scores. These statements isolate, respectively, the contribution from a single weighted window and the dependence between overlapping windows.

\begin{lemmav}{lem:phiw-window-second-moment}[Window second moment]
Let \(T,n_X,n_H\in\mathbb N\), let \(k\in\mathbb N\), let \(L\in\mathbb R\), and let \(M\) be a raw POMDP experiment on finite observed and hidden alphabets. Suppose that:
\begin{itemize}
\item \textbf{(Sequential ignorability.)} \(M\) satisfies \cref{obj:ass:sequential-ignorability}.
\item \textbf{(Policy overlap.)} \(M\) satisfies \cref{obj:ass:policy-overlap} with constant \(L\).
\item \textbf{(Overlap level.)} \(L\ge 1\).
\item \textbf{(Kernel law.)} \(M\) satisfies \cref{obj:ass:pomdp-kernel}.
\item \textbf{(Bounded rewards.)} \(M\) satisfies \cref{obj:ass:bounded-reward}.
\item \textbf{(Window position.)} \(t\in\{0,\ldots,T-1\}\) and \(k\le t\).
\end{itemize}
For the depth-\(k\) score \(Z_t(k)\) of \cref{obj:def:phiw-estimator}, under the observed trajectory law of \(M\),
\[
\mathbb E_{\mathrm{obs}}\bigl[Z_t(k)^2\bigr]\le L^{k+1}.
\]
\end{lemmav}

\begin{lemmav}{lem:phiw-overlapping-window-covariance}[Overlapping-window covariance]
Let \(M\) be a raw POMDP experiment of length \(T\) on finite observed and hidden alphabets. Let \(k,h\in\mathbb N\), let \(L\in\mathbb R\), and let \(t,u\in\{0,\ldots,T-1\}\). Assume:
\begin{itemize}
\item \textbf{(Model conditions.)} \(M\) satisfies \cref{obj:ass:sequential-ignorability}, \cref{obj:ass:policy-overlap} with overlap constant \(L\), \cref{obj:ass:pomdp-kernel}, and \cref{obj:ass:bounded-reward}.
\item \textbf{(Overlap constant.)} \(L\ge 1\).
\item \textbf{(Initial window.)} \(k\le t\).
\item \textbf{(Lag identity.)} \(u=t+h\).
\item \textbf{(Overlapping lag.)} \(1\le h\le k\).
\end{itemize}
Then the depth-\(k\) scores \(Z_t(k)\) of \cref{obj:def:phiw-estimator}, evaluated under the observed trajectory law of \(M\), satisfy
\[
\bigl|\operatorname{Cov}_{\mathrm{obs}}\bigl(Z_t(k),Z_u(k)\bigr)\bigr|
\le
2L^{k+1-h}.
\]
\end{lemmav}

The analysis of separated windows uses both the observed law and the full trajectory law. The next definitions fix those probability spaces and the observation map connecting them.

\begin{definitionv}{synth_15}[Observed-trajectory law]
For a raw POMDP experiment \(M=(\mathcal X,\mathcal H,K,b,e,\mathcal L(\mathsf O_T))\), let
\[
\mathbb P_M^{\mathrm{obs}}:=\operatorname{Law}^{\mathrm{obs}}_M
\]
denote the probability law of the observed trajectory
\[
\mathsf O_T=((X_t,A_t,Y_t))_{t=0}^{T-1}
\]
induced by the behavior-policy experiment \(M\). Equivalently, under \(\mathbb P_M^{\mathrm{obs}}\), \(S_0\sim d_b\), \(A_t\sim b(\cdot\mid X_t)\), and \(K\) generates \((Y_t,S_{t+1})\) from \((S_t,A_t)\), with only \((X_t,A_t,Y_t)_{t=0}^{T-1}\) retained.
\end{definitionv}

\begin{definitionv}{synth_17}[Full trajectory law and observation map]
For a raw POMDP experiment \(M=(\mathcal X,\mathcal H,K,b,e,\mathcal L(\mathsf O_T))\) with joint state space \(\mathcal S=\mathcal X\times\mathcal H\), let \(\mathbb P_M\) denote the full behavior-policy trajectory law on
\[
(S_0,A_0,Y_0,S_1,\ldots,A_{T-1},Y_{T-1},S_T).
\]
Under \(\mathbb P_M\), the initial state satisfies \(S_0\sim d_b\), each action is drawn as \(A_t\sim b(\cdot\mid X_t)\), and the kernel \(K\) generates \((Y_t,S_{t+1})\) from \((S_t,A_t)\), for \(t=0,\ldots,T-1\).

Let \(O\) be the observation map from a full trajectory to its observed trajectory,
\[
O(S_0,A_0,Y_0,\ldots,A_{T-1},Y_{T-1},S_T)
:=((X_t,A_t,Y_t))_{t=0}^{T-1},
\]
where \(S_t=(X_t,H_t)\). The observed behavior-policy law is the pushforward
\[
\mathbb P_M^{\mathrm{obs}}:=\mathbb P_M\circ O^{-1}.
\]
\end{definitionv}

Together, \cref{obj:synth_15,obj:synth_17} let the covariance calculation move between observable scores and latent-state transition operators. The observed law is the sampling law for the estimator, while the full law retains the latent state needed to apply contraction after a gap.

\begin{lemmav}{lem:phiw-separated-window-peeling}[Separated window peeling]
Let \(M\) be a raw POMDP experiment with trajectory length \(T\), and let \(L\), the policy-overlap constant, be real. Let \(Z_s(k)\) denote the depth-\(k\) partial-history weighted reward score from \cref{obj:def:phiw-estimator}. Suppose:
\begin{itemize}
\item \textbf{(Model conditions.)} \(M\) satisfies \cref{obj:ass:sequential-ignorability,obj:ass:policy-overlap,obj:ass:pomdp-kernel,obj:ass:bounded-reward} with overlap constant \(L\).
\item \textbf{(Overlap scale.)} \(1\le L\).
\item \textbf{(Mature earlier window.)} The history depth \(k\) satisfies \(k\le t\).
\item \textbf{(Separated future window.)} For an epoch \(t\) and a gap \(\gamma\in\mathbb N\), the epoch
\[
u=t+1+\gamma+k
\]
lies in the observed horizon, equivalently \(t+1+\gamma+k<T\).
\end{itemize}
For \(p\in\{b,e\}\), let \(P_p\) be the policy-induced transition operator obtained from the reward-marginalized kernel under policy \(p\), let
\[
g_e(s)=\sum_a e(a\mid x(s))\int y\,K(dy\times\mathcal S\mid s,a),
\]
and put
\[
\Phi_{\gamma,k}(s)=P_b^\gamma P_e^k g_e(s).
\]
Then, with \(\mathbb P_M^{\mathrm{obs}}\) the observed law induced by the observed trajectory map \(O\),
\[
\int Z_t(k)(w)Z_u(k)(w)\,d\mathbb P_M^{\mathrm{obs}}(w)
=
\int Z_t(k)(O(\tau))\,\Phi_{\gamma,k}(S_{t+1}(\tau))\,d\mathbb P_M(\tau).
\]
\end{lemmav}

\Cref{obj:lem:phiw-separated-window-peeling} rewrites a separated score product as an earlier observable score multiplied by a future continuation value evaluated at \(S_{t+1}\). This identity supplies the point where the gap between windows becomes a behavior-transition power, followed by a target-policy window.

\begin{lemmav}{lem:behavior-support-operator-congruence}[Behavior-support operator congruence]
Let \(M\in\mathcal M_T(t_0,\zeta,C)\) satisfy \cref{obj:def:model-class}, with behavior policy \(b\), target policy \(e\), and the policy-overlap condition in \cref{obj:ass:policy-overlap}.  For \(p\in\{b,e\}\), write
\[
P_p(s,s')=\sum_{a\in\{0,1\}} p(a\mid x(s))\,K(\{S'=s'\}\mid s,a),
\]
and let \(P_p^0 h=h\) and \(P_p^{m+1}h=P_p(P_p^m h)\).  Let \(d_b\) denote the behavior stationary law of \(P_b\).

Assume:
\begin{itemize}
\item \textbf{(Policy choice.)} The policy \(p\) is either \(b\) or \(e\).
\item \textbf{(Terminal agreement.)} The functions \(f,g:\mathcal S\to\mathbb R\) satisfy
\[
f(s)=g(s)
\]
for every joint state \(s\) with \(d_b(s)>0\).
\item \textbf{(Initial support.)} The joint state \(s\) satisfies \(d_b(s)>0\).
\end{itemize}
Then, for every \(m\in\mathbb N\),
\[
(P_p^m f)(s)=(P_p^m g)(s).
\]
\end{lemmav}

The support congruence in \cref{obj:lem:behavior-support-operator-congruence} keeps transition-operator calculations on the behavior support. Because the overlap condition makes target transitions compatible with behavior support, functions that agree on the behavior-stationary support continue to agree after iterating either policy operator from a supported state.

\begin{lemmav}{lem:phiw-separated-window-covariance}[Separated window covariance]
Let \(T,n_X,n_H,k,h\in\mathbb N\), let \(t_0,\zeta,C\in\mathbb R\), and let \(M\in\mathcal M_T(t_0,\zeta,C)\) be a raw POMDP experiment as in \cref{obj:def:model-class}, with the uniform contraction restriction of \cref{obj:ass:uniform-contraction}. Write the contraction factor as \(\alpha=e^{-1/t_0}\). Let \(t,u\in\{0,\ldots,T-1\}\) satisfy
\[
k\le t,\qquad u=t+h,\qquad k+1\le h .
\]
Then the depth-\(k\) scores \(Z_t(k)\) of \cref{obj:def:phiw-estimator}, evaluated under the observed trajectory law of \(M\), satisfy
\[
\bigl|\operatorname{Cov}_{\mathrm{obs}}\bigl(Z_t(k),Z_u(k)\bigr)\bigr|
\le
2\alpha^{h-k-1}.
\]
\end{lemmav}

Together, \cref{obj:lem:phiw-window-second-moment,obj:lem:phiw-overlapping-window-covariance,obj:lem:phiw-separated-window-peeling,obj:lem:behavior-support-operator-congruence,obj:lem:phiw-separated-window-covariance} decompose the sampling fluctuation into a weighted-window term and a geometric dependence tail. The overlap condition controls the inflation inside the window, while contraction controls pairs whose windows are separated by at least one time step.

The next statement sums those covariance bounds over future times. It is the finite-sample accounting step that turns the local bounds into a variance inequality for the average score.

\begin{lemmav}{lem:phiw-future-covariance-sum}[Aggregate future-covariance bound]
Let \(T\ge1\), \(k\in\mathbb N\), \(t_0,\zeta,C\in\mathbb R\), and let \(M\in\mathcal M_T(t_0,\zeta,C)\) be as in \cref{obj:def:model-class}. Write \(L=e^{\zeta}\), \(\alpha=e^{-1/t_0}\), and \(\mathscr T_k=\{t:k\le t\le T-1\}\). Then, for every \(t\in\mathscr T_k\), the depth-\(k\) scores of \cref{obj:def:phiw-estimator} satisfy, under the observed trajectory law of \(M\),
\[
\sum_{\substack{u\in\mathscr T_k\\ t<u}}
\bigl|\operatorname{Cov}_{\mathrm{obs}}\bigl(Z_t(k),Z_u(k)\bigr)\bigr|
\le
\frac{2L^{k+1}}{L-1}+\frac{2}{1-\alpha}.
\]
\end{lemmav}

\Cref{obj:lem:phiw-future-covariance-sum} supplies the two terms that reappear in the assembled variance bound. The geometric sum over overlapping lags gives the policy-overlap contribution, and the contraction tail gives the dependence contribution.

\begin{lemmav}{lem:phiw-raw-variance}[Raw variance assembly]
Let \(T\) and \(k\) be nonnegative integers, let \(t_0,\zeta,C\in\mathbb R\), and let \(M\) be a raw POMDP experiment. Write \(L=e^\zeta\) for the policy-overlap factor and \(\alpha=e^{-1/t_0}\) for the contraction factor. Suppose that
\begin{itemize}
\item \textbf{(Horizon.)} \(T\ge 1\).
\item \textbf{(Model class.)} \(M\in\mathcal M_T(t_0,\zeta,C)\), with \(\mathcal M_T(t_0,\zeta,C)\) as in \cref{obj:def:model-class}.
\item \textbf{(History depth.)} \(k\le \lfloor T/2\rfloor\).
\end{itemize}
For the depth-\(k\) scores \(Z_t(k)\) of \cref{obj:def:phiw-estimator}, form the unclipped partial-history average
\[
\frac{1}{T-k}\sum_{t=k}^{T-1} Z_t(k).
\]
Under the observed trajectory law obtained by projecting the full trajectory law of \(M\) onto observed histories, its variance satisfies
\[
\operatorname{Var}\!\left(\frac{1}{T-k}\sum_{t=k}^{T-1} Z_t(k)\right)
\le
\frac{2}{T}\left\{
L^{k+1}\left(1+\frac{4}{L-1}\right)
+
\frac{4}{1-\alpha}
\right\}.
\]
\end{lemmav}

The two terms in \cref{obj:lem:phiw-raw-variance} have distinct origins: multiplying observable policy ratios over a window produces the \(L^{k+1}\) term, while serial dependence contributes the contraction term. The bound is stated for the unclipped average because the displayed variance is the component used before the clipped estimator's squared-error risk is assembled.

The next statement supplies the fixed-radius upper bound for the partial-history importance-weighted estimator. It gives both the rate bound for the balanced depth and the corresponding variance calculation for the unclipped average.

\begin{lemmav}{lem:phiw-upper}[Balanced PHIW upper bound]
Fix constants satisfying
\begin{itemize}
\item \textbf{(Mixing scale.)} \(t_0>0\).
\item \textbf{(Policy-overlap scale.)} \(\zeta>0\).
\item \textbf{(Latent-overlap radius.)} \(C\ge 1\).
\end{itemize}
Let \(k_T\) be the balanced history depth in \cref{obj:def:history-depth}, and set
\[
\beta=\frac{2}{2+t_0\zeta}.
\]
Then there exist constants \(c^\star>0\) and \(T_\star\in\mathbb N\) such that, for every \(T\ge T_\star\), the estimator-specific worst-case squared risk from \cref{obj:synth_13} of the observable PHIW estimator \(\widehat\theta_{k_T}\) from \cref{obj:def:phiw-estimator} satisfies
\[
R_T(t_0,\zeta,C;\widehat\theta_{k_T})
\le
c^\star T^{-\beta}.
\]
Moreover, for every \(T\ge T_\star\), every finite alphabet sizes \(n_X,n_H\), and every raw POMDP experiment \(M\in\mathcal M_T(t_0,\zeta,C)\), the variance under the observed trajectory law induced by \(M\) of the unclipped partial-history average
\[
\frac{1}{T-k_T}\sum_{t=k_T}^{T-1} Z_t(k_T)
\]
satisfies
\[
\operatorname{Var}
\!\left(\frac{1}{T-k_T}\sum_{t=k_T}^{T-1} Z_t(k_T)\right)
\le
\frac{2}{T}
\left[
e^{\zeta(k_T+1)}
\left(1+\frac{4}{e^\zeta-1}\right)
+
\frac{4}{1-e^{-1/t_0}}
\right].
\]
\end{lemmav}

The bound in \cref{obj:lem:phiw-upper} identifies the observable estimator used for the upper side of the fixed-\(C\) comparison. The variance display isolates the cost of multiplying observable policy ratios over a window of length \(k_T\) and the serial-dependence contribution controlled by contraction. Balancing those terms gives the exponent \(\beta\) under fixed overlap and fixed latent-stationary domination.

For the lower side, the appendix uses the signed-depth alternatives introduced in \cref{obj:def:signed-depth-family}. The next two statements record the contraction and stationary-law calculations that make the construction a member of the same finite-POMDP class.

\begin{lemmav}{lem:signed-depth-constant-policy-contraction}[Contraction of the constant-policy signed-depth kernel]
Let \(t_0>0\), \(\zeta>0\), \(C\ge1\), \(Q\ge1\), \(v\in\{-1,+1\}\) and \(r\in[0,1]\), and write \(\alpha=e^{-1/t_0}\). Let \(P^{(r)}\) be the transition kernel induced on the joint state space \(\mathcal S_Q=\{0\}\times(\{0,\ldots,Q\}\times\{-1,+1\})\) of the signed-depth alternative of \cref{obj:def:signed-depth-family} with terminal depth \(Q\) and index \(v\) by the constant policy \(a_r\) that plays action one with probability \(r\). Then, for all probability vectors \(\mu,\mu'\) on \(\mathcal S_Q\),
\[
\|\mu P^{(r)}-\mu'P^{(r)}\|_{\mathrm{TV}}
\le
\alpha\|\mu-\mu'\|_{\mathrm{TV}} .
\]
\end{lemmav}

\begin{lemmav}{lem:signed-depth-stationary-law}[Closed form of the signed-depth stationary law]
Let \(t_0>0\), \(\zeta>0\), \(C\ge1\), \(Q\ge1\), \(v\in\{-1,+1\}\) and \(r\in[0,1]\). Write \(\alpha=e^{-1/t_0}\), \(\varepsilon_C=\min\{(C-1)/2,1/2\}\), and \(w_j=(1-\alpha)\alpha^{j}/(1-\alpha^{Q+1})\). Let \(P^{(r)}\) be the kernel induced on \(\mathcal S_Q\) by the constant policy \(a_r\) for the signed-depth alternative of \cref{obj:def:signed-depth-family} with terminal depth \(Q\) and index \(v\), and let \(d_r\) be the stationary law selected for \(P^{(r)}\). Then, for every hidden state \(h=(j(h),u(h))\),
\[
d_r(0,h)
=
w_{j(h)}\,\frac{1+u(h)\varepsilon_Cr^{j(h)}}{2}.
\]
\end{lemmav}

\Cref{obj:lem:signed-depth-constant-policy-contraction} gives the uniform total-variation contraction of the signed-depth transition under any constant action-one probability. \Cref{obj:lem:signed-depth-stationary-law} then gives the stationary law explicitly: the depth mass is a truncated geometric sequence, and the sign imbalance is scaled by the action-one probability to the current depth.

\begin{lemmav}{lem:signed-depth-membership}[Signed-depth membership]
For every \(T\ge 1\), \(Q\ge 1\), \(t_0>0\), \(\zeta>0\), \(C>1\), and \(v\in\{-1,+1\}\), let \(M_{Q,v}^T\) be the signed-depth experiment of \cref{obj:def:signed-depth-family}.  Put
\[
\varepsilon_C=\min\{(C-1)/2,1/2\},
\qquad
L=\exp(\zeta),
\qquad
w_j=\frac{(1-\alpha(t_0))\alpha(t_0)^j}{1-\alpha(t_0)^{Q+1}},
\]
where \(\alpha(t_0)\) is the mixing factor and \(j=0,\ldots,Q\).  Then:
\begin{itemize}
\item \textbf{(Membership.)} \(M_{Q,v}^T\in \mathcal M_T(t_0,\zeta,C)\) in the sense of \cref{obj:def:model-class}.
\item \textbf{(Behavior stationary law.)} For every hidden state \(h\), with decoded depth \(j(h)\in\{0,\ldots,Q\}\) and sign \(u(h)\in\{-1,+1\}\),
\[
d_b(0,h)
=
w_{j(h)}
\frac{1+u(h)\varepsilon_C L^{-j(h)}}{2}.
\]
\item \textbf{(Constant action-one stationary laws.)} For every \(r\in[0,1]\), if \(d_r\) denotes the stationary law induced by the constant policy that chooses action one with probability \(r\), then for every hidden state \(h\),
\[
d_r(0,h)
=
w_{j(h)}
\frac{1+u(h)\varepsilon_C r^{j(h)}}{2}.
\]
\item \textbf{(Target-over-behavior domination.)} For every joint state \(s\),
\[
d_e(s)\le (1+\varepsilon_C)d_b(s),
\qquad
1+\varepsilon_C\le C.
\]
\end{itemize}
The stationary target value of \cref{obj:def:target-value} is
\[
\theta(K,e)
=
v\,c_0(t_0)\,\varepsilon_C\,w_Q .
\]
\end{lemmav}

The stationary laws in \cref{obj:lem:signed-depth-membership} show how the lower construction keeps the target-over-behavior ratio bounded while concentrating the signal at terminal depth. The terminal value is proportional to the stationary depth mass \(w_Q\), so increasing \(Q\) makes the two alternatives harder to separate from observed data while preserving the fixed latent-overlap radius.

The likelihood calculation rests on finite-prefix posterior bounds, a rare-action second moment, a bootstrap factor that makes the posterior control uniform in the terminal depth, signed-mass identities at terminal depth, exact conditional reward means, and finite-word KL inequalities. The next group develops these ingredients in the order in which the observed-path comparison uses them.

\begin{lemmav}{lem:posterior-lower-window-power}[Posterior window power bound]
Let \(N,Q,k,d\in\mathbb N\), let \(t_0,\zeta,C,b\in\mathbb R\), let \(v\in\{-1,+1\}\), and let \(o=((a_0,y_0),\ldots,(a_{k-1},y_{k-1}))\) be an observed prefix for the signed-depth family of \cref{obj:def:signed-depth-family}.  Write
\[
c_0(t_0)=\frac{1-\alpha(t_0)}{4},
\]
and, for \(0\le m\le k\), write \(o_{\le m}\) for the first \(m\) observed pairs.  For any observed prefix \(o'\), define the raw terminal-depth posterior by
\[
\pi_v(o')
=
\frac{\sum_h \mathbf 1\{J(h)=Q\}\lambda_v(h;o')}
     {\sum_h \lambda_v(h;o')}.
\]
Assume:
\begin{itemize}
\item \textbf{(Nonnegative amplitude.)} \(0\le c_0(t_0)\).
\item \textbf{(Nonnegative base.)} \(0\le 1-c_0(t_0)b\).
\item \textbf{(Window length.)} \(d\le k\).
\item \textbf{(Strict-prefix posterior bound.)} For every \(i\in\{0,\ldots,k-1\}\),
\[
\pi_v(o_{\le i})\le b.
\]
\end{itemize}
Define the final \(d\)-prefix lower window by
\[
\Lambda_d(o)
=
\prod_{\substack{0\le i<k\\ k-d\le i}}
\left(1-c_0(t_0)\pi_v(o_{\le i})\right).
\]
Then
\[
\bigl(1-c_0(t_0)b\bigr)^d\le \Lambda_d(o).
\]
\end{lemmav}

\begin{lemmav}{lem:terminal-posterior-mature-window}[Terminal posterior window bound]
Let \(N,Q\in\mathbb N\), let \(t_0,\zeta,C\in\mathbb R\), and let \(v\in\{-1,+1\}\).  Put \(\alpha=\exp\{-(1/t_0)\}\).  In the finite signed-depth model of \cref{obj:def:signed-depth-family}, for an observed prefix \(o=((a_1,y_1),\ldots,(a_k,y_k))\), write
\[
\pi_v(o)
=
\frac{
\sum_{h\in\{0,\ldots,Q\}\times\{-1,+1\}}
\mathbf 1\{J(h)=Q\}\lambda_v(h;o)
}{
\mathsf M_v(o)
}.
\]
For \(d\in\mathbb N\), define the lower posterior window
\[
\Lambda_d(o)
=
\prod_{\substack{0\le i<k\\ k-d\le i}}
\left(1-c_0(t_0)\,\pi_v(o_{\le i})\right),
\]
where \(o_{\le i}\) is the prefix of \(o\) containing its first \(i\) action--reward pairs.

Assume:
\begin{itemize}
\item \textbf{(Positive scale.)} \(t_0>0\).
\item \textbf{(Positive tilt.)} \(\zeta>0\).
\item \textbf{(Overlap level.)} \(C>1\).
\item \textbf{(Mature window.)} \(Q\le k\).
\end{itemize}
Then the terminal-depth posterior satisfies
\[
\pi_v(o)
\le
\frac{\alpha^Q}{\Lambda_Q(o)}.
\]
\end{lemmav}

\begin{lemmav}{lem:terminal-posterior-initial-window}[Initial-window posterior bound]
Let \(N,Q\in\mathbb N\), let \(t_0,\zeta,C\in\mathbb R\), let \(v\in\{-1,+1\}\), let \(k\in\mathbb N\), and let
\[
o:\{0,\ldots,k-1\}\to \{0,1\}\times\{0,1\}
\]
be an observed action/reward prefix for the signed-depth construction of \cref{obj:def:signed-depth-family}.  Write \(\alpha=e^{-1/t_0}\).  For any prefix \(o'\), define the raw terminal-depth posterior by
\[
\pi_v(o')
=
\frac{\sum_{h:\,J(h)=Q}\lambda_v(h;o')}
     {\sum_h \lambda_v(h;o')}.
\]
Define the initial lower posterior window over the available prefixes by
\[
\Lambda_k(o)
=
\prod_{i=0}^{k-1}
\left(1-c_0\,\pi_v(o_{\le i})\right),
\]
where \(o_{\le i}\) is the prefix of \(o\) of length \(i\).

Assume:
\begin{itemize}
\item \textbf{(Positive scale.)} \(t_0>0\).
\item \textbf{(Positive overlap exponent.)} \(\zeta>0\).
\item \textbf{(Nontrivial overlap radius.)} \(C>1\).
\item \textbf{(Initial window.)} \(k<Q\).
\end{itemize}
Then the terminal-depth posterior after observing \(o\) satisfies
\[
\pi_v(o)\le \frac{\alpha^Q}{\Lambda_k(o)}.
\]
\end{lemmav}

\begin{lemmav}{lem:terminal-posterior-crude-bound}[Crude terminal posterior bound]
Let \(N,Q,k\in\mathbb N\), let \(t_0,\zeta,C\in\mathbb R\), and let \(v\in\{-1,+1\}\). Suppose that
\begin{itemize}
\item \textbf{(Positive temperature.)} \(t_0>0\).
\item \textbf{(Positive log-overlap.)} \(\zeta>0\).
\item \textbf{(Overlap level.)} \(C>1\).
\end{itemize}
For the finite signed-depth model of \cref{obj:def:signed-depth-family}, write
\[
\alpha=\exp(-1/t_0),
\qquad
c_0=\frac{1-\alpha}{4}.
\]
For every observed prefix \(o\in(\{0,1\}\times\{0,1\})^k\), define the raw terminal-depth posterior by
\[
\pi_v(o)
=
\frac{
\sum_h \mathbf 1\{J(h)=Q\}\,\lambda_v(h;o)
}{
\mathsf M_v(o)
},
\qquad
\mathsf M_v(o)=\sum_h \lambda_v(h;o),
\]
where the sums range over the signed-depth hidden states at depth horizon \(Q\). Then
\[
\pi_v(o)
\le
\left(\frac{\alpha}{1-c_0}\right)^Q .
\]
\end{lemmav}

\begin{lemmav}{lem:refined-terminal-posterior}[Refined terminal posterior bound]
Let \(N,Q,k\) be nonnegative integers. Suppose that
\begin{itemize}
\item \textbf{(Positive scale.)} \(t_0>0\).
\item \textbf{(Positive perturbation.)} \(\zeta>0\).
\item \textbf{(Overlap level.)} \(C>1\).
\end{itemize}
Write \(\alpha=\exp(-1/t_0)\) for the contraction parameter and
\(c_0=(1-\alpha)/4\) for the signed-depth reward-amplitude constant. For an
alternative \(v\in\{-1,+1\}\) in the signed-depth family of
\cref{obj:def:signed-depth-family} and any observed prefix
\(o=(a_1,y_1,\ldots,a_k,y_k)\), let \(\pi_{N,Q,v}(o)\) be the raw
terminal-depth posterior: the run-local unnormalized filter mass at latent
depth \(Q\), divided by the total run-local unnormalized filter mass of the
prefix. Then
\[
\pi_{N,Q,v}(o)
\le
\frac{\alpha^Q}
{\left\{1-c_0\left(\frac{\alpha}{1-c_0}\right)^Q\right\}^Q}.
\]
\end{lemmav}

The posterior bounds in \cref{obj:lem:posterior-lower-window-power,obj:lem:terminal-posterior-mature-window,obj:lem:terminal-posterior-initial-window,obj:lem:terminal-posterior-crude-bound,obj:lem:refined-terminal-posterior} separate the survival of the terminal depth from the normalization accumulated along the observed prefix. The mature-window and initial-window cases cover prefixes on the two sides of the terminal depth \(Q\), the crude bound supplies a uniform starting point, and the refined statement packages these ingredients into the terminal-posterior control used below.

\begin{lemmav}{lem:rare-action-second-moment-exact}[Rare-action second moment]
Let \(Q,k\in\mathbb N\), let \(t_0,\zeta,C\in\mathbb R\), and let \(v\in\{-1,+1\}\). Assume:
\begin{itemize}
\item \textbf{(Time scale.)} \(t_0>0\).
\item \textbf{(Overlap exponent.)} \(\zeta>0\).
\item \textbf{(Radius constant.)} \(C>1\).
\end{itemize}
Write \(L=\exp(\zeta)\), let \(\ell=\min\{Q,k\}\), and consider the finite signed-depth model of \cref{obj:def:signed-depth-family} at horizon \(k+1\). For an action/reward-symbol prefix
\[
o=((a_1,y_1),\ldots,(a_k,y_k)),
\]
let \(\pi_k^v(o)\) be its prefix mass under this model, and define the rare-action-history factor from \cref{obj:def:filter-quantities} by
\[
\mathfrak a_k(o)
=
L^{-(Q-\ell)}
\prod_{i:\,k-\ell<i\le k}\mathbf 1\{a_i=1\},
\]
with the empty product equal to one. Then
\[
\sum_{o}\pi_k^v(o)\,\mathfrak a_k(o)^2
=
L^{-2(Q-\ell)}
\left(\frac{1}{L}\right)^{|\{i:\ k-\ell<i\le k\}|},
\]
where the sum ranges over all action/reward-symbol prefixes of length \(k\).
\end{lemmav}

\Cref{obj:lem:rare-action-second-moment-exact} gives the exact second moment of the visible rare-action factor. The displayed expression isolates the contribution of the forced terminal-depth action pattern, and this is the \(L^{-Q}\) term that enters the observed KL calculation after summing conditional mean differences.

\begin{lemmav}{lem:finite-signed-depth-kl}[Finite observed KL bound]
Let \(T,Q\in\mathbb N\) and \(t_0,\zeta,C,K_0\in\mathbb R\).  Write \(\alpha=e^{-1/t_0}\), \(L=e^\zeta\), \(c_0=(1-\alpha)/4\), and \(\varepsilon_C=\min\{(C-1)/2,1/2\}\).  Suppose that
\begin{itemize}
\item \textbf{(Positive scale.)} \(t_0>0\), \(\zeta>0\), \(C>1\), and \(K_0>0\).
\item \textbf{(Uniform terminal posterior.)} For every horizon \(N\), every alternative index \(v\in\{-1,+1\}\), every time \(t<N\), and every observed prefix \(o\) of actions and two-symbol rewards before \(t\), the finite conditional terminal-depth probability \(q_t^v(o)\) from \cref{obj:def:filter-quantities} satisfies
\[
q_t^v(o)\le K_0\alpha^Q .
\]
\end{itemize}
Then the finite observed-word laws of the signed-depth alternatives of \cref{obj:def:signed-depth-family}, taken before the reward symbols are decoded as real rewards, satisfy
\[
\operatorname{KL}\!\left(\mathbb P_{Q,+1}^{\mathrm{fin},T}\middle\|\mathbb P_{Q,-1}^{\mathrm{fin},T}\right)
\le
\frac{4c_0^2K_0^2}{1-c_0^2}\,
T\,\varepsilon_C^2\,\alpha^{2Q}\,L^{-Q}.
\]
\end{lemmav}

\begin{lemmav}{lem:signed-depth-bootstrap-factor}[Uniform bootstrap factor]
Let \(t_0>0\) and write \(\alpha=e^{-1/t_0}\), \(c_0=(1-\alpha)/4\), \(\eta_0=\alpha/(1-c_0)\). Then there exists \(K_0>0\) such that
\[
\frac{1}{\bigl(1-c_0\bigl(\alpha/(1-c_0)\bigr)^{Q}\bigr)^{Q}}
\le
K_0
\qquad(Q\in\mathbb N).
\]
\end{lemmav}

\begin{lemmav}{lem:signed-depth-terminal-sign-invariant}[Terminal signed mass invariant]
Let \(N,Q,k\in\mathbb N\), let \(t_0,\zeta,C\in\mathbb R\), let \(v\in\{-1,+1\}\), and consider the finite signed-depth model of \cref{obj:def:signed-depth-family}.  For an observed prefix \(o=((a_i,y_i))_{i=1}^k\), write \(\lambda_v(h;o)\) for the unnormalized filter mass of terminal hidden state \(h\), write \(J(h)\in\{0,\ldots,Q\}\) and \(U(h)\in\{-1,+1\}\) for its depth and sign coordinates, and set
\[
\varepsilon_C=\min\left\{\frac{C-1}{2},\frac12\right\}.
\]
Assume:
\begin{itemize}
\item \textbf{(Positive initial scale.)} \(t_0>0\).
\item \textbf{(Positive overlap exponent.)} \(\zeta>0\).
\item \textbf{(Nontrivial overlap radius.)} \(C>1\).
\end{itemize}
With \(L=e^\zeta\), define the depth-\(Q\) action-history factor
\[
a^{(Q)}(o)
=
L^{-\max\{Q-k,0\}}
\prod_{\substack{1\le i\le k\\ i>k-\min\{Q,k\}}}
\mathbf 1\{a_i=1\}.
\]
Then
\[
\sum_{h:\,J(h)=Q} U(h)\lambda_v(h;o)
=
\varepsilon_C\,a^{(Q)}(o)
\sum_{h:\,J(h)=Q}\lambda_v(h;o).
\]
\end{lemmav}

\begin{lemmav}{lem:signed-depth-next-reward-numerator}[Next reward numerator]
Let \(U,k,N,Q\in\mathbb N\), let \(v\in\{-1,+1\}\), and consider the finite signed-depth model of \cref{obj:def:signed-depth-family} with horizon \(N\), terminal depth \(Q\), initial weights \(p_v\), behavior policy \(b\), and reward-symbol/hidden-state kernel \(K_v\). Suppose that
\begin{itemize}
\item \textbf{(Positive time scale.)} \(t_0>0\).
\item \textbf{(Positive behavior perturbation.)} \(\zeta>0\).
\item \textbf{(Overlap radius.)} \(C>1\).
\end{itemize}
Write
\[
c_0(t_0)=\frac{1-\alpha}{4},
\]
where \(\alpha\) is the signed-depth mixing parameter, and write \(\phi(0)=-1\), \(\phi(1)=1\).  For any observed action/reward-symbol prefix \(o\in(\{0,1\}\times\{0,1\})^k\), let
\[
\lambda_v(h;o)
\]
denote the unnormalized signed-depth filter mass, after the prefix \(o\), assigned to observed state \(0\) and packed hidden state \(h\in\{0,\ldots,2(Q+1)-1\}\).  Decode the packed hidden state by
\[
J(h)=\bigl\lfloor h/2\bigr\rfloor,
\qquad
U(h)=
\begin{cases}
+1, & h \equiv 1 \pmod 2,\\
-1, & h \equiv 0 \pmod 2.
\end{cases}
\]
Then
\[
\sum_{\tau}
\mathbf 1\!\left\{ ((a_i,r_i))_{i=1}^k(\tau)=o\right\}
\,\omega_v(\tau)\,\phi\!\left(r_{k+1}(\tau)\right)
=
v\,c_0(t_0)
\sum_{h=0}^{2(Q+1)-1}
\mathbf 1\{J(h)=Q\}\,U(h)\,\lambda_v(h;o),
\]
where the outer sum is over all finite signed-depth trajectories \(\tau\) of horizon \(U+k+1\), and \(\omega_v(\tau)\) is their chronological path weight under \(p_v\), \(b\), and \(K_v\).
\end{lemmav}

\begin{lemmav}{lem:signed-depth-terminal-prefix-mass}[Terminal prefix mass identity]
Let \(U,k,Q\in\mathbb N\) with \(U\ge 1\), let \(t_0,\zeta,C\in\mathbb R\), let \(v\in\{-1,+1\}\), and consider the finite signed-depth model of \cref{obj:def:signed-depth-family} at horizon \(U+k\).  For an action/reward-symbol prefix \(o\) of length \(k\), define the joint prefix-and-terminal-depth mass at the cut after the first \(k\) epochs by
\[
m_v^Q(o)
=
\Pr_v\{(A_1,R_1),\ldots,(A_k,R_k)=o,\ J_{k+1}=Q\}.
\]
For each hidden coordinate \(h\in\{0,\ldots,2(Q+1)-1\}\), let \(J(h)=\lfloor h/2\rfloor\), and let \(\lambda_v(h;o)\) be the unnormalized filter mass, as in \cref{obj:def:filter-quantities}, assigned after observing \(o\) to the cut state with hidden coordinate \(h\).  Then
\[
m_v^Q(o)
=
\sum_{h=0}^{2(Q+1)-1}
\mathbf 1\{J(h)=Q\}\,\lambda_v(h;o).
\]
\end{lemmav}

\begin{lemmav}{lem:signed-depth-conditional-reward-mean}[Conditional reward mean identity]
Let \(T,Q\in\mathbb N\), let \(t_0,\zeta,C\in\mathbb R\), and let \(v\) be a Boolean alternative, written through
\[
\sigma(v)=
\begin{cases}
1, & v=\mathrm{true},\\
-1, & v=\mathrm{false}.
\end{cases}
\]
Assume
\begin{itemize}
\item \textbf{(Positive baseline.)} \(t_0>0\).
\item \textbf{(Positive policy scale.)} \(\zeta>0\).
\item \textbf{(Overlap radius.)} \(C>1\).
\end{itemize}
For every epoch \(t<T\) and every observed prefix \(o\in\{\,\mathrm{false},\mathrm{true}\,\}^{t}\times\{0,1\}^{t}\), the conditional next-reward mean satisfies
\[
\mathbb E_v[Y_t\mid \mathcal F_{t-1}^{\mathrm{obs}}=o]
=
\sigma(v)\,c_0(t_0)\,\varepsilon_C(C)\,
a_t^{(Q)}(o)\,q_t^v(o),
\]
where the left-hand side is the totalized ratio defining the finite-prefix conditional reward mean, \(c_0(t_0)=(1-\alpha)/4\), \(\varepsilon_C(C)=\min\{(C-1)/2,1/2\}\), and \(a_t^{(Q)}(o)\) and \(q_t^v(o)\) are the rare-action-history factor and terminal posterior of \cref{obj:def:filter-quantities}.
\end{lemmav}

\Cref{obj:lem:signed-depth-bootstrap-factor} turns the finite-prefix normalization in \cref{obj:lem:refined-terminal-posterior} into a constant depending on the mixing scale. \Cref{obj:lem:signed-depth-terminal-sign-invariant,obj:lem:signed-depth-next-reward-numerator,obj:lem:signed-depth-terminal-prefix-mass} connect the signed terminal mass, the next reward numerator, and the prefix mass at terminal depth. These identities convert the latent terminal sign into the observable conditional reward mean in \cref{obj:lem:signed-depth-conditional-reward-mean}, while \cref{obj:lem:finite-signed-depth-kl} converts the resulting terminal-posterior control into a finite observed-word divergence bound. The product \(\alpha^{2Q}L^{-Q}\) reflects the joint requirement that terminal depth survives and that the visible action history carries the rare sequence needed to reveal it.

The likelihood calculation next fixes the observed signed-depth law and combines the exact observed reward mean with a bound on the divergence between the two signs.

\begin{definitionv}{synth_18}[Observed signed-depth law]
For \(T,Q\in\mathbb N\), \(t_0,\zeta,C\in\mathbb R\), and two-point alternative index \(v\), define
\[
\mathbb P_{Q,v}^{\mathrm{obs},T}
\]
to be the observed-path law obtained by applying the observed-coordinate projection to the trajectory law of the embedded finite signed-depth model with horizon \(T\), parameters \((t_0,\zeta,C,Q)\), and alternative index \(v\).
\end{definitionv}

The law in \cref{obj:synth_18} is the distribution used in the two-point information comparison. It records the observable image of the embedded signed-depth experiment, so the divergence calculation is stated on the same data observed by an estimator.

\begin{lemmav}{lem:observed-path-kl}[Observed-path KL bound]
Fix \(t_0>0\), and write \(\alpha=\exp(-1/t_0)\). There exists a constant \(K_0>0\), depending on \(t_0\), such that whenever the following conditions hold:
\begin{itemize}
\item \textbf{(Overlap exponent.)} \(\zeta>0\), and \(L=\exp(\zeta)\).
\item \textbf{(Reset amplitude.)} \(C>1\), and \(\varepsilon_C=\min\{(C-1)/2,1/2\}\).
\item \textbf{(Sequential ignorability.)} For every horizon \(T\), terminal depth \(Q\ge 1\), and alternative \(v\in\{-1,+1\}\), the signed-depth experiment in \cref{obj:def:signed-depth-family} satisfies \cref{obj:ass:sequential-ignorability}.
\item \textbf{(Stationary start.)} For every horizon \(T\), terminal depth \(Q\ge 1\), and alternative \(v\in\{-1,+1\}\), the signed-depth experiment in \cref{obj:def:signed-depth-family} satisfies \cref{obj:ass:stationary-start}.
\end{itemize}
there exists a constant \(B_0>0\), depending on \(t_0,\zeta,C\), such that for every \(T\) and every \(Q\ge 1\), with \(\mathbb P_{Q,v}^{\mathrm{obs},T}\) denoting the observed-path law of the signed-depth alternative \(v\),
\[
\operatorname{KL}\!\left(\mathbb P_{Q,+1}^{\mathrm{obs},T}\,\middle\|\,\mathbb P_{Q,-1}^{\mathrm{obs},T}\right)
\le
B_0\,T\,\alpha^{2Q}L^{-Q}.
\]
Moreover, for each \(v\in\{-1,+1\}\), each \(t<T\), and each observed prefix \(o\in\mathcal F_{t-1}^{\mathrm{obs}}\), the conditional reward mean satisfies the exact finite-prefix identity
\[
\mathbb E_v[Y_t\mid \mathcal F_{t-1}^{\mathrm{obs}}=o]
=
v\,c_0\,\varepsilon_C\,a_t^{(Q)}(o)\,q_t^v(o),
\]
where \(c_0=(1-\alpha)/4\), and \(a_t^{(Q)}\) and \(q_t^v\) are the rare-action-history factor and terminal-depth posterior from \cref{obj:def:filter-quantities}. The posterior satisfies
\[
q_t^v(o)\le K_0\,\alpha^Q .
\]
The same observed-path laws also satisfy
\[
\operatorname{KL}\!\left(\mathbb P_{Q,+1}^{\mathrm{obs},T}\,\middle\|\,\mathbb P_{Q,-1}^{\mathrm{obs},T}\right)
\le
\frac{4c_0^2K_0^2}{1-c_0^2}\,
T\,\varepsilon_C^2\,\alpha^{2Q}L^{-Q}.
\]
\end{lemmav}

The identity in \cref{obj:lem:observed-path-kl} gives the exact observed conditional mean in terms of a rare visible action run and a terminal-depth posterior. The KL bound then combines the survival factor \(\alpha^{2Q}\) with the policy-overlap factor \(L^{-Q}\), matching the fixed-overlap exponent used in the minimax comparison. This is the point at which latent stationarity and observed histories enter the lower-bound calculation together.

The upper-bound argument also uses the stationary mean of each PHIW score. With the observed law already fixed in \cref{obj:synth_15}, the next statement bridges the observable score to the target transition operator that controls the truncation term.

\begin{lemmav}{lem:phiw-score-stationary-mean}[Stationary PHIW score mean]
Let \(T,n_X,n_H,k\in\mathbb N\), let \(t_0,\zeta,C\in\mathbb R\), and let \(M\) be a raw POMDP experiment. Suppose:
\begin{itemize}
\item \textbf{(Model class.)} \(M\in\mathcal M_T(t_0,\zeta,C)\), as in \cref{obj:def:model-class}.
\item \textbf{(Mature time.)} \(t\in\{0,\ldots,T-1\}\) and \(k\le t\).
\end{itemize}
Then the depth-\(k\) score \(Z_t(k)\) of \cref{obj:def:phiw-estimator}, evaluated under the observed trajectory law of \(M\), satisfies
\[
\int Z_t(k)\,d\mathbb P_M^{\mathrm{obs}}
=
\sum_s d_b(s)\,(P_e^k g_e)(s).
\]
Here \(d_b\) is the stationary law selected for the behavior-policy transition matrix
\[
P_b(s,s')=\sum_{a\in\{0,1\}} b(a\mid x_s)\,K(s,a)\{(r,s''):\ s''=s'\},
\]
and \(P_e^k g_e\) is the \(k\)-fold iterate of the target-policy transition operator applied to the target-policy reward regression
\[
g_e(s)=\sum_{a\in\{0,1\}} e(a\mid x_s)\int r\,K(s,a)(d r,d s').
\]
\end{lemmav}

\Cref{obj:lem:phiw-score-stationary-mean} states the exact stationary target of each mature score. Under stationarity, a length-\(k\) observable reweighting transports the behavior stationary distribution through \(k\) target-policy transitions, so the remaining discrepancy is the contraction-controlled distance between this finite-lag quantity and the stationary target value.

The bounded-reward normalization also pins down the range of the target value. This range fact is used when risks are compared over estimators whose outputs are clipped to the same unit interval.

\begin{lemmav}{lem:target-value-unit-interval}[Target value is bounded]
Let \(T\ge 1\), let \(t_0,\zeta,C\in\mathbb R\), and let
\(M=(\mathcal X,\mathcal H,K,b,e,\mathcal L(\mathsf O_T))\in
\mathcal M_T(t_0,\zeta,C)\) be a model in \cref{obj:def:model-class}. Assume the model satisfies the stationary-start, sequential-ignorability, POMDP-kernel, bounded-reward, latent-stationary-overlap, and policy-overlap conditions of
\cref{obj:ass:stationary-start,obj:ass:sequential-ignorability,obj:ass:pomdp-kernel,obj:ass:bounded-reward,obj:ass:latent-stationary-overlap,obj:ass:policy-overlap}. Then the stationary target-policy mean reward \(\theta(K,e)\) of \cref{obj:def:target-value} lies in the unit interval:
\[
\theta(K,e)\in[-1,1].
\]
\end{lemmav}

\Cref{obj:lem:target-value-unit-interval} records the bounded target range implied by the model-class reward normalization and the target stationary distribution. This range is the common scale for the clipped PHIW estimator and for the bounded observable estimators used in the minimax comparison.

The radius-sensitive upper calculation refines the estimator bound by keeping the latent-overlap distance \(q_C\) explicit.

\begin{lemmav}{lem:radius-sensitive-phiw}[Radius-sensitive PHIW risk]
For every trajectory length \(T\), observable alphabet size \(n_X\), latent alphabet size \(n_H\), and history depth \(k\), let
\[
\alpha=\exp\{-(1/t_0)\},\qquad L=\exp\{\zeta\},\qquad q_C=\frac{C-1}{C}.
\]
Suppose that
\begin{itemize}
\item \textbf{(Mixing scale.)} \(0<t_0\).
\item \textbf{(Overlap scale.)} \(0<\zeta\).
\item \textbf{(Radius parameter.)} \(1\le C\), with \(q_C\) as in \cref{obj:def:overlap-distance-radius}.
\item \textbf{(Trajectory length.)} \(1\le T\).
\item \textbf{(Model class.)} The experiment \(M\) belongs to \(\mathcal M_T(t_0,\zeta,C)\) in the sense of \cref{obj:def:model-class}.
\item \textbf{(History depth.)} \(k\le T/2\).
\end{itemize}
Then the clipped partial-history importance-weighted estimator \(\widehat\theta_k\) from \cref{obj:def:phiw-estimator}, evaluated under the observed trajectory law induced by \(M\), satisfies
\[
\mathbb E_M\!\left[\{\widehat\theta_k-\theta(K,e)\}^2\right]
\le
4q_C^2\alpha^{2k}
+\frac{2}{T}\left\{
L^{k+1}\left(1+\frac{4}{L-1}\right)
+\frac{4}{1-\alpha}
\right\},
\]
where \(\theta(K,e)\) is the stationary target-policy mean reward from \cref{obj:def:target-value}.
\end{lemmav}

In \cref{obj:lem:radius-sensitive-phiw}, the first term is the squared truncation contribution after \(k\) observable lags, scaled by the radius \(q_C\). The second term is the sampling contribution from estimating with length-\(k\) products of observable policy ratios. Choosing the radius-calibrated depth in \cref{obj:def:radius-adaptive-history-depth} balances these terms in the shrinking-overlap results.

The matching radius-explicit KL statement records the lower-bound calculation with the same radius normalization.

\begin{lemmav}{lem:radius-explicit-observed-kl}[Observed KL radius bound]
Let \(t_0>0\).  Define
\[
\alpha(t_0)=\exp(-1/t_0),\qquad
c_0(t_0)=\frac{1-\alpha(t_0)}{4}.
\]
Then there exists a constant \(K_0=K_0(t_0)>0\) such that, for every \(\zeta>0\), \(C>1\), \(T\in\mathbb N\), and \(Q\in\mathbb N\) with \(Q\ge1\), if
\[
\varepsilon_C=\min\left\{\frac{C-1}{2},\frac12\right\},\qquad
L(\zeta)=\exp(\zeta),
\]
and if \(q_C\) is the radius in \cref{obj:def:overlap-distance-radius}, then
\[
\frac{q_C}{2}\le \varepsilon_C \le q_C
\]
and the observed-path laws \(\mathbb P_{Q,+1}^{\mathrm{obs},T}\) and
\(\mathbb P_{Q,-1}^{\mathrm{obs},T}\) of the signed-depth alternatives satisfy
\[
\operatorname{KL}\!\left(
\mathbb P_{Q,+1}^{\mathrm{obs},T}\,\middle\|\,
\mathbb P_{Q,-1}^{\mathrm{obs},T}
\right)
\le
\frac{4c_0(t_0)^2K_0^2}{1-c_0(t_0)^2}\,
T\,\varepsilon_C^2\,\alpha(t_0)^{2Q}\,L(\zeta)^{-Q}.
\]
\end{lemmav}

The inequalities in \cref{obj:lem:radius-explicit-observed-kl} connect the construction amplitude \(\varepsilon_C\) to the normalized radius from \cref{obj:def:overlap-distance-radius}. Consequently, the lower calculation can be stated on the same radius scale as the estimator bound in \cref{obj:lem:radius-sensitive-phiw}.

The final group records the parametric two-experiment comparison used to anchor the risk at the usual \(T^{-1}\) scale across all radii in the model class. The first two statements compute the separation in target value and the observed KL budget for the singleton-state pair; the following bounded-risk and information inequalities convert those calculations into risk floors.

\begin{lemmav}{lem:parametric-pair-target-value}[Parametric pair target value]
Let \(T\in\mathbb N\), let \(v\in\{0,1\}\), and define the sign
\[
\sigma(v)=
\begin{cases}
+1, & v=1,\\
-1, & v=0.
\end{cases}
\]
Put
\[
\eta_T=\frac{1}{4\sqrt T},
\]
with value \(0\) when \(T=0\). Suppose the scalar parameters satisfy:
\begin{itemize}
\item \textbf{(Initial scale.)} \(t_0>0\).
\item \textbf{(Overlap scale.)} \(\zeta>0\).
\item \textbf{(Radius scale.)} \(C\ge 1\).
\end{itemize}
Let \(M_T^v\) be the singleton-state finite experiment with one observed state and one hidden state, fair behavior and target policies \(b(a\mid x_\star)=e(a\mid x_\star)=1/2\), reward support \(\{-1,+1\}\), and reward-transition kernel that assigns, from every state-action pair, probability
\[
\frac{1+\sigma(v)y\eta_T}{2}
\]
to reward \(y\in\{-1,+1\}\) and next state \((x_\star,h_\star)\). Then the stationary target-policy mean reward of \cref{obj:def:target-value}, evaluated in this experiment, satisfies
\[
\theta(K_{M_T^v},e_{M_T^v})=\sigma(v)\eta_T .
\]
\end{lemmav}

\begin{lemmav}{lem:parametric-pair-observed-kl}[Observed KL budget]
Let \(T\in\mathbb N\) satisfy \(T\ge 1\), and set \(\eta_T=1/(4\sqrt T)\). Let \(M_T^{-}\) and \(M_T^{+}\) be the two finite singleton-state experiments with one observed state \(x_\star\), one hidden state, initial mass at the single joint state, fair behavior and target policies \(b=e\) on \(\{0,1\}\), and reward code \(\mathsf r\in\{0,1\}\) decoded as \(2\mathsf r-1\). Under \(M_T^\sigma\), for \(\sigma\in\{-1,+1\}\), the reward-transition kernel returns to the single joint state and assigns reward \(y\in\{-1,+1\}\) probability
\[
\frac{1+\sigma y\eta_T}{2}.
\]
Writing \(\mathbb P^{\mathrm{obs}}_{M_T^\sigma}\) for the image of the decoded trajectory law under the observed-path projection, the required KL budget holds in the direction
\[
\operatorname{KL}\!\left(
\mathbb P^{\mathrm{obs}}_{M_T^-}
\,\middle\|\,
\mathbb P^{\mathrm{obs}}_{M_T^+}
\right)
\le \frac14 .
\]
\end{lemmav}

Before applying the two-point inequality, the next statement records a uniform boundedness fact for clipped observable-data estimators. It uses the target-value range from \cref{obj:lem:target-value-unit-interval} and the estimator range built into \cref{obj:def:minimax-risk}.

\begin{lemmav}{lem:observed-risk-bounded-by-four}[Observed risk bound]
Let \(T\ge 1\) and let \(t_0,\zeta,C\in\mathbb R\). Let \(\widehat\theta\) be an observable-data estimator of the admissible class used in \cref{obj:def:minimax-risk}: for each observed-alphabet size and policy pair, \(\widehat\theta\) is a measurable real-valued function of the observed path taking values in \([-1,1]\). For every
\[
M=(\mathcal X,\mathcal H,K,b,e,\mathcal L(\mathsf O_T))\in \mathcal M_T(t_0,\zeta,C)
\]
as in \cref{obj:def:model-class}, with \(\widehat\theta\) evaluated at the observed-alphabet size and policies of \(M\), the squared risk for the target value \(\theta(K,e)\) of \cref{obj:def:target-value} satisfies
\[
E_M\bigl[(\widehat\theta-\theta(K,e))^2\bigr]\le 4 .
\]
\end{lemmav}

\Cref{obj:lem:observed-risk-bounded-by-four} places all admissible estimator losses on a common bounded scale. This supplies the integrability and range control used when the two-point comparison is embedded into the minimax risk.

\begin{lemmav}{lem:two-point-risk-floor}[Two-point risk floor]
Let \(T\ge 1\), \(t_0,\zeta,C\in\mathbb R\), and let \(M_0,M_1\in\mathcal M_T(t_0,\zeta,C)\) as in \cref{obj:def:model-class}.  After identifying their common observed alphabet, write \(\mathbb P_j\) for the observed-data law of \(M_j\) and \(\theta_j\) for the target value of \(M_j\) from \cref{obj:def:target-value}, \(j\in\{0,1\}\).  Let \(s\) be a nonnegative separation radius and let \(\chi\) be a nonnegative KL upper bound.  Assume:
\begin{itemize}
\item \textbf{(Common observable structure.)} The two models have the same observed-alphabet cardinality, the same behavior policy, and the same target policy.
\item \textbf{(Information bound.)} \(\mathbb P_0\ll\mathbb P_1\),
\[
\operatorname{KL}(\mathbb P_0\|\mathbb P_1)\le \chi,
\qquad
\operatorname{KL}(\mathbb P_0\|\mathbb P_1)<\infty .
\]
\item \textbf{(Target separation.)}
\[
2s\le |\theta_0-\theta_1| .
\]
\end{itemize}
Then every raw observable-data estimator \(\widehat\theta\), viewed as a real-valued measurable function of the observed-alphabet size, the policy pair, and the observed path, with \((\widehat\theta-\theta_j)^2\) integrable under \(\mathbb P_j\) at \(M_j\) for \(j=0,1\), satisfies
\[
\frac{s^2\exp(-\chi)}{4}
\le
\max\left\{
\mathbb E_{M_0}\!\left[(\widehat\theta-\theta_0)^2\right],
\mathbb E_{M_1}\!\left[(\widehat\theta-\theta_1)^2\right]
\right\}.
\]
Consequently, the minimax squared-error risk of \cref{obj:def:minimax-risk} satisfies
\[
\frac{s^2\exp(-\chi)}{4}
\le
R_T(t_0,\zeta,C).
\]
\end{lemmav}

\Cref{obj:lem:parametric-pair-target-value} sets the value separation for the singleton-state Rademacher pair, and \cref{obj:lem:parametric-pair-observed-kl} gives the corresponding observed KL budget. The bounded-risk statement in \cref{obj:lem:observed-risk-bounded-by-four} fixes the loss scale for admissible clipped estimators, and the generic reduction in \cref{obj:lem:two-point-risk-floor} then translates a common-policy, low-divergence pair with target separation into a lower bound for every observable-data estimator.

\begin{lemmav}{lem:uniform-parametric-floor}[Uniform parametric floor]
Fix a positive mixing scale \(t_0>0\) and a positive log-overlap scale \(\zeta>0\). For every horizon \(T\in\mathbb N\) with \(T\ge 1\) and every radius \(C\ge 1\), let \(M_T^{-}\) and \(M_T^{+}\) be the two raw POMDP experiments generated by the singleton-state, singleton-observation Rademacher construction with rewards in \(\{-1,+1\}\), fair behavior and target policies \(b=e\), and the two opposite signs of the construction's parametric reward tilt. Then:
\begin{itemize}
\item \textbf{(Class membership.)} Both \(M_T^{-}\) and \(M_T^{+}\) belong to the latent-overlap model class \(\mathcal M_T(t_0,\zeta,C)\) of \cref{obj:def:model-class}.
\item \textbf{(Common policies.)} Each experiment has equal behavior and target policies, and the two experiments share these policies:
\[
b_{M_T^{-}}=e_{M_T^{-}},
\qquad
b_{M_T^{+}}=e_{M_T^{+}},
\qquad
b_{M_T^{-}}=b_{M_T^{+}},
\qquad
e_{M_T^{-}}=e_{M_T^{+}}.
\]
\item \textbf{(Uniform minimax floor.)} The minimax squared-error risk \(R_T(t_0,\zeta,C)\) of \cref{obj:def:minimax-risk} satisfies
\[
\frac{\exp(-1/4)}{64T}\le R_T(t_0,\zeta,C).
\]
\item \textbf{(Pointwise two-experiment floor.)} For every observable-data estimator \(\widehat\theta\), viewed as a measurable function of the revealed alphabet and the policies, under the usual squared-error integrability conditions for the observed trajectory laws of \(M_T^{-}\) and \(M_T^{+}\),
\[
\frac{\exp(-1/4)}{64T}
\le
\max\left\{
\mathbb E_{M_T^{-}}^{\mathrm{obs}}
\left[
\left(\widehat\theta(1,b_{M_T^{-}},e_{M_T^{-}};\mathsf O_T)-\theta(K_{M_T^{-}},e_{M_T^{-}})\right)^2
\right],
\mathbb E_{M_T^{+}}^{\mathrm{obs}}
\left[
\left(\widehat\theta(1,b_{M_T^{+}},e_{M_T^{+}};\mathsf O_T)-\theta(K_{M_T^{+}},e_{M_T^{+}})\right)^2
\right]
\right\},
\]
where \(\theta(K,e)\) is the stationary target-policy mean reward of \cref{obj:def:target-value}.
\end{itemize}
\end{lemmav}

\Cref{obj:lem:uniform-parametric-floor} complements the signed-depth lower construction by giving a radius-uniform benchmark at the parametric scale. The experiments share behavior and target policies, so the displayed floor reflects value uncertainty in the observed rewards themselves and applies throughout the latent-overlap model class.
