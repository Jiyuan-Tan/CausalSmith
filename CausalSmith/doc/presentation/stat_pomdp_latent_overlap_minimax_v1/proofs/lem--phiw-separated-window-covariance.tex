\begin{proof}[Proof of \cref{obj:lem:phiw-separated-window-covariance}]
Put
\[
\gamma:=h-k-1 .
\]
Then \(0\le \gamma\) by \(k+1\le h\), and
\[
u=t+h=t+1+\gamma+k .
\]
Let \(\mathbb P_M\) denote the full trajectory law and \(\mathbb P_M^{\mathrm{obs}}\) its pushforward under the observation map \(O\). For \(p\in\{b,e\}\), write
\[
P_p(s,s')=\sum_{a\in\mathcal A}p(a\mid x(s))K(\{S'=s'\}\mid s,a),
\qquad
g_e(s)=\sum_{a\in\mathcal A}e(a\mid x(s))\int y\,K(dy\times\mathcal S\mid s,a),
\]
and define
\[
\Phi_{\gamma,k}(s):=P_b^\gamma P_e^k g_e(s).
\]

\begin{enumerate}
\item Since \(L=\exp(\zeta)\) and \(\zeta>0\) in \cref{obj:def:model-class}, \(1\le L\). The maturity and separation hypotheses give \(t+1+\gamma+k<T\), so \cref{obj:lem:phiw-separated-window-peeling} applies and yields
\[
\int Z_t(k)(w)Z_u(k)(w)\,d\mathbb P_M^{\mathrm{obs}}(w)
=
\int Z_t(k)(O(\tau))\,\Phi_{\gamma,k}(S_{t+1}(\tau))\,d\mathbb P_M(\tau).
\]
For the companion mean identity, let \(r_{+}:=t+1\) and \(q_{\gamma}:=r_{+}+\gamma\), so \(u=q_{\gamma}+k\). Since \(u=q_{\gamma}+k\), the depth-\(k\) window in \(Z_u(k)\) is exactly \(q_{\gamma},q_{\gamma}+1,\ldots,u\). We claim
\[
\int Z_u(k)(w)\,d\mathbb P_M^{\mathrm{obs}}(w)
=
\int \Phi_{\gamma,k}(S_{t+1}(\tau))\,d\mathbb P_M(\tau).
\]
To prove this identity, condition first on \(S_{q_{\gamma}}\). Write
\[
\rho(a,x):=
\begin{cases}
 e(a\mid x)/b(a\mid x),& b(a\mid x)>0,\\
 0,& b(a\mid x)=0.
\end{cases}
\]
The zero-cell convention and \cref{obj:ass:policy-overlap} imply \(e(a\mid x)=0\) whenever \(b(a\mid x)=0\). Hence, for each pre-terminal epoch \(m_{\mathrm{win}}\in\{q_{\gamma},\ldots,u-1\}\),
\[
\sum_{a\in\mathcal A} b(a\mid x(s))\rho(a,x(s))
        K(\{S'=s'\}\mid s,a)
=
\sum_{a\in\mathcal A} e(a\mid x(s))K(\{S'=s'\}\mid s,a)
=P_e(s,s'),
\]
while at the terminal epoch
\[
\sum_{a\in\mathcal A} b(a\mid x(s))\rho(a,x(s))
        \int y\,K(dy\times\mathcal S\mid s,a)
=
\sum_{a\in\mathcal A} e(a\mid x(s))
        \int y\,K(dy\times\mathcal S\mid s,a)
=g_e(s).
\]
Thus the target-weighted terminal window has conditional mean \(P_e^k g_e(S_{q_{\gamma}})\), with the terminal display alone applying when \(k=0\). Conditioning backward over the gap epochs \(t+1,\ldots,q_{\gamma}-1\), no score ratio appears, and \cref{obj:ass:sequential-ignorability,obj:ass:pomdp-kernel} give the behavior propagation \(P_b^\gamma\); when \(\gamma=0\), this is the identity operator. Bounded rewards and the finite state and action spaces justify the displayed integrals. Therefore the conditional mean given \(S_{t+1}\) is \(P_b^\gamma P_e^k g_e(S_{t+1})=\Phi_{\gamma,k}(S_{t+1})\), proving the claim. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.integral_phiwScore_eq_futureFunction
Also, by definition of \(\mathbb P_M^{\mathrm{obs}}\),
\[
\int Z_t(k)(w)\,d\mathbb P_M^{\mathrm{obs}}(w)
=
\int Z_t(k)(O(\tau))\,d\mathbb P_M(\tau).
\]
Thus the covariance equals the centered full-law product
\[
\operatorname{Cov}_{\mathrm{obs}}(Z_t(k),Z_u(k))
=
\int Z_t(k)(O(\tau))\,\Phi_{\gamma,k}(S_{t+1}(\tau))\,d\mathbb P_M(\tau)
-
\Bigl(\int Z_t(k)(O(\tau))\,d\mathbb P_M(\tau)\Bigr)
\Bigl(\int \Phi_{\gamma,k}(S_{t+1}(\tau))\,d\mathbb P_M(\tau)\Bigr).
\]
The covariance identity is legitimate because
\[
Z_t(k),Z_u(k)\in L^2(\mathbb P_M^{\mathrm{obs}}).
\]
Indeed, write the window product in \(Z_m(k)\) as
\[
R_m^{(k)}(w):=\prod_{\substack{j\in\{0,\ldots,T-1\}:\, j\le m,\ m-j\le k}}
\rho_j(w),
\qquad m\in\{t,u\}.
\]
The product contains at most \(k+1\) factors. The zero-cell convention and \cref{obj:ass:policy-overlap} make each factor nonnegative and at most \(L\) on observed behavior-support cells, while \cref{obj:ass:bounded-reward} gives \(|Y_m|\le 1\) almost surely. Hence \(|Z_m(k)|\le L^{k+1}\) almost surely under \(\mathbb P_M^{\mathrm{obs}}\), and the two scores are square-integrable. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.integral_phiwScore_mul_eq_futureFunction; CausalSmith.Stat.PomdpLatentOverlapMinimax.integral_phiwScore_eq_futureFunction

\item The first score has unit \(L^1\) norm:
\[
\int |Z_t(k)(O(\tau))|\,d\mathbb P_M(\tau)\le 1 .
\]
Indeed, with
\[
R_t^{(k)}(w):=\prod_{j=t-k}^{t}\rho_j(w),
\]
the zero-cell convention and \cref{obj:ass:policy-overlap} give
\[
\mathbb E_{\mathrm{obs}}\!\left[R_t^{(k)}\right]=1
\]
by peeling the ratios from \(j=t\) down to \(j=t-k\): conditionally at each peeled epoch,
\[
\sum_{a\in\mathcal A} b(a\mid X_j)\rho_j(a,X_j)
=
\sum_{a:\,b(a\mid X_j)>0} e(a\mid X_j)
=
\sum_{a\in\mathcal A} e(a\mid X_j)
=1 .
\]
By \cref{obj:ass:bounded-reward}, \(|Z_t(k)|\le R_t^{(k)}\) almost surely, which proves the displayed \(L^1\) bound after passing between the observed law and the full law by the observation map. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.integral_abs_phiwScore_le_one

\item Let \(d_b\) be the stationary law of \(P_b\), and define the support-truncated regression on all of \(\mathcal S\) by
\[
\widetilde g_e(s):=
\begin{cases}
g_e(s),& d_b(s)>0,\\
0,& d_b(s)=0.
\end{cases}
\]
For every \(s\) with \(d_b(s)>0\),
\[
|g_e(s)|\le 1 .
\]
To see this, fix such an \(s\). For an action \(a\) with \(b(a\mid x(s))>0\), consider epoch \(0\). The history before epoch \(0\) is the unique empty history, so the full history-action singleton with current state \(s\) and action \(a\) has probability
\[
d_b(s)b(a\mid x(s))>0,
\]
by \cref{obj:ass:stationary-start} and the epoch-zero action factorization in \cref{obj:ass:sequential-ignorability}. Under the history-next-pair factorization supplied by \cref{obj:ass:pomdp-kernel}, \cref{obj:ass:bounded-reward} gives, for almost every history-action pair, a kernel row whose reward coordinate is almost surely in \([-1,1]\). Since this epoch-zero singleton has positive mass, that almost-sure row statement holds at \((s,a)\), and hence
\[
\left|\int y\,K(dy\times\mathcal S\mid s,a)\right|\le 1 .
\]
If \(e(a\mid x(s))>0\), then \cref{obj:ass:policy-overlap} forces \(b(a\mid x(s))>0\). Therefore
\[
|g_e(s)|
\le
\sum_{a\in\mathcal A} e(a\mid x(s))
\left|\int y\,K(dy\times\mathcal S\mid s,a)\right|
\le
\sum_{a\in\mathcal A} e(a\mid x(s))
=1.
\]
For \(d_b(s)=0\), the definition gives \(|\widetilde g_e(s)|=0\). Consequently
\[
\sup_{s,s'}|\widetilde g_e(s)-\widetilde g_e(s')|\le 2 .
\] % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.rewardRegression_abs_le_one_of_behaviorSupport

\item Since \(t_0>0\), \(\alpha=\exp(-1/t_0)\) satisfies \(0\le \alpha\le 1\). For a function \(f:\mathcal S\to\mathbb R\), write
\[
\operatorname{osc}(f):=\sup_{s,s'\in\mathcal S}|f(s)-f(s')|.
\]
For each \(p_\star\in\{b,e\}\), the rows of \(P_{p_\star}\) are probability vectors. Nonnegativity follows from the nonnegativity of the policy probabilities and of the kernel rows. For the row sums,
\[
\sum_{s'\in\mathcal S}P_{p_\star}(s,s')
=
\sum_{a\in\mathcal A}p_\star(a\mid x(s))
   \sum_{s'\in\mathcal S}K(\{S'=s'\}\mid s,a)
=
\sum_{a\in\mathcal A}p_\star(a\mid x(s))
=1.
\]
Here \cref{obj:ass:pomdp-kernel} supplies the kernel row mass, \cref{obj:ass:sequential-ignorability} supplies the behavior probability vector when \(p_\star=b\), and the target-carrier part of \cref{obj:ass:policy-overlap} supplies the target probability vector when \(p_\star=e\).

We first convert the total-variation contraction in \cref{obj:ass:uniform-contraction} into the oscillation contraction used below. Let \(\nu\) and \(\nu'\) be probability vectors on the finite state space, and let \(f:\mathcal S\to\mathbb R\). Put
\[
f_{\min}:=\min_{x\in\mathcal S} f(x),
\qquad
f_0(x):=f(x)-f_{\min}.
\]
Then \(0\le f_0(x)\le \operatorname{osc}(f)\) for every \(x\), and, since \(\nu\) and \(\nu'\) have equal total mass,
\[
\sum_{x\in\mathcal S} f(x)(\nu(x)-\nu'(x))
=
\sum_{x\in\mathcal S} f_0(x)(\nu(x)-\nu'(x)).
\]
The layer-cake identity on the finite state space gives
\[
\sum_{x\in\mathcal S} f_0(x)(\nu(x)-\nu'(x))
=
\int_{0}^{\operatorname{osc}(f)}
\bigl(\nu\{x:f_0(x)>\lambda_{\mathrm{lev}}\}
      -\nu'\{x:f_0(x)>\lambda_{\mathrm{lev}}\}\bigr)
\,d\lambda_{\mathrm{lev}},
\]
and therefore
\[
\left|
\sum_{x\in\mathcal S} f(x)(\nu(x)-\nu'(x))
\right|
\le
\operatorname{osc}(f)\,\|\nu-\nu'\|_{\mathrm{TV}}.
\]
Now fix states \(s,s'\). Applying \cref{obj:ass:uniform-contraction} to the point masses \(\delta_s\) and \(\delta_{s'}\) gives
\[
\|\delta_s P_{p_\star}-\delta_{s'}P_{p_\star}\|_{\mathrm{TV}}
\le
\alpha\,\|\delta_s-\delta_{s'}\|_{\mathrm{TV}}
\le \alpha .
\]
Combining this with the preceding dual bound yields the one-step estimate
\[
|P_{p_\star}f(s)-P_{p_\star}f(s')|
\le
\alpha\,\operatorname{osc}(f).
\]
Taking the supremum over \(s,s'\) and iterating gives the finite-state oscillation form: whenever \(\operatorname{osc}(f)\le B_{\mathrm{osc}}\) for some \(B_{\mathrm{osc}}\ge0\),
\[
\operatorname{osc}(P_{p_\star}^{m_{\mathrm{iter}}}f)
\le
\alpha^{m_{\mathrm{iter}}}B_{\mathrm{osc}},
\qquad m_{\mathrm{iter}}\in\mathbb N.
\]
Applying this first with \(p_\star=e\), \(f=\widetilde g_e\), and \(B_{\mathrm{osc}}=2\), and then using \(\alpha^k\le1\), gives
\[
\sup_{s,s'}|P_e^k\widetilde g_e(s)-P_e^k\widetilde g_e(s')|\le 2.
\]
Applying it next with \(p_\star=b\) to \(P_e^k\widetilde g_e\) gives
\[
\sup_{s,s'}
\left|
P_b^\gamma P_e^k\widetilde g_e(s)
-
P_b^\gamma P_e^k\widetilde g_e(s')
\right|
\le
2\alpha^\gamma .
\] % lean: Causalean.Mathlib.Probability.FiniteMarkovOscillation.oscillationBound_markovOperatorIter

\item Define
\[
\widetilde\Phi_{\gamma,k}(s):=P_b^\gamma P_e^k\widetilde g_e(s).
\]
Because \(g_e=\widetilde g_e\) on \(\{s:d_b(s)>0\}\), \cref{obj:lem:behavior-support-operator-congruence} applied with policy \(e\) gives, for every behavior-supported state \(x\),
\[
P_e^k g_e(x)=P_e^k\widetilde g_e(x)
\qquad\text{whenever }d_b(x)>0.
\]
This pointwise behavior-supported equality is the terminal-agreement hypothesis needed for the second application of \cref{obj:lem:behavior-support-operator-congruence}, now with policy \(b\). Thus, for every initial state \(s\) with \(d_b(s)>0\),
\[
\Phi_{\gamma,k}(s)=P_b^\gamma P_e^k g_e(s)
=P_b^\gamma P_e^k\widetilde g_e(s)
=\widetilde\Phi_{\gamma,k}(s).
\]
We next record the behavior-stationarity identity used at time \(t+1\). For every bounded \(F:\mathcal S\to\mathbb R\) and every epoch \(m_{\mathrm{time}}\),
\[
\int F(S_{m_{\mathrm{time}}}(\tau))\,d\mathbb P_M(\tau)
=
\sum_{s\in\mathcal S}d_b(s)F(s).
\]
This follows by induction on \(m_{\mathrm{time}}\). At the initial epoch the identity is \cref{obj:ass:stationary-start}. For the induction step, \cref{obj:ass:sequential-ignorability} and \cref{obj:ass:pomdp-kernel} give the behavior peeling identity
\[
\int F(S_{m_{\mathrm{time}}+1}(\tau))\,d\mathbb P_M(\tau)
=
\int \sum_{x\in\mathcal S}P_b(S_{m_{\mathrm{time}}}(\tau),x)F(x)\,d\mathbb P_M(\tau).
\]
Using the induction hypothesis with the bounded function \(s\mapsto \sum_x P_b(s,x)F(x)\), the right-hand side becomes
\[
\sum_{s\in\mathcal S}d_b(s)\sum_{x\in\mathcal S}P_b(s,x)F(x)
=
\sum_{x\in\mathcal S}\left(\sum_{s\in\mathcal S}d_b(s)P_b(s,x)\right)F(x)
=
\sum_{x\in\mathcal S}d_b(x)F(x),
\]
where the last equality is the stationarity equation for the behavior stationary law \(d_b\), part of the model-class data in \cref{obj:def:model-class}. Taking the zero-support indicator
\[
\chi_{\mathrm{null}}(s):=\mathbf 1\{d_b(s)=0\}
\]
at \(m_{\mathrm{time}}=t+1\) gives
\[
\int \mathbf 1\{d_b(S_{t+1}(\tau))=0\}\,d\mathbb P_M(\tau)=0,
\]
so \(d_b(S_{t+1}(\tau))>0\) for \(\mathbb P_M\)-almost every \(\tau\). Hence
\[
\Phi_{\gamma,k}(S_{t+1}(\tau))
=
\widetilde\Phi_{\gamma,k}(S_{t+1}(\tau))
\quad\text{for }\mathbb P_M\text{-almost every }\tau .
\] % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.integral_curState_eq_behaviorStationary

\item Put
\[
A(\tau):=Z_t(k)(O(\tau)),
\qquad
\overline\Phi:=
\int \widetilde\Phi_{\gamma,k}(S_{t+1}(\tau))\,d\mathbb P_M(\tau).
\]
Using the almost-sure replacement from the previous step, both
\[
\int A(\tau)\Phi_{\gamma,k}(S_{t+1}(\tau))\,d\mathbb P_M(\tau)
=
\int A(\tau)\widetilde\Phi_{\gamma,k}(S_{t+1}(\tau))\,d\mathbb P_M(\tau)
\]
and
\[
\int \Phi_{\gamma,k}(S_{t+1}(\tau))\,d\mathbb P_M(\tau)
=
\int \widetilde\Phi_{\gamma,k}(S_{t+1}(\tau))\,d\mathbb P_M(\tau)
=
\overline\Phi
\]
hold; the integrability needed for these replacements follows from the finite state space and the score bound above. The centered representation from the first step therefore gives
\[
\left|\operatorname{Cov}_{\mathrm{obs}}(Z_t(k),Z_u(k))\right|
=
\left|
\int A(\tau)\bigl(\widetilde\Phi_{\gamma,k}(S_{t+1}(\tau))-\overline\Phi\bigr)
\,d\mathbb P_M(\tau)
\right|.
\]
The oscillation bound in the fourth step implies, for every \(\tau\),
\[
\left|\widetilde\Phi_{\gamma,k}(S_{t+1}(\tau))-\overline\Phi\right|
\le 2\alpha^\gamma,
\]
because \(\overline\Phi\) is an average of values of \(\widetilde\Phi_{\gamma,k}\). Therefore, using the unit \(L^1\) bound from the second step,
\[
\left|\operatorname{Cov}_{\mathrm{obs}}(Z_t(k),Z_u(k))\right|
\le
\int |A(\tau)|\,2\alpha^\gamma\,d\mathbb P_M(\tau)
\le
2\alpha^\gamma
=
2\alpha^{h-k-1}.
\] % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.abs_integral_mul_sub_mul_integral_le_oscillation
\end{enumerate}
\end{proof}
