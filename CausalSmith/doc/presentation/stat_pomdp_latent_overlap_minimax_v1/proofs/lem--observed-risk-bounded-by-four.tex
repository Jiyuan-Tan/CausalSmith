\begin{proof}[Proof of \cref{obj:lem:observed-risk-bounded-by-four}]
\begin{enumerate}
\item Fix
\[
M=(\mathcal X,\mathcal H,K,b,e,\mathcal L(\mathsf O_T))\in
\mathcal M_T(t_0,\zeta,C),
\]
and write an observed path as
\[
\omega_{\mathrm{obs}}=\bigl((X_t,A_t,Y_t)\bigr)_{t=0}^{T-1}\in \mathsf O_T .
\]
For this path set
\[
r_{\mathrm{err}}(\omega_{\mathrm{obs}})
:=
\widehat\theta(n_X,b,e,\omega_{\mathrm{obs}})-\theta(K,e).
\]
The admissibility condition in \cref{obj:def:minimax-risk} gives
\[
-1\le \widehat\theta(n_X,b,e,\omega_{\mathrm{obs}})\le 1,
\]
and \cref{obj:lem:target-value-unit-interval} gives
\[
-1\le \theta(K,e)\le 1 .
\]
Hence
\[
-2\le r_{\mathrm{err}}(\omega_{\mathrm{obs}})\le 2 .
\]
Equivalently,
\[
0\le r_{\mathrm{err}}(\omega_{\mathrm{obs}})+2,
\qquad
0\le 2-r_{\mathrm{err}}(\omega_{\mathrm{obs}}).
\]
Multiplying these two nonnegative quantities yields
\[
0\le
\bigl(r_{\mathrm{err}}(\omega_{\mathrm{obs}})+2\bigr)
\bigl(2-r_{\mathrm{err}}(\omega_{\mathrm{obs}})\bigr)
=
4-r_{\mathrm{err}}(\omega_{\mathrm{obs}})^2,
\]
so
\[
\bigl(\widehat\theta(n_X,b,e,\omega_{\mathrm{obs}})-\theta(K,e)\bigr)^2
\le 4
\qquad
\text{for every }\omega_{\mathrm{obs}}\in\mathsf O_T .
\] % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.observedRisk_le_four

\item The function
\[
\omega_{\mathrm{obs}}
\longmapsto
\bigl(\widehat\theta(n_X,b,e,\omega_{\mathrm{obs}})-\theta(K,e)\bigr)^2
\]
is measurable because \(\widehat\theta(n_X,b,e,\cdot)\) is measurable by admissibility and the target value is constant in the observed path. The observed-path law \(\mathcal L_M^{\mathrm{obs}}\) is a probability law, and the pointwise bound from the previous step gives integrability together with
\[
\int
\bigl(\widehat\theta(n_X,b,e,\omega_{\mathrm{obs}})-\theta(K,e)\bigr)^2
\,d\mathcal L_M^{\mathrm{obs}}(\omega_{\mathrm{obs}})
\le
\int 4\,d\mathcal L_M^{\mathrm{obs}}(\omega_{\mathrm{obs}})
=
4 .
\]
By the squared-risk definition in \cref{obj:def:minimax-risk}, the left-hand side is exactly
\[
E_M\bigl[(\widehat\theta-\theta(K,e))^2\bigr].
\]
Therefore
\[
E_M\bigl[(\widehat\theta-\theta(K,e))^2\bigr]\le 4 .
\] % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.observedRisk_le_four
\end{enumerate}
\end{proof}
