\begin{proof}[Proof of \cref{obj:lem:insulin-bias-interval-evaluation}]
All arithmetic below is exact rational interval arithmetic with the operations described in \cref{obj:lem:certified-stationary-reward-enclosure}.
\begin{enumerate}
\item The residual sums over the \(360\) states of \cref{obj:def:insulin-grid} evaluate to
\[
\delta_{\mathrm b}
=
\sum_{s\in\mathsf S_{\mathrm{grid}}}
\bigl|\mathfrak T_{\mathrm b,0}(s)\ominus
\mathfrak T_{\mathrm b,1}(s)\bigr|_{\max}
=
\frac{2596399843}{200000000000000},
\]
and
\[
\delta_{\mathrm e}
=
\sum_{s\in\mathsf S_{\mathrm{grid}}}
\bigl|\mathfrak T_{\mathrm e,0}(s)\ominus
\mathfrak T_{\mathrm e,1}(s)\bigr|_{\max}
=
\frac{16950307061}{1000000000000000}.
\]
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.insulinGeneratedResidual

\item Since \(\mathfrak r_{\mathrm{ins}}(s)=[r_{\mathrm{ins}}(s),r_{\mathrm{ins}}(s)]\), the interval expectations computed from the terminal rows are
\[
\mathfrak E_{\mathrm b}
=
\sum_{s\in\mathsf S_{\mathrm{grid}}}
\mathfrak T_{\mathrm b,0}(s)\cdot\mathfrak r_{\mathrm{ins}}(s)
=
\left[
-\frac{157888864566941}{300000000000000},
-\frac{63155545824091}{120000000000000}
\right],
\]
and
\[
\mathfrak E_{\mathrm e}
=
\sum_{s\in\mathsf S_{\mathrm{grid}}}
\mathfrak T_{\mathrm e,0}(s)\cdot\mathfrak r_{\mathrm{ins}}(s)
=
\left[
-\frac{312236715379949}{600000000000000},
-\frac{312236715366659}{600000000000000}
\right].
\]
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.insulinRewardIntervals

\item Because \(1-1/2=1/2\), the expansion radius in each case is \(2\delta_\star\). Hence
\[
\mathfrak J_{\mathrm b}
=
\left[
-\frac{157888864566941}{300000000000000}
-2\frac{2596399843}{200000000000000},
-\frac{63155545824091}{120000000000000}
+2\frac{2596399843}{200000000000000}
\right]
=
\left[
-\frac{15789665376647}{30000000000000},
-\frac{315762150721397}{600000000000000}
\right],
\]
and
\[
\mathfrak J_{\mathrm e}
=
\left[
-\frac{312236715379949}{600000000000000}
-2\frac{16950307061}{1000000000000000},
-\frac{312236715366659}{600000000000000}
+2\frac{16950307061}{1000000000000000}
\right]
=
\left[
-\frac{1561285278742111}{3000000000000000},
-\frac{1561081874990929}{3000000000000000}
\right].
\]
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.insulinChunkedStationaryRewardInterval

\item Taking the interval difference,
\[
\mathfrak B
=
\mathfrak J_{\mathrm b}\ominus\mathfrak J_{\mathrm e}
=
\left[
\ell(\mathfrak J_{\mathrm b})-u(\mathfrak J_{\mathrm e}),
u(\mathfrak J_{\mathrm b})-\ell(\mathfrak J_{\mathrm e})
\right],
\]
so
\[
\mathfrak B
=
\left[
-\frac{17884662673771}{3000000000000000},
-\frac{2920912477479}{500000000000000}
\right].
\]
The endpoint comparisons are then the rational inequalities
\[
-\frac{6}{1000}
=
-\frac{18000000000000}{3000000000000000}
<
-\frac{17884662673771}{3000000000000000}
=
\ell(\mathfrak B),
\]
and
\[
u(\mathfrak B)
=
-\frac{2920912477479}{500000000000000}
<
-\frac{2900000000000}{500000000000000}
=
-\frac{58}{10000}.
\]
This proves both asserted endpoint bounds.
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.insulinBiasInterval_bounds
\end{enumerate}
\end{proof}
