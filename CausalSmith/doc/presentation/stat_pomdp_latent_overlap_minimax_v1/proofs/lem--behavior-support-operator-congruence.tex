\begin{proof}[Proof of \cref{obj:lem:behavior-support-operator-congruence}]
Fix a policy \(p\in\{b,e\}\).  The argument first records the support-closure fact used at the induction step.  For joint states \(\sigma,\tau\in\mathcal S\) and an action \(a\in\{0,1\}\), write
\[
\kappa_{\sigma,\tau}(a)
:=
K(\{S'=\tau\}\mid \sigma,a).
\]
Then
\[
P_p(\sigma,\tau)
=
\sum_{a\in\{0,1\}}p(a\mid x(\sigma))\,\kappa_{\sigma,\tau}(a).
\]

\begin{enumerate}
\item Suppose \(d_b(\sigma)>0\) and \(P_p(\sigma,\tau)>0\).  We claim that
\[
d_b(\tau)>0 .
\]
First show that \(P_b(\sigma,\tau)>0\).  If \(p=b\), this is immediate.  If \(p=e\), suppose instead that \(P_b(\sigma,\tau)=0\).  By \cref{obj:def:model-class}, the behavior policy is a probability vector and \(K\) is a Markov kernel, so every summand in
\[
P_b(\sigma,\tau)
=
\sum_{a\in\{0,1\}} b(a\mid x(\sigma))\,\kappa_{\sigma,\tau}(a)
\]
is nonnegative.  Hence, for each action \(a\),
\[
b(a\mid x(\sigma))\,\kappa_{\sigma,\tau}(a)=0 .
\]
For such an \(a\), if \(\kappa_{\sigma,\tau}(a)=0\) then
\[
e(a\mid x(\sigma))\,\kappa_{\sigma,\tau}(a)=0.
\]
If \(\kappa_{\sigma,\tau}(a)>0\), the preceding zero product gives \(b(a\mid x(\sigma))=0\); by \cref{obj:ass:policy-overlap},
\[
e(a\mid x(\sigma))
\le
L\,b(a\mid x(\sigma))
=
0,
\]
and the target policy is nonnegative by \cref{obj:def:model-class}, so \(e(a\mid x(\sigma))=0\).  Thus again
\[
e(a\mid x(\sigma))\,\kappa_{\sigma,\tau}(a)=0 .
\]
Summing over \(a\) gives
\[
P_e(\sigma,\tau)
=
\sum_{a\in\{0,1\}}e(a\mid x(\sigma))\,\kappa_{\sigma,\tau}(a)
=
0,
\]
contradicting \(P_p(\sigma,\tau)>0\) in the case \(p=e\).  Therefore \(P_b(\sigma,\tau)>0\).

By \cref{obj:def:model-class}, \(d_b\) is stationary for \(P_b\), so
\[
d_b(\tau)
=
\sum_{\omega\in\mathcal S} d_b(\omega)P_b(\omega,\tau).
\]
All terms in this sum are nonnegative.  Therefore
\[
0
<
d_b(\sigma)P_b(\sigma,\tau)
\le
\sum_{\omega\in\mathcal S} d_b(\omega)P_b(\omega,\tau)
=
d_b(\tau),
\]
which proves the claim. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.behaviorSupport_closed_policyKernel

\item We now prove the asserted identity by induction on \(m\).  For \(m=0\),
\[
(P_p^0f)(s)=f(s)=g(s)=(P_p^0g)(s),
\]
because \(d_b(s)>0\) and \(f=g\) on the behavior support.

Assume the statement holds for \(m\): for every joint state \(\sigma\) with \(d_b(\sigma)>0\),
\[
(P_p^m f)(\sigma)=(P_p^m g)(\sigma).
\]
Let \(s\) satisfy \(d_b(s)>0\).  Using the recursive definition of the iterated operator,
\[
(P_p^{m+1}f)(s)
=
\sum_{\tau\in\mathcal S}P_p(s,\tau)(P_p^m f)(\tau),
\qquad
(P_p^{m+1}g)(s)
=
\sum_{\tau\in\mathcal S}P_p(s,\tau)(P_p^m g)(\tau).
\]
Fix \(\tau\in\mathcal S\).  If \(P_p(s,\tau)=0\), then the two \(\tau\)-summands are both zero.  If \(P_p(s,\tau)\neq 0\), the row nonnegativity of \(P_p\) gives \(P_p(s,\tau)>0\).  The support-closure claim from the previous step then gives \(d_b(\tau)>0\), so the induction hypothesis yields
\[
(P_p^m f)(\tau)=(P_p^m g)(\tau).
\]
Multiplying by \(P_p(s,\tau)\), the two \(\tau\)-summands agree.  Since this holds for every \(\tau\), the sums agree:
\[
(P_p^{m+1}f)(s)=(P_p^{m+1}g)(s).
\]
The induction proves the result for every \(m\in\mathbb N\). % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.markovOperatorIter_congr_on_behaviorSupport
\end{enumerate}
\end{proof}
