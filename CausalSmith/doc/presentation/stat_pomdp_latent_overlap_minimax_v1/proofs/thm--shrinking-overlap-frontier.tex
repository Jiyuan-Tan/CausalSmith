\begin{proof}[Proof of \cref{obj:thm:shrinking-overlap-frontier}]
Put
\[
\alpha=\exp(-1/t_0),\qquad L=\exp(\zeta),\qquad
\beta=\frac{2}{2+t_0\zeta},\qquad
\gamma=2(1-\beta).
\]
Since \(t_0>0\) and \(\zeta>0\), one has \(0<\alpha<1\), \(1<L\),
\(0<\beta<1\), and hence \(\gamma>0\).

\emph{1. A radius-depth calibration.}
Let
\[
\Lambda:=2\log(1/\alpha)+\log L=\frac{2}{t_0}+\zeta .
\]
Then \(\Lambda>0\),
\[
\beta=\frac{2/t_0}{\Lambda},
\qquad
1-\beta=\frac{\zeta}{\Lambda}.
\]
There are \(c_R>0\) and \(T_R\ge1\), depending only on \(t_0,\zeta\), such that for every \(T\ge T_R\) and every \(C\ge1\),
\begin{equation}\label{eq:shrinking-overlap-frontier-radius-calibration}
q_C^2\alpha^{2\kappa_T^{\mathrm{rad}}(C)}
+\frac{L^{\kappa_T^{\mathrm{rad}}(C)}}{T}
\le
c_R\left\{
T^{-1}+T^{-\beta}q_C^{2(1-\beta)}
\right\}.
\end{equation}
Indeed, choose \(T_R\) so that \(|\log T|\le(\Lambda/2)T\) for all
\(T\ge T_R\).  Fix such a \(T\), put \(q=q_C\), and set \(x=Tq^2\).
The relation \(C\ge1\) and \cref{obj:def:overlap-distance-radius} give
\(0\le q\le1\), hence \(0\le x\le T\).  If \(x\le1\), then
\(\kappa_T^{\mathrm{rad}}(C)=0\) by \cref{obj:def:radius-adaptive-history-depth}, and
\(q^2\le T^{-1}\), so
\[
q^2+\frac1T
\le 2T^{-1}
\le
2\left\{T^{-1}+T^{-\beta}q^{2(1-\beta)}\right\}.
\]
If \(x>1\), then \(q>0\).  With
\[
k=\left\lfloor\frac{\log x}{\Lambda}\right\rfloor ,
\]
the choice of \(T_R\) gives \(k\le T/2\), so the cap in
\cref{obj:def:radius-adaptive-history-depth} is inactive and
\(\kappa_T^{\mathrm{rad}}(C)=k\).  The floor bounds
\[
\frac{\log x}{\Lambda}-1<k\le\frac{\log x}{\Lambda}
\]
imply
\[
\alpha^{2k}
=\exp\!\left(-\frac{2k}{t_0}\right)
\le
\exp(2/t_0)x^{-\beta},
\qquad
\frac{L^k}{T}
\le
\frac{x^{1-\beta}}{T}.
\]
Since
\[
q^2x^{-\beta}
=
T^{-\beta}q^{2(1-\beta)},
\qquad
\frac{x^{1-\beta}}{T}
=
T^{-\beta}q^{2(1-\beta)},
\]
the \(x>1\) case is bounded by
\[
\bigl(\exp(2/t_0)+1\bigr)T^{-\beta}q^{2(1-\beta)}.
\]
Taking
\[
c_R=\max\{2,\exp(2/t_0)+1\}
\]
proves \cref{eq:shrinking-overlap-frontier-radius-calibration}. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.radiusAdaptiveDepth_rate

\emph{2. Constants.}
Apply \cref{obj:thm:uniform-overlap-frontier} to obtain constants
\(c_{\mathrm{lo}},c_{\mathrm{hi}}>0\), with
\(c_{\mathrm{lo}}\le c_{\mathrm{hi}}\), and a threshold \(T_U\ge1\), all
depending only on \(t_0,\zeta\), such that its two-sided risk comparison and
radius-calibrated estimator bound hold for all \(C\ge1\) and \(T\ge T_U\).
Define
\[
\eta_{\mathrm{rad}}:=2^{-\gamma},
\]
and
\[
\mathsf A_{\mathrm{bd}}
:=
\max\left\{
4,\,
2L\left(1+\frac{4}{L-1}\right),\,
\frac{8}{1-\alpha}
\right\}.
\]
Finally set
\[
c_{\mathrm{loc}}:=c_{\mathrm{lo}}\eta_{\mathrm{rad}},
\qquad
c_{\mathrm{att}}:=\mathsf A_{\mathrm{bd}}(c_R+1),
\qquad
C_{\mathrm{loc}}:=\max\{c_{\mathrm{hi}},c_{\mathrm{att}}\}.
\]
Then \(0<c_{\mathrm{loc}}\le C_{\mathrm{loc}}\), because
\(0<\eta_{\mathrm{rad}}\le1\), \(c_{\mathrm{lo}}\le c_{\mathrm{hi}}\), and
\(\mathsf A_{\mathrm{bd}}>0\). % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.shrinking_overlap_frontier

\emph{3. Comparing the two rate surfaces.}
Fix a deterministic sequence \((\delta_T)_{T\ge1}\) with
\(\delta_T\ge0\) and \(\delta_T\to0\).  Choose \(T_\Delta\) such that
\[
0\le\delta_T<1\qquad\text{for all }T\ge T_\Delta .
\]
Let \(T_\star=\max\{T_U,T_R,T_\Delta\}\).  For \(T\ge T_\star\), put
\(C_T=1+\delta_T\).  Then \(C_T\ge1\), and
\[
q_{C_T}
=
\frac{C_T-1}{C_T}
=
\frac{\delta_T}{1+\delta_T}.
\]
Since \(0\le\delta_T<1\),
\[
0\le q_{C_T}\le\delta_T,
\qquad
\frac{\delta_T}{2}\le q_{C_T}.
\]
Raising these inequalities to the positive power \(\gamma\) gives
\[
\eta_{\mathrm{rad}}\delta_T^\gamma
\le
q_{C_T}^\gamma
\le
\delta_T^\gamma .
\]
Thus the theorem's local rate, evaluated along the fixed sequence \((\delta_T)\), and the fixed-radius rate at \(C_T\) satisfy
\begin{equation}\label{eq:shrinking-overlap-frontier-rate-comparison}
\eta_{\mathrm{rad}}\left\{T^{-1}+T^{-\beta}\delta_T^\gamma\right\}
\le
\mathfrak r_T(t_0,\zeta,C_T)
\le
T^{-1}+T^{-\beta}\delta_T^\gamma.
\end{equation}
Equivalently, since \(r_T(\delta)\) in the statement denotes this fixed-sequence quantity,
\[
\eta_{\mathrm{rad}}\,r_T(\delta)
\le
\mathfrak r_T(t_0,\zeta,C_T)
\le
r_T(\delta).
\]
Combining \cref{eq:shrinking-overlap-frontier-rate-comparison} with the two-sided comparison from
\cref{obj:thm:uniform-overlap-frontier} gives
\begin{equation}\label{eq:shrinking-overlap-frontier-risk-comparison}
c_{\mathrm{loc}}\,r_T(\delta)
\le
R_T(t_0,\zeta,1+\delta_T)
\le
C_{\mathrm{loc}}\,r_T(\delta).
\end{equation}
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.shrinking_overlap_frontier

\emph{4. Attainment by the displayed shrinking depth.}
For \(T\ge T_\star\), define
\[
C_T^\dagger:=\frac{1}{1-\delta_T}.
\]
Then \(C_T^\dagger\ge1\), and
\[
q_{C_T^\dagger}
=
\frac{C_T^\dagger-1}{C_T^\dagger}
=
\delta_T .
\]
Therefore \cref{obj:def:radius-adaptive-history-depth} gives
\begin{equation}\label{eq:shrinking-overlap-frontier-depth-identity}
\kappa_T^{\mathrm{rad}}(C_T^\dagger)=\kappa_T.
\end{equation}
where \(\kappa_T\) is the depth displayed in the theorem statement.  In
particular \(\kappa_T\le T/2\).  Applying \cref{eq:shrinking-overlap-frontier-radius-calibration} with \(C=C_T^\dagger\) yields
\begin{equation}\label{eq:shrinking-overlap-frontier-local-depth-rate}
\delta_T^2\alpha^{2\kappa_T}
+\frac{L^{\kappa_T}}{T}
\le
c_R r_T(\delta).
\end{equation}
Now fix \(M\in\mathcal M_T(t_0,\zeta,1+\delta_T)\).  By
\cref{obj:lem:radius-sensitive-phiw}, the estimator from
\cref{obj:def:phiw-estimator} satisfies
\[
\mathbb E_M\!\left[
\bigl(\widehat\theta_{\kappa_T}-\theta(M)\bigr)^2
\right]
\le
4q_{C_T}^2\alpha^{2\kappa_T}
+\frac{2}{T}
\left\{
L^{\kappa_T+1}\left(1+\frac{4}{L-1}\right)
+\frac{4}{1-\alpha}
\right\}.
\]
Using \(q_{C_T}\le\delta_T\) and the definition of
\(\mathsf A_{\mathrm{bd}}\),
\[
\mathbb E_M\!\left[
\bigl(\widehat\theta_{\kappa_T}-\theta(M)\bigr)^2
\right]
\le
\mathsf A_{\mathrm{bd}}
\left\{
\delta_T^2\alpha^{2\kappa_T}
+\frac{L^{\kappa_T}}{T}
+T^{-1}
\right\}.
\]
Together with \cref{eq:shrinking-overlap-frontier-local-depth-rate} and \(T^{-1}\le r_T(\delta)\), this gives
\[
\mathbb E_M\!\left[
\bigl(\widehat\theta_{\kappa_T}-\theta(M)\bigr)^2
\right]
\le
c_{\mathrm{att}}\,r_T(\delta)
\le
C_{\mathrm{loc}}\,r_T(\delta).
\]
Taking the supremum over
\(M\in\mathcal M_T(t_0,\zeta,1+\delta_T)\) proves the asserted estimator
bound. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.shrinking_overlap_frontier

\emph{5. The bounded local regime.}
Assume that there is a constant \(D_{\mathrm{bd}}\ge0\) such that
\[
T\delta_T^2\le D_{\mathrm{bd}}
\]
for all sufficiently large \(T\).  Since \(q_{1+\delta_T}\le\delta_T\), the
same tail of \(T\)'s satisfies
\[
Tq_{1+\delta_T}^2\le D_{\mathrm{bd}}.
\]
Applying the final assertion of \cref{obj:thm:uniform-overlap-frontier} to
the sequence \(C_T=1+\delta_T\) gives a constant \(c_{\mathrm{imm}}>0\) such
that, for all sufficiently large \(T\),
\[
\sup_{M\in\mathcal M_T(t_0,\zeta,1+\delta_T)}
\mathbb E_M\!\left[
\bigl(\widehat\theta_0-\theta(M)\bigr)^2
\right]
\le
\frac{c_{\mathrm{imm}}}{T}.
\]
This is the claimed immediate-depth conclusion. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.shrinking_overlap_frontier
\end{proof}
