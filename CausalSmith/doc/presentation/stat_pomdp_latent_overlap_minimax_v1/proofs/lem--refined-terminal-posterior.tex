\begin{proof}[Proof of \cref{obj:lem:refined-terminal-posterior}]
\begin{enumerate}
\item
Let
\[
\Gamma:=\frac{\alpha}{1-c_0}.
\]
Since \(t_0>0\), the contraction parameter satisfies \(0<\alpha<1\).  Hence
\[
0\le c_0=\frac{1-\alpha}{4}\le \frac14,
\qquad
1-c_0>0,
\]
and therefore \(\Gamma\ge0\).  Moreover,
\[
\Gamma<1
\quad\Longleftrightarrow\quad
\alpha<1-\frac{1-\alpha}{4}
\quad\Longleftrightarrow\quad
\alpha<1.
\]
Thus \(\Gamma^Q\le1\).  Consequently
\[
B_Q:=1-c_0\Gamma^Q
\]
is strictly positive; indeed \(c_0\Gamma^Q\le c_0\le1/4\). % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.signedDepthRawPosterior_le_refined

\item
For \(0\le d\le k\), define the lower posterior window
\[
\Lambda_d(o)
:=
\prod_{s=k-d+1}^{k}
\left(1-c_0\,\pi_{N,Q,v}(o_{\le s-1})\right),
\]
where \(o_{\le m}\) denotes the first \(m\) observed action--reward pairs, with \(o_{\le0}\) the empty prefix; for \(d=0\) this is the empty product and equals \(1\).  If \(Q\le k\), \(\cref{obj:lem:terminal-posterior-mature-window}\) applies under the present hypotheses and gives
\[
\pi_{N,Q,v}(o)
\le
\frac{\alpha^Q}{\Lambda_Q(o)}.
\]
If \(k<Q\), \(\cref{obj:lem:terminal-posterior-initial-window}\) applies under the present hypotheses and gives
\[
\pi_{N,Q,v}(o)
\le
\frac{\alpha^Q}{\Lambda_k(o)}.
\]
These two estimates are exactly the two window cases needed below. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.signedDepthRawPosterior_le_product_of_le; CausalSmith.Stat.PomdpLatentOverlapMinimax.signedDepthRawPosterior_le_product_of_lt

\item
The crude terminal-posterior estimate in \(\cref{obj:lem:terminal-posterior-crude-bound}\) applies to every observed prefix \(o'\), of any length, under the same assumptions \(t_0>0\), \(\zeta>0\), and \(C>1\).  With the present notation it states
\[
\pi_{N,Q,v}(o')
\le
\left(\frac{\alpha}{1-c_0}\right)^Q
=
\Gamma^Q.
\]
This supplies the prefix-uniform bootstrap bound used in the lower-window estimate. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.signedDepthRawPosterior_le_ratio_pow

\item
Apply \(\cref{obj:lem:posterior-lower-window-power}\) with
\[
b:=\Gamma^Q.
\]
Its nonnegative-amplitude hypothesis is \(0\le c_0\), established in the first step, and its nonnegative-base hypothesis is
\[
0\le 1-c_0\Gamma^Q=B_Q,
\]
also established there.  Its strict-prefix posterior hypothesis holds because the preceding step gives, for every prefix appearing in the window,
\[
\pi_{N,Q,v}(o_{\le s-1})\le \Gamma^Q.
\]
The indexing in \(\cref{obj:lem:posterior-lower-window-power}\) ranges over the final \(d\) strict prefixes of \(o\), precisely the factors in \(\Lambda_d(o)\).  Hence, for every \(d\le k\),
\[
B_Q^d
=
\left(1-c_0\Gamma^Q\right)^d
\le
\Lambda_d(o).
\] % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.pow_one_sub_le_signedDepthPosteriorLowerWindow

\item
If \(Q\le k\), the terminal-window bound with \(d=Q\) and the preceding display give
\[
\pi_{N,Q,v}(o)
\le
\frac{\alpha^Q}{\Lambda_Q(o)}
\le
\frac{\alpha^Q}{B_Q^Q}.
\]
If \(k<Q\), the terminal-window bound with \(d=k\) gives
\[
\pi_{N,Q,v}(o)
\le
\frac{\alpha^Q}{\Lambda_k(o)}
\le
\frac{\alpha^Q}{B_Q^k}.
\]
Since \(0<B_Q\le1\) and \(k<Q\), we have \(B_Q^Q\le B_Q^k\), hence
\[
\frac{\alpha^Q}{B_Q^k}
\le
\frac{\alpha^Q}{B_Q^Q}.
\]
In both cases,
\[
\pi_{N,Q,v}(o)
\le
\frac{\alpha^Q}
{\left(1-c_0\left(\frac{\alpha}{1-c_0}\right)^Q\right)^Q},
\]
which is the claimed bound. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.signedDepthRawPosterior_le_refined
\end{enumerate}
\end{proof}
