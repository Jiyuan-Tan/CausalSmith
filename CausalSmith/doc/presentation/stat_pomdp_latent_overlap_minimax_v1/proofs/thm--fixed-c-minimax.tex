\begin{proof}[Proof of \cref{obj:thm:fixed-c-minimax}]
\begin{enumerate}
\item Set
\[
\alpha:=\exp(-1/t_0),
\qquad
L:=\exp(\zeta),
\qquad
D:=\frac{2}{t_0}+\zeta .
\]
By \(t_0>0\) and \(\zeta>0\), the constants satisfy
\[
0<\alpha<1,\qquad L>1,\qquad D>0,
\]
and, using the definitions in \cref{obj:synth_7,obj:synth_8},
\[
\log\alpha=-\frac1{t_0},
\qquad
\log L=\zeta .
\]
The exponent in the theorem can be written as
\[
\beta=\frac{2}{2+t_0\zeta}
      =\frac{2/t_0}{D}.
\]
Apply \cref{obj:lem:phiw-upper} with \(C\ge1\). It gives constants
\(c_{\mathrm{up},0}>0\) and \(T_{\mathrm{up}}\) such that, for all
\(T\ge T_{\mathrm{up}}\),
\[
\sup_{M\in\mathcal M_T(t_0,\zeta,C)}
\mathbb E_M\!\left[
\{\widehat\theta_{k_T}-\theta(M)\}^2
\right]
\le
c_{\mathrm{up},0}T^{-\beta}.
\]
For the signed-depth family, the behavior policy in \cref{obj:def:signed-depth-family} is a memoryless function of the single observed state, independent of the latent coordinate and the past. Consequently, for every horizon \(T\), terminal depth \(Q\ge1\), and sign \(v\in\{-1,+1\}\), the action kernel satisfies \cref{obj:ass:sequential-ignorability}. The same alternatives start from the behavior stationary law for every horizon, including the zero-horizon case. Indeed, for
\[
r:=L^{-1},
\]
the signed-depth constant-policy stationary law at action-one probability \(r\) has masses
\[
w_j\frac{1+u\varepsilon_C r^j}{2},
\qquad
w_j=\frac{(1-\alpha)\alpha^j}{1-\alpha^{Q+1}},
\]
and the behavior policy in \cref{obj:def:signed-depth-family} chooses action one with probability \(L^{-1}=r\). Hence its stationary law is precisely the initial hidden-state law
\[
w_j\frac{1+u\varepsilon_C L^{-j}}{2}.
\]
The chronological path law is generated from this initial distribution, so the time-zero joint-state marginal is this stationary law; when \(T=0\), the same identity is the whole marginal check, and for \(T\ge1\) it agrees with the behavior-stationary masses displayed in \cref{obj:lem:signed-depth-membership}. Thus \cref{obj:ass:stationary-start} holds for every horizon. These are the two structural hypotheses needed for \cref{obj:lem:observed-path-kl}. Therefore that lemma gives a constant \(B_0>0\), depending on \(t_0,\zeta,C\), such that for all \(T\) and \(Q\ge1\),
\[
\operatorname{KL}\!\left(
\mathbb P_{Q,+1}^{\mathrm{obs},T}
\,\middle\|\,
\mathbb P_{Q,-1}^{\mathrm{obs},T}
\right)
\le
B_0T\alpha^{2Q}L^{-Q}.
\]
Let
\[
\varepsilon_C:=\min\left\{\frac{C-1}{2},\frac12\right\},
\qquad
c_0:=\frac{1-\alpha}{4},
\]
and define
\[
c_\star:=
\{c_0\varepsilon_C(1-\alpha)\alpha\}^2
(8B_0)^{-\beta}\frac{\exp(-1/8)}{4}.
\]
Since \(C>1\), every factor in this product is positive, so \(c_\star>0\). Finally set
\[
c^\star:=\max\{c_{\mathrm{up},0},c_\star\}.
\]
Since \(c_{\mathrm{up},0}>0\) and \(c_\star>0\), the maximum is positive; moreover the second argument is bounded by the maximum. Thus
\[
0<c^\star,
\qquad
0<c_\star\le c^\star .
\] % lean: fixed_c_minimax

\item Define
\[
T_{\mathrm{pos}}:=\left\lceil\frac{1}{8B_0}\right\rceil+1,
\qquad
T_\star:=\max\{T_{\mathrm{up}},T_{\mathrm{pos}},1\}.
\]
Fix an integer \(T\ge T_\star\). Then \(T\ge1\), \(T\ge T_{\mathrm{up}}\), and
\[
1<8B_0T .
\]
Indeed,
\[
\frac{1}{8B_0}
\le
\left\lceil\frac{1}{8B_0}\right\rceil
<
T,
\]
and multiplication by \(8B_0>0\) gives the displayed inequality. Put
\[
\tau_T:=\frac{\log(8B_0T)}{D},
\qquad
Q:=\lceil \tau_T\rceil .
\]
Since \(\log(8B_0T)>0\) and \(D>0\), one has \(Q\ge1\). Moreover,
\[
\tau_T\le Q<\tau_T+1
\]
and therefore
\[
DQ\ge \log(8B_0T).
\] % lean: fixed_c_minimax

\item The preceding depth choice makes the two signed-depth alternatives statistically close. First,
\[
\alpha^{2Q}L^{-Q}
=
\exp\{Q(2\log\alpha-\log L)\}
=
\exp(-DQ),
\]
because \(D=2/t_0+\zeta\), \(\log\alpha=-1/t_0\), and \(\log L=\zeta\). Hence
\[
\exp(-DQ)
\le
\exp\{-\log(8B_0T)\}
=
(8B_0T)^{-1}.
\]
Combining this inequality with the KL bound from \cref{obj:lem:observed-path-kl} gives
\[
\operatorname{KL}\!\left(
\mathbb P_{Q,+1}^{\mathrm{obs},T}
\,\middle\|\,
\mathbb P_{Q,-1}^{\mathrm{obs},T}
\right)
\le
\frac18 .
\] % lean: fixed_c_minimax

\item Let \(M_{Q,+1}^T\) and \(M_{Q,-1}^T\) be the two signed-depth experiments of \cref{obj:def:signed-depth-family}. They have one observed state and \(2(Q+1)\) latent states. Their revealed behavior and target policies are common, because the policy part of the construction is independent of the sign \(v\). By \cref{obj:lem:signed-depth-membership}, both experiments belong to \(\mathcal M_T(t_0,\zeta,C)\), and their target values are
\[
\theta(M_{Q,+1}^T)=c_0\varepsilon_C w_Q,
\qquad
\theta(M_{Q,-1}^T)=-c_0\varepsilon_C w_Q,
\]
where
\[
w_Q
:=
\frac{(1-\alpha)\alpha^Q}{1-\alpha^{Q+1}} .
\]
Set
\[
s_T:=c_0\varepsilon_C w_Q .
\]
Then \(s_T\ge0\), and the two target values satisfy the separation bound
\[
2s_T
\le
\bigl|\theta(M_{Q,+1}^T)-\theta(M_{Q,-1}^T)\bigr|.
\]
The two embedded observed laws have the same finite real-reward support,
\[
\Omega_T^\circ
:=
\left\{
((0,a_t,y_t))_{t=1}^T:
 a_t\in\{0,1\},\ y_t\in\{-1,+1\}
\right\}.
\]
They assign total mass one to \(\Omega_T^\circ\), because the construction has one observed state, Boolean actions, and rewards in \(\{-1,+1\}\). We now check that every atom in this common support has positive mass under both signs. Fix
\(o=((0,a_t,y_t))_{t=1}^T\in\Omega_T^\circ\), and choose a fixed latent sign \(u_0\in\{-1,+1\}\). Consider the full hidden lift that keeps the latent state at depth zero and sign \(u_0\):
\[
S_t=(0,(0,u_0)),
\qquad
A_t=a_t,
\qquad
Y_t=y_t,
\qquad t=1,\\ldots,T.
\]
Since \(0<\alpha<1\) and \(Q\ge1\),
\[
1-\alpha^{Q+1}>0,
\qquad
w_0=\frac{1-\alpha}{1-\alpha^{Q+1}}>0.
\]
Also \(0<\varepsilon_C\le1/2\), so the initial mass of the chosen hidden state is positive:
\[
w_0\frac{1+u_0\varepsilon_C}{2}>0.
\]
For each action in the observed word,
\[
b(1\mid0)=\frac1L>0,
\qquad
b(0\mid0)=1-\frac1L>0,
\]
because \(L>1\). Along the displayed hidden lift, the current depth is \(0<Q\). Hence the reward factor is
\[
\frac{1+y_tvc_0u_0\mathbf 1\{0=Q\}}{2}=\frac12>0,
\]
and the hidden transition that resets to the same depth-zero sign has weight
\[
(1-\alpha)\frac{1+u_0\varepsilon_C}{2}>0.
\]
Thus this single compatible hidden lift has strictly positive finite-path probability for each sign \(v\in\{-1,+1\}\), and summing over all lifts gives
\[
\mathbb P_{Q,v}^{\mathrm{obs},T}(\{o\})>0,
\qquad o\in\Omega_T^\circ,
\qquad v\in\{-1,+1\}.
\]
The reward decoding used in the embedding is the same for the two signs, so the embedded laws preserve this common atomic support. Therefore any measurable set with zero mass under one embedded observed law contains no atom of \(\Omega_T^\circ\), and consequently has zero mass under the other embedded observed law. Hence
\[
\mathbb P_{Q,+1}^{\mathrm{obs},T}
\ll
\mathbb P_{Q,-1}^{\mathrm{obs},T},
\qquad
\mathbb P_{Q,-1}^{\mathrm{obs},T}
\ll
\mathbb P_{Q,+1}^{\mathrm{obs},T},
\]
and the finite KL displayed above is an ordinary KL bound between mutually absolutely continuous observed laws. % lean: signedDepth_fullSupportObs

\item Apply \cref{obj:lem:two-point-risk-floor} to the present pair with \(s=s_T\), taking the \(+1\) alternative as the lemma's first model and the \(-1\) alternative as its second: \(M_0=M_{Q,+1}^T\) and \(M_1=M_{Q,-1}^T\). With this assignment the lemma's divergence hypothesis \(\operatorname{KL}(\mathbb P_0\|\mathbb P_1)\le\chi\) reads
\[
\operatorname{KL}\!\left(
\mathbb P_{Q,+1}^{\mathrm{obs},T}
\,\middle\|\,
\mathbb P_{Q,-1}^{\mathrm{obs},T}
\right)
\le
\frac18 ,
\]
which is exactly the direction bounded above, and its absolute-continuity hypothesis reads \(\mathbb P_{Q,+1}^{\mathrm{obs},T}\ll\mathbb P_{Q,-1}^{\mathrm{obs},T}\), which the preceding step supplies. The common observable structure was checked in the preceding step, the KL bound gives the information hypothesis with \(\chi=1/8\), and the displayed target-value calculation gives
\[
2s_T
\le
\bigl|\theta(M_{Q,+1}^T)-\theta(M_{Q,-1}^T)\bigr|.
\]
Therefore every measurable observable estimator \(\widehat\theta\), under the stated squared-error integrability conditions for both alternatives, satisfies
\[
s_T^2\frac{\exp(-1/8)}{4}
\le
\max\left\{
\mathbb E_{M_{Q,+1}^T}\!\left[
\{\widehat\theta-\theta(M_{Q,+1}^T)\}^2
\right],
\mathbb E_{M_{Q,-1}^T}\!\left[
\{\widehat\theta-\theta(M_{Q,-1}^T)\}^2
\right]
\right\}.
\]
Commuting the two entries of the maximum, this is exactly
\[
s_T^2\frac{\exp(-1/8)}{4}
\le
\max_{v\in\{-1,+1\}}
\mathbb E_{M_{Q,v}^T}\!\left[
\{\widehat\theta-\theta(M_{Q,v}^T)\}^2
\right].
\]
The two signed-depth alternatives belong to \(\mathcal M_T(t_0,\zeta,C)\) and share the revealed inputs \((n_X,b,e)\). For the witness clause of the theorem the two squared-error integrability conditions are hypotheses, imposed on the estimator and discharged by whoever supplies it; nothing above derives them. They are derived only when \(\widehat\theta\) is further restricted to the admissible carrier of \cref{obj:def:minimax-risk}, whose members are measurable and uniformly \([-1,1]\)-valued by definition: for such an estimator the squared error at either alternative is a measurable function bounded by \((1+|\theta_\pm|)^2\le4\) on a probability space, hence integrable, so the preceding pointwise lower bound applies to every admissible estimator. Taking the supremum over the model class, then the infimum over these admissible estimators, yields
\[
s_T^2\frac{\exp(-1/8)}{4}
\le
R_T(t_0,\zeta,C).
\] % lean: two_point_floor_explicit

\item It remains to compare \(s_T\) with \(T^{-\beta/2}\). Since \(0<\alpha<1\) and \(Q\ge1\), also \(Q+1\ge1\). Thus
\[
0<\alpha^{Q+1}<1,
\]
so the denominator in the depth mass is strictly positive and satisfies
\[
0<1-\alpha^{Q+1}\le1.
\]
The numerator \((1-\alpha)\alpha^Q\) is nonnegative, and division by a positive number at most one gives
\[
w_Q
=
\frac{(1-\alpha)\alpha^Q}{1-\alpha^{Q+1}}
\ge
(1-\alpha)\alpha^Q .
\]
The upper ceiling inequality \(Q<\tau_T+1\) and the identity \(\log\alpha=-1/t_0\) give
\[
\alpha^Q
=
\exp\!\left(-\frac{Q}{t_0}\right)
\ge
\exp\!\left(-\frac{\tau_T+1}{t_0}\right)
=
\alpha(8B_0T)^{-(1/t_0)/D}.
\]
Therefore
\[
s_T
\ge
c_0\varepsilon_C(1-\alpha)\alpha
(8B_0T)^{-(1/t_0)/D}.
\]
Squaring and using \(\beta=(2/t_0)/D\) yields
\[
s_T^2
\ge
\{c_0\varepsilon_C(1-\alpha)\alpha\}^2
(8B_0T)^{-\beta}
=
\{c_0\varepsilon_C(1-\alpha)\alpha\}^2
(8B_0)^{-\beta}T^{-\beta}.
\]
Multiplying by \(\exp(-1/8)/4\) gives
\[
c_\star T^{-\beta}
\le
s_T^2\frac{\exp(-1/8)}{4}.
\]
Combining this with the two-point floor proves
\[
c_\star T^{-\beta}
\le
R_T(t_0,\zeta,C),
\]
and, for every measurable observable-data estimator satisfying the two integrability conditions,
\[
c_\star T^{-\beta}
\le
\max_{v\in\{-1,+1\}}
\mathbb E_{M_{Q,v}^T}\!\left[
\{\widehat\theta-\theta(M_{Q,v}^T)\}^2
\right].
\] % lean: fixed_c_minimax

\item The upper bound follows from the estimator bound in the first step. Since \(T\ge T_{\mathrm{up}}\),
\[
\sup_{M\in\mathcal M_T(t_0,\zeta,C)}
\mathbb E_M\!\left[
\{\widehat\theta_{k_T}-\theta(M)\}^2
\right]
\le
c_{\mathrm{up},0}T^{-\beta}
\le
c^\star T^{-\beta}.
\]
By \cref{obj:def:minimax-risk}, the minimax risk is bounded above by the worst-case risk of any admissible observable estimator, in particular by that of \(\widehat\theta_{k_T}\). Thus
\[
R_T(t_0,\zeta,C)
\le
c^\star T^{-\beta}.
\]
Together with the lower bound from the preceding step, this proves all asserted inequalities for every \(T\ge T_\star\), and the chosen \(Q=\lceil \log(8B_0T)/D\rceil\) supplies the stated signed-depth witnesses with one observed state and \(2(Q+1)\) latent states. % lean: fixed_c_minimax
\end{enumerate}
\end{proof}
