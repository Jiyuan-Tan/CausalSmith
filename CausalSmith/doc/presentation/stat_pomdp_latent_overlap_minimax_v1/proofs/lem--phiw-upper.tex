\begin{proof}[Proof of \cref{obj:lem:phiw-upper}]
\begin{enumerate}
\item Define
\[
\alpha=\exp(-1/t_0),\qquad L=\exp(\zeta),\qquad
\beta=\frac{2}{2+t_0\zeta},
\qquad q_C=\frac{C-1}{C}.
\]
The assumptions \(t_0>0\) and \(\zeta>0\) give
\[
0<\alpha<1,\qquad 1<L,\qquad 0<\beta<1.
\]
Also \(C\ge1\) implies
\[
0\le q_C\le1,\qquad q_C^2\le1.
\]
Let
\[
\mathfrak d
=
2\log(1/\alpha)+\log L
=
\frac{2}{t_0}+\zeta .
\]
Then \(\mathfrak d>0\) and
\[
\beta=\frac{2/t_0}{\mathfrak d}.
\]
Since \(\log T=o(T)\), there is an integer \(T_{\mathrm{bal}}\ge1\) such that, for every
\(T\ge T_{\mathrm{bal}}\),
\[
0\le \frac{\log T}{\mathfrak d}\le \frac{T}{2}.
\]
For such \(T\), \cref{obj:def:history-depth} gives
\[
k_T=\left\lfloor\frac{\log T}{\mathfrak d}\right\rfloor ,
\]
and hence
\[
\frac{\log T}{\mathfrak d}-1<k_T\le \frac{\log T}{\mathfrak d}.
\]
Therefore
\[
\alpha^{2k_T}
=
\exp(-2k_T/t_0)
\le
\exp(2/t_0)\,T^{-\beta},
\]
and
\[
\frac{L^{k_T+1}}{T}
=
\exp\{(k_T+1)\zeta-\log T\}
\le
\exp(\zeta)\,T^{-\beta}.
\]
With
\[
c_{\mathrm{bal}}:=\exp(2/t_0)+\exp(\zeta)>0,
\]
we have, for every \(T\ge T_{\mathrm{bal}}\),
\[
\alpha^{2k_T}+\frac{L^{k_T+1}}{T}
\le
c_{\mathrm{bal}}T^{-\beta}.
\]
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.historyDepth_rate

\item Set
\[
\mathsf A:=1+\frac{4}{L-1},
\qquad
\mathsf B:=\frac{4}{1-\alpha},
\qquad
\mathsf D:=\max\{4,2\mathsf A\},
\]
and define
\[
c^\star:=\mathsf D\,c_{\mathrm{bal}}+2\mathsf B,
\qquad
T_\star:=T_{\mathrm{bal}}.
\]
Since \(L>1\) and \(\alpha<1\), the constants \(\mathsf A,\mathsf B,\mathsf D\) are positive, and so
\(c^\star>0\).
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.phiw_upper

\item Fix \(T\ge T_\star\). Then \(1\le T\), \(k_T\le T/2\), and \(k_T<T\). For every
\(M\in\mathcal M_T(t_0,\zeta,C)\), \cref{obj:lem:radius-sensitive-phiw}, applied with
\(k=k_T\), gives
\[
\mathbb E_M\!\left[
\{\widehat\theta_{k_T}-\theta(K,e)\}^2
\right]
\le
4q_C^2\alpha^{2k_T}
+
\frac{2}{T}\left(L^{k_T+1}\mathsf A+\mathsf B\right).
\]
Using \(q_C^2\le1\), \(\mathsf D\ge4\), and \(\mathsf D\ge2\mathsf A\),
\[
4q_C^2\alpha^{2k_T}
+
\frac{2}{T}\left(L^{k_T+1}\mathsf A+\mathsf B\right)
\le
\mathsf D\left(\alpha^{2k_T}+\frac{L^{k_T+1}}{T}\right)
+
2\mathsf B\,T^{-1}.
\]
The balance bound from the first step and the inequality \(T^{-1}\le T^{-\beta}\), valid because
\(T\ge1\) and \(0<\beta<1\), yield
\[
\mathbb E_M\!\left[
\{\widehat\theta_{k_T}-\theta(K,e)\}^2
\right]
\le
\mathsf D c_{\mathrm{bal}}T^{-\beta}
+
2\mathsf B T^{-\beta}
=
c^\star T^{-\beta}.
\]
There are two cases. If the indexed class \(\mathcal M_T(t_0,\zeta,C)\) is nonempty, the pointwise bound over every member of the class, together with the estimator-specific risk definition in \cref{obj:synth_13}, gives
\[
R_T(t_0,\zeta,C;\widehat\theta_{k_T})
\le
c^\star T^{-\beta}.
\]
If the indexed class is empty, the real-valued worst-case-risk functional used for \(R_T\) has empty supremum equal to \(0\). Since \(c^\star>0\), \(T\ge1\), and \(T^{-\beta}\ge0\), the bound also holds in this case.
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.phiw_upper

\item It remains to record the variance assertion.
For every \(T\ge T_\star\), the definition of \(T_\star\) gives \(1\le T\), and the minimum in \cref{obj:def:history-depth} gives
\(k_T\le\lfloor T/2\rfloor\).  Apply \cref{obj:lem:phiw-raw-variance} with
\(k=k_T\), \(L=e^\zeta\), and \(\alpha=e^{-1/t_0}\).  Its conclusion is exactly
\[
\operatorname{Var}_M\!\left(
\frac1{T-k_T}\sum_{t=k_T}^{T-1}Z_t(k_T)
\right)
\le
\frac{2}{T}\left[
e^{\zeta(k_T+1)}\left(1+\frac{4}{e^\zeta-1}\right)
+\frac{4}{1-e^{-1/t_0}}
\right],
\]
uniformly over the finite alphabet sizes and every
\(M\in\mathcal M_T(t_0,\zeta,C)\), as required.
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.variance_phiwRaw_le
\end{enumerate}
\end{proof}
