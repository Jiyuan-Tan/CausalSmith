\begin{proof}[Proof of \cref{obj:lem:signed-depth-membership}]
Write \(\alpha=\alpha(t_0)=\exp(-1/t_0)\), \(L=\exp(\zeta)\), and
\(\varepsilon=\varepsilon_C\).  Since \(t_0>0\) and \(\zeta>0\), we have
\(0<\alpha<1\) and \(1<L\).  Also \(C>1\) gives \(C\ge 1\), so
\[
0\le \varepsilon=\min\{(C-1)/2,1/2\}\le \frac12 .
\]
For a hidden state \(h\), abbreviate \(j=j(h)\) and \(u=u(h)\in\{-1,+1\}\).

\begin{enumerate}
\item For \(r\in[0,1]\), let \(a_r\) be the constant policy with
\(\Pr(A_t=1)=r\), and let \(P^{(r)}\) be the induced transition kernel on the
joint state space
\[
\mathcal S_Q=\{(0,h):h\in\{0,\ldots,Q\}\times\{-1,+1\}\}.
\]
By \cref{obj:lem:signed-depth-constant-policy-contraction}, for all probability
laws \(\mu\) and \(\mu'\) on \(\mathcal S_Q\),
\[
\|\mu P^{(r)}-\mu'P^{(r)}\|_{\mathrm{TV}}
\le
\alpha\|\mu-\mu'\|_{\mathrm{TV}}.
\]
The behavior policy in \cref{obj:def:signed-depth-family} is the case
\(r=L^{-1}\), while the target policy is the case \(r=1\).  The displayed
inequality therefore gives the contraction required in
\cref{obj:ass:uniform-contraction} for both policies. % lean: signedDepth_uniformContraction

\item For \(r\in[0,1]\), \cref{obj:lem:signed-depth-stationary-law} identifies
the selected stationary law of the constant-policy kernel \(P^{(r)}\): for
every hidden state \(h\),
\[
d_r(0,h)
=
w_{j(h)}
\frac{1+u(h)\varepsilon r^{j(h)}}2 .
\]
Taking \(r=L^{-1}\) gives the behavior stationary law because the behavior
policy in \cref{obj:def:signed-depth-family} chooses action one with
probability \(1/L\), and \((L^{-1})^{j(h)}=L^{-j(h)}\).  Thus
\[
d_b(0,h)
=
w_{j(h)}
\frac{1+u(h)\varepsilon L^{-j(h)}}2 .
\]
Taking \(r=1\) gives the target stationary law
\[
d_e(0,h)
=
w_{j(h)}
\frac{1+u(h)\varepsilon}{2}.
\] % lean: signedDepth_stationaryLaw_apply

\item We now verify the model-class predicates in \cref{obj:def:model-class}.
The embedded experiment is generated by the initial law, the behavior policy,
and the time-homogeneous reward-transition kernel in
\cref{obj:def:signed-depth-family}.  Therefore, for each epoch
\(t=0,\ldots,T-1\),
\[
\mathcal L(Y_t,X_{t+1},H_{t+1}\mid X_{0:t},H_{0:t},A_{0:t},Y_{0:t-1})
=
K(\cdot\mid X_t,H_t,A_t),
\]
which is \cref{obj:ass:pomdp-kernel}.  The behavior action law is
\[
b(1\mid0)=L^{-1},
\qquad
b(0\mid0)=1-L^{-1},
\qquad
b(0\mid0)+b(1\mid0)=1,
\]
and these two masses are nonnegative because \(L>1\).  Since the observed
state is the singleton \(\{0\}\), the chronological law factors at every epoch
\(t=0,\ldots,T-1\) as
\[
\begin{aligned}
&\mathbb P\!\left\{
\bigl((X_s,H_s)_{s<t},(A_s,Y_s)_{s<t},(X_t,H_t),A_t\bigr)
\in d\mathsf h\,da
\right\} \\
&\qquad=
\mathbb P\!\left\{
\bigl((X_s,H_s)_{s<t},(A_s,Y_s)_{s<t},(X_t,H_t)\bigr)
\in d\mathsf h
\right\}\,b(da\mid X_t(\mathsf h)),
\end{aligned}
\]
which is \cref{obj:ass:sequential-ignorability}.  The finite reward symbols are
decoded as \(-1\) and \(+1\), so
\[
\Pr(|Y_t|\le1)=1,
\]
giving \cref{obj:ass:bounded-reward}.  The policy-overlap inequality is
\[
e(1\mid0)=1=L\,b(1\mid0),
\qquad
e(0\mid0)=0\le L\,b(0\mid0),
\]
so \cref{obj:ass:policy-overlap} holds with factor \(L\).  The initial law is
\[
\Pr\{S_0=(0,h)\}
=
w_{j(h)}
\frac{1+u(h)\varepsilon L^{-j(h)}}2
=
d_b(0,h),
\]
where the last equality is the stationary-law formula from the case
\(r=L^{-1}\).  Thus \cref{obj:ass:stationary-start} holds.

For the latent stationary overlap, first identify the two selected stationary
laws and the initial law on the same cells.  The target and behavior formulas
from Step 2 give
\[
d_e(0,h)
=
w_j\frac{1+u(h)\varepsilon}{2},
\qquad
d_b(0,h)
=
w_j\frac{1+u(h)\varepsilon L^{-j}}{2}.
\]
The displayed initialization in \cref{obj:def:signed-depth-family} gives
\[
\Pr\{S_0=(0,h)\}
=
w_j\frac{1+u(h)\varepsilon L^{-j}}{2}
=
d_b(0,h),
\]
where the last equality is the behavior stationary-law formula from Step 2.
The depth mass is nonnegative: since
\[
0<\alpha<1,
\qquad
1-\alpha^{Q+1}>0,
\]
we have
\[
w_j
=
\frac{(1-\alpha)\alpha^j}{1-\alpha^{Q+1}}
\ge0 .
\]
Here \(L^{-j}=(L^j)^{-1}\).  Since \(L>1\), we have \(0<L^{-j}\le1\).
For \(u(h)=-1\),
\[
d_e(0,h)
=
w_j\frac{1-\varepsilon}{2}
\le
w_j\frac{1-\varepsilon L^{-j}}2
=
d_b(0,h)
\le
(1+\varepsilon)d_b(0,h),
\]
where the first inequality uses \(\varepsilon L^{-j}\le\varepsilon\), and the
last uses \(\varepsilon\ge0\) and nonnegativity of the behavior cell.  For
\(u(h)=+1\),
\[
d_e(0,h)
=
w_j\frac{1+\varepsilon}{2}
\le
w_j\frac{1+\varepsilon}{2}(1+\varepsilon L^{-j})
=
(1+\varepsilon)d_b(0,h),
\]
because \(w_j\ge0\) and \(1+\varepsilon\ge0\), so the multiplier
\(w_j(1+\varepsilon)/2\) is nonnegative, while
\(1+\varepsilon L^{-j}\ge1\).  Multiplying this last inequality by that
nonnegative multiplier gives the displayed inequality, and its right-hand
side is \((1+\varepsilon)d_b(0,h)\) by the formula for \(d_b\).  Every joint state has the unique
observed coordinate \(0\), so these two sign cases give
\[
d_e(s)\le (1+\varepsilon)d_b(s)
\qquad\text{for every }s\in\mathcal S_Q.
\]
Moreover
\[
1+\varepsilon
=
1+\min\{(C-1)/2,1/2\}
\le C,
\]
because either \(\varepsilon=(C-1)/2\), giving
\(1+\varepsilon=(C+1)/2\le C\), or \(\varepsilon=1/2\), which is possible only
when \(C\ge2\).  Since \(d_b(s)\ge0\),
\[
d_e(s)\le (1+\varepsilon)d_b(s)\le C d_b(s),
\]
so \cref{obj:ass:latent-stationary-overlap} holds.

It remains to collect the remaining requirements of \cref{obj:def:model-class}.
The horizon requirement \(1\le T\) is the hypothesis \(T\ge1\) of the present
lemma.  The observed alphabet of \(M_{Q,v}^T\) is the singleton \(\{0\}\), the
latent alphabet is the \(2(Q+1)\)-element set
\(\{0,\ldots,Q\}\times\{-1,+1\}\), and the action alphabet is \(\{0,1\}\); all
three are finite, and the action alphabet has exactly two elements, as the
class requires.  The parameter requirements \(0<t_0\) and \(0<\zeta\) are
hypotheses of the lemma, and they fix \(L=\exp(\zeta)\) and
\(\alpha=\exp(-1/t_0)\) as the overlap and contraction constants used
throughout.  Together with the seven assumptions verified above -- the Markov
kernel, sequential ignorability, stationary start, bounded rewards, policy
overlap, uniform contraction, and latent stationary overlap -- this is every
defining property of \cref{obj:def:model-class}, and therefore
\(M_{Q,v}^T\in\mathcal M_T(t_0,\zeta,C)\).
% lean: signedDepth_membership

\item It remains to compute the target value.  The two-point alternative label
enters the reward law only through its numeric sign: the positive label
contributes \(+1\), the negative label contributes \(-1\), and in the statement
this multiplier is denoted by \(v\in\{-1,+1\}\).  Under the target policy the
action is always one.  From the reward weights in
\cref{obj:def:signed-depth-family},
\[
g_e(0,h)
=
\sum_a e(a\mid0)\,E_K[Y_t\mid S_t=(0,h),A_t=a]
=
v\,c_0(t_0)\,u(h)\,\mathbf 1\{j(h)=Q\}.
\]
Using \cref{obj:def:target-value} and the already computed target stationary
law,
\[
\theta(K,e)
=
\sum_h
w_{j(h)}
\frac{1+u(h)\varepsilon}{2}\,
v\,c_0(t_0)\,u(h)\,\mathbf 1\{j(h)=Q\}.
\]
Only depth \(Q\) contributes, and summing over the two signs gives
\[
\theta(K,e)
=
v\,c_0(t_0)\,w_Q
\left[
\frac{1+\varepsilon}{2}
-
\frac{1-\varepsilon}{2}
\right]
=
v\,c_0(t_0)\,\varepsilon\,w_Q .
\]
This is the claimed value formula. % lean: signedDepth_targetValue
\end{enumerate}
\end{proof}
