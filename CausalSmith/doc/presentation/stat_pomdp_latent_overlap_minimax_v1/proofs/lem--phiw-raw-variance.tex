\begin{proof}[Proof of \cref{obj:lem:phiw-raw-variance}]
Let \(\mathbb P_{\mathrm{obs}}\) denote the observed trajectory law and let
\(\mathbb E_{\mathrm{obs}}\) denote expectation under this law.  Let
\(\mathbb P_{\mathrm{full}}\) denote the full trajectory law carried by \(M\),
and let \(\mathbb E_{\mathrm{full}}\) denote expectation under that law.  Put
\[
\mathscr T_k:=\{t\in\{0,\ldots,T-1\}: k\le t\},
\qquad
\mathfrak B_k:=
L^{k+1}\left(1+\frac{4}{L-1}\right)+\frac{4}{1-\alpha}.
\]
For \(t\in\mathscr T_k\), write
\[
\Gamma_t^{(k)}
:=
\prod_{\substack{0\le j\le T-1\\ j\le t,\ t-j\le k}}\rho_j,
\qquad
Z_t(k)=Y_t\Gamma_t^{(k)} .
\]
Since \(M\in\mathcal M_T(t_0,\zeta,C)\), \cref{obj:def:model-class} gives
\(t_0>0\), \(\zeta>0\), hence
\[
L=e^\zeta>1,
\qquad
0\le \alpha=e^{-1/t_0}<1.
\]
Also \(k\le \lfloor T/2\rfloor\) implies
\[
|\mathscr T_k|=T-k>0,
\qquad
\frac{T}{2}\le T-k.
\]
In particular \(\mathfrak B_k\ge0\).  The raw average is
\[
\frac{1}{T-k}\sum_{t=k}^{T-1} Z_t(k)
=
\frac{1}{T-k}\sum_{t\in\mathscr T_k}Z_t(k).
\]
% lean: variance_phiwRaw_le

\begin{enumerate}
\item
For each \(t\in\mathscr T_k\), \cref{obj:lem:phiw-window-second-moment} applies with the sequential-ignorability, policy-overlap, kernel-law, and bounded-reward clauses supplied by \cref{obj:def:model-class}, together with \(L=e^\zeta\ge 1\) and \(k\le t\).  It gives
\[
\mathbb E_{\mathrm{obs}}\bigl[Z_t(k)^2\bigr]
\le L^{k+1}.
\]
Using the standard identity
\[
\operatorname{Var}_{\mathrm{obs}}(Z_t(k))
=
\mathbb E_{\mathrm{obs}}\bigl[Z_t(k)^2\bigr]
-\bigl(\mathbb E_{\mathrm{obs}}Z_t(k)\bigr)^2,
\]
we obtain
\[
\operatorname{Var}_{\mathrm{obs}}(Z_t(k))
\le L^{k+1}.
\]
% lean: integral_sq_phiwScore_le

\item
Fix \(t\in\mathscr T_k\).  The aggregate covariance estimate in \cref{obj:lem:phiw-future-covariance-sum} applies to the same model class and the same future index set \(\mathscr T_k\).  Therefore
\[
\sum_{\substack{u\in\mathscr T_k\\ t<u}}
\left|\operatorname{Cov}_{\mathrm{obs}}\bigl(Z_t(k),Z_u(k)\bigr)\right|
\le
\frac{2L^{k+1}}{L-1}+\frac{2}{1-\alpha}.
\]
% lean: sum_abs_covariance_future_le

\item
Now expand the variance of the finite sum:
\[
\operatorname{Var}_{\mathrm{obs}}\!\left(\sum_{t\in\mathscr T_k}Z_t(k)\right)
=
\sum_{t\in\mathscr T_k}\sum_{u\in\mathscr T_k}
\operatorname{Cov}_{\mathrm{obs}}\bigl(Z_t(k),Z_u(k)\bigr).
\]
Bounding each covariance by its absolute value and then using covariance symmetry gives
\[
\operatorname{Var}_{\mathrm{obs}}\!\left(\sum_{t\in\mathscr T_k}Z_t(k)\right)
\le
\sum_{t\in\mathscr T_k}\operatorname{Var}_{\mathrm{obs}}(Z_t(k))
+
2\sum_{t\in\mathscr T_k}
\sum_{\substack{u\in\mathscr T_k\\ t<u}}
\left|\operatorname{Cov}_{\mathrm{obs}}\bigl(Z_t(k),Z_u(k)\bigr)\right|.
\]
Using the diagonal and future-sum bounds,
\[
\operatorname{Var}_{\mathrm{obs}}\!\left(\sum_{t\in\mathscr T_k}Z_t(k)\right)
\le
|\mathscr T_k|L^{k+1}
+
2|\mathscr T_k|
\left(
\frac{2L^{k+1}}{L-1}
+
\frac{2}{1-\alpha}
\right)
=
(T-k)\mathfrak B_k.
\]
% lean: sum_abs_symmetric_eq_diag_add_two_upper

\item
Finally,
\[
\operatorname{Var}_{\mathrm{obs}}\!\left(
\frac{1}{T-k}\sum_{t\in\mathscr T_k}Z_t(k)
\right)
=
\frac{1}{(T-k)^2}
\operatorname{Var}_{\mathrm{obs}}\!\left(\sum_{t\in\mathscr T_k}Z_t(k)\right)
\le
\frac{\mathfrak B_k}{T-k}.
\]
Since \(T/2\le T-k\), \(T>0\), and \(\mathfrak B_k\ge0\),
\[
\frac{\mathfrak B_k}{T-k}
\le
\frac{2\mathfrak B_k}{T}.
\]
Substituting the definition of \(\mathfrak B_k\) gives
\[
\operatorname{Var}_{\mathrm{obs}}\!\left(\frac{1}{T-k}\sum_{t=k}^{T-1} Z_t(k)\right)
\le
\frac{2}{T}\left\{
L^{k+1}\left(1+\frac{4}{L-1}\right)
+
\frac{4}{1-\alpha}
\right\}.
\]
% lean: variance_phiwRaw_le
\end{enumerate}
\end{proof}
