\begin{proof}[Proof of \cref{obj:lem:signed-depth-terminal-sign-invariant}]
Put \(\alpha=e^{-1/t_0}\) and \(L=e^\zeta\).  For a prefix
\(\omega=((\tilde a_i,\tilde y_i))_{i=1}^m\) and a depth
\(j\in\{0,\ldots,Q\}\), define
\[
\operatorname{Un}_j(\omega)
:=
\sum_{h:\,J(h)=j}\lambda_v(h;\omega),
\qquad
\operatorname{Sg}_j(\omega)
:=
\sum_{h:\,J(h)=j}U(h)\lambda_v(h;\omega),
\]
and
\[
A_j(\omega)
:=
L^{-(j-\min\{j,m\})}
\prod_{i=m-\min\{j,m\}+1}^{m}\mathbf 1\{\tilde a_i=1\},
\]
with the empty product equal to one.  Thus \(A_Q(o)=a^{(Q)}(o)\).

\begin{enumerate}
\item We first prove the depthwise invariant
\[
\operatorname{Sg}_j(\omega)
=
\varepsilon_C A_j(\omega)\operatorname{Un}_j(\omega)
\qquad
\text{for every }m,\ \omega,\ \text{and }j\in\{0,\ldots,Q\}.
\]
For \(m=0\), the initial mass at depth \(j\) and sign \(u\in\{-1,+1\}\) is
\[
w_j\frac{1+u\varepsilon_C L^{-j}}{2}.
\]
Summing over \(u=+1\) and \(u=-1\) gives
\[
\operatorname{Un}_j(\omega)=w_j,
\qquad
\operatorname{Sg}_j(\omega)
=
w_j\varepsilon_C L^{-j}
=
\varepsilon_C A_j(\omega)\operatorname{Un}_j(\omega),
\]
since \(A_j(\omega)=L^{-j}\) when the prefix is empty.

\item Suppose the invariant holds for every prefix of length \(m\), and let
\(\omega^+=((\tilde a_i,\tilde y_i))_{i=1}^{m+1}\).  Write \(\omega\) for its first \(m\) observations and \(a_+=\tilde a_{m+1}\).  At reset depth \(0\), direct summation of the two reset signs in \cref{obj:def:signed-depth-family} gives, for each previous hidden state \(h\),
\[
\sum_{u'\in\{-1,+1\}}
u'\,p\{J'=0,U'=u'\mid h,a_+\}
=
\varepsilon_C
\sum_{u'\in\{-1,+1\}}
p\{J'=0,U'=u'\mid h,a_+\}.
\]
After multiplying by the previous unnormalized filter mass, the behavior-policy weight, and the reward-emission weight, and then summing over \(h\), this yields
\[
\operatorname{Sg}_0(\omega^+)
=
\varepsilon_C\operatorname{Un}_0(\omega^+).
\]
Because \(A_0(\omega^+)=1\), the invariant holds at depth \(0\).

\item For a positive target depth, write it as \(j+1\) with \(j<Q\).  The last observation of \(\omega^+\) is \((a_+,\tilde y_{m+1})\), and the preceding prefix \(\omega\) is the restriction of \(\omega^+\) to its first \(m\) coordinates.  Let \(\Gamma_{a_+}((\delta,u),(\delta',u'))\) be the hidden-transition weight from \((\delta,u)\) to \((\delta',u')\) under action \(a_+\), for \(\delta,\delta'\in\{0,\ldots,Q\}\) and \(u,u'\in\{-1,+1\}\).  The raw filter recursion gives
\[
\operatorname{Un}_{j+1}(\omega^+)
=
b(a_+)\sum_{\delta=0}^Q\sum_{u\in\{-1,+1\}}
\lambda_v((\delta,u);\omega)
\frac{1+\tilde y_{m+1}vc_0u\mathbf 1\{\delta=Q\}}{2}
\sum_{u'\in\{-1,+1\}}
\Gamma_{a_+}((\delta,u),(j+1,u')),
\]
where \(b(a_+)\) is the behavior-policy probability of the observed action and \(c_0=(1-\alpha)/4\).  By the transition weights in \cref{obj:def:signed-depth-family}, reaching the positive depth \(j+1\) is possible exactly by advancing from depth \(j\), and
\[
\sum_{u'\in\{-1,+1\}}
\Gamma_{a_+}((\delta,u),(j+1,u'))
=
\alpha\mathbf 1\{\delta=j\}.
\]
Because \(j<Q\), the reward factor on the surviving terms is \(1/2\).  Hence
\[
\operatorname{Un}_{j+1}(\omega^+)
=
b(a_+)\,\alpha\,\frac12\,
\operatorname{Un}_j(\omega).
\]

\item With the same notation, the signed advancing aggregate is obtained by inserting the new sign in the inner sum:
\[
\operatorname{Sg}_{j+1}(\omega^+)
=
b(a_+)\sum_{\delta=0}^Q\sum_{u\in\{-1,+1\}}
\lambda_v((\delta,u);\omega)
\frac{1+\tilde y_{m+1}vc_0u\mathbf 1\{\delta=Q\}}{2}
\sum_{u'\in\{-1,+1\}}
{u'}\,\Gamma_{a_+}((\delta,u),(j+1,u')) .
\]
Again only \(\delta=j\) survives.  For that depth, the transition weights in \cref{obj:def:signed-depth-family} give
\[
\sum_{u'\in\{-1,+1\}}
{u'}\,\Gamma_{a_+}((\delta,u),(j+1,u'))
=
\alpha\mathbf 1\{\delta=j\}\mathbf 1\{a_+=1\}u.
\]
Indeed, action \(1\) preserves the old sign on an advancing transition, while action \(0\) assigns equal total advancing weight to the two possible new signs.  Using once more that the reward factor is \(1/2\) at depth \(j<Q\), we obtain
\[
\operatorname{Sg}_{j+1}(\omega^+)
=
b(a_+)\,\alpha\,\frac12\,
\mathbf 1\{a_+=1\}\operatorname{Sg}_j(\omega).
\]

\item The action-history factor has the corresponding one-step recursion.  From the definition of \(A_j\),
\[
A_{j+1}(\omega^+)
=
L^{-((j+1)-\min\{j+1,m+1\})}
\prod_{i=m+2-\min\{j+1,m+1\}}^{m+1}
\mathbf 1\{\tilde a_i=1\}.
\]
The identities
\[
\min\{j+1,m+1\}=\min\{j,m\}+1,
\qquad
(j+1)-\min\{j+1,m+1\}=j-\min\{j,m\},
\]
and
\[
m+2-\min\{j+1,m+1\}=m+1-\min\{j,m\}
\]
show that the displayed product is the product defining \(A_j(\omega)\), followed by the last factor \(\mathbf 1\{a_+=1\}\).  Thus
\[
A_{j+1}(\omega^+)
=
A_j(\omega)\mathbf 1\{a_+=1\},
\]
with the same empty-product convention as before.

\item Applying the induction hypothesis at depth \(j\) to the preceding prefix \(\omega\) gives
\[
\operatorname{Sg}_{j+1}(\omega^+)
=
b(a_+)\,\alpha\,\frac12\,
\mathbf 1\{a_+=1\}\,
\varepsilon_C A_j(\omega)\operatorname{Un}_j(\omega).
\]
The unsigned recursion and the action-factor recursion give
\[
\varepsilon_C A_{j+1}(\omega^+)
\operatorname{Un}_{j+1}(\omega^+)
=
\varepsilon_C A_j(\omega)\mathbf 1\{a_+=1\}\,
 b(a_+)\,\alpha\,\frac12\operatorname{Un}_j(\omega),
\]
which is the same quantity.  Equivalently, splitting on the last action, when \(a_+=1\) the common factor is \(b(a_+)\alpha/2\) times the induction hypothesis, and when \(a_+=0\) both sides are zero.  This completes the induction.

\item Finally take \(j=Q\) and \(\omega=o\).  Reindexing the hidden-state sum by its depth and sign coordinates gives
\[
\sum_{h:\,J(h)=Q}\lambda_v(h;o)=\operatorname{Un}_Q(o),
\qquad
\sum_{h:\,J(h)=Q}U(h)\lambda_v(h;o)=\operatorname{Sg}_Q(o).
\]
The depthwise invariant at \(j=Q\), together with \(A_Q(o)=a^{(Q)}(o)\), therefore gives
\[
\sum_{h:\,J(h)=Q} U(h)\lambda_v(h;o)
=
\varepsilon_C\,a^{(Q)}(o)
\sum_{h:\,J(h)=Q}\lambda_v(h;o),
\]
as claimed.
\end{enumerate}
\end{proof}
