\begin{proof}[Proof of \cref{obj:lem:phiw-future-covariance-sum}]
Fix \(t\in\mathscr T_k\). By \cref{obj:def:model-class}, \(0<\zeta\) and \(0<t_0\), so
\[
L=e^\zeta>1,
\qquad
0\le \alpha=e^{-1/t_0}<1 .
\]
The same membership supplies the model restrictions used by the covariance estimates below.

\begin{enumerate}
\item Define the future index set and its two lag classes by
\[
\mathscr U_t:=\{u\in\mathscr T_k:t<u\},
\qquad
\lambda_{t,u}:=u-t\quad (u\in\mathscr U_t),
\]
and
\[
\mathscr N_t:=\{u\in\mathscr U_t:\lambda_{t,u}\le k\},
\qquad
\mathscr G_t:=\{u\in\mathscr U_t:k+1\le \lambda_{t,u}\}.
\]
For each \(u\in\mathscr U_t\), the lag \(\lambda_{t,u}\) is a positive integer, hence exactly one of
\(\lambda_{t,u}\le k\) and \(k+1\le\lambda_{t,u}\) holds. Thus
\[
\mathscr U_t=\mathscr N_t\,\dot\cup\,\mathscr G_t .
\]
Writing
\[
\Gamma_t(u):=
\bigl|\operatorname{Cov}_{\mathrm{obs}}\bigl(Z_t(k),Z_u(k)\bigr)\bigr|,
\qquad u\in\mathscr U_t,
\]
we get
\[
\sum_{u\in\mathscr U_t}\Gamma_t(u)
=
\sum_{u\in\mathscr N_t}\Gamma_t(u)
+
\sum_{u\in\mathscr G_t}\Gamma_t(u).
\]
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.sum_abs_covariance_future_le

\item For \(u\in\mathscr N_t\), the lag identity is \(u=t+\lambda_{t,u}\) and
\(1\le\lambda_{t,u}\le k\). Since \(k\le t\) and \(L\ge1\), \cref{obj:lem:phiw-overlapping-window-covariance} gives
\[
\Gamma_t(u)\le 2L^{k+1-\lambda_{t,u}}.
\]
Therefore
\[
\sum_{u\in\mathscr N_t}\Gamma_t(u)
\le
2L\sum_{u\in\mathscr N_t}L^{k-\lambda_{t,u}}.
\]
The map
\[
j_{\mathrm{near}}(u):=k-\lambda_{t,u},
\qquad u\in\mathscr N_t,
\]
is injective and takes its values in \(\{0,\ldots,k-1\}\). Since \(L\ge0\),
\[
\sum_{u\in\mathscr N_t}L^{k-\lambda_{t,u}}
\le
\sum_{0\le j<k}L^j .
\]
For \(L>1\),
\[
\sum_{0\le j<k}L^j
=
\frac{L^k-1}{L-1}
\le
\frac{L^k}{L-1}.
\]
Combining the last three displays yields
\[
\sum_{u\in\mathscr N_t}\Gamma_t(u)
\le
\frac{2L^{k+1}}{L-1}.
\]
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.geom_partial_le_pow_div

\item For \(u\in\mathscr G_t\), the lag identity is \(u=t+\lambda_{t,u}\) and
\(k+1\le\lambda_{t,u}\). Hence \cref{obj:lem:phiw-separated-window-covariance} gives
\[
\Gamma_t(u)
\le
2\alpha^{\lambda_{t,u}-k-1}.
\]
Thus
\[
\sum_{u\in\mathscr G_t}\Gamma_t(u)
\le
2\sum_{u\in\mathscr G_t}\alpha^{\lambda_{t,u}-k-1}.
\]
The map
\[
j_{\mathrm{sep}}(u):=\lambda_{t,u}-k-1,
\qquad u\in\mathscr G_t,
\]
is injective and takes its values in \(\{0,\ldots,T-1\}\). Since \(\alpha\ge0\),
\[
\sum_{u\in\mathscr G_t}\alpha^{\lambda_{t,u}-k-1}
\le
\sum_{0\le j<T}\alpha^j .
\]
Because \(0\le\alpha<1\),
\[
\sum_{0\le j<T}\alpha^j
\le
\sum_{j=0}^{\infty}\alpha^j
=
\frac{1}{1-\alpha}.
\]
Consequently,
\[
\sum_{u\in\mathscr G_t}\Gamma_t(u)
\le
\frac{2}{1-\alpha}.
\]
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.geom_partial_le_inv

\item Adding the two bounds over the disjoint decomposition of \(\mathscr U_t\) gives
\[
\sum_{\substack{u\in\mathscr T_k\\ t<u}}
\bigl|\operatorname{Cov}_{\mathrm{obs}}\bigl(Z_t(k),Z_u(k)\bigr)\bigr|
=
\sum_{u\in\mathscr U_t}\Gamma_t(u)
\le
\frac{2L^{k+1}}{L-1}+\frac{2}{1-\alpha}.
\]
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.sum_abs_covariance_future_le
\end{enumerate}
\end{proof}
