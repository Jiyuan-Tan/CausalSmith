\begin{proof}[Proof of \cref{obj:lem:insulin-bias-certificate}]
Fix \(T\in\mathbb N\). Write \(P_b^{(T)}\) and \(P_e^{(T)}\) for the policy-induced transition matrices on \(\mathsf S_{\mathrm{grid}}\) obtained from \cref{obj:lem:insulin-policy-contraction} by taking \(p=b\) and \(p=e\), respectively, and write \(d_b^{(T)}\) and \(d_e^{(T)}\) for their selected stationary laws.

\begin{enumerate}
\item The behavior and target kernels both satisfy the one-half contraction estimate
\[
\|\nu P_b^{(T)}-\nu'P_b^{(T)}\|_1
\le {1\over2}\|\nu-\nu'\|_1,
\qquad
\|\nu P_e^{(T)}-\nu'P_e^{(T)}\|_1
\le {1\over2}\|\nu-\nu'\|_1
\]
for all probability vectors \(\nu,\nu'\) on \(\mathsf S_{\mathrm{grid}}\). This is exactly the \(\ell^1\)-form supplied by \cref{obj:lem:insulin-policy-contraction}, instantiated first at \(p=b\) and then at \(p=e\). The corresponding selected laws are probability vectors and satisfy
\[
d_b^{(T)}P_b^{(T)}=d_b^{(T)},
\qquad
d_e^{(T)}P_e^{(T)}=d_e^{(T)},
\]
by \cref{obj:lem:insulin-stationary-law}.

\item Let
\[
r_{\mathrm{ins}}(s)=
\begin{cases}
-1,&g\le70,\\
-2/3,&g>150,\\
-1/3,&70<g\le80\ \text{or}\ 120<g\le150,\\
0,&\text{otherwise},
\end{cases}
\qquad s=(g,d_1,d_2,x_1,x_2,a_1).
\]
For \(\star\in\{\mathrm b,\mathrm e\}\), with \(P_\star^{(T)}\) denoting \(P_b^{(T)}\) or \(P_e^{(T)}\), let \(\mathfrak A_\star\) be the exact rational interval matrix recorded by the finite-grid certificate. Its entrywise enclosure is
\[
P_\star^{(T)}(s,s')\in \mathfrak A_\star(s,s'),
\qquad s,s'\in\mathsf S_{\mathrm{grid}}.
\]
The same certificate supplies an exact rational initial probability row \(\mu_{\star,0}\), interval
rows \(\mathfrak T_{\star,0}\) and \(\mathfrak T_{\star,1}\), and an exact chunked recurrence from the
first interval row to the second. In the notation of
\cref{obj:lem:certified-stationary-reward-enclosure}, its rational inclusions verify
\[
[\mu_{\star,0}(s),\mu_{\star,0}(s)]
\subseteq
\mathfrak T_{\star,0}(s),
\qquad s\in\mathsf S_{\mathrm{grid}}.
\]
For every output state, the recorded blocks partition \(\mathsf S_{\mathrm{grid}}\), every block has at
most \(40\) states, and the block and partial-sum intervals satisfy exactly the inclusion chain in
\cref{obj:lem:certified-stationary-reward-enclosure}. Because the certificate has zero approximation
steps, this verified recurrence links its initial interval row directly to its successor row. Passing
from the chunked certificate to the finite recurrence used by that lemma preserves both rows and all
the rational inclusions. For the reward certificate, put
\[
\mathfrak r_{\mathrm{ins}}(s)=[r_{\mathrm{ins}}(s),r_{\mathrm{ins}}(s)].
\]
Since \(|r_{\mathrm{ins}}(s)|\le1\) for every grid state, the reward bound in \cref{obj:lem:certified-stationary-reward-enclosure} is \(B=1\). Define
\[
\delta_\star
=
\sum_{s\in\mathsf S_{\mathrm{grid}}}
\bigl|
\mathfrak T_{\star,0}(s)\ominus
\mathfrak T_{\star,1}(s)
\bigr|_{\max},
\qquad
\mathfrak E_\star
=
\sum_{s\in\mathsf S_{\mathrm{grid}}}
\mathfrak T_{\star,0}(s)\cdot \mathfrak r_{\mathrm{ins}}(s) .
\]
After that conversion, the terminal row is \(\mathfrak T_{\star,0}\) and the successor row is
\(\mathfrak T_{\star,1}\), so the residual and stationary-reward interval are
\[
\delta_\star
=
\sum_{s\in\mathsf S_{\mathrm{grid}}}
\bigl|
\mathfrak T_{\star,0}(s)\ominus
\mathfrak T_{\star,1}(s)
\bigr|_{\max},
\qquad
\mathfrak J_\star
=
\left[
\ell(\mathfrak E_\star)-{\delta_\star\over 1-1/2},
u(\mathfrak E_\star)+{\delta_\star\over 1-1/2}
\right].
\]
The initial-row, recurrence, reward-bound, contraction, and stationarity hypotheses of \cref{obj:lem:certified-stationary-reward-enclosure} are therefore satisfied for both policies after passing each chunked recurrence certificate to its associated finite recurrence certificate. Under this conversion, the terminal interval row is \(\mathfrak T_{\star,0}\), the successor interval row is \(\mathfrak T_{\star,1}\), and the stationary-reward interval produced by \cref{obj:lem:certified-stationary-reward-enclosure} is exactly
\[
\mathfrak J_\star
=
\left[
\ell(\mathfrak E_\star)-{\delta_\star\over 1-1/2},
u(\mathfrak E_\star)+{\delta_\star\over 1-1/2}
\right],
\]
which is the chunked interval \(\mathfrak J_\star\) used here. Hence
\[
\sum_{s\in\mathsf S_{\mathrm{grid}}} d_b^{(T)}(s)r_{\mathrm{ins}}(s)\in\mathfrak J_{\mathrm b},
\qquad
\sum_{s\in\mathsf S_{\mathrm{grid}}} d_e^{(T)}(s)r_{\mathrm{ins}}(s)\in\mathfrak J_{\mathrm e}.
\] % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.insulinChunkedStationaryRewardInterval_eq

\item Define the subtraction interval
\[
\mathfrak B:=\mathfrak J_{\mathrm b}\ominus\mathfrak J_{\mathrm e}
=
\bigl[
\ell(\mathfrak J_{\mathrm b})-u(\mathfrak J_{\mathrm e}),
u(\mathfrak J_{\mathrm b})-\ell(\mathfrak J_{\mathrm e})
\bigr].
\]
Interval subtraction is sound, so the two inclusions from the previous step give
\[
\sum_s d_b^{(T)}(s)r_{\mathrm{ins}}(s)
-
\sum_s d_e^{(T)}(s)r_{\mathrm{ins}}(s)
\in \mathfrak B .
\]

\item The reward-regression identities in
\cref{obj:lem:insulin-immediate-reward-regression,obj:lem:insulin-target-reward-regression}
give, for every \(s\in\mathsf S_{\mathrm{grid}}\),
\[
\sum_{a\in\{0,1\}}
b(a\mid x(s))\,\rho(x(s),a)
\int p_1\,K_T(dp\mid s,a)
=
r_{\mathrm{ins}}(s),
\]
and
\[
\sum_{a\in\{0,1\}}
e(a\mid x(s))
\int p_1\,K_T(dp\mid s,a)
=
r_{\mathrm{ins}}(s).
\]
The finite-grid embedding preserves the policy-induced kernels entry by entry:
\[
P_b^{(T)}(s,s')=
\sum_a b(a\mid x(s))K_T(\mathbb R\times\{s'\}\mid s,a),
\qquad
P_e^{(T)}(s,s')=
\sum_a e(a\mid x(s))K_T(\mathbb R\times\{s'\}\mid s,a).
\]
Consequently the stationary laws in the interval certificate are exactly the behavior- and
target-policy stationary laws entering the two regression identities. Combining these kernel
identifications, the two regression identities, and \cref{obj:def:target-value} yields
\[
\Delta_{\mathrm{imm}}(T)
=
\sum_s d_b^{(T)}(s)r_{\mathrm{ins}}(s)
-
\sum_s d_e^{(T)}(s)r_{\mathrm{ins}}(s).
\]
Therefore
\[
\Delta_{\mathrm{imm}}(T)\in\mathfrak B .
\]

\item By the exact rational interval evaluation in \cref{obj:lem:insulin-bias-interval-evaluation}, the endpoints of the subtraction interval \(\mathfrak B\) satisfy
\[
-\frac{6}{1000}<\ell(\mathfrak B),
\qquad
u(\mathfrak B)<-\frac{58}{10000}.
\]
Since \(\Delta_{\mathrm{imm}}(T)\in\mathfrak B\), we have
\[
\ell(\mathfrak B)
\le
\Delta_{\mathrm{imm}}(T)
\le
u(\mathfrak B).
\]
The last two displays imply
\[
-\frac{6}{1000}
<
\Delta_{\mathrm{imm}}(T)
<
-\frac{58}{10000},
\]
as claimed. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.insulinBiasInterval_bounds
\end{enumerate}
\end{proof}
