\begin{proof}[Proof of \cref{obj:lem:uniform-parametric-floor}]
\begin{enumerate}
\item Fix \(T\ge 1\) and put
\[
\eta_T:=\frac{1}{4\sqrt T}.
\]
Since \(T\ge1\), \(0\le \eta_T\le 1/4\).  For each sign
\(\sigma\in\{-1,+1\}\), define a singleton-state experiment with
\(\mathcal X=\{x_\star\}\), \(\mathcal H=\{h_\star\}\), fair policies
\[
b_\sigma(a\mid x_\star)=e_\sigma(a\mid x_\star)=\frac12,
\qquad a\in\{0,1\},
\]
and reward-transition kernel
\[
\Pr_\sigma\{Y_t=y,\ X_{t+1}=x_\star,\ H_{t+1}=h_\star
       \mid X_t=x_\star,H_t=h_\star,A_t=a\}
=
\frac{1+\sigma y\eta_T}{2},
\qquad y\in\{-1,+1\}.
\]
Let \(M_T^{-}\) be the experiment with \(\sigma=-1\), and \(M_T^{+}\) the
experiment with \(\sigma=+1\).  The weights above are nonnegative and sum to
one.  The hypotheses give the positive parameter-scale requirements
\(0<t_0\) and \(0<\zeta\) appearing in \cref{obj:def:model-class}.  The
Markov-kernel, sequential-ignorability, stationary-start, and bounded-reward
requirements in
\cref{obj:ass:pomdp-kernel,obj:ass:sequential-ignorability,obj:ass:stationary-start,obj:ass:bounded-reward}
hold by this generated singleton construction.  The policy-overlap condition
in \cref{obj:ass:policy-overlap} holds because \(b_\sigma=e_\sigma\) and
\(\exp(\zeta)\ge1\).  For either policy \(p\in\{b_\sigma,e_\sigma\}\),
\[
\nu P_p=\nu'P_p=\delta_{(x_\star,h_\star)}
\]
for all probability laws \(\nu,\nu'\) on the joint state space.  Put
\[
\alpha_{t_0}:=\exp(-1/t_0).
\]
Since \(t_0>0\), \(\alpha_{t_0}>0\), and therefore
\(\alpha_{t_0}\|\nu-\nu'\|_{\mathrm{TV}}\ge0\).  Thus
\[
\|\nu P_p-\nu'P_p\|_{\mathrm{TV}}
=0
\le
\alpha_{t_0}\|\nu-\nu'\|_{\mathrm{TV}},
\]
which is the uniform-contraction requirement in
\cref{obj:ass:uniform-contraction}.  Finally, the latent stationary overlap condition in
\cref{obj:ass:latent-stationary-overlap} is immediate:
\(d_{e_\sigma}=d_{b_\sigma}=\delta_{(x_\star,h_\star)}\), and hence
\(d_{e_\sigma}(s)\le C\,d_{b_\sigma}(s)\) for every state whenever \(C\ge1\).
Thus both alternatives belong to \(\mathcal M_T(t_0,\zeta,C)\) as defined in
\cref{obj:def:model-class}.  The same displayed definition of the policies
also gives
\[
b_{M_T^-}=e_{M_T^-},\qquad
b_{M_T^+}=e_{M_T^+},\qquad
b_{M_T^-}=b_{M_T^+},\qquad
e_{M_T^-}=e_{M_T^+}.
\]
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.uniform_parametric_floor_certificates

\item The singleton alternatives just constructed are the two alternatives covered
by \cref{obj:lem:parametric-pair-target-value}.  With the sign convention of
Step 1, that result gives
\[
\theta_-
:=\theta(K_{M_T^-},e_{M_T^-})
=-\eta_T,
\qquad
\theta_+
:=\theta(K_{M_T^+},e_{M_T^+})
=\eta_T.
\]
Consequently
\[
2\eta_T
=
|\theta_{-}-\theta_{+}|.
\]
For the divergence calculation, let \(\mathsf r\in\{0,1\}\) be the reward code
and set
\[
\mathsf y(\mathsf r):=2\mathsf r-1,
\qquad
\mathsf y(0)=-1,
\qquad
\mathsf y(1)=+1.
\]
Define the one-epoch coded observed law
\(\Lambda_T^\sigma\in\Delta(\{x_\star\}\times\{0,1\}\times\{0,1\})\) by
\[
\Lambda_T^\sigma\{(x_\star,a,\mathsf r)\}
:=
\frac12\,\frac{1+\sigma\mathsf y(\mathsf r)\eta_T}{2},
\qquad a,\mathsf r\in\{0,1\}.
\]
Since \(0\le\eta_T\le1/4\), every displayed cell has positive mass.  Thus
\(\Lambda_T^-\ll\Lambda_T^+\), and the log likelihood ratio is integrable.  If
\(\Xi_T^-\) and \(\Xi_T^+\) denote the reward-code marginals of
\(\Lambda_T^-\) and \(\Lambda_T^+\), then
\[
\Xi_T^-\{1\}=\frac{1-\eta_T}{2},
\qquad
\Xi_T^+\{1\}=\frac{1+\eta_T}{2}.
\]
The fair action factor is common to the two signs, so the product rule for KL
with a common second factor gives
\[
\operatorname{KL}(\Lambda_T^-\|\Lambda_T^+)
=
\operatorname{KL}(\Xi_T^-\|\Xi_T^+).
\]
Writing out the two Bernoulli masses gives
\[
\operatorname{KL}(\Xi_T^-\|\Xi_T^+)
=
\frac{1-\eta_T}{2}\log\frac{1-\eta_T}{1+\eta_T}
+
\frac{1+\eta_T}{2}\log\frac{1+\eta_T}{1-\eta_T}
=
\eta_T\log\frac{1+\eta_T}{1-\eta_T}.
\]
For \(0\le u\le1/4\),
\[
\log\frac{1+u}{1-u}
=
\log(1+u)-\log(1-u)
\le
u+\frac{u}{1-u}
\le 4u,
\]
using \(\log(1+u)\le u\) and \(-\log(1-u)\le u/(1-u)\).  Applying this with
\(u=\eta_T\) yields the one-epoch bound
\[
\operatorname{KL}(\Lambda_T^-\|\Lambda_T^+)
\le 4\eta_T^2.
\]
Let
\[
\Pi_{\sigma,T}^{\mathrm{code}}
:=
\bigotimes_{i=1}^T\Lambda_T^\sigma
\quad\text{on}\quad
(\{x_\star\}\times\{0,1\}\times\{0,1\})^T.
\]
The generated finite trajectory construction has this product as its coded
observed-word law.  By finite-product KL tensorization, with the absolute
continuity and integrability supplied by the positive one-epoch masses,
\[
\operatorname{KL}(\Pi_{-,T}^{\mathrm{code}}\|
\Pi_{+,T}^{\mathrm{code}})
\le
T\operatorname{KL}(\Lambda_T^-\|\Lambda_T^+)
\le
4T\eta_T^2.
\]
Because \(T\ge1\), \(\sqrt T>0\), and \(\eta_T=1/(4\sqrt T)\),
\[
4T\eta_T^2=\frac14.
\]
Finally, let \(\mathfrak d_T\) be the decoding map that replaces each reward
code \(\mathsf r\) by \(\mathsf y(\mathsf r)\) and keeps the observed state and
action coordinates.  The observed laws in the statement are the pushforwards
\[
\mathbb P_{M_T^-}^{\mathrm{obs}}
=(\mathfrak d_T)_\#\Pi_{-,T}^{\mathrm{code}},
\qquad
\mathbb P_{M_T^+}^{\mathrm{obs}}
=(\mathfrak d_T)_\#\Pi_{+,T}^{\mathrm{code}}.
\]
KL divergence is monotone under measurable pushforward, so the preceding coded
product bound gives
\[
\operatorname{KL}\!\left(
\mathbb P_{M_T^-}^{\mathrm{obs}}
\middle\|
\mathbb P_{M_T^+}^{\mathrm{obs}}
\right)
\le
\operatorname{KL}(\Pi_{-,T}^{\mathrm{code}}\|
\Pi_{+,T}^{\mathrm{code}})
\le
\frac14.
\]
This is the observed KL budget in \cref{obj:lem:parametric-pair-observed-kl}.
The positive one-epoch masses give
\(\Pi_{-,T}^{\mathrm{code}}\ll\Pi_{+,T}^{\mathrm{code}}\), hence also
\(\mathbb P_{M_T^-}^{\mathrm{obs}}\ll\mathbb P_{M_T^+}^{\mathrm{obs}}\), and the
displayed upper bound gives finite KL divergence. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.parametricPair_observed_klDiv_le

\item Let \(\widehat\theta\) be any observable-data estimator satisfying the
measurability and two integrability assumptions in the statement, and write
\[
\varphi(o):=
\widehat\theta(1,b_{M_T^-},e_{M_T^-};o)
=
\widehat\theta(1,b_{M_T^+},e_{M_T^+};o),
\]
where the equality uses the common policies from Step 1.  The two-point risk
inequality applied here is \cref{obj:lem:two-point-risk-floor}, used in its
raw-estimator form for the estimator \(\widehat\theta\).  Its structural
hypotheses are available here: the observed alphabets both have one state and
the behavior and target policies are common by Step 1.  Step 2 supplies
\[
2\eta_T\le |\theta_- - \theta_+|,
\qquad
\mathbb P_{M_T^-}^{\mathrm{obs}}\ll
\mathbb P_{M_T^+}^{\mathrm{obs}},
\qquad
\operatorname{KL}\!\left(
\mathbb P_{M_T^-}^{\mathrm{obs}}
\middle\|
\mathbb P_{M_T^+}^{\mathrm{obs}}
\right)
\le \frac14,
\]
together with finite KL divergence.  The present step supplies the required
measurability of \(\widehat\theta(n_X,b,e;\cdot)\) for every input
\((n_X,b,e)\), and the two squared-error integrability assumptions for the two
displayed risks.

Apply \cref{obj:lem:two-point-risk-floor} directly to the raw estimator
\(\widehat\theta\), the two observed laws
\[
\mathbb P_{M_T^-}^{\mathrm{obs}},
\qquad
\mathbb P_{M_T^+}^{\mathrm{obs}},
\]
the target values
\[
\theta_-=
\theta(K_{M_T^-},e_{M_T^-}),
\qquad
\theta_+=
\theta(K_{M_T^+},e_{M_T^+}),
\]
and the numerical choices
\[
s_\star=\eta_T,
\qquad
\chi_{\mathrm{KL}}=\frac14.
\]
The separation condition is supplied by Step 2, and the absolute-continuity and
finite-KL hypotheses are supplied by the final display of Step 2.  Hence
\[
\frac{\eta_T^2\exp(-1/4)}{4}
\le
\max\left\{
\mathbb E_{M_T^-}^{\mathrm{obs}}
\!\left[
\{\varphi(\mathsf O_T)-\theta_-\}^2
\right],
\mathbb E_{M_T^+}^{\mathrm{obs}}
\!\left[
\{\varphi(\mathsf O_T)-\theta_+\}^2
\right]
\right\}.
\]
Finally,
\[
\frac{\eta_T^2\exp(-1/4)}{4}
=
\frac{\exp(-1/4)}{64T},
\]
which is exactly the pointwise two-experiment floor in the statement.
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.two_point_floor_explicit

\item It remains to record the minimax lower bound.  For the supplied radius \(C\),
the two singleton alternatives with the same kernels and policies satisfy the
hypotheses of the minimax conclusion in \cref{obj:lem:two-point-risk-floor}:
Step 1 gives their common observable cardinality, common behavior policy,
common target policy, and membership in \(\mathcal M_T(t_0,\zeta,C)\); Step 2
gives
\[
2\eta_T
\le
|\theta_- - \theta_+|,
\qquad
\mathbb P_{M_T^-}^{\mathrm{obs}}\ll
\mathbb P_{M_T^+}^{\mathrm{obs}},
\qquad
\operatorname{KL}\!\left(
\mathbb P_{M_T^-}^{\mathrm{obs}}
\middle\|
\mathbb P_{M_T^+}^{\mathrm{obs}}
\right)
\le
\frac14,
\]
with finite KL divergence.  Applying \cref{obj:lem:two-point-risk-floor} with
\[
s_\star=\eta_T,
\qquad
\chi_{\mathrm{KL}}=\frac14
\]
yields
\[
\frac{\eta_T^2\exp(-1/4)}{4}
\le
R_T(t_0,\zeta,C),
\]
where \(R_T(t_0,\zeta,C)\) is the observable minimax risk of
\cref{obj:def:minimax-risk}.  Since
\[
\frac{\eta_T^2\exp(-1/4)}{4}
=
\frac{\exp(-1/4)}{64T},
\]
the desired minimax floor follows.  Together with the membership and policy
identities from Step 1 and the pointwise inequality from Step 3, this proves
all four asserted conclusions.
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.uniform_parametric_floor
\end{enumerate}
\end{proof}
