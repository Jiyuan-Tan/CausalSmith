\begin{proof}[Proof of \cref{obj:thm:uniform-overlap-frontier}]
Write
\[
\alpha=\exp(-1/t_0),\qquad L=\exp(\zeta),\qquad
\beta=\frac{2}{2+t_0\zeta}.
\]
Then \(0<\alpha<1\), \(1<L\), and \(0<\beta<1\).

\emph{Step 1. Adaptive-depth calibration.}
Set
\[
\mathcal D=2\log(1/\alpha)+\log L=\frac{2}{t_0}+\zeta,
\qquad
q=q_C,\qquad \xi=Tq^2 .
\]
There are \(c_{\mathrm{rad}}>0\) and \(T_{\mathrm{rad}}\ge1\), depending only on
\(t_0,\zeta\), such that, for all \(T\ge T_{\mathrm{rad}}\) and all \(C\ge1\),
\[
q_C^2\alpha^{2\kappa_T^{\mathrm{rad}}(C)}
+\frac{L^{\kappa_T^{\mathrm{rad}}(C)}}{T}
\le
c_{\mathrm{rad}}
\left\{
T^{-1}+T^{-\beta}q_C^{2(1-\beta)}
\right\}.
\]
Indeed, since \(\log T=o(T)\), enlarge \(T_{\mathrm{rad}}\) so that
\[
|\log T|\le \frac{\mathcal D}{2}T
\qquad (T\ge T_{\mathrm{rad}}).
\]
If \(\xi\le1\), then \(\kappa_T^{\mathrm{rad}}(C)=0\) by
\cref{obj:def:radius-adaptive-history-depth}, and \(q_C^2\le T^{-1}\), hence
\[
q_C^2+\frac1T
\le
2T^{-1}
\le
2\left\{
T^{-1}+T^{-\beta}q_C^{2(1-\beta)}
\right\}.
\]
If \(\xi>1\), then \(q_C>0\), \(0<\xi\le T\), and the same logarithmic bound gives
\[
\left\lfloor\frac{\log \xi}{\mathcal D}\right\rfloor\le \frac T2 .
\]
Thus \(\kappa_T^{\mathrm{rad}}(C)=\lfloor\log\xi/\mathcal D\rfloor\). The floor
bounds
\[
\frac{\log\xi}{\mathcal D}-1
<
\kappa_T^{\mathrm{rad}}(C)
\le
\frac{\log\xi}{\mathcal D}
\]
give
\[
\alpha^{2\kappa_T^{\mathrm{rad}}(C)}
\le
e^{2/t_0}\xi^{-\beta},
\qquad
L^{\kappa_T^{\mathrm{rad}}(C)}
\le
\xi^{1-\beta}.
\]
Consequently,
\[
q_C^2\alpha^{2\kappa_T^{\mathrm{rad}}(C)}
+\frac{L^{\kappa_T^{\mathrm{rad}}(C)}}{T}
\le
\{e^{2/t_0}+1\}\,T^{-\beta}q_C^{2(1-\beta)}.
\]
Taking \(c_{\mathrm{rad}}=\max\{2,e^{2/t_0}+1\}\) proves the calibration. % lean: radiusAdaptiveDepth_rate

\emph{Step 2. Upper bound.}
Let
\[
A=\max\left\{
4,\,
2L\left(1+\frac4{L-1}\right),\,
\frac8{1-\alpha}
\right\},
\qquad
c_{\mathrm{att}}=A(c_{\mathrm{rad}}+1).
\]
For \(T\ge T_{\mathrm{rad}}\), put \(k=\kappa_T^{\mathrm{rad}}(C)\).  By
\cref{obj:def:radius-adaptive-history-depth}, \(k\le T/2\), and
\cref{obj:lem:radius-sensitive-phiw} gives, uniformly over
\(M\in\mathcal M_T(t_0,\zeta,C)\),
\[
\mathbb E_M[(\widehat\theta_k-\theta(K,e))^2]
\le
4q_C^2\alpha^{2k}
+\frac2T\left\{
L^{k+1}\left(1+\frac4{L-1}\right)+\frac4{1-\alpha}
\right\}.
\]
By the definition of \(A\),
\[
4q_C^2\alpha^{2k}
+\frac2T\left\{
L^{k+1}\left(1+\frac4{L-1}\right)+\frac4{1-\alpha}
\right\}
\le
A\left\{
q_C^2\alpha^{2k}+\frac{L^k}{T}+\frac1T
\right\}.
\]
The calibration in Step 1 and the inequality
\[
T^{-1}\le T^{-1}+T^{-\beta}q_C^{2(1-\beta)}
\]
therefore imply
\[
\sup_{M\in\mathcal M_T(t_0,\zeta,C)}
\mathbb E_M[(\widehat\theta_{\kappa_T^{\mathrm{rad}}(C)}-\theta(K,e))^2]
\le
c_{\mathrm{att}}\mathfrak r_T(t_0,\zeta,C).
\]
Since \(R_T(t_0,\zeta,C)\) is the infimum over observable-data estimators in
\cref{obj:def:minimax-risk}, this also yields
\[
R_T(t_0,\zeta,C)\le c_{\mathrm{att}}\mathfrak r_T(t_0,\zeta,C).
\] % lean: uniform_overlap_frontier

\emph{Step 3. Constants for the lower comparison.}
From \cref{obj:lem:radius-explicit-observed-kl}, choose
\(K_0=K_0(t_0)>0\).  With \(c_0=(1-\alpha)/4\), define
\[
\mathsf B=\frac{4c_0^2K_0^2}{1-c_0^2}>0,
\qquad
x_\star=\max\left\{1,\frac1{4\mathsf B}\right\}.
\]
Then \(1<8\mathsf Bx_\star\).  Put
\[
c_{\mathrm{par}}=\frac{e^{-1/4}}{64},
\qquad
c_{\mathrm{hid}}
=
\left\{\frac{c_0(1-\alpha)\alpha}{2}\right\}^2
(8\mathsf B)^{-\beta}\frac{e^{-1/8}}4,
\]
\[
s_\star=1+x_\star^{1-\beta},
\qquad
c_{\mathrm{lo}}
=
\min\left\{\frac{c_{\mathrm{par}}}{2s_\star},
\frac{c_{\mathrm{hid}}}{2}\right\}.
\]
All these constants are positive and depend only on \(t_0,\zeta\). % lean: uniform_overlap_frontier

\emph{Step 4. Lower bound in the small-radius branch.}
By \cref{obj:lem:uniform-parametric-floor},
\[
\frac{c_{\mathrm{par}}}{T}\le R_T(t_0,\zeta,C)
\qquad (T\ge1,\ C\ge1).
\]
If \(\xi=Tq_C^2<x_\star\), then
\[
T^{-\beta}q_C^{2(1-\beta)}
=
T^{-1}\xi^{1-\beta}
\le
T^{-1}x_\star^{1-\beta},
\]
with both sides equal to \(0\) when \(q_C=0\). Hence
\[
\mathfrak r_T(t_0,\zeta,C)
\le
\frac{s_\star}{T},
\]
and therefore
\[
c_{\mathrm{lo}}\mathfrak r_T(t_0,\zeta,C)
\le
\frac{c_{\mathrm{par}}}{T}
\le
R_T(t_0,\zeta,C).
\] % lean: uniform_overlap_frontier

\emph{Step 5. Lower bound in the large-radius branch.}
Assume \(\xi\ge x_\star\). Then \(C>1\). Define
\[
Q=\left\lceil\frac{\log(8\mathsf B\xi)}{\mathcal D}\right\rceil .
\]
Because \(8\mathsf B\xi\ge8\mathsf Bx_\star>1\), we have \(Q\ge1\), and
\[
\mathcal D Q\ge \log(8\mathsf B\xi),
\qquad
Q<\frac{\log(8\mathsf B\xi)}{\mathcal D}+1 .
\]
Let \(M_+\) and \(M_-\) be the two signed-depth alternatives of depth \(Q\) from
\cref{obj:def:signed-depth-family}, and write \(\theta_+\) and \(\theta_-\) for
their target values.  By \cref{obj:lem:signed-depth-membership}, both alternatives
belong to \(\mathcal M_T(t_0,\zeta,C)\).  With \(w_Q\) denoting the terminal
depth mass in that construction, the same result gives
\[
\theta_+=c_0\varepsilon_C w_Q,
\qquad
\theta_-=-c_0\varepsilon_C w_Q.
\]
Define the separation radius
\[
\sigma_Q:=c_0\varepsilon_C w_Q .
\]
The target-value identities give the separation condition needed for the two-point
bound,
\[
2\sigma_Q\le |\theta_+-\theta_-| .
\]
Also, by \cref{obj:lem:radius-explicit-observed-kl},
\[
\frac{q_C}{2}\le\varepsilon_C\le q_C
\]
and, orienting the two observed laws as \(\mathbb P_{Q,+}^{\mathrm{obs},T}\)
against \(\mathbb P_{Q,-}^{\mathrm{obs},T}\),
\[
\operatorname{KL}(\mathbb P_{Q,+}^{\mathrm{obs},T}\,\|\,\mathbb P_{Q,-}^{\mathrm{obs},T})
\le
\mathsf B\,T\varepsilon_C^2\alpha^{2Q}L^{-Q}.
\]
Since \(\alpha^{2Q}L^{-Q}=e^{-\mathcal DQ}\), the choice of \(Q\) gives
\[
\mathsf B\,T\varepsilon_C^2\alpha^{2Q}L^{-Q}
\le
\mathsf B\,Tq_C^2(8\mathsf B\,Tq_C^2)^{-1}
=
\frac18 .
\]
The two alternatives have the same observed alphabet and the same revealed policies
by their construction; the displayed KL bound, in the same orientation, supplies
the information and finiteness hypotheses with \(\chi=1/8\).  Applying
\cref{obj:lem:two-point-risk-floor} therefore gives
\[
R_T(t_0,\zeta,C)
\ge
\sigma_Q^2\frac{e^{-1/8}}4
=
(c_0\varepsilon_Cw_Q)^2\frac{e^{-1/8}}4 .
\]
The depth mass in \cref{obj:lem:signed-depth-membership} satisfies
\[
w_Q=\frac{(1-\alpha)\alpha^Q}{1-\alpha^{Q+1}}
\ge
(1-\alpha)\alpha^Q,
\]
and the ceiling upper bound implies
\[
\alpha^Q
\ge
\alpha(8\mathsf B\xi)^{-1/(t_0\mathcal D)}.
\]
Thus
\[
(c_0\varepsilon_Cw_Q)^2\frac{e^{-1/8}}4
\ge
c_{\mathrm{hid}}\,q_C^2\xi^{-\beta}
=
c_{\mathrm{hid}}\,T^{-\beta}q_C^{2(1-\beta)}.
\]
Combining this hidden lower bound with the parametric floor and using
\(c_{\mathrm{lo}}\le c_{\mathrm{par}}/2\) and
\(c_{\mathrm{lo}}\le c_{\mathrm{hid}}/2\), we get
\[
c_{\mathrm{lo}}\mathfrak r_T(t_0,\zeta,C)
\le
\frac12 R_T(t_0,\zeta,C)+\frac12 R_T(t_0,\zeta,C)
=
R_T(t_0,\zeta,C).
\] % lean: uniform_overlap_frontier

\emph{Step 6. Assemble the frontier.}
Set
\[
c_{\mathrm{hi}}=\max\{c_{\mathrm{att}},c_{\mathrm{lo}}\},
\qquad
T_\star=T_{\mathrm{rad}}.
\]
Here \(c_{\mathrm{att}}=A(c_{\mathrm{rad}}+1)\) uses the same calibration constant
fixed in Step 1.  Thus \(0<c_{\mathrm{lo}}\le c_{\mathrm{hi}}\) and
\(c_{\mathrm{att}}\le c_{\mathrm{hi}}\).  Since
\(\mathfrak r_T(t_0,\zeta,C)\ge0\), the two upper bounds from Step 2 with constant
\(c_{\mathrm{att}}\) also hold with constant \(c_{\mathrm{hi}}\).  Together,
Steps 2, 4, and 5 prove, for all \(C\ge1\) and \(T\ge T_\star\),
\[
c_{\mathrm{lo}}\mathfrak r_T(t_0,\zeta,C)
\le
R_T(t_0,\zeta,C)
\le
c_{\mathrm{hi}}\mathfrak r_T(t_0,\zeta,C),
\]
together with
\[
\sup_{M\in\mathcal M_T(t_0,\zeta,C)}
\mathbb E_M[
(\widehat\theta_{\kappa_T^{\mathrm{rad}}(C)}-\theta(K,e))^2]
\le
c_{\mathrm{hi}}\mathfrak r_T(t_0,\zeta,C).
\] % lean: uniform_overlap_frontier

\emph{Step 7. Immediate estimator in the \(Tq_C^2=O(1)\) regime.}
Let \(C_T\ge1\) and suppose there is \(D_{\mathrm{imm}}\ge0\) such that,
eventually,
\[
Tq_{C_T}^2\le D_{\mathrm{imm}} .
\]
Define
\[
V=L\left(1+\frac4{L-1}\right)+\frac4{1-\alpha},
\qquad
c_{\mathrm{imm}}=4D_{\mathrm{imm}}+2V>0.
\]
For all sufficiently large \(T\), apply
\cref{obj:lem:radius-sensitive-phiw} with \(k=0\).  Since
\(q_{C_T}^2\le D_{\mathrm{imm}}/T\),
\[
\sup_{M\in\mathcal M_T(t_0,\zeta,C_T)}
\mathbb E_M[(\widehat\theta_0-\theta(K,e))^2]
\le
4q_{C_T}^2+\frac2T V
\le
\frac{4D_{\mathrm{imm}}+2V}{T}
=
\frac{c_{\mathrm{imm}}}{T}.
\]
This is the asserted eventual parametric bound for the immediate estimator. % lean: uniform_overlap_frontier
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:fixed-c-minimax}]
\begin{enumerate}
\item Set
\[
\alpha:=\exp(-1/t_0),
\qquad
L:=\exp(\zeta),
\qquad
D:=\frac{2}{t_0}+\zeta .
\]
By \(t_0>0\) and \(\zeta>0\), the constants satisfy
\[
0<\alpha<1,\qquad L>1,\qquad D>0,
\]
and, using the definitions in \cref{obj:synth_7,obj:synth_8},
\[
\log\alpha=-\frac1{t_0},
\qquad
\log L=\zeta .
\]
The exponent in the theorem can be written as
\[
\beta=\frac{2}{2+t_0\zeta}
      =\frac{2/t_0}{D}.
\]
Apply \cref{obj:lem:phiw-upper} with \(C\ge1\). It gives constants
\(c_{\mathrm{up},0}>0\) and \(T_{\mathrm{up}}\) such that, for all
\(T\ge T_{\mathrm{up}}\),
\[
\sup_{M\in\mathcal M_T(t_0,\zeta,C)}
\mathbb E_M\!\left[
\{\widehat\theta_{k_T}-\theta(M)\}^2
\right]
\le
c_{\mathrm{up},0}T^{-\beta}.
\]
For the signed-depth family, the behavior policy in \cref{obj:def:signed-depth-family} is a memoryless function of the single observed state, independent of the latent coordinate and the past. Consequently, for every horizon \(T\), terminal depth \(Q\ge1\), and sign \(v\in\{-1,+1\}\), the action kernel satisfies \cref{obj:ass:sequential-ignorability}. The same alternatives start from the behavior stationary law for every horizon, including the zero-horizon case. Indeed, for
\[
r:=L^{-1},
\]
the signed-depth constant-policy stationary law at action-one probability \(r\) has masses
\[
w_j\frac{1+u\varepsilon_C r^j}{2},
\qquad
w_j=\frac{(1-\alpha)\alpha^j}{1-\alpha^{Q+1}},
\]
and the behavior policy in \cref{obj:def:signed-depth-family} chooses action one with probability \(L^{-1}=r\). Hence its stationary law is precisely the initial hidden-state law
\[
w_j\frac{1+u\varepsilon_C L^{-j}}{2}.
\]
The chronological path law is generated from this initial distribution, so the time-zero joint-state marginal is this stationary law; when \(T=0\), the same identity is the whole marginal check, and for \(T\ge1\) it agrees with the behavior-stationary masses displayed in \cref{obj:lem:signed-depth-membership}. Thus \cref{obj:ass:stationary-start} holds for every horizon. These are the two structural hypotheses needed for \cref{obj:lem:observed-path-kl}. Therefore that lemma gives a constant \(B_0>0\), depending on \(t_0,\zeta,C\), such that for all \(T\) and \(Q\ge1\),
\[
\operatorname{KL}\!\left(
\mathbb P_{Q,+1}^{\mathrm{obs},T}
\,\middle\|\,
\mathbb P_{Q,-1}^{\mathrm{obs},T}
\right)
\le
B_0T\alpha^{2Q}L^{-Q}.
\]
Let
\[
\varepsilon_C:=\min\left\{\frac{C-1}{2},\frac12\right\},
\qquad
c_0:=\frac{1-\alpha}{4},
\]
and define
\[
c_\star:=
\{c_0\varepsilon_C(1-\alpha)\alpha\}^2
(8B_0)^{-\beta}\frac{\exp(-1/8)}{4}.
\]
Since \(C>1\), every factor in this product is positive, so \(c_\star>0\). Finally set
\[
c^\star:=\max\{c_{\mathrm{up},0},c_\star\}.
\]
Since \(c_{\mathrm{up},0}>0\) and \(c_\star>0\), the maximum is positive; moreover the second argument is bounded by the maximum. Thus
\[
0<c^\star,
\qquad
0<c_\star\le c^\star .
\] % lean: fixed_c_minimax

\item Define
\[
T_{\mathrm{pos}}:=\left\lceil\frac{1}{8B_0}\right\rceil+1,
\qquad
T_\star:=\max\{T_{\mathrm{up}},T_{\mathrm{pos}},1\}.
\]
Fix an integer \(T\ge T_\star\). Then \(T\ge1\), \(T\ge T_{\mathrm{up}}\), and
\[
1<8B_0T .
\]
Indeed,
\[
\frac{1}{8B_0}
\le
\left\lceil\frac{1}{8B_0}\right\rceil
<
T,
\]
and multiplication by \(8B_0>0\) gives the displayed inequality. Put
\[
\tau_T:=\frac{\log(8B_0T)}{D},
\qquad
Q:=\lceil \tau_T\rceil .
\]
Since \(\log(8B_0T)>0\) and \(D>0\), one has \(Q\ge1\). Moreover,
\[
\tau_T\le Q<\tau_T+1
\]
and therefore
\[
DQ\ge \log(8B_0T).
\] % lean: fixed_c_minimax

\item The preceding depth choice makes the two signed-depth alternatives statistically close. First,
\[
\alpha^{2Q}L^{-Q}
=
\exp\{Q(2\log\alpha-\log L)\}
=
\exp(-DQ),
\]
because \(D=2/t_0+\zeta\), \(\log\alpha=-1/t_0\), and \(\log L=\zeta\). Hence
\[
\exp(-DQ)
\le
\exp\{-\log(8B_0T)\}
=
(8B_0T)^{-1}.
\]
Combining this inequality with the KL bound from \cref{obj:lem:observed-path-kl} gives
\[
\operatorname{KL}\!\left(
\mathbb P_{Q,+1}^{\mathrm{obs},T}
\,\middle\|\,
\mathbb P_{Q,-1}^{\mathrm{obs},T}
\right)
\le
\frac18 .
\] % lean: fixed_c_minimax

\item Let \(M_{Q,+1}^T\) and \(M_{Q,-1}^T\) be the two signed-depth experiments of \cref{obj:def:signed-depth-family}. They have one observed state and \(2(Q+1)\) latent states. Their revealed behavior and target policies are common, because the policy part of the construction is independent of the sign \(v\). By \cref{obj:lem:signed-depth-membership}, both experiments belong to \(\mathcal M_T(t_0,\zeta,C)\), and their target values are
\[
\theta(M_{Q,+1}^T)=c_0\varepsilon_C w_Q,
\qquad
\theta(M_{Q,-1}^T)=-c_0\varepsilon_C w_Q,
\]
where
\[
w_Q
:=
\frac{(1-\alpha)\alpha^Q}{1-\alpha^{Q+1}} .
\]
Set
\[
s_T:=c_0\varepsilon_C w_Q .
\]
Then \(s_T\ge0\), and the two target values satisfy the separation bound
\[
2s_T
\le
\bigl|\theta(M_{Q,+1}^T)-\theta(M_{Q,-1}^T)\bigr|.
\]
The two embedded observed laws have the same finite real-reward support,
\[
\Omega_T^\circ
:=
\left\{
((0,a_t,y_t))_{t=1}^T:
 a_t\in\{0,1\},\ y_t\in\{-1,+1\}
\right\}.
\]
They assign total mass one to \(\Omega_T^\circ\), because the construction has one observed state, Boolean actions, and rewards in \(\{-1,+1\}\). We now check that every atom in this common support has positive mass under both signs. Fix
\(o=((0,a_t,y_t))_{t=1}^T\in\Omega_T^\circ\), and choose a fixed latent sign \(u_0\in\{-1,+1\}\). Consider the full hidden lift that keeps the latent state at depth zero and sign \(u_0\):
\[
S_t=(0,(0,u_0)),
\qquad
A_t=a_t,
\qquad
Y_t=y_t,
\qquad t=1,\\ldots,T.
\]
Since \(0<\alpha<1\) and \(Q\ge1\),
\[
1-\alpha^{Q+1}>0,
\qquad
w_0=\frac{1-\alpha}{1-\alpha^{Q+1}}>0.
\]
Also \(0<\varepsilon_C\le1/2\), so the initial mass of the chosen hidden state is positive:
\[
w_0\frac{1+u_0\varepsilon_C}{2}>0.
\]
For each action in the observed word,
\[
b(1\mid0)=\frac1L>0,
\qquad
b(0\mid0)=1-\frac1L>0,
\]
because \(L>1\). Along the displayed hidden lift, the current depth is \(0<Q\). Hence the reward factor is
\[
\frac{1+y_tvc_0u_0\mathbf 1\{0=Q\}}{2}=\frac12>0,
\]
and the hidden transition that resets to the same depth-zero sign has weight
\[
(1-\alpha)\frac{1+u_0\varepsilon_C}{2}>0.
\]
Thus this single compatible hidden lift has strictly positive finite-path probability for each sign \(v\in\{-1,+1\}\), and summing over all lifts gives
\[
\mathbb P_{Q,v}^{\mathrm{obs},T}(\{o\})>0,
\qquad o\in\Omega_T^\circ,
\qquad v\in\{-1,+1\}.
\]
The reward decoding used in the embedding is the same for the two signs, so the embedded laws preserve this common atomic support. Therefore any measurable set with zero mass under one embedded observed law contains no atom of \(\Omega_T^\circ\), and consequently has zero mass under the other embedded observed law. Hence
\[
\mathbb P_{Q,+1}^{\mathrm{obs},T}
\ll
\mathbb P_{Q,-1}^{\mathrm{obs},T},
\qquad
\mathbb P_{Q,-1}^{\mathrm{obs},T}
\ll
\mathbb P_{Q,+1}^{\mathrm{obs},T},
\]
and the finite KL displayed above is an ordinary KL bound between mutually absolutely continuous observed laws. % lean: signedDepth_fullSupportObs

\item Apply \cref{obj:lem:two-point-risk-floor} to the present pair with \(s=s_T\), taking the \(+1\) alternative as the lemma's first model and the \(-1\) alternative as its second: \(M_0=M_{Q,+1}^T\) and \(M_1=M_{Q,-1}^T\). With this assignment the lemma's divergence hypothesis \(\operatorname{KL}(\mathbb P_0\|\mathbb P_1)\le\chi\) reads
\[
\operatorname{KL}\!\left(
\mathbb P_{Q,+1}^{\mathrm{obs},T}
\,\middle\|\,
\mathbb P_{Q,-1}^{\mathrm{obs},T}
\right)
\le
\frac18 ,
\]
which is exactly the direction bounded above, and its absolute-continuity hypothesis reads \(\mathbb P_{Q,+1}^{\mathrm{obs},T}\ll\mathbb P_{Q,-1}^{\mathrm{obs},T}\), which the preceding step supplies. The common observable structure was checked in the preceding step, the KL bound gives the information hypothesis with \(\chi=1/8\), and the displayed target-value calculation gives
\[
2s_T
\le
\bigl|\theta(M_{Q,+1}^T)-\theta(M_{Q,-1}^T)\bigr|.
\]
Therefore every measurable observable estimator \(\widehat\theta\), under the stated squared-error integrability conditions for both alternatives, satisfies
\[
s_T^2\frac{\exp(-1/8)}{4}
\le
\max\left\{
\mathbb E_{M_{Q,+1}^T}\!\left[
\{\widehat\theta-\theta(M_{Q,+1}^T)\}^2
\right],
\mathbb E_{M_{Q,-1}^T}\!\left[
\{\widehat\theta-\theta(M_{Q,-1}^T)\}^2
\right]
\right\}.
\]
Commuting the two entries of the maximum, this is exactly
\[
s_T^2\frac{\exp(-1/8)}{4}
\le
\max_{v\in\{-1,+1\}}
\mathbb E_{M_{Q,v}^T}\!\left[
\{\widehat\theta-\theta(M_{Q,v}^T)\}^2
\right].
\]
The two signed-depth alternatives belong to \(\mathcal M_T(t_0,\zeta,C)\) and share the revealed inputs \((n_X,b,e)\). For the witness clause of the theorem the two squared-error integrability conditions are hypotheses, imposed on the estimator and discharged by whoever supplies it; nothing above derives them. They are derived only when \(\widehat\theta\) is further restricted to the admissible carrier of \cref{obj:def:minimax-risk}, whose members are measurable and uniformly \([-1,1]\)-valued by definition: for such an estimator the squared error at either alternative is a measurable function bounded by \((1+|\theta_\pm|)^2\le4\) on a probability space, hence integrable, so the preceding pointwise lower bound applies to every admissible estimator. Taking the supremum over the model class, then the infimum over these admissible estimators, yields
\[
s_T^2\frac{\exp(-1/8)}{4}
\le
R_T(t_0,\zeta,C).
\] % lean: two_point_floor_explicit

\item It remains to compare \(s_T\) with \(T^{-\beta/2}\). Since \(0<\alpha<1\) and \(Q\ge1\), also \(Q+1\ge1\). Thus
\[
0<\alpha^{Q+1}<1,
\]
so the denominator in the depth mass is strictly positive and satisfies
\[
0<1-\alpha^{Q+1}\le1.
\]
The numerator \((1-\alpha)\alpha^Q\) is nonnegative, and division by a positive number at most one gives
\[
w_Q
=
\frac{(1-\alpha)\alpha^Q}{1-\alpha^{Q+1}}
\ge
(1-\alpha)\alpha^Q .
\]
The upper ceiling inequality \(Q<\tau_T+1\) and the identity \(\log\alpha=-1/t_0\) give
\[
\alpha^Q
=
\exp\!\left(-\frac{Q}{t_0}\right)
\ge
\exp\!\left(-\frac{\tau_T+1}{t_0}\right)
=
\alpha(8B_0T)^{-(1/t_0)/D}.
\]
Therefore
\[
s_T
\ge
c_0\varepsilon_C(1-\alpha)\alpha
(8B_0T)^{-(1/t_0)/D}.
\]
Squaring and using \(\beta=(2/t_0)/D\) yields
\[
s_T^2
\ge
\{c_0\varepsilon_C(1-\alpha)\alpha\}^2
(8B_0T)^{-\beta}
=
\{c_0\varepsilon_C(1-\alpha)\alpha\}^2
(8B_0)^{-\beta}T^{-\beta}.
\]
Multiplying by \(\exp(-1/8)/4\) gives
\[
c_\star T^{-\beta}
\le
s_T^2\frac{\exp(-1/8)}{4}.
\]
Combining this with the two-point floor proves
\[
c_\star T^{-\beta}
\le
R_T(t_0,\zeta,C),
\]
and, for every measurable observable-data estimator satisfying the two integrability conditions,
\[
c_\star T^{-\beta}
\le
\max_{v\in\{-1,+1\}}
\mathbb E_{M_{Q,v}^T}\!\left[
\{\widehat\theta-\theta(M_{Q,v}^T)\}^2
\right].
\] % lean: fixed_c_minimax

\item The upper bound follows from the estimator bound in the first step. Since \(T\ge T_{\mathrm{up}}\),
\[
\sup_{M\in\mathcal M_T(t_0,\zeta,C)}
\mathbb E_M\!\left[
\{\widehat\theta_{k_T}-\theta(M)\}^2
\right]
\le
c_{\mathrm{up},0}T^{-\beta}
\le
c^\star T^{-\beta}.
\]
By \cref{obj:def:minimax-risk}, the minimax risk is bounded above by the worst-case risk of any admissible observable estimator, in particular by that of \(\widehat\theta_{k_T}\). Thus
\[
R_T(t_0,\zeta,C)
\le
c^\star T^{-\beta}.
\]
Together with the lower bound from the preceding step, this proves all asserted inequalities for every \(T\ge T_\star\), and the chosen \(Q=\lceil \log(8B_0T)/D\rceil\) supplies the stated signed-depth witnesses with one observed state and \(2(Q+1)\) latent states. % lean: fixed_c_minimax
\end{enumerate}
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:unit-overlap-boundary}]
Write
\[
\alpha:=\exp(-1/t_0),
\qquad
L:=\exp(\zeta).
\]
Since \(t_0>0\) and \(\zeta>0\), one has \(0<\alpha<1\) and \(1<L\). % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.unit_overlap_boundary

1. Define
\[
c_{\mathrm{imm}}
:=
2\left\{
L\left(1+\frac{4}{L-1}\right)+\frac{4}{1-\alpha}
\right\}.
\]
The preceding inequalities imply \(c_{\mathrm{imm}}>0\). For any \(T\ge1\), \cref{obj:lem:radius-sensitive-phiw} applied at depth \(k=0\) and overlap constant \(C=1\) gives, uniformly over \(M\in\mathcal M_T(t_0,\zeta,1)\),
\[
\mathbb E_M\!\left[
\bigl(\widehat\theta_0-\theta(K,e)\bigr)^2
\right]
\le
\frac{2}{T}
\left\{
L\left(1+\frac{4}{L-1}\right)+\frac{4}{1-\alpha}
\right\}
=
\frac{c_{\mathrm{imm}}}{T}.
\]
Taking the supremum over the model class yields
\[
\sup_{M\in\mathcal M_T(t_0,\zeta,1)}
\mathbb E_M\!\left[
\bigl(\widehat\theta_0-\theta(K,e)\bigr)^2
\right]
\le
\frac{c_{\mathrm{imm}}}{T}.
\]
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.radius_sensitive_phiw

2. Let \(M\in\mathcal M_T(t_0,\zeta,1)\). Write \(d_b\) and \(d_e\) for the behavior and target stationary laws on the finite joint state space \(\mathcal S\), and define
\[
\left\|d_b-d_e\right\|_{\mathrm{TV}}
:=
\frac12\sum_{s\in\mathcal S}|d_b(s)-d_e(s)| .
\]
The stationary-law total-variation bound at overlap radius \(q_C\) gives
\[
\left\|d_b-d_e\right\|_{\mathrm{TV}}
\le q_1,
\]
and at unit overlap \cref{obj:def:overlap-distance-radius} gives
\[
q_1=\frac{1-1}{1}=0.
\]
Thus
\[
\left\|d_b-d_e\right\|_{\mathrm{TV}}
\le0.
\]
Since the displayed total-variation quantity is a half-sum of absolute values, it is nonnegative, so
\[
\left\|d_b-d_e\right\|_{\mathrm{TV}}=0.
\]
Unfolding the definition of total variation yields
\[
\sum_{s\in\mathcal S}|d_b(s)-d_e(s)|=0.
\]
For each joint state \(s\), the nonnegative summand is bounded by the whole finite sum,
\[
0\le |d_b(s)-d_e(s)|
\le
\sum_{u\in\mathcal S}|d_b(u)-d_e(u)|
=0,
\]
and hence
\[
|d_b(s)-d_e(s)|=0.
\]
Therefore \(d_b(s)=d_e(s)\) for every \(s\in\mathcal S\), and consequently \(d_e=d_b\). % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.stationary_tv_le_overlapRadius

3. Set
\[
c_{\mathrm{lo}}:=\frac{\exp(-1/4)}{64},
\qquad
c_{\mathrm{hi}}:=\max\{c_{\mathrm{imm}},c_{\mathrm{lo}}\},
\qquad
T_\star:=1.
\]
Then \(c_{\mathrm{lo}}>0\), \(c_{\mathrm{lo}}\le c_{\mathrm{hi}}\), and therefore \(c_{\mathrm{hi}}>0\); also \(T_\star\ge1\). For every \(T\ge T_\star\), \cref{obj:lem:uniform-parametric-floor} at \(C=1\) gives
\[
\frac{\exp(-1/4)}{64T}
\le
R_T(t_0,\zeta,1),
\]
which is
\[
\frac{c_{\mathrm{lo}}}{T}
\le
R_T(t_0,\zeta,1).
\]
For the upper bound, the minimax risk is bounded above by the worst-case risk of any admissible estimator with nonnegative squared risk; applying this to \(\widehat\theta_0\) and using Step 1 gives
\[
R_T(t_0,\zeta,1)
\le
\sup_{M\in\mathcal M_T(t_0,\zeta,1)}
\mathbb E_M\!\left[
\bigl(\widehat\theta_0-\theta(K,e)\bigr)^2
\right]
\le
\frac{c_{\mathrm{imm}}}{T}
\le
\frac{c_{\mathrm{hi}}}{T}.
\]
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.uniform_parametric_floor

4. Finally,
\[
t_0\zeta>0,
\qquad\text{hence}\qquad
2+t_0\zeta>2.
\]
Therefore
\[
\frac{2}{2+t_0\zeta}<1,
\]
which is the asserted bound on the hidden-state exponent. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.unit_overlap_boundary
\end{proof}

\begin{proof}[Proof of \cref{obj:thm:shrinking-overlap-frontier}]
Put
\[
\alpha=\exp(-1/t_0),\qquad L=\exp(\zeta),\qquad
\beta=\frac{2}{2+t_0\zeta},\qquad
\gamma=2(1-\beta).
\]
Since \(t_0>0\) and \(\zeta>0\), one has \(0<\alpha<1\), \(1<L\),
\(0<\beta<1\), and hence \(\gamma>0\).

\emph{1. A radius-depth calibration.}
Let
\[
\Lambda:=2\log(1/\alpha)+\log L=\frac{2}{t_0}+\zeta .
\]
Then \(\Lambda>0\),
\[
\beta=\frac{2/t_0}{\Lambda},
\qquad
1-\beta=\frac{\zeta}{\Lambda}.
\]
There are \(c_R>0\) and \(T_R\ge1\), depending only on \(t_0,\zeta\), such that for every \(T\ge T_R\) and every \(C\ge1\),
\begin{equation}\label{eq:shrinking-overlap-frontier-radius-calibration}
q_C^2\alpha^{2\kappa_T^{\mathrm{rad}}(C)}
+\frac{L^{\kappa_T^{\mathrm{rad}}(C)}}{T}
\le
c_R\left\{
T^{-1}+T^{-\beta}q_C^{2(1-\beta)}
\right\}.
\end{equation}
Indeed, choose \(T_R\) so that \(|\log T|\le(\Lambda/2)T\) for all
\(T\ge T_R\).  Fix such a \(T\), put \(q=q_C\), and set \(x=Tq^2\).
The relation \(C\ge1\) and \cref{obj:def:overlap-distance-radius} give
\(0\le q\le1\), hence \(0\le x\le T\).  If \(x\le1\), then
\(\kappa_T^{\mathrm{rad}}(C)=0\) by \cref{obj:def:radius-adaptive-history-depth}, and
\(q^2\le T^{-1}\), so
\[
q^2+\frac1T
\le 2T^{-1}
\le
2\left\{T^{-1}+T^{-\beta}q^{2(1-\beta)}\right\}.
\]
If \(x>1\), then \(q>0\).  With
\[
k=\left\lfloor\frac{\log x}{\Lambda}\right\rfloor ,
\]
the choice of \(T_R\) gives \(k\le T/2\), so the cap in
\cref{obj:def:radius-adaptive-history-depth} is inactive and
\(\kappa_T^{\mathrm{rad}}(C)=k\).  The floor bounds
\[
\frac{\log x}{\Lambda}-1<k\le\frac{\log x}{\Lambda}
\]
imply
\[
\alpha^{2k}
=\exp\!\left(-\frac{2k}{t_0}\right)
\le
\exp(2/t_0)x^{-\beta},
\qquad
\frac{L^k}{T}
\le
\frac{x^{1-\beta}}{T}.
\]
Since
\[
q^2x^{-\beta}
=
T^{-\beta}q^{2(1-\beta)},
\qquad
\frac{x^{1-\beta}}{T}
=
T^{-\beta}q^{2(1-\beta)},
\]
the \(x>1\) case is bounded by
\[
\bigl(\exp(2/t_0)+1\bigr)T^{-\beta}q^{2(1-\beta)}.
\]
Taking
\[
c_R=\max\{2,\exp(2/t_0)+1\}
\]
proves \cref{eq:shrinking-overlap-frontier-radius-calibration}. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.radiusAdaptiveDepth_rate

\emph{2. Constants.}
Apply \cref{obj:thm:uniform-overlap-frontier} to obtain constants
\(c_{\mathrm{lo}},c_{\mathrm{hi}}>0\), with
\(c_{\mathrm{lo}}\le c_{\mathrm{hi}}\), and a threshold \(T_U\ge1\), all
depending only on \(t_0,\zeta\), such that its two-sided risk comparison and
radius-calibrated estimator bound hold for all \(C\ge1\) and \(T\ge T_U\).
Define
\[
\eta_{\mathrm{rad}}:=2^{-\gamma},
\]
and
\[
\mathsf A_{\mathrm{bd}}
:=
\max\left\{
4,\,
2L\left(1+\frac{4}{L-1}\right),\,
\frac{8}{1-\alpha}
\right\}.
\]
Finally set
\[
c_{\mathrm{loc}}:=c_{\mathrm{lo}}\eta_{\mathrm{rad}},
\qquad
c_{\mathrm{att}}:=\mathsf A_{\mathrm{bd}}(c_R+1),
\qquad
C_{\mathrm{loc}}:=\max\{c_{\mathrm{hi}},c_{\mathrm{att}}\}.
\]
Then \(0<c_{\mathrm{loc}}\le C_{\mathrm{loc}}\), because
\(0<\eta_{\mathrm{rad}}\le1\), \(c_{\mathrm{lo}}\le c_{\mathrm{hi}}\), and
\(\mathsf A_{\mathrm{bd}}>0\). % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.shrinking_overlap_frontier

\emph{3. Comparing the two rate surfaces.}
Fix a deterministic sequence \((\delta_T)_{T\ge1}\) with
\(\delta_T\ge0\) and \(\delta_T\to0\).  Choose \(T_\Delta\) such that
\[
0\le\delta_T<1\qquad\text{for all }T\ge T_\Delta .
\]
Let \(T_\star=\max\{T_U,T_R,T_\Delta\}\).  For \(T\ge T_\star\), put
\(C_T=1+\delta_T\).  Then \(C_T\ge1\), and
\[
q_{C_T}
=
\frac{C_T-1}{C_T}
=
\frac{\delta_T}{1+\delta_T}.
\]
Since \(0\le\delta_T<1\),
\[
0\le q_{C_T}\le\delta_T,
\qquad
\frac{\delta_T}{2}\le q_{C_T}.
\]
Raising these inequalities to the positive power \(\gamma\) gives
\[
\eta_{\mathrm{rad}}\delta_T^\gamma
\le
q_{C_T}^\gamma
\le
\delta_T^\gamma .
\]
Thus the theorem's local rate, evaluated along the fixed sequence \((\delta_T)\), and the fixed-radius rate at \(C_T\) satisfy
\begin{equation}\label{eq:shrinking-overlap-frontier-rate-comparison}
\eta_{\mathrm{rad}}\left\{T^{-1}+T^{-\beta}\delta_T^\gamma\right\}
\le
\mathfrak r_T(t_0,\zeta,C_T)
\le
T^{-1}+T^{-\beta}\delta_T^\gamma.
\end{equation}
Equivalently, since \(r_T(\delta)\) in the statement denotes this fixed-sequence quantity,
\[
\eta_{\mathrm{rad}}\,r_T(\delta)
\le
\mathfrak r_T(t_0,\zeta,C_T)
\le
r_T(\delta).
\]
Combining \cref{eq:shrinking-overlap-frontier-rate-comparison} with the two-sided comparison from
\cref{obj:thm:uniform-overlap-frontier} gives
\begin{equation}\label{eq:shrinking-overlap-frontier-risk-comparison}
c_{\mathrm{loc}}\,r_T(\delta)
\le
R_T(t_0,\zeta,1+\delta_T)
\le
C_{\mathrm{loc}}\,r_T(\delta).
\end{equation}
% lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.shrinking_overlap_frontier

\emph{4. Attainment by the displayed shrinking depth.}
For \(T\ge T_\star\), define
\[
C_T^\dagger:=\frac{1}{1-\delta_T}.
\]
Then \(C_T^\dagger\ge1\), and
\[
q_{C_T^\dagger}
=
\frac{C_T^\dagger-1}{C_T^\dagger}
=
\delta_T .
\]
Therefore \cref{obj:def:radius-adaptive-history-depth} gives
\begin{equation}\label{eq:shrinking-overlap-frontier-depth-identity}
\kappa_T^{\mathrm{rad}}(C_T^\dagger)=\kappa_T.
\end{equation}
where \(\kappa_T\) is the depth displayed in the theorem statement.  In
particular \(\kappa_T\le T/2\).  Applying \cref{eq:shrinking-overlap-frontier-radius-calibration} with \(C=C_T^\dagger\) yields
\begin{equation}\label{eq:shrinking-overlap-frontier-local-depth-rate}
\delta_T^2\alpha^{2\kappa_T}
+\frac{L^{\kappa_T}}{T}
\le
c_R r_T(\delta).
\end{equation}
Now fix \(M\in\mathcal M_T(t_0,\zeta,1+\delta_T)\).  By
\cref{obj:lem:radius-sensitive-phiw}, the estimator from
\cref{obj:def:phiw-estimator} satisfies
\[
\mathbb E_M\!\left[
\bigl(\widehat\theta_{\kappa_T}-\theta(M)\bigr)^2
\right]
\le
4q_{C_T}^2\alpha^{2\kappa_T}
+\frac{2}{T}
\left\{
L^{\kappa_T+1}\left(1+\frac{4}{L-1}\right)
+\frac{4}{1-\alpha}
\right\}.
\]
Using \(q_{C_T}\le\delta_T\) and the definition of
\(\mathsf A_{\mathrm{bd}}\),
\[
\mathbb E_M\!\left[
\bigl(\widehat\theta_{\kappa_T}-\theta(M)\bigr)^2
\right]
\le
\mathsf A_{\mathrm{bd}}
\left\{
\delta_T^2\alpha^{2\kappa_T}
+\frac{L^{\kappa_T}}{T}
+T^{-1}
\right\}.
\]
Together with \cref{eq:shrinking-overlap-frontier-local-depth-rate} and \(T^{-1}\le r_T(\delta)\), this gives
\[
\mathbb E_M\!\left[
\bigl(\widehat\theta_{\kappa_T}-\theta(M)\bigr)^2
\right]
\le
c_{\mathrm{att}}\,r_T(\delta)
\le
C_{\mathrm{loc}}\,r_T(\delta).
\]
Taking the supremum over
\(M\in\mathcal M_T(t_0,\zeta,1+\delta_T)\) proves the asserted estimator
bound. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.shrinking_overlap_frontier

\emph{5. The bounded local regime.}
Assume that there is a constant \(D_{\mathrm{bd}}\ge0\) such that
\[
T\delta_T^2\le D_{\mathrm{bd}}
\]
for all sufficiently large \(T\).  Since \(q_{1+\delta_T}\le\delta_T\), the
same tail of \(T\)'s satisfies
\[
Tq_{1+\delta_T}^2\le D_{\mathrm{bd}}.
\]
Applying the final assertion of \cref{obj:thm:uniform-overlap-frontier} to
the sequence \(C_T=1+\delta_T\) gives a constant \(c_{\mathrm{imm}}>0\) such
that, for all sufficiently large \(T\),
\[
\sup_{M\in\mathcal M_T(t_0,\zeta,1+\delta_T)}
\mathbb E_M\!\left[
\bigl(\widehat\theta_0-\theta(M)\bigr)^2
\right]
\le
\frac{c_{\mathrm{imm}}}{T}.
\]
This is the claimed immediate-depth conclusion. % lean: CausalSmith.Stat.PomdpLatentOverlapMinimax.shrinking_overlap_frontier
\end{proof}

\begin{proof}[Proof of \cref{obj:prop:finite-insulin-demonstration}]
Fix \(T\). We verify the five asserted clauses.

\begin{enumerate}
\item By \cref{obj:def:insulin-grid}, the observed coordinate has
\[
|\mathcal X|
=
|\{50,100,150,200,250\}|\,
|\{0,31,850\}|^2\,|\{0,1\}|
=
5\cdot 3^2\cdot 2
=
90,
\]
and the latent diet-memory coordinate has
\[
|\mathcal H|=|\{0,78\}|^2=2^2=4.
\]
Thus the joint-state alphabet \(\mathcal S=\mathcal X\times\mathcal H\) has
\[
|\mathcal S|=90\cdot4=360.
\]
% lean: insulinGrid_card

\item For a memoryless policy \(p\in\{b,e\}\), let \(P_p^{(T)}:\mathcal S\times\mathcal S\to[0,1]\) be the reward-marginalized joint-state transition matrix
\[
P_p^{(T)}(s,s')
:=
\sum_{a\in\{0,1\}}p(a\mid s_X)\,
K_T\{(r_{\mathrm{out}},\bar s):\bar s=s'\mid s,a\},
\qquad s,s'\in\mathcal S .
\]
By the construction in \cref{obj:def:insulin-grid}, the embedded policy kernel of the refreshed insulin experiment is this matrix. Hence, for every pair of probability laws \(\nu,\nu'\) on \(\mathcal S\), \cref{obj:lem:insulin-policy-contraction} gives
\[
\|\nu P_p^{(T)}-\nu'P_p^{(T)}\|_{\mathrm{TV}}
\le
\frac12\|\nu-\nu'\|_{\mathrm{TV}} .
\]
Applying this with \(p=b\) and \(p=e\) is exactly \cref{obj:ass:uniform-contraction} with \(\alpha=1/2\). % lean: insulinGrid_contraction

\item The behavior probabilities are
\[
b(1\mid x)=\frac3{10},
\qquad
b(0\mid x)=\frac7{10},
\]
and the target policy is a probability law with \(0\le e(a\mid x)\le1\). Therefore
\[
e(1\mid x)\le 1=\frac{10}{3}\,\frac3{10}
=\frac{10}{3}b(1\mid x),
\]
and
\[
e(0\mid x)\le 1\le \frac73
=\frac{10}{3}\,\frac7{10}
=\frac{10}{3}b(0\mid x).
\]
Thus \cref{obj:ass:policy-overlap} holds with \(L=10/3\). % lean: insulinGrid_policyOverlap

\item Let \(d_b\) and \(d_e\) be the selected stationary laws of \(P_b^{(T)}\) and \(P_e^{(T)}\). The behavior and target policies in \cref{obj:def:insulin-grid} are probability vectors; together with the contraction estimate supplied by \cref{obj:lem:insulin-policy-contraction}, \cref{obj:lem:insulin-stationary-law} gives
\[
d_p(s')
=
\sum_{s\in\mathcal S}d_p(s)P_p^{(T)}(s,s'),
\qquad
\sum_{s\in\mathcal S}d_p(s)=1,
\qquad
 d_p(s)\ge0,
\]
for \(p\in\{b,e\}\) and all \(s'\in\mathcal S\).
Write \(S_t\) for the joint state at time \(t\). For \(a\in\{0,1\}\) and \(s,s'\in\mathcal S\), define the nonrefresh action kernel as follows. If
\[
s=(g,d_1,d_2,x_1,x_2,a_1),
\qquad
s'=(g',d'_1,d'_2,x'_1,x'_2,a'_1),
\]
let \(\varphi_{s,a}(g')\) be the probability that the Gaussian clipping-and-rounding rule in \cref{obj:def:insulin-grid} produces next glucose value \(g'\), and set this probability to zero when \(g'\) is outside the glucose grid. Let
\[
\pi_D(0)=\frac45,
\qquad
\pi_D(78)=\frac15,
\qquad
\pi_E(0)=\pi_E(31)=\frac25,
\qquad
\pi_E(850)=\frac15,
\]
with \(\pi_D\) and \(\pi_E\) equal to zero off their displayed supports. Conditional on the nonrefresh branch and action \(a\), \cref{obj:def:insulin-grid} draws the new diet and activity coordinates with laws \(\pi_D\) and \(\pi_E\), shifts the old memory coordinates, and stores the current action. Hence the nonrefresh probability of moving from \(s\) to \(s'\) under action \(a\) is
\[
\omega_a(s,s')
:=
\varphi_{s,a}(g')\,\pi_D(d'_1)\,\pi_E(x'_1)\,
\mathbf 1\{d'_2=d_1\}\,
\mathbf 1\{x'_2=x_1\}\,
\mathbf 1\{a'_1=a\}.
\]
All factors in this display are nonnegative, so \(\omega_a(s,s')\ge0\). The refresh branch in \cref{obj:def:insulin-grid} has probability \(1/2\) and draws the next full state uniformly from the \(360\)-point set \(\mathcal S\), so its contribution to any fixed next state \(s'\) is
\[
\frac12\cdot\frac1{360}=\frac1{720}.
\]
The remaining branch has probability \(1/2\), and the policy-induced kernel averages the action-specific nonrefresh probabilities with weights \(p(a\mid s_X)\). Thus, for every memoryless policy \(p\) on \(\{0,1\}\),
\[
P_p^{(T)}(s,s')
=
\frac1{720}
+
\frac12\sum_{a\in\{0,1\}}p(a\mid s_X)\omega_a(s,s'),
\qquad s,s'\in\mathcal S .
\]
Since the behavior policy has nonnegative action probabilities, this identity gives the pointwise minorization
\[
P_b^{(T)}(s,s')
=
\frac1{720}
+\frac12\sum_{a\in\{0,1\}}b(a\mid s_X)\omega_a(s,s')
\ge
\frac1{720},
\qquad s,s'\in\mathcal S .
\]
The stationary-minorization calculation is then
\[
d_b(s')
=
\sum_{s\in\mathcal S}d_b(s)P_b^{(T)}(s,s')
\ge
\sum_{s\in\mathcal S}d_b(s)\frac1{720}
=
\frac1{720},
\qquad s'\in\mathcal S .
\]
Since \(d_e\) is a probability vector,
\[
d_e(s')\le\sum_{\bar s\in\mathcal S}d_e(\bar s)=1.
\]
Combining the last two displays yields
\[
d_e(s')\le 1\le 720\,d_b(s'),
\qquad s'\in\mathcal S,
\]
and \(720\ge1\). This proves \cref{obj:ass:latent-stationary-overlap} with \(C=720\). % lean: insulinGrid_stationaryOverlap

\item For the same horizon \(T\), write \(K\) for the transition kernel of \(\mathcal I_{\mathrm{grid}}\), \(b\) and \(e\) for its behavior and target policies, and \(d_b\) for the stationary law induced by \(b\). The stationary immediate-weighting bias diagnostic is
\[
\Delta_{\mathrm{imm}}(T)
=
\sum_s d_b(s)
\sum_{a\in\{0,1\}}
b(a\mid s_X)\,
\frac{e(a\mid s_X)}{b(a\mid s_X)}
\int r\,K(ds',dr\mid s,a)
-
\theta(K,e).
\]
For this diagnostic on the refreshed finite insulin adaptation \(\mathcal I_{\mathrm{grid}}\) of \cref{obj:def:insulin-grid}, \cref{obj:lem:insulin-bias-certificate} supplies the interval certificate
\[
-\frac6{1000}
<
\Delta_{\mathrm{imm}}(T)
<
-\frac{58}{10000},
\]
which is the displayed bound
\[
-0.006<\Delta_{\mathrm{imm}}(T)<-0.0058.
\]
% lean: insulinGrid_bias_certificate
\end{enumerate}

The five displayed conclusions are precisely the conjunction asserted for the refreshed insulin experiment. % lean: finite_insulin_demonstration
\end{proof}
