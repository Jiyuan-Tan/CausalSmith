\section*{exp\_\allowbreak{}supply\_\allowbreak{}kway\_\allowbreak{}uniform\_\allowbreak{}dosage}

\noindent\textit{Specialization:} v1\par

\paragraph{TL;DR.} For independent product-Bernoulli assignment and the fixed complete Fourier basis through order \(k\), the population \(A\)-criterion is an explicit elementary-symmetric polynomial. Its unique shared-supply optimizer is the uniform dosage \(u=\min\{L/p,1/2\}\), with an exact criterion value and a global quadratic stability certificate whose coefficient is sharp when \(L\ge p/2\). Exact leverage and a rare-activation rank obstruction complement a two-sided threshold-OLS risk theorem. A fixed spectral threshold, the mean-zero whitened Gram, and a second-order inverse identity yield an explicit uniform \(O(n^{-1})\) relative-risk expansion at fixed dimension. This is oracle attainment at the population-optimal independent design, not finite-sample dosage minimaxity.

\section{Project justification}

\paragraph{Gap.} The modern PFED paper resolves limited supply analytically only for additive responses and reports numerical evidence for interaction models. Classical product-design theory gives the inverse-trace identity but not the shared expected-supply optimizer, its stability certificate, or the fixed-threshold second-order risk translation with fallback bias.

\paragraph{Niche.} The fixed dosage-independent Fourier basis and downward closure permit an exact orthonormal triangular reduction, while the rank-one whitened rows expose both exact leverage and the second moment needed for a mean-cancelled inverse-Gram expansion.

\paragraph{Fill.} The project derives the exact all-order criterion, proves uniform dosage with a global modulus whose coefficient is sharp when \(L\ge p/2\), and couples the design to exact leverage, rank-failure probability, the exact spectral-fallback decomposition, and an explicit uniform \(O(n^{-1})\) fixed-threshold oracle-risk bound.

\section{Related work}

Shyamal, Zhang, and Uhler introduce the same product-Bernoulli factorial model and bounded-order Fourier regression, but their limited-supply closed form is additive-only. Schwabe and Wong provide the classical downward-closed product-design inverse-trace identity specialized here, so that identity is not presented as new. Tropp supplies the matrix-Chernoff engine for the empirical Gram event; the present second-order inverse calculation additionally uses the exact product-Fourier leverage bound and mean-zero whitened Gram to obtain the explicit relative-risk remainder. Hsu, Kakade, and Zhang motivate separating Gram conditioning from response noise in random-design regression. Dereziński and Warmuth obtain exact inverse-Gram expectations under volume sampling, a dependent design outside the admissible class here. Freise and Schwabe and Ravichandran and coauthors study restricted factorial regions or cell allocations under dependent randomization rather than independent coordinate dosage. Yao and coauthors are only an idealized high-multiplicity interaction-screen consumer: their application does not establish independent delivery or the homoskedastic Fourier regression assumed here.

\section{Assumptions}

\begin{assumption}[ass:factor-count]\label{ass:factor-count}
\(p\ge2\).
\par\smallskip\noindent\emph{Standard} (multifactor design, \cite{ShyamalZhangUhler2025}).
\end{assumption}

\begin{assumption}[ass:interaction-order]\label{ass:interaction-order}
\(k\in\{1,\ldots,p\}\).
\par\smallskip\noindent\emph{Standard} (bounded-order Fourier interaction model, \cite{ShyamalZhangUhler2025}).
\end{assumption}

\begin{assumption}[ass:sample-size]\label{ass:sample-size}
\(n\ge1\).
\par\smallskip\noindent\emph{Standard} (finite random-design sample, \cite{HsuKakadeZhang2014}).
\end{assumption}

\begin{assumption}[ass:supply-range]\label{ass:supply-range}
\(L\in(0,p]\).
\par\smallskip\noindent\emph{Standard} (expected-treatment supply bound, \cite{ShyamalZhangUhler2025}).
\end{assumption}

\begin{assumption}[ass:box-dosage]\label{ass:box-dosage}
\(d\in[0,1]^p\).
\par\smallskip\noindent\emph{Standard} (Bernoulli dosage box, \cite{ShyamalZhangUhler2025}).
\end{assumption}

\begin{assumption}[ass:expected-supply]\label{ass:expected-supply}
\(\sum_{i=1}^p d_i\le L\).
\par\smallskip\noindent\emph{Standard} (limited expected supply, \cite{ShyamalZhangUhler2025}).
\end{assumption}

\begin{assumption}[ass:product-randomization]\label{ass:product-randomization}
The variables \(Z_{ai}\) are mutually independent with \(Z_{ai}\sim\operatorname{Bernoulli}(d_i)\).
\par\smallskip\noindent\emph{Standard} (product-Bernoulli probabilistic factorial design, \cite{ShyamalZhangUhler2025}).
\end{assumption}

\begin{assumption}[ass:coefficient-ball]\label{ass:coefficient-ball}
\(\|\beta\|_2\le B\).
\par\smallskip\noindent\emph{Standard} (bounded Fourier coefficient class, \cite{ShyamalZhangUhler2025}).
\end{assumption}

\begin{assumption}[ass:linear-response]\label{ass:linear-response}
\(Y_a=\phi(X_a)^{\mathsf T}\beta+\epsilon_a\) for every \(a\in\{1,\ldots,n\}\).
\par\smallskip\noindent\emph{Standard} (bounded-order Fourier regression, \cite{ShyamalZhangUhler2025}).
\end{assumption}

\begin{assumption}[ass:conditional-centering]\label{ass:conditional-centering}
\(\mathbb E[\epsilon\mid F]=0\).
\par\smallskip\noindent\emph{Standard} (conditionally centered regression noise, \cite{HsuKakadeZhang2014}).
\end{assumption}

\begin{assumption}[ass:conditional-homoskedasticity]\label{ass:conditional-homoskedasticity}
\(\operatorname{Cov}(\epsilon\mid F)=\sigma^2I_n\).
\par\smallskip\noindent\emph{Standard} (conditionally homoskedastic uncorrelated noise, \cite{ShyamalZhangUhler2025}).
\end{assumption}

\begin{assumption}[ass:variance-range]\label{ass:variance-range}
\(\sigma\in[\sigma_{\min},\sigma_{\max}]\).
\par\smallskip\noindent\emph{Standard} (positive bounded noise variance, \cite{HsuKakadeZhang2014}).
\end{assumption}

\begin{assumption}[ass:spectral-threshold]\label{ass:spectral-threshold}
\(0<\varepsilon<1/2\).
\par\smallskip\noindent\emph{Standard} (relative Gram spectral threshold, \cite{Tropp2012}).
\end{assumption}

\begin{assumption}[ass:fixed-dimension]\label{ass:fixed-dimension}
\(p,k,L,B,\sigma_{\min},\sigma_{\max}\) remain fixed as \(n\to\infty\).
\par\smallskip\noindent\emph{Standard} (fixed-dimensional random-design asymptotics, \cite{HsuKakadeZhang2014}).
\end{assumption}

\section{Main results}

\begin{lemma}[lem:trace-identity]\label{lem:trace-identity}
For every \(d\in\mathcal D_L\cap(0,1)^p\), the population Gram matrix is positive definite and \[\Phi_k(d)=\sum_{T\in\mathcal I_k}\prod_{i\in T}r(d_i)=\sum_{j=0}^{k}e_j(r(d_1),\ldots,r(d_p)).\] If any coordinate of \(d\) equals zero or one, then \(\Sigma_k(d)\) is singular and \(\Phi_k(d)=+\infty\). The interior identity is obtained by the triangular change from the fixed monomials to product-orthonormal features and is the Bernoulli specialization of the downward-closed product-design identity of Schwabe and Wong.
\end{lemma}
\begin{proof}
Put \(m_i=\mathbb E_d[X_{ai}]=2d_i-1\) and \(v_i=1-m_i^2=4d_i(1-d_i)\).  First let \(d\in\mathcal D_L\cap(0,1)^p\).  Then \(v_i>0\) for every \(i\).  For each \(T\in\mathcal I_k\), define the proof-intermediate feature
\[
 \psi_T(X_a)=\prod_{i\in T}\frac{X_{ai}-m_i}{\sqrt{v_i}}.
\]
The product-randomization assumption makes the coordinate factors independent.  Each factor has mean zero and second moment one.  Hence, for \(T,U\in\mathcal I_k\),
\[
 \mathbb E_d[\psi_T(X_a)\psi_U(X_a)]=\mathbf 1\{T=U\}.
\]
Indeed, if \(T\ne U\), any coordinate in \(T\mathbin\triangle U\) occurs exactly once in the product and contributes its zero mean; if \(T=U\), independence and unit second moments give one.

Expanding the centered products gives
\[
 \psi_T
 =\sum_{S\subseteq T}
 \frac{\prod_{i\in T\setminus S}(-m_i)}
      {\prod_{i\in T}\sqrt{v_i}}\,\phi_S.
\]
Because \(\mathcal I_k\) is downward closed, every \(S\subseteq T\in\mathcal I_k\) is again retained.  Thus \(\psi=A\phi\), where
\[
 A_{T,S}=\mathbf 1\{S\subseteq T\}
 \frac{\prod_{i\in T\setminus S}(-m_i)}
      {\prod_{i\in T}\sqrt{v_i}}.
\]
Order \(\mathcal I_k\) compatibly with inclusion.  The matrix \(A\) is triangular and
\[
 A_{T,T}=\prod_{i\in T}v_i^{-1/2}>0,
\]
so \(A\) is invertible.  Orthonormality of the \(\psi_T\)'s and the definition of \(\Sigma_k(d)\) imply
\[
 A\Sigma_k(d)A^{\mathsf T}=I_q.
\]
Consequently \(\Sigma_k(d)=A^{-1}A^{-\mathsf T}\) is positive definite and
\[
 \Sigma_k(d)^{-1}=A^{\mathsf T}A.
\]
Taking traces and using the displayed entries of \(A\),
\[
\begin{aligned}
 \Phi_k(d)
 &=\operatorname{tr}(A^{\mathsf T}A)\\
 &=\sum_{T\in\mathcal I_k}\sum_{S\subseteq T}A_{T,S}^2\\
 &=\sum_{T\in\mathcal I_k}
   \frac{1}{\prod_{i\in T}v_i}
   \sum_{S\subseteq T}\prod_{i\in T\setminus S}m_i^2\\
 &=\sum_{T\in\mathcal I_k}
   \prod_{i\in T}\frac{1+m_i^2}{1-m_i^2}.
\end{aligned}
\]
The third-to-fourth-line transition is the finite product expansion
\(\sum_{S\subseteq T}\prod_{i\in T\setminus S}m_i^2
=\prod_{i\in T}(1+m_i^2)\).  Since \(m_i=2d_i-1\),
\[
 \frac{1+m_i^2}{1-m_i^2}
 =\frac{d_i^2+(1-d_i)^2}{2d_i(1-d_i)}
 =\frac{1}{2d_i(1-d_i)}-1
 =r(d_i).
\]
It follows that
\[
 \Phi_k(d)=\sum_{T\in\mathcal I_k}\prod_{i\in T}r(d_i)
 =\sum_{j=0}^{k}e_j\bigl(r(d_1),\ldots,r(d_p)\bigr),
\]
where the last equality groups the retained subsets by cardinality.  This direct triangular calculation also specializes the cited Schwabe--Wong identity: the marginal information matrix for \((1,X_i)\) is
\[
 M_i=\begin{pmatrix}1&m_i\\m_i&1\end{pmatrix},
 \qquad
 \operatorname{tr}(M_i^{-1})-1
 =\frac{1+m_i^2}{1-m_i^2}=r(d_i),
\]
and substitution in that downward-closed product-design identity yields the same sum.

It remains to treat the boundary.  If \(d_i=0\), then \(X_{ai}=-1\) almost surely, so the column \(\phi_{\{i\}}\) equals minus the intercept column almost surely.  If \(d_i=1\), the same singleton column equals the intercept column almost surely.  The interaction-order assumption gives \(k\ge1\), so both columns belong to \(\mathcal I_k\).  In either case the feature vector satisfies a nonzero deterministic linear relation, and therefore \(\Sigma_k(d)=\mathbb E_d[\phi(X_a)\phi(X_a)^{\mathsf T}]\) is singular.  The extended-real definition of \(\Phi_k\) then gives \(\Phi_k(d)=+\infty\).

As the required smallest exact check, take \(p=3\) and \(k=2\).  At \(d=(1/3,1/3,1/3)\), each coordinate factor is \(5/4\), and the proved identity gives
\[
 \Phi_2(d)=1+3\frac54+3\left(\frac54\right)^2=\frac{151}{16}.
\]
At \(d=(1/4,1/3,5/12)\), the three factors are \(5/3\), \(5/4\), and \(37/35\), whence
\[
 \Phi_2(d)=1+\frac{1669}{420}+\frac{217}{42}=\frac{4259}{420}.
\]
Thus both direct inverse-trace checks agree with the general formula.
\end{proof}
\par\smallskip\noindent \textit{Justification.} This representation removes the higher-order characteristic-polynomial obstruction and exposes separable curvature. \textit{Gap.} SchwabeWong1999 contains the general identity, while ShyamalZhangUhler2025 does not specialize it to solve the shared-supply all-order problem. \textit{Consumer.} The identity turns the interaction-design problem into an explicit scalar symmetric-polynomial optimization usable by the dosage theorem.

\begin{theorem}[thm:uniform-optimum]\label{thm:uniform-optimum}
The unique minimizer of \(\Phi_k(d)\) over \(d\in\mathcal D_L\) is \(d^\star=u\mathbf 1_p\), where \(u=\min\{L/p,1/2\}\), and \[\Phi^\star=\sum_{j=0}^{k}\binom pj r(u)^j.\] Moreover, every \(d\in\mathcal D_L\) satisfies \[\Phi_k(d)-\Phi^\star\ge 8C_{p,k}\|d-d^\star\|_2^2.\] The coefficient \(8C_{p,k}\) is the best global strong-convexity certificate when \(L\ge p/2\).
\end{theorem}
\begin{proof}
By \texttt{ass:supply-range} and \texttt{def:uniform-dosage},
\(u=\min\{L/p,1/2\}\in(0,1/2]\), and \(d^\star=u\mathbf 1_p\) is an interior point of the cube and belongs to \(\mathcal D_L\).  Indeed, if \(L<p/2\), then \(\sum_i d_i^\star=pu=L\), whereas if \(L\ge p/2\), then \(\sum_i d_i^\star=p/2\le L\).

We first establish the curvature certificate on the open cube.  For \(0<v<1\), put \(m=2v-1\).  The definition \texttt{def:coordinate-factor} gives
\[
 r(v)=\frac{1+m^2}{1-m^2}\ge 1,
\qquad
 r'(v)=-\frac{1-2v}{2v^2(1-v)^2},
\]
where \(|m|<1\) makes the first denominator positive.  Direct differentiation also gives the exact log-curvature identity
\[
 (\log r)''(v)
 =16\frac{1+3m^4}{(1-m^4)^2}
 =16+16\frac{m^4(5-m^4)}{(1-m^4)^2}
 \ge 16. \tag{1}
\]
The last inequality follows from \(0\le m^4<1\), so every factor and denominator in the remainder is nonnegative or positive as appropriate.

By the already established \texttt{lem:trace-identity}, on \((0,1)^p\) one has
\[
 \Phi_k(d)=\sum_{T\in\mathcal I_k} f_T(d),
 \qquad
 f_T(d):=\prod_{i\in T}r(d_i).
\]
For a nonempty \(T\), define the vector \(a_T(d)\in\mathbb R^p\) and diagonal matrix \(D_T(d)\in\mathbb R^{p\times p}\) by
\[
 (a_T(d))_i=\mathbf 1\{i\in T\}(\log r)'(d_i),
 \qquad
 (D_T(d))_{ii}=\mathbf 1\{i\in T\}(\log r)''(d_i).
\]
Twice differentiating the finite product yields
\[
 \nabla^2 f_T(d)
 =f_T(d)\bigl(D_T(d)+a_T(d)a_T(d)^{\mathsf T}\bigr).
\]
The matrix \(a_Ta_T^{\mathsf T}\) is positive semidefinite.  Moreover, \(f_T(d)\ge1\) by the first display and every active diagonal entry of \(D_T\) is at least \(16\) by (1).  Hence
\[
 \nabla^2 f_T(d)\succeq
 16\operatorname{diag}\bigl((\mathbf 1\{i\in T\})_{i=1}^p\bigr). \tag{2}
\]
For a fixed coordinate \(i\), the number of nonempty sets \(T\in\mathcal I_k\) containing \(i\) is, by \texttt{def:index-family} and \texttt{def:curvature-multiplicity},
\[
 \sum_{h=0}^{k-1}\binom{p-1}{h}=C_{p,k};
\]
one chooses the other \(h=|T|-1\) elements from the remaining \(p-1\) coordinates.  Summing (2), while noting that the empty-set summand is constant, proves
\[
 \nabla^2\Phi_k(d)\succeq16C_{p,k}I_p
 \qquad(d\in(0,1)^p). \tag{3}
\]

Let \(d\in\mathcal D_L\cap(0,1)^p\), set \(\delta=d-d^\star\), and apply the one-dimensional fundamental theorem of calculus twice to \(g(t)=\Phi_k(d^\star+t\delta)\).  The whole segment lies in the open cube, and (3) gives
\[
\begin{aligned}
 \Phi_k(d)-\Phi_k(d^\star)
 &=\nabla\Phi_k(d^\star)^{\mathsf T}\delta
   +\int_0^1(1-t)\,\delta^{\mathsf T}
       \nabla^2\Phi_k(d^\star+t\delta)\delta\,dt\\
 &\ge \nabla\Phi_k(d^\star)^{\mathsf T}\delta
   +8C_{p,k}\|\delta\|_2^2. \tag{4}
\end{aligned}
\]
It remains to verify the sign of the first-order term rather than discard it without justification.  Differentiating the trace identity and using the fact that all coordinates of \(d^\star\) equal \(u\) gives, for every \(i\),
\[
 \frac{\partial\Phi_k}{\partial d_i}(d^\star)
 =r'(u)\sum_{h=0}^{k-1}\binom{p-1}{h}r(u)^h
 =:\gamma_u. \tag{5}
\]
If \(u=1/2\), the displayed formula for \(r'\) gives \(\gamma_u=0\).  If \(u=L/p<1/2\), then \(r'(u)<0\), while the sum in (5) is strictly positive because its \(h=0\) term is one; thus \(\gamma_u<0\).  Feasibility from \texttt{ass:expected-supply} gives
\[
 \sum_{i=1}^p\delta_i=\sum_{i=1}^p d_i-pu\le L-L=0,
\]
and consequently
\(\nabla\Phi_k(d^\star)^{\mathsf T}\delta
=\gamma_u\sum_i\delta_i\ge0\).  In either supply regime, (4) therefore proves
\[
 \Phi_k(d)-\Phi_k(d^\star)
 \ge8C_{p,k}\|d-d^\star\|_2^2. \tag{6}
\]
If \(d\) is on the boundary of \([0,1]^p\), at least one coordinate is zero or one, and \texttt{lem:trace-identity} gives \(\Phi_k(d)=+\infty\); hence (6) remains true in the extended-real sense.  Since \(C_{p,k}\ge1\), (6) is strict for every \(d\ne d^\star\), proving that \(d^\star\) is the unique minimizer over \(\mathcal D_L\).

Finally, substitution of the uniform dosage into the trace identity, followed by grouping sets according to their cardinality, gives the claimed exact value:
\[
 \Phi^\star=\Phi_k(d^\star)
 =\sum_{T\in\mathcal I_k}r(u)^{|T|}
 =\sum_{j=0}^k\binom pj r(u)^j. \tag{7}
\]

It remains to prove sharpness when \(L\ge p/2\).  Then \(d^\star=(1/2)\mathbf 1_p\).  At this point every \(f_T=1\), every \(a_T=0\), and every active diagonal entry of \(D_T\) equals \(16\) by (1).  The same counting used in (3) therefore gives the equality
\[
 \nabla^2\Phi_k(d^\star)=16C_{p,k}I_p. \tag{8}
\]
The assumption \(p\ge2\), inherited through \texttt{lem:trace-identity}, permits the nonzero direction \(h=e_1-e_2\).  For all sufficiently small \(|t|\), the point \(d(t)=d^\star+th\) lies in \((0,1)^p\), has coordinate sum \(p/2\le L\), and hence is feasible.  Applying the same twice-integrated identity as in (4), using \(\nabla\Phi_k(d^\star)=0\), continuity of the finite-sum Hessian, and (8), yields
\[
 \lim_{t\to0}
 \frac{\Phi_k(d(t))-\Phi_k(d^\star)}
      {\|d(t)-d^\star\|_2^2}
 =\frac{\frac12 h^{\mathsf T}(16C_{p,k}I_p)h}{\|h\|_2^2}
 =8C_{p,k}. \tag{9}
\]
Thus any larger coefficient in a global inequality of the form (6) would fail along this feasible path.  Together with (6), this proves that \(8C_{p,k}\) is the best global quadratic certificate when \(L\ge p/2\).
\end{proof}
\par\smallskip\noindent \textit{Justification.} This is the exact design choice and a quantitative certificate of the loss from dosage imbalance. \textit{Gap.} ShyamalZhangUhler2025 proves the limited-supply closed form only for \(k=1\); SchwabeWong1999 does not impose or optimize the shared expected-supply simplex. \textit{Consumer.} An independent-delivery interaction screen can set all dosages to \(u\) and quantify the population precision penalty of implementation error.

\begin{proposition}[prop:exact-leverage]\label{prop:exact-leverage}
At the optimal dosage, the exact maximum whitened leverage is \[\mu^\star=\sum_{j=0}^{k}\binom pj\left(\frac{1-u}{u}\right)^j.\]
\end{proposition}
\begin{proof}
By \(\textnormal{ass:supply-range}\) and \(\textnormal{def:uniform-dosage}\),
\[
 0<u=\min\{L/p,1/2\}\le \frac12<1,
\]
so \(d^\star=u\mathbf 1_p\) is an interior dosage.  In particular, the positive-definiteness conclusion of \(\textnormal{lem:trace-identity}\) applies to \(\Sigma_k(d^\star)\).

We make the triangular change of basis explicit.  Put
\[
 m=2u-1,
 \qquad
 s=\sqrt{4u(1-u)}>0,
 \qquad
 g_i(x_i)=\frac{x_i-m}{s},
 \qquad
 \psi_T(x)=\prod_{i\in T}g_i(x_i)
 \quad(T\in\mathcal I_k).
\]
Under the product randomization at \(d^\star\), the coordinates \(X_i\) are independent and satisfy
\[
 \mathbb E_{d^\star}X_i=m,
 \qquad
 \mathbb E_{d^\star}(X_i-m)^2=s^2.
\]
Consequently \(\mathbb E_{d^\star}g_i(X_i)=0\) and \(\mathbb E_{d^\star}g_i(X_i)^2=1\).  Independence then gives, for \(T,U\in\mathcal I_k\),
\[
 \mathbb E_{d^\star}\{\psi_T(X)\psi_U(X)\}
 =\prod_{i\in T\cap U}\mathbb E_{d^\star}g_i(X_i)^2
   \prod_{i\in T\mathbin\triangle U}\mathbb E_{d^\star}g_i(X_i)
 =\mathbf 1\{T=U\}.
\]

For each \(T\in\mathcal I_k\), expansion of the product yields
\[
 \psi_T(x)
 =s^{-|T|}\prod_{i\in T}(x_i-m)
 =s^{-|T|}\sum_{R\subseteq T}(-m)^{|T\setminus R|}\phi_R(x).
\]
Because \(R\subseteq T\) implies \(R\in\mathcal I_k\), this defines a square matrix \(A\), indexed by \(\mathcal I_k\), such that \(\psi(x)=A\phi(x)\).  Ordering subsets by nondecreasing cardinality makes \(A\) triangular, and its diagonal entry indexed by \(T\) is \(s^{-|T|}>0\); hence \(A\) is invertible.  The preceding orthonormality calculation and the definition of \(\Sigma_k\) give
\[
 I_q
 =\mathbb E_{d^\star}\{\psi(X)\psi(X)^{\mathsf T}\}
 =A\Sigma_k(d^\star)A^{\mathsf T}.
\]
Therefore
\[
 \Sigma_k(d^\star)^{-1}=A^{\mathsf T}A,
\]
and, for every \(x\in\{-1,1\}^p\),
\[
 \phi(x)^{\mathsf T}\Sigma_k(d^\star)^{-1}\phi(x)
 =\|A\phi(x)\|_2^2
 =\sum_{T\in\mathcal I_k}\psi_T(x)^2.
\]

For each coordinate, direct substitution gives
\[
 g_i(x_i)^2
 =\begin{cases}
 \dfrac{1-u}{u},&x_i=1,\\[6pt]
 \dfrac{u}{1-u},&x_i=-1.
 \end{cases}
\]
Since \(0<u\le1/2\),
\[
 \frac{u}{1-u}\le\frac{1-u}{u}.
\]
Every term \(\psi_T(x)^2=\prod_{i\in T}g_i(x_i)^2\) is a product of nonnegative coordinate factors.  Thus all terms in their sum are simultaneously maximized at \(x=\mathbf 1_p\).  By \(\textnormal{def:maximum-leverage}\),
\[
 \mu^\star
 =\sum_{T\in\mathcal I_k}\prod_{i\in T}\frac{1-u}{u}
 =\sum_{j=0}^{k}\binom pj\left(\frac{1-u}{u}\right)^j,
\]
where the last equality groups subsets by cardinality.  This proves the asserted exact maximum.
\end{proof}
\par\smallskip\noindent \textit{Justification.} The leverage is the exact concentration scale of the whitened empirical Gram matrix. \textit{Gap.} Generic bounded-design analyses use dimension-dependent envelopes and do not compute this supply-sensitive Boolean leverage. \textit{Consumer.} The expression gives an explicit sample-size diagnostic before running a sparse-supply interaction experiment.

\begin{proposition}[prop:rank-obstruction]\label{prop:rank-obstruction}
For every fixed \(T\in\mathcal I_k\) with \( |T|=k\), if no sampled unit has \(Z_{ai}=1\) for all \(i\in T\), then \(F\) is rank deficient. Consequently, under \(d^\star\), \[\mathbb P_{d^\star}(\operatorname{rank}(F)<q)\ge(1-u^k)^n.\]
\end{proposition}
\begin{proof}
Fix \(T\subseteq[p]\) with \(|T|=k\). Because \(Z_{ai}=(1+X_{ai})/2\), the retained polynomial
\[
 g_T(X_a):=\prod_{i\in T}Z_{ai}
 =2^{-k}\prod_{i\in T}(1+X_{ai})
 =2^{-k}\sum_{S\subseteq T}\phi_S(X_a)
\]
is a nonzero linear combination of columns of \(F\); every \(S\subseteq T\) lies in \(\mathcal I_k\). If no sampled unit activates every coordinate of \(T\), then \(g_T(X_a)=0\) for every row \(a\). Hence the nonzero coefficient vector having entry \(2^{-k}\) on each \(S\subseteq T\) and zero elsewhere belongs to the null space of \(F\), so \(\operatorname{rank}(F)<q\).

Under \(d^\star\), product randomization gives
\[
 \mathbb P_{d^\star}\{Z_{ai}=1\text{ for all }i\in T\}=u^k
\]
for each unit. Independence across the \(n\) units then gives probability \((1-u^k)^n\) that no unit realizes this activation. That event is contained in \(\{\operatorname{rank}(F)<q\}\), proving
\[
 \mathbb P_{d^\star}\{\operatorname{rank}(F)<q\}\ge(1-u^k)^n.
\]
\end{proof}
\par\smallskip\noindent \textit{Justification.} This discrete null vector proves that rare joint activations are a necessary obstacle, not an artifact of matrix concentration. \textit{Gap.} HsuKakadeZhang2014 treats generic random-design conditioning and does not exploit Boolean activation polynomials. \textit{Consumer.} The bound warns a screen designer when a proposed sample size cannot reliably identify all \(k\)-fold coefficients.

\begin{theorem}[thm:risk-sandwich]\label{thm:risk-sandwich}
Under \(d^\star\), let \(b_{n,\varepsilon}=\mathbb P_{d^\star}(G_n(\varepsilon)^c)\). On \(G_n(\varepsilon)\),
\[
 \operatorname{Cov}(\widehat\beta_{\varepsilon}\mid F)
 =\sigma^2(F^{\mathsf T}F)^{-1},
\]
and
\[
 \frac{\sigma^2}{n(1+\varepsilon)}\Sigma_k(d^\star)^{-1}
 \preceq\operatorname{Cov}(\widehat\beta_{\varepsilon}\mid F)
 \preceq
 \frac{\sigma^2}{n(1-\varepsilon)}\Sigma_k(d^\star)^{-1}.
\]
Unconditionally,
\[
 R_n(\beta,\sigma,\varepsilon)
 =\sigma^2\mathbb E_{d^\star}\!\left[
 \mathbf 1_{G_n(\varepsilon)}
 \operatorname{tr}\{(F^{\mathsf T}F)^{-1}\}\right]
 +\|\beta\|_2^2b_{n,\varepsilon}.
\]
Furthermore,
\[
 b_{n,\varepsilon}\le
 \min\!\left\{1,2q\exp\!\left(-\frac{n\varepsilon^2}{3\mu^\star}\right)\right\}
 =\min\{1,\delta_{n,\varepsilon}\},
\]
and
\[
 \frac{\sigma^2\Phi^\star}{n}\frac{1-b_{n,\varepsilon}}{1+\varepsilon}
 \le R_n(\beta,\sigma,\varepsilon)
 \le \frac{\sigma^2\Phi^\star}{n(1-\varepsilon)}+B^2b_{n,\varepsilon}.
\]
In particular,
\[
 R_n(\beta,\sigma,\varepsilon)
 \le \frac{\sigma^2\Phi^\star}{n}(1+2\varepsilon)
 +2B^2q\exp\!\left(-\frac{n\varepsilon^2}{3\mu^\star}\right).
\]
For the fixed threshold \(\bar\varepsilon\in(0,1/2)\), the second-order inverse identity gives the sharper relative-risk bound
\[
\begin{aligned}
 \left|
 \frac{R_n(\beta,\sigma,\bar\varepsilon)}{\sigma^2\Phi^\star/n}-1
 \right|
 &\le
 \frac{\mu^\star-1}{(1-\bar\varepsilon)n}
 +\left(1+\mu^\star+
 \frac{n\|\beta\|_2^2}{\sigma^2\Phi^\star}\right)b_{n,\bar\varepsilon}\\
 &\le
 \frac{\mu^\star-1}{(1-\bar\varepsilon)n}
 +2q\left(1+\mu^\star+
 \frac{nB^2}{\sigma_{\min}^2\Phi^\star}\right)
 \exp\!\left(-\frac{n\bar\varepsilon^2}{3\mu^\star}\right).
\end{aligned}
\]
\end{theorem}
\begin{proof}
Write \(\Sigma=\Sigma_k(d^\star)\).  The uniform-dosage definition and the supply range give \(0<u\le1/2\) and \(pu\le L\), so \(d^\star\in\mathcal D_L\cap(0,1)^p\).  Lemma \(\textnormal{lem:trace-identity}\) therefore gives
\[
 \Sigma\succ0,
 \qquad
 0<\Phi^\star=\operatorname{tr}(\Sigma^{-1})<\infty .
 \tag{1}
\]
Thus every inverse and inverse square root used below exists.

For each experimental unit define the proof-intermediate whitened row and rank-one matrix
\[
 w_a=\Sigma^{-1/2}x_a,
 \qquad
 K_a=w_aw_a^{\mathsf T}.
\]
Product randomization across units and the definition of \(\Sigma\) show that \(K_1,\ldots,K_n\) are independent positive-semidefinite random matrices and
\[
 \mathbb E_{d^\star}K_a
 =\Sigma^{-1/2}\mathbb E_{d^\star}(x_ax_a^{\mathsf T})\Sigma^{-1/2}
 =I_q.
 \tag{2}
\]
Moreover, Proposition \(\textnormal{prop:exact-leverage}\) and the definition of maximum leverage imply at every assignment point that
\[
 \lambda_{\max}(K_a)=\|w_a\|_2^2
 =x_a^{\mathsf T}\Sigma^{-1}x_a\le\mu^\star.
 \tag{3}
\]
In particular, \(0\preceq K_a\preceq\mu^\star I_q\).  The exact formula in Proposition \(\textnormal{prop:exact-leverage}\) also gives \(\mu^\star>1\).

Apply Lemma \(\textnormal{lem:tropp-matrix-chernoff}\) with \(R=\mu^\star\).  Equation \((2)\) gives
\[
 \sum_{a=1}^n\mathbb E K_a=nI_q,
 \qquad
 \mu_{\min}=\mu_{\max}=n.
 \tag{4}
\]
For \(0\le t\le1/2\), put
\[
 h_+(t)=(1+t)\log(1+t)-t,
 \qquad
 h_-(t)=t+(1-t)\log(1-t).
\]
Each function and its first derivative vanish at zero, whereas
\[
 h_+''(t)=\frac1{1+t}\ge\frac23,
 \qquad
 h_-''(t)=\frac1{1-t}\ge1.
\]
Integrating twice from zero yields \(h_+(t)\ge t^2/3\) and \(h_-(t)\ge t^2/2\ge t^2/3\).  On the other hand,
\[
 \frac1n\sum_{a=1}^nK_a
 =\Sigma^{-1/2}(F^{\mathsf T}F/n)\Sigma^{-1/2}
 =I_q+H_n(\varepsilon).
 \tag{5}
\]
The two matrix-Chernoff inequalities and the union bound therefore give, for \(0<\varepsilon<1/2\),
\[
\begin{aligned}
 \mathbb P_{d^\star}\!\left\{\lambda_{\min}\!\left(\frac1n\sum_{a=1}^nK_a\right)\le1-\varepsilon\right\}
 &\le q\exp\!\left(-\frac{n\varepsilon^2}{3\mu^\star}\right),\\
 \mathbb P_{d^\star}\!\left\{\lambda_{\max}\!\left(\frac1n\sum_{a=1}^nK_a\right)\ge1+\varepsilon\right\}
 &\le q\exp\!\left(-\frac{n\varepsilon^2}{3\mu^\star}\right).
\end{aligned}
\]
Consequently,
\[
 b_{n,\varepsilon}
 \le\min\!\left\{1,2q\exp\!\left(-\frac{n\varepsilon^2}{3\mu^\star}\right)\right\}
 =\min\{1,\delta_{n,\varepsilon}\}.
 \tag{6}
\]
This derives the stated concentration constant from independence, isotropy, and the exact leverage bound.

On \(G_n(\varepsilon)\), equation \((5)\) and the operator-norm definition of this event imply
\[
 (1-\varepsilon)\Sigma
 \preceq F^{\mathsf T}F/n
 \preceq(1+\varepsilon)\Sigma.
 \tag{7}
\]
Since \(\varepsilon<1\) and \(\Sigma\succ0\), the empirical Gram matrix is invertible on this event.  Inversion reverses Loewner order, and hence
\[
 \frac1{n(1+\varepsilon)}\Sigma^{-1}
 \preceq(F^{\mathsf T}F)^{-1}
 \preceq\frac1{n(1-\varepsilon)}\Sigma^{-1}.
 \tag{8}
\]
The linear-response assumption and the estimator definition give, conditionally on any \(F\in G_n(\varepsilon)\),
\[
 \widehat\beta_\varepsilon-\beta
 =(F^{\mathsf T}F)^{-1}F^{\mathsf T}\epsilon.
 \tag{9}
\]
Conditional centering makes the right-hand side conditionally mean zero.  Conditional homoskedasticity then gives
\[
\begin{aligned}
 \operatorname{Cov}(\widehat\beta_\varepsilon\mid F)
 &=(F^{\mathsf T}F)^{-1}F^{\mathsf T}(\sigma^2I_n)F(F^{\mathsf T}F)^{-1}\\
 &=\sigma^2(F^{\mathsf T}F)^{-1}.
\end{aligned}
 \tag{10}
\]
Multiplying \((8)\) by \(\sigma^2>0\) proves the asserted conditional covariance sandwich.

The causal coefficient vector is distinct from its observed-law identifying functional.  Indeed, linear response, iterated expectation, and conditional centering give
\[
\begin{aligned}
 \mathbb E_{d^\star}[x_aY_a]
 &=\mathbb E_{d^\star}[x_ax_a^{\mathsf T}]\beta
   +\mathbb E_{d^\star}[x_a\epsilon_a]\\
 &=\Sigma\beta+
 \mathbb E_{d^\star}\!\left[x_a\mathbb E(\epsilon_a\mid F)\right]
 =\Sigma\beta.
\end{aligned}
\]
Because \(\Sigma\) is invertible by \((1)\), this proves, rather than defines, the relationship \(\beta=\Sigma^{-1}\mathbb E_{d^\star}[x_aY_a]\).

The event \(G_n(\varepsilon)\) is measurable with respect to \(F\).  On that event, equations \((9)\)--\((10)\) imply
\[
 \mathbb E\!\left(\|\widehat\beta_\varepsilon-\beta\|_2^2\mid F\right)
 =\sigma^2\operatorname{tr}\{(F^{\mathsf T}F)^{-1}\}.
\]
On the complementary event the estimator equals zero, so its squared error is exactly \(\|\beta\|_2^2\).  Splitting the expectation over the two events yields the exact unconditional decomposition
\[
 R_n(\beta,\sigma,\varepsilon)
 =\sigma^2\mathbb E_{d^\star}\!\left[
 \mathbf 1_{G_n(\varepsilon)}\operatorname{tr}\{(F^{\mathsf T}F)^{-1}\}\right]
 +\|\beta\|_2^2b_{n,\varepsilon}.
 \tag{11}
\]
Thus the fallback bias is retained.  Taking traces in \((8)\), using \(\operatorname{tr}(\Sigma^{-1})=\Phi^\star\), and invoking \(\|\beta\|_2\le B\), we obtain
\[
 \frac{\sigma^2\Phi^\star}{n}\frac{1-b_{n,\varepsilon}}{1+\varepsilon}
 \le R_n(\beta,\sigma,\varepsilon)
 \le\frac{\sigma^2\Phi^\star}{n(1-\varepsilon)}+B^2b_{n,\varepsilon}.
 \tag{12}
\]
Since \((1-\varepsilon)^{-1}\le1+2\varepsilon\) for \(0<\varepsilon<1/2\), substituting \((6)\) into the upper bound in \((12)\) proves the first-order risk bound.

It remains to prove the fixed-threshold second-order assertion.  Fix \(\bar\varepsilon\in(0,1/2)\) as in Definition \(\textnormal{def:oracle-threshold}\), abbreviate
\[
 \overline G=G_n(\bar\varepsilon),
 \qquad b=b_{n,\bar\varepsilon},
 \qquad H=H_n(\bar\varepsilon),
 \qquad D=\Sigma^{-1}.
\]
Equations \((2)\) and \((5)\) imply
\[
 H=\frac1n\sum_{a=1}^n(K_a-I_q),
 \qquad \mathbb EH=0.
 \tag{13}
\]
Because \(K_a\) has rank one, \(K_a^2=\|w_a\|_2^2K_a\).  Equation \((3)\) therefore yields
\[
 K_a^2\preceq\mu^\star K_a,
 \qquad
 \mathbb E(K_a-I_q)^2=\mathbb EK_a^2-I_q
 \preceq(\mu^\star-1)I_q.
 \tag{14}
\]
For \(a\ne c\), independence and centering give \(\mathbb E[(K_a-I_q)(K_c-I_q)]=0\).  Expanding the square in \((13)\) and applying \((14)\) gives
\[
 \mathbb EH^2
 =\frac1{n^2}\sum_{a=1}^n\mathbb E(K_a-I_q)^2
 \preceq\frac{\mu^\star-1}{n}I_q.
 \tag{15}
\]

On \(\overline G\), the spectrum of \(I_q+H\) is contained in \([1-\bar\varepsilon,1+\bar\varepsilon]\).  Direct multiplication by \(I_q+H\) verifies the exact identity
\[
 (I_q+H)^{-1}=I_q-H+H^2(I_q+H)^{-1}.
 \tag{16}
\]
Since \(H\) commutes with \((I_q+H)^{-1}\), functional calculus gives
\[
 0\preceq H^2(I_q+H)^{-1}
 \preceq\frac1{1-\bar\varepsilon}H^2
 \quad\text{on }\overline G.
 \tag{17}
\]
Because \(D\succeq0\), equations \((15)\)--\((17)\) imply
\[
 0\le
 \mathbb E\!\left[\mathbf 1_{\overline G}
 \operatorname{tr}\{DH^2(I_q+H)^{-1}\}\right]
 \le\frac{\mu^\star-1}{(1-\bar\varepsilon)n}\operatorname{tr}(D).
 \tag{18}
\]
Indeed, after using \((17)\), dropping the indicator can only increase the trace, and \((15)\) applies after expectation.

The pointwise bounds \(0\preceq n^{-1}\sum_aK_a\preceq\mu^\star I_q\), obtained from \((3)\), imply
\[
 -I_q\preceq H\preceq(\mu^\star-1)I_q,
 \qquad
 \|H\|_{\mathrm{op}}\le\mu^\star.
\]
Consequently,
\[
 |\operatorname{tr}(DH)|
 \le\|H\|_{\mathrm{op}}\operatorname{tr}(D)
 \le\mu^\star\Phi^\star.
 \tag{19}
\]
Taking the \(D\)-weighted trace of \((16)\), restricting to \(\overline G\), and using \(\mathbb E\operatorname{tr}(DH)=0\) from \((13)\), we find
\[
\begin{aligned}
 &\mathbb E\!\left[\mathbf 1_{\overline G}
 \operatorname{tr}\{D(I_q+H)^{-1}\}\right]-\Phi^\star\\
 &\quad=-b\Phi^\star
 +\mathbb E\!\left[\mathbf 1_{\overline G^c}\operatorname{tr}(DH)\right]
 +\mathbb E\!\left[\mathbf 1_{\overline G}
 \operatorname{tr}\{DH^2(I_q+H)^{-1}\}\right].
\end{aligned}
 \tag{20}
\]
Equations \((18)\)--\((20)\) and the triangle inequality prove
\[
 \left|
 \frac{\mathbb E[\mathbf 1_{\overline G}
 \operatorname{tr}\{D(I_q+H)^{-1}\}]}{\Phi^\star}-1
 \right|
 \le\frac{\mu^\star-1}{(1-\bar\varepsilon)n}
 +(1+\mu^\star)b.
 \tag{21}
\]
Finally, on \(\overline G\), congruence in \((5)\) gives
\[
 \operatorname{tr}\{(F^{\mathsf T}F)^{-1}\}
 =\frac1n\operatorname{tr}\{D(I_q+H)^{-1}\}.
\]
Substitute this identity into \((11)\), divide by the positive quantity \(\sigma^2\Phi^\star/n\), and apply \((21)\).  This yields
\[
 \left|
 \frac{R_n(\beta,\sigma,\bar\varepsilon)}{\sigma^2\Phi^\star/n}-1
 \right|
 \le\frac{\mu^\star-1}{(1-\bar\varepsilon)n}
 +\left(1+\mu^\star+
 \frac{n\|\beta\|_2^2}{\sigma^2\Phi^\star}\right)b_{n,\bar\varepsilon}.
\]
Using \(\|\beta\|_2\le B\), \(\sigma\ge\sigma_{\min}>0\), and the fixed-threshold instance of \((6)\) proves the theorem's final displayed inequality.  The argument uses only the stipulated conditional first and second noise moments; it assumes neither conditional Gaussianity nor finite-sample optimality over dosage designs.
\end{proof}
\par\smallskip\noindent \textit{Justification.} The theorem verifies the exact Chernoff constant, preserves fallback bias, and exploits mean-zero whitened-Gram fluctuations to give an explicit second-order relative-risk error. \textit{Gap.} Tropp supplies the matrix tail inequality and general random-design regression work supplies conditioning templates, but neither gives this product-Fourier leverage calibration, fallback branch, or fixed-threshold second-order constant. \textit{Consumer.} The result turns the exact population design into an auditable finite-sample precision guarantee and supplies the quantitative input for fixed-dimensional oracle attainment.

\begin{theorem}[thm:oracle-attainment]\label{thm:oracle-attainment}
For any fixed \(\bar\varepsilon\in(0,1/2)\), with \(p,k,L,B,\sigma_{\min},\sigma_{\max}\) fixed, for every \(n\ge1\),
\[
\begin{aligned}
 &\sup_{\substack{\|\beta\|_2\le B,\;\sigma\in[\sigma_{\min},\sigma_{\max}]\\
 \textnormal{admissible conditional-noise laws}}}
 \left|
 \frac{R_n(\beta,\sigma,\bar\varepsilon)}{\sigma^2\Phi^\star/n}-1
 \right|\\
 &\qquad\le
 \frac{\mu^\star-1}{(1-\bar\varepsilon)n}
 +2q\left(1+\mu^\star+
 \frac{nB^2}{\sigma_{\min}^2\Phi^\star}\right)
 \exp\!\left(-\frac{n\bar\varepsilon^2}{3\mu^\star}\right)
 =O(n^{-1}).
\end{aligned}
\]
Consequently, the displayed supremum converges to zero. This is a uniform fixed-dimensional oracle-risk expansion for the specified estimator at \(d^\star\); it is not a finite-\(n\) minimax-optimality statement over dosage designs.
\end{theorem}
\begin{proof}
Fix \(\bar\varepsilon\in(0,1/2)\) according to Definition \(\textnormal{def:oracle-threshold}\).  Under the fixed-dimension assumption, \(q\), \(\mu^\star\), \(\Phi^\star\), \(B\), and \(\sigma_{\min}\) are constants as \(n\to\infty\).  The trace and exact-leverage results used in Theorem \(\textnormal{thm:risk-sandwich}\) give
\[
 1<\mu^\star<\infty,
 \qquad
 0<\Phi^\star<\infty.
\]
Hence \(\sigma^2\Phi^\star/n>0\) uniformly for \(\sigma\in[\sigma_{\min},\sigma_{\max}]\).

For an admissible coefficient, noise scale, and conditional-noise law, define the proof-intermediate relative risk
\[
 \rho_n(\beta,\sigma)
 =\frac{R_n(\beta,\sigma,\bar\varepsilon)}{\sigma^2\Phi^\star/n}.
\]
The fixed-threshold conclusion of Theorem \(\textnormal{thm:risk-sandwich}\), together with \(\|\beta\|_2\le B\) and \(\sigma^2\ge\sigma_{\min}^2\), gives simultaneously over the entire displayed class
\[
 \left|\rho_n(\beta,\sigma)-1\right|
 \le
 \frac{\mu^\star-1}{(1-\bar\varepsilon)n}
 +2q\left(1+\mu^\star+
 \frac{nB^2}{\sigma_{\min}^2\Phi^\star}\right)
 \exp\!\left(-\frac{n\bar\varepsilon^2}{3\mu^\star}\right).
 \tag{1}
\]
The right-hand side is deterministic and independent of \(\beta\), \(\sigma\), and the particular admissible conditional-noise law.  Taking the stated supremum therefore preserves \((1)\).

Set \(c=\bar\varepsilon^2/(3\mu^\star)>0\).  For each fixed integer \(m\ge0\), differentiation of \(f_m(x)=x^me^{-cx}\) on \([1,\infty)\) shows that its only possible interior maximizer is \(x=m/c\), while \(f_m(x)\to0\) as \(x\to\infty\).  Thus \(ne^{-cn}\) and \(n^2e^{-cn}\) are bounded.  After multiplying the right-hand side of \((1)\) by \(n\), its first term is constant and its exponential terms are fixed linear combinations of \(ne^{-cn}\) and \(n^2e^{-cn}\).  The entire right-hand side of \((1)\) is therefore \(O(n^{-1})\), and in particular converges to zero.

This proves the explicit uniform fixed-dimensional oracle-risk expansion for the known-information, fixed-threshold estimator under independent product-Bernoulli delivery at \(d^\star\).  Because no lower bound over competing finite-sample dosage designs enters the argument, the conclusion is not a finite-\(n\) dosage-minimax statement.
\end{proof}
\par\smallskip\noindent \textit{Justification.} This gives a finite-sample, uniform \(O(n^{-1})\) relative-risk expansion for the fixed-threshold known-information estimator at the exact population-optimal dosage. \textit{Gap.} PFED does not provide an all-order shared-supply oracle constant, and the generic concentration references do not derive the mean-cancelled second-order remainder with this exact leverage and fallback term. \textit{Consumer.} An independent-delivery interaction screen can use a fixed conditioning threshold and read off a uniform finite-sample error bound around the population oracle risk, without interpreting it as finite-sample design minimaxity.

\begin{proposition}[prop:additive-reduction]\label{prop:additive-reduction}
When \(k=1\), the criterion reduces to \(\Phi_1(d)=1+\sum_{i=1}^p r(d_i)\), and the optimizer reduces to the uniform limited-supply dosage in the additive PFED setting. When \(L\ge p/2\), it further reduces to half dosage with \(\Phi^\star=1+p\).
\end{proposition}
\begin{proof}
Suppose \(k=1\).  By \texttt{def:index-family}, the retained family is
\(\mathcal I_1=\{\varnothing,\{1\},\ldots,\{p\}\}\).  Therefore the already established \texttt{lem:trace-identity} gives, for every interior feasible dosage,
\[
 \Phi_1(d)
 =\prod_{i\in\varnothing}r(d_i)
  +\sum_{i=1}^p\prod_{\ell\in\{i\}}r(d_\ell)
 =1+\sum_{i=1}^p r(d_i). \tag{1}
\]
At a boundary dosage the same trace lemma assigns \(\Phi_1(d)=+\infty\), consistently with the singular-design convention in \texttt{def:population-criterion}.

Applying \texttt{thm:uniform-optimum} with \(k=1\) shows that the unique optimizer is the uniform limited-supply dosage
\[
 d_i=u=\min\{L/p,1/2\}\qquad(1\le i\le p).
\]
This is precisely the additive product-Bernoulli allocation asserted in the proposition.  If \(L\ge p/2\), then \(u=1/2\), and \texttt{def:coordinate-factor} gives
\[
 r(1/2)=\frac{1}{2(1/2)(1/2)}-1=1.
\]
Substituting into (1), or equivalently into the exact value in \texttt{thm:uniform-optimum}, yields \(\Phi^\star=1+p\), as required.
\end{proof}
\par\smallskip\noindent \textit{Justification.} This is the required sanity reduction to the known additive and unconstrained designs. \textit{Gap.} The reduction is known from ShyamalZhangUhler2025; it is included as a fidelity check rather than a novelty claim. \textit{Consumer.} Readers can verify that the all-order theorem preserves the established additive benchmark.

\begin{lemma}[lem:schwabe-wong-product-trace]\label{lem:schwabe-wong-product-trace}
Let \(\Delta\) be a downward-closed collection of subsets and let \(\xi=\xi_1\otimes\cdots\otimes\xi_p\) be a product design for the associated interaction model, with nonsingular marginal information matrices \(M_i(\xi_i)\). Then the product-design inverse-trace identity of Schwabe and Wong is
\[
 \operatorname{tr}\{M(\xi)^{-1}\}
 =\sum_{T\in\Delta}\prod_{i\in T}
 \bigl[\operatorname{tr}\{M_i(\xi_i)^{-1}\}-1\bigr].
\]
\end{lemma}

\begin{lemma}[lem:tropp-matrix-chernoff]\label{lem:tropp-matrix-chernoff}
Let \(K_1,\ldots,K_n\) be independent positive-semidefinite random matrices of dimension \(q\), and suppose for a constant \(R>0\) that \(\lambda_{\max}(K_a)\le R\) almost surely. Define
\[
 \mu_{\min}=\lambda_{\min}\!\left(\sum_{a=1}^n\mathbb E K_a\right),
 \qquad
 \mu_{\max}=\lambda_{\max}\!\left(\sum_{a=1}^n\mathbb E K_a\right).
\]
For \(0<\varepsilon<1\),
\[
\begin{aligned}
 \mathbb P\!\left\{\lambda_{\min}\!\left(\sum_{a=1}^nK_a\right)\le(1-\varepsilon)\mu_{\min}\right\}
 &\le q\left[\frac{e^{-\varepsilon}}{(1-\varepsilon)^{1-\varepsilon}}\right]^{\mu_{\min}/R},\\
 \mathbb P\!\left\{\lambda_{\max}\!\left(\sum_{a=1}^nK_a\right)\ge(1+\varepsilon)\mu_{\max}\right\}
 &\le q\left[\frac{e^{\varepsilon}}{(1+\varepsilon)^{1+\varepsilon}}\right]^{\mu_{\max}/R}.
\end{aligned}
\]
\end{lemma}

\section*{Technical internal limitation (diagnostic only)}

At \(\beta=0\), spectral fallback can make finite-sample truncated risk smaller than the population \(A\)-risk. Thus the exact population optimizer is not asserted to minimize finite-sample risk over designs, and no minimax claim follows from the risk sandwich.

\section{Honest scope}

The note establishes the exact population identity and optimizer, a global stability modulus whose coefficient is sharp only when \(L\ge p/2\), exact leverage, the rank obstruction, the finite-sample risk sandwich, and a fixed-threshold uniform \(O(n^{-1})\) oracle-risk expansion at fixed \(p,k,L,B,\sigma_{\min},\sigma_{\max}\). Exact finite-sample minimax dosage remains open. The core excludes dependent or fixed-count assignment, growing dimension, heteroskedastic outcomes, adaptive rounds, model misspecification, and uniformity as supply vanishes. The Yao application is only an idealized consumer under independent delivery and the stated regression model.
