For \(\eta=(m,\Gamma_1,\Gamma_2)\in\mathfrak I_{\Delta,p}^{\mathrm{WH}}\), set
\[
 r=\operatorname{rank}(\Gamma_1),\qquad S=\operatorname{im}(\Gamma_1),
\]
choose any \(O\in\mathbb R^{p\times r}\) with \(O^{\mathsf T}O=I_r\) and \(\operatorname{im}(O)=S\), and define
\[
 R=O^{\mathsf T}\Gamma_1,\quad
 F=O^{\mathsf T}\Gamma_2R^{\mathsf T}(RR^{\mathsf T})^{-1},
\]
\[
 L^{\mathrm{WH}}=\Delta^{-1}\operatorname{Log}(F),\quad
 H^{\mathrm{WH}}=L^{\mathrm{WH}}(F-I_r)^{-1},\quad
 P^{\mathrm{WH}}=(H^{\mathrm{WH}})^2R,
\]
\[
 K^{\mathrm{WH}}=OP^{\mathrm{WH}},\qquad
 C_{+}^{\mathrm{WH}}(t)=Oe^{L^{\mathrm{WH}}t}P^{\mathrm{WH}},\qquad t>0.
\]
Set \(\lambda^{\mathrm{WH}}=m/\Delta\), \(D^{\mathrm{WH}}=\operatorname{diag}(\lambda^{\mathrm{WH}})\), and
\[
 g^{\mathrm{WH}}(t)=
 \begin{cases}
 C_{+}^{\mathrm{WH}}(t)(D^{\mathrm{WH}})^{-1},&t>0,\\
 C_{+}^{\mathrm{WH}}(-t)^{\mathsf T}(D^{\mathrm{WH}})^{-1},&t<0,\\
 0,&t=0.
 \end{cases}
\]
For \(x>0\), define the finite-dimensional response transform
\[
 \mathcal R^{\mathrm{WH}}(x)
 =\int_0^\infty e^{-xt}C_{+}^{\mathrm{WH}}(t)^{\mathsf T}\,dt
 =(P^{\mathrm{WH}})^{\mathsf T}(xI_r-(L^{\mathrm{WH}})^{\mathsf T})^{-1}O^{\mathsf T}.
\]
Define \((\beta^{\#},A^{\#})\) as the unique pair \((x,M)\in(0,\infty)\times\mathbb R^{p\times p}\) satisfying
\[
 MO=O(L^{\mathrm{WH}}+xI_r),\qquad
 M[D^{\mathrm{WH}}+\mathcal R^{\mathrm{WH}}(x)]=K^{\mathrm{WH}}.
\]
Equivalently, put
\[
 \Omega^{\mathrm{WH}}
 =\{x>0:\det[D^{\mathrm{WH}}+\mathcal R^{\mathrm{WH}}(x)]\neq0\},
\qquad
 \mathcal A^{\mathrm{WH}}(x)
 =K^{\mathrm{WH}}[D^{\mathrm{WH}}+\mathcal R^{\mathrm{WH}}(x)]^{-1},
\]
let \(\beta^{\#}\) be the unique \(x\in\Omega^{\mathrm{WH}}\) such that
\[
 \mathcal A^{\mathrm{WH}}(x)O=O(L^{\mathrm{WH}}+xI_r),
\]
and set
\[
 A^{\#}=\mathcal A^{\mathrm{WH}}(\beta^{\#}),\qquad
 \mu^{\#}=(I_p-A^{\#}/\beta^{\#})\lambda^{\mathrm{WH}}.
\]
Finally define
\[
 \Phi_{\Delta,p}^{\mathrm{WH}}(m,\Gamma_1,\Gamma_2)
 =(\beta^{\#},A^{\#},\mu^{\#}).
\]
