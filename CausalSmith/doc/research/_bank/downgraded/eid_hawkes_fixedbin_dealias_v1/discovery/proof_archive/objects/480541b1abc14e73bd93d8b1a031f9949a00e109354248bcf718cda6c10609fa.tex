For \(\eta=(m,\Gamma_1,\Gamma_2)\in\mathfrak I_{\Delta,p}^{\mathrm{ib}}\), set
\[
 S=\operatorname{im}(\Gamma_1),\qquad
 I^\circ=\{i\in\{1,\ldots,p\}:e_i\in S\},\qquad
 r^\circ=|I^\circ|,
\]
and let \(E^\circ=E_{I^\circ}\) be the ordered coordinate selector.  With \(\Gamma_1^\dagger\) denoting the Moore--Penrose inverse, define
\[
 F^\circ=(E^\circ)^{\mathsf T}\Gamma_2\Gamma_1^\dagger E^\circ,\qquad
 L^\circ=\Delta^{-1}\operatorname{Log}(F^\circ),\qquad
 J^\circ=\int_0^\Delta e^{L^\circ u}\,du,
\]
\[
 G_1^\circ=(E^\circ)^{\mathsf T}\Gamma_1E^\circ,\qquad
 K^\circ=(J^\circ)^{-2}G_1^\circ,\qquad
 \lambda^\circ=m/\Delta,\qquad
 D_I^\circ=(E^\circ)^{\mathsf T}\operatorname{diag}(\lambda^\circ)E^\circ,
\]
\[
 Q^\circ=\int_0^\infty e^{L^\circ u}D_I^\circ e^{(L^\circ)^{\mathsf T}u}\,du,\qquad
 s^\circ=(r^\circ)^{-1}\operatorname{tr}\{K^\circ(Q^\circ)^{-1}+(L^\circ)^2\},
\]
\[
 \beta^\circ=(s^\circ)^{1/2},\qquad
 A^\circ=E^\circ(L^\circ+\beta^\circ I_{r^\circ})(E^\circ)^{\mathsf T},\qquad
 \mu^\circ=(I_p-A^\circ/\beta^\circ)\lambda^\circ.
\]
Here \((J^\circ)^{-2}=(J^\circ)^{-1}(J^\circ)^{-1}\), as required by the positive-lag identity \(\Gamma_1=J^2K\).  Define
\[
 \Phi_{\Delta,p}^{\mathrm{ib}}:\mathfrak I_{\Delta,p}^{\mathrm{ib}}
 \longrightarrow\mathbb R_+\times\mathbb R^{p\times p}\times\mathbb R^p,
 \qquad
 \Phi_{\Delta,p}^{\mathrm{ib}}(\eta)=(\beta^\circ,A^\circ,\mu^\circ).
\]
