For \(\eta=(m,\Gamma_1,\Gamma_2)\in\mathfrak I_{\Delta,p}^{0,+,\mathrm{rs}}\), set
\[
 r=\operatorname{rank}(\Gamma_1),\quad S=\operatorname{im}(\Gamma_1),
\]
choose any \(O\in\mathbb R^{p\times r}\) with \(O^{\mathsf T}O=I_r\) and \(\operatorname{im}(O)=S\), and define
\[
 R=O^{\mathsf T}\Gamma_1,\quad
 F=O^{\mathsf T}\Gamma_2R^{\mathsf T}(RR^{\mathsf T})^{-1},
\]
\[
 \lambda^\sharp=m/\Delta,\quad D^\sharp=\operatorname{diag}(\lambda^\sharp),
 \quad L^\sharp=\Delta^{-1}\operatorname{Log}(F),
\]
\[
 \beta^\sharp=-\operatorname{tr}(L^\sharp)/r,\quad
 A_S^\sharp=L^\sharp+\beta^\sharp I_r,\quad
 H^\sharp=L^\sharp(F-I_r)^{-1},
\]
\[
 K^\sharp=O(H^\sharp)^2R,\quad
 Z^\sharp=K^\sharp-(K^\sharp)^{\mathsf T},\quad
 P^\sharp=OO^{\mathsf T},\quad
 W^\sharp=O^{\mathsf T}(D^\sharp)^{-1}O,
\]
\[
 \mathsf C^\sharp=
 \{OA_S^\sharp-Z^\sharp(I_p-P^\sharp)(D^\sharp)^{-1}O\}
 (W^\sharp)^{-1}O^{\mathsf T}+Z^\sharp(I_p-P^\sharp),
\]
and
\[
 A^\sharp=\mathsf C^\sharp(D^\sharp)^{-1},\quad
 \mu^\sharp=(I_p-A^\sharp/\beta^\sharp)\lambda^\sharp.
\]
Define the domain-restricted map
\[
 \Phi_{\Delta,p}^{0,+,\mathrm{rs}}:\mathfrak I_{\Delta,p}^{0,+,\mathrm{rs}}
 \longrightarrow\mathbb R_+\times\mathbb R^{p\times p}\times\mathbb R^p,
 \quad
 \Phi_{\Delta,p}^{0,+,\mathrm{rs}}(\eta)
 =(\beta^\sharp,A^\sharp,\mu^\sharp).
\]
