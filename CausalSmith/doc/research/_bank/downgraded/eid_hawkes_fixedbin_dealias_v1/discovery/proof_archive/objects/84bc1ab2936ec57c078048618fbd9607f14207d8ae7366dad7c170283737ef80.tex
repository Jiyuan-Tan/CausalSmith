For \(\eta=(m,\Gamma_1,\Gamma_2)\in\mathfrak I_{\Delta,p}^{\mathrm{WH}}\), set
\[
 r=\operatorname{rank}(\Gamma_1),\qquad S=\operatorname{im}(\Gamma_1),
\]
choose any \(O\in\mathbb R^{p\times r}\) with \(O^{\mathsf T}O=I_r\) and \(\operatorname{im}(O)=S\), and define
\[
 R=O^{\mathsf T}\Gamma_1,\quad
 F=O^{\mathsf T}\Gamma_2R^{\mathsf T}(RR^{\mathsf T})^{-1},
\]
\[
 L^{\mathrm{WH}}=\Delta^{-1}\operatorname{Log}(F),\quad
 H^{\mathrm{WH}}=L^{\mathrm{WH}}(F-I_r)^{-1},\quad
 K^{\mathrm{WH}}=O(H^{\mathrm{WH}})^2R,
\]
and, for \(t>0\),
\[
 C_{+}^{\mathrm{WH}}(t)=Oe^{L^{\mathrm{WH}}t}(H^{\mathrm{WH}})^2R.
\]
Set \(\lambda^{\mathrm{WH}}=m/\Delta\), \(D^{\mathrm{WH}}=\operatorname{diag}(\lambda^{\mathrm{WH}})\), and
\[
 g^{\mathrm{WH}}(t)=
 \begin{cases}
 C_{+}^{\mathrm{WH}}(t)(D^{\mathrm{WH}})^{-1},&t>0,\\
 C_{+}^{\mathrm{WH}}(-t)^{\mathsf T}(D^{\mathrm{WH}})^{-1},&t<0,\\
 0,&t=0.
 \end{cases}
\]
Let \(\chi^{\mathrm{WH}}\) be the unique matrix function whose entries are causal and in \(L^1(\mathbb R_+)\) and which satisfies
\[
 g^{\mathrm{WH}}(t)=\chi^{\mathrm{WH}}(t)
 +(\chi^{\mathrm{WH}}*g^{\mathrm{WH}})(t),\qquad t>0,
\]
where
\[
 (\chi^{\mathrm{WH}}*g^{\mathrm{WH}})(t)
 =\int_0^\infty\chi^{\mathrm{WH}}(s)g^{\mathrm{WH}}(t-s)\,ds
\]
uses ordinary matrix multiplication inside the integral.  Finally set
\[
 M_0^{\mathrm{WH}}=\int_0^\infty\chi^{\mathrm{WH}}(t)\,dt,\quad
 M_1^{\mathrm{WH}}=\int_0^\infty t\chi^{\mathrm{WH}}(t)\,dt,
\]
\[
 u_0^{\mathrm{WH}}=\mathbf 1_p^{\mathsf T}M_0^{\mathrm{WH}}\mathbf 1_p,\quad
 u_1^{\mathrm{WH}}=\mathbf 1_p^{\mathsf T}M_1^{\mathrm{WH}}\mathbf 1_p,
\]
\[
 \beta^{\mathrm{WH}}=u_0^{\mathrm{WH}}/u_1^{\mathrm{WH}},\qquad
 A^{\mathrm{WH}}=\beta^{\mathrm{WH}}M_0^{\mathrm{WH}},\qquad
 \mu^{\mathrm{WH}}=(I_p-M_0^{\mathrm{WH}})\lambda^{\mathrm{WH}}.
\]
Define
\[
 \Phi_{\Delta,p}^{\mathrm{WH}}(m,\Gamma_1,\Gamma_2)
 =(\beta^{\mathrm{WH}},A^{\mathrm{WH}},\mu^{\mathrm{WH}}).
\]
