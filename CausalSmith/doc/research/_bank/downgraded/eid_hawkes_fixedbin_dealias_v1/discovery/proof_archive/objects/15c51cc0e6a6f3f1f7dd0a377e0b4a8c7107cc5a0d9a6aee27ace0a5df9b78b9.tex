\[
 r=\operatorname{rank}(\Gamma_1),\qquad S=\operatorname{im}(\Gamma_1),
\]

\[
 R=O^{\mathsf T}\Gamma_1,\qquad
 F=O^{\mathsf T}\Gamma_2R^{\mathsf T}(RR^{\mathsf T})^{-1},
\]
\[
 L^\sharp=\Delta^{-1}\operatorname{Log}(F),\qquad
 H^\sharp=L^\sharp(F-I_r)^{-1},\qquad
 \widetilde C_{+}^{\mathrm{rs}}(t)
 =Oe^{L^\sharp t}(H^\sharp)^2R,

\[
 D^0=\Delta^{-1}\!\left[
 \Gamma_0-\int_0^\Delta(\Delta-t)
 \{\widetilde C_{+}^{\mathrm{rs}}(t)
  +\widetilde C_{+}^{\mathrm{rs}}(t)^{\mathsf T}\}\,dt\right],
 \qquad \lambda^0=\operatorname{diag}(D^0),
 \qquad m^0=\Delta\lambda^0.

Define
\[
 \Psi_{\Delta,p}^{0,+,\mathrm{rs},0}(\gamma)
 =\Phi_{\Delta,p}^{0,+,\mathrm{rs}}(m^0,\Gamma_1,\Gamma_2).
