 q_{00}(x)=q_{11}(x)=\frac18,
 \qquad
 \frac1{40}\le q_{01}(x),q_{10}(x)\le\frac{29}{40},

 \mu_{D,P_{j,\ell}}^{-}(x)=\frac{1-w_{j,\ell}(x)}2,

 L_n(\kappa_n)

Along every admissible triangular sequence,

Let \(I_n\) be an arbitrary measurable connected interval rule satisfying the uniform coverage condition in \(\mathrm{def:minimax\mbox{-}length}\).  By \(\mathrm{prop:outcome\mbox{-}invariant\mbox{-}witness\mbox{-}high\mbox{-}separation}\), each component of \(Q_{+,n}\) is in \(\mathcal P_n\) and has causal target \(a_\star=7/10\), while each component of \(Q_{-,n}\) is in \(\mathcal P_n\) and has target \(-a_\star=-7/10\).  Averaging the componentwise honesty inequalities therefore gives
\[
 Q_{+,n}\{a_\star\in I_n\}\ge1-\alpha,
 \qquad
 Q_{-,n}\{-a_\star\in I_n\}\ge1-\alpha.
 \tag{31}
\]
For probability measures \(Q\ll R\), Cauchy--Schwarz applied to the density ratio yields the elementary inequality
\[
 \operatorname{TV}(Q,R)
 =\frac12E_R\left|\frac{dQ}{dR}-1\right|
 \le\frac12\sqrt{E_R\left(\frac{dQ}{dR}-1\right)^2}
 =\frac12\sqrt{\chi^2(Q,R)}.
 \tag{32}
\]
The triangle inequality for total variation, (32), and the equal chi-square identities for the two labels give
\[
 \operatorname{TV}(Q_{+,n},Q_{-,n})
 \le\operatorname{TV}(Q_{+,n},R_\star^{\otimes n})
    +\operatorname{TV}(Q_{-,n},R_\star^{\otimes n})
 \le
 \sqrt{\frac{\left(1+C_\psi\kappa_n^2h_n\right)^n-1}{J_n}}.
 \tag{33}
\]
Transfer the second event in (31) from \(Q_{-,n}\) to \(Q_{+,n}\) using the definition of total variation, and intersect it with the first event in (31).  The union bound and (33) imply
\[
 Q_{+,n}\{-a_\star,a_\star\in I_n\}
 \ge1-2\alpha-
 \sqrt{\frac{\left(1+C_\psi\kappa_n^2h_n\right)^n-1}{J_n}}.
 \tag{34}
\]
A connected subset of \(\Theta=[-1,1]\) containing both labels has Lebesgue length at least \(2a_\star=7/5\).  Since \(Q_{+,n}\) is the uniform mixture of in-class product laws,
\[
 \sup_{P\in\mathcal P_n}E_P[|I_n|]
 \ge E_{Q_{+,n}}[|I_n|]
 \ge\frac75\left[
 1-2\alpha-
 \sqrt{\frac{\left(1+C_\psi\kappa_n^2h_n\right)^n-1}{J_n}}
 \right]_+.
 \tag{35}
\]
Taking the infimum over all honest rules proves the finite-sample assertion.

We now evaluate (35).  From the definition of \(h_n\),
\[
 h_n=C_h\kappa_n^{1/s},
 \qquad
 n\kappa_n^2h_n=C_hT_n.
 \tag{36}
\]
Whenever \(h_n\le1/32\), the number \(x=(16h_n)^{-1}\) is at least two.  Since \(\lfloor x\rfloor\ge x/2\) for \(x\ge2\),
\[
 J_n\ge\frac1{32h_n}.
 \tag{37}
\]
Using \(1+x\le e^x\), (36)--(37) bound the chi-square term in (35) by
\[
 \frac{\left(1+C_\psi\kappa_n^2h_n\right)^n-1}{J_n}
 \le32h_n\exp\{C_\psi n\kappa_n^2h_n\}
 =32C_h\kappa_n^{1/s}
   \exp\{C_\psi C_hT_n\}.
 \tag{38}
\]
The logarithm of the final expression is
\[
 \log(32C_h)-\frac1s\log(1/\kappa_n)+C_\psi C_hT_n.
 \tag{39}
\]
If
\[
 \frac1s\log(1/\kappa_n)-C_\psi C_hT_n\longrightarrow+\infty,
 \tag{40}
\]
then (40) first implies \(\log(1/\kappa_n)\to\infty\), because \(T_n>0\).  Hence \(\kappa_n\to0\), so \(h_n\to0\) and (37)--(39) apply eventually.  Equation (39) tends to \(-\infty\); therefore the square-root term in (35) tends to zero.  This proves
\[
 \liminf_{n\to\infty}L_n(\kappa_n)
 \ge\frac75(1-2\alpha).
 \tag{41}
\]

For the ratio corollary, suppose
\[
 \limsup_n\frac{T_n}{\log(1/\kappa_n)}<\lambda_w.
 \tag{42}
\]
Condition (42) forces \(\kappa_n\to0\): otherwise, along a subsequence bounded away from zero, \(T_n=n\kappa_n^{2+1/s}\) diverges at least linearly while \(\log(1/\kappa_n)\) stays bounded, contradicting (42).  Choose \(\varepsilon>0\) so that eventually
\[
 \frac{T_n}{\log(1/\kappa_n)}\le\lambda_w-\varepsilon.
\]
Using \(C_\psi C_h\lambda_w=1/s\), we obtain
\[
 \frac1s\log(1/\kappa_n)-C_\psi C_hT_n
 \ge C_\psi C_h\varepsilon\log(1/\kappa_n)
 \longrightarrow+\infty.
 \tag{43}
\]
Thus (42) implies (40), and (41) proves the stated corollary.  Substitution of \(s=1\) into the definition of \(\lambda_w\) gives
\[
 \lambda_w=\frac{315}{256}\frac{M}{32}=\frac{315M}{8192}.
\]

Finally, fix \(0<\gamma<(2+1/s)^{-1}\) and, for all sufficiently large \(n\), set
\[
 \kappa_n=\rho_n(\log n)^\gamma
 =n^{-s/(2s+1)}(\log n)^\gamma.
 \tag{44}
\]
This sequence is positive and eventually at most \(1/8\).  Direct calculation gives
\[
 T_n=n\kappa_n^{2+1/s}
 =(log n)^{\gamma(2+1/s)}=o(\log n),
 \tag{45}
\]
whereas
\[
 \log(1/\kappa_n)
 =\frac{s}{2s+1}\log n-\gamma\log\log n
 \asymp\log n.
 \tag{46}
\]
Equations (45)--(46) imply (40).  Also
\[
 \frac{\rho_n}{\kappa_n}=(\log n)^{-\gamma}\longrightarrow0.
\]
The upper inequality in \(\mathrm{thm:known\mbox{-}cutoff\mbox{-}benchmark}\) now yields \(L_n^{\mathrm{known}}(\kappa_n)\to0\), while (41) remains in force.  This establishes the oracle-shrinking comparison.  Since no converse or attaining construction is supplied at or above \(\lambda_w\), the proof makes only the asserted one-sided claim.
