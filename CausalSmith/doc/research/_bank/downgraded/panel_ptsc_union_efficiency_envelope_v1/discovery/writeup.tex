\paragraph{Fill.} The project derives corrected PT and SC affine influence classes at strict-interior locally constant-rank intersection points, solves their centered intersection by stratum covariance-weighted quadratic programs, and adds \(E[\chi_0^2]\) to the union and oracle variances.   \par\smallskip\noindent\emph{Definition.} At the fixed reference law, \(\chi_0=\chi_{\mathrm{PT}}(P_0)\).
  \item \texttt{QPx} --- \texttt{stratum union-\allowbreak{}projection program}; \texttt{finite covariance-\allowbreak{}weighted quadratic programs}; \texttt{computational envelope}
  \par\smallskip\noindent\emph{Definition.} Program constructed in \(\mathrm{def{:}union\text{-}qp}\).
  \item \texttt{aUx} --- \texttt{union PT-\allowbreak{}normal coefficient}; \(\mathbb R^{d_{\mathrm{PT}}}\); \texttt{union projection coefficient}
  \par\smallskip\noindent\emph{Definition.} \(a_x^U\) is the PT-normal component of the Euclidean minimum-norm solution of \(\mathrm{QP}_x\).
  \item \texttt{dUx} --- \texttt{union SC-\allowbreak{}normal coefficient}; \(\mathbb R^{d_{\mathrm{SC}}}\); \texttt{union projection coefficient}
  \par\smallskip\noindent\emph{Definition.} \(d_x^U\) is the SC-normal component of the Euclidean minimum-norm solution of \(\mathrm{QP}_x\).

\section{Assumptions}

\begin{assumption}[ass:iid-sampling]\label{ass:iid-sampling}
\(O_1,\ldots,O_n\) are independent draws from \(P_O\).
\par\smallskip\noindent\emph{Standard} (Independent panel sampling, \cite{SunXieZhang2025}).
\end{assumption}

\begin{assumption}[ass:no-anticipation]\label{ass:no-anticipation}
\(Y_t(1)=Y_t(0)\) almost surely for every \(t<T\).
\par\smallskip\noindent\emph{Standard} (No anticipation, \cite{SunXieZhang2025}).
\end{assumption}

\begin{assumption}[ass:cell-mass]\label{ass:cell-mass}
\(\min_{x\in\mathcal X}p_X(x)\ge\kappa_0\).
\par\smallskip\noindent\emph{Standard} (Finite-cell probability margin, \cite{TchetgenTchetgenRobinsRotnitzky2010}).
\end{assumption}

\begin{assumption}[ass:group-overlap]\label{ass:group-overlap}
\(\min_{x\in\mathcal X}\min_{1\le g\le J+1}p_g(x)\ge\epsilon_0\).
\par\smallskip\noindent\emph{Standard} (Uniform conditional group overlap, \cite{SunXieZhang2025}).
\end{assumption}

\begin{assumption}[ass:fourth-moments]\label{ass:fourth-moments}
\(\max_{1\le g\le J+1}\max_{1\le t\le T}E_O[|Y_t|^4\mid G=g]\le M_4\).
\par\smallskip\noindent\emph{Standard} (Uniform bounded fourth moments, \cite{SunXieZhang2025}).
\end{assumption}

\begin{assumption}[ass:parallel-trends]\label{ass:parallel-trends}
\(E_F[Y_T(0)-Y_{T-1}(0)\mid G=1,X=x]=E_F[Y_T(0)-Y_{T-1}(0)\mid G=g,X=x]\) for every \(x\in\mathcal X\) and \(g=2,\ldots,J+1\).
\par\smallskip\noindent\emph{Standard} (Conditional parallel trends, \cite{SunXieZhang2025}).
\end{assumption}

\begin{assumption}[ass:synthetic-mean]\label{ass:synthetic-mean}
\(E_F[Y_t(0)\mid G=1,X=x]=\sum_{g=2}^{J+1}w_g(x)E_F[Y_t(0)\mid G=g,X=x]\) for every \(x\in\mathcal X\) and \(t=1,\ldots,T\).
\par\smallskip\noindent\emph{Standard} (Time-invariant covariate-indexed group-level synthetic-control mean representation, \cite{SunXieZhang2025}).
\end{assumption}

\begin{assumption}[ass:sc-rank]\label{ass:sc-rank}
\(\inf_{x\in\mathcal X}\lambda_{\min}\{B(x)^{\prime}B(x)\}\ge\kappa_0\).
\par\smallskip\noindent\emph{Standard} (Uniform full-column-rank group-level preperiod mean system, \cite{SunXieZhang2025}).
\end{assumption}

\begin{assumption}[ass:normal-rank]\label{ass:normal-rank}
\(\operatorname{rank}(\Omega_x)\) is locally constant at \(P_0\) for every \(x\in\mathcal X\).
\par\smallskip\noindent\emph{Standard} (Constant-rank smooth moment manifold, \cite{Newey1994}).
\end{assumption}

\begin{assumption}[ass:normal-gap]\label{ass:normal-gap}
\(\inf_{x\in\mathcal X}\lambda_{\min}^{+}(\Omega_x)\ge\kappa_0\), where \(\lambda_{\min}^{+}\) denotes the smallest nonzero eigenvalue.
\par\smallskip\noindent\emph{Standard} (Separated nonzero Gram spectrum, \cite{AndrewsCheng2012}).
\end{assumption}

\begin{assumption}[ass:local-qmd]\label{ass:local-qmd}
For the observed laws induced by \(P_n\) and \(P_0\), \(dP_{n,O}^{1/2}/dP_{0,O}^{1/2}=1+(2\sqrt n)^{-1}h_n+o_{L^2(P_{0,O})}(n^{-1/2})\) with \(\sup_nE_{P_{0,O}}[h_n^2]<\infty\).
\par\smallskip\noindent\emph{Standard} (Quadratic-mean differentiable local alternatives, \cite{VanDerVaart1998}).
\end{assumption}

\begin{assumption}[ass:union-variance-floor]\label{ass:union-variance-floor}
\(V_U^{\mathrm{loc}}(P_n;b_n,P_0)\ge\kappa_0\) for every \(n\).
\par\smallskip\noindent\emph{Standard} (Nondegenerate asymptotic variance, \cite{ChernozhukovChetverikovKato2016}).
\end{assumption}

\section{Main results}

\begin{lemma}[lem:tangent-normal-form]\label{lem:tangent-normal-form}
At every \(P_0\in\mathcal M_I\) such that, for every \(x\in\mathcal X\) and \(1\le g\le J+1\), \(p_X(x)>\kappa_0\), \(p_g(x)\in(\epsilon_0,1-\epsilon_0)\), \(\max_{g,t}E_{P_{0,O}}[|Y_t|^4\mid G=g]<M_4\), \(\lambda_{\min}\{B(x)^{\prime}B(x)\}>\kappa_0\), every nonzero eigenvalue of \(\Omega_x\) is greater than \(\kappa_0\), and \(\operatorname{rank}(\Omega_x)\) is locally constant, profiling \(q_{\mathrm{PT},x}=\delta_1^0(x)\mathbf 1_J-D(x)\) over the unrestricted path of \(\delta_1^0(x)\) yields the observed normal \(R_{\mathrm{PT},x}\) for \(-A_{\mathrm{PT}}D(x)\). Its Gram rank is at most \(J-1\), with equality under positive definiteness on the donor-contrast range. The SC normal Gram rank is at most \(T-J\), with equality under positive definiteness on the residualized complement of \(\operatorname{col}B(x)\). For either branch, \[\mathcal T_b(P_0)=\{s\in\mathcal H_{P_0}:E_{P_{0,O}}[R_{b,x}s]=0\text{ for every }x\in\mathcal X\},\] and \[\dot\theta_b(s)=E_{P_{0,O}}\!\left[\left\{\chi_b+\sum_xg^{\circ}_{b,x}\right\}s\right],\qquad \mathcal I_b(P_0)=\chi_b+\sum_xg^{\circ}_{b,x}+\operatorname{span}\{R_{b,x}:x\in\mathcal X\}.\] Consequently, \[\phi_b=\chi_b+\sum_xg^{\circ}_{b,x}-R_b^{\prime}\{E_{P_{0,O}}[R_bR_b^{\prime}]\}^{\dagger}E_{P_{0,O}}\!\left[R_b\sum_xg^{\circ}_{b,x}\right],\] and \(V_b=E_{P_{0,O}}[\chi_b^2]\) plus the minimum variance of the centered branch face. Redundant displayed rows are handled by the Moore--Penrose inverse.
\end{lemma}
\begin{proof}
Write \(E_0=E_{P_{0,O}}\).  All paths below are constructed at the strict-interior, locally constant-rank point specified in the statement.  We first profile the PT restriction.  By \(\mathrm{def{:}restriction\text{-}maps}\),
\[
 q_{\mathrm{PT},x}=\delta_1^0(x)\mathbf 1_J-D(x),
 \qquad
 A_{\mathrm{PT}}=I_J-J^{-1}\mathbf 1_J\mathbf 1_J^{\prime}.
 \tag{1}
\]
Since \(\mathbf 1_J^{\prime}\mathbf 1_J=J\),
\[
 A_{\mathrm{PT}}\mathbf 1_J=0,
 \qquad
 A_{\mathrm{PT}}\dot q_{\mathrm{PT},x}=-A_{\mathrm{PT}}\dot D(x).
 \tag{2}
\]
Thus profiling the unrestricted scalar path \(\dot\delta_1^0(x)\) leaves exactly the observed derivative of \(-A_{\mathrm{PT}}D(x)\).  The matrix \(A_{\mathrm{PT}}\) is the orthogonal projector onto \(\{v:\mathbf 1_J^{\prime}v=0\}\), so it has rank \(J-1\).  If \(\widetilde R_{D,x}\) represents \(\dot D(x)\), then \(R_{\mathrm{PT},x}=-A_{\mathrm{PT}}\widetilde R_{D,x}\), whence
\[
 \operatorname{rank}E_0[R_{\mathrm{PT},x}R_{\mathrm{PT},x}^{\prime}]
 \le J-1.
 \tag{3}
\]
The rank is \(J-1\) when the indicated covariance form is positive definite on \(\operatorname{range}(A_{\mathrm{PT}})\).

For SC put \(\Pi_x=B(x)\{B(x)^{\prime}B(x)\}^{-1}B(x)^{\prime}\).  The strict SC spectral margin makes this the orthogonal projector onto the \(J\)-dimensional column space of \(B(x)\).  At an SC law, \(a(x)=B(x)w(x)\).  Differentiating \(\Pi_xB(x)=B(x)\) and then \(r_{\mathrm{SC},x}=(I_T-\Pi_x)a(x)\) gives
\[
 \dot r_{\mathrm{SC},x}
 =(I_T-\Pi_x)\{\dot a(x)-\dot B(x)w(x)\}.
 \tag{4}
\]

\begin{theorem}[thm:uniform-attainment-obstruction]\label{thm:uniform-attainment-obstruction}
Under the frozen definition of \(\mathfrak L_U(P_0)\), the requested uniform asymptotic-linear conclusion, and hence the requested four-part affirmative attainment result, is false in general. In particular, take \(P_0=P^{\star}\) and fixed margins for which the witness has strict slack in cell mass, two-sided group overlap, fourth moments, the SC spectrum, every nonzero normal eigenvalue, and the positive local-score variance bound, with locally constant displayed ranks. For every sufficiently small nonzero \(\delta\), let \(Q_\delta\) retain the donor means and innovation laws of \(P^{\star}\) but replace the treated untreated mean path by \((-2\delta,-4\delta,-4\delta)\), and define \[\Delta_\delta=\phi_U^{\mathrm{loc}}(Q_\delta;\mathrm{PT},P^{\star})-\phi_U^{\mathrm{loc}}(Q_\delta;\mathrm{SC},P^{\star}).\] Then \(Q_\delta\in\mathcal M_I\) and \[E_{Q_\delta}[\Delta_\delta^2]\ge\frac{15}{2}\delta^2>0.\] For every sample size, finite-coordinate splicing produces two elements of \(\mathfrak L_U(P^{\star})\) having the same observed law at that coordinate but different admissible chart labels. No single estimator can have both chart expansions uniformly. This obstruction proves no affirmative sequencewise continuation theorem; it merely does not rule out such a theorem after imposing a common local neighborhood, common bounds on the QMD scores and remainders, and exclusion of arbitrary finite-coordinate replacements.
\end{theorem}
\begin{proof}
Choose the fixed margins so that \(P^{\star}\) satisfies with strict slack the cell-mass, two-sided group-overlap, fourth-moment, SC-spectrum, nonzero-normal-spectrum, and positive local-score-variance inequalities, and so that the displayed ranks are locally constant.  For sufficiently small fixed \(\delta\ne0\), retain both donor mean paths and all innovation laws but replace the treated untreated mean path by
\[
 (1/2+\delta)(-1,-2,-2)+(1/2-\delta)(1,2,2)
 =(-2\delta,-4\delta,-4\delta).
 \tag{1}
\]
Give the treated group the same additive final treatment effect and call the law \(Q_\delta\).  Every untreated final mean change remains zero, so PT holds.  The donor matrix is unchanged, and the treated path is reproduced by the unique weights
\[
 w_\delta=(1/2+\delta,1/2-\delta)^{\prime},
 \tag{2}
\]
so SC also holds.  The finitely many probabilities, fourth moments, Gram matrices, positive eigenvalues, and local-score variances vary continuously in \(\delta\).  The strict slack at \(P^{\star}\) therefore ensures that every displayed margin remains valid for all sufficiently small \(|\delta|\).  Thus \(Q_\delta\in\mathcal M_I\).

There is one covariate stratum.  On either branch \(C_b(X)=\theta_b\) is constant, so
\[
 \chi_{\mathrm{PT}}(Q_\delta)=\chi_{\mathrm{SC}}(Q_\delta)=0.
 \tag{3}
\]
The branch-specific composition terms in \(\mathrm{def{:}local\text{-}score\text{-}extension}\) therefore do not affect the witness calculation.

Fix an integer \(m\).  Construct two triangular sequences that equal \(Q_\delta\) at coordinate \(m\) and \(P^{\star}\) at every other coordinate, labeling coordinate \(m\) by PT in the first and by SC in the second.  Each sequence belongs to \(\mathfrak L_U(P^{\star})\): after its single exceptional coordinate it is identically \(P^{\star}\), so the QMD score and remainder are eventually zero, and every coordinate obeys the fixed margins.  Thus, at every sample size, the frozen sequence class contains two admissibly labeled copies of the same observed law.

Let \(z_{g,t}=3\mathbf 1\{G=g\}\{Y_t-\mu_{g,t}\}\).  At \(Q_\delta\),
\[
 \operatorname{Var}(z_{2,3})=\frac32,
 \qquad
 \operatorname{Var}(z_{3,3})=6.
 \tag{4}
\]
At \(P^{\star}\), the common score has final donor coefficients \((-1/2,-1/2)\).  The PT extension freezes its center normal coefficient, and the donor laws are unchanged, so those coefficients remain \((-1/2,-1/2)\) at \(Q_\delta\).  The SC normal uses only preperiod coordinates, whereas (2) makes the SC final donor coefficients \((-1/2-\delta,-1/2+\delta)\).  Hence, for
\[
 \Delta_\delta
 =\phi_U^{\mathrm{loc}}(Q_\delta;\mathrm{PT},P^{\star})
 -\phi_U^{\mathrm{loc}}(Q_\delta;\mathrm{SC},P^{\star}),
 \tag{5}
\]
the final-innovation component is \(\delta(z_{2,3}-z_{3,3})\).  Every remaining component uses other group-period innovations and is orthogonal to it.  Therefore
\[
 E_{Q_\delta}[\Delta_\delta^2]
 \ge\delta^2\left(\frac32+6\right)
 =\frac{15}{2}\delta^2>0.
 \tag{6}
\]
Both chart scores are mean zero, so \(E_{Q_\delta}[\Delta_\delta]=0\).

If a single estimator had both chart expansions uniformly over the frozen class, couple the two sequences at sample size \(m\) through the same independent sample from \(Q_\delta\) and subtract their expansions.  Since both branches identify the same ATT at \(Q_\delta\), the estimator and target terms cancel, leaving
\[
 S_m=m^{-1/2}\sum_{i=1}^m\Delta_\delta(O_i)=o_{Q_\delta}(1).
 \tag{7}
\]
Independent sampling and (6) imply \(E[S_m^2]=E[\Delta_\delta^2]\ge15\delta^2/2\).  Because \(\Delta_\delta\) is a bounded linear combination of finitely many Rademacher innovations,
\[
 E[S_m^4]
 =m^{-1}E[\Delta_\delta^4]
 +3(1-m^{-1})E[\Delta_\delta^2]^2
 \tag{8}
\]
is uniformly bounded.  Thus \((S_m^2)\) is uniformly integrable.  Convergence in probability in (7) would imply \(E[S_m^2]\to0\), contradicting (6).

No estimator can therefore possess both chart expansions uniformly over \(\mathfrak L_U(P^{\star})\).  Because asymptotic linearity is a necessary conjunct of the requested one-step, variance-consistency, multiplier-consistency, and coverage result, the full affirmative theorem is false.  The argument does not rule out a different theorem on a redesigned common neighborhood with common QMD-score and remainder bounds and without arbitrary finite-coordinate splicing.
\end{proof}
\par\smallskip\noindent \textit{Justification.} Strict slack keeps both the witness and its perturbations inside the frozen class, while the one-stratum composition terms vanish and leave the quantitative chart separation unchanged. \textit{Gap.} The integrated alternative-hypothesis laws in SunXieZhang2025 do not validate uniformity over finite-coordinate-spliced sequences. \textit{Consumer.} A branch-robust confidence interval requires a redesigned uniformity class and a new positive theorem.

\begin{lemma}[lem:sc-profiled-normal-rank]\label{lem:sc-profiled-normal-rank}
At every \(P_0\in\mathcal M_I\) such that, for every \(x\in\mathcal X\) and \(1\le g\le J+1\), \(p_X(x)>\kappa_0\), \(p_g(x)\in(\epsilon_0,1-\epsilon_0)\), \(\max_{g,t}E_{P_{0,O}}[|Y_t|^4\mid G=g]<M_4\), \(\lambda_{\min}\{B(x)^{\prime}B(x)\}>\kappa_0\), every nonzero eigenvalue of \(\Omega_x\) is greater than \(\kappa_0\), and \(\operatorname{rank}(\Omega_x)\) is locally constant, the derivative of the profiled SC restriction in stratum \(x\) is \[\dot r_{\mathrm{SC},x}=\{I_T-B(x)[B(x)^{\prime}B(x)]^{-1}B(x)^{\prime}\}\{\dot a(x)-\dot B(x)w(x)\}.\] Consequently, \[\operatorname{rank}E_{P_{0,O}}[R_{\mathrm{SC},x}R_{\mathrm{SC},x}^{\prime}]\le T-J.\] Equality holds when the covariance form of the residualized coordinates is positive definite on the \((T-J)\)-dimensional orthogonal complement of \(\operatorname{col}B(x)\).
\end{lemma}
\begin{proof}
Fix a stratum \(x\) at the strict-interior, locally constant-rank point in the statement and put \(\Pi_x=B(x)\{B(x)^{\prime}B(x)\}^{-1}B(x)^{\prime}\).  The strict SC Gram margin makes \(\Pi_x\) the orthogonal projector onto the \(J\)-dimensional column space of \(B(x)\), so \(I_T-\Pi_x\) has rank \(T-J\).  At an SC law, \(a(x)=B(x)w(x)\).  Differentiating \(\Pi_xB(x)=B(x)\) yields
\[
 \dot\Pi_xB(x)w(x)=(I_T-\Pi_x)\dot B(x)w(x).
 \tag{1}
\]
Since \(r_{\mathrm{SC},x}=(I_T-\Pi_x)a(x)\),
\[
 \dot r_{\mathrm{SC},x}
 =(I_T-\Pi_x)\{\dot a(x)-\dot B(x)w(x)\}.
 \tag{2}
\]
The definition of \(R_{\mathrm{SC},x}\) uses the conditional-residual Riesz representative of exactly this derivative.  Conditional centering does not change the deterministic range in (2).  Thus the covariance range of \(R_{\mathrm{SC},x}\) is contained in \(\operatorname{range}(I_T-\Pi_x)\), proving the rank bound \(T-J\).  If the residualized covariance form is positive definite on that range, its covariance rank equals the range dimension, proving equality.
\end{proof}
\par\smallskip\noindent \textit{Justification.} The lemma certifies the finite SC normal dimension at the same strict-interior constant-rank points as the envelope. \textit{Gap.} The profiled derivative and its exact residual-space rank are needed for the QP geometry. \textit{Consumer.} This is a rank calculation, not an inference guarantee.

\begin{proposition}[prop:saturated-donor-reduction]\label{prop:saturated-donor-reduction}

\begin{proof}

 \tag{1}
\]

 \mathcal T_{\mathrm{SC}}(P_0)=\mathcal H_{P_0},

 \mathcal I_{\mathrm{PT}}(P_0)\cap\mathcal I_{\mathrm{SC}}(P_0)

\section{Interpretation}

At a strict-interior constant-rank intersection law, the cost of requiring regularity under either identifying restriction is the squared-distance cost of forcing the PT and SC affine influence faces to agree. The common treated-composition term does not affect that distance but does contribute \(E[\chi_0^2]\) to every variance. A zero Gram discrepancy means local robustness is free in the specified direction; a positive discrepancy is a local efficiency cost, not an estimated diagnostic with its own sampling theory.

\section*{Technical internal limitation (diagnostic only)}

The corrected union envelope is a local regularity lower bound only at strict-interior intersection laws with locally constant rank and strict support, moment, SC-spectrum, and nonzero-normal-spectrum slack. Its common treated-composition component is identified analytically, but no attaining estimator, usable standard error, or multiplier-confidence procedure is established. The negative theorem uses a witness with strict slack, including in the local-score variance margin, and depends on the frozen sequence class's closure under arbitrary finite-coordinate replacements. It does not settle a redesigned common neighborhood with common QMD-score and remainder bounds. Boundary overlap, saturated moment inequalities, rank changes, covariance degeneracy, and weight restrictions require different local experiments.

\section{Honest scope}

The proved deliverables are strict-interior, locally constant-rank versions of the corrected tangent-normal characterization \(\mathcal I_b=\chi_b+\sum_xg^{\circ}_{b,x}+\operatorname{span}R_b\), the finite centered union convolution lower bound with \(E[\chi_0^2]\), observable oracle-adaptation tests, the exact strict three-group witness, and the single-donor and saturated-donor reductions. Sun, Xie, and Zhang supply one integrated estimator and one multiplier construction analyzed under alternative PT and SC hypotheses; this paper preserves that integrated-construction description and provides no new implemented inference. Lower-bound and impossibility conclusions transfer to estimators regular on the broader published union by restriction to the repaired subclass, while sharpness remains local to that subclass. The frozen sequence class makes a uniformly chartwise expansion false by finite-coordinate splicing, and no positive sequencewise continuation theorem is claimed.
