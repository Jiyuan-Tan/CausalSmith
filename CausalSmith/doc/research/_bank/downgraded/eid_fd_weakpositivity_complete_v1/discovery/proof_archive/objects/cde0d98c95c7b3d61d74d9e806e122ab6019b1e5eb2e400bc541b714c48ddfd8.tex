Let \(i_Z,i_A,i_X,i_Y\) be distinct vertices, put
\(\iota=(i_Z,i_A,i_X,i_Y)\), and suppose that
\(D=\{i_Z,i_A,i_X,i_Y\}\), that the edge set of \(G\) is exactly
\[
 \{i_Z\mathbin{\to}i_A,\ i_A\mathbin{\to}i_X,
   \ i_A\mathbin{\to}i_Y,\ i_X\mathbin{\to}i_Y\},
\]
that \(V=\{i_Z,i_X,i_Y\}\), and that \(W=\{i_A\}\).  Write
\(\mathcal Z=\mathcal A_{i_Z}\), \(\mathcal B=\mathcal A_{i_A}\),
\(\mathcal X=\mathcal A_{i_X}\), \(\mathcal Y=\mathcal A_{i_Y}\),
and \(B=\lvert\mathcal B\rvert\).  Define the family-local
arbitrary-law class \(\mathfrak F_0(G,\mathcal A,V,W;\iota)\) to consist
of the recursive mechanisms \(M=(G,\mathcal A,\theta)\) whose
observational law has the recursive product factorization, whose
intervention distributions mentioned below have the corresponding
truncated product factorization, and whose row at \(i_A\), conditional on
each state of \(i_Z\), is a point mass.  The definition of
\(\mathfrak F_0\) imposes no positivity condition.

Let \(p\) be an arbitrary probability law on
\(\mathcal Z\times\mathcal X\times\mathcal Y\).  Put
\[
 p(z)=\sum_{x,y}p(z,x,y),\qquad
 p(z,x)=\sum_y p(z,x,y),\qquad
 \mathcal Z_+=\{z:p(z)>0\},\qquad
 \mathcal Z_0=\mathcal Z\setminus\mathcal Z_+.
\]
For \(z\in\mathcal Z_+\), define, under the displayed strict guards,
\[
 e_z(x)=\frac{p(z,x)}{p(z)},\qquad
 m_{zx}(y)=\frac{p(z,x,y)}{p(z,x)}
 \quad\text{when }p(z,x)>0,
\]
and define the positive-root profile
\[
 \rho_z=\bigl(e_z,(m_{zx})_{x:e_z(x)>0}\bigr).
\]
Thus two positive roots have the same profile exactly when their
\(e\)-rows agree and their \(m\)-rows agree on the resulting common
positive support.  Put
\[
 \mathcal R_+(p)=\{\rho_z:z\in\mathcal Z_+\},\qquad
 R=\lvert\mathcal R_+(p)\rvert,
\]
and, for \(\rho\in\mathcal R_+(p)\), put
\[
 \mathcal Z_\rho=\{z\in\mathcal Z_+:\rho_z=\rho\},\qquad
 H_\rho=\{x\in\mathcal X:e_z(x)=0\text{ for }z\in\mathcal Z_\rho\}.
\]
The definition of \(H_\rho\) is independent of the chosen member of
\(\mathcal Z_\rho\).

For an integer \(b\geq1\), let \(\Pi_b^0(p)\) be the set of partitions
\(\pi\) of every state in \(\mathcal Z\) into at most \(b\) nonempty
blocks such that
\[
 z,z'\in C\cap\mathcal Z_+\quad\Longrightarrow\quad \rho_z=\rho_{z'}
 \qquad(C\in\pi).
\]
Thus zero-root states are also assigned to blocks, while the positive
part of a block is empty or is contained in one profile class.  For
\(\pi\in\Pi_b^0(p)\), let \(\mathcal K_\pi^0(p)\) be the set of arrays
\(k=(k(z,x,\cdot))_{z,x}\), with every
\(k(z,x,\cdot)\in\Delta(\mathcal Y)\), such that
\[
 \begin{split}
 k(z,x,\cdot)&=k(z',x,\cdot)
 &&(C\in\pi,\ z,z'\in C,\ x\in\mathcal X),\\
 k(z,x,\cdot)&=m_{zx}
 &&(z\in\mathcal Z_+,\ e_z(x)>0).
 \end{split}
\]
Define the augmented sharp candidate and the unconstrained-row candidate
by
\[
 \mathcal K_b^0(p)=\bigcup_{\pi\in\Pi_b^0(p)}\mathcal K_\pi^0(p)
\]
and
\[
 \mathcal K_{\mathrm{ind}}^0(p)
 =\left\{k:
 k(z,x,\cdot)=m_{zx}\text{ if }z\in\mathcal Z_+
 \text{ and }e_z(x)>0,
 \text{ with all remaining rows free in }\Delta(\mathcal Y)
 \right\}.
\]
Finally define
\[
 B_{\mathrm{sat}}^0(p)
 =R+\lvert\mathcal Z_0\rvert+
 \sum_{\rho\in\mathcal R_+(p):H_\rho\ne\varnothing}
 \bigl(\lvert\mathcal Z_\rho\rvert-1\bigr).
\]
For a mechanism \(M\), distinguish the causal response kernel
\[
 K_M(z,x,y)=P_M(i_Y=y\mid do(i_Z=z,i_X=x))
\]
from the observed conditionals \(m_{zx}\).  Then the following conclusions
hold.
\begin{enumerate}
\item The law \(p\) is induced by a mechanism in
\(\mathfrak F_0(G,\mathcal A,V,W;\iota)\) if and only if \(R\leq B\).
On every feasible observed-law fiber, the partitions of \(\mathcal Z\)
into the nonempty fibers of the qualitative functional map
\(f:\mathcal Z\to\mathcal B\) are exactly the members of
\(\Pi_B^0(p)\).
\item On every feasible law, the sharp full response-kernel region is
\[
 \left\{K_M:M\in\mathfrak F_0(G,\mathcal A,V,W;\iota),\ p_M=p\right\}
 =\mathcal K_B^0(p).
\]
Every member of this union is attained simultaneously on all
\((z,x,y)\) without changing any alphabet.  If \(p\) and the selected
kernel are rational, the attaining mechanism can be chosen with rational
conditional tables.
\item If \(\lvert\mathcal Y\rvert\geq2\), then
\[
 \mathcal K_b^0(p)=\mathcal K_{\mathrm{ind}}^0(p)
 \quad\Longleftrightarrow\quad
 b\geq B_{\mathrm{sat}}^0(p)
\]
for every \(b\geq R\).  Consequently
\(B_{\mathrm{sat}}^0(p)\) is the least hidden-alphabet cardinality at
which every unobserved response row can vary independently.
\item On a feasible law the full response kernel is point identified
automatically when \(\lvert\mathcal Y\rvert=1\).  If
\(\lvert\mathcal Y\rvert\geq2\), it is point identified if and only if
\[
 H_\rho=\varnothing\quad(\rho\in\mathcal R_+(p))
\]
and one of the following two mutually exclusive conditions holds:
\begin{enumerate}
\item \(\mathcal Z_0=\varnothing\);
\item \(\mathcal Z_0\ne\varnothing\), \(B=R\), and the complete arrays
\[
 g_\rho=(m_{zx})_{x\in\mathcal X},\qquad z\in\mathcal Z_\rho,
\]
are equal for all \(\rho\in\mathcal R_+(p)\).
\end{enumerate}
At \(B=R\), every zero-root state must join one of the existing positive
profile blocks; at \(B>R\), a zero-root state can instead form a free
singleton block.
\item For the single-coordinate intervention, put
\(Q_M^X(x,\cdot)=P_M(i_Y\in\cdot\mid do(i_X=x))\).  Its sharp projection
is
\[
 \left\{(Q_M^X(x,\cdot))_{x\in\mathcal X}:
 M\in\mathfrak F_0(G,\mathcal A,V,W;\iota),\ p_M=p\right\}
 =\prod_{x\in\mathcal X}\bigl(L_x+M_x\Delta(\mathcal Y)\bigr),
\]
where
\[
 L_x(y)=\sum_{z\in\mathcal Z_+:p(z,x)>0}p(z)m_{zx}(y),
 \qquad
 M_x=\sum_{z\in\mathcal Z_+:p(z,x)=0}p(z).
\]
The projection is attained simultaneously across \(x\), on the original
alphabets, and rationally when \(p\) and the selected projected vector are
rational.  If \(p(z,x)>0\) for every \((z,x)\), then
\(\mathcal Z_0=\varnothing\) and the full response kernel collapses to the
guarded observed-law circuit
\[
 K_M(z,x,y)=m_{zx}(y)=\frac{p(z,x,y)}{p(z,x)}.
\]
If merely \(\mathcal Z_0=\varnothing\), then the augmented partitions,
kernel union, and saturation threshold reduce respectively to
\(\Pi_b(p)\), \(\mathcal K_b(p)\), and
\(B_{\mathrm{sat}}(p)\) obtained by omitting the superscript \(0\) and
the term \(\lvert\mathcal Z_0\rvert\).
\item After exact profile comparison, feasibility and
\(B_{\mathrm{sat}}^0(p)\) require
\(O(\lvert\mathcal Z\rvert\lvert\mathcal X\rvert
\lvert\mathcal Y\rvert)\) guarded arithmetic operations.
The family \(\Pi_B^0(p)\) is enumerable by at most
\(B^{\lvert\mathcal Z\rvert}\) labeled assignments, and each
\(\mathcal K_\pi^0(p)\) has
\(O(\lvert\mathcal Z\rvert\lvert\mathcal X\rvert
\lvert\mathcal Y\rvert)\) simplex constraints and linear equalities.
Every division above is licensed by the displayed tested strict guard.
\end{enumerate}
