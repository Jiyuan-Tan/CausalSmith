Set
\[
 \widehat\Omega_T=T^{-1}\sum_{t=1}^T\widehat\xi_t\widehat\xi_t^\top,
 \qquad
 \mathcal H_T=\{\sigma_{\min}(\widehat H_T)\ge h_-/2\}.
\]
Define the total nonnegative scalar
\[
 \widehat q_T(c)=
 \begin{cases}
 c^\top\widehat H_T^{-1}\widehat\Omega_T\widehat H_T^{-\top}c,
   &\text{on }\mathcal H_T,\\
 0,&\text{on }\mathcal H_T^c,
 \end{cases}
\]
where the inverse is evaluated only on \(\mathcal H_T\), and set
\[
 \mathcal E_T(c)=\mathcal H_T\cap\{\widehat q_T(c)\ge\omega_{c,T}/2\}.
\]
Define the total early-scale event-truncated sandwich by
\[
 \widehat V_T(c)=
 \begin{cases}
 \widehat H_T^{-1}\widehat\Omega_T\widehat H_T^{-\top},
   &\text{on }\mathcal E_T(c),\\
 I_d,&\text{on }\mathcal E_T(c)^c.
 \end{cases}
\]
Thus every inverse is evaluated only on \(\mathcal H_T\), and \(c^\top\widehat V_T(c)c>0\) for every realization because \(c\ne0\).
