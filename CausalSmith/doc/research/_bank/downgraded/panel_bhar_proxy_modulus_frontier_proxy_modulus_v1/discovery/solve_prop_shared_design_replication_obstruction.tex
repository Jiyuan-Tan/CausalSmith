%%% FIELD proof-prop-shared-design-replication-obstruction
Fix
\[
H=0,
\qquad p=S=d=1,
\qquad \lambda=\kappa=1,
\qquad \pi_1=1.
\tag{1}
\]
Choose
\[
B=16,
\quad c_X=C_X=c_\Sigma=C_\Sigma=1,
\quad \delta_0=\frac14,
\quad c_{\mathrm{mod}}=C_{\mathrm{mod}}=1,
\tag{2}
\]
take any fixed \(\beta_0>0\) and \(\rho_0\in(0,1)\), choose \(L\) at least the \(\mathcal H_1(1,\cdot)\)-radius of the affine function introduced below, and let
\[
K(u)=\mathbf 1\{|u|\le 1\}.
\tag{3}
\]
This kernel is nonnegative, bounded, compactly supported, and equal to one on a neighborhood of zero.  Choose \(0<c_G\le 1\).  By the definitions of \(r\), \(b_r\), and \(e_0\),
\[
r=\lceil 1\rceil-1=0,
\qquad b_0(u)=e_0=1.
\tag{4}
\]
Let \(E_0\) be constant, set \(z_0=1/2\) and \(\mathcal Z_+=\{1/2\}\), and put \(n_1=N=T^2\).

We construct a pair for every \(\eta\in(0,1)\).  Write \(X_t=X_{1t}\), and let \(X_1,\ldots,X_T\) be independent uniform random variables on \([0,1]\).  Set
\[
m(x)=8+x-\frac12,
\qquad
Q_{1ti}=m(X_t)+\frac12U_{ti},
\tag{5}
\]
where the \(U_{ti}\) are mutually independent Rademacher random variables, independent of the complete design.  Since \(m([0,1])=[15/2,17/2]\),
\[
7\le Q_{1ti}\le 9\le B,
\qquad
\mathbb E[Q_{1ti}\mid X_t=x,E_0]=m(x).
\tag{6}
\]
Thus (6) verifies \(\texttt{ass:bounded-historical-wealth}\) and \(\texttt{ass:historical-regression}\).  The choice of \(L\) and the affine form in (5) verify \(\texttt{ass:holder-smoothness}\).  The density of every \(X_t\) is one on \([0,1]\), so \(\texttt{ass:design-density}\) holds.  Calendar independence gives, for every \(\ell_{\mathrm{mix}}\ge1\),
\[
\beta_P(\ell_{\mathrm{mix}}\mid E_0)=0
\le \beta_0\rho_0^{\ell_{\mathrm{mix}}},
\tag{7}
\]
which is \(\texttt{ass:temporal-mixing}\).  Finally set \(X_{1t}^{\mathrm{obs}}=X_t\), as required by \(\texttt{ass:exact-training-coordinates}\).

Independently of the historical sample, let
\[
Z_j=\frac12+V_j,
\tag{8}
\]
where \(V_1,\ldots,V_N\) are independent Rademacher random variables.  Equations (2) and (8) give
\[
\mathbb E[Z_j\mid E_0]=z_0,
\qquad
\Sigma=\operatorname{Var}(Z_j\mid E_0)=1,
\tag{9}
\]
and
\[
\mathbb E\!\left[
\left|\Sigma^{-1/2}(Z_j-z_0)\right|^4\,\middle|\,E_0
\right]=1\le B.
\tag{10}
\]
Thus (8)--(10) verify \(\texttt{ass:donor-replication}\), \(\texttt{ass:donor-ellipticity}\), and \(\texttt{ass:donor-fourth-moment}\); the stipulated independence verifies \(\texttt{ass:donor-independence}\).  Since \(D=Sd=1\), (2) also gives the strictly compatible envelope \(B>D^2\).

For \(0<\delta\le\delta_0\), every \(z_0+u\) with \(|u|\le\delta\) belongs to \([0,1]\).  From \(\Sigma=1\), \(\pi_1=1\), \(\texttt{def:local-modulus}\), and (5),
\[
\begin{aligned}
\omega_P(\delta;z_0)
&=\sup_{|u|\le\delta}|m(z_0+u)-m(z_0)|\\
&=\sup_{|u|\le\delta}|u|
=\delta.
\end{aligned}
\tag{11}
\]
Equation (11) verifies both inequalities in \(\texttt{ass:local-two-sided-modulus}\).  Moreover,
\[
\operatorname{dist}(z_0,\partial[0,1])=\frac12\ge\delta_0,
\qquad
P\{z_0\in\mathcal Z_+\}=1\ge c_X,
\tag{12}
\]
so \(\texttt{ass:interior-event-path}\) and \(\texttt{ass:positive-query-region}\) hold.

Define
\[
\Gamma_\eta=\mathbf 1\{X_1\le\eta\}.
\tag{13}
\]
The two causal completions have the same historical and donor variables.  Under the zero-effect completion \(P_{0,\eta}\), set
\[
B_{1i}(0)=B_{1i}(1)=B_{1i}^{\mathrm{obs}}=8.
\tag{14}
\]
Under the nonzero-effect completion \(P_{1,\eta}\), set
\[
B_{1i}(0)=8,
\qquad
B_{1i}(1)=B_{1i}^{\mathrm{obs}}=8+\Gamma_\eta.
\tag{15}
\]
For both completions define \(B_{1i}(a,\mathbf a_{-1i})=B_{1i}(a)\) for every other-unit assignment and define \(Q_{1ti}(a)=Q_{1ti}\) for \(a\in\{0,1\}\).  Equations (14)--(15) and these extensions verify, respectively, \(\texttt{ass:consistency}\), \(\texttt{ass:historical-consistency}\), \(\texttt{ass:no-spillovers}\), and \(\texttt{ass:limited-anticipation}\).  They also give
\[
0\le B_{1i}(a)\le9<B,
\tag{16}
\]
which verifies \(\texttt{ass:bounded-event-wealth}\).  To realize \(\texttt{def:buyhold-level}\), take
\[
W_{1i,-1}=1,
\qquad
R_{1i0}(a)=B_{1i}(a)-1>-1.
\tag{17}
\]
Because \(H=0\), equation (17) gives \(W_{1i,-1}\{1+R_{1i0}(a)\}=B_{1i}(a)\).

The untreated event potential outcome is eight under both completions.  Since \(m(z_0)=8\),
\[
\frac1{n_1}\sum_{i=1}^{n_1}
\mathbb E_{P_{\ell,\eta}}[B_{1i}(0)\mid z_0,E_0]
=8=m(z_0),
\qquad \ell\in\{0,1\},
\tag{18}
\]
so \(\texttt{ass:event-transport}\) holds.  Conditional on the complete design and \(E_0\), the event outcomes in (14) or (15) are degenerate, while unit \(i\)'s historical path is a function only of the independent vector \((U_{ti})_{t=1}^{T}\).  Hence the complete outcome vectors are independent across \(i\) under this conditioning, which verifies \(\texttt{ass:cross-unit-independence}\).  Equations (5)--(18) therefore verify every member property in \(\texttt{def:panel-law-class}\), and
\[
P_{0,\eta},P_{1,\eta}
\in\mathcal M_{T^2,T,(T^2)}(1).
\tag{19}
\]

The construction also identifies which extra centering condition fails.  From \(\varepsilon_{1ti}=U_{ti}/2\) and independence of the Rademacher errors from the design,
\[
\mathbb E_{P_{\ell,\eta}}\!\left[
\varepsilon_{1ti}\,\middle|\,(X_t)_{t=1}^{T},E_0
\right]=0,
\qquad \ell\in\{0,1\},
\tag{20}
\]
so both laws satisfy \(\texttt{ass:historical-full-design-centering}\).  In contrast, under \(P_{1,\eta}\),
\[
\mathbb E[B_{1i}^{\mathrm{obs}}\mid(X_t)_{t=1}^{T},E_0]
=8+\Gamma_\eta,
\qquad
\mathbb E[B_{1i}^{\mathrm{obs}}\mid E_0]=8+\eta.
\tag{21}
\]
Because \(0<\eta<1\), \(\Gamma_\eta\ne\eta\) almost surely, and (21) violates \(\texttt{ass:event-full-design-mean-stability}\).  The zero-effect completion satisfies that condition because both conditional means equal eight.

We now separate the causal targets from the realized observed event means.  Since \(z_0\) and \(E_0\) are constant, \(\pi_1=1\), and \(P(X_1\le\eta)=\eta\), \(\texttt{def:target}\) and (14)--(15) give
\[
\tau_{0,\eta}:=\tau_n^{P_{0,\eta}}(z_0)=0,
\qquad
\tau_{1,\eta}:=\tau_n^{P_{1,\eta}}(z_0)
=\mathbb E[\Gamma_\eta]=\eta.
\tag{22}
\]
The realized event mean under \(P_{1,\eta}\) is instead \(8+\Gamma_\eta\), a single design-shared Bernoulli shock which replication over \(i\) does not average.

Let \(\mathcal O\) be the sigma-field generated by all observed variables, and let \(\nu_{\ell,\eta}\) be its conditional law given \(E_0\) under \(P_{\ell,\eta}\).  On
\[
A_\eta^c=\{X_1>\eta\},
\tag{23}
\]
the two observed-data maps coincide.  On \(A_\eta\), their event vectors are respectively the all-eight and all-nine vectors, and hence have disjoint supports.  Therefore the greatest measure dominated by both observed-data laws is their common restriction to (23), and its mass, the conditional observed-data overlap, is exactly
\[
\|\nu_{0,\eta}\wedge\nu_{1,\eta}\|
=P(A_\eta^c\mid E_0)=1-\eta.
\tag{24}
\]

Let \(\widetilde\tau\) be any observed-data-measurable estimator and put \(\mu_\eta=\nu_{0,\eta}\wedge\nu_{1,\eta}\).  Pointwise, for every \(x\in\mathbb R\),
\[
x^2+(x-\eta)^2
=2(x-\eta/2)^2+\frac{\eta^2}{2}
\ge\frac{\eta^2}{2}.
\tag{25}
\]
Since \(\mu_\eta\le\nu_{\ell,\eta}\), equations (22)--(25) imply
\[
\begin{aligned}
&\mathbb E_{P_{0,\eta}}[(\widetilde\tau-\tau_{0,\eta})^2\mid E_0]
+\mathbb E_{P_{1,\eta}}[(\widetilde\tau-\tau_{1,\eta})^2\mid E_0]\\
&\qquad\ge
\int\{\widetilde\tau^2+(\widetilde\tau-\eta)^2\}\,d\mu_\eta
\ge(1-\eta)\frac{\eta^2}{2}.
\end{aligned}
\tag{26}
\]
At least one summand in (26) is at least one half of its right-hand side.  By (19),
\[
\inf_{\widetilde\tau}
\sup_{P\in\mathcal M_{T^2,T,(T^2)}(1)}
\mathbb E_P[\{\widetilde\tau-\tau_n(z_0)\}^2\mid E_0]
\ge\frac{(1-\eta)\eta^2}{4}.
\tag{27}
\]
Taking \(\eta=1/4\) in (27) yields \(3/256\).

For this array, \(\texttt{def:rate-frontier}\) gives
\[
R_{\mathrm{tr}}=T^{-2},
\qquad
R_{\mathrm{sm}}=(T^3)^{-2/3}=T^{-2},
\qquad
R_{\mathrm{don}}=T^{-2},
\tag{28}
\]
and
\[
R_{\mathrm{sp},\beta}
=\left\{\frac{\log(T\vee2)}T\right\}^{2}.
\tag{29}
\]
Thus, for all sufficiently large \(T\),
\[
R_{\star,\beta}=R_{\mathrm{sp},\beta}\longrightarrow0,
\tag{30}
\]
while the lower bound in (27) is fixed and positive.  This proves the universal risk obstruction over the original class.

For the interval obstruction, fix \(\alpha\in(0,1/2)\) and define
\[
\eta_\alpha=\frac{1-2\alpha}{2}.
\tag{31}
\]
Then \(\eta_\alpha\in(0,1/2)\) and
\[
1-\eta_\alpha-2\alpha=\eta_\alpha>0,
\qquad
1-\eta_\alpha>2\alpha.
\tag{32}
\]
Let \(I\) be an observed-data-measurable interval that is conditionally honest uniformly over the original class.  Applied to the pair with \(\eta=\eta_\alpha\), honesty and (22) give
\[
P_{0,\eta_\alpha}\{0\notin I\mid E_0\}\le\alpha,
\qquad
P_{1,\eta_\alpha}\{\eta_\alpha\notin I\mid E_0\}\le\alpha.
\tag{33}
\]
Using \(\mu_{\eta_\alpha}\le\nu_{0,\eta_\alpha}\), \(\mu_{\eta_\alpha}\le\nu_{1,\eta_\alpha}\), (24), (33), and subadditivity,
\[
\begin{aligned}
\mu_{\eta_\alpha}\{0,\eta_\alpha\in I\}
&\ge \mu_{\eta_\alpha}(\Omega)
-\mu_{\eta_\alpha}\{0\notin I\}
-\mu_{\eta_\alpha}\{\eta_\alpha\notin I\}\\
&\ge1-\eta_\alpha-2\alpha
=\eta_\alpha.
\end{aligned}
\tag{34}
\]
Whenever an interval contains both zero and \(\eta_\alpha\), its length is at least \(\eta_\alpha\).  Since \(\mu_{\eta_\alpha}\le\nu_{0,\eta_\alpha}\), (34) implies
\[
\mathbb E_{P_{0,\eta_\alpha}}[\operatorname{length}(I)\mid E_0]
\ge\int\operatorname{length}(I)\,d\mu_{\eta_\alpha}
\ge\eta_\alpha^2.
\tag{35}
\]
Taking the supremum over the original class and then the infimum over all uniformly honest intervals proves the asserted honest-length lower bound.  Equations (20)--(21) show precisely that this impossibility proves rate-essentiality of the event-side restriction, but not separate necessity of the historical restriction.

It remains to prove the estimator-specific corollaries for \(\eta=1/4\).  Work henceforth under \(P_{1,1/4}\), abbreviate \(\Gamma=\Gamma_{1/4}\), and note from (15) and (22) that
\[
\bar B_1^{\mathrm{obs}}=8+\Gamma,
\qquad
\tau_n(z_0)=\frac14.
\tag{36}
\]
The generated and projected centers in \(\texttt{def:feasible-estimator}\) satisfy
\[
\bar Z=\frac1N\sum_{j=1}^{N}Z_j,
\qquad
\widetilde z=\Pi_{[0,1]}(\bar Z).
\tag{37}
\]
Projection fixes \(z_0\) and is nonexpansive.  Using (8)--(9) and \(N=T^2\),
\[
|\widetilde z-z_0|
\le\left|\frac1N\sum_{j=1}^{N}V_j\right|,
\qquad
\mathbb E[(\widetilde z-z_0)^2\mid E_0]
\le N^{-1}=T^{-2}.
\tag{38}
\]
For all sufficiently large \(T\), the truncation at one in the beta-mixing initial bandwidth is inactive, and (28)--(29) imply
\[
\underline h_T
=C_h\left[T^{-1}\vee\frac{\log T}{T}\vee T^{-1}\right]
=C_h\frac{\log T}{T}.
\tag{39}
\]
Define
\[
I_T=\{t:|X_t-\widetilde z|\le\underline h_T\},
\qquad
\mathcal A_T=\{|I_T|>0\}.
\tag{40}
\]
For every \(\widetilde z\in[0,1]\), the intersection of \([\widetilde z-\underline h_T,\widetilde z+\underline h_T]\) with \([0,1]\) has Lebesgue measure at least \(\underline h_T\).  Donor independence and the independent uniform design therefore give
\[
P(\mathcal A_T^c\mid E_0,\widetilde z)
\le(1-\underline h_T)^T
\le e^{-T\underline h_T}
=T^{-C_h},
\tag{41}
\]
and hence \(P(\mathcal A_T^c\mid E_0)\to0\).

On \(\mathcal A_T\), the first candidate has
\[
L_1=n_1|I_T|,
\qquad
G_1=n_1|I_T|,
\qquad
\lambda_{\min}(G_1/L_1)=1\ge c_G.
\tag{42}
\]
Thus the selector accepts that candidate, and the local-constant fit is
\[
\widehat m_1
=\frac1{n_1|I_T|}
\sum_{t\in I_T}\sum_{i=1}^{n_1}Q_{1ti}.
\tag{43}
\]
The conditional-mean part of (43) differs from \(m(z_0)=8\) by at most
\[
\frac1{|I_T|}\sum_{t\in I_T}|m(X_t)-m(z_0)|
\le\underline h_T+|\widetilde z-z_0|.
\tag{44}
\]
Given the design and donors, the noise part has mean zero and variance
\[
\operatorname{Var}\!\left(
\frac1{2n_1|I_T|}\sum_{t\in I_T}\sum_{i=1}^{n_1}U_{ti}
\,\middle|\,(X_t)_{t=1}^{T},(Z_j)_{j=1}^{N},E_0
\right)
=\frac1{4n_1|I_T|}
\le\frac1{4n_1}.
\tag{45}
\]
Every successful local-constant intercept is an average of values in \([7,9]\), while the failed-fit rule sets it to zero.  Hence \((\widehat m_1-8)^2\le64\) on \(\mathcal A_T^c\).  Splitting over \(\mathcal A_T\), using \((x+y)^2\le2x^2+2y^2\), and applying (38), (39), (41), (44), and (45), there is a fixed \(C<\infty\) such that
\[
\mathbb E[(\widehat m_1-8)^2\mid E_0]
\le C\{\underline h_T^2+N^{-1}+n_1^{-1}
+P(\mathcal A_T^c\mid E_0)\}
\longrightarrow0.
\tag{46}
\]
Put \(D_T=\widehat m_1-8\).  By (36) and \(\texttt{def:feasible-estimator}\),
\[
\widehat\tau-\tau_n(z_0)=\Gamma-\frac14-D_T.
\tag{47}
\]
Cauchy--Schwarz and (46) yield
\[
\begin{aligned}
&\left|
\mathbb E[(\widehat\tau-\tau_n(z_0))^2\mid E_0]
-\mathbb E[(\Gamma-1/4)^2\mid E_0]
\right|\\
&\quad\le
2\{\mathbb E[(\Gamma-1/4)^2\mid E_0]\}^{1/2}
\{\mathbb E[D_T^2\mid E_0]\}^{1/2}
+\mathbb E[D_T^2\mid E_0]
\longrightarrow0.
\end{aligned}
\tag{48}
\]
Since \(\Gamma\) is Bernoulli with success probability \(1/4\),
\[
\lim_{T\to\infty}
\mathbb E[(\widehat\tau-\tau_n(z_0))^2\mid E_0]
=\operatorname{Var}(\Gamma)
=\frac14\left(1-\frac14\right)
=\frac3{16}.
\tag{49}
\]
The no-log spacing term is \(R_{\mathrm{sp}}=T^{-2}\); together with (28), this gives \(R_\star=T^{-2}\).

Finally consider \(C_\alpha\).  On \(\mathcal A_T\), (42)--(43) give
\[
a_{1ti}=\frac{\mathbf1\{t\in I_T\}}{n_1|I_T|},
\qquad
\sum_t|a_{1ti}|=\frac1{n_1},
\qquad
\sum_{t,i}|a_{1ti}|=1.
\tag{50}
\]
For fixed \(\alpha\), substitution into \(\texttt{def:honest-interval}\) yields
\[
R_{\mathrm{hist}}
=B\left\{\frac12\log\left(\frac2{\alpha_{\mathrm{hist}}}\right)n_1^{-1}\right\}^{1/2}
=O(T^{-1}),
\tag{51}
\]
and
\[
R_{\mathrm{evt}}
=B\left\{\frac12\log\left(\frac2{\alpha_{\mathrm{evt}}}\right)n_1^{-1}\right\}^{1/2}
=O(T^{-1}).
\tag{52}
\]
The support of \(K\), (39), and (50) give
\[
A_L\le C_{0,1}L\underline h_T
=O\!\left(\frac{\log T}{T}\right).
\tag{53}
\]
Here \(D=1\), so the exact donor calibration satisfies
\[
M_4(N)=\frac{16}{N}+3\left(1-\frac1N\right)\le16,
\qquad
q_N=\{M_4(N)/\alpha_{\mathrm{don}}\}^{1/4}=O(1).
\tag{54}
\]
By (2), \(N=T^2\), and \(\kappa=1\),
\[
\delta_N=\frac{q_N}{\sqrt N}=O(T^{-1}),
\qquad
A_{\mathrm{don}}=\delta_N=O(T^{-1}).
\tag{55}
\]
Thus \(\delta_N\le\delta_0\) eventually.  On \(\mathcal A_T\), the selected Gram matrix is positive and all required quantities are finite.  Consequently the fallback event is contained in \(\mathcal A_T^c\) for all sufficiently large \(T\), and, throughout \(\mathcal A_T\),
\[
H_\alpha
=R_{\mathrm{hist}}+R_{\mathrm{evt}}+A_L+A_{\mathrm{don}}
=O\!\left(\frac{\log T}{T}\right)
=o(1).
\tag{56}
\]
In particular, \(H_\alpha<1/8\) eventually on \(\mathcal A_T\).

If the nonfallback interval contains \(\tau_n(z_0)\), then \(|\widehat\tau-\tau_n(z_0)|\le H_\alpha\).  Yet \(|\Gamma-1/4|\ge1/4\) surely, so (47) and the reverse triangle inequality imply on \(\{|D_T|<1/16\}\) that
\[
|\widehat\tau-\tau_n(z_0)|
\ge\frac14-\frac1{16}
=\frac3{16}>\frac18.
\tag{57}
\]
Equations (56)--(57) show that, eventually,
\[
\{\tau_n(z_0)\in C_\alpha\}
\subseteq \mathcal A_T^c\cup\{|D_T|\ge1/16\}.
\tag{58}
\]
Markov's inequality and (46) now give
\[
\begin{aligned}
P\{\tau_n(z_0)\in C_\alpha\mid E_0\}
&\le P(\mathcal A_T^c\mid E_0)
+P(|D_T|\ge1/16\mid E_0)\\
&\le P(\mathcal A_T^c\mid E_0)
+256\,\mathbb E[D_T^2\mid E_0]
\longrightarrow0.
\end{aligned}
\tag{59}
\]
This proves both estimator-specific corollaries and completes the proof.
