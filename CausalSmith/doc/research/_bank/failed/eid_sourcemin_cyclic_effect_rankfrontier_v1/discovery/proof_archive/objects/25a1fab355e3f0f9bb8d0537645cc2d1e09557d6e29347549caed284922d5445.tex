\[
 \rho_x(\widetilde{\mathcal M})
 =\left(

 \bigl(V_i^{\widetilde{\mathcal M}}\mid\operatorname{do}(V_x^{\widetilde{\mathcal M}}=t)\bigr)
 \right)_{i\in\mathsf O\setminus\{x\}}
