Fix \(n\geq1\), \(\alpha\in(0,1)\), an encoded covariance-region family \(\mathfrak C^{(n)}_\alpha=(\varphi^{(n)}_{e,\alpha}:e\in E)\), and rational thresholds
\[
 \gamma_0>0,
 \qquad \zeta_0>0,
 \qquad \lambda_0>0,
 \qquad \overline\kappa\geq1,
 \qquad \overline M\geq1.
\]
For a symmetric matrix \(G\), write
\[
 \begin{aligned}
 \mathsf{PD}(G)
 &\Longleftrightarrow
 G=G^{\mathsf T}\ \wedge
 \forall v\left[
   \sum_{i=1}^p v_i^2>0
   \Rightarrow v^{\mathsf T}Gv>0
 \right],\\
 \mathsf{PSD}(G)
 &\Longleftrightarrow
 G=G^{\mathsf T}\ \wedge
 \forall v\;v^{\mathsf T}Gv\geq0,\\
 \mathsf{Reg}_e(G)
 &\Longleftrightarrow
 \mathsf{PD}(G)\wedge\varphi^{(n)}_{e,\alpha}(G).
 \end{aligned}
\]
Let \(\mathfrak C_p\) be the finite set of directed simple cycles of length at least two on \(\{1,\ldots,p\}\), and for \(C=(i_1,\ldots,i_r)\in\mathfrak C_p\) put
\[
 \pi_C(A)=\prod_{j=1}^r (I_p-A)_{i_j i_{j+1}},
 \qquad i_{r+1}=i_1.
\]
For each \(K\subseteq E\) with \(|K|=h\), define the full feasibility formula
\[
\begin{aligned}
 F_K(A,\Psi,t;\Gamma)\Longleftrightarrow{}&
 \bigwedge_{i=1}^p A_{ii}=1
 \ \wedge\ (\det A)^2>0\\
 &\wedge\bigwedge_{C\in\mathfrak C_p}
   \{-1<\pi_C(A)<1\}
 \ \wedge\ \mathsf{PSD}(\Psi)\\
 &\wedge\bigwedge_{\substack{e\in K\\1\leq j\leq p}}t_{ej}\geq0
 \ \wedge\bigwedge_{e\in K}
 \left\{A\Gamma_eA^{\mathsf T}
 =\Psi+\operatorname{diag}(t_e)\right\},\\
 W_K(A,\Psi,t,\Gamma)\Longleftrightarrow{}&
 \left(\bigwedge_{e\in E}\mathsf{Reg}_e(\Gamma_e)\right)
 \wedge F_K(A,\Psi,t;\Gamma).
\end{aligned}
\]
Put \(q=h-c\).  For a witness of \(W_K\), define the candidate analogues of \(\gamma_H,\zeta_D,\lambda_H,\kappa_D,M_H\) by
\[
\begin{aligned}
 \widehat\gamma(K,t)
 &=0 &&\text{if }q<3,\\
 \widehat\gamma(K,t)
 &=\min_{1\leq k<\ell\leq p}
   \min_{\substack{S\subseteq K\\|S|=q}}
   \max_{\{e,f,g\}\subseteq S}
   \left|\det
   \begin{pmatrix}
    1&t_{ek}&t_{e\ell}\\
    1&t_{fk}&t_{f\ell}\\
    1&t_{gk}&t_{g\ell}
   \end{pmatrix}\right| &&\text{if }q\geq3,\\
 \widehat\zeta(A)
 &=1-\max_{C\in\mathfrak C_p}|\pi_C(A)|,\\
 \widehat\lambda(K,\Psi,\Gamma)
 &=\min\left\{\lambda_{\min}(\Psi),
       \min_{e\in K}\lambda_{\min}(\Gamma_e)\right\},\\
 \widehat\kappa(A)
 &=\|A\|_{\mathrm{op}}\|A^{-1}\|_{\mathrm{op}},\\
 \widehat M(K,\Psi,t,\Gamma)
 &=1+\|\Psi\|_{\mathrm{op}}
   +\max_{e\in K}\|\Gamma_e\|_{\mathrm{op}}
   +\max_{e\in K}\|\operatorname{diag}(t_e)\|_{\mathrm{op}}.
\end{aligned}
\]
For
\[
 \Delta_{efg}^{k\ell}(t)=
 \det
 \begin{pmatrix}
  1&t_{ek}&t_{e\ell}\\
  1&t_{fk}&t_{f\ell}\\
  1&t_{gk}&t_{g\ell}
 \end{pmatrix},
\]
define \(\mathsf{Good}_K(A,\Psi,t,\Gamma)\) as the conjunction of the following first-order conditions:
\[
\begin{aligned}
 \mathsf G_\gamma:
 &\quad q\geq3\ \wedge
 \bigwedge_{1\leq k<\ell\leq p}
 \bigwedge_{\substack{S\subseteq K\\|S|=q}}
 \bigvee_{\{e,f,g\}\subseteq S}
 \left\{(\Delta_{efg}^{k\ell}(t))^2\geq\gamma_0^2\right\},\\
 \mathsf G_\zeta:
 &\quad \bigwedge_{C\in\mathfrak C_p}
 \left\{1-\zeta_0-\pi_C(A)\geq0\right\}
 \wedge
 \left\{1-\zeta_0+\pi_C(A)\geq0\right\},\\
 \mathsf G_\lambda:
 &\quad \mathsf{PSD}(\Psi-\lambda_0I_p)
 \wedge\bigwedge_{e\in K}\mathsf{PSD}(\Gamma_e-\lambda_0I_p),\\
 \mathsf G_\kappa:
 &\quad \exists r>0:\quad
 \mathsf{PSD}(A^{\mathsf T}A-rI_p)
 \wedge
 \mathsf{PSD}(\overline\kappa^{2}rI_p-A^{\mathsf T}A),\\
 \mathsf G_M:
 &\quad \exists x,y,z\geq0:\ 1+x+y+z\leq\overline M,\\
 &\qquad \mathsf{PSD}(xI_p-\Psi)
 \wedge\bigwedge_{e\in K}\mathsf{PSD}(yI_p-\Gamma_e)
 \wedge\bigwedge_{\substack{e\in K\\1\leq j\leq p}}\{t_{ej}\leq z\}.
\end{aligned}
\]
Apply a fixed effective real quantifier-elimination algorithm to
\[
 \Phi(A)=
 \bigvee_{\substack{K\subseteq E\\|K|=h}}
 \exists\Gamma\,\exists\Psi\,\exists t\;W_K(A,\Psi,t,\Gamma)
\]
to obtain a quantifier-free rational-polynomial formula \(\phi_{\mathcal U}(A)\).  Apply the same algorithm to the closed audit sentence
\[
 \mathsf{Bad}=
 \bigvee_{\substack{K\subseteq E\\|K|=h}}
 \exists A\,\exists\Gamma\,\exists\Psi\,\exists t\;
 \left[W_K(A,\Psi,t,\Gamma)
 \wedge\neg\mathsf{Good}_K(A,\Psi,t,\Gamma)\right].
\]
The construction \(\operatorname{CertificateInferenceHandle}\) returns \(\mathsf{abstain}\) when \(\mathsf{Bad}\) is true and otherwise returns the code \(\phi_{\mathcal U}\).  It makes exactly \(\binom mh\) subset-formula branches; the quantifier-elimination calls, rather than an unspecified feasibility oracle, are part of this terminating mathematical specification.
